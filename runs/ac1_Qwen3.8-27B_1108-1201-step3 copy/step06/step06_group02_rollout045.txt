import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def objective(a):
    """Compute 2*n*max(conv(a,a)) for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()

def subgradient_method(f0, time_limit, lower_bound=1.0, verbose=False):
    """Subgradient method with adaptive step size and momentum."""
    f = np.maximum(f0, 0.0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = objective(f)
    t0 = time.monotonic()
    
    # Momentum
    v = np.zeros(n)
    beta = 0.9
    
    it = 0
    step_size = 1.0 / np.sqrt(it + 1)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1.0:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        obj = 2.0 * n * M
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            if verbose and it % 50 == 0:
                print(f"    SG iter {it}: obj={best_obj:.10f}, M={M:.8f}")
        
        # Subgradient: focus on top peaks
        k_top = min(20, len(conv))
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        
        # Weight peaks by their excess over the mean
        mean_val = 1.0 / (2 * n - 1)
        excess = np.maximum(pv - mean_val, 0)
        w = excess ** 2
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        # Compute subgradient
        sg = np.zeros(n)
        for idx, k in enumerate(peaks):
            if w[idx] < 1e-12:
                continue
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            i_arr = np.arange(js, je + 1)
            sg[i_arr] += 2.0 * w[idx] * f[k - i_arr]
        
        # Polyak step size
        step_size = (obj - lower_bound) / (np.linalg.norm(sg)**2 + 1e-10)
        step_size = min(step_size, 0.5)
        
        # Update with momentum
        v = beta * v - step_size * sg
        f_new = f + v
        f_new = project_simplex(f_new)
        
        # Check if we improved
        f = f_new
        if it % 1000 == 0:
            beta = max(0.5, beta * 0.999)
    
    return best_f, best_obj

Actually, let me reconsider the whole approach. The key challenge is that the objective is non-convex (bilinear), so we can't solve it directly with convex optimization. The Frank-Wolfe approach works by alternating:
- Fix f, solve LP for g (linear in g)
- Move from f towards g

The LP is:

subject to (f*g)[k] <= M for all k = 0, ..., 2n-2
g >= 0

This is a standard LP. The constraint (f*g)[k] = sum_i f_i * g_{k-i} (with valid indices) is linear in g.

The issue is the choice of M. If M is the current max(f*f), the LP might be infeasible or give a direction that doesn't improve things. Using a relaxed M (larger than current max) gives a more feasible LP but a less targeted direction.

**Approach: Direct optimization of the max using surrogate**

Instead of minimizing max(conv) directly, minimize a smooth surrogate like:
- log-sum-exp of conv values: LSE(t * conv) / t for large t
- Power mean: (sum(conv^p))^(1/p) for large p

The power mean approach is nice because it's differentiable and we can use gradient descent.

For the power mean: F_p(a) = (sum_k (conv_k)^p)^(1/p)

As p -> infinity, this approaches max(conv). For finite p, it's a smooth upper bound on max(conv).

The gradient of F_p with respect to a_i is:
dF_p/da_i = (1/p) * (sum conv_k^p)^((p-1)/p) * sum_k p * conv_k^(p-1) * dconv_k/da_i

And dconv_k/da_i = f_{k-i} (the contribution of a_i to conv_k is a_i * f_{k-i}).

So the gradient is:
grad_i = (sum conv_k^p)^((p-1)/p) * sum_k conv_k^(p-1) * f_{k-i}

This is computable efficiently using convolution!
