```python
def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Project v onto {x: x>=0, sum(x)=target_sum}"""
    n = len(v)
    v = v.copy()
    # Scale to sum=target_sum approximately, then project
    # Standard projection onto simplex with sum=target_sum
    # Equivalent to projecting (v * target_sum) onto the standard simplex then dividing by target_sum... 
    # Actually, let's just do: project onto {x>=0, sum=x_target}
    # This is the same as: x = proj_{simplex}(v) * target_sum
    # where proj_{simplex} projects onto {x>=0, sum=1}
```

```python
def line_search_convex(a, d, tol=1e-12, max_iter=100):
    """Find t that minimizes max_k conv(a+t*d, a+t*d)[k] over valid t range."""
    n = len(a)
    # Determine valid range: a + t*d >= 0
    t_min = 0.0
    t_max = np.inf
    for i in range(n):
        if d[i] < -tol:
            t_max = min(t_max, -a[i] / d[i])
        elif d[i] > tol:
            t_min = max(t_min, -a[i] / d[i])  # this should be <= 0
    t_min = max(t_min, 0.0)
    if not np.isfinite(t_max):
        t_max = 1.0  # cap
    
    def f(t):
        at = a + t * d
        conv = fast_convolve(at)
        return conv.max()
    
    # Bisection
    lo, hi = t_min, t_max
    fo = f(lo)
    fh = f(hi)
    
    for _ in range(max_iter):
        mid = (lo + hi) / 2
        fm = f(mid)
        if fm < fo:
            fo = fm
            lo = mid
        if fm < fh:
            fh = fm
            hi = mid
        if hi - lo < tol:
            break
        # Better: find the minimum of the piecewise quadratic
        # For bisection, we need to know which side is lower
        # f is convex, so if f(mid) < f(lo), minimum is to the right of lo
        # if f(mid) < f(hi), minimum is to the left of hi
    
    # Refine around the best
    best_t = (lo + hi) / 2
    best_f = f(best_t)
    return best_t, best_f
```

```python
def golden_section_min(f, a, b, tol=1e-10, max_iter=100):
    """Minimize unimodal f on [a, b] using golden section search."""
    phi = (1 + np.sqrt(5)) / 2
    c = b - (b - a) / phi
    d = a + (b - a) / phi
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - (b - a) / phi
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + (b - a) / phi
            fd = f(d)
    return (a + b) / 2
```

```python
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def gs_search_conv(a, d, t_lo, t_hi, tol=1e-13, max_iter=60):
    """Golden section search to minimize max(conv(a+t*d)) on [t_lo, t_hi]."""
    if t_hi - t_lo < tol:
        return (t_lo + t_hi) / 2.0
    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    c = t_hi - invphi * (t_hi - t_lo)
    dd = t_lo + invphi * (t_hi - t_lo)
    fc = fast_convolve(a + c * d).max()
    fd = fast_convolve(a + dd * d).max()
    for _ in range(max_iter):
        if t_hi - t_lo < tol:
            break
        if fc < fd:
            t_hi = dd
            dd = c
            fd = fc
            c = t_hi - invphi * (t_hi - t_lo)
            fc = fast_convolve(a + c * d).max()
        else:
            t_lo = c
            c = dd
            fc = fd
            dd = t_lo + invphi * (t_hi - t_lo)
            fd = fast_convolve(a + dd * d).max()
    return (t_lo + t_hi) / 2.0


def compute_subgrad(a, k_peaks=10, temperature=0.001):
    """Compute weighted subgradient of max_k conv[k]. Returns (gradient, conv)."""
    n = len(a)
    conv = fast_convolve(a)
    k_actual = min(k_peaks, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = (pv - pv.max()) / temperature
    w = np.exp(scaled)
    w /= w.sum()
    g = np.zeros(n)
    for idx in range(k_actual):
        if w[idx] < 1e-15:
            continue
        k = int(peaks[idx])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        g[lo:hi+1] += w[idx] * 2.0 * a[k - hi : k - lo + 1][::-1]
    return g, conv

                d2 = -a.copy()
                d2[away] = 1.0 - a[away]
                t2 = gs_search_conv(a, d2, 0.0, 1.0, tol=1e-14, max_iter=45)
                v2 = fast_convolve(np.maximum(a + t2 * d2, 0)).max()
                if v2 < best_val:
                    best_val = v2
                    best_d = d2
                    best_t = t2

                d3[i_src] = -1.0
                d3[j_dst] = 1.0

                v3 = fast_convolve(a + t3 * d3).max()
                if v3 < best_val:
                    best_val = v3
                    best_d = d3
                    best_t = t3

                        # Now sum(d4) should be 0

                            if v4 < best_val:

                                best_d = d4
                                best_t = t4

                temp = 0.001
    
    return best_a, best_obj


def multiplicative_descent(a0, time_limit, verbose=False):

            break

        temp *= 1.005

            a = norm_simplex(best_a + rng.standard_normal(n) * 0.002)

            no_improve = 0

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return norm_simplex(f_interp)


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    return norm_simplex(a)


def make_bell(n, width_frac=0.6):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0.0, 1.0)
    a = 1.0 - t ** 4
    return norm_simplex(a)


def make_cosine_mod(n, freq, amp):
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
    return norm_simplex(a)
```

```python
- Systematic pairwise coordinate descent (expensive-to-cheap mass transfer)

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_simplex(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def gs_search_conv(a, d, t_lo, t_hi, tol=1e-14, max_iter=55):
    """Golden section search: minimize max(conv(a+t*d)) for t in [t_lo, t_hi]."""
    if t_hi - t_lo < tol:
        return (t_lo + t_hi) / 2.0
    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    c = t_hi - invphi * (t_hi - t_lo)
    dd = t_lo + invphi * (t_hi - t_lo)
    fc = fast_convolve(a + c * d).max()
    fd = fast_convolve(a + dd * d).max()
    for _ in range(max_iter):
        if t_hi - t_lo < tol:
            break
        if fc < fd:
            t_hi = dd
            dd = c
            fd = fc
            c = t_hi - invphi * (t_hi - t_lo)
            fc = fast_convolve(a + c * d).max()
        else:
            t_lo = c
            c = dd
            fc = fd
            dd = t_lo + invphi * (t_hi - t_lo)
            fd = fast_convolve(a + dd * d).max()
    return (t_lo + t_hi) / 2.0


def compute_subgrad(a, k_peaks=12, temperature=0.001):
    """Weighted subgradient of max_k conv[k] at point a. Returns (grad, conv)."""
    n = len(a)
    conv = fast_convolve(a)
    k_actual = min(k_peaks, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = (pv - pv.max()) / temperature
    w = np.exp(scaled)
    w /= w.sum()
    g = np.zeros(n)
    for idx in range(k_actual):
        if w[idx] < 1e-15:
            continue
        k = int(peaks[idx])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        g[lo:hi + 1] += w[idx] * 2.0 * a[k - hi:k - lo + 1][::-1]
    return g, conv


def fw_optimize(a0, time_limit, verbose=False):

    a = norm_simplex(np.asarray(a0, dtype=np.float64).copy())
    n = len(a)
    t0 = time.monotonic()

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    no_improve = 0
    it = 0
    k_peaks = 12
    temp = 0.001
    rng = np.random.default_rng(99999)
    
    # Track which strategies work best
    strat_counts = [0, 0, 0]  # fw, away, pairwise

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.2:
            break
        if no_improve > 800:
            break

        g, conv = compute_subgrad(a, k_peaks=k_peaks, temperature=temp)
        current_peak = conv.max()

        best_val = current_peak
        best_d = None
        best_t = 0.0
        best_strat = -1

        # --- Strategy 1: FW vertex (argmin subgradient) ---
        j_fw = int(np.argmin(g))
        d1 = np.zeros(n)
        d1[j_fw] = 1.0
        d1 -= a
        t1 = gs_search_conv(a, d1, 0.0, 1.0, tol=1e-14, max_iter=50)
        v1 = fast_convolve(a + t1 * d1).max()
        if v1 < best_val:
            best_val = v1
            best_d = d1
            best_t = t1
            best_strat = 0

        # --- Strategy 2: Away step ---
        support = np.where(a > 1e-12)[0]
        if len(support) > 3:
            away = int(support[np.argmin(a[support])])
            if away != j_fw:
                d2 = -a.copy()
                d2[away] = 1.0 - a[away]
                t2 = gs_search_conv(a, d2, 0.0, 1.0, tol=1e-14, max_iter=45)
                at2 = np.maximum(a + t2 * d2, 0.0)
                if at2.sum() > 1e-15:
                    v2 = fast_convolve(at2).max()
                    if v2 < best_val:
                        best_val = v2
                        best_d = d2
                        best_t = t2
                        best_strat = 1

        # --- Strategy 3: Systematic pairwise (top expensive -> top cheap) ---
        if it % 2 == 0:
            m = min(6, n // 4)
            expensive = np.argsort(g)[-m:]
            cheap = np.argsort(g)[:m]
            for i_src in expensive:
                if a[i_src] < 1e-12:
                    continue
                for j_dst in cheap:
                    if j_dst == i_src:
                        continue
                    d3 = np.zeros(n)
                    d3[int(i_src)] = -1.0
                    d3[int(j_dst)] = 1.0
                    t3_max = a[i_src]
                    t3 = gs_search_conv(a, d3, 0.0, t3_max, tol=1e-14, max_iter=40)
                    v3 = fast_convolve(a + t3 * d3).max()
                    if v3 < best_val:
                        best_val = v3
                        best_d = d3
                        best_t = t3
                        best_strat = 2

        # --- Strategy 4: Random pairwise (extra diversity) ---
        if it % 3 == 0:
            for _ in range(4):
                i_r = int(rng.integers(n))
                j_r = int(rng.integers(n))
                if i_r == j_r or a[i_r] < 1e-12:
                    continue
                d4 = np.zeros(n)
                d4[i_r] = -1.0
                d4[j_r] = 1.0
                t4 = gs_search_conv(a, d4, 0.0, a[i_r], tol=1e-14, max_iter=35)
                v4 = fast_convolve(a + t4 * d4).max()
                if v4 < best_val:
                    best_val = v4
                    best_d = d4
                    best_t = t4
                    best_strat = 2

        if best_d is not None and best_val < current_peak - 1e-16:
            a_new = np.maximum(a + best_t * best_d, 0.0)
            a = norm_simplex(a_new)
            no_improve = 0
            if best_strat >= 0:
                strat_counts[best_strat] += 1
            conv_new = fast_convolve(a)
            new_obj = 2.0 * n * conv_new.max()
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
                if verbose and it % 500 == 0:
                    print(f"      FW it={it}: obj={best_obj:.10f} strat={best_strat}")
        else:
            no_improve += 1
            temp = min(temp * 1.015, 0.05)
            if no_improve > 0 and no_improve % 150 == 0:
                noise = rng.standard_normal(n) * 0.002
                a = norm_simplex(best_a + noise)
                temp = 0.001

    return best_a, best_obj


def multiplicative_descent(a0, time_limit, verbose=False):
    """Multiplicative weight descent with peak-weighted subgradient."""
    a = norm_simplex(np.asarray(a0, dtype=np.float64).copy())
    n = len(a)
    t0 = time.monotonic()

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    lr = 0.004
    no_improve = 0
    temp = 0.002
    it = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.2:
            break
        if no_improve > 600:
            break

        g, conv = compute_subgrad(a, k_peaks=15, temperature=temp)

        a_new = a * np.exp(-lr * g)
        a = norm_simplex(a_new)

        conv_new = fast_convolve(a)
        new_obj = 2.0 * n * conv_new.max()

        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"      MD it={it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        lr *= 0.9998
        temp *= 1.005

        if no_improve > 250:
            a = norm_simplex(best_a + rng.standard_normal(n) * 0.003)
            lr = 0.003
            no_improve = 0

    return best_a, best_obj


def pairwise_descent(a0, time_limit, verbose=False):
    """Focused pairwise coordinate descent with systematic sweeps."""
    a = norm_simplex(np.asarray(a0, dtype=np.float64).copy())
    n = len(a)
    t0 = time.monotonic()

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(55555)

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.2:
            break
        if no_improve > 400:
            break

        g, conv = compute_subgrad(a, k_peaks=8, temperature=0.001)
        current_obj = 2.0 * n * conv.max()

        # Systematic sweep: for each of top expensive coords, try transfer to top cheap
        m_exp = min(8, n // 5)
        m_che = min(8, n // 5)
        exp_idx = np.argsort(g)[-m_exp:]
        che_idx = np.argsort(g)[:m_che]

        improved = False
        for i_s in exp_idx:
            if a[i_s] < 1e-12:
                continue
            for j_d in che_idx:
                if j_d == i_s:
                    continue
                d = np.zeros(n)
                d[int(i_s)] = -1.0
                d[int(j_d)] = 1.0
                t_opt = gs_search_conv(a, d, 0.0, a[i_s], tol=1e-14, max_iter=40)
                if t_opt < 1e-14:
                    continue
                a_try = a + t_opt * d
                v = fast_convolve(a_try).max()
                if v < conv.max() - 1e-16:
                    a = norm_simplex(np.maximum(a_try, 0.0))
                    improved = True
                    conv = fast_convolve(a)

        # Also try some random pairs
        for _ in range(6):
            i_r = int(rng.integers(n))
            j_r = int(rng.integers(n))
            if i_r == j_r or a[i_r] < 1e-12:
                continue
            d = np.zeros(n)
            d[i_r] = -1.0
            d[j_r] = 1.0
            t_opt = gs_search_conv(a, d, 0.0, a[i_r], tol=1e-14, max_iter=35)
            if t_opt < 1e-14:
                continue
            a_try = a + t_opt * d
            v = fast_convolve(a_try).max()
            if v < conv.max() - 1e-16:
                a = norm_simplex(np.maximum(a_try, 0.0))
                improved = True
                conv = fast_convolve(a)

        if improved:
            no_improve = 0
            new_obj = 2.0 * n * conv.max()
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
                if verbose and it % 100 == 0:
                    print(f"      PD it={it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            if no_improve > 100:
                a = norm_simplex(best_a + rng.standard_normal(n) * 0.003)
                no_improve = 0

    return best_a, best_obj


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return norm_simplex(f_interp)


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-shaped initial sequence normalized to sum=1."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    return norm_simplex(a)


def make_bell(n, width_frac=0.6):
    """Flat-topped bell sequence."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0.0, 1.0)
    a = 1.0 - t ** 4
    return norm_simplex(a)


def make_cosine_mod(n, freq, amp):
    """Cosine-modulated uniform sequence."""
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
    return norm_simplex(a)


def make_multi_cosine(n, n_freqs=4, max_amp=0.2, seed=0):
    """Multi-frequency cosine modulated."""
    rng = np.random.default_rng(seed)
    x = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_freqs + 1):
        amp = rng.uniform(0.03, max_amp)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * m * x / n + phase)
    return norm_simplex(a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimizer: FW with exact line search, systematic pairwise descent,
    multiplicative updates, and multi-length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    rng_main = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    # --- Load parent ---
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and p.sum() > 1e-10:
            parent = norm_simplex(p)
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_score = p_score
                best_seq = parent.copy()
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768, 0.25)
        sc = float(evaluate_sequence(parent.tolist()))
        best_score = sc
        best_seq = parent.copy()
        print(f"Default init: n=768, score={sc:.8f}")

    n_parent = len(parent)

    # === Phase 1: Build candidate pool ===
    print("\n=== Phase 1: Candidate Pool ===")
    t0 = time.monotonic()

    candidates = []
    # Parent at various lengths
    for target_n in [n_parent, 640, 768, 896, 1024, 1280]:
        if target_n == n_parent:
            candidates.append((parent, f"parent_{n_parent}"))
        else:
            candidates.append((interpolate_seq(parent, target_n), f"interp_{target_n}"))

    # Structured inits
    for n_try in [n_parent, 1024, 768]:
        candidates.append((make_gaussian(n_try, 0.25), f"gauss_{n_try}"))
        candidates.append((make_gaussian(n_try, 0.35), f"gauss_w_{n_try}"))
        candidates.append((make_bell(n_try, 0.5), f"bell_{n_try}"))
        candidates.append((make_cosine_mod(n_try, 1, 0.12), f"cos1_{n_try}"))
        candidates.append((make_multi_cosine(n_try, 4, 0.15, seed=n_try), f"mc_{n_try}"))

    # Evaluate and rank
    scored = []
    for a_c, label in candidates:
        a_c = norm_simplex(a_c)
        conv = fast_convolve(a_c)
        sc = 2.0 * len(a_c) * conv.max()
        scored.append((sc, a_c, label))
        # Also check with evaluate_sequence for top candidates
        if sc < 1.52:
            es = float(evaluate_sequence(a_c.tolist()))
            if np.isfinite(es) and es < best_score:
                best_score = es
                best_seq = a_c.copy()
                print(f"  NEW BEST [{label}]: {best_score:.8f}")

    scored.sort(key=lambda x: x[0])
    print(f"  Best normalized init: {scored[0][2]}, obj={scored[0][0]:.8f}")
    print(f"  Phase 1 time: {time.monotonic()-t0:.1f}s")

    # === Phase 2: FW optimization on top candidates ===
    print("\n=== Phase 2: FW Optimization ===")
    top_candidates = scored[:5]
    for idx in range(len(top_candidates)):
        rem = deadline - time.monotonic() - 100
        if rem < 20:
            break
        sc_c, a_c, label = top_candidates[idx]
        # Allocate time: more for better candidates
        weight = max(0.3, 1.0 - idx * 0.15)
        time_alloc = min(rem * weight * 0.4, 350)
        if time_alloc < 10:
            break
        print(f"  FW [{label}] n={len(a_c)}: alloc={time_alloc:.0f}s (init={sc_c:.6f})")
        a_opt, obj_opt = fw_optimize(a_c, time_limit=time_alloc, verbose=True)
        sc_opt = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"    -> NEW BEST: {best_score:.8f}")
        elif np.isfinite(obj_opt) and obj_opt < sc_c:
            # Track even if not global best yet
            pass

    # === Phase 3: Pairwise descent on current best ===
    print("\n=== Phase 3: Pairwise Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            time_alloc = min(rem, 150)
            print(f"  PD: n={len(best_seq)}, alloc={time_alloc:.0f}s")
            a_opt, obj_opt = pairwise_descent(best_seq, time_limit=time_alloc, verbose=True)
            sc_opt = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc_opt) and sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"    -> NEW BEST: {best_score:.8f}")

    # === Phase 4: Multiplicative descent ===
    print("\n=== Phase 4: Multiplicative Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            time_alloc = min(rem, 120)
            print(f"  MD: n={len(best_seq)}, alloc={time_alloc:.0f}s")
            a_opt, obj_opt = multiplicative_descent(best_seq, time_limit=time_alloc, verbose=True)
            sc_opt = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc_opt) and sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"    -> NEW BEST: {best_score:.8f}")

    # === Phase 5: Perturbation + re-optimize ===
    print("\n=== Phase 5: Perturbation Exploration ===")
    if best_seq is not None:
        n_best = len(best_seq)
        perturbations = []

        # Shifts
        for s in [1, 2, 3, 5]:
            perturbations.append((f"shift{s}", np.roll(best_seq, s)))

        # Flip
        perturbations.append(("flip", best_seq[::-1].copy()))

        # Blend with uniform
        for mix in [0.03, 0.05, 0.08, 0.12]:
            perturbations.append((f"mix_{mix}", (1.0 - mix) * best_seq + mix * (np.ones(n_best) / n_best)))

        # Small structured noise
        for amp in [0.001, 0.002, 0.004]:
            noise = rng_main.standard_normal(n_best) * amp
            perturbations.append((f"noise_{amp}", best_seq + noise))

        # Nudge specific regions
        for frac in [0.1, 0.2]:
            region = n_best // 8
            for start in [0, n_best // 2]:
                p = best_seq.copy()
                p[start:start + region] *= (1.0 + frac)
                p[start + region:start + 2 * region] *= (1.0 - frac * 0.5)
                perturbations.append((f"region_{start}_{frac}", p))

        for label, a_p in perturbations:
            rem = deadline - time.monotonic() - 20
            if rem < 8:
                break
            a_p = norm_simplex(a_p)
            # Quick check
            conv_p = fast_convolve(a_p)
            sc_p = 2.0 * len(a_p) * conv_p.max()
            if sc_p > best_score * 1.001:
                continue  # Skip if clearly worse
            time_alloc = min(rem * 0.25, 50)
            if time_alloc < 5:
                break
            a_opt, obj_opt = fw_optimize(a_p, time_limit=time_alloc)
            sc_opt = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc_opt) and sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"  NEW BEST [{label}]: {best_score:.8f}")

    # === Phase 6: Multi-length final ===
    print("\n=== Phase 6: Multi-Length Final ===")
    if best_seq is not None:
        n_cur = len(best_seq)
        target_lengths = sorted(set([
            max(256, n_cur - 32),
            max(256, n_cur - 16),
            n_cur + 16,
            n_cur + 32,
            round(n_cur * 0.9),
            round(n_cur * 1.1),
        ]))
        for tn in target_lengths:
            if tn == n_cur or tn < 128 or tn > 2048:
                continue
            rem = deadline - time.monotonic() - 10
            if rem < 8:
                break
            a_interp = interpolate_seq(best_seq, tn)
            sc_i = float(evaluate_sequence(a_interp.tolist()))
            if np.isfinite(sc_i) and sc_i < best_score:
                best_score = sc_i
                best_seq = a_interp.copy()
                print(f"  NEW BEST [interp_{tn}]: {best_score:.8f}")
            time_alloc = min(rem * 0.35, 40)
            if time_alloc > 5:
                a_opt, obj_opt = fw_optimize(a_interp, time_limit=time_alloc)
                sc_opt = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(sc_opt) and sc_opt < best_score:
                    best_score = sc_opt
                    best_seq = a_opt.copy()
                    print(f"  NEW BEST [fw_interp_{tn}]: {best_score:.8f}")

    # === Phase 7: Final polish ===
    print("\n=== Phase 7: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 3
        if rem > 8:
            time_alloc = min(rem, 20)
            print(f"  Final FW: alloc={time_alloc:.0f}s")
            a_opt, obj_opt = fw_optimize(best_seq, time_limit=time_alloc)
            sc_opt = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc_opt) and sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"  FINAL NEW BEST: {best_score:.8f}")

        # Final pairwise squeeze
        rem = deadline - time.monotonic() - 2
        if rem > 5:
            time_alloc = min(rem, 12)
            a_opt, obj_opt = pairwise_descent(best_seq, time_limit=time_alloc)
            sc_opt = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc_opt) and sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"  FINAL PD NEW BEST: {best_score:.8f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768.0
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    total_time = time.monotonic() - (deadline - max(1.0, float(budget_s) - 15.0))
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Total time: {total_time:.1f}s")
    return [float(x) for x in best_seq]
```
