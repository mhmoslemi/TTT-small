    else:
        max_c = np.max(c)
        if max_c < 1e-15:
            return np.zeros(n)
        c_norm = c / max_c
        w = c_norm ** (power - 1)
    
    w_rev = w[::-1].copy()
    conv_len = n + len(w_rev) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(F * W, fft_size)[:conv_len]
    indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
    g = 2.0 * conv[indices]
    return g


def nesterov_gd(f0, time_limit, power_schedule=None, verbose=False):
    """Nesterov-accelerated gradient descent with power-smoothed objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    if power_schedule is None:
        power_schedule = [2, 4, 8, 16, 32, 64, 128]
    
    y = f.copy()
    best_f = f.copy()
    c = fast_convolve(f)
    best_obj = c.max()
    
    t0 = time.monotonic()
    
    for power in power_schedule:
        if time.monotonic() - t0 > time_limit * 0.9:
            break
        
        it = 0
        no_improve = 0
        beta = 1.0
        lr = 2.0 / (power * (power + 1) * np.max(fast_convolve(f)) * n)
        lr = min(lr, 0.5 / np.sqrt(n))
        
        stage_start = time.monotonic()
        stage_time = min(time_limit * 0.06, (time_limit * 0.85 - (time.monotonic() - t0)) * 0.2)
        if stage_time < 10:
            continue
        
        while time.monotonic() - stage_start < stage_time and time.monotonic() - t0 < time_limit * 0.85:
            it += 1
            c = fast_convolve(y)
            g = gradient_entropy(y, c, power)
            
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-15:
                break
            
            beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
            y_new = y + ((beta - 1.0) / beta_new) * (y - f) - lr * g
            f = y
            y = y_new
            beta = beta_new
            
            y = project_simplex(y)
            
            c_y = fast_convolve(y)
            obj_y = c_y.max()
            
            if obj_y < best_obj - 1e-13:
                best_obj = obj_y
                best_f = y.copy()
                no_improve = 0
                lr = min(lr * 1.03, 1.0 / np.sqrt(n) * 0.5)
            else:
                no_improve += 1
                if no_improve % 20 == 0:
                    lr *= 0.7
            
            if no_improve > 100:
                rng = np.random.default_rng(it * 7 + power)
                noise = rng.standard_normal(n) * 0.003
                y = project_simplex(best_f + noise)
                f = y.copy()
                beta = 1.0
                lr = 2.0 / (power * np.max(fast_convolve(y)) * n)
                lr = min(lr, 0.5 / np.sqrt(n))
                no_improve = 0
            
            if verbose and it % 200 == 0:
                print(f"    NGD p={power} it={it}: obj={c_y.max():.10f} best={best_obj:.10f}")
    
    return best_f, best_obj


def active_set_lp(f, M, n, num_active=30):
    """Solve LP with only the most active constraints."""
    c = fast_convolve(f)
    if num_active >= len(c):
        top_k = np.arange(len(c))
    else:
        top_k = np.argpartition(c, -num_active)[-num_active:]
    
    A_ub = np.zeros((len(top_k), n))
    for idx, k in enumerate(top_k):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A_ub[idx, js:je + 1] = f[js:je + 1]
    
    b_ub = np.full(len(top_k), M)
    obj_vec = -np.ones(n)
    bounds = [(0, None)] * n
    
    res = linprog(obj_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_active(f0, time_limit, num_active=30, verbose=False):
    """LP-Frank-Wolfe with active set constraints and golden section line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    c = fast_convolve(f)
    M = c.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_factor = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        g = active_set_lp(f, M * m_factor, n, num_active)
        if g is None:
            m_factor *= 1.15
            if m_factor > 8.0:
                rng = np.random.default_rng(it * 13 + 5)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                c = fast_convolve(f)
                M = c.max()
                m_factor = 1.0
                no_improve = 0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            m_factor *= 1.1
            continue
        g = g / g_sum
        
        diff = np.max(np.abs(g - f))
        if diff < 1e-11:
            no_improve += 1
            if no_improve > 80:
                m_factor = min(m_factor * 1.08, 6.0)
            if no_improve > 250:
                rng = np.random.default_rng(it * 29 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                c = fast_convolve(f)
                M = c.max()
                m_factor = 1.0
                no_improve = 0
            continue
        else:
            if diff < 1e-7:
                no_improve += 1
            else:
                no_improve = 0
        
        # Golden section line search
        best_alpha = 0.0
        best_val = M
        
        # Coarse search
        for alpha in [0.1, 0.2, 0.35, 0.5, 0.7, 0.9, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Fine search with golden section
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(15):
                m1 = hi - (hi - lo) * 0.618
                m2 = lo + (hi - lo) * 0.618
                for m in [m1, m2]:
                    ft = f + m * (g - f)
                    ft = np.maximum(ft, 0.0)
                    st = ft.sum()
                    if st < 1e-15:
                        continue
                    ft = ft / st
                    val = fast_convolve(ft).max()
                    if val < best_val:
                        best_val = val
                        best_alpha = m
                ft1 = f + m1 * (g - f)
                ft1 = np.maximum(ft1, 0.0) / ft1.sum()
                v1 = fast_convolve(ft1).max()
                ft2 = f + m2 * (g - f)
                ft2 = np.maximum(ft2, 0.0) / ft2.sum()
                v2 = fast_convolve(ft2).max()
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
        
        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            st = f_new.sum()
            if st < 1e-15:
                f_new = f.copy()
            else:
                f_new = f_new / st
            c = fast_convolve(f_new)
            M = c.max()
            f = f_new
            m_factor = max(1.0, m_factor * 0.9)
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 50 == 0:
                    print(f"    LP-FW-A it={it}: obj={best_obj:.12f}")
        else:
            no_improve += 1
            m_factor = min(m_factor * 1.05, 5.0)
    
    return best_f, best_obj


def peak_shrinkage(f0, time_limit, verbose=False):
    """Directly target the peak of the convolution by shrinking contributors."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    lr = 0.05
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(f)
        M = c.max()
        peak_k = int(np.argmax(c))
        
        # Find contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contributions = f[idx] * f[peak_k - idx]
        total_contrib = contributions.sum()
        
        if total_contrib < 1e-15:
            break
        
        # Shrink the largest contributors
        num_shrink = min(10, len(idx))
        top_local = np.argsort(contributions)[-num_shrink:]
        
        shrink_mask = np.zeros(n)
        for ti in top_local:
            j = int(idx[ti])
            shrink_mask[j] = 1.0
        
        # Compute where to add mass: positions with low convolution impact
        # Use the current convolution as a proxy for "cost" of adding mass
        add_scores = np.zeros(n)
        for j in range(n):
            # Cost of adding to position j is approximately the current conv values
            # that involve f[j]
            k_min = max(0, j)
            k_max = min(2 * n - 2, j + n - 1)
            add_scores[j] = np.mean(c[k_min:k_max + 1])
        
        add_scores[shrink_mask > 0] = np.inf
        add_scores = np.maximum(add_scores, 0.0)
        
        # Normalize and use as weight for adding mass
        add_scores[~np.isinf(add_scores)] -= np.nanmin(add_scores)
        if add_scores[~np.isinf(add_scores)].max() < 1e-15:
            add_scores[~np.isinf(add_scores)] = 1.0
        add_weights = 1.0 / (add_scores + 1e-10)
        add_weights[shrink_mask > 0] = 0
        aw_sum = add_weights.sum()
        if aw_sum < 1e-15:
            break
        add_weights /= aw_sum
        
        # Shrink and add
        f[shrink_mask > 0] *= (1.0 - lr)
        removed = lr * f[shrink_mask > 0].sum()
        
        # Redistribute removed mass
        f += removed * add_weights
        
        f = np.maximum(f, 0.0)
        fs = f.sum()
        if fs < 1e-15:
            f = best_f.copy()
            lr *= 0.5
            continue
        f = f / fs
        
        c_new = fast_convolve(f)
        obj_new = c_new.max()
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
            lr = min(lr * 1.02, 0.15)
        else:
            no_improve += 1
            lr *= 0.92
        
        if no_improve > 100:
            rng = np.random.default_rng(it * 7 + 11)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            lr = 0.05
            no_improve = 0
        
        if lr < 1e-5:
            break
        
        if verbose and it % 100 == 0:
            print(f"    Shrink it={it}: obj={obj_new:.10f} best={best_obj:.10f}")
    
    return best_f, best_obj


def coordinate_polish(f0, time_limit, verbose=False):
    """Coordinate descent polish."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f0, fast_convolve(f0).max()
    f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False
        c = fast_convolve(f)
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in [0.75, 0.85, 0.92, 1.08, 1.15, 1.25]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.4:
                    continue
                
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                f_trial[mask] *= (1.0 - new_val) / denom
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= (1.0 - best_j_val) / denom
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    kernel = np.array([0.1, 0.2, 0.4, 0.2, 0.1])
    f_smooth = np.convolve(f_interp, kernel, mode='same')
    f_smooth = np.maximum(f_smooth, 0)
    ss = f_smooth.sum()
    if ss > 1e-15:
        f_smooth /= ss
    return f_smooth


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining:
    - Diverse initialization screening
    - Nesterov accelerated gradient descent with power smoothing
    - Active-set LP-Frank-Wolfe (fast, top-K constraints)
    - Peak-targeted shrinkage heuristic
    - Coordinate descent polish
    - Multi-length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass
    
    if parent is None:
        parent = make_gaussian(512, 0.3)
    
    n_parent = len(parent)
    
    # === Phase 1: Initialization Screening ===
    print("=== Phase 1: Init Screening ===")
    t1s = time.monotonic()
    
    init_candidates = []
    init_candidates.append(("parent", parent.copy()))
    init_candidates.append(("gauss_25", make_gaussian(n_parent, 0.25)))
    init_candidates.append(("gauss_35", make_gaussian(n_parent, 0.35)))
    init_candidates.append(("gauss_45", make_gaussian(n_parent, 0.45)))
    init_candidates.append(("bell_60", make_bell(n_parent, 0.6, 4)))
    init_candidates.append(("bell_70", make_bell(n_parent, 0.7, 6)))
    init_candidates.append(("raised_cos", make_raised_cosine(n_parent)))
    init_candidates.append(("tukey_03", make_tukey(n_parent, 0.3)))
    init_candidates.append(("tukey_05", make_tukey(n_parent, 0.5)))
    init_candidates.append(("tukey_07", make_tukey(n_parent, 0.7)))
    
    # Random perturbations
    for i in range(8):
        noise = rng.standard_normal(n_parent) * 0.02
        a_pert = np.maximum(parent + noise, 0)
        a_pert = a_pert / a_pert.sum()
        init_candidates.append((f"pert{i}", a_pert))
    
    # Edge-weighted
    for edge_w in [0.3, 0.5]:
        x = np.arange(n_parent)
        center = n_parent / 2.0
        a_edge = np.exp(-edge_w * (x - center) ** 2 / n_parent)
        a_edge = np.maximum(a_edge, 1e-15)
        a_edge /= a_edge.sum()
        init_candidates.append((f"edge_{edge_w}", a_edge))
    
    best_init = None
    best_init_obj = float('inf')
    for name, a in init_candidates:
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            continue
        a = a / s
        c = fast_convolve(a)
        obj = 2 * len(a) * c.max()
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a.copy()
        try_seq(a, f"init_{name}")
    
    print(f"  Best init obj: {best_init_obj:.8f}")
    print(f"  Phase 1 time: {time.monotonic()-t1s:.1f}s")
    
    # === Phase 2: Nesterov GD ===
    print(f"\n=== Phase 2: Nesterov GD (n={len(best_init)}) ===")
    t2s = time.monotonic()
    rem = deadline - time.monotonic() - 500
    t_alloc = max(20, min(rem, 300))
    print(f"  Alloc: {t_alloc:.0f}s")
    
    a_opt, obj = nesterov_gd(best_init, time_limit=t_alloc, verbose=True)
    try_seq(a_opt, "ngd")
    
    # Try from parent if different
    if not np.allclose(best_init, parent, atol=1e-6):
        rem = deadline - time.monotonic() - 400
        if rem > 100:
            t_a2 = min(rem * 0.3, 150)
            a_opt2, obj2 = nesterov_gd(parent, time_limit=t_a2)
            try_seq(a_opt2, "ngd_parent")
    
    print(f"  Phase 2 time: {time.monotonic()-t2s:.1f}s")
    
    # === Phase 3: Peak Shrinkage ===
    print(f"\n=== Phase 3: Peak Shrinkage ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 40:
            t_alloc = min(rem * 0.3, 150)
            print(f"  Alloc: {t_alloc:.0f}s")
            a_opt, obj = peak_shrinkage(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "shrink")
    
    # === Phase 4: Active-set LP-FW ===
    print(f"\n=== Phase 4: LP-FW Active Set ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 60:
            t_alloc = min(rem * 0.5, 250)
            print(f"  Alloc: {t_alloc:.0f}s")
            a_opt, obj = lp_fw_active(best_seq, time_limit=t_alloc, num_active=40, verbose=True)
            try_seq(a_opt, "lpfw_active")
    
    # === Phase 5: More LP-FW with different active set sizes ===
    print(f"\n=== Phase 5: LP-FW Variants ===")
    if best_seq is not None:
        for na in [20, 60, 100]:
            rem = deadline - time.monotonic() - 200
            if rem < 60:
                break
            t_alloc = min(rem * 0.3, 120)
            a_opt, obj = lp_fw_active(best_seq, time_limit=t_alloc, num_active=na)
            try_seq(a_opt, f"lpfw_na{na}")
    
    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-Length ===")
    base = best_seq if best_seq is not None else best_init
    for target_n in [1024, 896, 640, 1280]:
        rem = deadline - time.monotonic() - 150
        if rem < 80:
            break
        
        f_interp = interpolate_sequence(base, target_n)
        try_seq(f_interp, f"interp_{target_n}")
        
        # Quick Nesterov GD
        t_gd = min(rem * 0.3, 80)
        if t_gd > 20:
            a_opt, obj = nesterov_gd(f_interp, time_limit=t_gd)
            try_seq(a_opt, f"ngd_{target_n}")
        
        # LP-FW
        rem2 = deadline - time.monotonic() - 100
        if rem2 > 50:
            t_lpfw = min(rem2 * 0.4, 100)
            a_opt, obj = lp_fw_active(a_opt, time_limit=t_lpfw, num_active=40)
            try_seq(a_opt, f"lpfw_{target_n}")
    
    # === Phase 7: Deep LP-FW with restarts ===
    print(f"\n=== Phase 7: Deep LP-FW ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(5):
            rem = deadline - time.monotonic() - 50
            if rem < 40:
                break
            t_alloc = min(rem / (5 - p), 100)
            
            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.002
                f_init = project_simplex(best_seq + noise)
            elif p == 2:
                rc = make_raised_cosine(n_best)
                f_init = 0.7 * best_seq + 0.3 * rc
                f_init = f_init / f_init.sum()
            elif p == 3:
                # Reverse blend
                f_init = 0.8 * best_seq + 0.2 * best_seq[::-1]
                f_init = f_init / f_init.sum()
            else:
                # Mix with Gaussian
                g_init = make_gaussian(n_best, 0.35)
                f_init = 0.85 * best_seq + 0.15 * g_init
                f_init = f_init / f_init.sum()
            
            print(f"  Variant {p}: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_active(f_init, time_limit=t_alloc, num_active=50)
            try_seq(a_opt, f"deep_{p}")
    
    # === Phase 8: Final coordination ===
    print(f"\n=== Phase 8: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 30:
            t_alloc = min(rem, 60)
            a_opt, obj = coordinate_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord_final")
        
        # One more LP-FW
        rem = deadline - time.monotonic() - 5
        if rem > 20:
            a_opt, obj = lp_fw_active(a_opt, time_limit=min(rem, 40), num_active=50)
            try_seq(a_opt, "final_lpfw")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
