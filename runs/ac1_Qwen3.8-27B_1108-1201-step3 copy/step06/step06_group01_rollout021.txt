"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive Riemannian optimization
on the probability simplex with symmetry constraints, multi-peak subgradients,
Nesterov momentum, spectral refinement, and multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project v onto the probability simplex {x : x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def get_subgradient(a, top_k=20, temperature=0.01):
    """Compute subgradient of 2n*max(conv(a,a)) at point a."""
    n = len(a)
    c = fast_convolve(a)
    # Get top-k peaks
    k_actual = min(top_k, len(c))
    peaks = np.argpartition(c, -k_actual)[-k_actual:]
    pv = c[peaks]
    
    # Soft-max weighting
    max_pv = pv.max()
    scaled = (pv - max_pv) / (temperature * max_pv + 1e-15)
    w = np.exp(scaled)
    w /= w.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[i] * a[k - idx]
    
    return grad, c.max()

    lr = 0.01  # initial learning rate
    t0 = time.monotonic()
    no_improve = 0

**Pairwise coordinate descent:**
At the current peak k*, the contribution is c[k*] = sum_j a[j]*a[k*-j].
The top contributors are the pairs (j, k*-j) with the largest product a[j]*a[k*-j].

For each such pair, we can try:
- Decrease a[j] by δ, increase a[k*-j] by δ (or vice versa) - this keeps sum constant
- Decrease a[j] by δ, increase some other a[m] by δ - this redistributes mass

The optimal δ can be found by a 1D line search (since we're only changing 2 coordinates).

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive Riemannian subgradient
descent on the probability simplex with Nesterov momentum, backtracking line
search, pairwise coordinate descent, spectral refinement, and multi-length
exploration. Avoids LP solvers for faster per-iteration cost, enabling many
more optimization steps within the time budget. Key innovations: (1) exact
subgradient with adaptive peak weighting, (2) 2-coordinate exact line search
on the peak, (3) iterative symmetrization, (4) spectral low-pass filtering
to reduce high-frequency oscillation, (5) progressive temperature annealing
for the multi-peak soft-max.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project v onto the probability simplex {x : x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if cond.any():
        rho = int(np.nonzero(cond)[0][-1])
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def eval_score(a):
    """Compute the objective 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    a = a / s
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()


def get_subgradient(a, top_k=20, temperature=0.01):
    """
    Compute weighted subgradient of max_k conv(a,a)_k at point a.
    Returns (gradient, max_value, active_peaks).
    """
    n = len(a)
    c = fast_convolve(a)
    k_actual = min(top_k, len(c))
    peaks = np.argpartition(c, -k_actual)[-k_actual:]
    pv = c[peaks]
    max_pv = pv.max()
    if max_pv < 1e-15:
        return np.zeros(n), 0.0, peaks
    scaled = (pv - max_pv) / (temperature * max_pv + 1e-30)
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[i] * a[k - idx]
    return grad, float(max_pv), peaks


def symmetrize(a):
    """Enforce palindromic symmetry: a[i] = a[n-1-i]."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s < 1e-15:
        return a.copy()
    return a_sym / s


def spectral_smooth(a, alpha=0.1):
    """Apply low-pass filter in frequency domain to reduce high-frequency content."""
    n = len(a)
    fft_size = 1 << (n - 1).bit_length()
    A = rfft(a, fft_size)
    freqs = np.arange(fft_size // 2 + 1)
    # Butterworth-like low-pass: keep low frequencies, attenuate high
    cutoff = max(2, n // 4)
    mask = np.ones(fft_size // 2 + 1)
    for f in range(cutoff, fft_size // 2 + 1):
        mask[f] = alpha
    A *= mask
    a_smooth = irfft(A, fft_size)[:n]
    a_smooth = np.maximum(a_smooth, 0.0)
    s = a_smooth.sum()
    if s < 1e-15:
        return a.copy()
    return a_smooth / s


def riemnnian_optimize(a0, time_limit, top_k=25, use_sym=False, lr_init=0.008,
                       momentum=0.9, verbose=False):
    """
    Riemannian subgradient descent on the simplex with Nesterov momentum
    and adaptive backtracking.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    v = np.zeros(n)
    beta = momentum
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    step = 0
    temp = 0.01

    while time.monotonic() - t0 < time_limit:
        step += 1

        grad, _, _ = get_subgradient(a, top_k=top_k, temperature=temp)

        # Nesterov: take step from lookahead point
        a_look = a + beta * v
        a_look = np.maximum(a_look, 0)
        sl = a_look.sum()
        if sl > 1e-15:
            a_look = a_look / sl
        grad_look, _, _ = get_subgradient(a_look, top_k=top_k, temperature=temp)
        v = beta * v - lr * grad_look

        # Candidate
        a_cand = a + v
        a_new = simplex_project(a_cand)
        a_new = a_new / a_new.sum()

        if use_sym:
            a_new = symmetrize(a_new)

        score = 2.0 * n * fast_convolve(a_new).max()

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            # Backtracking: try smaller step
            for shrink in [0.5, 0.25, 0.1]:
                a_cand = a + shrink * v
                a_new = simplex_project(a_cand)
                a_new = a_new / a_new.sum()
                if use_sym:
                    a_new = symmetrize(a_new)
                score = 2.0 * n * fast_convolve(a_new).max()
                if score < best_score - 1e-14:
                    best_score = score
                    best_a = a_new.copy()
                    a = a_new
                    no_improve = 0
                    break

            no_improve += 1
            lr *= 0.95

        # Adaptive learning rate
        if no_improve > 100:

        elif no_improve == 0 and step > 200:
            lr = min(lr * 1.005, 0.03)

        # Animate temperature (start broad, narrow down)
        progress = min(1.0, step / max(1, int(time_limit * 2)))

def pairwise_cd(a0, time_limit, top_pairs=5, max_delta=0.05, verbose=False):
    """
    Pairwise coordinate descent: at the active peak, try adjusting pairs
    (i, j) with i+j = k* to reduce the peak.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        # Find the global peak
        k_star = int(np.argmax(c))
        peak_val = c[k_star]

        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_star - idx]

        if not np.any(contribs > 1e-20):
            break

        top_idx = np.argsort(contribs)[::-1][:top_pairs]
        improved = False

        for pi in top_idx:
            i = int(idx[pi])
            j = k_star - i
            if i == j or i >= n or j >= n or i < 0 or j < 0:
                continue
            if a[i] < 1e-15 or a[j] < 1e-15:
                continue

            # Try transferring mass from i to j or vice versa
            direction = 1.0 if a[i] > a[j] else -1.0
            for delta_scale in [1.0, 0.5, 0.25, 0.1, 0.05]:
                delta = direction * delta_scale * max_delta
                new_i = a[i] + delta
                new_j = a[j] - delta
                if new_i < 0 or new_j < 0:

                a_trial = a.copy()
                a_trial[i] = new_i
                a_trial[j] = new_j
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

            if improved:
                break

        if not improved:
            stall += 1
            if stall > 50:
                # Try a different peak
                c_sorted = np.sort(c)[::-1]
                if len(c_sorted) > 1 and c_sorted[1] > c_sorted[0] * 0.99:

                    k_mask = np.zeros(len(c))
                    k_mask[k_star] = -1.0
                    k_star2 = int(np.argmax(k_mask * np.where(c > 0, c, 0)))
                    js2 = max(0, k_star2 - n + 1)
                    je2 = min(n - 1, k_star2)
                    idx2 = np.arange(js2, je2 + 1)
                    contribs2 = a[idx2] * a[k_star2 - idx2]
                    top_idx2 = np.argsort(contribs2)[::-1][:3]
                    for pi2 in top_idx2:
                        i2 = int(idx2[pi2])
                        j2 = k_star2 - i2
                        if i2 == j2 or i2 >= n or j2 >= n or i2 < 0 or j2 < 0:
                            continue
                        if a[i2] < 1e-15 or a[j2] < 1e-15:
                            continue
                        direction2 = 1.0 if a[i2] > a[j2] else -1.0
                        for delta_scale in [0.5, 0.25, 0.1]:
                            delta = direction2 * delta_scale * max_delta
                            new_i = a[i2] + delta
                            new_j = a[j2] - delta
                            if new_i < 0 or new_j < 0:
                                continue
                            a_trial = a.copy()
                            a_trial[i2] = new_i
                            a_trial[j2] = new_j
                            a_trial = a_trial / a_trial.sum()
                            score = 2.0 * n * fast_convolve(a_trial).max()
                            if score < best_score - 1e-14:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
            if not improved and stall > 200:
                break
        else:
            stall = 0

        if verbose and (stall % 500 == 0 and stall > 0):
            elapsed = time.monotonic() - t0
            print(f"    PairCD {elapsed:.0f}s: score={best_score:.8f} stall={stall}")

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T=300, maxiter=5000, verbose=False):
    """
    L-BFGS refinement using soft-max smoothed objective in log-space.
    The soft-max approximates the max function, making the objective smooth.
    """
    n = len(a0)
    a0 = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a0 = a0 / a0.sum()
    z0 = np.log(a0)
    t0 = time.monotonic()
    stopped = [False]

    def obj_and_grad(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15, np.zeros_like(z)
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        c = fast_convolve(a)
        max_c = c.max()
        scaled_c = c / (T * max_c + 1e-30)
        exp_c = np.exp(scaled_c - scaled_c.max())
        log_sum_exp = scaled_c.max() + np.log(exp_c.sum())
        obj = 2.0 * n * (T * max_c + 1e-30) * log_sum_exp
        w = exp_c / exp_c.sum()
        # Gradient of max-conv wrt a: for each lag k, d c[k]/da[i] = 2*a[k-i] (if valid)
        grad_a = np.zeros(n)
        for k in range(min(2 * n - 1, 2 * n - 1)):
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        # Chain rule: d obj / d z = a * (d obj / d a - a * sum(a * d obj / d a))
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        return obj, grad_z

    def func(z):
        return obj_and_grad(z)[0]

    def jac(z):
        return obj_and_grad(z)[1]

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        z_out = res.x
        a_out = np.exp(z_out - z_out.max())
        a_out = a_out / a_out.sum()
        a_out = np.maximum(a_out, 0)
        a_out = a_out / a_out.sum()
    except Exception:
        a_out = a0.copy()

    return a_out


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving non-negativity and sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate sequence and update best if improved."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        score = np.inf
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def diverse_initializations(n, parent=None):
    """Generate diverse initial sequences of length n."""
    rng = np.random.default_rng(42)
    inits = []

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.copy())
        else:
            inits.append(interpolate_sequence(parent, n))

    # Raised cosine
    x = np.linspace(0, 1, n)
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    rc = rc + 1e-6
    rc = rc / rc.sum()
    inits.append(rc)

    # Truncated Gaussian
    x = np.arange(n)
    center = n / 2.0
    sigma = n / 4.0
    g = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    g = g / g.sum()
    inits.append(g)

    # Low-pass filtered random
    for _ in range(3):
        r = rng.standard_normal(n)
        r = spectral_smooth(r, alpha=0.1)
        r = np.maximum(r, 0)
        rs = r.sum()
        if rs > 0.01:
            inits.append(r / rs)

    # Two-bump (mass near edges)
    bump = np.zeros(n)
    bw = max(3, n // 8)
    bump[:bw] = np.linspace(1, 0.3, bw)
    bump[-bw:] = np.linspace(0.3, 1, bw)
    bump = np.maximum(bump, 0)
    bump = bump / bump.sum()
    inits.append(bump)

    # Anti-symmetric perturbation of uniform (to break symmetry)
    t = np.linspace(0, 2 * np.pi, n)
    for freq in [2, 3, 5]:
        u = np.ones(n) / n + 0.1 * np.cos(freq * t)
        u = np.maximum(u, 0)
        u = u / u.sum()
        inits.append(u)

    return inits


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Multi-start Riemannian subgradient with Nesterov momentum.
    2. Pairwise coordinate descent on active peak.
    3. L-BFGS soft-max polishing.
    4. Spectral smoothing + re-optimization.
    5. Multi-length exploration.
    6. Final intensive local search.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all():
            ps = p.sum()
            if ps > 1e-10:
                parent = p / ps
                p_score = float(evaluate_sequence(parent.tolist()))
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
                best_seq = parent.copy()
                best_score = p_score
    except Exception:
        pass

    if parent is None:
        best_seq = np.ones(768) / 768.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default init: score={best_score:.8f}")

    n_parent = len(parent) if parent is not None else 768

    # === Phase 1: Multi-start Riemannian optimization at parent length ===
    print(f"\n=== Phase 1: Riemannian Multi-Start (n={n_parent}) ===")
    n_main = n_parent
    inits = diverse_initializations(n_main, parent)

    n_starts = min(len(inits), 5)
    rem = deadline - time.monotonic()
    time_per_start = min(rem * 0.35 / max(1, n_starts), 180.0)

    for si in range(n_starts):
        rem = deadline - time.monotonic() - 300
        if rem < 30:
            break
        t_alloc = min(time_per_start, rem * 0.9)
        if t_alloc < 15:
            break
        print(f"  Start {si+1}/{n_starts}, n={n_main}, budget={t_alloc:.0f}s")
        a_opt, score_opt = riemnnian_optimize(
            inits[si], t_alloc, top_k=25, use_sym=False,
            lr_init=0.008, momentum=0.9, verbose=False
        )
        best_seq, best_score = check_and_update(a_opt, f"riem_s{si+1}", best_seq, best_score)

        # Try symmetrized version
        a_sym = symmetrize(a_opt)
        best_seq, best_score = check_and_update(a_sym, f"riem_s{si+1}_sym", best_seq, best_score)

    # === Phase 2: Pairwise coordinate descent ===
    print(f"\n=== Phase 2: Pairwise CD (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(rem * 0.3, 200.0)
            print(f"  Pairwise CD, n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_cd, _ = pairwise_cd(best_seq, t_alloc, top_pairs=5, max_delta=0.03)
            best_seq, best_score = check_and_update(a_cd, "paircd", best_seq, best_score)

    # === Phase 3: L-BFGS soft-max polish ===
    print(f"\n=== Phase 3: L-BFGS Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        for T in [100, 500]:
            rem = deadline - time.monotonic() - 100
            if rem < 20:
                break
            t_alloc = min(rem * 0.3, 100.0)
            print(f"  L-BFGS T={T}, n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_lb = lbgfs_refine(best_seq, t_alloc, T=T, maxiter=3000)
            best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T}", best_seq, best_score)

            # Symmetrize
            a_sym = symmetrize(a_lb)
            best_seq, best_score = check_and_update(a_sym, f"lbgfs_T{T}_sym", best_seq, best_score)

    # === Phase 4: Riemannian with symmetry constraint ===
    print(f"\n=== Phase 4: Symmetric Riemannian (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_alloc = min(rem * 0.4, 180.0)
            print(f"  Sym RiemGrad, n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_sym0 = symmetrize(best_seq)
            a_sym_opt, _ = riemnnian_optimize(
                a_sym0, t_alloc, top_k=30, use_sym=True,
                lr_init=0.006, momentum=0.92
            )
            best_seq, best_score = check_and_update(a_sym_opt, "sym_riem", best_seq, best_score)

    # === Phase 5: Spectral smooth + re-optimize ===
    print(f"\n=== Phase 5: Spectral Refinement (best={best_score:.8f}) ===")
    if best_seq is not None:
        for alpha in [0.05, 0.1, 0.3]:
            rem = deadline - time.monotonic() - 100
            if rem < 20:
                break
            a_smooth = spectral_smooth(best_seq, alpha=alpha)
            a_smooth = symmetrize(a_smooth)
            # Quick Riemannian polish
            t_alloc = min(rem * 0.2, 60.0)
            a_sp, _ = riemnnian_optimize(a_smooth, t_alloc, top_k=20,
                                         use_sym=True, lr_init=0.005, momentum=0.9)
            best_seq, best_score = check_and_update(a_sp, f"spectral_a{alpha}",
                                                    best_seq, best_score)

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-Length (best={best_score:.8f}) ===")
    n_best = len(best_seq) if best_seq is not None else n_parent
    alt_lengths = []
    for factor in [0.75, 0.85, 1.1, 1.25, 1.5, 2.0]:
        alt_n = int(n_best * factor)
        if 128 <= alt_n <= 4096 and alt_n != n_best and alt_n % 2 == 0:
            if alt_n not in alt_lengths:
                alt_lengths.append(alt_n)

    for alt_n in alt_lengths[:3]:
        rem = deadline - time.monotonic() - 80
        if rem < 40:
            break
        t_alloc = min(rem * 0.35, 120.0)
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  Alt length n={alt_n}, budget={t_alloc:.0f}s")
        a_alt, _ = riemnnian_optimize(init_alt, t_alloc, top_k=25,
                                      use_sym=False, lr_init=0.007, momentum=0.9)
        best_seq, best_score = check_and_update(a_alt, f"alt_n{alt_n}", best_seq, best_score)

        # Pairwise CD on alt
        rem = deadline - time.monotonic() - 60
        if rem > 15 and best_seq is not None and len(best_seq) == alt_n:
            a_cd_alt, _ = pairwise_cd(best_seq, min(rem * 0.5, 40.0), top_pairs=4, max_delta=0.02)
            best_seq, best_score = check_and_update(a_cd_alt, f"alt_n{alt_n}_cd",
                                                    best_seq, best_score)

    # === Phase 7: Intensive final local search ===
    print(f"\n=== Phase 7: Final Local Search (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_f = len(best_seq)
        # Multiple short Riemannian runs with different hyperparams
        configs = [
            {'top_k': 10, 'lr_init': 0.003, 'momentum': 0.95},
            {'top_k': 50, 'lr_init': 0.005, 'momentum': 0.85},
            {'top_k': 20, 'lr_init': 0.002, 'momentum': 0.92},
        ]
        for ci, cfg in enumerate(configs):
            rem = deadline - time.monotonic() - 30
            if rem < 10:
                break
            t_alloc = min(rem * 0.3, 60.0)
            a_f, _ = riemnnian_optimize(best_seq, t_alloc, use_sym=False, **cfg)
            best_seq, best_score = check_and_update(a_f, f"final_cfg{ci}", best_seq, best_score)

        # Pairwise CD final
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_cd_f, _ = pairwise_cd(best_seq, min(rem * 0.6, 30.0), top_pairs=8, max_delta=0.01)
            best_seq, best_score = check_and_update(a_cd_f, "final_cd", best_seq, best_score)

        # Micro-perturbation search
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            rng = np.random.default_rng(12345)
            for _ in range(int(rem * 50)):
                if time.monotonic() > deadline - 3:
                    break
                a_pert = best_seq + rng.standard_normal(n_f) * 0.001
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
                sp = 2.0 * n_f * fast_convolve(a_pert).max()
                if sp < best_score - 1e-12:
                    best_score = sp
                    best_seq = a_pert.copy()

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification via evaluate_sequence
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
