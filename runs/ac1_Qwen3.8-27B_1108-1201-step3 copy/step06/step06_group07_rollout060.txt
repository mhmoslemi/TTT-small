Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding
combined with mirror descent, adaptive line search, and diverse asymmetric
initializations. Key innovation: LP-based optimal direction search that
finds the best step direction while maintaining the convolution bound.

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def cross_convolve(f, g):
    """Linear cross-convolution via FFT."""
    n, m = len(f), len(g)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    G = np.fft.rfft(g, fft_size)
    return np.fft.irfft(F * G, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_convolution_matrix(n, f):
    """Build the (2n-1) x n matrix A where A[k,j] = f[k-j] if valid."""
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je >= js:
            j_idx = np.arange(js, je+1)
            A[k, js:je+1] = f[k - j_idx]
    return A

def solve_lp_direction(f, M):
    """
    Solve: max sum(g) s.t. g >= 0, A@f... 
    Wait, the constraint is max_k sum_j A[k,j]*g_j <= M where A[k,j] = f[k-j].
    So: A @ g <= M (for all rows), g >= 0, maximize sum(g).
    """
    n = len(f)
    A = build_convolution_matrix(n, f)
    b = np.full(2*n-1, M)
    c = -np.ones(n)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x, -res.fun
    else:
        return f.copy(), f.sum()

def lp_optimize(f0, time_limit, verbose=False, label=""):
    """LP-based optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f0.copy(), float('inf')
    f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Solve LP
        g_lp, sum_g = solve_lp_direction(f, M)
        
        # Check if LP found improvement
        f_sum = f.sum()  # should be 1
        if sum_g <= f_sum * (1.001):
            # Try slightly tighter constraint
            g_lp, sum_g = solve_lp_direction(f, M * 0.998)
            if sum_g <= f_sum * (1.001):
                no_improve += 1
                if no_improve > 30:
                    break
                continue
        
        # Normalize direction to simplex
        g_dir = g_lp / sum_g
        
        # Line search
        best_t = 0.0
        best_trial = best_obj
        for t in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            trial = (1 - t) * f + t * g_dir
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            trial = trial / ts
            obj = fast_convolve(trial).max()
            if obj < best_trial:
                best_trial = obj
                best_t = t
        
        if best_t > 0:
            f = (1 - best_t) * f + best_t * g_dir
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            new_obj = fast_convolve(f).max()
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 5 == 0:
                    print(f"    {label} LP iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
    
    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based optimal direction finding
(the SOTA approach), combined with adaptive line search, diverse asymmetric
initializations, and mirror descent fallback. The LP at each step finds the
direction g that maximizes sum(g) subject to g>=0 and max(cross_conv(f,g))<=M,
identifying the best feasible step that reduces the convolution peak.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(f, g):
    """Linear cross-convolution via FFT."""
    n, m = len(f), len(g)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    G = np.fft.rfft(g, fft_size)
    return np.fft.irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(n, f):
    """Build (2n-1) x n matrix where A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je >= js:
            j_idx = np.arange(js, je + 1)
            A[k, js:je + 1] = f[k - j_idx]
    return A


def solve_lp_direction(f, M):
    """
    Solve: max sum(g) s.t. g >= 0, (f*g)_k <= M for all k.
    Returns (g, sum_g).
    """
    n = len(f)
    A = build_conv_matrix(n, f)
    b = np.full(2 * n - 1, M)
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return f.copy(), f.sum()


def lp_optimize(f0, time_limit, verbose=False, label="", use_tight=False):
    """
    LP-based optimization: iteratively find best direction via LP,
    then do line search to update.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f0.copy(), float('inf')
    f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    lp_time_total = 0.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Solve LP for direction
        t_lp = time.monotonic()
        g_lp, sum_g = solve_lp_direction(f, M)
        lp_time_total += time.monotonic() - t_lp

        if sum_g <= 1.0001:
            # Try tighter M for more aggressive movement
            if use_tight:
                g_lp, sum_g = solve_lp_direction(f, M * 0.995)
            if sum_g <= 1.0001:
                no_improve += 1
                if no_improve > 40:
                    break
                continue

        # Normalize direction to simplex
        g_dir = g_lp / sum_g

        # Line search
        best_t = 0.0
        best_trial = best_obj
        for t in [0.01, 0.02, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            trial = (1.0 - t) * f + t * g_dir
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            trial = trial / ts
            obj = fast_convolve(trial).max()
            if obj < best_trial:
                best_trial = obj
                best_t = t

        if best_t > 0:
            f = (1.0 - best_t) * f + best_t * g_dir
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            new_obj = fast_convolve(f).max()
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    avg_lp = lp_time_total / it
                    print(f"    {label} LP iter {it}: obj={best_obj:.10f}, "
                          f"t={best_t:.3f}, sum_g={sum_g:.6f}, avg_lp={avg_lp*1000:.1f}ms")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter {it}: best={best_obj:.10f}, "
                  f"elapsed={elapsed:.0f}s, no_improve={no_improve}")

    return best_f, best_obj


def mirror_descent(a0, time_limit, top_k=30, temp_init=0.0005, lr_init=0.01,
                   lr_decay=0.9999, verbose=False, label=""):
    """Fast multiplicative weight mirror descent on the simplex."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        K = min(top_k, len(conv))
        if K < len(conv):
            peak_indices = np.argpartition(conv, -K)[-K:]
        else:
            peak_indices = np.arange(len(conv))

        pv = conv[peak_indices]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peak_indices))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient
        g = np.zeros(n)
        for idx in range(len(peak_indices)):
            k = int(peak_indices[idx])
            w_k = w[idx]
            if w_k < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            g[js:je + 1] += 2.0 * w_k * a[k - np.arange(js, je + 1)]

        # Backtracking
        best_lr = lr * 0.1
        best_trial_obj = M * 1.01
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            if trial_lr < 1e-10:
                continue
            log_update = np.clip(-trial_lr * g, -50, 50)
            a_trial = a * np.exp(log_update)
            a_trial = np.maximum(a_trial, 1e-15)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial = a_trial / ss
            trial_obj = fast_convolve(a_trial).max()
            if trial_obj <= best_trial_obj:
                best_lr = trial_lr
                best_trial_obj = trial_obj
                if trial_obj < M * 0.999:
                    break

        log_update = np.clip(-best_lr * g, -50, 50)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        new_obj = fast_convolve(a).max()
        if new_obj < best_obj - 1e-13:
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    {label} MD iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        temp = min(temp * 1.002, 0.05)
        if no_improve > 100:
            lr *= 0.5
            no_improve = 0
        lr = max(lr * lr_decay, 1e-8)

        if no_improve > 500:
            rng = np.random.default_rng(it * 7919 + 13)
            noise = 0.001 + 0.002 * min(no_improve / 5000, 1.0)
            a = np.maximum(best_a + rng.standard_normal(n) * noise, 1e-15)
            a = a / a.sum()
            lr = lr_init * 0.3
            no_improve = 0

        if lr < 1e-9:
            break

    return best_a, best_obj


def make_asymmetric_ramp(n, alpha=1.0):
    """Linearly decaying sequence (asymmetric)."""
    x = np.arange(n)
    a = np.exp(-alpha * x / n)
    a = a / a.sum()
    return a


def make_asymmetric_gaussian(n, center_frac=0.35, sigma_frac=0.25):
    """Gaussian centered off-center (asymmetric)."""
    x = np.arange(n)
    center = n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = a / a.sum()
    return a


def make_waterfall(n, p=0.7):
    """Waterfall-like: decays but with random perturbations."""
    rng = np.random.default_rng(12345)
    a = np.ones(n)
    for i in range(1, n):
        a[i] = a[i-1] * p * rng.uniform(0.8, 1.2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell_asymmetric(n, width=0.5, shift=0.1):
    """Bell shape shifted to one side."""
    x = np.linspace(0, 1, n)
    center = 0.5 + shift
    a = np.exp(-((x - center) / width) ** 2)
    a = a / a.sum()
    return a

def _coordinate_descent(a0, time_limit, verbose=False, label=""):
    """Fine coordinate descent: adjust one coordinate at a time."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    rounds = 0

    while time.monotonic() - t0 < time_limit:
        rounds += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-13:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 0.5]:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0.0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial = trial / ss
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a
