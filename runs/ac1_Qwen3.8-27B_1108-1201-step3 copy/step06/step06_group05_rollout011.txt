import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def golden_section_line_search(f, g, f0, max_iter=30):
    """
    Golden section line search for alpha in [0, 1] minimizing f_f(alpha) = objective of (1-alpha)*f0 + alpha*(g-f0).
    """
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    
    def obj(alpha):
        ft = (1 - alpha) * f0 + alpha * g
        ft = np.maximum(ft, 0)
        s = ft.sum()
        if s < 1e-15:
            return np.inf
        ft = ft / s
        return fast_convolve(ft).max()
    
    a = 0.0
    b = 1.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = obj(c)
    fd = obj(d)
    
    for _ in range(max_iter):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = obj(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = obj(d)
        if abs(b - a) < 1e-10:
            break
    
    best_alpha = (a + b) / 2
    return best_alpha, obj(best_alpha)


def lp_direction(f, M, n):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0, sum(g) = 1.
    Returns the optimal g or None if infeasible.
    """
    # Variables: g_0, ..., g_{n-1}
    # Maximize sum(g) = -minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (f*g)_k <= M for k = 0, ..., 2n-2
    # (f*g)_k = sum_j f_j * g_{k-j} (where g_i = 0 for i outside [0, n-1])
    # For a given k, the valid j are those where 0 <= k-j <= n-1, i.e., max(0, k-n+1) <= j <= min(n-1, k)
    
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j+n, j] = f
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        g = res.x
        s = g.sum()
        if s > 1e-15:
            return g / s
        return None
    return None

            obj_trial = c_trial.max()

                if obj_trial < best_obj - 1e-13:

                    no_improve = 0

                momentum = np.zeros(n)
                lr = 1.0 / (beta * 2.0 * np.sqrt(n))

def peak_shaving(f0, time_limit, verbose=False):
    """
    Directly target the top peaks using a subgradient method with stepsize decay.
    Uses the insight that for the top K peaks, the subgradient is a weighted sum of
    the "rows" of the convolution matrix corresponding to those peaks.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    lr = 0.02
    K = 20  # Number of top peaks to target
    momentum = np.zeros(n)
    mu = 0.7
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(f)
        M = c.max()
        
        # Select top K peaks with weights based on their excess
        k_peaks = min(K, len(c))
        peak_idx = np.argpartition(c, -k_peaks)[-k_peaks:]
        
        # Weights: exponential of normalized excess
        peak_vals = c[peak_idx]
        max_peak = peak_vals.max()
        min_peak = peak_vals.min()
        if max_peak - min_peak < 1e-15:
            w = np.ones(len(peak_idx)) / len(peak_idx)
        else:
            w = np.exp(50 * (peak_vals - max_peak) / (min_peak - max_peak + 1e-15))
            w = w / w.sum()
        
        # Compute weighted subgradient
        sg = np.zeros(n)
        for i in range(len(peak_idx)):
            if w[i] < 1e-10:
                continue
            k = int(peak_idx[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += w[i] * 2 * f[k - idx]
        
        # Update with momentum
        momentum = mu * momentum - lr * sg
        f_trial = f + momentum
        f_trial = project_simplex(f_trial)
        
        c_trial = fast_convolve(f_trial)
        obj_trial = c_trial.max()
        
        if obj_trial < best_obj - 1e-14:
            best_obj = obj_trial
            best_f = f.copy()
            f = f_trial
            no_improve = 0
            lr = min(lr * 1.01, 0.05)
        else:
            no_improve += 1
            lr *= 0.95
            f = f  # Don't update if no improvement
        
        if no_improve > 100:
            rng = np.random.default_rng(it * 31 + 17)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            momentum = np.zeros(n)
            lr = 0.015
            no_improve = 0
        
        if lr < 1e-7:
            break
    
    return best_f, best_obj


def coordinate_polish(f0, time_limit, verbose=False):
    """
    Coordinate descent: adjust pairs of coordinates to reduce max convolution.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f0, fast_convolve(f0).max()
    f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    # Target the top peaks
    c = fast_convolve(f)
    top_peaks = np.argpartition(c, -10)[-10:]
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        c = fast_convolve(f)
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-10:
                continue
            
            best_j_obj = c.max()
            best_j_val = old_val
            
            # Try shifting mass from j to a "good" location
            # The best location to add mass is where it causes the least increase in peaks
            for mult in [0.8, 0.9, 1.1, 1.2]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                
                # Redistribute the difference to other coordinates proportionally
                diff = new_val - old_val
                if abs(diff) < 1e-12:
                    continue
                
                f_trial = f.copy()
                f_trial[j] = new_val
                # Distribute -diff to other coordinates
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                others_sum = f[mask].sum()
                if others_sum < 1e-10:
                    continue
                f_trial[mask] -= diff * (f[mask] / others_sum)
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-10:
                    continue
                f_trial = f_trial / ss
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                diff = best_j_val - old_val
                f[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                others_sum = f[mask].sum()
                if others_sum > 1e-10:
                    f[mask] -= diff * (f[mask] / others_sum)
                f = np.maximum(f, 0)
                f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            # Try a different strategy: swap mass between coordinates
            if time.monotonic() - t0 < time_limit * 0.5:
                c = fast_convolve(f)
                max_idx = c.argmax()
                # Find coordinates that contribute to the max peak
                js = max(0, int(max_idx) - n + 1)
                je = min(n - 1, int(max_idx))
                contributors = np.arange(js, je + 1)
                
                # Try moving mass from high contributors to low contributors
                for j_from in contributors:
                    if time.monotonic() - t0 > time_limit:
                        break
                    if f[j_from] < 1e-8:
                        continue
                    
                    # Try moving a small amount to a random other coordinate
                    for _ in range(5):
                        j_to = np.random.randint(n)
                        if j_to == j_from:
                            continue
                        
                        for frac in [0.05, 0.1, 0.2]:
                            amount = f[j_from] * frac
                            f_trial = f.copy()
                            f_trial[j_from] -= amount
                            f_trial[j_to] += amount
                            f_trial = np.maximum(f_trial, 0)
                            f_trial = f_trial / f_trial.sum()
                            val = fast_convolve(f_trial).max()
                            if val < best_obj - 1e-14:
                                f = f_trial
                                best_obj = val
                                best_f = f.copy()
                                improved = True
                                break
            else:
                break
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_hann(n):
    x = np.linspace(-1, 1, n)
    a = 0.5 * (1 + np.cos(np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_raised_cosine(n, power=3):
    x = np.linspace(-1, 1, n)
    a = np.cos(np.pi * x / 2) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

Let me think about an alternative: **Direct QP approach**

At each iteration, we want to find the best step direction. Instead of solving an LP, we can solve a QP:
minimize ||a_new - a||² + lambda * max_k (a_new * a_new)_k
subject to a_new >= 0, sum(a_new) = 1

This is non-convex, but we can linearize the max term:
max_k (a_new * a_new)_k ≈ max_k [(a+a_new-a) * (a+a_new-a)]_k
≈ max_k [c_k + 2 * (A_k @ (a_new - a))]

So we get: maximize sum of top K linearized peaks + lambda * ||a_new - a||²
