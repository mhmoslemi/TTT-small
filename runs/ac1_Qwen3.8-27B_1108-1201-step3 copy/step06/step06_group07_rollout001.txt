def solve_lp_direction(a, M):
    """Solve the LP to find direction g that maximizes sum(g) subject to (a*g)_k <= M."""
    from scipy.optimize import linprog
    n = len(a)
    size = 2 * n - 1
    
    # Build constraint matrix A (sparse)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # We need to build this efficiently
    
    # For scipy linprog, we need a sparse matrix
    import scipy.sparse as sp
    
    rows = []
    cols = []
    vals = []
    for k in range(size):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = sp.coo_matrix((vals, (rows, cols)), shape=(size, n)).tocsr()
    
    # Objective: maximize sum(g) = minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: A @ g <= M (for all k), g >= 0
    b = np.full(size, M)
    
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='hihghs')
    
    if result.success:
        g = result.x
        s = g.sum()
        return g, s
    else:
        return None, 1.0

"""
LP-guided optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find optimal descent directions (maximizing sum(g)
subject to cross-convolution constraint), combined with mirror descent fallback,
adaptive line search, symmetry exploitation, and multi-start diversification.
Key innovation over previous approach: LP-based direction finding provides
provably improving directions, leading to faster and more reliable convergence.

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_lp_indices(n):
    """Precompute row/col indices for the LP constraint matrix structure."""
    size = 2 * n - 1
    rows = []
    cols = []
    for k in range(size):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32), size


def solve_direction_lp(a, M, row_idx, col_idx, size):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    # A[k, j] = a[k-j] where valid
    vals = a[row_idx - col_idx]
    A = sp.coo_matrix((vals, (row_idx, col_idx)), shape=(size, n)).tocsr()
    c = -np.ones(n)
    b = np.full(size, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})
        if result.success:
            g = result.x
            s = float(np.sum(g))
            if s > 1.001:
                return g, s
    except Exception:
        pass
    return None, 1.0


def line_search_t(a, b, M, max_iter=60):
    """Find max t in [0,1] such that max(conv((1-t)*a + t*b, (1-t)*a + t*b)) <= M."""
    c = fast_convolve(a)
    d = cross_convolve(a, b)
    e = fast_convolve(b)

    lo, hi = 0.0, 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        conv_mid = (1.0 - mid) ** 2 * c + 2.0 * mid * (1.0 - mid) * d + mid ** 2 * e
        if conv_mid.max() <= M + 1e-14:
            lo = mid
        else:
            hi = mid
    return lo


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, peak_indices, weights):
    """Compute weighted subgradient of max_k (a*a)_k."""
    n = len(a)
    g = np.zeros(n)
    for idx in range(len(peak_indices)):
        k = int(peak_indices[idx])
        w_k = weights[idx]
        if w_k < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        g[js:je + 1] += 2.0 * w_k * a[k - np.arange(js, je + 1)]
    return g

    best_a = a.copy()

    no_improve = 0
    lp_fail_count = 0

        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        M = conv.max()

        if g is None or g_sum <= 1.005:
            lp_fail_count += 1
            no_improve += 1

                                                  top_k=30, temp_init=0.001,

                        best_a = a_md.copy()
                        a = a_md.copy()
                        no_improve = 0

        lp_fail_count = 0

        # Normalize direction

            no_improve += 1
            continue

        t_use = min(t_opt * 0.9, 0.3)

        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()

        new_obj = fast_convolve(a_new).max()

        if new_obj < best_obj - 1e-14:

            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0

            # Try smaller step

            if obj_try < best_obj - 1e-14:

                best_obj = obj_try

                no_improve = 0

def coordinate_descent(a0, time_limit, verbose=False, label=""):
    """Fine coordinate descent for final polishing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            best_j_obj = best_obj
            best_j_val = old_val
            for mult in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0.0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial = trial / ss
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0.0) / np.maximum(a.sum(), 1e-15)
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        if not improved:
            break

    return best_a, best_obj


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    return a / a.sum()


def make_bell(n, width_frac=0.6):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    return a / a.sum()


def make_bimodal(n, c1=0.3, c2=0.7, sigma=0.08):
    x = np.arange(n) / n
    a = np.exp(-0.5 * ((x - c1) / sigma) ** 2) + np.exp(-0.5 * ((x - c2) / sigma) ** 2)
    return a / a.sum()


def symmetrize(a):
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym /= ss
    return a_sym


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

            best_init = parent
        score = try_seq(symmetrize(parent), "parent_sym")
        if score < best_init_score:
            best_init_score = score
            best_init = symmetrize(parent)

    if best_init is None:
        best_init = make_gaussian(768, 0.3)
        best_init_score = try_seq(best_init, "default_gauss")

    print(f"  Best init: {best_init_score:.6f} (n={len(best_init)})")
    n_best = len(best_init)

    # === Phase 2: LP-based optimization (main) ===
    print(f"\n=== Phase 2: LP Optimization (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    t_lp = remaining * 0.45
    print(f"  Time allocated: {t_lp:.0f}s")

    # Precompute LP indices
    row_idx, col_idx, size = build_lp_indices(n_best)

    a_lp, obj_lp = lp_optimize(best_init, time_limit=t_lp, verbose=True, label="LP_main")
    try_seq(a_lp, "LP_main")

    # Run LP on symmetrized version too
    if time.monotonic() < deadline - 60:
        a_sym_init = symmetrize(best_seq if best_seq is not None else best_init)
        t_lp2 = min(60.0, (deadline - time.monotonic()) * 0.15)
        if t_lp2 > 10:
            a_lp2, obj_lp2 = lp_optimize(a_sym_init, time_limit=t_lp2, verbose=True, label="LP_sym")
            try_seq(a_lp2, "LP_sym")

    # === Phase 3: Try different lengths ===
    print(f"\n=== Phase 3: Length exploration ===")
    for n_try in [n_best - 64, n_best + 64, n_best + 128]:
        if n_try < 100 or n_try > 2000:
            continue
        if time.monotonic() > deadline - 120:
            break
        # Interpolate best to new length
        a_interp = interpolate_sequence(best_seq if best_seq is not None else best_init, n_try)
        score = try_seq(a_interp, f"interp_n{n_try}")
        if score < best_score + 0.01:
            # Worth optimizing at this length
            t_short = min(60.0, (deadline - time.monotonic()) * 0.1)
            if t_short > 15:
                row_idx_s, col_idx_s, size_s = build_lp_indices(n_try)
                a_opt, _ = lp_optimize(a_interp, time_limit=t_short, verbose=False, label=f"LP_n{n_try}")
                try_seq(a_opt, f"LP_n{n_try}")

    # === Phase 4: Mirror descent polishing ===
    print(f"\n=== Phase 4: Mirror Descent Polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_md = remaining * 0.25
        if t_md > 20:
            print(f"  Time allocated: {t_md:.0f}s")
            a_md, obj_md = mirror_descent(best_seq, time_limit=t_md, top_k=50,
                                          temp_init=0.0001, lr_init=0.005,
                                          verbose=True, label="MD_polish")
            try_seq(a_md, "MD_polish")

            # MD on symmetrized
            if time.monotonic() < deadline - 30:
                a_sym = symmetrize(best_seq)
                a_md2, obj_md2 = mirror_descent(a_sym, time_limit=min(40, (deadline - time.monotonic()) * 0.1),
                                                top_k=30, temp_init=0.0002, lr_init=0.004,
                                                verbose=False, label="MD_sym2")
                try_seq(a_md2, "MD_sym2")

    # === Phase 5: Coordinate descent final polish ===
    print(f"\n=== Phase 5: Coordinate Descent ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_cd = min(remaining * 0.15, 120)
        if t_cd > 10:
            print(f"  Time allocated: {t_cd:.0f}s")
            a_cd, obj_cd = coordinate_descent(best_seq, time_limit=t_cd, verbose=True, label="CD")
            try_seq(a_cd, "CD")

    # === Phase 6: Final LP rounds ===
    print(f"\n=== Phase 6: Final LP Rounds ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            n_cur = len(best_seq)
            row_idx_f, col_idx_f, size_f = build_lp_indices(n_cur)
            t_final = min(remaining - 10, 180)
            a_final, obj_final = lp_optimize(best_seq, time_limit=t_final, verbose=True, label="LP_final")
            try_seq(a_final, "LP_final")

            # Try LP with slightly relaxed constraint
            remaining = deadline - time.monotonic()
            if remaining > 20:
                conv = fast_convolve(best_seq)
                M_cur = conv.max()
                # Use 1.005 * M to get a more relaxed direction
                a_relaxed_init = best_seq.copy()
                # Manually solve with relaxed M
                g, g_sum = solve_direction_lp(a_relaxed_init, M_cur * 1.01, row_idx_f, col_idx_f, size_f)
                if g is not None and g_sum > 1.01:
                    b = g / g_sum
                    t_opt = line_search_t(a_relaxed_init, b, M_cur)
                    if t_opt > 0.01:
                        a_try = (1 - t_opt * 0.8) * a_relaxed_init + t_opt * 0.8 * b
                        a_try = np.maximum(a_try, 0) / np.maximum(a_try.sum(), 1e-15)
                        try_seq(a_try, "LP_relaxed")

    # === Phase 7: Multi-restart with noise ===
    print(f"\n=== Phase 7: Multi-restart ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        n_rest = min(3, int(remaining // 60))
        for r in range(n_rest):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            noise_scale = 0.003 * (r + 1)
            noise = rng.standard_normal(len(best_seq)) * noise_scale
            f_init = np.maximum(best_seq + noise, 1e-15)
            f_init = f_init / f_init.sum()
            t_alloc = min(remaining - 20, 80)
            a_r, _ = mirror_descent(f_init, time_limit=t_alloc, top_k=30,
                                    temp_init=0.001, lr_init=0.008,
                                    verbose=False, label=f"restart_{r}")
            try_seq(a_r, f"restart_{r}")

    # === Final: try symmetrized best ===
    if best_seq is not None:
        a_s = symmetrize(best_seq)
        try_seq(a_s, "final_sym")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
