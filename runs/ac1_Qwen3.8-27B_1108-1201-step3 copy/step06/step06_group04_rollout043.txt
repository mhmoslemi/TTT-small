import numpy as np

def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_value(f):
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def make_gaussian(n, sigma_frac=0.25):

    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    s = a.sum()
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def subgradient_step(f, weights, lr, n):
    """Apply weighted subgradient step for multi-peak reduction."""
    s = np.zeros(n)
    for k, w in zip(*weights):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            # subgradient contribution: 2 * p_{k-j} for index j
            # and 2 * p_j for index k-j
            s[j] += w * 2.0 * f[k - j]
            ki = k - j
            if ki != j:
                s[ki] += w * 2.0 * f[j]
    # Project: subtract mean to handle sum constraint
    s_centered = s - s.mean()
    f_new = f - lr * s_centered
    f_new = np.maximum(f_new, 0.0)
    sn = f_new.sum()
    if sn < 1e-15:
        return f.copy()
    return f_new / sn

def build_lp_matrix(f, n):
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        A_ub[k_start:k_end + 1, j] = f[:length]
    return A_ub


def solve_lp_direction(f, M, n):
    A_ub = build_lp_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def coordinate_descent(f0, time_limit, verbose=False):
    """Peak-targeted coordinate descent with smart redistribution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 50:
        conv = fast_convolve(f)
        M = conv.max()
        improved = False
        
        # Focus on top peaks
        top_k = min(8, len(conv))
        peak_indices = np.argsort(conv)[::-1][:top_k]
        
        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            
            idx = np.arange(js, je + 1)
            products = f[idx] * f[k - idx]
            
            # Try reducing top contributors
            order = np.argsort(products)[::-1]
            for oi in range(min(3, len(order))):
                j_max = int(idx[order[oi]])
                old_val = f[j_max]
                if old_val < 1e-12:
                    continue
                
                for mult in [0.5, 0.6, 0.7, 0.8, 1.2, 1.4]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.5:
                        continue
                    
                    trial = f.copy()
                    trial[j_max] = new_val
                    d = old_val - new_val
                    if abs(d) > 1e-15:
                        # Redistribute to non-contributing indices
                        non_contrib = np.ones(n, dtype=bool)
                        non_contrib[js:je + 1] = False
                        if non_contrib.sum() > 0:
                            w = f[non_contrib]
                            wsum = w.sum()
                            if wsum > 1e-15:
                                trial[non_contrib] += d * w / wsum
                    
                    trial = np.maximum(trial, 0)
                    tsum = trial.sum()
                    if tsum < 1e-15:
                        continue
                    trial = trial / tsum
                    obj = 2.0 * n * fast_convolve(trial).max()
                    if obj < best_obj - 1e-13:
                        best_obj = obj
                        best_f = trial.copy()
                        f = trial
                        improved = True
                        break  # Move to next peak
        
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
    
    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-peak subgradient descent,
LP-Frank-Wolfe refinement, coordinate descent, and structural exploration.
Key innovations: multi-peak weighted subgradients with adaptive step size,
cosine/periodic structured initializations, load-balancing redistribution,

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def make_gaussian(n, sigma_frac=0.25):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    s = a.sum()
    return a / s


def make_cosine(n, num_peaks=1):
    x = np.arange(n)
    a = np.abs(np.cos(np.pi * num_peaks * x / n)) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_gapped(n, gap):
    a = np.zeros(n)
    a[::gap] = 1.0
    s = a.sum()
    if s < 1e-15:

    return a / s


def make_bell(n, width=0.7):

    half_w = n * width / 2.0
    t = np.clip(np.abs(x) / half_w, 0, 1.0)

    return a / s


def make_asymmetric(n, decay=3.0):

    a = np.exp(-decay * x / n)

def make_two_gaussian(n, frac=0.5, width=0.15):

    c1 = n * frac
    c2 = n * (1 - frac)
    w = n * width
    a = np.exp(-0.5 * ((x - c1) / w) ** 2) + np.exp(-0.5 * ((x - c2) / w) ** 2)
    s = a.sum()
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def compute_subgradient(f, peak_idx, n):
    """Compute the subgradient of max_k (p*p)(k) at the given peak."""
    s = np.zeros(n)
    for k in peak_idx:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            ki = k - j
            s[j] += 2.0 * f[ki]
            if ki != j:
                s[ki] += 2.0 * f[j]
    return s


def subgradient_step(f, peaks, weights, lr, n):
    """Apply weighted multi-peak subgradient step."""
    s = np.zeros(n)
    for i in range(len(peaks)):
        k = int(peaks[i])
        w = weights[i]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            ki = k - j
            s[j] += w * 2.0 * f[ki]
            if ki != j:
                s[ki] += w * 2.0 * f[j]
    s_centered = s - s.mean()
    f_new = f - lr * s_centered
    f_new = np.maximum(f_new, 0.0)
    sn = f_new.sum()
    if sn < 1e-15:
        return f.copy()
    return f_new / sn

def build_lp_matrix(f, n):
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        A_ub[k_start:k_end + 1, j] = f[:length]
    return A_ub


def solve_lp_direction(f, M, n):
    A_ub = build_lp_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

            no_improve = 0
        else:
            no_improve += 1

        if verbose and it % 20 == 0:
            print(f"    LP-FW iter {it}: obj={best_obj:.10f}, m_exp={m_exp:.2f}")

        if no_improve > 80:
            break

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Peak-targeted coordinate descent with smart redistribution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 40:
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        top_k = min(10, len(conv))
        peak_indices = np.argsort(conv)[::-1][:top_k]

        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[k - idx]

            order = np.argsort(products)[::-1]
            for oi in range(min(4, len(order))):
                j_max = int(idx[order[oi]])
                old_val = f[j_max]
                if old_val < 1e-12:
                    continue

                for mult in [0.4, 0.5, 0.6, 0.7, 0.8, 1.2, 1.4, 1.6]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.5:
                        continue

                    trial = f.copy()
                    trial[j_max] = new_val
                    d = old_val - new_val
                    if abs(d) > 1e-15:
                        non_contrib = np.ones(n, dtype=bool)
                        non_contrib[js:je + 1] = False
                        if non_contrib.sum() > 0:
                            w = f[non_contrib]
                            wsum = w.sum()
                            if wsum > 1e-15:
                                trial[non_contrib] += d * w / wsum

                    trial = np.maximum(trial, 0)
                    tsum = trial.sum()
                    if tsum < 1e-15:
                        continue
                    trial = trial / tsum
                    obj = 2.0 * n * fast_convolve(trial).max()
                    if obj < best_obj - 1e-12:
                        best_obj = obj
                        best_f = trial.copy()
                        f = trial
                        improved = True
                        break

        if not improved:
            no_improve += 1
            if no_improve > 10:
                # Aggressive: scale down all peak contributors
                k = int(np.argmax(fast_convolve(f)))
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                trial = f.copy()
                trial[js:je + 1] *= 0.9
                d = f.sum() - trial.sum()
                non_contrib = np.ones(n, dtype=bool)
                non_contrib[js:je + 1] = False
                if non_contrib.sum() > 0:
                    nc_w = f[non_contrib]
                    nw = nc_w.sum()
                    if nw > 1e-15:
                        trial[non_contrib] += d * nc_w / nw
                trial = np.maximum(trial, 0)
                trial = trial / trial.sum()
                obj = 2.0 * n * fast_convolve(trial).max()
                if obj < best_obj:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    no_improve = 0
                else:
                    break
        else:
            no_improve = 0

        if verbose and no_improve == 0:
            print(f"    CD: obj={best_obj:.10f}")

    return best_f, best_obj


def sparsify_search(f0, time_limit, n):
    """Try various sparsity patterns to find better structure."""
    best_f = f0.copy()
    best_obj = 2.0 * n * fast_convolve(f0).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)

    while time.monotonic() - t0 < time_limit:
        density = rng.uniform(0.3, 0.8)
        mask = (rng.random(n) < density).astype(float)
        if mask.sum() < max(4, n // 8):
            continue
        trial = f0 * mask
        ts = trial.sum()
        if ts < 1e-10:
            continue
        trial = trial / ts
        obj = 2.0 * n * fast_convolve(trial).max()
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_f = trial.copy()
            print(f"    Sparsify: obj={best_obj:.10f}, density={density:.2f}")

    return best_f, best_obj


def deflation_step(f, n, alpha=0.05):
    """Deflate the peak by scaling down contributing elements."""
    conv = fast_convolve(f)
    k_peak = int(np.argmax(conv))
    js = max(0, k_peak - n + 1)
    je = min(n - 1, k_peak)
    f_new = f.copy()
    f_new[js:je + 1] *= (1.0 - alpha)
    d = f.sum() - f_new.sum()
    non_contrib = np.ones(n, dtype=bool)
    non_contrib[js:je + 1] = False
    if non_contrib.sum() > 0:
        w = f[non_contrib]
        wsum = w.sum()
        if wsum > 1e-15:
            f_new[non_contrib] += d * w / wsum
    f_new = np.maximum(f_new, 0)
    f_new = f_new / f_new.sum()
    return f_new
