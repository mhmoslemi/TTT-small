"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated subgradient descent
with Nesterov momentum, adaptive temperature, LP-based escape directions,
multi-length exploration, and multi-stage polishing (L-BFGS soft-max + coordinate descent).

Key improvements:
- FFT-based cross-correlation for subgradient (O(n log n) vs O(n^2))
- Much faster per-iteration enables 10-50x more iterations in same time
- LP-guided escape directions to break out of local minima
- Systematic multi-length exploration
- Aggressive final polishing with coordinate descent
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr_fft(w, a):
    """
    Compute g[j] = sum_k w[k] * a[k-j] for valid j, via FFT.
    w has length 2n-1, a has length n, result has length n.
    """
    n = len(a)
    a_flip = a[::-1]
    size = len(w) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    B = rfft(a_flip, fft_size)
    conv_wb = irfft(W * B, fft_size)
    return conv_wb[n - 1:n - 1 + n]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_obj(a):
    """Compute max(conv(a,a)) assuming sum(a)=1."""
    b = fast_convolve(a)
    return float(b.max())


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def soft_max_obj_grad(z, T, n):
    """Soft-max smoothed objective and gradient in log-space."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    grad_a = 2.0 * n * cross_corr_fft(w, a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z


def lbgfs_refine(a0, T, time_limit, maxiter=5000):
    """L-BFGS in log-space with soft-max objective."""
    t0 = time.monotonic()
    n = len(a0)
    z0 = z_from_a(a0)
    best_obj = soft_max_obj_grad(z0, T, n)[0]
    best_z = z0.copy()

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        obj, _ = soft_max_obj_grad(z, T, n)
        return obj

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = soft_max_obj_grad(z, T, n)
        return g

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        final_obj = func(res.x)
        if np.isfinite(final_obj) and final_obj < best_obj:
            best_z = res.x
            best_obj = final_obj
    except Exception:
        pass

    return softmax_from_z(best_z), best_obj


def coordinate_polish(a0, time_limit):
    """Coordinate descent: try scaling each element by various multipliers."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = eval_obj(a)

    multipliers = np.array([0.3, 0.5, 0.7, 0.85, 0.9, 1.0, 1.1, 1.2, 1.4, 1.7, 2.5])

    improved_global = True
    while improved_global and time.monotonic() - t0 < time_limit:
        improved_global = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                val = eval_obj(a_trial)
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_global = True
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_struct_init(n, kind="gauss", sigma=0.3):
    """Create structured initialization."""
    x = np.linspace(0, 1, n)
    if kind == "gauss":
        a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    elif kind == "cosine":
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif kind == "tukey":
        a = np.ones(n)
        taper = int(sigma * n // 2)
        if taper > 0:
            t = np.linspace(0, 1, taper)
            a[:taper] = 0.5 * (1 + np.cos(np.pi * (2 * t / sigma - 1)))
            a[-taper:] = a[:taper][::-1]
    elif kind == "uniform":
        a = np.ones(n)
    elif kind == "edge":
        a = np.exp(-4 * ((x - 0.5) ** 2)) + 0.3
        a = x * (1 - x) + 0.1
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def lp_escape_direction(f, M, n_constraints=None):
    """
    Solve LP: maximize sum(g) subject to g >= 0, ||f*g||_inf <= M.
    Returns normalized g or None if infeasible.
    Use subset of constraints for speed.
    """
    n = len(f)
    if n > 2000:
        # Too large for dense LP, use sampling
        num_lags = min(2 * n - 1, n_constraints or 500)
        lag_indices = np.random.default_rng(42).choice(2 * n - 1, size=num_lags, replace=False)
    else:
        lag_indices = np.arange(2 * n - 1)

    A_ub = np.zeros((len(lag_indices), n))
    for idx, k in enumerate(lag_indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A_ub[idx, j] = f[k - j]

    b_ub = np.full(len(lag_indices), M)
    c = -np.ones(n)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
        if res.success:
            g = np.maximum(res.x, 0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

    print(f"\n=== Phase 7: Final L-BFGS (best={best_state[1]:.6f}) ===")
    if best_state[0] is not None:
        for T in [100, 300]:
            rem = deadline - time.monotonic() - 80.0
            if rem < 10:
                break
            t_alloc = min(rem * 0.4, 40.0)
            if t_alloc < 5:
                continue
            print(f"  Final L-BFGS T={T}")
            a_fin, _ = lbgfs_refine(best_state[0], T=T, time_limit=t_alloc, maxiter=5000)
            try_seq(a_fin, f"final_T{T}")

    # === Phase 8: Coordinate descent polish ===
    print(f"\n=== Phase 8: CD polish (best={best_state[1]:.6f}) ===")
    if best_state[0] is not None:
        rem = deadline - time.monotonic() - 30.0
        if rem > 10:
            t_cd = min(rem, 60.0)
            print(f"  CD polish, t={t_cd:.0f}s")
            a_cd, _ = coordinate_polish(best_state[0], time_limit=t_cd)
            try_seq(a_cd, "cd_polish")

    # === Phase 9: Micro-optimization ===
    print(f"\n=== Phase 9: Micro (best={best_state[1]:.6f}) ===")
    if best_state[0] is not None:
        n_best = len(best_state[0])
        rem = deadline - time.monotonic() - 5.0
        if rem > 8:
            t_alloc = min(rem * 0.5, 20.0)
            a_micro, _ = subgradient_optimize_fast(
                best_state[0], time_limit=t_alloc,
                lr_init=0.0003, momentum=0.97,
                temp_init=0.00005, temp_growth=1.001,
                kick_interval=5000, kick_size=0.0005,
                lp_interval=50000, seed=999999
            )
            try_seq(a_micro, "micro")

            # Small perturbations
            rem = deadline - time.monotonic() - 3.0
            if rem > 5:
                for attempt in range(min(5, int(rem / 3))):
                    rem = deadline - time.monotonic() - 1
                    if rem < 2:
                        break
                    noise = rng.standard_normal(n_best) * 0.002
                    a_pert = project_simplex(best_state[0] + noise)
                    a_p, _ = subgradient_optimize_fast(
                        a_pert, time_limit=min(rem * 0.8, 5.0),
                        lr_init=0.001, momentum=0.93,
                        temp_init=0.0005, temp_growth=1.01,
                        kick_interval=500, kick_size=0.002,
                        lp_interval=5000, seed=attempt * 31337 + 1
                    )
                    try_seq(a_p, f"perturb_{attempt}")

    # Final verification
    if best_state[0] is None:
        best_state = (parent.copy(), float(evaluate_sequence(parent.tolist())))

    fs = float(evaluate_sequence(best_state[0].tolist()))
    if np.isfinite(fs):
        best_state = (best_state[0], fs)
    else:
        best_state = (parent.copy(), float(evaluate_sequence(parent.tolist())))

    print(f"\n{'=' * 50}")
    print(f"FINAL SCORE: {best_state[1]:.6f} (n={len(best_state[0])})")
    print(f"{'=' * 50}")

    return [float(x) for x in best_state[0]]
