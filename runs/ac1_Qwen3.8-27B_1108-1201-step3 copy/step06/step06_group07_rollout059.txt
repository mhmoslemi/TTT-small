LP-based optimization minimizing 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Uses cross-correlation LP for global directions, combined with multiplicative mirror descent
polishing, diverse multi-start, and adaptive restarts. Key innovation: solving an LP at each
step to find the maximal feasible direction (b) such that cross-correlation (a*b)_k <= M,
which provides a globally optimal step direction unlike local gradient methods.

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix_indices(n):
    """Precompute row/col indices for the convolution matrix (structure is fixed)."""
    rows = []
    cols = []
    for i in range(n):
        k_start = i
        k_end = min(i + n - 1, 2 * n - 2)
        n_ks = k_end - k_start + 1
        rows.extend(range(k_start, k_end + 1))
        cols.extend([i] * n_ks)
    return np.array(rows), np.array(cols)


def solve_lp_direction(a, M, row_idx, col_idx):

    n = len(a)
    vals = np.empty(len(row_idx))
    for i in range(n):
        k_start = i
        k_end = min(i + n - 1, 2 * n - 2)
        n_ks = k_end - k_start + 1
        # Find position in arrays for column i
        start_pos = col_idx[col_idx == i][0] if np.any(col_idx == i) else 0
        # Actually let's just build vals properly
        pass
    
    # Rebuild vals (faster with the known structure)
    vals = np.empty(2 * n * n - n, dtype=np.float64)
    pos = 0
    for i in range(n):
        k_start = i
        k_end = min(i + n - 1, 2 * n - 2)
        n_ks = k_end - k_start + 1
        vals[pos:pos + n_ks] = a[:n_ks]
        pos += n_ks
    
    A_mat = sparse.csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)

        if res.success and res.x is not None:

    except Exception:
        pass
    return None

def coordinate_descent_polish(a0, time_limit, verbose=False, label=""):
    """Fine coordinate descent: adjust pairs of coordinates for peak reduction."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_convolve(a)
        
        # Find the peak
        k_peak = int(np.argmax(conv))
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        
        if je < js:
            break
        
        idx = np.arange(js, je + 1)
        products = a[idx] * a[k_peak - idx]
        
        # Try reducing the top contributors
        top_contrib = idx[np.argsort(products)[-min(5, len(idx)):]]
        
        for j in top_contrib:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            best_j_dest = -1
            
            # Try reducing a[j] and adding to a non-contributing position
            non_contrib = np.ones(n, dtype=bool)
            non_contrib[js:je + 1] = False
            
            if non_contrib.sum() == 0:
                continue
            
            # Find best destination (position that minimizes new peak)
            for mult in [0.7, 0.8, 0.9]:
                new_val = old_val * mult
                diff = old_val - new_val
                if diff < 1e-15:
                    continue
                # Distribute to non-contrib positions proportional to current values
                nc_mask = non_contrib
                nc_vals = a[nc_mask]
                nc_sum = nc_vals.sum()
                if nc_sum < 1e-15:
                    continue
                trial = a.copy()
                trial[j] = new_val
                trial[nc_mask] += diff * nc_vals / nc_sum
                trial = np.maximum(trial, 0.0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial /= ss
                obj = fast_convolve(trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = new_val
                    best_j_dest = j
                    best_trial = trial.copy()
            
            if best_j_dest >= 0:
                a = best_trial
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            # Try broad uniform reduction of peak contributors
            for scale in [0.95, 0.9, 0.85]:
                trial = a.copy()
                trial[idx] *= scale
                non_contrib = np.ones(n, dtype=bool)
                non_contrib[idx] = False
                if non_contrib.sum() > 0:
                    nc = a[non_contrib]
                    nc_sum = nc.sum()
                    if nc_sum > 1e-15:
                        trial[non_contrib] += (1.0 - trial.sum()) * nc / nc_sum
                        trial = np.maximum(trial, 0.0)
                        ss = trial.sum()
                        if ss > 1e-15:
                            trial /= ss
                            obj = fast_convolve(trial).max()
                            if obj < best_obj - 1e-14:
                                a = trial
                                best_obj = obj
                                best_a = a.copy()
                                improved = True
                                break
            if not improved:
                break
    
    return best_a, best_obj


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.7):
    x = np.arange(n) - n / 2.0
    half_w = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_w, 0, 1.0)
    a = 1.0 - t ** 6
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_symmetric_gaussian(n, sigma_frac=0.35):
    """Gaussian centered exactly in the middle for even n."""
    center = (n - 1) / 2.0
    x = np.arange(n) - center
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = (a + a[::-1]) / 2.0
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
