Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive smoothed mirror descent
with momentum, multi-length exploration, and LP-Frank-Wolfe refinement.
Key improvements over previous: adaptive temperature based on progress,
momentum in multiplicative updates, broader n exploration, and 
peak-specific gradient targeting.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_gradient(conv, f, tau, top_k_frac=0.1):
    """Compute gradient of LSE approximation of max(conv)."""
    n = len(f)
    len_c = len(conv)
    k_top = max(10, int(len_c * top_k_frac))
    peaks = np.argpartition(conv, -k_top)[-k_top:]
    pv = conv[peaks]
    scaled = tau * pv
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        idx = np.arange(js, je + 1)
        contrib = f[k - idx] + f[idx]
        # Handle the diagonal: if k = 2*idx[i], we double-count
        # Actually, f[k-idx] + f[idx] correctly handles this:
        # when k-idx[i] == idx[i] (i.e., idx[i] = k/2), we get 2*f[k/2]
        # which is correct because d(conv[k])/da[k/2] = 2*a[k/2] when k is even
        grad[js:je + 1] += w[i] * contrib
    return grad


def smd_optimize(f0, time_limit, tau_init=10.0, tau_max=2000.0,
                 lr_init=0.01, momentum=0.5, verbose=False):
    """
    Adaptive Smoothed Mirror Descent with momentum.
    """
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    tau = tau_init
    lr = lr_init
    no_improve = 0
    iter_count = 0
    momentum_buf = np.zeros(n)
    best_tau = tau

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Adaptive top-k fraction: use more peaks when tau is low
        top_k_frac = min(0.3, max(0.05, 50.0 / tau))
        grad = smooth_max_gradient(conv, f, tau, top_k_frac=top_k_frac)

        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            grad = grad / g_norm

        # Momentum
        momentum_buf = momentum * momentum_buf + (1 - momentum) * grad

        # Multiplicative (mirror) descent step
        f_new = f * np.exp(-lr * momentum_buf)
        f_new = np.maximum(f_new, 1e-15)
        s = f_new.sum()
        if s < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / s

        # Backtracking line search
        obj_new = fast_convolve(f_new).max()
        if obj_new > M * 1.02:
            for damp in [0.5, 0.3, 0.2, 0.1, 0.05]:
                f_trial = f * (1.0 - damp) + f_new * damp
                f_trial = np.maximum(f_trial, 1e-15)
                f_trial = f_trial / f_trial.sum()
                obj_trial = fast_convolve(f_trial).max()
                if obj_trial < M:
                    f_new = f_trial
                    obj_new = obj_trial
                    break
            else:
                lr = max(lr * 0.7, 1e-6)
                continue

        f = f_new

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
            # Cool down: increase tau (sharper gradient)
            best_tau = tau
            if no_improve == 0 and iter_count > 10:
                tau = min(tau * 1.01, tau_max)
        else:
            no_improve += 1
            # Heat up: decrease tau (smoother gradient) to escape
            if no_improve > 20:
                tau = max(tau * 0.95, tau_init)
            if no_improve > 50:
                lr = min(lr * 1.05, lr_init * 3.0)

        # Adaptive learning rate
        if no_improve == 0:
            lr = min(lr * 1.002, lr_init * 2.0)
        else:
            lr = max(lr * 0.99, 1e-6)

        # Periodic restart
        if no_improve > 200:
            rng = np.random.default_rng(iter_count * 7 + 13)
            noise_scale = 0.002 + 0.003 * (no_improve / 200.0)
            f = np.maximum(best_f + rng.standard_normal(n) * noise_scale, 1e-15)
            f = f / f.sum()
            lr = lr_init * 0.5
            tau = best_tau
            momentum_buf = np.zeros(n)
            no_improve = 0

        if lr < 1e-7:
            break

        if verbose and iter_count % 200 == 0:
            print(f"      SMD iter {iter_count}: obj={best_obj:.10f}, tau={tau:.1f}, lr={lr:.6f}, no_improve={no_improve}")

    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with multiple M values."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Try multiple M values
        found_direction = False
        for m_factor in [1.0, 1.01, 1.02, 1.05, 1.1]:
            M_trial = M * m_factor
            g = _solve_lp_direction(f, M_trial, n)
            if g is not None and g.sum() > 1e-10:
                g = g / g.sum()
                found_direction = True
                break

        if not found_direction:
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(it * 17 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
            if no_improve > 100:
                break
            continue

        no_improve = 0

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            # Binary search refinement
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 10 == 0:
                    print(f"      LP-FW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        if no_improve > 50:
            break

    return best_f, best_obj


def _solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def peak_pair_reduction(f0, time_limit, verbose=False):
    """
    Targeted reduction of the top peaks by identifying and reducing
    the specific pairs that contribute most to each peak.
    """
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.005
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Find top 5 peaks
        top_k = min(5, len(conv))
        peaks = np.argsort(conv)[-top_k:]
        
        # For each peak, identify the top contributing pairs
        paired_reduction = np.zeros(n)
        for k in peaks:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = f[k - idx] * f[idx]
            
            # Find the pairs with highest contribution
            if len(contrib) > 0:
                # Weight by how much this peak contributes to the max
                peak_weight = conv[k] / M
                # Reduce the indices that have highest product
                threshold = np.median(contrib) if len(contrib) > 1 else contrib[0] * 0.5
                mask = contrib > threshold
                paired_reduction[js:je+1] += peak_weight * (f[k - idx][mask] * 0.5 + f[idx][mask] * 0.5)

        # Also add a small uniform component to maintain feasibility
        paired_reduction += 0.01 * f
        
        # Multiplicative step
        f_new = f * np.exp(-lr * paired_reduction)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        obj_new = fast_convolve(f_new).max()
        if obj_new < M - 1e-16:
            f = f_new
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.01, 0.02)
            else:
                no_improve = 1
        else:
            lr *= 0.8
            no_improve += 1

        if no_improve > 50:
            rng = np.random.default_rng(it * 31 + 7)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.002, 1e-15)
            f = f / f.sum()
            lr = 0.005
            no_improve = 0

        if lr < 1e-7:
            break

        if verbose and it % 100 == 0:
            print(f"      PPR iter {it}: obj={best_obj:.10f}, lr={lr:.6f}")

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_init(n, variant, seed=42):
    """Generate asymmetric initial sequences."""
    rng = np.random.default_rng(seed + variant * 1000)
    x = np.arange(n) / n

    if variant == 0:
        a = np.exp(-x / 0.1)
    elif variant == 1:
        a = np.exp(-(1-x) / 0.1)
    elif variant == 2:
        a = np.exp(-x / 0.2) * (1 + 0.3 * np.sin(4 * np.pi * x))
    elif variant == 3:
        a = (1 - x) ** 2 + 0.01
    elif variant == 4:
        a = np.exp(-40 * (x - 0.2) ** 2) + 0.3 * np.exp(-40 * (x - 0.6) ** 2)
    elif variant == 5:
        a = x ** 0.5 + 0.01
    elif variant == 6:
        a = np.exp(-x / 0.15)
    elif variant == 7:
        a = np.exp(-x / 0.08)
    elif variant == 8:
        a = (1 - x) ** 3 + 0.01
    elif variant == 9:
        a = np.exp(-x / 0.25)
    elif variant == 10:
        # Two-scale: rapid decay then slow
        a = np.exp(-x / 0.05) * 0.5 + np.exp(-x / 0.3) * 0.5
    elif variant == 11:
        # Power law
        a = (x + 0.01) ** (-0.3)
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with 1D optimization per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    sweep = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False
        order = np.arange(n)
        rng = np.random.default_rng(sweep * 1000 + 42)
        rng.shuffle(order)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj

            old_val = f[j]
            if old_val < 1e-14:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            # Try several perturbations
            for frac in [-0.5, -0.3, -0.2, -0.1, 0.05, 0.1, 0.2, 0.3, 0.5]:
                new_val = old_val * (1 + frac)
                if new_val < 0 or new_val > 0.8:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-16:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    s2 = f.sum()
                    if s2 > 1e-15:
                        f = f / s2
                if best_j_obj < best_obj - 1e-16:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break
        if verbose and sweep % 5 == 0:
            print(f"      CD sweep {sweep}: obj={best_obj:.10f}")

    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive smoothed mirror descent
with momentum, alternating LP-Frank-Wolfe refinement, peak-pair targeting,
and multi-length exploration. Key improvements: adaptive temperature cooling
based on progress, momentum in multiplicative updates, alternating SMD/LP-FW
cycles, and broader length exploration around the parent's length.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_grad(conv, f, tau, top_k_frac=0.1):
    """Compute gradient of top-k LSE approximation of max(conv)."""
    n = len(f)
    len_c = len(conv)
    k_top = max(10, int(len_c * top_k_frac))
    peaks = np.argpartition(conv, -k_top)[-k_top:]
    pv = conv[peaks]
    scaled = tau * pv
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        idx = np.arange(js, je + 1)
        contrib = f[k - idx] + f[idx]
        grad[js:je + 1] += w[i] * contrib
    return grad


def smd_optimize(f0, time_limit, tau_init=20.0, tau_max=3000.0,
                 lr_init=0.006, momentum=0.3, top_k_init=0.15,
                 verbose=False, seed=42):
    """
    Adaptive Smoothed Mirror Descent with momentum and adaptive temperature.
    """
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    tau = tau_init
    lr = lr_init
    no_improve = 0
    iter_count = 0
    momentum_buf = np.zeros(n)
    best_tau = tau
    best_lr = lr
    top_k_frac = top_k_init
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        conv = fast_convolve(f)
        M = conv.max()

        grad = smooth_max_grad(conv, f, tau, top_k_frac=top_k_frac)

        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            grad = grad / g_norm

        momentum_buf = momentum * momentum_buf + (1.0 - momentum) * grad

        f_new = f * np.exp(-lr * momentum_buf)
        f_new = np.maximum(f_new, 1e-15)
        s = f_new.sum()
        if s < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / s

        obj_new = fast_convolve(f_new).max()
        if obj_new > M * 1.01:
            accepted = False
            for damp in [0.5, 0.3, 0.2, 0.1, 0.05]:
                f_trial = f * (1.0 - damp) + f_new * damp
                f_trial = np.maximum(f_trial, 1e-15)
                f_trial = f_trial / f_trial.sum()
                obj_trial = fast_convolve(f_trial).max()
                if obj_trial < M:
                    f_new = f_trial
                    obj_new = obj_trial
                    accepted = True
                    break
            if not accepted:
                lr = max(lr * 0.7, 1e-7)
                continue

        f = f_new

        if obj_new < best_obj - 1e-16:
            best_obj = obj_new
            best_f = f.copy()
            if no_improve > 0:
                best_tau = tau
                best_lr = lr
            no_improve = 0
            # Gradually increase temperature (sharper) and decrease top_k_frac
            tau = min(tau * 1.008, tau_max)
            top_k_frac = max(0.03, top_k_frac * 0.998)
        else:
            no_improve += 1
            if no_improve > 30:
                tau = max(tau * 0.97, tau_init)
                top_k_frac = min(0.3, top_k_frac * 1.02)
            if no_improve > 100:
                lr = min(lr * 1.03, lr_init * 3.0)

        if no_improve == 0:
            lr = min(lr * 1.001, lr_init * 2.5)
        else:
            lr = max(lr * 0.995, 1e-7)

        if no_improve > 300:
            noise_scale = 0.002 + 0.004 * min(1.0, (no_improve - 300) / 300.0)
            f = np.maximum(best_f + rng.standard_normal(n) * noise_scale, 1e-15)
            f = f / f.sum()
            lr = best_lr * 0.5
            tau = best_tau
            top_k_frac = top_k_init
            momentum_buf = np.zeros(n)
            no_improve = 0

        if lr < 1e-8:
            break

        if verbose and iter_count % 300 == 0:
            print(f"      SMD it={iter_count} obj={best_obj:.10f} tau={tau:.0f} "
                  f"lr={lr:.6f} topk={top_k_frac:.3f} ni={no_improve}")

    return best_f, best_obj


def solve_lp_dir(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_fw_optimize(f0, time_limit, verbose=False, seed=42):
    """LP-based Frank-Wolfe with adaptive M relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        found = False
        for mf in [1.0, 1.005, 1.01, 1.02, 1.05]:
            g = solve_lp_dir(f, M * mf, n)
            if g is not None and g.sum() > 1e-10:
                g = g / g.sum()
                found = True
                break

        if not found:
            no_improve += 1
            if no_improve > 20:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
            if no_improve > 80:
                break
            continue

        no_improve = 0
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 40:
                break
            continue

        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new

            if M < best_obj - 1e-16:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 5 == 0:
                    print(f"      LP-FW it={it} obj={best_obj:.10f}")
        else:
            no_improve += 1

        if no_improve > 40:
            break

    return best_f, best_obj


def peak_pair_descent(f0, time_limit, verbose=False, seed=42):
    """Target the top peaks by reducing their highest-contributing pairs."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.004
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        top_k = min(5, len(conv))
        peaks = np.argsort(conv)[-top_k:]

        reduction = np.zeros(n)
        for k in peaks:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            products = f[k - idx] * f[idx]
            peak_weight = conv[k] / M
            if len(products) > 2:
                thresh = np.percentile(products, 60)
                mask = products > thresh
                reduction[js:je + 1] += peak_weight * 0.5 * (
                    f[k - idx] * mask + f[idx] * mask)
            else:
                reduction[js:je + 1] += peak_weight * 0.5 * (f[k - idx] + f[idx])

        reduction += 0.005 * f

        f_new = f * np.exp(-lr * reduction)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        obj_new = fast_convolve(f_new).max()
        if obj_new < M - 1e-18:
            f = f_new
            if obj_new < best_obj - 1e-16:
                best_obj = obj_new
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.005, 0.015)
            else:
                no_improve = 1
        else:
            lr = max(lr * 0.85, 1e-7)
            no_improve += 1

        if no_improve > 60:
            f = np.maximum(best_f + rng.standard_normal(n) * 0.002, 1e-15)
            f = f / f.sum()
            lr = 0.004
            no_improve = 0

        if lr < 1e-8:
            break

        if verbose and it % 100 == 0:
            print(f"      PPD it={it} obj={best_obj:.10f} lr={lr:.6f}")

    return best_f, best_obj


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_init(n, variant, seed=42):
    """Generate asymmetric initial sequences."""
    x = np.arange(n) / n
    if variant == 0:
        a = np.exp(-x / 0.1)
    elif variant == 1:
        a = np.exp(-x / 0.15)
    elif variant == 2:
        a = np.exp(-x / 0.2) * (1.0 + 0.3 * np.sin(4.0 * np.pi * x))
        a = np.maximum(a, 0)
    elif variant == 3:
        a = (1.0 - x) ** 2 + 0.01
    elif variant == 4:
        a = np.exp(-40.0 * (x - 0.2) ** 2) + 0.3 * np.exp(-40.0 * (x - 0.6) ** 2)
    elif variant == 5:
        a = np.exp(-x / 0.08)
    elif variant == 6:
        a = np.exp(-x / 0.25)
    elif variant == 7:
        a = (x + 0.01) ** (-0.3)
    elif variant == 8:
        a = 0.5 * np.exp(-x / 0.05) + 0.5 * np.exp(-x / 0.3)
    elif variant == 9:
        a = np.exp(-x / 0.12)
    elif variant == 10:
        a = np.exp(-x / 0.18) * (1.0 + 0.2 * np.cos(6.0 * np.pi * x))
        a = np.maximum(a, 0)
    else:
        a = np.exp(-x / 0.1)

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def coord_descent(f0, time_limit, verbose=False):
    """Coordinate descent with 1D optimization per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    sweep = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False
        order = np.arange(n)
        rng = np.random.default_rng(sweep * 1000 + 42)
        rng.shuffle(order)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj

            old_val = f[j]
            if old_val < 1e-14:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for frac in [-0.4, -0.2, -0.1, 0.05, 0.1, 0.2, 0.4]:
                new_val = old_val * (1.0 + frac)
                if new_val < 0 or new_val > 0.7:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-16:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    s2 = f.sum()
                    if s2 > 1e-15:
                        f = f / s2
                if best_j_obj < best_obj - 1e-16:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break
        if verbose and sweep % 3 == 0:
            print(f"      CD sweep={sweep} obj={best_obj:.10f}")

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: adaptive SMD with momentum, alternating
    LP-Frank-Wolfe, peak-pair descent, multi-length exploration, and
    coordinate descent polishing.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 60.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # === Phase 1: Quick SMD on parent to get into basin ===
    print("\n=== Phase 1: Quick SMD on parent ===")
    if parent is not None:
        n_p = len(parent)
        t_alloc = 90
        print(f"  SMD parent n={n_p} t={t_alloc}s")
        a_opt, _ = smd_optimize(parent, time_limit=t_alloc, tau_init=15.0,
                                tau_max=500.0, lr_init=0.01, momentum=0.2,
                                top_k_init=0.2, verbose=True, seed=seed)
        try_update(a_opt, "smd_parent_quick")

    # === Phase 2: Alternating SMD / LP-FW cycles ===
    print("\n=== Phase 2: Alternating SMD / LP-FW ===")
    if best_seq is not None:
        cur = best_seq.copy()
        for cycle in range(3):
            rem = deadline - time.monotonic() - 300
            if rem < 60:
                break
            # SMD phase
            t_smd = min(rem * 0.45, 150)
            if t_smd > 20:
                tau_s = 30.0 + cycle * 40.0
                lr_s = 0.005 / (1.0 + 0.3 * cycle)
                print(f"  Cycle {cycle} SMD: tau={tau_s:.0f} lr={lr_s:.4f} t={t_smd:.0f}s")
                cur, _ = smd_optimize(cur, time_limit=t_smd, tau_init=tau_s,
                                      tau_max=tau_s * 20.0, lr_init=lr_s,
                                      momentum=0.3 + 0.1 * cycle,
                                      top_k_init=0.15 - 0.03 * cycle,
                                      verbose=False, seed=seed + cycle)
                try_update(cur, f"alt_smd_{cycle}")

            # LP-FW phase
            rem = deadline - time.monotonic() - 250
            if rem < 30:
                break
            t_lpfw = min(rem * 0.35, 80)
            if t_lpfw > 10:
                print(f"  Cycle {cycle} LP-FW: t={t_lpfw:.0f}s")
                cur, _ = lp_fw_optimize(cur, time_limit=t_lpfw, verbose=False,
                                        seed=seed + 100 + cycle)
                try_update(cur, f"alt_lpfw_{cycle}")

    # === Phase 3: Multi-length exploration ===
    print("\n=== Phase 3: Multi-Length Exploration ===")
    if best_seq is not None:
        n_best = len(best_seq)
        test_ns = set()
        for delta in [-64, -32, -16, 16, 32, 64, 128]:
            n_try = n_best + delta
            if 200 <= n_try <= 2000:
                test_ns.add(n_try)
        # Also try some absolute values
        for alt_n in [640, 800, 960, 1152]:
            if abs(alt_n - n_best) > 16:
                test_ns.add(alt_n)

        for n_try in sorted(test_ns):
            rem = deadline - time.monotonic() - 200
            if rem < 30:
                break
            f_interp = interpolate_seq(best_seq, n_try)
            sc = try_update(f_interp, f"interp_n{n_try}")
            if sc < best_score + 0.01:
                t_alloc = min(rem * 0.25, 50)
                if t_alloc > 10:
                    print(f"  SMD n={n_try}: t={t_alloc:.0f}s")
                    a_opt, _ = smd_optimize(f_interp, time_limit=t_alloc,
                                            tau_init=50.0, tau_max=2000.0,
                                            lr_init=0.004, momentum=0.3,
                                            top_k_init=0.12, seed=seed + n_try)
                    try_update(a_opt, f"smd_n{n_try}")

    # === Phase 4: Fresh asymmetric init + SMD ===
    print("\n=== Phase 4: Fresh Initializations ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for v in [0, 1, 5, 6, 8, 9]:
            rem = deadline - time.monotonic() - 150
            if rem < 20:
                break
            a_init = make_init(n_best, variant=v, seed=seed + v)
            conv = fast_convolve(a_init)
            obj_init = conv.max()
            # Only optimize if init is competitive
            if obj_init < fast_convolve(best_seq).max() * 1.1:
                t_alloc = min(rem * 0.15, 40)
                if t_alloc > 10:
                    a_opt, _ = smd_optimize(a_init, time_limit=t_alloc,
                                            tau_init=20.0, tau_max=1000.0,
                                            lr_init=0.008, momentum=0.3,
                                            top_k_init=0.15, seed=seed + v * 7)
                    try_update(a_opt, f"fresh_v{v}")

    # === Phase 5: Peak-pair descent ===
    print("\n=== Phase 5: Peak-Pair Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 120
        if rem > 30:
            t_alloc = min(rem * 0.4, 80)
            print(f"  PPD: t={t_alloc:.0f}s")
            a_opt, _ = peak_pair_descent(best_seq, time_limit=t_alloc,
                                         verbose=True, seed=seed + 500)
            try_update(a_opt, "ppd")

    # === Phase 6: SMD with very high temperature (fine tuning) ===
    print("\n=== Phase 6: Fine SMD ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_alloc = min(rem * 0.5, 100)
            print(f"  Fine SMD: t={t_alloc:.0f}s")
            a_opt, _ = smd_optimize(best_seq, time_limit=t_alloc,
                                    tau_init=200.0, tau_max=10000.0,
                                    lr_init=0.002, momentum=0.5,
                                    top_k_init=0.05, verbose=True,
                                    seed=seed + 900)
            try_update(a_opt, "fine_smd")

    # === Phase 7: Coordinate descent ===
    print("\n=== Phase 7: Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_alloc = min(rem * 0.5, 100)
            print(f"  CD: t={t_alloc:.0f}s")
            a_opt, _ = coord_descent(best_seq, time_limit=t_alloc, verbose=True)
            try_update(a_opt, "cd")

    # === Phase 8: Final LP-FW ===
    print("\n=== Phase 8: Final LP-FW ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            t_alloc = min(rem * 0.5, 50)
            print(f"  Final LP-FW: t={t_alloc:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_alloc, seed=seed + 999)
            try_update(a_opt, "final_lpfw")

    # === Phase 9: Very fine SMD ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_alloc = min(rem * 0.6, 25)
            print(f"  Very Fine SMD: t={t_alloc:.0f}s")
            a_opt, _ = smd_optimize(best_seq, time_limit=t_alloc,
                                    tau_init=500.0, tau_max=20000.0,
                                    lr_init=0.001, momentum=0.5,
                                    top_k_init=0.03, seed=seed + 1111)
            try_update(a_opt, "vfine_smd")

    # Fallback
    if best_seq is None:
        best_seq = make_init(768, variant=0, seed=seed)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
