                         method='highs',
                         options={'time_limit': lp_time_limit, 'presolve': True})
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def lp_alternating(f0, time_limit, verbose=False, label=""):
    """
    LP-based alternating optimization.
    Solve LP to find descent direction, then interpolate toward it.
    """
    a = np.maximum(f0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break

        conv = fast_convolve(a)
        M = conv.max()

        lp_time = min(remaining * 0.3, 8.0)
        b_opt = solve_primal_lp(a, M, lp_time_limit=lp_time)

        if b_opt is None:
            # Fallback: mirror descent step
            if it % 5 == 0:
                a_md, obj_md = mirror_descent(a, time_limit=min(remaining * 0.1, 5.0),
                                              top_k=20, temp_init=0.0005, lr_init=0.01)
                if obj_md < best_obj:
                    best_obj = obj_md
                    best_a = a_md.copy()
                    a = a_md.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            continue

        b_sum = b_opt.sum()
        if b_sum < 1e-10:
            no_improve += 1
            continue
        b_scaled = b_opt / b_sum

        # Line search for interpolation step
        best_t = 0.0
        best_trial_obj = best_obj + 1e-15
        for t in [0.9, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
            trial = (1.0 - t) * a + t * b_scaled
            trial = np.maximum(trial, 0.0)
            ss = trial.sum()
            if ss < 1e-10:
                continue
            trial = trial / ss
            trial_obj = fast_convolve(trial).max()
            if trial_obj < best_trial_obj:
                best_trial_obj = trial_obj
                best_t = t

        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * b_scaled
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            new_obj = fast_convolve(a).max()
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_a = a.copy()
                no_improve = 0
                if verbose and it % 5 == 0:
                    print(f"  {label} LP iter {it}: obj={best_obj:.10f}, t={best_t:.3f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 15:
            rng = np.random.default_rng(it * 100003 + 7)
            scale = 0.002 + 0.003 * min(no_improve / 50.0, 1.0)
            a = np.maximum(best_a + rng.standard_normal(n) * scale, 1e-15)
            a = a / a.sum()
            no_improve = 0

        if it % 25 == 0:
            a_sym = 0.5 * (a + a[::-1])
            a_sym = np.maximum(a_sym, 0.0)
            s = a_sym.sum()
            if s > 1e-10:
                a_sym /= s
                sym_obj = fast_convolve(a_sym).max()
                if sym_obj < best_obj - 1e-14:
                    best_obj = sym_obj
                    best_a = a_sym.copy()
                    a = a_sym.copy()

    return best_a, best_obj


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.5):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bimodal(n, c1=0.3, c2=0.7, sigma=0.08):
    x = np.arange(n) / max(n, 1)
    a = np.exp(-0.5 * ((x - c1) / sigma) ** 2) + np.exp(-0.5 * ((x - c2) / sigma) ** 2)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def symmetrize(a):
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym /= ss
    return a_sym


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization: multi-scale LP alternating + mirror descent polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})", flush=True)
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}", flush=True)
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Multi-scale LP optimization ===
    print("=== Phase 1: Multi-scale LP ===", flush=True)

    target_n = 768
    if parent is not None:
        target_n = len(parent)

    # Determine scales to try
    scales = [32, 64, 128, 256, 512]
    if target_n > 512:
        scales.append(target_n)
    elif target_n > 256:
        if 512 not in scales:
            scales.append(512)
        if target_n not in scales:
            scales.append(target_n)

    # For each scale, allocate time proportional to complexity
    # Small scales are fast, so give them less time
    remaining = deadline - time.monotonic()
    # Allocate: 5% for n<=64, 10% for n=128, 20% for n=256, 30% for n=512, rest for target
    time_weights = {}
    total_w = 0
    for s in scales:
        w = float(s) ** 1.5
        time_weights[s] = w
        total_w += w
    # Normalize to use 70% of remaining time for multi-scale
    ms_budget = remaining * 0.70
    for s in scales:
        time_weights[s] = ms_budget * time_weights[s] / total_w

    current_a = None
    for scale_n in scales:
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        t_alloc = min(time_weights[scale_n], remaining - 5)
        if t_alloc < 2:
            continue

        print(f"  Scale n={scale_n}, time={t_alloc:.0f}s", flush=True)

        if current_a is None:
            # Initialize with Gaussian
            a_init = make_gaussian(scale_n, 0.3)
            if parent is not None and scale_n == scales[-1]:
                # Use parent as initialization for the final scale
                a_init = interpolate_sequence(parent, scale_n)
        else:
            a_init = interpolate_sequence(current_a, scale_n)

        # Run LP alternating
        a_opt, obj = lp_alternating(a_init, time_limit=t_alloc, verbose=True,
                                    label=f"LP_n{scale_n}")
        try_seq(a_opt, f"LP_n{scale_n}")

        # Quick MD polish
        remaining = deadline - time.monotonic()
        t_md = min(remaining * 0.05, 20)
        if t_md > 2:
            a_md, obj_md = mirror_descent(a_opt, time_limit=t_md, top_k=20,
                                          temp_init=0.0005, lr_init=0.01)
            sc = try_seq(a_md, f"MD_n{scale_n}")
            if np.isfinite(sc) and sc < best_score:
                a_opt = a_md

        # Symmetrize
        a_sym = symmetrize(a_opt)
        sc = try_seq(a_sym, f"SYM_n{scale_n}")

        current_a = a_opt if best_seq is None else best_seq
        if a_sym is not None:
            current_a = a_sym

    # === Phase 2: Full-scale LP at target_n ===
    print("\n=== Phase 2: Full-scale LP ===", flush=True)
    remaining = deadline - time.monotonic()
    if remaining > 30:
        if best_seq is not None:
            a_start = best_seq
        elif current_a is not None:
            a_start = interpolate_sequence(current_a, target_n)
        else:
            a_start = make_gaussian(target_n, 0.3)

        t_lp = remaining * 0.45
        print(f"  LP time: {t_lp:.0f}s, n={len(a_start)}", flush=True)
        a_opt, obj = lp_alternating(a_start, time_limit=t_lp, verbose=True,
                                    label="LP_full")
        try_seq(a_opt, "LP_full")

    # === Phase 3: Mirror descent polish ===
    print("\n=== Phase 3: MD Polish ===", flush=True)
    remaining = deadline - time.monotonic()
    if remaining > 20 and best_seq is not None:
        t_md = remaining * 0.25
        a_opt, obj = mirror_descent(best_seq, time_limit=t_md, top_k=30,
                                    temp_init=0.0002, lr_init=0.008, verbose=True,
                                    label="MD_polish")
        try_seq(a_opt, "MD_polish")

    # === Phase 4: Multi-restart ===
    print("\n=== Phase 4: Multi-restart ===", flush=True)
    remaining = deadline - time.monotonic()
    if remaining > 60 and best_seq is not None:
        n_restarts = 3
        t_per = min(remaining * 0.2 / n_restarts, 120)
        for r in range(n_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            if r == 0:
                f_init = symmetrize(best_seq)
            elif r == 1:
                noise = rng.standard_normal(len(best_seq)) * 0.003
                f_init = np.maximum(best_seq + noise, 1e-15)
                f_init = f_init / f_init.sum()
            else:
                f_init = 0.85 * best_seq + 0.15 * (np.ones(len(best_seq)) / len(best_seq))
                f_init = f_init / f_init.sum()

            t_alloc = min(t_per, remaining - 30)
            if t_alloc < 10:
                break
            a_opt, obj = lp_alternating(f_init, time_limit=t_alloc, verbose=(r == 0),
                                        label=f"restart_{r}")
            try_seq(a_opt, f"restart_{r}")

    # === Phase 5: Final fine-tuning ===
    print("\n=== Phase 5: Final tuning ===", flush=True)
    remaining = deadline - time.monotonic()
    if remaining > 15 and best_seq is not None:
        # Try different initializations at the best length
        n_best = len(best_seq)
        for sigma in [0.2, 0.25, 0.35, 0.4]:
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            a_init = make_gaussian(n_best, sigma)
            a_opt, obj = mirror_descent(a_init, time_limit=min(remaining * 0.1, 30),
                                        top_k=20, temp_init=0.001, lr_init=0.02)
            try_seq(a_opt, f"gauss_{sigma}")

        # Final symmetrization
        a_s = symmetrize(best_seq)
        try_seq(a_s, "final_sym")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})", flush=True)
    return [float(x) for x in best_seq]
