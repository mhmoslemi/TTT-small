from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_direction(a, M, n):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0."""
    c_obj = -np.ones(n)
    # Build constraint matrix: (a*b)_k = sum_j a_{k-j} * b_j for valid k-j
    # Instead of building the full matrix, use structure
    A_ub = np.zeros((2*n-1, n))
    for j in range(n):
        A_ub[j:j+n, j] = a
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, res
    return None, None

def lp_fw_step(a, M, t):
    """One LP-FW step."""
    n = len(a)
    b, res = solve_lp_direction(a, M, n)
    if b is None:
        return a, M, None
    b_sum = b.sum()
    if b_sum < 1e-15:
        return a, M, None
    g = b / b_sum
    # Line search
    best_t = 0
    best_val = fast_convolve(a).max()
    for alpha in np.linspace(0, 1, 20):
        a_trial = a + alpha * (g - a)
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-15:

        a_trial = a_trial / s
        val = fast_convolve(a_trial).max()
        if val < best_val:

    if best_t > 0:

        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        return a_new, fast_convolve(a_new).max(), res

    n = len(a)
    size = 1 << (2*n-2).bit_length()
    a = a.copy()

import time

    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

    direction = g - a
    # Initial step
    alpha = 1.0
    for _ in range(max_iter):
        a_trial = a + alpha * direction
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-15:
            alpha *= 0.5
            continue
        a_trial = a_trial / s
        c_trial = fast_convolve(a_trial)
        f_trial = c_trial.max()
        if f_trial < c_val:
            return alpha, a_trial, f_trial
        alpha *= 0.5
    return 0.0, a, c_val

def mirror_descent_lp(a0, time_limit, verbose=False):
    """
    Mirror descent (multiplicative weights) using LP dual variables
    as weighted subgradients.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    eta = 0.05
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(a)
        M = c.max()
        
        # Get subgradient from LP dual
        b, res = solve_lp_direction(a, M * 1.01, n)
        
        if res is not None and res.unevaluated is None:
            # Use dual variables if available
            # res.y gives dual variables for inequality constraints
            if res.y is not None and len(res.y) == 2 * n - 1:
                y = np.maximum(res.y, 0)
                y_sum = y.sum()
                if y_sum > 1e-15:
                    y = y / y_sum
                    # Weighted subgradient
                    # sg_j = 2 * sum_k y_k * a_{k-j} for valid k-j
                    A_w = build_conv_matrix(a)  # A_w[k,j] = a_{k-j}
                    sg = 2.0 * A_w.T @ y
                else:
                    sg = get_subgradient(a, c, M)
            else:
                sg = get_subgradient(a, c, M)
        else:
            sg = get_subgradient(a, c, M)
        
        # Multiplicative weight update
        a = a * np.exp(-eta * sg)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        obj = fast_convolve(a).max()
        
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            eta = min(eta * 1.01, 0.1)
        else:
            no_improve += 1
            eta *= 0.97
        
        if verbose and it % 20 == 0:
            print(f"    MD-LP iter {it}: obj={obj:.10f}, best={best_obj:.10f}, eta={eta:.4f}")
        
        if no_improve > 150:
            rng = np.random.default_rng(it * 17 + 3)
            a = np.maximum(best_a + rng.standard_normal(n) * 0.002, 1e-15)
            a = a / a.sum()
            eta = 0.04
            no_improve = 0
        
        if eta < 1e-5:

    return best_a, best_obj

def get_subgradient(a, c, M, top_k=20):
    """Get subgradient of max(black_conv) using top-k peaks."""
    n = len(a)
    c_max = c.max()
    # Weight by excess above threshold
    threshold = c_max * 0.95
    weights = np.maximum(c - threshold, 0)
    if weights.sum() < 1e-15:
        # Use top-k
        k = min(top_k, len(c))
        idx = np.argpartition(c, -k)[-k:]
        weights = np.zeros_like(c)
        weights[idx] = c[idx]
    
    w_sum = weights.sum()
    if w_sum < 1e-15:
        return np.zeros(n)
    weights = weights / w_sum
    
    # Subgradient: sg_j = 2 * sum_k weights_k * a_{k-j} for valid k-j
    # This is 2 * (weights * a_flipped)[j] correlated properly
    # Using matrix: sg = 2 * A.T @ weights where A[k,j] = a_{k-j}
    A = build_conv_matrix(a)
    sg = 2.0 * A.T @ weights
    return sg

def coordinate_descent_fine(a0, time_limit, verbose=False):
    """
    Coordinate descent with wider search and adaptive step sizes.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, fast_convolve(a0).max()
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    full_pass = 0
    
    while time.monotonic() - t0 < time_limit:
        full_pass += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-14:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            # Try a range of multipliers
            for mult in [0.5, 0.7, 0.8, 0.85, 0.9, 0.95, 1.05, 1.1, 1.2, 1.5, 2.0]:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                
                denom = 1.0 - old_val
                if abs(denom) < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if abs(denom) > 1e-12:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                if best_j_obj < best_obj:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            if full_pass > 3:
                break
            # Perturb to escape
            rng = np.random.default_rng(full_pass * 23 + 1)
            a = np.maximum(best_a + rng.standard_normal(n) * 0.001, 0)
            a = a / a.sum()
    
    return best_a, best_obj

def interpolate_sequence(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a)
    kernel = np.array([0.1, 0.2, 0.4, 0.2, 0.1])
    a_smooth = np.convolve(a_interp, kernel, mode='same')
    a_smooth = np.maximum(a_smooth, 0)
    ss = a_smooth.sum()
    if ss > 1e-15:
        a_smooth /= ss
    return a_smooth

def frequency_flatten(a, iters=50):
    """Try to flatten the power spectrum."""
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    a = a.copy()
    for _ in range(iters):
        A = np.fft.rfft(a, size)
        P = np.abs(A) ** 2
        target = P.mean()
        ratio = np.sqrt(target / (P + 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        a = a_new
    return a

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build Toeplitz matrix A where (Ab)_k = (a*b)_k for linear convolution."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_direction(a, M, n):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0."""
    c_obj = -np.ones(n)
    A_ub = build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, res
    return None, None


def get_subgradient(a, c, top_k=30):
    """Compute weighted subgradient of max(c) using top-k peaks."""
    n = len(a)
    c_max = c.max()
    threshold = c_max * 0.90
    weights = np.maximum(c - threshold, 0)
    if weights.sum() < 1e-15:
        k = min(top_k, len(c))
        idx = np.argpartition(c, -k)[-k:]
        weights = np.zeros_like(c)
        weights[idx] = np.maximum(c[idx] - c_max * 0.5, 0)
    w_sum = weights.sum()
    if w_sum < 1e-15:
        idx = np.argmax(c)
        weights = np.zeros_like(c)
        weights[idx] = 1.0
    weights = weights / weights.sum()
    A = build_conv_matrix(a)
    sg = 2.0 * (A.T @ weights)
    return sg

def mirror_descent_lp(a0, time_limit, verbose=False):
    """Mirror descent (multiplicative weights) using LP dual for subgradient."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    eta = 0.05
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(a)
        M = c.max()

        # Get subgradient (try LP dual first, fallback to top-k)
        b, res = solve_lp_direction(a, M * 1.02, n)
        used_dual = False
        if res is not None and hasattr(res, 'y') and res.y is not None:
            y = np.asarray(res.y)
            if len(y) == 2 * n - 1 and np.all(np.isfinite(y)):
                y = np.maximum(y, 0)
                y_sum = y.sum()
                if y_sum > 1e-15:
                    y = y / y_sum
                    A = build_conv_matrix(a)
                    sg = 2.0 * (A.T @ y)
                    used_dual = True

        if not used_dual:
            sg = get_subgradient(a, c, top_k=30)

        # Multiplicative weight update
        a_new = a * np.exp(-eta * sg)
        a_new = np.maximum(a_new, 1e-15)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        obj = fast_convolve(a_new).max()

        if obj < best_obj - 1e-14:
            a = a_new
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            eta = min(eta * 1.02, 0.12)
        else:
            no_improve += 1
            eta *= 0.96

            a = a / a.sum()

def interpolate_sequence(a, new_n):
    """Interpolate a sequence to a new length with smoothing."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a)
    kernel = np.array([0.1, 0.2, 0.4, 0.2, 0.1])
    a_smooth = np.convolve(a_interp, kernel, mode='same')
    a_smooth = np.maximum(a_smooth, 0)
    ss = a_smooth.sum()
    if ss > 1e-15:
        a_smooth /= ss
    return a_smooth


def frequency_flatten(a, iters=40):
    """Flatten the power spectrum via alternating projections."""
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    a = a.copy()
    for _ in range(iters):
        A = np.fft.rfft(a, size)
        P = np.abs(A) ** 2
        target = P.mean()
        ratio = np.sqrt(target / (P + 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        a = a_new
    return a


def raised_cosine(n):
    """Raised cosine (Hann window) initialization."""
    x = np.linspace(-1, 1, n)
    a = 0.5 * (1 + np.cos(np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def gaussian_init(n, sigma_frac=0.3):
    """Gaussian initialization."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a
