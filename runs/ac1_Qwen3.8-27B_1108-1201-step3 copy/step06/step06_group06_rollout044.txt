def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(f, tau):
    """Compute gradient of log-sum-exp smooth approximation of max-conv."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    h = conv_result[n - 1:2 * n - 1][::-1]
    grad = (2.0 / tau) * h
    return grad, conv

        A = np.fft.fft(a, N)
        conv = np.real(np.fft.ifft(A * A, N))[:2 * n - 1]
        
        # Compute how flat conv is
        conv_mean = conv.mean()
        conv_std = conv.std()
        flatness = conv_mean / (conv.max() + 1e-30)
        
        if flatness > 0.95:
            break
        
        # Modify spectrum: reduce peaks in |A^2| by reducing |A| where |A| is large
        mag = np.abs(A)
        mag_mean = np.mean(mag)
        # Compress dynamic range
        compression = 0.3 + 0.4 * (1.0 - flatness)
        new_mag = mag_mean + compression * (mag - mag_mean)
        new_mag = np.maximum(new_mag, 0.1 * mag_mean)
        phase = A / (mag + 1e-30)
        A_new = new_mag * phase
        a_full = np.real(np.fft.ifft(A_new))[:n]
        a_new = np.maximum(a_full, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new / s
    return a


def adam_optimize(f0, time_limit, tau_init=None, lr_init=0.005, verbose=False):
    """
    Adam optimization of smoothed max-conv with adaptive temperature
    and aggressive restart strategy.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    if tau_init is None:
        tau = 0.06 * conv.max()
    else:
        tau = tau_init

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    lr = lr_init
    t = 0

    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    best_no_improve = 0
    t0 = time.monotonic()
    tau_decay = 0.9995

    while time.monotonic() - t0 < time_limit:
        t += 1
        grad, conv = compute_smooth_grad(f, tau)
        M = conv.max()

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        tau *= tau_decay
        if tau < 2e-7 * best_obj:
            tau = 2e-7 * best_obj

        # Adaptive learning rate and restart
        if no_improve > 50:
            lr *= 0.9
        if no_improve > 200:
            # Warm restart: perturb best with noise
            rng = np.random.default_rng(t * 7 + 3)
            noise_scale = 0.003 + 0.005 * min(no_improve / 500.0, 2.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0
            lr = lr_init * 0.8
            conv_f = fast_convolve(f)
            tau = 0.04 * conv_f.max()

        if verbose and t % 1000 == 0:
            obj_score = 2 * n * best_obj
            print(f"    Adam iter {t}: obj={obj_score:.8f}, tau={tau:.2e}, lr={lr:.5f}, no_imp={no_improve}")

    return best_f, 2 * n * best_obj


def peak_pair_cd(f0, time_limit, verbose=False):
    """
    Aggressive multi-peak coordinate descent.
    Simultaneously targets all significant peaks by redistributing mass
    from peak-contributing pairs to non-contributing positions.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        # Get all significant peaks (above 95th percentile)
        threshold = np.percentile(conv, 90)
        peak_indices = np.where(conv > threshold)[0]
        peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]
        peak_indices = peak_indices[:min(10, len(peak_indices))]

        # Build a mask of "dangerous" positions (contributing to top peaks)
        danger = np.zeros(n, dtype=float)
        for pk in peak_indices[:5]:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            for j in range(js, je + 1):
                danger[j] += f[j] * f[pk - j] / (M + 1e-30)

        for pk in peak_indices:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[pk - idx]
            
            # Target the top-3 contributing pairs
            top_pair_count = min(3, len(idx))
            pair_order = np.argsort(products)[::-1][:top_pair_count]

            for po in pair_order:
                j_main = idx[po]
                j_other = pk - j_main
                
                if f[j_main] < 1e-10:
                    continue

                # Try reducing j_main and redistributing to safe positions
                safe_mask = danger < np.percentile(danger, 30)
                safe_mask[j_main] = False
                if j_main != j_other:
                    safe_mask[j_other] = False
                
                if safe_mask.sum() < 3:
                    continue

                for factor in [0.4, 0.6, 0.8]:
                    trial = f.copy()
                    old_val = trial[j_main]
                    reduction = old_val * (1 - factor)
                    trial[j_main] = old_val * factor
                    # Redistribute proportionally to safe positions weighted by current value
                    safe_weights = trial[safe_mask] + 1e-10
                    safe_weights /= safe_weights.sum()
                    trial[safe_mask] += reduction * safe_weights
                    trial = np.maximum(trial, 0)
                    ts = trial.sum()
                    if ts < 1e-15:
                        continue
                    trial /= ts
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-15:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break

        if not improved:
            no_improve += 1
            if no_improve > 20:
                rng = np.random.default_rng(it * 13 + 7)
                # Larger perturbation
                noise = rng.standard_normal(n) * 0.005
                f = project_simplex(best_f + noise)
                no_improve = 0
        else:
            no_improve = 0

        if verbose and it % 50 == 0:
            print(f"    CD iter {it}: obj={2*n*best_obj:.8f}, no_imp={no_improve}")

    return best_f, 2 * n * best_obj


def lp_fw_optimize(f0, time_limit, verbose=False):
    """
    LP-based Frank-Wolfe with adaptive constraint relaxation.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Solve LP: max sum(g) s.t. (f*g)[k] <= M*m_exp, g >= 0
        g = _solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp = min(m_exp * 1.15, 5.0)
            if m_exp > 4.0:
                rng = np.random.default_rng(it * 17 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp = min(m_exp * 1.1, 5.0)
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 30:
                m_exp = min(m_exp * 1.1, 4.0)
            if no_improve > 100:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue
        else:
            no_improve = 0

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = np.maximum(f + mid * (g - f), 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = np.maximum(f + best_alpha * (g - f), 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={2*n*best_obj:.8f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 5.0)

        if no_improve > 150:
            break

    return best_f, 2 * n * best_obj


def _solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def coordinate_descent_refine(f0, time_limit, verbose=False):
    """
    Fine-grained coordinate descent: try small multiplicative updates
    on individual coordinates to reduce the objective.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        # Random coordinate updates
        for _ in range(min(200, n)):
            j = rng.integers(0, n)
            if f[j] < 1e-12:
                continue
            
            # Try multiplicative updates
            for factor in [0.7, 0.85, 1.2, 1.4]:
                trial = f.copy()
                old = trial[j]
                trial[j] = old * factor
                diff = trial[j] - old
                # Redistribute to maintain sum=1
                if abs(diff) > 1e-15:
                    # Add/subtract from a random other coordinate
                    k = rng.integers(0, n)
                    if k == j:
                        k = (j + 1) % n
                    trial[k] -= diff
                    if trial[k] < 0:
                        # Redistribute diff across all others
                        others = np.ones(n)
                        others[j] = 0
                        trial[others] -= diff * others[others.sum() > 0] / others.sum()
                
                trial = np.maximum(trial, 0)
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                trial /= ts
                obj = fast_convolve(trial).max()
                if obj < best_obj - 1e-15:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    improved = True
                    break

        if not improved:
            # Try a pair update
            j1 = rng.integers(0, n)
            j2 = rng.integers(0, n)
            if j1 == j2 or f[j1] < 1e-12 or f[j2] < 1e-12:
                continue
            for ratio in [0.5, 0.7, 1.5, 2.0]:
                trial = f.copy()
                old1 = trial[j1]
                old2 = trial[j2]
                trial[j1] = old1 * ratio
                trial[j2] = old2 / ratio
                trial = np.maximum(trial, 0)
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                trial /= ts
                obj = fast_convolve(trial).max()
                if obj < best_obj - 1e-15:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    improved = True
                    break

    return best_f, 2 * n * best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_init_gaussian(n, center=0.5, width=0.3):
    """Gaussian initialization."""
    x = np.arange(n) - n * center
    sigma = n * width
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_init_flat_top(n, width=0.6):
    """Flat-topped initialization with smooth edges."""
    x = np.arange(n) - n / 2.0
    half = n * width / 2.0
    t = np.clip(np.abs(x) / half, 0, 1.0)
    a = (1.0 - t) ** 3
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_init_bimodal(n, sep=0.3):
    """Bimodal initialization."""
    a1 = make_init_gaussian(n, 0.5 - sep, 0.15)
    a2 = make_init_gaussian(n, 0.5 + sep, 0.15)
    a = (a1 + a2) / 2
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_init_random_smooth(n, seed=42):
    """Random smooth initialization via filtered noise."""
    rng = np.random.default_rng(seed)
    a = rng.standard_normal(n)
    # Smooth with a Gaussian kernel
    kernel_size = max(3, n // 32)
    kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size // 2) ** 2 / (kernel_size // 4) ** 2)
    kernel /= kernel.sum()
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a
