"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase optimization strategy:
1. LP-based fictitious play for coarse optimization
2. Nesterov-accelerated exact subgradient descent
3. L-BFGS with softmax-smoothed max (multiple temperatures)
4. Fourier-parameterized optimization for smoothness exploration
5. Simulated annealing for escaping local minima
6. Aggressive peak shaving for final polish
The key insight: the linear convolution (not autocorrelation) peak can be 
significantly below the Cauchy-Schwarz bound, allowing scores well below 2.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the objective value (assuming a is non-negative)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    max_c = fast_convolve(a).max()
    return 2.0 * n * max_c / (s * s)


def normalize(a):
    """Normalize to be non-negative with sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    try:
        _lp_solver_cached
        return _lp_solver_cached
    except NameError:
        pass
    global _lp_solver_cached
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.PDLP, cp.SCS, cp.ECOS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.SCS
    return cp.SCS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max conv(a,b)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.ECOS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def fictitious_play(n, time_limit, initial=None, solver=None, verbose=False):
    """Run fictitious play for the auto-convolution game."""
    if solver is None:
        solver = get_lp_solver()
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
        t = np.linspace(0, 2 * np.pi, n)
        a = a + 0.01 * np.cos(t)
        a = normalize(a)
    else:
        a = normalize(np.asarray(initial, dtype=np.float64))

    b = a.copy()
    best_score = compute_score(a)
    best_a = a.copy()

    iters = 0
    while time.monotonic() - t0 < time_limit:
        iters += 1
        b_new = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        candidates = [a, b, 0.5 * (a + b)]
        for cand in candidates:
            cand = normalize(cand)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()

        if verbose and iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP iter={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def subgradient_step(a, lr, mom_v, momentum):
    """One step of Nesterov-accelerated subgradient descent."""
    n = len(a)
    c = fast_convolve(a)
    max_c = c.max()
    # Find all indices achieving (near) the max
    threshold = max_c - 1e-10 * max_c
    peak_indices = np.where(c >= threshold)[0]
    # Weight subgradients uniformly over peaks
    w = np.ones(len(peak_indices)) / len(peak_indices)
    sg = np.zeros(n)
    for i, k in enumerate(peak_indices):
        if w[i] < 1e-15:
            continue
        js = max(0, int(k) - n + 1)
        je = min(n - 1, int(k))
        idx = np.arange(js, je + 1)
        sg[idx] += w[i] * a[int(k) - idx]
    # Normalize gradient (the objective is max_c, factor of 2n/(sum^2) is constant near optimum)
    s = a.sum()
    scale = 2.0 * n / (s * s)
    sg *= scale
    mom_v = momentum * mom_v - lr * sg
    return sg, mom_v


def nesterov_subgrad(a0, time_limit, lr_init=0.001, momentum=0.9, decay=0.9998):
    """Nesterov-accelerated subgradient descent on the simplex."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    v = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        a_lookahead = a + momentum * v
        # Project to simplex
        a_lookahead = _project_simplex(a_lookahead)
        # Compute gradient at lookahead
        sg, v = subgradient_step(a_lookahead, lr, v, momentum)
        # Update
        a_new = _project_simplex(a + v)
        a_new = normalize(a_new)

        score = compute_score(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        lr *= decay
        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 100000)
            a = best_a + rng.standard_normal(n) * 0.005
            a = normalize(a)
            v = np.zeros(n)
            lr = max(lr * 5, 0.0005)
            no_improve = 0

    return best_a, best_score


def _project_simplex(v):
    """Project vector onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    cond = u * rho_candidates > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1] + 1
        theta = (cssv[rho - 1] - 1.0) / rho
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


def softmax_obj_grad(a, T, n):
    """Compute softmax-smoothed objective and gradient."""
    c = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return 1e10, np.zeros(n)
    max_c = c.max()
    scaled = T * (c - max_c)
    exp_c = np.exp(scaled)
    log_sum_exp = max_c + np.log(exp_c.sum()) / T
    obj = 2.0 * n * log_sum_exp / (s * s)
    w = exp_c / exp_c.sum()
    # Gradient of max_c w.r.t. a
    grad_c = np.zeros(n)
    for i, k in enumerate(np.where(w > 1e-15)[0]):
        ki = int(k)
        js = max(0, ki - n + 1)
        je = min(n - 1, ki)
        idx = np.arange(js, je + 1)
        grad_c[idx] += w[i] * a[ki - idx]
    # Chain rule: d/d a [2n * log_sum_exp / s^2]
    grad = (2.0 * n / (s * s)) * grad_c - 2.0 * (2.0 * n * log_sum_exp / (s * s * s)) * a
    return obj, grad


def lbgfs_soft(a0, T, time_limit, maxiter=5000):
    """L-BFGS with softmax-smoothed max objective."""
    n = len(a0)
    a_start = normalize(a0.copy())
    t0 = time.monotonic()
    stopped = [False]

    def func(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        a = _project_simplex(z)
        a = normalize(a)
        obj, _ = softmax_obj_grad(a, T, n)
        return obj

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros_like(z)
        a = _project_simplex(z)
        a = normalize(a)
        _, g = softmax_obj_grad(a, T, n)
        return g

    try:
        res = minimize(fun=func, x0=a_start, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                                'maxfun': maxiter * 40, 'disp': False})
        a_out = normalize(_project_simplex(res.x))
    except Exception:
        a_out = a_start
    return a_out


def fourier_optimize(n, time_limit, n_fourier=15):
    """Optimize in Fourier parameter space."""
    t0 = time.monotonic()

    def fourier_to_seq(params):
        x = np.arange(n)
        a = np.ones(n) * params[0]
        for m in range(1, min(n_fourier, n // 2)):
            c = params[2 * m - 1]
            s = params[2 * m]
            a += c * np.cos(2 * np.pi * m * x / n) + s * np.sin(2 * np.pi * m * x / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()

    n_params = 2 * min(n_fourier, n // 2) + 1
    rng = np.random.default_rng(42)
    best_params = np.zeros(n_params)
    best_params[0] = 1.0
    best_score = compute_score(fourier_to_seq(best_params))

    def func(params):
        if time.monotonic() - t0 > time_limit * 1.5:
            return 1e15
        a = fourier_to_seq(params)
        return compute_score(a)

    # Multiple restarts
    n_restarts = max(1, int((time_limit - (time.monotonic() - t0)) / 30))
    for r in range(min(n_restarts, 8)):
        if time.monotonic() - t0 > time_limit - 10:
            break
        if r == 0:
            params0 = best_params.copy()
        else:
            params0 = rng.standard_normal(n_params) * 0.1
            params0[0] = 1.0
        try:
            res = minimize(func, params0, method='Nelder-Mead',
                           options={'maxiter': 10000, 'xatol': 1e-8, 'fatol': 1e-10,
                                    'adaptive': True})
            a_test = fourier_to_seq(res.x)
            sc = compute_score(a_test)
            if sc < best_score:
                best_score = sc
                best_params = res.x.copy()
                print(f"    Fourier restart {r}: score={sc:.6f}")
        except Exception:
            continue

    a_out = fourier_to_seq(best_params)
    return a_out, best_score


def simulated_annealing(a0, time_limit, t_init=0.02, t_decay=0.9995, noise_scale=0.05):
    """Simulated annealing on the simplex."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t = t_init
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    step = 0

    while time.monotonic() - t0 < time_limit:
        # Generate perturbation
        noise = rng.standard_normal(n) * noise_scale * t
        a_trial = a + noise
        a_trial = _project_simplex(a_trial)
        a_trial = normalize(a_trial)

        score_trial = compute_score(a_trial)
        score_current = compute_score(a)

        if score_trial < score_current or rng.random() < np.exp(-(score_trial - score_current) / max(t, 1e-15)):
            a = a_trial
            if score_trial < best_score:
                best_score = score_trial
                best_a = a.copy()

        t *= t_decay
        step += 1
        if step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA step={step} ({elapsed:.0f}s): best={best_score:.6f} t={t:.6f}")

        # Periodically reheat
        if step % 20000 == 0 and t < 0.001:
            t = t_init * 0.3
            noise_scale = max(0.01, noise_scale * 0.5)

    return best_a, best_score


def peak_shave_aggressive(a0, time_limit, n_steps=5000):
    """Aggressive peak shaving with larger moves."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        # Find peaks (top few)
        top_k = min(5, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]

        improved = False
        for pk in peak_indices:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            # Find top contributors
            top_contrib = np.argsort(contribs)[::-1][:5]

            for ci in top_contrib:
                i = int(idx[ci])
                old_val = a[i]
                if old_val < 1e-10:
                    continue
                for factor in [0.9, 0.8, 0.7, 0.5, 0.3]:
                    diff = old_val * (1 - factor)
                    if diff < 1e-12:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others = a[mask]
                    if others.sum() < 1e-15:
                        break
                    a_trial = a.copy()
                    a_trial[i] = old_val * factor
                    a_trial[mask] = others + diff * others / others.sum()
                    a_trial = normalize(a_trial)
                    score = compute_score(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try reducing two elements simultaneously
            pk = int(peak_indices[0])
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            top_contrib = np.argsort(contribs)[::-1][:3]
            found = False
            for ci1 in top_contrib:
                for ci2 in top_contrib:
                    if ci1 == ci2:
                        continue
                    i1, i2 = int(idx[ci1]), int(idx[ci2])
                    if a[i1] < 1e-10 or a[i2] < 1e-10:
                        continue
                    for f in [0.8, 0.6, 0.4]:
                        a_trial = a.copy()
                        a_trial[i1] *= f
                        a_trial[i2] *= f
                        a_trial = normalize(a_trial)
                        score = compute_score(a_trial)
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            found = True
                            break
                    if found:
                        break
                if found:
                    break
            if not found:
                break

    return best_a, best_score


def coordinate_descent(a0, time_limit, group_size=4):
    """Block coordinate descent: optimize small groups of coordinates at a time."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(77)

    while time.monotonic() - t0 < time_limit:
        # Pick a random group of coordinates
        start = rng.integers(0, max(1, n - group_size))
        group = np.arange(start, min(n, start + group_size))

        # For each coordinate in the group, try to reduce it
        improved = False
        for i in group:
            old_val = a[i]
            if old_val < 1e-10:
                continue
            for factor in [0.95, 0.9, 0.8, 0.7]:
                diff = old_val * (1 - factor)
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                if others.sum() < 1e-15:
                    break
                a_trial = a.copy()
                a_trial[i] = old_val * factor
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = normalize(a_trial)
                score = compute_score(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
                elif score < best_score - 1e-16:
                    a = a_trial
        if not improved:
            # Try increasing a random coordinate (redistributing from others)
            i = int(rng.integers(0, n))
            mask = np.ones(n, dtype=bool)
            mask[i] = False
            others = a[mask]
            if others.sum() < 1e-15:
                continue
            for frac in [0.05, 0.1, 0.2]:
                diff = frac * others.sum()
                a_trial = a.copy()
                a_trial[i] = a[i] + diff
                a_trial[mask] = others * (1 - frac * a[i] / (a[i] + diff)) if a[i] + diff > 0 else others
                # Simpler: redistribute
                a_trial = a.copy()
                a_trial[i] = a[i] + diff
                a_trial[mask] = others - diff * others / others.sum()
                a_trial = np.maximum(a_trial, 0)
                a_trial = normalize(a_trial)
                score = compute_score(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
        if not improved:
            # Small random perturbation to escape
            a = a + rng.standard_normal(n) * 0.002
            a = normalize(a)

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        score = float('inf')
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining LP fictitious play, subgradient methods,
    L-BFGS with softmax smoothing, Fourier parameterization, simulated annealing,
    and peak shaving.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        # Initialize with uniform at n=768
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized uniform: n={n_init}, score={best_score:.6f}")

    n_parent = len(parent)

    # === Phase 1: Fictitious Play (coarse) ===
    print("\n=== Phase 1: Fictitious Play ===")
    rem = deadline - time.monotonic() - 400
    if rem > 60:
        t_fp = min(rem * 0.5, 200.0)
        print(f"  FP n={n_parent}, budget={t_fp:.0f}s")
        a_fp, sc_fp = fictitious_play(n_parent, t_fp, initial=parent, solver=solver, verbose=True)
        best_seq, best_score = check_and_update(a_fp, "FP_main", best_seq, best_score)

    # Also try a slightly different length
    alt_n = int(n_parent * 1.25)
    rem = deadline - time.monotonic() - 350
    if rem > 60 and best_seq is not None:
        t_fp2 = min(rem * 0.4, 100.0)
        init_alt = interpolate_sequence(parent, alt_n)
        print(f"  FP n={alt_n}, budget={t_fp2:.0f}s")
        a_fp2, sc_fp2 = fictitious_play(alt_n, t_fp2, initial=init_alt, solver=solver, verbose=False)
        best_seq, best_score = check_and_update(a_fp2, f"FP_alt_n{alt_n}", best_seq, best_score)

    # === Phase 2: L-BFGS with multiple temperatures ===
    print(f"\n=== Phase 2: L-BFGS Softmax (best={best_score:.6f}) ===")
    if best_seq is not None:
        for T in [100, 300, 1000, 3000]:
            rem = deadline - time.monotonic() - 200
            if rem < 20:
                break
            t_alloc = min(rem * 0.25, 100.0)
            print(f"  L-BFGS T={T}, budget={t_alloc:.0f}s")
            a_lb = lbgfs_soft(best_seq, T, t_alloc, maxiter=3000)
            best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T}", best_seq, best_score)

    # === Phase 3: Nesterov Subgradient ===
    print(f"\n=== Phase 3: Nesterov Subgradient (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_sg = min(rem * 0.3, 150.0)
            n_best = len(best_seq)
            lr_init = 0.002 * (n_best / 768.0)
            print(f"  SubGrad n={n_best}, lr={lr_init:.5f}, budget={t_sg:.0f}s")
            a_sg, _ = nesterov_subgrad(best_seq, t_sg, lr_init=lr_init, momentum=0.92, decay=0.9997)
            best_seq, best_score = check_and_update(a_sg, "nesterov_sg", best_seq, best_score)

            # Second round with different params
            rem = deadline - time.monotonic() - 150
            if rem > 20:
                t_sg2 = min(rem * 0.4, 80.0)
                a_sg2, _ = nesterov_subgrad(best_seq, t_sg2, lr_init=0.0005, momentum=0.95, decay=0.9999)
                best_seq, best_score = check_and_update(a_sg2, "nesterov_sg2", best_seq, best_score)

    # === Phase 4: Fourier optimization ===
    print(f"\n=== Phase 4: Fourier Optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:

            print(f"  Fourier n={n_best}, budget={t_four:.0f}s")
            a_fou, sc_fou = fourier_optimize(n_best, t_four, n_fourier=20)
            best_seq, best_score = check_and_update(a_fou, "fourier", best_seq, best_score)

    # === Phase 5: Simulated Annealing ===
    print(f"\n=== Phase 5: Simulated Annealing (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_sa = min(rem * 0.4, 120.0)
            print(f"  SA budget={t_sa:.0f}s")
            a_sa, _ = simulated_annealing(best_seq, t_sa, t_init=0.01, t_decay=0.9998, noise_scale=0.03)
            best_seq, best_score = check_and_update(a_sa, "sim_anneal", best_seq, best_score)

    # === Phase 6: Aggressive Peak Shaving ===
    print(f"\n=== Phase 6: Peak Shaving (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_ps = min(rem * 0.5, 100.0)
            print(f"  PeakShave budget={t_ps:.0f}s")
            a_ps, _ = peak_shave_aggressive(best_seq, t_ps, n_steps=10000)
            best_seq, best_score = check_and_update(a_ps, "peak_shave", best_seq, best_score)

    # === Phase 7: Coordinate Descent ===
    print(f"\n=== Phase 7: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_cd = min(rem * 0.5, 60.0)
            print(f"  CoordDesc budget={t_cd:.0f}s")
            a_cd, _ = coordinate_descent(best_seq, t_cd, group_size=3)
            best_seq, best_score = check_and_update(a_cd, "coord_desc", best_seq, best_score)

    # === Phase 8: Final L-BFGS polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_final = min(rem * 0.5, 50.0)
            print(f"  Final L-BFGS T=5000, budget={t_final:.0f}s")
            a_fin = lbgfs_soft(best_seq, 5000, t_final, maxiter=5000)
            best_seq, best_score = check_and_update(a_fin, "final_lbgfs", best_seq, best_score)

        rem = deadline - time.monotonic() - 15
        if rem > 10:
            t_ps2 = min(rem * 0.6, 30.0)
            a_ps2, _ = peak_shave_aggressive(best_seq, t_ps2, n_steps=5000)
            best_seq, best_score = check_and_update(a_ps2, "final_ps", best_seq, best_score)

    # === Phase 9: Multi-length exploration if time remains ===
    print(f"\n=== Phase 9: Multi-length (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for factor in [0.75, 1.5, 2.0, 0.5]:
            rem = deadline - time.monotonic() - 20
            if rem < 25:
                break
            alt_n = int(n_best * factor)
            if alt_n < 64 or alt_n > 4096 or alt_n == n_best:
                continue
            t_alloc = min(rem * 0.4, 60.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            # Quick L-BFGS at new length
            a_alt = lbgfs_soft(init_alt, 500, t_alloc, maxiter=2000)
            best_seq, best_score = check_and_update(a_alt, f"alt_n{alt_n}", best_seq, best_score)

    # Fallback
    if best_seq is None:
        best_seq = parent if parent is not None else np.ones(512) / 512
        best_score = compute_score(best_seq)

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
