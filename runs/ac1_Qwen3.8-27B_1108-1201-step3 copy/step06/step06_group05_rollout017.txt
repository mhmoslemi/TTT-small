import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lp_gradient(f, c, p):
    """
    Compute gradient of (1/p)*(sum c_k^p)^(1/p) where c = f*f.
    Uses FFT for efficiency.
    """
    n = len(f)
    max_c = np.max(c)
    if max_c < 1e-18:
        return np.zeros(n)
    c_norm = c / max_c
    w = c_norm ** (p - 1)
    # Gradient w.r.t. f_j: 2 * sum_i f_i * w_{j+i} (normalized)
    w_rev = w[::-1].copy()
    conv_len = n + len(w_rev) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(F * W, fft_size)[:conv_len]
    indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
    g = 2.0 * conv[indices] * max_c ** (p - 1) / max_c
    return g


def compute_subgradient_max(f, c):
    """Compute subgradient of max(c) where c = f*f."""
    n = len(f)
    max_c = np.max(c)
    mask = c >= max_c - 1e-12
    w = mask.astype(np.float64)
    w /= w.sum()
    w_rev = w[::-1].copy()
    conv_len = n + len(w_rev) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(F * W, fft_size)[:conv_len]
    indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
    g = 2.0 * conv[indices]
    return g


def make_gaussian(n, sigma_frac=0.25):
    """Gaussian centered at (n-1)/2."""
    x = np.arange(n) - (n - 1) / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_hann(n):
    """Hann window (raised cosine)."""
    a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_chebyshev(n):
    """Chebyshev-like (1-cos) profile."""
    x = np.arange(n) / (n - 1) * np.pi
    a = 1 - np.cos(x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_flat_topped(n, flat_frac=0.3):
    """Flat-topped distribution - uniform in center, tapering at edges."""
    a = np.ones(n)
    taper = int(n * (1 - flat_frac) / 2)

        for i in range(taper):
            t = (i + 1) / taper
            a[i] = 0.5 * (1 - np.cos(np.pi * t))
            a[n - 1 - i] = a[i]
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_three_point(n):
    """Distribution concentrated at 3 symmetric points."""
    a = np.zeros(n)
    center = (n - 1) / 2.0
    spread = n * 0.3
    for frac in [0.0, 0.5, 1.0]:
        pos = int(round(center + (frac - 0.5) * 2 * spread))
        if 0 <= pos < n:
            a[pos] = 1.0
    a /= a.sum()
    return a


def make_spectral_init(n):
    """
    Initialize using frequency domain: choose |S(omega)|^2 to be as flat as possible,
    then take inverse DFT and project.
    """
    # Start with a raised cosine in time domain as base
    a = make_hann(n)
    # Compute DFT
    S = np.fft.rfft(a)
    # Make |S|^2 more flat (reduce contrast)
    mag = np.abs(S)
    mag_min = mag.min() * 0.5
    mag = (mag + mag_min) / (mag.max() + mag_min)
    mag = mag / mag.max()
    # Reconstruct with flattened spectrum
    phase = np.angle(S)
    S_new = mag * np.exp(1j * phase)
    a_new = np.fft.irfft(S_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = make_hann(n)
    return a_new


def make_palindromic(base):
    """Make a sequence palindromic by averaging with its reverse."""
    n = len(base)
    return (base + base[::-1]) / (2.0 * base.sum())


def nesterov_lp_descent(f0, time_limit, p_start=4, p_end=200, verbose=False):
    """
    Nesterov-accelerated gradient descent on L^p smoothed objective.
    Minimize (1/p)*(sum (f*f)_k^p)^(1/p) subject to f >= 0, sum(f) = 1.
    p increases over time from p_start to p_end for sharper approximation to L_inf.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    t0 = time.monotonic()
    t_nest = 1.0
    y = f.copy()
    x_prev = f.copy()
    
    # Schedule: increase p over time
    total_time = time_limit
    p_schedule_stages = 8
    stage_duration = total_time / p_schedule_stages
    
    for stage in range(p_schedule_stages):
        p = int(p_start * (p_end / p_start) ** (stage / max(1, p_schedule_stages - 1)))
        p = max(2, min(p, 1000))
        
        stage_start = time.monotonic()
        it = 0
        no_improve = 0
        staleness = 0.0
        
        # Adaptive step size
        c = fast_convolve(y)
        max_c = c.max()
        # Lipschitz constant estimate for L^p gradient
        L_est = 4.0 * p * max_c ** (p - 1) * n
        step = 1.0 / max(L_est, 1e-10)
        step = min(step, 0.5 / n)
        
        while time.monotonic() - stage_start < stage_duration:
            it += 1
            
            # Nesterov extrapolation
            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_nest ** 2)) / 2.0
            beta = (t_nest - 1.0) / t_next
            y = f + beta * (f - x_prev)
            y = np.maximum(y, 0)
            ys = y.sum()
            if ys > 1e-15:
                y = y / ys
            
            # Compute gradient at y
            c_y = fast_convolve(y)
            g = compute_lp_gradient(y, c_y, p)
            
            # Gradient norm for step size adaptation
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-15:
                break
            
            # Step update
            x_prev = f.copy()
            f_new = y - step * g
            f_new = project_simplex(f_new)
            
            # Evaluate
            c_new = fast_convolve(f_new)
            obj_new = c_new.max()
            
            # Backtracking if objective increased significantly
            if obj_new > best_obj * 1.01 + 1e-10:
                step *= 0.5
                f_new = y - step * g
                f_new = project_simplex(f_new)
                c_new = fast_convolve(f_new)
                obj_new = c_new.max()
            
            f = f_new
            t_nest = t_next
            
            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_f = f.copy()
                no_improve = 0
                step = min(step * 1.01, 2.0 / (g_norm + 1e-10))
            else:
                no_improve += 1
                if no_improve > 50:
                    step *= 0.9
            
            # Periodic restart from best
            if no_improve > 200:
                rng = np.random.default_rng(int(time.monotonic() * 1000) + stage * 7)
                noise_scale = 0.003 / np.sqrt(stage + 1)
                f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
                y = f.copy()
                x_prev = f.copy()
                t_nest = 1.0
                no_improve = 0
                best_obj = fast_convolve(f).max()
                if best_obj < obj_new:
                    best_f = f.copy()
            
            if verbose and it % 50 == 0:
                print(f"      Nesterov p={p} iter={it}: obj={obj_new:.10f}, best={best_obj:.10f}, step={step:.2e}")
    
    return best_f, best_obj


def akl_fw(f0, time_limit, verbose=False):
    """
    Away-step Frank-Wolfe (AKL) for minimizing max(f*f) on the simplex.
    Uses LP oracle for the linear minimization step.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    c = fast_convolve(f)
    M = c.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    # AKL maintains a set of active vertices
    v_set = [f.copy()]
    v_weights = [1.0]
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute subgradient at current f
        c = fast_convolve(f)
        M = c.max()
        g = compute_subgradient_max(f, c)
        
        # Solve LP: min g^T v s.t. v >= 0, sum(v) = 1
        # This is equivalent to: find the vertex that minimizes g^T v
        # For simplex, the optimal is v = e_{argmin g}
        # But we want the LP direction that reduces max(f*f)
        # Actually for FW, we minimize <g, v> over the feasible set
        # The feasible set is the simplex, so min is at a vertex: v = e_j where j = argmin g
        # But this is too crude. Let's use the LP formulation:
        # LP: min g^T v s.t. (f*v)_k <= M for all k, v >= 0, sum(v) = 1
        # This gives a better direction than just the greedy vertex.
        
        # Actually for standard FW, the direction is simply argmin g^T v over the feasible set.
        # For the simplex, this is e_{argmin g}. But for AKL, we can do better.
        
        # Let's use the LP-based direction (like the previous approach)
        # LP: max sum(v) - lambda * max_k (f*v)_k
        # Simplified: max sum(v) s.t. (f*v)_k <= M, v >= 0
        c_obj = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M * 1.01)
        bounds = [(0, None)] * n
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        
        if not res.success:
            # Fallback: greedy direction
            v = np.zeros(n)
            v[np.argmin(g)] = 1.0
        else:
            v = res.x
            vs = v.sum()
            if vs < 1e-15:
                v = np.zeros(n)
                v[np.argmin(g)] = 1.0
            else:
                v = v / vs
        
        # Fixed point check
        if np.max(np.abs(v - f)) < 1e-12:
            no_improve += 1
            if no_improve > 100:
                break
            continue
        
        # AKL: consider both toward v and away from each vertex in v_set
        best_alpha = 0.0
        best_val = M
        best_mode = 0  # 0 = toward v, 1 = away from v_set[i]
        best_idx = 0
        
        # Toward v (standard FW step)
        for alpha in np.linspace(0.05, 1.0, 10):
            ft = f + alpha * (v - f)
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
                best_mode = 0
                best_idx = 0
        
        # Away steps: for each vertex in v_set
        for wi, wv in enumerate(v_weights):
            if wv < 0.05:
                continue
            vi = v_set[wi]
            for alpha in np.linspace(0.05, 0.5, 6):
                # Move away from vi: f_new = (1 - gamma) * f + gamma * (v - f) but also away
                # AKL: if gamma < 1/(1+w_i), move toward v
                #       if gamma >= 1/(1+w_i), move away from v_i
                # Simplified: try f_new = normalize((1-alpha)*f - alpha*w_i*vi + alpha*(1+w_i)*v)
                # Even simpler: try removing mass from vi direction
                ft = f.copy()
                # Remove fraction alpha of the weight on vi
                ft -= alpha * wv * vi
                ft += alpha * wv * v
                ft = np.maximum(ft, 0)
                st = ft.sum()
                if st < 1e-15:
                    continue
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha
                    best_mode = 1
                    best_idx = wi
        
        if best_mode == 0:
            f_new = f + best_alpha * (v - f)
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            # Update v_set (simplified: just add v and renormalize)
            if len(v_set) < 10:
                v_set.append(v.copy())
                v_weights.append(best_alpha)
            else:
                # Remove smallest weight
                min_i = int(np.argmin(v_weights))
                v_set.pop(min_i)
                v_weights.pop(min_i)
                v_set.append(v.copy())
                v_weights.append(best_alpha)
        elif best_mode == 1:
            vi = v_set[best_idx]
            f_new = f - best_alpha * v_weights[best_idx] * vi
            f_new += best_alpha * v_weights[best_idx] * v
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            v_weights[best_idx] -= best_alpha
            if v_weights[best_idx] < 1e-6:
                v_set.pop(best_idx)
                v_weights.pop(best_idx)
        
        f = f_new
        M = f_new.sum()  # placeholder
        
        c = fast_convolve(f)
        M = c.max()
        
        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and it % 20 == 0:
            print(f"      AKL iter={it}: obj={M:.10f}, best={best_obj:.10f}")
        
        if no_improve > 300:
            # Restart with perturbation
            rng = np.random.default_rng(it * 17 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            v_set = [f.copy()]
            v_weights = [1.0]
            no_improve = 0
    
    return best_f, best_obj


def subgradient_avg(f0, time_limit, verbose=False):
    """
    Subgradient method with Polyak averaging for robustness.
    Step size: t_k = c / sqrt(k) for the L_inf objective.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    # Initialize x_bar (averaged iterate)
    x_bar = f.copy()
    t0 = time.monotonic()
    
    c_level = best_obj
    
    while time.monotonic() - t0 < time_limit:
        it = int((time.monotonic() - t0) * 100) + 1
        t_k = 1.0 / np.sqrt(max(it, 1)) * 0.1
        
        # Subgradient
        c = fast_convolve(f)
        M = c.max()
        g = compute_subgradient_max(f, c)
        
        # Update
        f_new = project_simplex(f - t_k * g)
        
        # Update average
        x_bar = x_bar + (f_new - x_bar) / (it + 1)
        
        # Check averaged iterate
        c_bar = fast_convolve(x_bar)
        obj_bar = c_bar.max()
        
        if obj_bar < best_obj - 1e-13:
            best_obj = obj_bar
            best_f = x_bar.copy()
        
        c_new = fast_convolve(f_new)
        if c_new.max() < best_obj - 1e-13:
            best_obj = c_new.max()
            best_f = f_new.copy()
        
        f = f_new
    
    return best_f, best_obj


def coordinate_polish(f0, time_limit, verbose=False):
    """
    Fine coordinate descent: adjust individual coordinates to reduce max convolution.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f0, fast_convolve(np.maximum(f0, 0)).max()
    f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        c = fast_convolve(f)
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in [0.7, 0.8, 0.9, 0.95, 1.05, 1.1, 1.2, 1.4]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.3:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length with smoothing."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    kernel = np.array([0.1, 0.2, 0.4, 0.2, 0.1])
    f_smooth = np.convolve(f_interp, kernel, mode='same')
    f_smooth = np.maximum(f_smooth, 0)
    ss = f_smooth.sum()
    if ss > 1e-15:
        f_smooth /= ss
    return f_smooth

            t_alloc = min(rem * 0.4, 100)
            print(f"  Subgrad avg, t={t_alloc:.0f}s")
            f_opt, obj = subgradient_avg(best_seq, time_limit=t_alloc)
            try_seq(f_opt, "subgrad_avg")
    
    print(f"  Phase 5 time: {time.monotonic()-t5:.1f}s")
    
    # === Phase 6: Deep AKL with diverse restarts ===
    print("\n=== Phase 6: Deep AKL ===")
    t6 = time.monotonic()
    
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(3):
            rem = deadline - time.monotonic() - 60
            if rem < 40:
                break
            t_alloc = min(rem / (3 - p), 80)
            
            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.003
                f_init = project_simplex(best_seq + noise)
            else:
                rc = make_hann(n_best)
                f_init = 0.7 * best_seq + 0.3 * rc
                f_init = f_init / f_init.sum()
            
            print(f"  Deep AKL variant {p}, t={t_alloc:.0f}s")
            f_opt, obj = akl_fw(f_init, time_limit=t_alloc)
            try_seq(f_opt, f"deep_akl_{p}")
    
    print(f"  Phase 6 time: {time.monotonic()-t6:.1f}s")
    
    # === Phase 7: Coordinate polish ===
    print("\n=== Phase 7: Coordinate Polish ===")
    t7 = time.monotonic()
    
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_alloc = min(rem, 60)
            print(f"  Coord descent, t={t_alloc:.0f}s")
            f_opt, obj = coordinate_polish(best_seq, time_limit=t_alloc)
            try_seq(f_opt, "coord_polish")
            
            # One more AKL squeeze
            rem = deadline - time.monotonic() - 10
            if rem > 30:
                t_alloc = min(rem, 50)
                f_opt, obj = akl_fw(f_opt, time_limit=t_alloc)
                try_seq(f_opt, "final_akl")
    
    print(f"  Phase 7 time: {time.monotonic()-t7:.1f}s")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
