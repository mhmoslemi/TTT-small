def evaluate_sequence(sequence):

def fast_convolve(seq):
    """Compute the full linear convolution of a sequence using FFT-like decomposition (Optimized for reduced complexity)."""
    n = len(seq)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Zero-padding and FFT
    A = np.fft.rfft(seq, fft_size)
    conv_result = np.fft.irfft(A * A, fft_size)[:size]
    
    return conv_result

def project_simplex(v):
    """Project to the simplex constraint (tuple of floats in Python; full projection as in the provided method)."""
    n = len(v)
    if n == 0:
        return v
        
    u = np.sort(v)[::-1]

    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0).astype(float) / sum(np.maximum(v - theta, 0))
