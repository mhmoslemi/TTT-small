def project_simplex(v):
    """Euclidean projection on the probability simplex allowing zeros. Returns a vector with elements >=0 and sum=1."""
    n = len(v)
    if n == 1:

    u = np.sort(v)[::-1]

    return np.maximum(v[rho:] - theta, 0.0).reshape(v.shape)
