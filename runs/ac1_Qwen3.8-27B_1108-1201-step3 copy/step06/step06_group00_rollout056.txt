def coordinate_descent(a, time_limit, max_passes=100, verbose=True):
    """
    Coordinate descent on the TRUE objective (max of convolution values).
    Uses golden section search for each coordinate.
    Works on simplex (sum=1, a>=0).
    """
    t0 = time.monotonic()
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    best_a = a.copy()
    best_max = conv_max(a)
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    
    for path in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        n_improved = 0
        
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = best_a[i]
            
            # If old_val is essentially 0 or 1, skip
            if old_val < 1e-12 or old_val > 1 - 1e-8:
                continue
            
            # Golden section search for optimal x = a[i]
            # a'[i] = x, a'[j] = a[j]*(1-x)/(1-old_val) for j != i
            # Objective: max of conv(a', a')
            
            # Initial bracket: try to narrow down
            # The optimal is usually near old_val
            delta = max(0.05, 0.3 * old_val)
            lo = max(0.0, old_val - delta)
            hi = min(1.0 - 1e-10, old_val + delta)
            
            # Evaluate at current point
            f_mid = best_max
            
            # Check if moving down helps
            x_test_down = max(0.0, old_val - delta * 0.5)
            if x_test_down < old_val:
                a_t = best_a.copy()
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                s_val = (1 - x_test_down) / (1 - old_val)
                a_t[mask] *= s_val
                a_t[i] = x_test_down
                f_down = conv_max(a_t)
                if f_down < f_mid:
                    # Optimal is below current
                    hi = old_val
                    f_mid = f_down
                    lo = max(0.0, old_val - 2 * delta)
                else:
                    f_up_test = old_val + delta * 0.5
                    if f_up_test < 1 - 1e-10:
                        a_t = best_a.copy()
                        mask = np.ones(n, dtype=bool)
                        mask[i] = False
                        s_val = (1 - f_up_test) / (1 - old_val)
                        a_t[mask] *= s_val
                        a_t[i] = f_up_test
                        f_up = conv_max(a_t)
                        if f_up < f_mid:
                            lo = old_val
                            f_mid = f_up
                            hi = min(1.0 - 1e-10, old_val + 2 * delta)
            else:
                # old_val is small, try moving up
                x_test_up = old_val + delta
                if x_test_up < 1 - 1e-10:
                    a_t = best_a.copy()
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    s_val = (1 - x_test_up) / (1 - old_val)
                    a_t[mask] *= s_val
                    a_t[i] = x_test_up
                    f_up = conv_max(a_t)
                    if f_up < f_mid:
                        lo = old_val
                        f_mid = f_up
                        hi = min(1.0 - 1e-10, old_val + 2 * delta)
            
            # Now golden section on [lo, hi]
            if lo >= hi - 1e-10:
                lo = max(0.0, old_val - 0.1)
                hi = min(1.0 - 1e-10, old_val + 0.1)
            
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            
            def eval_at_x(x):
                if x < 0 or x > 1 - 1e-10:
                    return 1e10
                a_t = best_a.copy()
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                s_val = (1 - x) / (1 - old_val)
                a_t[mask] *= s_val
                a_t[i] = x
                return conv_max(a_t)
            
            fc = eval_at_x(c)
            fd = eval_at_x(d)
            
            # If neither point improves, skip
            if fc >= best_max - 1e-10 and fd >= best_max - 1e-10:
                continue
            
            for _ in range(30):
                if abs(hi - lo) < 1e-7:
                    break
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = eval_at_x(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = eval_at_x(d)
            
            x_opt = (lo + hi) / 2
            f_opt = eval_at_x(x_opt)
            
            if f_opt < best_max - 1e-10:
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                s_val = (1 - x_opt) / (1 - old_val)
                best_a[mask] *= s_val
                best_a[i] = x_opt
                best_a = best_a / best_a.sum()
                best_max = conv_max(best_a)
                improved = True
                n_improved += 1
        
        if verbose and (path % 5 == 0 or not improved):
            elapsed = time.monotonic() - t0
            print(f"    CD pass {path}: max={best_max:.8f}, improved_coords={n_improved}, t={elapsed:.1f}s")
        if not improved:
            break
    
    return best_a, best_max


def pairwise_cd(a, time_limit, max_passes=20, verbose=False):
    """
    Pairwise coordinate descent: optimize pairs of coordinates simultaneously.
    Try pairs that are "far apart" (to reduce correlation peaks).
    """
    t0 = time.monotonic()
    n = len(a)
    best_a = a.copy().astype(np.float64)
    s = best_a.sum()
    if s < 1e-15:
        return best_a, float('inf')
    best_a = best_a / s
    best_max = conv_max(best_a)
    
    rng = np.random.default_rng(12345)
    
    for path in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        
        # Try several random pairs
        n_pairs = min(50, n // 4)
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            
            i, j = rng.choice(n, size=2, replace=False)
            if abs(i - j) < n // 8:
                continue
            
            vi, vj = best_a[i], best_a[j]
            if vi < 1e-10 or vj < 1e-10:
                continue
            
            # 2D search: vary a[i] and a[j], rescale others
            # a'[i] = x, a'[j] = y, a'[k] = a[k]*(1-x-y)/(1-vi-vj) for k!=i,j
            # We want to minimize max(conv(a',a'))
            
            # First try: move mass from one to the other (x+y = vi+vj)
            # x = vi + t, y = vj - t for t in [-vi, vj]
            best_t = 0.0
            best_f = best_max
            
            # Sample several values of t
            t_candidates = np.linspace(-0.8 * vi, 0.8 * vj, 20)
            for t in t_candidates:
                x = vi + t
                y = vj - t
                if x < 0 or y < 0 or x + y > 1 - 1e-10:
                    continue
                a_t = best_a.copy()
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                mask[j] = False
                s_val = (1 - x - y) / (1 - vi - vj)
                if s_val < 0:
                    continue
                a_t[mask] *= s_val
                a_t[i] = x
                a_t[j] = y
                f = conv_max(a_t)
                if f < best_f:
                    best_f = f
                    best_t = t
            
            if best_t != 0 and best_f < best_max - 1e-10:
                x = vi + best_t
                y = vj - best_t
                a_t = best_a.copy()
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                mask[j] = False
                s_val = (1 - x - y) / (1 - vi - vj)
                a_t[mask] *= s_val
                a_t[i] = x
                a_t[j] = y
                best_a = a_t / a_t.sum()
                best_max = conv_max(best_a)
                improved = True
        
        if not improved:
            break
    
    return best_a, best_max


def make_diverse_init(n, seed=42):
    """Create several diverse initializations and return the best one."""
    rng = np.random.default_rng(seed)
    candidates = []
    
    # Exponentially decreasing
    a = np.exp(-np.linspace(0, 3, n))
    candidates.append(a / a.sum())
    
    # Power law decreasing
    idx = np.arange(1, n + 1)
    a = 1.0 / idx
    candidates.append(a / a.sum())
    
    # Power law with different exponent
    a = 1.0 / idx**1.5
    candidates.append(a / a.sum())
    
    # Asymmetric Gaussian (shifted)
    x = np.linspace(0, 1, n)
    for mu in [0.3, 0.4, 0.5]:
        a = np.exp(-2 * ((x - mu) / 0.3) ** 2)
        candidates.append(a / a.sum())
    
    # Sigmoid-like
    a = 1.0 / (1.0 + np.exp(5 * (x - 0.4)))
    a = np.maximum(a, 1e-10)
    candidates.append(a / a.sum())
    
    # Beta distribution shaped
    from scipy.stats import beta as beta_dist
    for alpha, beta_p in [(5, 1), (8, 1), (3, 1), (10, 2)]:
        a = beta_dist.pdf(x, alpha, beta_p)
        a = np.maximum(a, 1e-10)
        candidates.append(a / a.sum())
    
    # First element large, rest spread
    a = np.ones(n) * 0.01
    a[0] = 0.3
    candidates.append(a / a.sum())
    
    a = np.ones(n) * 0.01
    a[0] = 0.2
    a[1] = 0.1
    candidates.append(a / a.sum())
    
    best_a = None
    best_obj = float('inf')
    for a in candidates:
        obj = conv_max(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a
    
    return best_a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

            return float('inf')
