"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Diverse initialization with broad Gaussian shapes (sigma ~ 0.35-0.45 * n)
2. Adam smooth optimization with log-sum-exp surrogate and FFT gradients
3. LP-Frank-Wolfe refinement (primary workhorse, extended iterations)
4. Multiplicative weights / mirror descent for complementary descent
5. Peak-targeted coordinate descent for final refinement
6. Multi-length exploration (n=1024 primary, n=1536, n=512)

n=1024 primary length, and multiplicative weights as new method.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(f, tau):
    """
    Compute gradient of F(a) = log(sum_k exp((a*a)[k]/tau)).
    Returns (gradient vector, convolution).
    """
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()

    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    h = conv_result[n - 1:2 * n - 1][::-1]

    grad = (2.0 / tau) * h
    return grad, conv


def adam_smooth_optimize(f0, time_limit, tau_init=None, lr_init=0.004, verbose=False):
    """Adam on smooth surrogate log-sum-exp."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    if tau_init is None:
        tau = 0.06 * conv.max()
    else:
        tau = tau_init

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    lr = lr_init
    t = 0

    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        t += 1
        grad, conv = compute_smooth_grad(f, tau)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        tau *= 0.9995
        if tau < 1e-7:
            tau = 1e-7

        if no_improve > 80:
            lr *= 0.9
        if no_improve > 350:
            rng = np.random.default_rng(t * 13 + 7)
            noise_scale = 0.002 + 0.003 * min(no_improve / 1000.0, 3.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = lr_init * 0.7
            tau = 0.04 * fast_convolve(f).max()
            if verbose:
                print(f"    Adam restart at iter {t}: obj={2*n*best_obj:.8f}")

        if verbose and t % 200 == 0:
            print(f"    Adam iter {t}: obj={2*n*best_obj:.8f}, tau={tau:.2e}")

    return best_f, 2 * n * best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        row_start = 0
        row_end = min(n - 1, j)
        col_start = max(0, j)
        col_end = min(2 * n - 2, j + n - 1)
        # (f*g)[k] = sum_j f[j] * g[k-j], so A[k,j] = f[j] if 0<=k-j<n
        # i.e., k-j in [0,n-1], so k in [j, j+n-1]
        for k in range(j, min(2 * n - 1, j + n)):
            A_ub[k, j] = f[j]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def solve_lp_sparse(f, M, n):
    """Faster LP using scipy sparse matrices."""
    from scipy.sparse import lil_matrix
    c = -np.ones(n)
    A_ub = lil_matrix((2 * n - 1, n))
    for j in range(n):
        k_start = max(0, j)
        k_end = min(2 * n - 1, j + n)
        for k in range(k_start, k_end):
            A_ub[k, j] = f[j]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive relaxation and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0

        g = solve_lp_sparse(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.15
            if m_exp > 8.0:
                rng = np.random.default_rng(it * 17 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.1
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 30:
                m_exp = min(m_exp * 1.1, 5.0)
            if no_improve > 100:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue
        else:
            no_improve = 0

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0 and best_alpha < 1.0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = np.maximum(f + mid * (g - f), 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = np.maximum(f + best_alpha * (g - f), 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 50 == 0:
                    print(f"    LP-FW iter {it}: obj={2*n*best_obj:.8f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 5.0)

        if no_improve > 200:
            break

    return best_f, 2 * n * best_obj


def multiplicative_weights_optimize(f0, time_limit, verbose=False):
    """
    Multiplicative weights / mirror descent approach.
    Alternates between:
    - Inner: projected gradient on a^T Q(p) a (QP with p-weighted quadratic)
    - Outer: update p via multiplicative weights on conv values
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    p = np.ones(2 * n - 1) / (2 * n - 1)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    lr_p = 0.2
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Inner: minimize a^T Q(p) a via few projected gradient steps
        a_inner = f.copy()
        inner_iters = max(5, min(30, 500 // (it + 1)))
        inner_lr = 1.0 / (n * 0.5)

        for _ in range(inner_iters):
            # grad of a^T Q(p) a: (Q(p)*a)[j] = sum_l p[j+l] * a[l] (with bounds)
            p_rev = p[::-1]
            total_len = n + len(p) - 1
            fft_size = 1 << (total_len - 1).bit_length()
            A = np.fft.rfft(a_inner, fft_size)
            P = np.fft.rfft(p_rev, fft_size)
            conv_r = np.fft.irfft(A * P, fft_size)
            grad = conv_r[n - 1:2 * n - 1][::-1]
            grad = 2.0 * grad

            a_inner = a_inner - inner_lr * grad
            a_inner = project_simplex(a_inner)

        f = a_inner
        conv = fast_convolve(f)
        M = conv.max()

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            if verbose and it % 50 == 0:
                print(f"    MW iter {it}: obj={2*n*best_obj:.8f}")

        # Outer: multiplicative weight update
        scale = 0.02 + 0.03 * min(it / 200.0, 1.0)
        scaled = conv / (scale * M + 1e-15)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        p = (1 - lr_p) * p + lr_p * w
        p = np.maximum(p, 1e-30)
        p /= p.sum()

        if it % 80 == 0:
            lr_p = max(0.05, lr_p * 0.95)

    return best_f, 2 * n * best_obj


def peak_targeted_cd(f0, time_limit, verbose=False):
    """
    Peak-targeted coordinate descent.
    Identifies the peak of the convolution and the pair contributing most,
    then redistributes mass to reduce the peak.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        # Find all near-peak positions
        threshold = M * 0.95
        peak_indices = np.where(conv >= threshold)[0]

        for pk in peak_indices[:8]:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[pk - idx]
            j_max = idx[np.argmax(products)]
            j_other = pk - j_max

            if j_max == j_other:
                old_val = f[j_max]
                if old_val < 1e-10:
                    continue
                for factor in [0.5, 0.7, 0.85]:
                    trial = f.copy()
                    trial[j_max] *= factor
                    diff = old_val - trial[j_max]
                    # Redistribute to low-contribution elements
                    contrib_mask = np.ones(n, dtype=bool)
                    for pk2 in peak_indices[:4]:
                        pk2 = int(pk2)
                        js2 = max(0, pk2 - n + 1)
                        je2 = min(n - 1, pk2)
                        contrib_mask[js2:je2 + 1] = False
                    if contrib_mask.sum() > 0 and f[contrib_mask].sum() > 0:
                        trial[contrib_mask] += diff * f[contrib_mask] / f[contrib_mask].sum()
                    trial = np.maximum(trial, 0)
                    trial /= trial.sum()
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break
            else:
                old_jm = f[j_max]
                old_jo = f[j_other]
                for scale in [0.5, 0.7, 0.85]:
                    trial = f.copy()
                    trial[j_max] = old_jm * scale
                    diff = old_jm * (1 - scale)
                    mask = np.ones(n, dtype=bool)
                    mask[j_max] = False
                    mask[j_other] = False
                    if mask.sum() > 0 and f[mask].sum() > 0:
                        trial[mask] += diff * f[mask] / f[mask].sum()
                    trial = np.maximum(trial, 0)
                    trial /= trial.sum()
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break

        if not improved:
            no_improve += 1
            if no_improve > 20:
                rng = np.random.default_rng(it * 7 + int(time.monotonic() * 100))
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                no_improve = 0
        else:
            no_improve = 0

        if verbose and it % 100 == 0:
            print(f"    PeakCD iter {it}: obj={2*n*best_obj:.8f}")

    return best_f, 2 * n * best_obj
