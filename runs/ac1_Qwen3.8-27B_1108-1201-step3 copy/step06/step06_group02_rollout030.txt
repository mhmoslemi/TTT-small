"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-strategy optimization:

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def objective_normalized(a):
    """Compute 2*n*max(conv)/sum^2 assuming sum(a)=1."""
    conv = fast_convolve(a)
    return float(2.0 * len(a) * conv.max())


def project_nonneg_sum1(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(len(v)) / len(v)
    return v / s


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-like initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    """Bell-shaped with flat top."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine_mod(n, freq, amp):
    """Cosine-modulated uniform."""
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_multi_cosine(n, n_freqs=5, max_amp=0.25, seed=0):
    """Multi-frequency cosine modulation."""
    rng = np.random.default_rng(seed)
    a = np.ones(n)
    for m in range(1, n_freqs + 1):
        amp = rng.uniform(0.05, max_amp)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * m * np.arange(n) / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_power_law(n, p=0.5):
    """Power-law shaped sequence."""
    x = np.arange(n) - n / 2.0
    a = 1.0 / (1.0 + (0.5 * np.abs(x) / n) ** (2.0 * p))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate to new length, normalize to sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def gerchberg_saxton(a0, max_iter=2000, tol=1e-10):
    """
    Spectral flattening via Gerchberg-Saxton iterations.
    Alternates between:
    - Time domain: enforce non-negativity and sum=1
    - Frequency domain: flatten |FFT|^2
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    fft_a = np.fft.rfft(a, fft_size)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    for it in range(max_iter):
        # Frequency domain: flatten magnitude
        mag = np.abs(fft_a)
        phase = np.angle(fft_a)
        # Target: flat magnitude (geometric mean)
        target_mag = np.full_like(mag, np.median(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0)
        # Blend: keep some of original to avoid oscillation
        new_mag = 0.3 * mag + 0.7 * target_mag
        fft_a = new_mag * np.exp(1j * phase)

        # Time domain: inverse FFT, enforce non-negativity
        a = np.fft.irfft(fft_a, fft_size)[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a /= s

        # Update FFT for next iteration
        fft_a = np.fft.rfft(a, fft_size)

        # Check objective
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

    return best_a


def solve_lp_top_peaks(f, M, peak_indices, n):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in peak_indices, g >= 0.
    Uses sparse structure for efficiency.
    """
    n_vars = n
    n_cons = len(peak_indices)

    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n_vars)

    # Constraints: (f*g)[k] <= M for each peak k
    # (f*g)[k] = sum_j f[j]*g[k-j] where 0<=j<n and 0<=k-j<n
    # So (f*g)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[j]*g[k-j]
    # Let i = k-j, so i ranges from max(0,k-n+1) to min(n-1,k), and j = k-i
    # (f*g)[k] = sum_i f[k-i]*g[i] for valid i
    # So row k of A has A[k, i] = f[k-i] for valid i

    A_ub = np.zeros((n_cons, n_vars))
    b_ub = np.full(n_cons, M)

    for row, k in enumerate(peak_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            i = k - j  # index of g
            A_ub[row, i] = f[j]

    bounds = [(0, None)] * n_vars
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

    best_obj = M

    it = 0

    m_idx = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1.0:
            break

        top_k = min(n_top, len(conv))
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        peak_indices = sorted(peak_indices.tolist())

        # Try LP with escalating M levels

        for level_try in range(m_idx, len(m_levels)):
            M_try = conv.max() * m_levels[level_try]
            g = solve_lp_top_peaks(f, M_try, peak_indices, n)
            if g is not None and g.sum() > 1e-12:
                m_idx = level_try
                break
            m_idx = level_try + 1

        if g is None or g.sum() < 1e-12:
            # Perturb and retry

            conv = fast_convolve(f)
            m_idx = 0
            no_improve += 1
            if no_improve > 100:
                break
            continue

        g = g / g.sum()
        diff = np.max(np.abs(g - f))
        if diff < 1e-12:
            no_improve += 1
            m_idx = min(m_idx + 1, len(m_levels) - 1)
            if no_improve > 80:
                break
            continue
        no_improve = 0

        # Line search: coarse then fine
        best_alpha = 0.0
        best_val = conv.max()
        alphas = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]

            ft = f + alpha * (g - f)

            if val < best_val:

        if 0 < best_alpha < 1.0:
            idx = alphas.index(best_alpha)
            lo = alphas[max(0, idx - 1)]
            hi = alphas[min(len(alphas) - 1, idx + 1)]
            for _ in range(10):
                a1 = hi - 0.618 * (hi - lo)
                a2 = lo + 0.618 * (hi - lo)

                ft2 = project_nonneg_sum1(f + a2 * (g - f))

                if v1 < v2:

                    if v1 < best_val:

                        best_alpha = a1
                else:
                    lo = a1
                    if v2 < best_val:
                        best_val = v2
                        best_alpha = a2

            m_idx = min(m_idx + 1, len(m_levels) - 1)

        f_new = project_nonneg_sum1(f + best_alpha * (g - f))

def subgradient_momentum(a0, time_limit, verbose=False):
    """
    Subgradient method with momentum for minimizing max(conv(a,a)).
    """
    a = project_nonneg_sum1(a0.copy())

    best_a = a.copy()

    lr = 0.01
    beta = 0.9
    top_k = 30
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1:
            break

        conv = fast_convolve(a)
        M = conv.max()

        # Get top peaks for smoothed subgradient
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        # Softmax weighting
        scaled = pv / (0.01 * M + 1e-12)
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute smoothed subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-10:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)

            for j in range(j_start, j_end + 1):
                idx = k - j
                sg[idx] += 2.0 * w[i] * a[j]

        # Momentum update
        momentum = beta * momentum + (1 - beta) * sg
        a_new = a - lr * momentum
        a_new = project_nonneg_sum1(a_new)

        obj_new = fast_convolve(a_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new

            a = a_new
            no_improve = 0
            if verbose and it % 100 == 0:

            a = a_new  # still accept for exploration
            no_improve += 1

        # Adaptive learning rate
        if no_improve > 50:
            lr *= 0.8

                # Restart from best with perturbation
                rng = np.random.default_rng(it * 31 + 7)
                a = project_nonneg_sum1(best_a + rng.standard_normal(n) * 0.01)
                momentum = np.zeros(n)
                lr = 0.01
                no_improve = 0

def peak_shaving(a0, time_limit, verbose=False):
    """
    Iterative peak shaving: directly reduce the largest convolution peak
    by optimally redistributing mass.
    """
    a = project_nonneg_sum1(a0.copy())
    n = len(a)
    t0 = time.monotonic()

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1:
            break
        if no_improve > 200:
            break

        conv = fast_convolve(a)
        M = conv.max()
        peak_k = int(np.argmax(conv))

        # Find contributing pairs
        j_start = max(0, peak_k - n + 1)
        j_end = min(n - 1, peak_k)
        contrib_j = np.arange(j_start, j_end + 1)
        if len(contrib_j) == 0:
            break

        # Products a[j] * a[peak_k - j]
        products = a[contrib_j] * a[peak_k - contrib_j]
        total = products.sum()
        if total < 1e-15:
            break

        # Find the most contributing coordinate pair
        max_idx = np.argmax(products)
        j_star = int(contrib_j[max_idx])
        partner = peak_k - j_star

        # Source: reduce the larger of the pair
        if a[j_star] >= a[partner]:
            source = j_star
        else:
            source = partner

        # Target: distribute to non-contributing coordinates
        non_contrib = np.ones(n, dtype=bool)
        non_contrib[contrib_j] = False
        if non_contrib.sum() < 3:
            non_contrib[:] = True
            non_contrib[source] = False

        # Try different fractions
        best_trial = None
        best_trial_obj = M
        nc_vals = a[non_contrib]
        nc_sum = nc_vals.sum()
        if nc_sum < 1e-15:
            no_improve += 1
            continue

        for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            diff = a[source] * frac
            if diff < 1e-15:
                continue
            trial = a.copy()
            trial[source] -= diff
            trial[non_contrib] += diff * nc_vals / nc_sum
            trial = project_nonneg_sum1(trial)
            obj = fast_convolve(trial).max()
            if obj < best_trial_obj - 1e-15:
                best_trial_obj = obj
                best_trial = trial

        if best_trial is not None:
            a = best_trial
            if best_trial_obj < best_obj - 1e-14:
                best_obj = best_trial_obj
                best_a = a.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"      PS iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            # Try second peak
            peak_vals = np.sort(conv)[::-1]
            if len(peak_vals) > 1 and peak_vals[1] > 0.9 * peak_vals[0]:
                peak_k2 = int(np.argsort(conv)[-2])
                if peak_k2 != peak_k:
                    j_start2 = max(0, peak_k2 - n + 1)
                    j_end2 = min(n - 1, peak_k2)
                    cj2 = np.arange(j_start2, j_end2 + 1)
                    prods2 = a[cj2] * a[peak_k2 - cj2]
                    if len(cj2) > 0:
                        mi2 = int(np.argmax(prods2))
                        js2 = int(cj2[mi2])
                        src2 = js2 if a[js2] >= a[peak_k2 - js2] else (peak_k2 - js2)
                        nc2 = np.ones(n, dtype=bool)
                        nc2[cj2] = False
                        if nc2.sum() >= 2:
                            nc2_vals = a[nc2]
                            nc2_sum = nc2_vals.sum()
                            if nc2_sum > 1e-15:
                                for frac2 in [0.1, 0.2, 0.3]:
                                    diff2 = a[src2] * frac2
                                    trial = a.copy()
                                    trial[src2] -= diff2
                                    trial[nc2] += diff2 * nc2_vals / nc2_sum
                                    trial = project_nonneg_sum1(trial)
                                    obj2 = fast_convolve(trial).max()
                                    if obj2 < best_obj - 1e-14:
                                        a = trial
                                        best_obj = obj2
                                        best_a = a.copy()
                                        no_improve = 0
                                        break
            no_improve += 1

    return best_a, best_obj

        j_start = max(0, peak_k - n + 1)
        j_end = min(n - 1, peak_k)
        contrib_j = np.arange(j_start, j_end + 1)
        if len(contrib_j) == 0:
            break
        products = a[contrib_j] * a[peak_k - contrib_j]
        center = int(contrib_j[np.argmax(products)])

        # Define block
        b_start = max(0, center - block_size // 2)
        b_end = min(n, b_start + block_size)
        block = np.arange(b_start, b_end)
        n_block = len(block)

        if n_block < 2:
            no_improve += 1
            continue

        # Optimize within the block: try transferring mass from high to low
        # within the block, checking if it reduces the peak
        a_block = a[block].copy()
        best_block = a_block.copy()
        best_block_obj = M

        # Try: reduce the max in block, add to min in block
        for _ in range(5):
            if a_block.max() - a_block.min() < 1e-10:
                break
            max_in_block = np.argmax(a_block)
            min_in_block = np.argmin(a_block)
            diff = (a_block[max_in_block] - a_block[min_in_block]) * 0.1
            if diff < 1e-12:
                break
            trial = a.copy()
            trial[block[max_in_block]] -= diff
            trial[block[min_in_block]] += diff
            trial = project_nonneg_sum1(trial)
            obj = fast_convolve(trial).max()
            if obj < best_block_obj - 1e-15:
                best_block_obj = obj
                best_block = trial[block].copy()

        if best_block_obj < M - 1e-14:
            a = project_nonneg_sum1(a)
            a[block] = best_block
            a = project_nonneg_sum1(a)
            new_obj = fast_convolve(a).max()
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_a = a.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"      BCD iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Occasionally try a random block
        if it % 10 == 0:
            rng = np.random.default_rng(it * 13 + 1)
            b_start = rng.integers(0, max(1, n - block_size))
            block = np.arange(b_start, b_start + block_size)
            # Equalize within block (smoothing)
            mean_val = a[block].mean()
            trial = a.copy()
            trial[block] = 0.7 * trial[block] + 0.3 * mean_val
            trial = project_nonneg_sum1(trial)
            obj = fast_convolve(trial).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = trial.copy()
                a = trial
                no_improve = 0

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization for minimizing the auto-convolution peak ratio.
    """
    rng = np.random.default_rng(seed)
    total_budget = max(1.0, float(budget_s))
    deadline = time.monotonic() + total_budget - 30  # Leave 30s buffer

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768, 0.3)
        try_seq(parent, "default")

    # === Phase 1: Generate diverse initializations across multiple lengths ===
    print("\n=== Phase 1: Initializations ===")
    t0 = time.monotonic()

    init_candidates = []
    n_values = [512, 768, 1024, 1536, 2048]

    for n_try in n_values:
        init_candidates.append((n_try, make_gaussian(n_try, 0.25)))
        init_candidates.append((n_try, make_gaussian(n_try, 0.35)))
        init_candidates.append((n_try, make_bell(n_try, 0.5)))
        init_candidates.append((n_try, make_bell(n_try, 0.7)))
        init_candidates.append((n_try, make_cosine_mod(n_try, 1, 0.15)))
        init_candidates.append((n_try, make_multi_cosine(n_try, 4, 0.2, seed=n_try % 100)))
        init_candidates.append((n_try, make_multi_cosine(n_try, 7, 0.15, seed=n_try % 200 + 50)))
        init_candidates.append((n_try, make_power_law(n_try, 0.5)))
        init_candidates.append((n_try, make_power_law(n_try, 1.0)))

    # Add parent and interpolated versions
    if parent is not None:
        init_candidates.append((len(parent), parent.copy()))
        for n_try in [512, 1024, 1536, 2048]:
            if n_try != len(parent):
                init_candidates.append((n_try, interpolate_sequence(parent, n_try)))

    # Score all initializations
    scored_inits = []
    for n_try, a_init in init_candidates:
        sc = objective_normalized(a_init)
        scored_inits.append((sc, a_init, n_try))
    scored_inits.sort(key=lambda x: x[0])

    # Evaluate top initializations with full evaluator
    for sc, a_init, n_try in scored_inits[:10]:
        try_seq(a_init, f"init_n{n_try}")

    print(f"  Best init obj: {scored_inits[0][0]:.6f} (n={scored_inits[0][2]})")
    print(f"  Phase 1 time: {time.monotonic()-t0:.1f}s")

    # === Phase 2: Gerchberg-Saxton spectral optimization on top inits ===
    print("\n=== Phase 2: Spectral Optimization ===")
    for sc, a_init, n_try in scored_inits[:5]:
        rem = deadline - time.monotonic() - 600
        if rem < 10:
            break
        if n_try > 2048:
            continue
        t_alloc = min(rem * 0.3, 30.0)
        a_gs = gerchberg_saxton(a_init, max_iter=int(t_alloc * 100))
        try_seq(a_gs, f"gs_n{n_try}")

    # === Phase 3: LP-FW on best candidates ===
    print("\n=== Phase 3: LP-Frank-Wolfe ===")

    # Select top candidates for LP-FW
    lp_candidates = []
    seen_n = set()
    for sc, a_init, n_try in scored_inits:
        if n_try not in seen_n and n_try <= 2048:
            lp_candidates.append((sc, a_init, n_try))
            seen_n.add(n_try)
        if len(lp_candidates) >= 5:
            break

    # Also add GS results if they improved
    if best_seq is not None and len(best_seq) not in seen_n:
        lp_candidates.append((objective_normalized(best_seq), best_seq.copy(), len(best_seq)))

    for sc, a_init, n_try in lp_candidates:
        rem = deadline - time.monotonic() - 400
        if rem < 30:
            break
        t_alloc = min(rem * 0.4, 150.0)
        n_top = 50 if n_try <= 1024 else 30
        print(f"  LP-FW n={n_try}: alloc={t_alloc:.0f}s (init_obj={sc:.6f})")
        a_opt, obj = lp_fw_optimize(a_init, time_limit=t_alloc, n_top=n_top, verbose=True)
        try_seq(a_opt, f"lpfw_n{n_try}")

    # === Phase 4: Subgradient with momentum ===
    print("\n=== Phase 4: Subgradient Momentum ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_alloc = min(rem * 0.5, 120.0)
            n_best = len(best_seq)
            print(f"  SG n={n_best}: alloc={t_alloc:.0f}s")
            a_opt, obj = subgradient_momentum(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, f"sg_n{n_best}")

    # === Phase 5: Peak shaving ===
    print("\n=== Phase 5: Peak Shaving ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_alloc = min(rem * 0.6, 100.0)
            n_best = len(best_seq)
            print(f"  PS n={n_best}: alloc={t_alloc:.0f}s")
            a_opt, obj = peak_shaving(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, f"ps_n{n_best}")

    # === Phase 6: Block coordinate descent ===
    print("\n=== Phase 6: Block Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_alloc = min(rem * 0.6, 80.0)
            n_best = len(best_seq)
            print(f"  BCD n={n_best}: alloc={t_alloc:.0f}s")
            a_opt, obj = block_coordinate_descent(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, f"bcd_n{n_best}")

    # === Phase 7: Second round of LP-FW on best ===
    print("\n=== Phase 7: LP-FW Round 2 ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_alloc = min(rem * 0.5, 80.0)
            n_best = len(best_seq)
            n_top = 50 if n_best <= 1024 else 30
            print(f"  LP-FW r2 n={n_best}: alloc={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, n_top=n_top, verbose=True)
            try_seq(a_opt, f"lpfw2_n{n_best}")

    # === Phase 8: Multi-length exploration from best ===
    print("\n=== Phase 8: Multi-Length Exploration ===")
    if best_seq is not None:
        base_n = len(best_seq)
        target_ns = [n for n in [512, 768, 1024, 1536, 2048] if n != base_n]
        for target_n in target_ns:
            rem = deadline - time.monotonic() - 60
            if rem < 15:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")
            if sc < best_score + 0.02:
                t_alloc = min(rem * 0.3, 40.0)
                if t_alloc > 10:
                    n_top = 40 if target_n <= 1024 else 25
                    a_opt, obj = lp_fw_optimize(f_interp, time_limit=t_alloc, n_top=n_top)
                    try_seq(a_opt, f"lpfw_interp_n{target_n}")

    # === Phase 9: Final polish ===
    print("\n=== Phase 9: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_alloc = min(rem * 0.4, 30.0)
            a_opt, obj = peak_shaving(best_seq, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, "final_ps")

        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t_alloc = min(rem * 0.5, 20.0)
            n_best = len(best_seq)
            n_top = 40 if n_best <= 1024 else 25
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, n_top=n_top)
            try_seq(a_opt, "final_lpfw")

        rem = deadline - time.monotonic() - 5
        if rem > 5:
            t_alloc = min(rem, 10.0)
            a_opt, obj = subgradient_momentum(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_sg")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"Total time: {time.monotonic() - (deadline - (total_budget - 30)):.1f}s")

    # Scale up to avoid the sum < 0.01 check (shouldn't be needed but safe)
    result = best_seq.copy()
    s = result.sum()
    if s < 0.1:
        result = result * (1.0 / s)

    return [float(x) for x in result]
