"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LP-based Frank-Wolfe with constraint reduction
2. Log-sum-exp smoothed gradient descent with adaptive step sizes
3. Multi-length exploration with diverse initializations
4. Simulated annealing restarts for local minima escape

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_reduced(f, M, n, k_active=50):
    """Solve reduced LP: max sum(g) s.t. (f*g)[k] <= M for active k, g >= 0."""
    c_conv = fast_convolve(f)
    # Find the most active constraints
    if k_active < len(c_conv):
        active_idx = np.argpartition(c_conv, -k_active)[-k_active:]
    else:
        active_idx = np.arange(len(c_conv))
    
    # Only include constraints that are close to active
    threshold = M * 0.85
    active_mask = c_conv[active_idx] > threshold
    active_idx = active_idx[active_mask]
    
    if len(active_idx) == 0:
        active_idx = np.array([np.argmax(c_conv)])
    
    c_obj = -np.ones(n)
    A_ub = np.zeros((len(active_idx), n))
    for i, k in enumerate(active_idx):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # (f*g)[k] = sum_j f[j] * g[k-j]
        # For fixed k, g appears with coefficient f[k-j] for j in [js, je]
        # A[i, j] = f[k-j] for j in [js, je]
        for j in range(js, je + 1):
            A_ub[i, j] = f[k - j]
    b_ub = np.full(len(active_idx), M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def lp_fw_fast(f0, time_limit, k_active=50, verbose=False):
    """Fast LP-based Frank-Wolfe with constraint reduction."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    c = fast_convolve(f)
    M = c.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        g = solve_lp_reduced(f, M * m_exp, n, k_active=k_active)
        if g is None:
            m_exp *= 1.15
            if m_exp > 5.0:
                # Random restart
                rng = np.random.default_rng(it * 19 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                c = fast_convolve(f)
                M = c.max()
                m_exp = 1.0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.1
            continue
        g = g / g_sum
        
        # Check fixed point
        diff = np.max(np.abs(g - f))
        if diff < 1e-10:
            no_improve += 1
            if no_improve > 30:
                m_exp = min(m_exp * 1.05, 4.0)
            if no_improve > 100:
                rng = np.random.default_rng(it * 23 + 11)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                c = fast_convolve(f)
                M = c.max()
                m_exp = 1.0
                no_improve = 0
            continue
        else:
            if diff < 1e-7:
                no_improve += 1
            else:
                no_improve = 0
        
        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = np.linspace(0.05, 1.0, 20)
        for alpha in alphas:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            c = fast_convolve(f_new)
            M = c.max()
            f = f_new
            m_exp = 1.0
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 50 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.12f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.03, 4.0)
        
        if no_improve > 300:
            break
    
    return best_f, best_obj

def gradient_descent_lse(f0, time_limit, verbose=False):
    """Log-sum-exp smoothed gradient descent with adaptive step sizes."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    c = fast_convolve(f)
    best_obj = c.max()
    
    t0 = time.monotonic()
    momentum = np.zeros(n)
    mu = 0.85
    
    beta_schedule = [4, 8, 16, 32, 64, 128, 256, 512, 1024]
    
    for beta in beta_schedule:
        if time.monotonic() - t0 > time_limit * 0.9:
            break
        
        it = 0
        no_improve = 0
        lr = 0.5 / (beta * np.max(fast_convolve(f)) * n ** 0.5 + 1e-10)
        
        stage_start = time.monotonic()
        stage_time = min(time_limit * 0.06, (time_limit - (time.monotonic() - t0)) * 0.12)
        
        while time.monotonic() - stage_start < stage_time and time.monotonic() - t0 < time_limit * 0.9:
            it += 1
            c = fast_convolve(f)
            max_c = c.max()
            
            # Softmax weights
            scaled = beta * (c - max_c)
            exp_scaled = np.exp(scaled)
            w = exp_scaled / exp_scaled.sum()
            
            # Gradient
            w_rev = w[::-1].copy()
            conv_len = n + len(w_rev) - 1
            fft_size = 1 << (conv_len - 1).bit_length()
            F = np.fft.rfft(f, fft_size)
            W = np.fft.rfft(w_rev, fft_size)
            conv = np.fft.irfft(F * W, fft_size)[:conv_len]
            indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
            g = 2.0 * conv[indices]
            
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-15:
                break
            
            lr = min(lr, 0.5 / (g_norm + 1e-10))
            momentum = mu * momentum - lr * g
            f_trial = f + momentum
            f_trial = project_simplex(f_trial)
            
            c_trial = fast_convolve(f_trial)
            obj_trial = c_trial.max()
            
            if obj_trial <= c.max() + 1e-12:
                f = f_trial
                if obj_trial < best_obj - 1e-13:
                    best_obj = obj_trial
                    best_f = f.copy()
                    no_improve = 0
                    lr = min(lr * 1.03, 1.0 / (g_norm * 0.3 + 1e-10))
                else:
                    no_improve += 1
            else:
                no_improve += 1
                lr *= 0.6
                momentum = mu * momentum
            
            if no_improve > 60:
                rng = np.random.default_rng(it * 13 + int(beta) * 7)
                noise_scale = 0.003 / np.sqrt(beta / 4.0)
                f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
                momentum = np.zeros(n)
                lr = 0.5 / (beta * 2.0 * n ** 0.5 + 1e-10)
                no_improve = 0
        
        if verbose:
            print(f"  GD LSE beta={beta}: best={best_obj:.10f}")
    
    return best_f, best_obj

def make_tilted_gaussian(n, center_frac=0.4, sigma_frac=0.3, tilt=0.3):
    """Gaussian with a linear tilt to break symmetry."""
    x = np.arange(n)
    center = n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2) * np.exp(tilt * (x - center) / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_hann(n):

    s = a.sum()

    return a

def make_blackman(n):

def make_kaiser(n, beta=4.0):
    """Kaiser window normalized to sum to 1."""
    from scipy.special import i0
    x = np.linspace(-1, 1, n)
    a = i0(beta * np.sqrt(1 - x**2)) / i0(beta)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast constraint-reduced LP-based Frank-Wolfe (key improvement over prior)
2. Log-sum-exp smoothed gradient descent with adaptive step sizes
3. Multi-length exploration with diverse window initializations
4. Continuation strategy for LP-FW (loose-to-tight M)
5. Final full-constraint LP polish for precision

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_reduced(f, M, n, k_active=60):
    """Solve reduced LP: max sum(g) s.t. (f*g)[k] <= M for active constraints, g>=0.
    Only includes constraints where conv value is close to M."""
    c_conv = fast_convolve(f)
    total_constraints = len(c_conv)
    if k_active < total_constraints:
        active_idx = np.argpartition(c_conv, -k_active)[-k_active:]
    else:
        active_idx = np.arange(total_constraints)

    threshold = M * 0.75
    active_mask = c_conv[active_idx] > threshold
    active_idx = active_idx[active_mask]

    if len(active_idx) == 0:
        active_idx = np.array([int(np.argmax(c_conv))])

    c_obj = -np.ones(n)
    A_ub = np.zeros((len(active_idx), n))
    for i, k in enumerate(active_idx):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            indices = np.arange(js, je + 1)
            A_ub[i, indices] = f[k - indices]
    b_ub = np.full(len(active_idx), M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def solve_lp_full(f, M, n):

    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    return None

def gradient_descent_lse(f0, time_limit, verbose=False):
    """Log-sum-exp smoothed gradient descent with adaptive step sizes and momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    c = fast_convolve(f)
    best_obj = c.max()

    t0 = time.monotonic()
    momentum = np.zeros(n)
    mu = 0.85

    beta_schedule = [4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048]

    for beta in beta_schedule:
        if time.monotonic() - t0 > time_limit * 0.92:
            break

        it = 0
        no_improve = 0
        lr = 0.5 / (beta * max(np.max(fast_convolve(f)), 1e-10) * np.sqrt(n) + 1e-10)
        stage_start = time.monotonic()
        stage_time = min(time_limit * 0.05, max(5.0, (time_limit - (time.monotonic() - t0)) * 0.1))

        while time.monotonic() - stage_start < stage_time and time.monotonic() - t0 < time_limit * 0.92:
            it += 1
            c = fast_convolve(f)
            max_c = c.max()

            scaled = beta * (c - max_c)
            exp_scaled = np.exp(scaled)
            w = exp_scaled / exp_scaled.sum()

            w_rev = w[::-1].copy()
            conv_len = n + len(w_rev) - 1
            fft_size = 1 << (conv_len - 1).bit_length()
            F = np.fft.rfft(f, fft_size)
            W = np.fft.rfft(w_rev, fft_size)
            conv = np.fft.irfft(F * W, fft_size)[:conv_len]
            indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
            g = 2.0 * conv[indices]

            g_norm = np.linalg.norm(g)
            if g_norm < 1e-15:
                break

            lr = min(lr, 0.5 / (g_norm + 1e-10))
            momentum = mu * momentum - lr * g
            f_trial = f + momentum
            f_trial = project_simplex(f_trial)

            c_trial = fast_convolve(f_trial)
            obj_trial = c_trial.max()

            if obj_trial <= c.max() + 1e-12:
                f = f_trial
                if obj_trial < best_obj - 1e-13:
                    best_obj = obj_trial
                    best_f = f.copy()
                    no_improve = 0
                    lr = min(lr * 1.04, 1.0 / (g_norm * 0.3 + 1e-10))
                else:
                    no_improve += 1
            else:
                no_improve += 1
                lr *= 0.6
                momentum *= mu

            if no_improve > 50:
                rng = np.random.default_rng(it * 13 + int(beta) * 7)
                noise_scale = 0.003 / np.sqrt(max(beta / 4.0, 1.0))
                f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
                momentum = np.zeros(n)
                lr = 0.5 / (beta * 2.0 * np.sqrt(n) + 1e-10)
                no_improve = 0

        if verbose:
            print(f"  GD beta={beta}: best={best_obj:.10f}")

    return best_f, best_obj


def make_hann(n):
    a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_blackman(n):
    if n < 3:
        return np.ones(n) / n
    a = (0.42 - 0.5 * np.cos(2 * np.pi * np.arange(n) / (n - 1))
         + 0.08 * np.cos(4 * np.pi * np.arange(n) / (n - 1)))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_kaiser(n, beta=5.0):
    from scipy.special import i0
    x = np.linspace(-1, 1, n)
    a = i0(beta * np.sqrt(np.maximum(1 - x**2, 0))) / i0(beta)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_tilted_gaussian(n, center_frac=0.4, sigma_frac=0.3, tilt=0.4):
    x = np.arange(n)
    center = n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2) * np.exp(tilt * (x - center) / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_flat_cosine(n, power=2):
    x = np.linspace(-1, 1, n)
    a = np.cos(np.pi * x / 2.0) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    return f_interp
