def mwu_best_response(p, eta, n_iters=100):
    """
    Approximate best response to p using multiplicative weight updates.
    Returns q that approximately minimizes max_k (p*q)[k].
    """
    n = len(p)
    q = np.ones(n) / n
    
    for _ in range(n_iters):
        conv = fast_convolve(p, q)  # cross-correlation
        M = conv.max()
        k_peak = conv.argmax()
        # Gradient w.r.t. q: d/dq_i (p*q)[k_peak] = p[k_peak - i] (with appropriate indexing)
        # Actually, (p*q)[k] = sum_i p[i] * q[k-i] (with valid range)
        # So d/dq_j (p*q)[k] = p[k-j] if 0 <= k-j < n, else 0
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        grad = np.zeros(n)
        # For j in [js, je]: grad[j] = p[k_peak - j]
        grad[js:je+1] = p[k_peak - je:k_peak - js + 1][::-1]
        # Normalize gradient
        grad = grad / max(grad.max(), 1e-15)
        q = q * np.exp(-eta * grad)
        q = q / q.sum()
    
    return q

                grad[i] = np.dot(lam[k_lo:k_hi+1], a[k_lo-i:k_hi-i+1])
        
        # Adversary picks i* = argmin
        i_star = np.argmin(grad)
        
        # Update: lam[k] *= exp(eta * T[k, i_star])
        # T[k, i_star] = a[k - i_star] if 0 <= k - i_star < n
        k_lo = i_star
        k_hi = min(m - 1, i_star + n - 1)
        lam[k_lo:k_hi+1] *= np.exp(eta * a[k_lo - i_star:k_hi - i_star + 1])
        lam = lam / lam.sum()
    
    # Convert dual to primal: b[i] proportional to (T^T @ lam)[i] - nu
    # At optimum: b[i] > 0 implies (T^T @ lam)[i] = nu
    # For MWU, the primal is approximately: b[i] = max(0, grad[i] - nu)
    # Use the fact that at the LP optimum, b is supported on columns where (T^T lam)_i = nu
    # Approximate: b[i] = max(0, grad[i] - min(grad)) / sum
    
    # Actually, for the primal-weak duality, the best primal solution is:
    # b[i] proportional to exp(lam_xt * T[:,i]) ... no
    # Let's just use: the "best response" direction is given by the gradient
    # b[i] = max(0, grad[i] - threshold) where threshold = min(grad)
    threshold = grad.min()
    b = np.maximum(grad - threshold, 0)
    bsum = b.sum()
    if bsum < 1e-15:
        b = np.ones(n) / n
    else:
        b = b / bsum
    
    return b, lam


def mwu_fictitious_play(a0, time_limit, eta=None, verbose=True):
    """
    Fictitious play with MWU best responses.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    b = a.copy()
    a_avg = a.copy()
    b_avg = b.copy()
    
    if eta is None:
        eta = 1.0 / np.sqrt(np.log(2 * n - 1))
    
    best_score = score_normalized(a)
    best_a = a.copy()
    
    t0 = time.monotonic()
    iters = 0
    mwu_iters = 20  # MWU iterations per best response
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        
        # MWU best response for b given a
        b_new, _ = mwu_best_response(a, eta, mwu_iters)
        b = b_new
        
        # MWU best response for a given b
        a_new, _ = mwu_best_response(b, eta, mwu_iters)
        a = a_new
        
        # Update averages
        a_avg = (iters * a_avg + a) / (iters + 1)
        b_avg = (iters * b_avg + b) / (iters + 1)
        
        # Check candidates
        candidates = [a, b, 0.5*(a+b), a_avg, 0.5*(a_avg+b_avg)]
        for cand in candidates:
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = score_normalized(cand)
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
        
        # Adapt eta (decrease over time for convergence)
        eta *= 0.999
        mwu_iters = min(50, 20 + iters // 10)
        
        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MWU-FP iter={iters} ({elapsed:.0f}s): score={best_score:.6f} eta={eta:.4f}")
    
    return best_a, best_score


def soft_max_lbgfs(a0, time_limit, tau_schedule=None, verbose=True):
    """
    L-BFGS optimization with adaptive soft-max smoothing.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = score_normalized(a)
    
    # Try multiple temperature schedules
    if tau_schedule is None:
        tau_schedule = [100, 200, 500, 1000, 2000]
    
    for tau_val in tau_schedule:
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        # Work in log-space for non-negativity
        a_safe = np.maximum(a, 1e-30)
        z0 = np.log(a_safe)
        
        def obj_func(z):
            a_tmp = np.exp(z - z.max())
            a_tmp = a_tmp / a_tmp.sum()
            conv = fast_convolve_self(a_tmp)
            smax, _ = softmax_smooth_max(conv, tau_val)
            return 2.0 * n * smax
        
        def obj_grad(z):
            a_tmp = np.exp(z - z.max())
            a_tmp = a_tmp / a_tmp.sum()
            conv = fast_convolve_self(a_tmp)
            smax, w = softmax_smooth_max(conv, tau_val)
            
            # Gradient of (1/tau)*log(sum exp(tau*conv_k)) w.r.t. a
            # = sum_k w_k * d(conv_k)/da
            # d(conv_k)/da[j] = 2*a[k-j] if 0<=k-j<n
            grad_a = np.zeros(n)
            for k in range(len(conv)):
                if w[k] < 1e-15:
                    continue
                k_lo = max(0, k - n + 1)
                k_hi = min(n - 1, k)
                grad_a[k_lo:k_hi+1] += 2.0 * w[k] * a[k - k_hi:k - k_lo + 1]
            
            # Convert to z-space gradient (log-parameterization)
            # da/dz[j] = a[j] - a[j]*sum_l a[l]*grad_a[l] (projected gradient)
            S = np.dot(grad_a, a_tmp)
            grad_z = a_tmp * (grad_a - S)
            return 2.0 * n * grad_z
        
        # L-BFGS
        t_remaining = time_limit - (time.monotonic() - t0) - 3
        if t_remaining < 2:
            break
        
        maxiter = int(min(500, t_remaining * 100))
        
        try:
            res = minimize(
                fun=obj_func,
                x0=z0,
                jac=obj_grad,
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                         'maxfun': maxiter * 20, 'disp': False}
            )
            a_out = np.exp(res.x - res.x.max())
            a_out = a_out / a_out.sum()
            a_out = np.maximum(a_out, 0)
            a_out = a_out / a_out.sum()
        except Exception as e:
            a_out = a
        
        sc = score_normalized(a_out)
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
            a = a_out.copy()
        
        if verbose:
            print(f"    L-BFGS tau={tau_val}: score={best_score:.6f}")
    
    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=True):
    """
    Coordinate descent with 1D optimization per coordinate.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    sweep = 0
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False
        order = np.random.default_rng(sweep).permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-10:
                # Try setting to small positive value
                for trial in [old_val + 0.001, 0.001, 0.005]:
                    a_trial = a.copy()
                    a_trial[j] = trial
                    a_trial = a_trial / a_trial.sum()
                    sc = score_normalized(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                continue
            
            # Try multiplicative updates
            for factor in [1.05, 0.95, 1.1, 0.9, 1.2, 0.8, 1.5, 0.5]:
                a_trial = a.copy()
                a_trial[j] = old_val * factor
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = a_trial / s
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
        
        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc sweep={sweep} ({elapsed:.0f}s): score={best_score:.6f}")
        
        if not improved and sweep > 3:
            break
    
    return best_a, best_score


def peak_shave(a0, time_limit, verbose=True):
    """
    Aggressive peak shaving: reduce contributions to the maximum peaks.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    step = 0
    while time.monotonic() - t0 < time_limit:
        step += 1
        conv = fast_convolve_self(a)
        max_val = conv.max()
        
        # Find all peaks above 95% of max
        threshold = 0.95 * max_val
        peak_indices = np.where(conv >= threshold)[0]
        
        improved = False
        for k in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Find which (i, j) pairs contribute most to this peak
            k_lo = max(0, k - n + 1)
            k_hi = min(n - 1, k)
            idx = np.arange(k_lo, k_hi + 1)
            contribs = a[idx] * a[k - idx]
            
            # Get top contributors
            top_n = min(5, len(contribs))
            top_idx = np.argsort(contribs)[::-1][:top_n]
            
            for ci in top_idx:
                i = idx[ci]
                j = k - i
                
                if a[i] < 1e-8:
                    continue
                
                # Try reducing a[i] (or a[j] if different)
                reduce_targets = set()
                if i != j:
                    reduce_targets = {i, j}
                else:
                    reduce_targets = {i}
                
                for target in reduce_targets:
                    for factor in [0.95, 0.9, 0.85, 0.8]:
                        a_trial = a.copy()
                        old = a_trial[target]
                        a_trial[target] = old * factor
                        diff = old - a_trial[target]
                        if diff < 1e-12:
                            continue
                        # Redistribute to others proportionally
                        mask = np.ones(n, dtype=bool)
                        mask[target] = False
                        others = a_trial[mask]
                        if others.sum() < 1e-10:
                            continue
                        a_trial[mask] = others + diff * others / others.sum()
                        a_trial = a_trial / a_trial.sum()
                        sc = score_normalized(a_trial)
                        if sc < best_score - 1e-12:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
        
            if improved:
                break
        
        if not improved:
            # Try simultaneous reduction of top 2 contributors
            conv = fast_convolve_self(a)
            k = np.argmax(conv)
            k_lo = max(0, k - n + 1)
            k_hi = min(n - 1, k)
            idx = np.arange(k_lo, k_hi + 1)
            contribs = a[idx] * a[k - idx]
            top_idx = np.argsort(contribs)[::-1][:2]
            
            if len(top_idx) >= 2:
                i1, i2 = idx[top_idx[0]], idx[top_idx[1]]
                if i1 != i2 and a[i1] > 1e-8 and a[i2] > 1e-8:
                    for f1, f2 in [(0.9, 0.9), (0.85, 0.85), (0.95, 0.85)]:
                        a_trial = a.copy()
                        d1 = a_trial[i1] * (1 - f1)
                        d2 = a_trial[i2] * (1 - f2)
                        a_trial[i1] *= f1
                        a_trial[i2] *= f2
                        diff = d1 + d2
                        mask = np.ones(n, dtype=bool)
                        mask[i1] = False
                        mask[i2] = False
                        others = a_trial[mask]
                        if others.sum() < 1e-10:
                            continue
                        a_trial[mask] = others + diff * others / others.sum()
                        a_trial = a_trial / a_trial.sum()
                        sc = score_normalized(a_trial)
                        if sc < best_score - 1e-12:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
        
        if not improved:
            # Try adding to a low-contribution position
            # Find position with lowest self-contribution
            self_contrib = a**2
            min_pos = np.argmin(self_contrib)
            if a[min_pos] < best_a.min() * 1.5:
                for factor in [1.5, 2.0, 3.0]:
                    a_trial = a.copy()
                    old = a_trial[min_pos]
                    a_trial[min_pos] = old * factor if old > 0 else 0.001
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = score_normalized(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
        
        if not improved:
            break
        
        if verbose and step % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave step={step} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def subgradient_momentum(a0, time_limit, verbose=True):
    """
    Subgradient method with momentum and adaptive step size.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    step = 0
    lr = 0.01
    mom = 0.9
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        conv = fast_convolve_self(a)
        max_val = conv.max()
        
        # Find active peaks (within 1% of max)
        threshold = 0.99 * max_val
        active = np.where(conv >= threshold)[0]
        
        # Compute subgradient (uniform over active peaks)
        grad = np.zeros(n)
        for k in active:
            k_lo = max(0, k - n + 1)
            k_hi = min(n - 1, k)
            idx = np.arange(k_lo, k_hi + 1)
            grad[idx] += 2.0 * a[k - idx] / len(active)
        
        # Momentum update
        v = mom * v - lr * grad
        
        # Apply and project onto simplex
        a_new = a + v
        # Simplex projection
        u = np.sort(a_new)[::-1]
        cssv = np.cumsum(u)
        cond = u * np.arange(1, n + 1) > (cssv - 1)
        if cond.any():
            rho = np.nonzero(cond)[0][-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            a_new = np.maximum(a_new - theta, 0.0)
        else:
            a_new = np.ones(n) / n
        a_new = a_new / a_new.sum()
        
        sc = score_normalized(a_new)
        if sc < best_score - 1e-11:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
            lr = min(lr * 1.05, 0.05)
        else:
            no_improve += 1
            lr *= 0.98
        
        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(step)
            a = best_a + rng.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = 0.005
            no_improve = 0
        
        if verbose and step % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad step={step} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score
