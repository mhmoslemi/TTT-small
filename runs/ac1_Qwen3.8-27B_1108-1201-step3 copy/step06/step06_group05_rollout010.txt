import numpy as np

import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build matrix A where A[k,j] = f[k-j] for valid indices."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        end = min(2 * n - 1, j + n)
        length = end - start
        A[start:end, j] = f[:length]
    return A


def solve_lp_max_sum(f, M, n):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    A_ub = build_conv_matrix(f, n)
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.success:
        return res.x
    return None


def solve_lp_min_cross(f, n):
    """Solve: min M' s.t. (f*g)[k] <= M' for all k, sum(g)=1, g >= 0.
    Returns (g_opt, M'_opt)."""
    # Variables: g_0, ..., g_{n-1}, M'
    # x = [g_0, ..., g_{n-1}, M']
    nvar = n + 1
    c = np.zeros(nvar)
    c[-1] = 1.0  # minimize M'
    
    # Constraints: (f*g)_k - M' <= 0 for all k
    # A_ub has (2n-1) rows, nvar columns
    A_ub = np.zeros((2 * n - 1, nvar))
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A_ub[k, j] = f[k - j]
        A_ub[k, -1] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, nvar))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 60})
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def golden_section_line_search(f, g, func, lo=0.0, hi=1.0, tol=1e-6, max_iter=30):
    """Golden section search for alpha in [lo, hi] minimizing func((1-alpha)*f + alpha*g)."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a = lo
    b = hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = func(c)
    fd = func(d)
    
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = func(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = func(d)
    
    best_alpha = (a + b) / 2.0
    return best_alpha

            scaled -= scaled.max()

def make_shape(n, shape_type="raised_cos"):
    """Create various initial shapes."""
    x = np.linspace(-1, 1, n)
    if shape_type == "uniform":
        a = np.ones(n)
    elif shape_type == "gaussian":
        a = np.exp(-2.0 * x ** 2)
    elif shape_type == "raised_cos":
        a = 0.5 * (1 + np.cos(np.pi * x))
    elif shape_type == "bell4":
        a = (1 - np.abs(x) ** 4) ** 2
        a = np.maximum(a, 0)
    elif shape_type == "bimodal":
        a = np.exp(-10 * (np.abs(x) - 0.3) ** 2)
    elif shape_type == "plateau":
        a = np.where(np.abs(x) < 0.3, 1.0, 0.3)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    return f_interp


def coordinate_polish(f0, time_limit, verbose=False):
    """Coordinate descent: adjust individual coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f0, fast_convolve(f0).max()
    f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        c = fast_convolve(f)
        
        for j in range(0, n, max(1, n // 512)):  # subset of coordinates for speed
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-10:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in [0.7, 0.85, 1.15, 1.3]:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                f[j] = best_j_val
                f = np.maximum(f, 0)
                f /= f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj

            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if np.isfinite(p_score) and p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass
    
    if parent is None:
        parent = make_shape(768, "raised_cos")
    
    # === Phase 1: Quick multi-length screening ===
    print("=== Phase 1: Multi-length screening ===")
    t1 = time.monotonic()
    
    lengths_to_try = [512, 640, 768, 896, 1024, 1280, 1536]
    if len(parent) in lengths_to_try:
        lengths_to_try.remove(len(parent))
    
    best_screen_n = len(parent)
    best_screen_obj = 1e10
    screen_results = {}
    
    for n_try in lengths_to_try:
        if time.monotonic() - t1 > 60:
            break
        # Use interpolation from parent + different shapes
        f_interp = interpolate_sequence(parent, n_try)
        for shape in ["raised_cos", "gaussian", "bell4"]:
            f_shape = make_shape(n_try, shape)
            # Blend
            for mix in [0.0, 0.3, 0.7, 1.0]:
                f_test = (1 - mix) * f_interp + mix * f_shape
                f_test = np.maximum(f_test, 0)
                fs = f_test.sum()
                if fs < 1e-10:
                    continue
                f_test /= fs
                c = fast_convolve(f_test)
                obj = 2 * n_try * c.max()
                if obj < best_screen_obj:
                    best_screen_obj = obj
                    best_screen_n = n_try
                screen_results[(n_try, shape, mix)] = (f_test, obj)
    
    print(f"  Best screen: n={best_screen_n}, obj={best_screen_obj:.8f}")
    print(f"  Phase 1 time: {time.monotonic()-t1:.1f}s")
    
    # === Phase 2: LP-FW on parent length ===
    print(f"\n=== Phase 2: LP-FW on n={len(parent)} ===")
    t2 = time.monotonic()
    rem = deadline - time.monotonic() - 600
    t_alloc = max(60, min(rem, 500))
    print(f"  Allocation: {t_alloc:.0f}s")
    
    a_opt, obj = lp_fw_optimize(parent, time_limit=t_alloc, verbose=True)
    try_seq(a_opt, f"lpfw_n{len(parent)}")
    
    print(f"  Phase 2 time: {time.monotonic()-t2:.1f}s, best={best_score:.8f}")
    
    # === Phase 3: Subgradient refinement ===
    print(f"\n=== Phase 3: Subgradient refinement ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 60:
            t_alloc = min(rem * 0.4, 200)
            print(f"  Allocation: {t_alloc:.0f}s")
            a_opt, obj = subgradient_peak_reduction(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "subgrad")
    
    # === Phase 4: LP-FW on best sequence with restarts ===
    print(f"\n=== Phase 4: Deep LP-FW with restarts ===")
    if best_seq is not None:
        n_best = len(best_seq)
        restarts = [
            ("identity", best_seq.copy()),
            ("flip", best_seq[::-1].copy()),
            ("noise", project_simplex(best_seq + rng.standard_normal(n_best) * 0.001)),
        ]
        
        # Mix with different shapes
        for shape in ["raised_cos", "gaussian"]:
            f_shape = make_shape(n_best, shape)
            restarts.append((f"mix_{shape}", 0.85 * best_seq + 0.15 * f_shape))
            restarts[-1] = (restarts[-1][0], restarts[-1][1] / restarts[-1][1].sum())
        
        for i, (name, f_init) in enumerate(restarts):
            rem = deadline - time.monotonic() - 100
            if rem < 40:
                break
            t_alloc = min(rem / (len(restarts) - i), 120)
            print(f"  Restart '{name}': t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(f_init, time_limit=t_alloc, verbose=(i == 0))
            try_seq(a_opt, f"deep_lpfw_{name}")
    
    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length exploration ===")
    base = best_seq if best_seq is not None else parent
    
    # Try the best screening length if different
    candidate_lengths = set([len(base)])
    for (n_try, shape, mix), (f_test, obj) in list(screen_results.items())[:10]:
        if obj < best_screen_obj * 1.02 and n_try != len(base):
            candidate_lengths.add(n_try)
    
    candidate_lengths = sorted(candidate_lengths)
    
    for target_n in candidate_lengths:
        rem = deadline - time.monotonic() - 80
        if rem < 80:
            break
        
        if target_n == len(base):
            continue
        
        f_interp = interpolate_sequence(base, target_n)
        try_seq(f_interp, f"interp_n{target_n}")
        
        # Quick LP-FW
        t_gd = min(rem * 0.3, 100)
        if t_gd > 30:
            a_opt, obj = lp_fw_optimize(f_interp, time_limit=t_gd)
            try_seq(a_opt, f"lpfw_n{target_n}")
    
    # === Phase 6: Final polishing ===
    print(f"\n=== Phase 6: Final polishing ===")
    if best_seq is not None:
        # Coordinate descent
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_alloc = min(rem, 40)
            print(f"  Coord descent: t={t_alloc:.0f}s")
            a_opt, obj = coordinate_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord")
        
        # One more LP-FW squeeze
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            t_alloc = min(rem, 30)
            a_opt, obj = lp_fw_optimize(a_opt, time_limit=t_alloc)
            try_seq(a_opt, "final_lpfw")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
