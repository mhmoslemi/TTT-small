        length = k_end - k_start + 1
        A[k_start:k_end + 1, j] = f[:length]
    return A

                w = np.maximum(f[non_peak], 0)
                wsum = w.sum()
                if wsum > 1e-15:
                    f[non_peak] += diff * w / wsum
                else:
                    f[non_peak] += diff / non_peak.sum()
            f = np.maximum(f, 0)
            f = f / f.sum()
            no_improve = 0
            m_exp = 1.0
            continue

        # Solve LP direction
        M_eff = M * max(m_exp, 1.0)
        g, dual = solve_lp(f, M_eff, n, A_cache=A_cache)
        if g is None:
            m_exp *= 1.2
            if m_exp > 10:
                m_exp = 1.0
                # Fallback: use gradient direction
                k_peak = int(np.argmax(auto_conv(f)))
                js = max(0, k_peak - n + 1)
                je = min(n - 1, k_peak)
                g = f.copy()
                g[js:je + 1] *= 0.5
                g = np.maximum(g, 0)
                g = g / g.sum()
            continue

        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            m_exp *= 1.1
            continue
        g = g / gs

        # Check if direction is meaningful
        diff = np.max(np.abs(g - f))
        if diff < 1e-10:
            no_improve += 2
            m_exp = min(m_exp * 1.05, 5.0)
            continue

        # === Away Step: identify worst coordinate ===
        conv_now = auto_conv(f)
        k_max = int(np.argmax(conv_now))
        js_away = max(0, k_max - n + 1)
        je_away = min(n - 1, k_max)
        idx_away = np.arange(js_away, je_away + 1)
        if len(idx_away) > 0:
            products = f[idx_away] * f[k_max - idx_away]
            j_worst = int(idx_away[np.argmax(products)])
        else:
            j_worst = int(np.argmax(f))

        # Try away step: reduce mass at j_worst
        # Compute the "away" direction
        away = -1.0  # coefficient for e_{j_worst}
        # Try a few away amounts
        best_away_alpha = 0.0
        best_away_val = cur_obj

        # First, get the FW step
        alpha_fw, obj_fw = line_search_agg(f, g, M, n, max_evals=12)

        # Try combining FW step with away step
        if alpha_fw > 0.05:
            base_step = f + alpha_fw * (g - f)
            base_step = np.maximum(base_step, 0)
            bs = base_step.sum()
            if bs > 1e-15:
                base_step = base_step / bs

            # Try removing mass from worst coordinate
            for away_frac in [0.1, 0.2, 0.3, 0.5]:
                trial = base_step.copy()
                removal = trial[j_worst] * away_frac
                if removal > 1e-15:
                    trial[j_worst] -= removal
                    # Redistribute to non-peak coordinates
                    non_peak = np.ones(n, dtype=bool)
                    non_peak[js_away:je_away + 1] = False
                    if non_peak.sum() > 0:
                        w = np.maximum(trial[non_peak], 0)
                        wsum = w.sum()
                        if wsum > 1e-15:
                            trial[non_peak] += removal * w / wsum
                        else:
                            trial[non_peak] += removal / non_peak.sum()
                    trial = np.maximum(trial, 0)
                    ts = trial.sum()
                    if ts > 1e-15:
                        trial = trial / ts
                        obj_trial = 2.0 * n * auto_conv(trial).max()
                        if obj_trial < best_away_val:
                            best_away_val = obj_trial
                            best_away_alpha = away_frac
                            if obj_trial < obj_fw:
                                obj_fw = obj_trial

        # Apply the best step
        if best_away_alpha > 0 and best_away_val < cur_obj - 1e-14:
            f = base_step.copy()
            removal = f[j_worst] * best_away_alpha
            if removal > 1e-15:
                f[j_worst] -= removal
                non_peak = np.ones(n, dtype=bool)
                non_peak[js_away:je_away + 1] = False
                if non_peak.sum() > 0:
                    w = np.maximum(f[non_peak], 0)
                    wsum = w.sum()
                    if wsum > 1e-15:
                        f[non_peak] += removal * w / wsum
                    else:
                        f[non_peak] += removal / non_peak.sum()
                f = np.maximum(f, 0)
                f = f / f.sum()
        elif alpha_fw > 1e-6:
            f_new = f + alpha_fw * (g - f)
            f_new = np.maximum(f_new, 0)
            sn = f_new.sum()
            if sn > 1e-15:
                f = f_new / sn
            else:
                no_improve += 1
                m_exp *= 1.05
                continue
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.03, 3.0)
            continue

        # Update
        if best_obj < 2.0 * n * auto_conv(f).max():
            pass  # Keep best_f as is

        if verbose and it % 50 == 0:
            print(f"    AFW iter {it}: obj={best_obj:.10f}, m_exp={m_exp:.3f}, no_imp={no_improve}")

        if no_improve > 150:
            break

    return best_f, best_obj


def peak_cd_polish(f0, time_limit, verbose=False):
    """
    Peak-targeted coordinate descent with multi-peak awareness.
    Tries to flatten the top-K peaks simultaneously.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = auto_conv(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    K = 8

    while time.monotonic() - t0 < time_limit and no_improve < 60:
        conv = auto_conv(f)
        M = conv.max()
        improved = False

        # Get top K peaks
        top_k_indices = np.argsort(conv)[-K:]
        top_k_indices = top_k_indices[::-1]  # Sort by value descending

        for k in top_k_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[k - idx]
            total = products.sum()
            if total < 1e-15:
                continue

            # Try reducing top contributors
            order = np.argsort(products)[::-1]
            for oi in range(min(5, len(order))):
                j_max = int(idx[order[oi]])
                old_val = f[j_max]
                if old_val < 1e-12:
                    continue

                for mult in [0.3, 0.5, 0.7, 0.85, 1.2, 1.5, 2.0]:
                    new_val = old_val * mult
                    if new_val < 0:
                        continue
                    if new_val > 0.3:
                        continue

                    trial = f.copy()
                    trial[j_max] = new_val
                    d = old_val - new_val
                    if abs(d) > 1e-15:
                        # Redistribute away from all top peaks
                        peak_mask = np.zeros(n, dtype=bool)
                        for kp in top_k_indices:
                            kp = int(kp)
                            peak_mask[max(0, kp - n + 1):min(n, kp + 1)] = True
                        non_peak = ~peak_mask
                        if non_peak.sum() > 0:
                            w = np.maximum(f[non_peak], 0)
                            wsum = w.sum()
                            if wsum > 1e-15:
                                trial[non_peak] += d * w / wsum
                            else:
                                trial[non_peak] += d / non_peak.sum()
                        else:
                            trial[np.argmax(f)] += d

                    trial = np.maximum(trial, 0)
                    tsum = trial.sum()
                    if tsum < 1e-15:
                        continue
                    trial = trial / tsum
                    obj = 2.0 * n * auto_conv(trial).max()
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_f = trial.copy()
                        f = trial
                        improved = True

        # Also try: scale down the entire peak region
        if not improved and no_improve > 5:
            k_main = int(np.argmax(auto_conv(f)))
            js = max(0, k_main - n + 1)
            je = min(n - 1, k_main)

            for scale in [0.85, 0.8, 0.7, 0.9]:
                trial = f.copy()
                trial[js:je + 1] *= scale
                d = f.sum() - trial.sum()
                non_contrib = np.ones(n, dtype=bool)
                non_contrib[js:je + 1] = False
                if non_contrib.sum() > 0:
                    nc_w = np.maximum(f[non_contrib], 0)
                    nw = nc_w.sum()
                    if nw > 1e-15:
                        trial[non_contrib] += d * nc_w / nw
                    else:
                        trial[non_contrib] += d / non_contrib.sum()
                trial = np.maximum(trial, 0)
                tsum = trial.sum()
                if tsum > 1e-15:
                    trial = trial / tsum
                    obj = 2.0 * n * auto_conv(trial).max()
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_f = trial.copy()
                        f = trial
                        improved = True

        if not improved:
            no_improve += 1
            if no_improve > 30:
                # Aggressive random perturbation
                rng = np.random.default_rng(int(time.monotonic() * 1000))
                f = best_f.copy()
                scale = 0.005 * min(no_improve / 10.0, 3.0)
                f = np.maximum(best_f + rng.standard_normal(n) * scale, 0)
                f = f / f.sum()
                no_improve = 0
        else:
            no_improve = 0

        if verbose and it if 'it' in dir() else True:
            pass

    return best_f, best_obj


def mirror_descent(f0, time_limit, verbose=False, lr_init=0.003, n_peaks=20):
    """Multiplicative weight mirror descent with peak-aware gradients."""
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)

    best_obj = 2.0 * n * auto_conv(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    temperature = 0.003
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = auto_conv(f)
        M = conv.max()

        # Get top peaks with soft weighting
        k_actual = min(n_peaks, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute gradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            kk = int(peaks[i])
            js = max(0, kk - n + 1)
            je = min(n - 1, kk)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[kk - idx]

        # Multiplicative update
        f = f * np.exp(-lr * sg)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        co = 2.0 * n * auto_conv(f).max()
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9999
        temperature *= 1.005

        if no_improve > 100:
            rng = np.random.default_rng(it * 43 + 11)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-15)
            f = f / f.sum()
            lr = max(lr_init * 0.7, 1e-7)
            no_improve = 0

        if lr < 1e-9:
            break

    return best_f, best_obj


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.25):
    """Gaussian initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width=0.7, power=2.0):
    """Bell-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    half_w = n * width / 2.0
    t = np.clip(np.abs(x) / half_w, 0, 1.0)
    a = (1.0 - t) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_periodic_envelope(n, period, decay=0.5):
    """Periodic envelope: high at multiples of period, low elsewhere."""
    x = np.arange(n)
    a = np.ones(n)
    for i in range(n):
        dist = min(i % period, period - (i % period))
        a[i] = np.exp(-decay * (dist / period) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def try_seq(a, best_seq, best_score, label=""):
    """Evaluate and potentially update best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining away-step Frank-Wolfe, peak CD,
    mirror descent, and multi-length exploration with structured initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768, 0.25)
        best_seq, best_score = try_seq(parent, best_seq, best_score, "default")

    n_parent = len(parent)
    print(f"Starting with n={n_parent}, score={best_score:.6f}")

    # === Phase 1: Quick diverse initialization ===
    print("\n=== Phase 1: Diverse Initializations ===")
    best_init = None
    best_init_obj = float('inf')

    init_list = [parent.copy()]
    for n_try in [640, 768, 800, 1024]:
        if n_try == n_parent:
            init_list.append(parent.copy())
        else:
            init_list.append(interpolate_seq(parent, n_try))
        init_list.append(make_gaussian(n_try, 0.2))
        init_list.append(make_gaussian(n_try, 0.35))
        init_list.append(make_bell(n_try, 0.6, 2.0))
        init_list.append(make_bell(n_try, 0.8, 3.0))
        # Periodic structures
        for period in [16, 32, 64]:
            if n_try % period == 0:
                init_list.append(make_periodic_envelope(n_try, period, 0.8))

    for ic in init_list:
        s = ic.sum()
        if s < 1e-10:
            continue
        a_norm = ic / s
        obj = 2.0 * len(a_norm) * auto_conv(a_norm).max()
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_norm.copy()
        best_seq, best_score = try_seq(a_norm, best_seq, best_score, f"init_n{len(a_norm)}")

    print(f"  Best init: normalized_obj={best_init_obj:.6f} (n={len(best_init)})")

    # === Phase 2: Main AFW optimization ===
    print(f"\n=== Phase 2: Away-step FW (n={len(best_init)}) ===")
    rem = deadline - time.monotonic() - 800
    if rem > 60:
        t_alloc = min(rem, 400)
        print(f"  AFW: t={t_alloc:.0f}s")
        a_opt, obj = away_step_fw(best_init, time_limit=t_alloc, verbose=True)
        best_seq, best_score = try_seq(a_opt, best_seq, best_score, "afw_main")

    # === Phase 3: Peak CD polish ===
    if best_seq is not None:
        print(f"\n=== Phase 3: Peak CD (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 600
        if rem > 30:
            t_alloc = min(rem, 150)
            print(f"  Peak CD: t={t_alloc:.0f}s")
            a_opt, obj = peak_cd_polish(best_seq, time_limit=t_alloc)
            best_seq, best_score = try_seq(a_opt, best_seq, best_score, "peak_cd")

    # === Phase 4: Mirror descent ===
    if best_seq is not None:
        print(f"\n=== Phase 4: Mirror Descent (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_alloc = min(rem, 120)
            print(f"  MD: t={t_alloc:.0f}s")
            a_opt, obj = mirror_descent(best_seq, time_limit=t_alloc)
            best_seq, best_score = try_seq(a_opt, best_seq, best_score, "mirror_descent")

    # === Phase 5: Second AFW pass (deeper) ===
    if best_seq is not None:
        print(f"\n=== Phase 5: Deep AFW (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 350
        if rem > 40:
            t_alloc = min(rem, 180)
            print(f"  Deep AFW: t={t_alloc:.0f}s")
            a_opt, obj = away_step_fw(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = try_seq(a_opt, best_seq, best_score, "deep_afw")

    # === Phase 6: Multi-length exploration ===
    base = best_seq if best_seq is not None else best_init
    print(f"\n=== Phase 6: Multi-length Exploration ===")
    for target_n in [700, 832, 900, 1024, 1200]:
        if len(base) == target_n:
            continue
        rem = deadline - time.monotonic() - 200
        if rem < 50:
            break
        f_interp = interpolate_seq(base, target_n)
        best_seq, best_score = try_seq(f_interp, best_seq, best_score, f"interp_n{target_n}")
        t_alloc = min((deadline - time.monotonic() - 200) * 0.35, 60)
        if t_alloc < 20:
            continue
        print(f"  AFW n={target_n}: t={t_alloc:.0f}s")
        a_opt, obj = away_step_fw(f_interp, time_limit=t_alloc, verbose=False)
        best_seq, best_score = try_seq(a_opt, best_seq, best_score, f"afw_n{target_n}")

    # === Phase 7: Perturbed variants ===
    if best_seq is not None:
        print(f"\n=== Phase 7: Perturbed AFW ===")
        n_best = len(best_seq)
        for p in range(4):
            rem = deadline - time.monotonic() - 80
            if rem < 30:
                break
            t_alloc = min(rem / (4 - p), 70)

            if p == 0:
                noise = rng.standard_normal(n_best) * 0.003
                f_init = proj_simplex(best_seq + noise)
            elif p == 1:
                f_init = 0.9 * best_seq + 0.1 * (np.ones(n_best) / n_best)
                f_init = f_init / f_init.sum()
            elif p == 2:
                # Symmetric blend
                f_init = 0.5 * best_seq + 0.5 * best_seq[::-1]
                f_init = f_init / f_init.sum()
            else:
                # Peak-targeted perturbation
                conv_b = auto_conv(best_seq)
                k_peak = int(np.argmax(conv_b))
                js = max(0, k_peak - n_best + 1)
                je = min(n_best - 1, k_peak)
                f_init = best_seq.copy()
                f_init[js:je + 1] *= 0.85
                f_init[js:je + 1] += rng.standard_normal(je - js + 1) * 0.005
                f_init = np.maximum(f_init, 0)
                f_init = f_init / f_init.sum()

            print(f"  Variant {p}: t={t_alloc:.0f}s")
            a_opt, obj = away_step_fw(f_init, time_limit=t_alloc, verbose=False)
            best_seq, best_score = try_seq(a_opt, best_seq, best_score, f"perturb_{p}")

    # === Phase 8: Final polish ===
    if best_seq is not None:
        print(f"\n=== Phase 8: Final Polish ===")
        rem = deadline - time.monotonic() - 15
        if rem > 20:
            # Peak CD
            t1 = min(rem * 0.4, 40)
            print(f"  Final Peak CD: t={t1:.0f}s")
            a_opt, obj = peak_cd_polish(best_seq, time_limit=t1)
            best_seq, best_score = try_seq(a_opt, best_seq, best_score, "fin_peak")

            # Quick AFW
            rem2 = deadline - time.monotonic() - 5
            if rem2 > 10 and best_seq is not None:
                t2 = min(rem2, 40)
                print(f"  Final AFW: t={t2:.0f}s")
                a_opt, obj = away_step_fw(best_seq, time_limit=t2, verbose=False)
                best_seq, best_score = try_seq(a_opt, best_seq, best_score, "fin_afw")

            # Mirror descent
            rem3 = deadline - time.monotonic() - 3
            if rem3 > 8 and best_seq is not None:
                t3 = min(rem3, 25)
                print(f"  Final MD: t={t3:.0f}s")
                a_opt, obj = mirror_descent(best_seq, time_limit=t3)
                best_seq, best_score = try_seq(a_opt, best_seq, best_score, "fin_md")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
