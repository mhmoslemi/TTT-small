def compute_obj_grad(z, T):
    """Compute objective and gradient for softmax parameterization with soft-max surrogate."""
    n = len(z)
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a = a / a.sum()
    
    conv = fast_convolve(a)
    
    # Soft-max
    T_conv = T * (conv - conv.max())
    exp_T_conv = np.exp(T_conv)
    sum_exp = exp_T_conv.sum()
    soft_max_val = conv.max() + np.log(sum_exp) / T

- sum(a) = (1/n) * sum_j f(j/n) ≈ ∫_0^1 f(x) dx

where (f*f_cont)(y) = ∫ f(x) f(y-x) dx (continuous convolution).

So: 2n * max(conv(a,a)) / (sum a)^2 ≈ 2n * (1/n) * max(f*f_cont) / (∫f)^2 = 2 * max(f*f_cont) / (∫f)^2

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS with adaptive soft-max
surrogate in log-space (softmax parameterization), diverse parameterized
initializations, peak-targeted coordinate descent, subgradient polishing,
and multi-length exploration. Key innovations: smooth quasi-Newton optimization
with temperature annealing, structural parameter sweeps, and aggressive
peak-elimination coordinate descent with diverse redistribution strategies.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0.5, old_n - 0.5, old_n)
    new_x = np.linspace(0.5, new_n - 0.5, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    return f_interp


def make_gaussian(n, sigma_frac=0.25, center_frac=0.5):
    """Gaussian initial sequence."""
    x = np.arange(n) - n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width=0.7, sharpness=2.0):
    """Bell-shaped (raised cosine-like) initial sequence."""
    x = np.arange(n) - n / 2.0
    half_w = n * width / 2.0
    t = np.clip(np.abs(x) / half_w, 0, 1.0)
    a = (1.0 - t) ** sharpness
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_trapezoid(n, plateau_frac=0.3, transition_frac=0.3):
    """Trapezoidal initial sequence."""
    x = np.arange(n)
    center = n / 2.0
    half_plateau = n * plateau_frac / 2.0
    half_total = n * (plateau_frac + transition_frac) / 2.0
    a = np.zeros(n)
    for i in range(n):
        d = abs(x[i] - center)
        if d <= half_plateau:
            a[i] = 1.0
        elif d <= half_total:
            a[i] = 1.0 - (d - half_plateau) / (half_total - half_plateau)
        else:
            a[i] = 0.0
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_asymmetric(n, decay=3.0, side='left'):
    """Asymmetric exponential initial sequence."""
    x = np.arange(n)
    if side == 'left':
        a = np.exp(-decay * x / n)
    else:
        a = np.exp(-decay * (n - 1 - x) / n)
    a = a / a.sum()
    return a


def make_bimodal(n, sep=0.3, width=0.12):
    """Bimodal (two-peaked) initial sequence."""
    centers = [n * (0.5 - sep), n * (0.5 + sep)]
    a = np.zeros(n)
    for c in centers:
        a += np.exp(-0.5 * ((np.arange(n) - c) / (n * width)) ** 2)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def compute_softmax_obj_grad(z, T):
    """
    Compute objective and gradient for softmax parameterization
    with soft-max surrogate for the max convolution value.
    
    Parameters:
        z: log-weights vector (length n)
        T: temperature for soft-max approximation
    
    Returns:
        (objective_value, gradient_array)
    """
    n = len(z)
    # Compute a = softmax(z) with numerical stability
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a_sum = a.sum()
    a = a / a_sum

    # Compute convolution
    conv = fast_convolve(a)

    # Soft-max approximation of max(conv)
    T_conv = T * (conv - conv.max())
    # Clip for numerical stability
    T_conv = np.clip(T_conv, -700, 700)
    exp_T_conv = np.exp(T_conv)
    sum_exp = exp_T_conv.sum()
    if sum_exp < 1e-300:
        soft_max_val = conv.max()
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        soft_max_val = conv.max() + np.log(sum_exp) / T
        p = exp_T_conv / sum_exp

    obj = 2.0 * n * soft_max_val

    # Gradient: d(soft_max)/da[j] = sum_m a[m] * p[j+m] = cross-correlation(a, p)[j]
    fft_size = 1 << (3 * n - 2).bit_length()
    A_fft = np.fft.rfft(a, fft_size)
    P_fft = np.fft.rfft(p, fft_size)
    xcorr = np.fft.irfft(np.conj(A_fft) * P_fft, fft_size)
    grad_a = xcorr[:n]

    # Objective gradient w.r.t. a
    grad_F_a = 2.0 * n * grad_a

    # Chain rule through softmax: dF/dz[j] = a[j] * (grad_F_a[j] - dot(a, grad_F_a))
    mean_grad = np.dot(a, grad_F_a)
    grad_z = a * (grad_F_a - mean_grad)

    return obj, grad_z


def compute_hard_obj_grad(z, top_k=5):
    """
    Compute objective and subgradient using top-K convolution positions.
    More stable than pure soft-max for large T.
    """
    n = len(z)
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a_sum = a.sum()
    a = a / a_sum

    conv = fast_convolve(a)
    M = conv.max()
    obj = 2.0 * n * M

    # Find top-k positions
    k_actual = min(top_k, len(conv))
    peak_indices = np.argpartition(conv, -k_actual)[-k_actual:]
    peak_vals = conv[peak_indices]
    # Soft weights over the top-k peaks
    T_sub = 20.0
    w = np.exp(T_sub * (peak_vals - peak_vals.max()))
    w /= w.sum()

    # Gradient: sum over top-k peaks with weights
    p = np.zeros(len(conv))
    for i, k in enumerate(peak_indices):
        p[k] = w[i]

    fft_size = 1 << (3 * n - 2).bit_length()
    A_fft = np.fft.rfft(a, fft_size)
    P_fft = np.fft.rfft(p, fft_size)
    xcorr = np.fft.irfft(np.conj(A_fft) * P_fft, fft_size)
    grad_a = xcorr[:n]

    grad_F_a = 2.0 * n * grad_a
    mean_grad = np.dot(a, grad_F_a)
    grad_z = a * (grad_F_a - mean_grad)

    return obj, grad_z


def lbfgs_optimize(a_init, time_limit, T_schedule=None):
    """
    Run L-BFGS optimization with adaptive temperature schedule.
    
    Parameters:
        a_init: initial sequence (will be normalized)
        time_limit: seconds to spend on this optimization
        T_schedule: list of temperatures to try in sequence
    
    Returns:
        optimized sequence (sum-normalized)
    """
    n = len(a_init)
    a_init = np.maximum(a_init.astype(np.float64), 0.0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    if T_schedule is None:
        T_schedule = [15.0, 25.0, 40.0, 60.0, 90.0]

    # Initialize z from a_init
    a_safe = np.maximum(a_init, 1e-12)
    z0 = np.log(a_safe / a_safe.sum())

    best_a = a_init.copy()
    best_true_obj = 2.0 * n * fast_convolve(a_init).max()

    t0 = time.monotonic()
    max_total = time_limit

    for T in T_schedule:
        elapsed = time.monotonic() - t0
        if elapsed > max_total - 5:
            break
        remaining = max_total - elapsed
        if remaining < 5:
            break

        # Determine max iterations based on remaining time
        # Each iteration is roughly O(n log n) for the FFT
        max_iter = max(30, int(remaining * 8))

        def z_obj_grad(z):
            return compute_softmax_obj_grad(z, T)

        try:
            res = minimize(z_obj_grad, z0, method='L-BFGS-B',
                           options={'maxiter': max_iter, 'maxfun': max_iter * 2,
                                    'gtol': 1e-9, 'ftol': 1e-12})
            if res.x is not None:
                z0 = res.x
        except Exception:
            pass

        # Evaluate true objective at current z0
        z_shifted = z0 - z0.max()
        a_cur = np.exp(z_shifted)
        a_cur = a_cur / a_cur.sum()
        true_obj = 2.0 * n * fast_convolve(a_cur).max()
        if true_obj < best_true_obj:
            best_true_obj = true_obj
            best_a = a_cur.copy()

    return best_a

def peak_cd_polish(a_init, time_limit, verbose=False):
    """
    Peak-targeted coordinate descent: identifies top peaks and
    adjusts contributing elements to reduce them.
    """
    f = np.maximum(a_init.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit and no_improve < 60:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        top_k = min(15, len(conv))
        peak_indices = np.argsort(conv)[-top_k:]

        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[k - idx]
            total = products.sum()
            if total < 1e-15:
                continue

            # Try reducing top contributors and increasing non-contributors
            order = np.argsort(products)[::-1]
            for oi in range(min(6, len(order))):
                j_max = int(idx[order[oi]])
                old_val = f[j_max]
                if old_val < 1e-12:
                    continue

                for mult in [0.3, 0.5, 0.7, 0.85, 1.2, 1.5, 2.0]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.5:
                        continue

                    trial = f.copy()
                    trial[j_max] = new_val
                    d = old_val - new_val
                    if abs(d) > 1e-15:
                        # Redistribute to non-peak contributors
                        non_peak = np.ones(n, dtype=bool)
                        non_peak[js:je + 1] = False
                        cnt = non_peak.sum()
                        if cnt > 0:
                            w = f[non_peak]
                            wsum = w.sum()
                            if wsum > 1e-15:
                                trial[non_peak] += d * w / wsum
                            else:
                                trial[non_peak] += d / cnt
                    trial = np.maximum(trial, 0)
                    tsum = trial.sum()
                    if tsum < 1e-15:
                        continue
                    trial = trial / tsum
                    obj = 2.0 * n * fast_convolve(trial).max()
                    if obj < best_obj - 1e-12:
                        best_obj = obj
                        best_f = trial.copy()
                        f = trial
                        improved = True

        if not improved:
            no_improve += 1
            # Aggressive: scale down all peak contributors
            if no_improve > 10:
                conv_cur = fast_convolve(f)
                k = int(np.argmax(conv_cur))
                js = max(0, k - n + 1)
                je = min(n - 1, k)

                for scale in [0.9, 0.8, 0.7]:
                    trial = f.copy()
                    trial[js:je + 1] *= scale
                    d = f.sum() - trial.sum()
                    non_contrib = np.ones(n, dtype=bool)
                    non_contrib[js:je + 1] = False
                    if non_contrib.sum() > 0:
                        nc_w = f[non_contrib]
                        nw = nc_w.sum()
                        if nw > 1e-15:
                            trial[non_contrib] += d * nc_w / nw
                    trial = np.maximum(trial, 0)
                    trial = trial / trial.sum()
                    obj = 2.0 * n * fast_convolve(trial).max()
                    if obj < best_obj:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        no_improve = 0
                        break
                else:
                    # Try random perturbation
                    rng = np.random.default_rng(it * 7919)
                    noise = rng.standard_normal(n) * 0.003
                    trial = project_simplex(f + noise)
                    obj = 2.0 * n * fast_convolve(trial).max()
                    if obj < best_obj:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        no_improve = 0
                    else:
                        if no_improve > 30:
                            break
        else:
            no_improve = 0

    return best_f, best_obj


def subgradient_polish(a_init, time_limit):
    """
    Projected subgradient descent targeting the max convolution position.
    Uses adaptive step size and periodic restarts.
    """
    f = np.maximum(a_init.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_obj = 2.0 * n * fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    step = 0.01
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Use top-3 peaks with equal weight for a smoother subgradient
        k_peek = min(3, len(conv))
        peaks = np.argpartition(conv, -k_peek)[-k_peek:]
        peak_vals = conv[peaks]
        w = peak_vals / peak_vals.sum()

        # Subgradient: sum over peaks of w_i * a[k_i - j]
        grad = np.zeros(n)
        for i in range(len(peaks)):
            kk = int(peaks[i])
            js = max(0, kk - n + 1)
            je = min(n - 1, kk)
            idx = np.arange(js, je + 1)
            grad[idx] += w[i] * f[kk - idx]

        # Update: f = f - step * grad, then project to simplex
        f_new = f - step * grad
        f_new = project_simplex(f_new)
        f = f_new

        obj = 2.0 * n * fast_convolve(f).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            step *= 0.97
            if no_improve > 100:
                # Restart from best with perturbation
                rng = np.random.default_rng(it * 104729)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                step = 0.008
                no_improve = 0

        if step < 1e-6:
            break

    return best_f, best_obj


def multiplicative_descent(a_init, time_limit, lr_init=0.003, decay=0.9995):
    """Mirror descent with multiplicative weight updates targeting peaks."""
    f = np.maximum(a_init.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    best_obj = 2.0 * n * fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    temperature = 0.003
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)

        k_actual = min(30, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            kk = int(peaks[i])
            js = max(0, kk - n + 1)
            je = min(n - 1, kk)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[kk - idx]

        f = f * np.exp(-lr * sg)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        co = 2.0 * n * fast_convolve(f).max()
        if co < best_obj - 1e-12:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= decay
        temperature *= 1.005

        if no_improve > 200:
            rng = np.random.default_rng(it * 43 + 11)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.004, 1e-15)
            f = f / f.sum()
            lr = max(lr_init * 0.7, 1e-6)
            no_improve = 0

        if lr < 1e-9:
            break

    return best_f, best_obj


def try_sequence(a, best_seq, best_score, label=""):
    """Evaluate a candidate sequence, return updated best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score
