import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the objective score for a normalized sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    max_conv = fast_convolve(a_norm).max()
    return float(2.0 * n * max_conv)


def gs_initialize(n, max_iter=200, fft_size=None, seed=0):
    """
    Gerchberg-Saxton algorithm to find a non-negative sequence
    with approximately flat power spectrum (low peak auto-correlation).
    """
    if fft_size is None:
        fft_size = 1 << (2 * n - 1).bit_length()
    
    rng = np.random.default_rng(seed)
    
    # Start with random phase
    phi = rng.uniform(0, 2 * np.pi, fft_size)
    # Ensure Hermitian symmetry for real output
    # For real signal: X[-k] = conj(X[k])
    # We'll work with full FFT for simplicity
    
    A = np.exp(1j * phi)
    # Make Hermitian symmetric
    A = (A + np.conj(A[np.mod(np.arange(fft_size), fft_size)[-1::-1]])) / 2
    # Actually, simpler: just take real part at the end
    
    a = np.ones(n) / n  # initial
    a_prev = a.copy()
    
    for it in range(max_iter):
        # FFT (zero-padded)
        A_full = fft(a, fft_size)
        # Keep phase, set magnitude to constant
        mag = np.abs(A_full)
        mag_new = np.ones(fft_size)
        mag_new[0] = mag[0]  # preserve DC
        # Actually, let's just use uniform magnitude
        A_new = mag_new * np.exp(1j * np.angle(A_full))
        
        # Inverse FFT
        a_new = ifft(A_new).real[:n]
        
        # Project to non-negative simplex
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Check convergence
        if np.allclose(a_new, a, atol=1e-10):
            break
        a = a_new
    
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def gw_initialize(n, max_iter=100, seed=0):
    """
    Gerchberg-Wax algorithm variant: alternate between
    time-domain non-negativity constraint and frequency-domain flat magnitude.
    """
    fft_size = 1 << (2 * n - 1).bit_length()
    rng = np.random.default_rng(seed)
    
    # Initialize in time domain
    a = rng.random(n)
    a = a / a.sum()
    
    for it in range(max_iter):
        # Time -> Frequency
        A = fft(a, fft_size)
        
        # Frequency constraint: flat magnitude
        phase = np.angle(A)
        A_flat = np.exp(1j * phase)
        A_flat[0] = A[0]  # preserve DC component
        
        # Frequency -> Time
        a_new = ifft(A_flat).real[:n]
        
        # Time constraint: non-negative
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a = np.ones(n) / n
            break
        a_new = a_new / s
        a = a_new
    
    return a


def multiplicative_weights_fp(a0, time_budget, eta0=0.1, eta_decay=0.999, 
                                top_k=20, max_iter=100000):

    Fictitious play with multiplicative weights update.
    Uses the subgradient of the (soft-max) auto-correlation peak.
    """

    a = np.maximum(a0.astype(np.float64), 1e-30)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    eta = eta0
    
    # Temperature for soft-max (starts high, decreases)
    T = max(1.0 / (n * best_score * 0.1), 0.01)
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_budget:
            break
        
        # Compute auto-correlation
        b = fast_convolve(a)
        
        # Compute soft-max weights over the auto-correlation
        # Use top-k for efficiency
        if top_k < len(b):
            # Focus on top peaks
            peaks = np.argpartition(b, -top_k)[-top_k:]
            b_top = b[peaks]
            max_b_top = b_top.max()
            w_top = np.exp((b_top - max_b_top) / T)
            w_top = w_top / w_top.sum()

                je = min(n - 1, k)
                j_idx = np.arange(js, je + 1)

            w = np.exp((b - max_b) / T)
            w = w / w.sum()
            g = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-12:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                j_idx = np.arange(js, je + 1)
                k_minus_j = k - j_idx
                g[j_idx] += 2.0 * w[k] * a[k_minus_j]
        
        # Multiplicative weights update
        # a_new[j] = a[j] * exp(-eta * g[j]) / Z
        # Work in log domain for stability
        log_a = np.log(a)
        log_a -= eta * g
        # Normalize (sum of exp(log_a) should be 1)
        log_a -= log_a.max()
        exp_log_a = np.exp(log_a)
        Z = exp_log_a.sum()
        a_new = exp_log_a / Z
        
        # Ensure non-negativity (should be automatic from exp)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Evaluate
        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        # Decay learning rate and temperature
        eta *= eta_decay
        T *= 0.9995
        T = max(T, 0.001)
        
        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW-FP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f} eta={eta:.4f} T={T:.4f}")
    
    return best_a, best_score


def subgradient_momentum(a0, time_budget, lr_init=0.01, lr_decay=0.999, 
                          momentum=0.9, top_k=50):
    """
    Subgradient method with Nesterov momentum for the auto-correlation peak.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    v = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_budget:
            break
        
        b = fast_convolve(a)
        
        # Soft-max subgradient
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        b_peaks = b[peaks]
        T = 0.01 * max(b_peaks.max(), 1e-10)
        w = np.exp((b_peaks - b_peaks.max()) / T)
        w = w / w.sum()
        
        # Subgradient
        g = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            j_idx = np.arange(js, je + 1)
            g[j_idx] += 2.0 * w[idx] * a[k - j_idx]
        
        # Nesterov momentum step
        v = momentum * v - lr * g
        a_trial = a + v
        
        # Project onto simplex {x >= 0, sum(x) = 1}
        a_proj = _project_simplex(a_trial)
        
        score = compute_score(a_proj)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_proj.copy()
            a = a_proj
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= lr_decay
        
        if no_improve > 500:
            # Restart with perturbation
            rng = np.random.default_rng(int(time.monotonic() * 1e6))
            a = best_a + rng.standard_normal(n) * 0.01
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 2, 0.005)
            no_improve = 0
        
        if (it + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad+Mom iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def _project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def lbgfs_continuation(a0, time_budget, T_schedule=None, max_iter_per_T=2000):
    """
    L-BFGS optimization in log-space with continuation on soft-max temperature T.
    """
    n = len(a0)
    
    if T_schedule is None:
        T_schedule = [100.0, 50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1]
    
    a = np.maximum(a0.astype(np.float64), 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    for T in T_schedule:
        if time.monotonic() - t0 > time_budget:
            break
        
        z0 = np.log(a)
        t_T = time.monotonic()
        
        def func_and_grad(z):
            if time.monotonic() - t0 > time_budget:
                return 1e15, np.zeros_like(z)
            # Softmax
            z_shift = z - z.max()
            a_local = np.exp(z_shift)
            a_local = a_local / a_local.sum()
            
            # Compute soft-max of auto-correlation
            b = fast_convolve(a_local)
            max_b = b.max()
            exp_b = np.exp((b - max_b) / T)
            log_sum_exp = max_b + T * np.log(exp_b.sum())
            obj = 2.0 * n * log_sum_exp
            
            # Gradient
            w = exp_b / exp_b.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-12:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                j_idx = np.arange(js, je + 1)
                grad_a[j_idx] += 2.0 * w[k] * a_local[k - j_idx]
            
            # Chain rule to z-space
            S = np.dot(grad_a, a_local)
            grad_z = a_local * (grad_a - a_local * S) / T  # Wait, need to be careful
            
            # Actually: obj = 2n * LSE(b/T) where LSE(x) = log(sum(exp(x)))
            # d obj / d z_j = 2n * sum_k (w_k * db_k/dz_j)
            # db_k/dz_j = 2 * a_j * (a_{k-j} - a_j * S_b) where S_b = sum_i a_i * ... 
            # This is getting complicated. Let me use a simpler approach.
            
            # Simpler: use finite difference or just the direct gradient
            # grad_a is the gradient of max(b) w.r.t. a (soft-max version)
            # d a_j / d z_i = a_j * (delta[j,i] - a_i)
            # So grad_z_i = sum_j grad_a_j * a_j * (delta[j,i] - a_i)
            #             = a_i * (grad_a_i - sum_j grad_a_j * a_j)
            #             = a_i * (grad_a_i - S)
            
            grad_z = a_local * (grad_a - a_local * S)
            
            return obj, grad_z
        
        try:
            res = minimize(
                func=func_and_grad,
                x0=z0,
                jac=True,
                method='L-BFGS-B',
                options={'maxiter': max_iter_per_T, 'ftol': 1e-14, 
                         'gtol': 1e-8, 'disp': False}
            )
            
            # Extract solution
            z_out = res.x
            a_out = np.exp(z_out - z_out.max())
            a_out = a_out / a_out.sum()
            a_out = np.maximum(a_out, 0)
            a_out = a_out / a_out.sum()
            
            score = compute_score(a_out)
            if score < best_score - 1e-12:
                best_score = score
                best_a = a_out.copy()
                a = a_out
        except Exception as e:
            pass
        
        elapsed = time.monotonic() - t0
        print(f"    L-BFGS T={T} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def peak_shave(a0, time_budget, n_steps=5000):
    """
    Targeted peak shaving: reduce the values of a that contribute most to peaks.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_budget:
            break
        
        b = fast_convolve(a)
        
        # Find the global peak
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        
        if peak_val <= best_score / (2.0 * n) + 1e-15:
            break
        
        # Find the top contributors to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        j_idx = np.arange(js, je + 1)
        contribs = a[j_idx] * a[peak_k - j_idx]
        
        # Get top 5 contributors
        top = np.argsort(contribs)[::-1][:5]
        
        improved = False
        for ci in top:
            i = j_idx[ci]
            j = peak_k - i
            if a[i] < 1e-15:
                continue
            
            # Try reducing a[i] and redistributing
            for factor in [0.95, 0.9, 0.85, 0.8]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-15:
                    continue
                
                # Redistribute to positions that DON'T contribute to the peak
                a_trial = a.copy()
                a_trial[i] = new_val
                
                # Find positions that don't contribute to the peak
                # Position p contributes to peak_k if a[p] * a[peak_k - p] > 0
                # i.e., 0 <= peak_k - p < n, i.e., peak_k - n + 1 <= p <= peak_k
                contributes = np.zeros(n, dtype=bool)
                cs = max(0, peak_k - n + 1)
                ce = min(n - 1, peak_k)
                contributes[cs:ce+1] = True
                
                # Redistribute to non-contributing positions
                mask = ~contributes & (np.arange(n) != i)
                if mask.sum() == 0:
                    # Redistribute to all other positions
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                
                others = a_trial[mask]
                if others.sum() < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                
                score = compute_score(a_trial)
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            
            if improved:
                break
        
        if not improved:
            # Try pairwise reduction
            for ci in top[:3]:
                i = j_idx[ci]
                j = peak_k - i
                if i == j or a[j] < 1e-15 or a[i] < 1e-15:
                    continue
                for f1, f2 in [(0.9, 0.9), (0.85, 0.85), (0.9, 0.8)]:
                    a_trial = a.copy()
                    a_trial[i] *= f1
                    a_trial[j] *= f2
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = compute_score(a_trial)
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            
            if not improved:
                # Try targeting second peak
                b_copy = b.copy()
                b_copy[peak_k] = -np.inf
                peak_k2 = int(np.argmax(b_copy))
                if b[peak_k2] > best_score / (2.0 * n) * 0.9:
                    # Apply same logic to second peak
                    js2 = max(0, peak_k2 - n + 1)
                    je2 = min(n - 1, peak_k2)
                    j_idx2 = np.arange(js2, je2 + 1)
                    contribs2 = a[j_idx2] * a[peak_k2 - j_idx2]
                    top2 = np.argsort(contribs2)[::-1][:3]
                    for ci in top2:
                        i = j_idx2[ci]
                        j = peak_k2 - i
                        if a[i] < 1e-15:
                            continue
                        for factor in [0.9, 0.85]:
                            new_val = a[i] * factor
                            diff = a[i] - new_val
                            a_trial = a.copy()
                            a_trial[i] = new_val
                            mask = np.ones(n, dtype=bool)
                            mask[i] = False
                            others = a_trial[mask]
                            if others.sum() < 1e-15:
                                break
                            a_trial[mask] = others + diff * others / others.sum()
                            a_trial = a_trial / a_trial.sum()
                            score = compute_score(a_trial)
                            if score < best_score - 1e-13:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
                if not improved:
                    break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def diverse_initializations(n, num_starts=8, budget=30.0):
    """
    Generate diverse initial sequences using different methods.
    Returns a list of (sequence, score) pairs.
    """
    results = []
    t0 = time.monotonic()
    
    # 1. Uniform
    a = np.ones(n) / n
    results.append((a, compute_score(a)))
    
    # 2. GS initialization with different seeds
    for seed in range(4):
        if time.monotonic() - t0 > budget:
            break
        a_gs = gs_initialize(n, max_iter=50, seed=seed + 100)
        score = compute_score(a_gs)
        results.append((a_gs, score))
    
    # 3. GW initialization
    for seed in range(2):
        if time.monotonic() - t0 > budget:
            break
        a_gw = gw_initialize(n, max_iter=50, seed=seed + 200)
        score = compute_score(a_gw)
        results.append((a_gw, score))
    
    # 4. Structured: cosine window
    t = np.linspace(0, np.pi, n)
    a_cos = np.cos(t) ** 2 + 0.01
    a_cos = a_cos / a_cos.sum()
    results.append((a_cos, compute_score(a_cos)))
    
    # 5. Structured: exponential decay from center
    center = n // 2
    dist = np.abs(np.arange(n) - center)
    a_exp = np.exp(-dist / (n * 0.3))
    a_exp = a_exp / a_exp.sum()
    results.append((a_exp, compute_score(a_exp)))
    
    results.sort(key=lambda x: x[1])
    return results
