                a1 = hi - gr * (hi - lo)
                a2 = lo + gr * (hi - lo)
                ft1 = np.maximum(f + a1 * (g - f), 0.0)
                ft2 = np.maximum(f + a2 * (g - f), 0.0)
                if ft1.sum() < 1e-15 or ft2.sum() < 1e-15:
                    break
                ft1 = ft1 / ft1.sum()
                ft2 = ft2 / ft2.sum()
                v1 = fast_convolve(ft1).max()
                v2 = fast_convolve(ft2).max()
                if v1 < v2:
                    hi = a2
                else:
                    lo = a1
                mid = (lo + hi) / 2.0
                ftm = np.maximum(f + mid * (g - f), 0.0)
                if ftm.sum() < 1e-15:
                    break
                ftm = ftm / ftm.sum()
                vm = fast_convolve(ftm).max()
                if vm < best_val:
                    best_val = vm
                    best_alpha = mid

        if best_alpha < 1e-10:
            m_idx = min(m_idx + 1, len(m_levels) - 1)
            no_improve += 1
            continue

        f = np.maximum(f + best_alpha * (g - f), 0.0)
        f = f / f.sum()
        m_idx = 0

        conv = fast_convolve(f)
        new_M = float(conv.max())
        if new_M < best_obj - 1e-14:
            best_obj = new_M
            best_f = f.copy()
            if verbose and it % 10 == 0:
                print(f"    LP-FW iter {it}: obj={best_obj:.10f}, level={m_levels[m_idx]}")

        if new_M < 1.0 / n * 1.001:
            break

    return best_f, best_obj


def peak_targeted_descent(f0, time_limit, verbose=False):
    """Peak-targeted coordinate descent with swap operations."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = float(conv.max())
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > 100:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1.0:
            break

        conv = fast_convolve(f)
        M = float(conv.max())
        peak_k = int(np.argmax(conv))

        # Find contributing indices
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        if js > je:
            no_improve += 1
            continue
        contrib_idx = np.arange(js, je + 1)
        products = f[contrib_idx] * f[peak_k - contrib_idx]
        total = products.sum()
        if total < 1e-15:
            break

        # Find best coordinate to reduce
        best_trial = None
        best_trial_obj = M

        # Try reducing top contributing coordinates
        top_contrib = np.argsort(products)[-min(5, len(products)):]
        for ti in top_contrib:
            src = int(contrib_idx[ti])
            for frac in [0.1, 0.2, 0.3, 0.5, 0.7]:
                diff = f[src] * frac
                if diff < 1e-12:
                    continue
                trial = f.copy()
                trial[src] -= diff
                # Distribute to non-contributing positions
                non_contrib = np.ones(n, dtype=bool)
                non_contrib[contrib_idx] = False
                if non_contrib.sum() < 2:
                    non_contrib[:] = True
                    non_contrib[src] = False
                nc_vals = trial[non_contrib]
                nc_sum = nc_vals.sum()
                if nc_sum < 1e-12:
                    continue
                trial[non_contrib] += diff * nc_vals / nc_sum
                trial = np.maximum(trial, 0.0)
                ts = trial.sum()
                if ts < 1e-12:
                    continue
                trial = trial / ts
                obj = fast_convolve(trial).max()
                if obj < best_trial_obj - 1e-14:
                    best_trial_obj = float(obj)
                    best_trial = trial.copy()

        # Also try swaps between contributing and non-contributing
        if best_trial is not None:
            f = best_trial
            if best_trial_obj < best_obj - 1e-14:
                best_obj = best_trial_obj
                best_f = f.copy()
                no_improve = 0
                if verbose and no_improve == 0:
                    pass
        else:
            # Try swap: move mass from src to a random non-contributing position
            found = False
            non_contrib = np.ones(n, dtype=bool)
            non_contrib[contrib_idx] = False
            nc_positions = np.where(non_contrib)[0]
            if len(nc_positions) > 0:
                rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
                for _ in range(min(3, len(nc_positions))):
                    src = int(contrib_idx[np.argmax(products)])
                    dst = int(nc_positions[rng.integers(len(nc_positions))])
                    for frac in [0.1, 0.2, 0.3]:
                        diff = f[src] * frac
                        if diff < 1e-12:
                            continue
                        trial = f.copy()
                        trial[src] -= diff
                        trial[dst] += diff
                        trial = np.maximum(trial, 0.0)
                        trial = trial / trial.sum()
                        obj = fast_convolve(trial).max()
                        if obj < best_obj - 1e-14:
                            f = trial
                            best_obj = float(obj)
                            best_f = f.copy()
                            no_improve = 0
                            found = True
                            break
                    if found:
                        break
            if not found:
                no_improve += 1

    return best_f, best_obj


def mirror_descent(f0, time_limit, verbose=False, lr_init=0.003, decay=0.9995):
    """Mirror descent with adaptive peak weighting."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    best_obj = float(fast_convolve(f).max())
    best_f = f.copy()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    temperature = 0.001
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1.0:
            break

        conv = fast_convolve(f)
        M = float(conv.max())

        # Weighted subgradient over top peaks
        k_actual = min(50, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        wsum = w.sum()
        if wsum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= wsum

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            for m in range(n):
                j = k - m
                if 0 <= j < n:
                    sg[m] += 2.0 * w[i] * f[j]

        # Multiplicative update
        f = f * np.exp(-lr * sg)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        co = float(fast_convolve(f).max())
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    MD iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        lr *= decay
        temperature *= 1.002

        if no_improve > 300:
            break
        if lr < 1e-9:
            break

    return best_f, best_obj


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine_mod(n, freq, amp):
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_multi_cosine(n, n_freqs=4, max_amp=0.25, seed=0):
    rng = np.random.default_rng(seed)
    a = np.ones(n)
    for m in range(1, n_freqs + 1):
        amp = rng.uniform(0.05, max_amp)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * m * np.arange(n) / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_power_law(n, alpha=0.5):
    x = np.arange(1, n + 1)
    a = x ** (-alpha)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_two_level(n, ratio=0.3, period=3, seed=0):
    rng = np.random.default_rng(seed)
    a = np.zeros(n)
    for i in range(n):
        a[i] = 1.0 if (i % period) < int(period * ratio) else 0.0
    # Smooth slightly
    a = a * 0.7 + 0.3 * make_gaussian(n, 0.4)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def coordinate_descent_refine(f0, time_limit, verbose=False):
    """Coordinate-wise optimization: try adjusting individual coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    best_obj = float(fast_convolve(f).max())
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > 200:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1.0:
            break

        conv = fast_convolve(f)
        M = float(conv.max())
        peak_k = int(np.argmax(conv))

        # Try adjusting coordinates that contribute to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contrib = np.arange(js, je + 1)
        if len(contrib) == 0:
            break

        products = f[contrib] * f[peak_k - contrib]
        # Sort by contribution
        order = np.argsort(products)[::-1]
        improved = False

        for idx in order[:min(10, len(order))]:
            m = int(contrib[idx])
            partner = peak_k - m
            
            # Try reducing m and adding to a non-peak-contributing position
            best_local = None
            best_local_val = M
            
            for frac in [0.05, 0.1, 0.2, 0.3]:
                diff = f[m] * frac
                if diff < 1e-12:
                    continue
                trial = f.copy()
                trial[m] -= diff
                # Add to partner's "opposite" side
                target = (n - 1 - m) % n
                trial[target] += diff
                trial = np.maximum(trial, 0.0)
                trial = trial / trial.sum()
                val = float(fast_convolve(trial).max())
                if val < best_local_val - 1e-14:
                    best_local_val = val
                    best_local = trial.copy()

            if best_local is not None:
                f = best_local
                improved = True
                if best_local_val < best_obj - 1e-14:
                    best_obj = best_local_val
                    best_f = f.copy()
                    no_improve = 0
                break

        if not improved:
            no_improve += 1

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization combining LP-Frank-Wolfe (reduced constraints),
    peak-targeted descent, mirror descent, and coordinate refinement.
    Explores multiple lengths and initializations with adaptive time allocation.
    """
    global height_sequence_1
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = try_seq(parent, "parent")
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception as e:
        print(f"Could not load parent: {e}")
        pass

    if parent is None:
        parent = make_gaussian(768, 0.3)
        try_seq(parent, "default_parent")

    print(f"\n=== Phase 1: Diverse Initializations ===")
    t0 = time.monotonic()

    # Generate diverse initializations at multiple lengths
    n_values = [512, 640, 768, 896, 1024, 1280]
    init_pool = []

    for n_try in n_values:
        inits = [
            ('gauss25', make_gaussian(n_try, 0.25)),
            ('gauss35', make_gaussian(n_try, 0.35)),
            ('gauss45', make_gaussian(n_try, 0.45)),
            ('bell50', make_bell(n_try, 0.5)),
            ('bell60', make_bell(n_try, 0.6)),
            ('cos1_2', make_cosine_mod(n_try, 1, 0.2)),
            ('cos2_15', make_cosine_mod(n_try, 2, 0.15)),
            ('mc4', make_multi_cosine(n_try, 4, 0.25, seed=n_try)),
            ('mc6', make_multi_cosine(n_try, 6, 0.2, seed=n_try + 50)),
            ('pow03', make_power_law(n_try, 0.3)),
            ('pow07', make_power_law(n_try, 0.7)),
            ('two3', make_two_level(n_try, 0.33, 3, seed=n_try)),
            ('flat', np.ones(n_try) / n_try),
        ]
        for label, a_init in inits:
            sc = objective_normalized(a_init)
            init_pool.append((sc, a_init, f"{label}_{n_try}"))

    # Add parent and its variants
    for label, p_var in [('parent', parent),
                         ('parent_flip', parent[::-1]),
                         ('parent_shift1', np.roll(parent, 1)),
                         ('parent_shift2', np.roll(parent, 2))]:
        sc = objective_normalized(p_var)
        init_pool.append((sc, p_var, label))

    # Sort by score
    init_pool.sort(key=lambda x: x[0])
    print(f"  Generated {len(init_pool)} initializations")
    print(f"  Top 5: {[(f'{l}', f'{s:.6f}') for s, a, l in init_pool[:5]]}")

    # Evaluate top initializations with the actual evaluator
    for sc, a_init, label in init_pool[:5]:
        try_seq(a_init, f"init_{label}")

    print(f"  Phase 1 time: {time.monotonic() - t0:.1f}s")

    # === Phase 2: LP-FW with reduced constraints on best inits ===
    print(f"\n=== Phase 2: LP-FW (reduced) ===")
    
    # Try top 4 initializations
    top_inits = init_pool[:4]
    total_rem = deadline - time.monotonic() - 400
    per_init_time = max(30, total_rem / max(1, len(top_inits)))

    for idx, (sc, a_init, label) in enumerate(top_inits):
        rem = deadline - time.monotonic() - 300
        if rem < 30:
            break
        t_alloc = min(per_init_time, rem)
        print(f"  LP-FW [{label}] n={len(a_init)}: t={t_alloc:.0f}s (obj={sc:.6f})")
        a_opt, obj = lp_fw_optimize(a_init, time_limit=t_alloc, verbose=True,
                                      use_reduced=True, top_k=40)
        try_seq(a_opt, f"lpfw_reduced_{label}")

    # === Phase 3: LP-FW with full constraints on current best ===
    print(f"\n=== Phase 3: LP-FW (full) on best ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(rem, 120)
            print(f"  LP-FW full: n={len(best_seq)}, t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True,
                                         use_reduced=False)
            try_seq(a_opt, "lpfw_full")

    # === Phase 4: Peak-targeted descent ===
    print(f"\n=== Phase 4: Peak-Targeted Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_alloc = min(rem, 100)
            print(f"  PTD: n={len(best_seq)}, t={t_alloc:.0f}s")
            a_opt, obj = peak_targeted_descent(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "ptd")

    # === Phase 5: Mirror descent ===
    print(f"\n=== Phase 5: Mirror Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 120
        if rem > 20:
            t_alloc = min(rem, 80)
            print(f"  MD: n={len(best_seq)}, t={t_alloc:.0f}s")
            a_opt, obj = mirror_descent(best_seq, time_limit=t_alloc, verbose=True,
                                         lr_init=0.003, decay=0.9995)
            try_seq(a_opt, "md")

    # === Phase 6: Coordinate descent refinement ===
    print(f"\n=== Phase 6: Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t_alloc = min(rem, 60)
            print(f"  CD: n={len(best_seq)}, t={t_alloc:.0f}s")
            a_opt, obj = coordinate_descent_refine(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "cd")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-Length Exploration ===")
    if best_seq is not None:
        base_n = len(best_seq)
        target_lengths = []
        for tn in [512, 640, 768, 896, 1024, 1280]:
            if tn != base_n:
                target_lengths.append(tn)
        target_lengths.sort(key=lambda x: abs(x - base_n))

        for target_n in target_lengths[:3]:
            rem = deadline - time.monotonic() - 80
            if rem < 15:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")

            if sc < best_score + 0.02:
                t_alloc = min(rem * 0.4, 50)
                if t_alloc > 10:
                    print(f"  LP-FW n={target_n}: t={t_alloc:.0f}s")
                    a_opt, obj = lp_fw_optimize(f_interp, time_limit=t_alloc,
                                                 use_reduced=True, top_k=30)
                    try_seq(a_opt, f"lpfw_n{target_n}")

    # === Phase 8: Second round of LP-FW on best with different params ===
    print(f"\n=== Phase 8: Second LP-FW Round ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_alloc = min(rem, 80)
            print(f"  LP-FW round2: n={len(best_seq)}, t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True,
                                         use_reduced=True, top_k=50)
            try_seq(a_opt, "lpfw_r2")

    # === Phase 9: Perturbation + LP-FW ===
    print(f"\n=== Phase 9: Perturbation Exploration ===")
    if best_seq is not None:
        n_best = len(best_seq)
        perturbations = []
        for shift in [1, 2, 3, 5, 8]:
            perturbations.append((f'shift{shift}', np.roll(best_seq, shift)))
        perturbations.append(('flip', best_seq[::-1]))
        for mix in [0.05, 0.1, 0.2]:
            perturbations.append((f'mix{mix}', (1 - mix) * best_seq + mix * (np.ones(n_best) / n_best)))
        for amp in [0.002, 0.005, 0.01]:
            rng = np.random.default_rng(seed + int(amp * 10000) + 7)
            noise = rng.standard_normal(n_best) * amp
            perturbations.append((f'noise{amp}', np.maximum(best_seq + noise, 0)))

        for label, f_pert in perturbations[:8]:
            rem = deadline - time.monotonic() - 40
            if rem < 10:
                break
            f_pert = project_sum1(f_pert)
            t_alloc = min(rem * 0.3, 30)
            if t_alloc < 5:
                break
            print(f"  LP-FW [{label}]: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(f_pert, time_limit=t_alloc,
                                         use_reduced=True, top_k=30)
            try_seq(a_opt, f"pert_{label}")

    # === Phase 10: Final polish ===
    print(f"\n=== Phase 10: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            t_alloc = min(rem * 0.5, 30)
            print(f"  PTD polish: t={t_alloc:.0f}s")
            a_opt, obj = peak_targeted_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_ptd")

        rem = deadline - time.monotonic() - 5
        if rem > 8:
            t_alloc = min(rem, 15)
            print(f"  MD polish: t={t_alloc:.0f}s")
            a_opt, obj = mirror_descent(best_seq, time_limit=t_alloc,
                                         lr_init=0.001, decay=0.9998)
            try_seq(a_opt, "final_md")

    # Fallback
    if best_seq is None:
        best_seq = make_gaussian(768, 0.3)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    elapsed = time.monotonic() - (deadline - max(1.0, float(budget_s) - 30.0))
    print(f"Total elapsed: {elapsed:.1f}s")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
