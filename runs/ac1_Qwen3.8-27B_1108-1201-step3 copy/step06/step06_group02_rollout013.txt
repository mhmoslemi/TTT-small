def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]

def fast_convolve(a):

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def make_gaussian(n, sigma_frac=0.25):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using efficient multiplicative descent
with peak-focused subgradients, targeted coordinate refinement with analytical
delta computation, and multi-length exploration. Avoids expensive LP solves
in favor of faster per-iteration methods to allow more total iterations.
"""

import numpy as np
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_objective(a):
    """Max of full convolution for a (assumed sum=1)."""
    conv = fast_convolve(a)
    return conv.max()


def conv_delta(a, i, j, delta):
    """
    Compute first-order change in convolution when a[i]+=delta, a[j]-=delta.
    Returns array of length 2n-1 with the delta values.
    """
    n = len(a)
    result = np.zeros(2 * n - 1)
    # Change from a[i] += delta: contributes delta * a[k-i] and delta * a[i+k-set]
    # C[k] = sum_l a[l]*a[k-l]
    # dC[k]/da[i] = a[k-i] (if valid) + a[k-i] (if k-i valid) -- actually:
    # dC[k]/da[i] = a[k-i] + a[k-i] = 2*a[k-i] when 0<=k-i<n
    # Wait, let me be precise:
    # C[k] = sum_{l: 0<=l<n, 0<=k-l<n} a[l]*a[k-l]
    # dC[k]/da[i] = a[k-i] * 1_{0<=k-i<n} (from l=i term)
    #             + a[k-i] * 1_{0<=k-i<n} (from k-l=i term, i.e., l=k-i)
    #             = 2*a[k-i] * 1_{0<=k-i<n}
    # unless i = k-i (i.e., k=2i), in which case it's still 2*a[i] (derivative of a[i]^2 is 2*a[i])
    # So always: dC[k]/da[i] = 2*a[k-i] if 0 <= k-i < n, else 0.

    # For a[i] += delta:
    #   ΔC[k] from i: 2*delta*a[k-i] for k in [i, i+n-1] (where 0<=k-i<n means i <= k <= i+n-1)
    # For a[j] -= delta:
    #   ΔC[k] from j: -2*delta*a[k-j] for k in [j, j+n-1]

    # Vectorized:
    # Contribution from i:
    k_min_i = i
    k_max_i = i + n - 1
    k_range_i = np.arange(k_min_i, min(k_max_i, 2 * n - 1) + 1)
    partners_i = k_range_i - i
    result[k_range_i] += 2.0 * delta * a[partners_i]

    # Contribution from j (negative):
    k_min_j = j
    k_max_j = j + n - 1
    k_range_j = np.arange(k_min_j, min(k_max_j, 2 * n - 1) + 1)
    partners_j = k_range_j - j
    result[k_range_j] -= 2.0 * delta * a[partners_j]

    return result


def multiplicative_descent_v3(f, time_limit, top_k=15, verbose=False):
    """
    Fast multiplicative descent with peak-focused weighted subgradient.
    Uses soft-softmax weighting over top-k peaks.
    """
    f = np.maximum(f.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    lr = 0.004
    decay = 0.9998
    temperature = 0.0002
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 500 == 0 and time.monotonic() - t0 > time_limit - 3:
            break

        conv = fast_convolve(f)
        M = conv.max()

        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        peak_values = conv[peak_indices]

        scaled = peak_values / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peak_indices))
            w[np.argmax(peak_values)] = 1.0
        else:
            w /= w_sum

        sg = np.zeros(n)
        for idx in range(len(peak_indices)):
            if w[idx] < 1e-14:
                continue
            k = int(peak_indices[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            i_vals = np.arange(js, je + 1)
            partners = k - i_vals
            sg[i_vals] += 2.0 * w[idx] * f[partners]

        f = f * np.exp(-lr * sg)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        co = fast_convolve(f).max()
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    MD3 iter {it}: obj={best_obj:.10f}, lr={lr:.2e}, temp={temperature:.2e}")
        else:
            no_improve += 1

        lr *= decay
        temperature = max(temperature * 1.004, 1e-7)

        if no_improve > 120:
            rng = np.random.default_rng(it * 43 + 11)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-15)
            f = f / f.sum()
            lr = max(lr, 0.003)
            temperature = 0.0002
            no_improve = 0

        if lr < 1e-7:
            break

    return best_f, best_obj


def targeted_coordinate_refine(f, time_limit, verbose=False):
    """
    Best-improvement coordinate descent using analytical convolution deltas.
    For each coordinate pair (i,j), try moving mass and check if max(conv) decreases.
    Uses the fast delta computation to avoid full FFT in many cases.
    """
    f = np.maximum(f.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve_passes = 0
    pass_count = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        if pass_count % 20 == 0:
            print(f"    TCR pass {pass_count}: obj={best_obj:.10f}")
        pass_improvements = 0
        rem = time_limit - (time.monotonic() - t0)

        conv = fast_convolve(f)
        M = conv.max()
        peak_k = int(np.argmax(conv))

        # Focus on coordinates that contribute to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contributors = np.arange(js, je + 1)
        non_contributors = np.setdiff1d(np.arange(n), contributors)

        # Get top 5 contributors by product
        products = f[contributors] * f[peak_k - contributors]
        top_contrib = contributors[np.argsort(products)[-8:]]
        top_contrib = top_contrib[np.argsort(f[top_contrib])[::-1]]

        time_per_trial = rem / (len(top_contrib) * 4 * 2)
        if time_per_trial < 0.001:
            break

        for src in top_contrib:
            if time.monotonic() - t0 > time_limit - 2:
                break
            if f[src] < 1e-8:
                continue

            # Try moving to non-contributors with smallest values
            if len(non_contributors) > 0:
                nc_sorted = non_contributors[np.argsort(f[non_contributors])[:4]]
            else:
                nc_sorted = np.array([])

            for tgt in nc_sorted:
                for frac in [0.2, 0.5, 0.8]:
                    delta = f[src] * frac * 0.5  # half to be safe
                    if delta < 1e-12:
                        continue

                    # Use analytical delta
                    d_conv = conv_delta(f, src, tgt, delta)
                    new_conv = conv + d_conv
                    new_max = new_conv.max()

                    if new_max < best_obj - 1e-12:
                        # Verify with actual computation
                        trial = f.copy()
                        trial[src] -= delta
                        trial[tgt] += delta
                        trial = np.maximum(trial, 0.0)
                        ts = trial.sum()
                        trial = trial / ts
                        actual_max = fast_convolve(trial).max()
                        if actual_max < best_obj - 1e-13:
                            f = trial
                            conv = fast_convolve(f)
                            best_obj = conv.max()
                            best_f = f.copy()
                            pass_improvements += 1
                            break
                else:
                    continue
                break

        if pass_improvements == 0:
            no_improve_passes += 1
            # Try random perturbation
            rng = np.random.default_rng(pass_count * 23 + 5)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.0015, 0.0)
            f = f / f.sum()
            if no_improve_passes > 15:
                break
        else:
            no_improve_passes = 0

    return best_f, best_obj

def greedy_direction_fw(f, time_limit, top_k=10, verbose=False):
    """
    Greedy Frank-Wolfe: find a direction by analyzing peak contributions,
    then do a line search. Cheaper than LP-based FW.
    """
    f = np.maximum(f.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 200 == 0 and time.monotonic() - t0 > time_limit - 3:
            break

        conv = fast_convolve(f)
        M = conv.max()
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]

        # Compute weighted blame
        blame = np.zeros(n)
        for k in peak_indices:
            w_k = conv[int(k)] / M
            js = max(0, int(k) - n + 1)
            je = min(n - 1, int(k))
            i_vals = np.arange(js, je + 1)
            partners = int(k) - i_vals
            blame[i_vals] += w_k * f[partners]

        # Direction: reduce high-blame, increase low-blame
        q = max(4, n // 16)
        top_blame = np.argsort(blame)[-q:]
        low_blame = np.argsort(blame)[:q]

        direction = np.zeros(n)
        direction[top_blame] = -1.0
        direction[low_blame] = 1.0

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            trial = f + alpha * direction
            trial = project_simplex(trial)
            val = fast_convolve(trial).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha < 1e-6:
            no_improve += 1
            if no_improve > 30:
                break
            # Perturb and retry
            rng = np.random.default_rng(it * 7 + 3)
            f = project_simplex(f + rng.standard_normal(n) * 0.002)
            continue

        f = project_simplex(f + best_alpha * direction)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        co = fast_convolve(f).max()
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    GFW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.25):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine_mod(n, freq, amp):
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_multi_cosine(n, n_freqs=4, max_amp=0.2, seed=0):
    rng = np.random.default_rng(seed)
    a = np.ones(n).astype(np.float64)
    for m in range(1, n_freqs + 1):
        amp = rng.uniform(0.03, max_amp)
        phase = rng.uniform(0, 2.0 * np.pi)
        a = a + amp * np.cos(2.0 * np.pi * m * np.arange(n) / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining fast multiplicative descent,
    targeted coordinate refinement with analytical deltas, greedy FW,
    and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)}, max_conv={compute_objective(a_norm):.8f})")
        return score

    # === Phase 0: Load parent ===
    print("=== Phase 0: Load Parent ===")
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = try_seq(parent, "parent")
            print(f"  Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception as e:
        print(f"  No parent: {e}")

    if parent is None:
        parent = make_gaussian(768)
        try_seq(parent, "gaussian_init")

    # === Phase 1: Fast multiplicative descent ===
    print("\n=== Phase 1: Multiplicative Descent ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t_alloc = min(rem * 0.28, 350.0)
        print(f"  Alloc={t_alloc:.0f}s")
        a_opt, obj = multiplicative_descent_v3(parent, time_limit=t_alloc, top_k=15, verbose=True)
        try_seq(a_opt, "md3")

    # === Phase 2: Targeted coordinate refinement ===
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 80:
            print("\n=== Phase 2: Targeted Coordinate Refinement ===")
            t_alloc = min(rem * 0.22, 250.0)
            print(f"  Alloc={t_alloc:.0f}s")
            a_opt, obj = targeted_coordinate_refine(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "tcr")

    # === Phase 3: Greedy FW ===
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 60:
            print("\n=== Phase 3: Greedy Frank-Wolfe ===")
            t_alloc = min(rem * 0.18, 180.0)
            print(f"  Alloc={t_alloc:.0f}s")
            a_opt, obj = greedy_direction_fw(best_seq, time_limit=t_alloc, top_k=8, verbose=True)
            try_seq(a_opt, "gfw")

    # === Phase 4: Another round of MD on best ===
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 80:
            print("\n=== Phase 4: Second MD Round ===")
            t_alloc = min(rem * 0.2, 200.0)
            print(f"  Alloc={t_alloc:.0f}s")
            a_opt, obj = multiplicative_descent_v3(best_seq, time_limit=t_alloc, top_k=10, verbose=True)
            try_seq(a_opt, "md3b")

    # === Phase 5: Random pair descent for fine-tuning ===
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 60:
            print("\n=== Phase 5: Random Pair Descent ===")
            t_alloc = min(rem * 0.15, 150.0)
            print(f"  Alloc={t_alloc:.0f}s")
            a_opt, obj = random_pair_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "rnd_pair")

    # === Phase 6: Multi-length exploration ===
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 120:
            print("\n=== Phase 6: Multi-Length Exploration ===")
            base_n = len(best_seq)
            candidates_n = sorted(set([
                base_n - 64, base_n - 32, base_n - 16, base_n - 8,
                base_n + 8, base_n + 16, base_n + 32, base_n + 64,
                800, 850, 900, 950, 1024
            ]))
            candidates_n = [x for x in candidates_n if 384 <= x <= 2048 and x != base_n]

            for tn in candidates_n:
                rem = deadline - time.monotonic()
                if rem < 40:
                    break
                f_interp = interpolate_sequence(best_seq, tn)
                sc = try_seq(f_interp, f"interp_n{tn}")

                if sc < best_score + 0.08:
                    t_alloc = min(rem * 0.12, 50.0)
                    if t_alloc > 8:
                        a_opt, obj = multiplicative_descent_v3(f_interp, time_limit=t_alloc, top_k=10)
                        try_seq(a_opt, f"md_n{tn}")

    # === Phase 7: Final polish ===
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 40:
            print("\n=== Phase 7: Final Polish ===")
            t_alloc = min(rem * 0.4, 60.0)
            if t_alloc > 10:
                a_opt, obj = targeted_coordinate_refine(best_seq, time_limit=t_alloc)
                try_seq(a_opt, "polish_tcr")

        rem = deadline - time.monotonic()
        if rem > 30:
            t_alloc = min(rem * 0.4, 40.0)
            if t_alloc > 8:
                a_opt, obj = multiplicative_descent_v3(best_seq, time_limit=t_alloc, top_k=5)
                try_seq(a_opt, "polish_md")

        rem = deadline - time.monotonic()
        if rem > 20:
            t_alloc = min(rem * 0.5, 20.0)
            if t_alloc > 5:
                a_opt, obj = greedy_direction_fw(best_seq, time_limit=t_alloc, top_k=5)
                try_seq(a_opt, "polish_gfw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)}, max_conv={compute_objective(best_seq):.8f})")
    return [float(x) for x in best_seq]
