def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_sparse_conv_matrix(f, n):
    """Build sparse matrix A where A[k,j] = f[j] if 0<=k-j<n else 0."""
    rows = np.tile(np.arange(n), n)
    cols = np.repeat(np.arange(n), n)
    data = np.tile(f, n)
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_direction_sparse(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0, using sparse matrix."""
    c = -np.ones(n)
    A = build_sparse_conv_matrix(f, n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def solve_lp_direction_dense(f, M, n, n_constraints=None):
    """Solve LP with optional subset of constraints for speed."""
    if n_constraints is not None and n_constraints < 2 * n - 1:

    for j in range(n):

        end = min(2 * n - 2, j + n - 1)
        for k_pos, k in enumerate(top_idx):

    b = np.full(len(top_idx), M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def golden_section_search(f_func, a, b, tol=1e-8, max_iter=50):
    """Golden section search for minimization on [a, b]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f_func(c)
    fd = f_func(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f_func(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f_func(d)
    return (a + b) / 2.0


def lse_smoothed_obj(f, c, beta):
    """Compute log-sum-exp smoothed max of convolution."""
    scaled = beta * c
    scaled -= scaled.max()
    return (1.0 / beta) * (np.log(np.sum(np.exp(scaled))) + beta * c.max())


def compute_lse_gradient(f, c, beta):
    """Compute gradient of LSE smoothed max w.r.t. f."""
    n = len(f)
    scaled = beta * c
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()
    
    # Gradient: g_j = 2 * sum_k w_k * f_{k-j} (valid indices)
    # This is 2 * cross_correlation(w, f) at appropriate indices
    w_rev = w[::-1].copy()
    conv_len = n + len(w_rev) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(F * W, fft_size)[:conv_len]
    
    indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
    g = 2.0 * conv[indices]
    return g

def create_asymmetric_init(n, rng, mode=0):
    """Create various asymmetric initializations."""
    x = np.arange(n, dtype=np.float64)
    
    if mode == 0:
        # Exponential decay from left
        a = np.exp(-x / (n * 0.4))
    elif mode == 1:
        # Exponential decay from right
        a = np.exp(-(n - 1 - x) / (n * 0.4))
    elif mode == 2:
        # Shifted Gaussian
        center = n * 0.35
        a = np.exp(-0.5 * ((x - center) / (n * 0.2)) ** 2)
    elif mode == 3:
        # Ramp up then flat
        a = np.minimum(x / (n * 0.5), 1.0)
    elif mode == 4:
        # Flat then ramp down
        a = np.minimum((n - 1 - x) / (n * 0.5), 1.0)
    elif mode == 5:
        # Bimodal (two bumps)
        c1, c2 = n * 0.3, n * 0.7
        a = np.exp(-0.5 * ((x - c1) / (n * 0.12)) ** 2) + np.exp(-0.5 * ((x - c2) / (n * 0.12)) ** 2)
    elif mode == 6:
        # Linear ramp
        a = x + 1.0
    elif mode == 7:
        # S-curve
        a = 1.0 / (1.0 + np.exp(-(x - n * 0.5) / (n * 0.08)))
    elif mode == 8:
        # Weighted toward center but asymmetric
        center = n * 0.5
        a = np.exp(-0.5 * ((x - center) / (n * 0.25)) ** 2) * (1.0 + 0.3 * np.sin(np.pi * x / n))
    else:
        a = np.exp(-0.5 * ((x - n * 0.5) / (n * 0.25)) ** 2)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_gaussian(n, sigma_frac=0.3):
    """Create a Gaussian initial sequence."""
    x = np.arange(n, dtype=np.float64) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_raised_cosine(n):
    """Hann window."""
    x = np.linspace(-1, 1, n)
    a = 0.5 * (1 + np.cos(np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_smooth = np.maximum(f_interp, 0)
    s = f_smooth.sum()
    if s > 1e-15:
        f_smooth /= s
    return f_smooth

        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except NameError:
        pass
    except Exception as e:
        print(f"Parent loading error: {e}")

        blend = 0.3 + 0.1 * trial
        a_mix = (1 - blend) * parent + blend * asym
        a_mix = np.maximum(a_mix, 0)
        a_mix = a_mix / a_mix.sum()
        init_list.append((f"asym_{mode}", a_mix))
    
    # Pure asymmetric shapes
    for mode in range(9):
        a_init = create_asymmetric_init(n_parent, rng, mode=mode)
        init_list.append((f"pure_asym_{mode}", a_init))
    
    # Standard shapes
    init_list.append(("gauss_narrow", make_gaussian(n_parent, 0.2)))
    init_list.append(("gauss_wide", make_gaussian(n_parent, 0.4)))
    init_list.append(("raised_cos", make_raised_cosine(n_parent)))
    
    # Evaluate all
    best_init = None
    best_init_obj = float('inf')
    for name, a in init_list:
        a = np.maximum(a, 0)
        a = a / a.sum()
        c = fast_convolve(a)
        obj = 2 * len(a) * c.max()
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a.copy()
        score = try_seq(a, f"init_{name}")

            break

        a_pert = a_pert / a_pert.sum()
