def lp_refine(a, M=None, time_limit=30):
    """
    Given a (probability vector), solve LP to find b that maximizes sum(b)
    subject to conv(a,b) <= M elementwise and b >= 0.
    Then return the rescaled b.
    """
    n = len(a)
    if M is None:
        b_conv = fast_convolve(a)
        M = b_conv.max()
    
    # conv(a,b)[k] = sum_i c[k,i] * b[i] where c[k,i] = a[k-i] if valid
    # We want: c @ b <= M for all k
    # Maximize: sum(b)
    
    # Build the constraint matrix (sparse)
    from scipy.sparse import csr_matrix
    import scipy.optimize as opt
    
    # c is (2n-1) x n, banded
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        lo = max(0, k-n+1)
        hi = min(n-1, k)
        for i in range(lo, hi+1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k-i])
    
    c_mat = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Maximize sum(b) = minimize -sum(b)
    c_obj = -np.ones(n)
    
    res = opt.linprog(c_obj, A_ub=c_mat, b_ub=np.ones(2*n-1)*M, 
                      bounds=[(0, None)]*n, method='highs')
    
    if res.status == 0:
        b = res.x
        b_sum = b.sum()
        if b_sum > 1e-10:
            b = b / b_sum
        return b
    return None

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])

- y ≥ 0

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative refinement.

Core idea: At each step, solve a linear program to find the "flattest" 
direction b (maximizing sum(b) subject to conv(a,b) <= current peak), 
then interpolate toward b. This is the approach from the SOTA described
in the problem statement, implemented with scipy's HiGHS solver.

Additional features:
- Subgradient initialization
- Adaptive step size
- Multi-length exploration
- Coordinate descent polishing

import numpy as np
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog

DEFAULT_N = 1152

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def eval_local(a):
    """Compute objective locally (assumes a is non-negative)."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))

def build_lp_indices(n):
    """Precompute row/col indices for the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)

def solve_lp(a, M, row_idx, col_idx, n, time_limit=20.0):
    """
    Solve: max sum(b) s.t. C @ b <= M, b >= 0
    where C[k,i] = a[k-i] for valid (k,i).
    Returns b (probability vector) or None.
    """
    # Build the sparse matrix with current a values
    n_rows = 2 * n - 1
    n_cols = n
    nnz = len(row_idx)
    
    # Map (k, i) -> a[k-i]
    vals = np.zeros(nnz)
    k_arr = row_idx.astype(np.int64)
    i_arr = col_idx.astype(np.int64)
    vals = a[k_arr - i_arr]
    
    C = csr_matrix((vals, (row_idx, col_idx)), shape=(n_rows, n_cols))
    
    c_obj = -np.ones(n_cols)
    b_ub = np.full(n_rows, M)
    
    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub,
                      bounds=[(0, None)] * n_cols,
                      method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0 and res.x is not None:
            b = np.maximum(res.x, 0.0)
            s = b.sum()
            if s > 1e-10:
                return b / s
            return None
    except Exception as e:
        print(f"  LP error: {e}")
    return None

def lp_refine(a, time_limit, t_init=0.1, min_t=0.001, stagnation_limit=30, seed=42):

    """

    M_best = M_cur

            stagnation += 5

                conv = fast_convolve(a_cur)
                M_cur = conv.max()
                stagnation = 0
            continue
        
        # Interpolate

        a_trial = np.maximum(a_trial, 0)

            M_cur = M_trial
            t = min(t * 1.1, 0.5)  # Slightly increase step

            if M_trial < M_best - 1e-12:
                M_best = M_trial
                a_best = a_cur.copy()
                print(f"    LP iter {iteration}: M={M_best:.8f} (t={t:.4f})")

            t = max(min_t, t * 0.7)

                stagnation = 0

        b_max = b.max()

        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:

        else:
            w /= w_sum

def coordinate_polish(a0, time_limit):
    """Coordinate descent: try scaling each element by multipliers."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
    return best_a, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative refinement 
(the SOTA approach) combined with subgradient initialization, adaptive step 
sizes, multi-length exploration, and coordinate descent polishing.

Core new idea: At each LP step, solve max sum(b) s.t. conv(a,b) <= M, b >= 0,
which gives the "flattest" feasible direction. Interpolating toward this 
direction provably non-increases the peak for small steps, and in practice 
drives the objective down significantly faster than pure subgradient.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_local(a):
    """Compute objective 2n*max(conv)/(sum)^2 locally."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def build_lp_indices(n):
    """Precompute row/col indices for the LP constraint matrix (convolution structure)."""
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            rows_list.append(k)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp_step(a, M, row_idx, col_idx, n, time_limit=15.0):
    """
    Solve LP: max sum(b) s.t. C@b <= M*1, b >= 0
    where C[k,i] = a[k-i] for valid indices (convolution matrix).
    Returns probability vector b or None.
    """
    nnz = len(row_idx)
    k_arr = row_idx.astype(np.int64)
    i_arr = col_idx.astype(np.int64)
    vals = a[k_arr - i_arr]

    C = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0 and res.x is not None:
            b = np.maximum(res.x, 0.0)
            s = b.sum()
            if s > 1e-10:
                return b / s
    except Exception:
        pass
    return None


def lp_refine(a, time_limit, t_init=0.15, min_t=0.005, stagnation_limit=25, seed=42):
    """
    Iterative LP-based refinement on the simplex.
    a: probability vector (sum=1, all >= 0)
    Returns best probability vector found.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a)
    a_cur = a.copy()
    a_best = a.copy()

    conv = fast_convolve(a_cur)
    M_cur = conv.max()
    M_best = M_cur

    row_idx, col_idx = build_lp_indices(n)

    t = t_init
    stagnation = 0
    iteration = 0
    print_count = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 2:
            break

        iteration += 1

        # Solve LP with time budget
        lp_time = min(15.0, remaining * 0.4)
        b_new = solve_lp_step(a_cur, M_cur, row_idx, col_idx, n, time_limit=lp_time)

        if b_new is None:
            t = max(min_t, t * 0.5)
            stagnation += 3
            if stagnation > stagnation_limit:
                a_cur = project_simplex(a_best + rng.standard_normal(n) * 0.01)
                conv = fast_convolve(a_cur)
                M_cur = conv.max()
                stagnation = 0
                t = t_init
            continue

        # Check if b_new is essentially the same as a_cur (fixed point)
        diff = np.abs(b_new - a_cur).max()
        if diff < 1e-8:
            stagnation += 5
            if stagnation > stagnation_limit:
                a_cur = project_simplex(a_best + rng.standard_normal(n) * 0.01)
                conv = fast_convolve(a_cur)
                M_cur = conv.max()
                stagnation = 0
                t = t_init
            continue

        # Try interpolation with current step size
        best_trial_M = M_cur
        best_trial_a = a_cur

        for trial_t in [t, t * 0.5, t * 0.25]:
            if trial_t < min_t:
                continue
            a_trial = (1.0 - trial_t) * a_cur + trial_t * b_new
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            conv_trial = fast_convolve(a_trial)
            M_trial = conv_trial.max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_trial_a = a_trial.copy()

        if best_trial_M < M_cur - 1e-14:
            a_cur = best_trial_a
            M_cur = best_trial_M
            t = min(t * 1.05, 0.4)
            stagnation = 0
            if M_cur < M_best - 1e-14:
                M_best = M_cur
                a_best = a_cur.copy()
                print_count += 1
                if print_count % 10 == 1:
                    print(f"    LP iter {iteration}: M={M_best:.8f}, t={t:.4f}")
        else:
            t = max(min_t, t * 0.7)
            stagnation += 1
            if stagnation > stagnation_limit:
                a_cur = project_simplex(a_best + rng.standard_normal(n) * 0.005)
                conv = fast_convolve(a_cur)
                M_cur = conv.max()
                stagnation = 0
                t = t_init

    return a_best


def subgradient_optimize(a0, time_limit, lr_init=0.005, momentum=0.9,
                         temp_init=0.001, temp_growth=1.02,
                         kick_interval=2000, kick_size=0.003,
                         n_top=None, seed=42):
    """Projected subgradient with Nesterov momentum and temperature-weighted peaks."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    if S < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / S
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    cur_temp = temp_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)

        scaled = b / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
        else:
            w /= w_sum

        if n_top is not None and n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx].copy()
            ws = w_sel.sum()
            if ws > 1e-30:
                w_sel /= ws
                w_new = np.zeros(len(b))
                w_new[top_idx] = w_sel
                w = w_new

        g = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += w[k] * a[k - idx]

        v = momentum * v - cur_lr * g
        a_new = project_simplex(a + v)

        b_new = fast_convolve(a_new)
        obj_new = b_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve > kick_interval:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.7, lr_init * 0.1)
            no_improve = 0
        elif no_improve > kick_interval // 2:
            cur_lr *= 0.95

        cur_temp *= temp_growth
        if cur_temp > 0.5:
            cur_temp = 0.5

    return best_a, best_obj


def coordinate_polish(a0, time_limit):
    """Coordinate descent: try scaling each element by multipliers."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    multipliers = np.array([0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0])

    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.9999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                ss = a_trial.sum()
                if ss < 1e-10:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                ss = a.sum()
                if ss > 1e-10:
                    a = a / ss
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian_init(n, sigma=0.3):
    """Gaussian-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_cosine_init(n):
    """Raised cosine window."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    """Tukey window."""
    a = np.ones(n)
    taper = max(1, int(r * n // 2))
    if taper < n // 2:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1 + np.cos(np.pi * (2 * t / r - 1)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-10)
    return a / a.sum()

    if t_alloc > 10:
        print(f"  Allocating {t_alloc:.0f}s")
        a_sg, obj_sg = subgradient_optimize(
            parent, time_limit=t_alloc,
            lr_init=0.005, momentum=0.9,
            temp_init=0.001, temp_growth=1.02,
            kick_interval=1500, kick_size=0.003,
            n_top=max(20, n_parent // 10), seed=42
        )
        a_sg = np.maximum(a_sg, 0.0)
        ss = a_sg.sum()
        if ss > 0.01:
            a_sg /= ss
            sc = float(evaluate_sequence(a_sg.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_sg.copy()
                print(f"  NEW BEST [sg_init]: {sc:.6f}")

    # If subgradient didn't improve, use parent
    if best_seq is None:
        best_seq = parent.copy()
        try:
            best_score = float(evaluate_sequence(best_seq.tolist()))
        except Exception:
            best_score = eval_local(best_seq)
        print(f"  Using parent: score={best_score:.6f}")

    # === Phase 2: LP-based refinement (main) ===
    print(f"\n=== Phase 2: LP refinement (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.45, 650.0)
    if t_alloc > 20:
        print(f"  n={n_best}, allocating {t_alloc:.0f}s")
        a_lp = lp_refine(best_seq, time_limit=t_alloc, t_init=0.15, seed=42)
        a_lp = np.maximum(a_lp, 0.0)
        ss = a_lp.sum()
        if ss > 0.01:
            a_lp /= ss
            sc = float(evaluate_sequence(a_lp.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_lp.copy()
                print(f"  NEW BEST [lp_main]: {sc:.6f}")

    # Second LP round with different params
    rem = deadline - time.monotonic()
    if rem > 60:
        t_alloc2 = min(rem * 0.2, 200.0)
        print(f"  Second LP round, t={t_alloc2:.0f}s")
        # Perturb slightly and re-run LP
        rng = np.random.default_rng(777)
        a_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)
        a_lp2 = lp_refine(a_pert, time_limit=t_alloc2, t_init=0.1, seed=777)
        a_lp2 = np.maximum(a_lp2, 0.0)
        ss = a_lp2.sum()
        if ss > 0.01:
            a_lp2 /= ss
            sc = float(evaluate_sequence(a_lp2.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_lp2.copy()
                print(f"  NEW BEST [lp_round2]: {sc:.6f}")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length (best={best_score:.6f}) ===")
    target_lengths = []
    for factor in [0.75, 1.25, 1.5, 2.0]:
        tl = int(n_parent * factor)
        tl = max(256, min(4096, tl))
        if tl != n_best and tl not in target_lengths:
            target_lengths.append(tl)

    target_lengths.sort(key=lambda x: abs(x - n_best))

    for target_n in target_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 90:
            break
        f_interp = interpolate_sequence(best_seq, target_n)
        t_alloc = min(rem * 0.25, 180.0)
        if t_alloc < 40:
            continue
        print(f"  Exploring n={target_n}, t={t_alloc:.0f}s")
        a_opt = lp_refine(f_interp, time_limit=t_alloc, t_init=0.15, seed=42 + target_n)
        a_opt = np.maximum(a_opt, 0.0)
        ss = a_opt.sum()
        if ss > 0.01:
            a_opt /= ss
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST [lp_n{target_n}]: {sc:.6f}")

    # === Phase 4: Different structural initializations ===
    print(f"\n=== Phase 4: Structural inits (best={best_score:.6f}) ===")
    inits = {
        "gauss02": make_gaussian_init(n_best, 0.2),
        "gauss03": make_gaussian_init(n_best, 0.3),
        "cosine": make_cosine_init(n_best),
        "tukey": make_tukey_init(n_best, 0.5),
    }
    for name, init_a in inits.items():
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(rem * 0.15, 80.0)
        if t_alloc < 20:
            continue
        print(f"  Trying {name}, t={t_alloc:.0f}s")
        a_opt = lp_refine(init_a, time_limit=t_alloc, t_init=0.2, seed=1000 + hash(name) % 9000)
        a_opt = np.maximum(a_opt, 0.0)
        ss = a_opt.sum()
        if ss > 0.01:
            a_opt /= ss
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST [{name}]: {sc:.6f}")

    # === Phase 5: Final polish ===
    print(f"\n=== Phase 5: Final polish (best={best_score:.6f}) ===")
    n_final = len(best_seq)

    # Coordinate descent
    rem = deadline - time.monotonic()
    if rem > 30:
        t_cd = min(rem * 0.3, 100.0)
        print(f"  CD polish, t={t_cd:.0f}s")
        a_cd, _ = coordinate_polish(best_seq, time_limit=t_cd)
        a_cd = np.maximum(a_cd, 0.0)
        ss = a_cd.sum()
        if ss > 0.01:
            a_cd /= ss
            sc = float(evaluate_sequence(a_cd.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_cd.copy()
                print(f"  NEW BEST [cd]: {sc:.6f}")

    # Fine subgradient
    rem = deadline - time.monotonic()
    if rem > 20:
        t_sg = min(rem * 0.3, 60.0)
        print(f"  Fine SG, t={t_sg:.0f}s")
        a_fin, _ = subgradient_optimize(
            best_seq, time_limit=t_sg,
            lr_init=0.001, momentum=0.95,
            temp_init=0.0001, temp_growth=1.005,
            kick_interval=3000, kick_size=0.001,
            n_top=max(30, n_final // 5), seed=99999
        )
        a_fin = np.maximum(a_fin, 0.0)
        ss = a_fin.sum()
        if ss > 0.01:
            a_fin /= ss
            sc = float(evaluate_sequence(a_fin.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_fin.copy()
                print(f"  NEW BEST [fin_sg]: {sc:.6f}")

    # Final LP round
    rem = deadline - time.monotonic()
    if rem > 40:
        t_lp = min(rem * 0.4, 120.0)
        print(f"  Final LP, t={t_lp:.0f}s")
        a_final_lp = lp_refine(best_seq, time_limit=t_lp, t_init=0.1,
                               min_t=0.01, stagnation_limit=40, seed=55555)
        a_final_lp = np.maximum(a_final_lp, 0.0)
        ss = a_final_lp.sum()
        if ss > 0.01:
            a_final_lp /= ss
            sc = float(evaluate_sequence(a_final_lp.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_final_lp.copy()
                print(f"  NEW BEST [final_lp]: {sc:.6f}")

    # Coord descent one more time on the absolute best
    rem = deadline - time.monotonic()
    if rem > 20:
        t_cd2 = min(rem * 0.4, 50.0)
        a_cd2, _ = coordinate_polish(best_seq, time_limit=t_cd2)
        a_cd2 = np.maximum(a_cd2, 0.0)
        ss = a_cd2.sum()
        if ss > 0.01:
            a_cd2 /= ss
            sc = float(evaluate_sequence(a_cd2.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_cd2.copy()
                print(f"  NEW BEST [cd2]: {sc:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()

    # Final verification
    try:
        fs = float(evaluate_sequence([float(x) for x in best_seq]))
        if np.isfinite(fs):
            best_score = fs
        else:
            best_seq = parent.copy()
            best_score = float(evaluate_sequence(parent.tolist()))
    except Exception:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
