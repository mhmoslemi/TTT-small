def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

        conv_mid[-1] = -1e10
        peak_k = int(np.argmax(conv_mid))
        peak_val = conv[peak_k]
        
        if peak_val <= best_score / (2.0 * n) + 1e-14:
            break
        
        # Find top contributors to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        products = a[idx] * a[peak_k - idx]
        
        # Top 5 contributors
        top = np.argsort(products)[::-1][:5]
        improved = False
        
        for ci in top:
            i = idx[ci]
            j = peak_k - i
            if a[i] < 1e-14:
                continue
            
            # Try reducing a[i]
            for factor in [0.95, 0.9, 0.85, 0.8, 0.7]:
                trial = a.copy()
                diff = a[i] * (1 - factor)
                trial[i] = a[i] * factor
                # Redistribute diff to other positions proportionally
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                osum = others.sum()
                if osum < 1e-14:
                    break
                trial[mask] = others + diff * others / osum
                s = trial.sum()
                trial = trial / s
                score = 2.0 * n * fast_convolve(trial).max()
                if score < best_score - 1e-13:
                    a = trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            
            if improved:
                break
        
        # Also try reducing both i and j simultaneously
        if not improved:
            for ci in top:
                i = idx[ci]
                j = peak_k - i
                if i == j or a[j] < 1e-14:
                    continue
                for f1 in [0.9, 0.8, 0.7]:
                    for f2 in [0.9, 0.8, 0.7]:
                        trial = a.copy()
                        trial[i] *= f1
                        trial[j] *= f2
                        trial = np.maximum(trial, 0)
                        s = trial.sum()
                        if s < 1e-14:
                            break
                        trial = trial / s
                        score = 2.0 * n * fast_convolve(trial).max()
                        if score < best_score - 1e-13:
                            a = trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
        
        if not improved:
            break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def generate_structured_init(n, rng, init_type='random'):
    """Generate a structured initial sequence."""
    t = np.linspace(0, 1, n)
    if init_type == 'uniform':
        a = np.ones(n)
    elif init_type == 'hann':
        a = np.sin(np.pi * t) ** 2
    elif init_type == 'hamming':
        a = 0.54 - 0.46 * np.cos(2 * np.pi * t)
    elif init_type == 'bimodal':
        a = (1 - np.sin(np.pi * t)) ** 2
    elif init_type == 'cos_mod':
        freq = rng.integers(1, 5)
        a = 1 + 0.3 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2*np.pi))
    elif init_type == 'random_smooth':
        # Sum of a few random sinusoids
        a = np.ones(n)
        for _ in range(rng.integers(2, 6)):
            freq = rng.uniform(0.5, 10)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * freq * t + phase)
    elif init_type == 'gauss':
        a = np.exp(-((t - 0.5)**2) / (2 * rng.uniform(0.02, 0.2)**2))
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a
