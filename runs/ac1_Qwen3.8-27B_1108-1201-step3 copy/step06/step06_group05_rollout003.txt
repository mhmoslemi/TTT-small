"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS-B with log-sum-exp smoothing and softmax parameterization (fast quasi-Newton convergence)
2. Spectral-domain initialization for flat auto-correlation design

4. LP-based Frank-Wolfe for high-precision polish
5. Coordinate descent final refinement

Key innovation over previous: Uses L-BFGS-B (quasi-Newton) instead of first-order

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_objective_and_grad(x, beta, n):
    """
    Compute log-sum-exp objective and gradient w.r.t. x (pre-softmax variables).
    a = softmax(x), c = a*a, f = (1/beta)*log(sum(exp(beta*c)))
    """
    # Softmax
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    a = exp_x / exp_x.sum()
    
    # Convolution
    c = fast_convolve(a)
    
    # Log-sum-exp
    scaled = beta * c
    scaled -= np.max(scaled)
    lse = np.max(scaled) + np.log(np.sum(np.exp(scaled)))
    obj = lse / beta
    
    # Softmax weights
    w = np.exp(scaled) / np.sum(np.exp(scaled))
    
    # Gradient w.r.t. a: g_j = 2 * sum_m a[m] * w[j+m]
    # This is cross-correlation of a and w
    w_rev = w[::-1].copy()
    conv_len = n + len(w_rev) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(A * W, fft_size)[:conv_len]
    indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
    grad_a = 2.0 * conv[indices]
    
    # Chain rule through softmax: grad_x_i = a_i * (grad_a_i - sum_j a_j * grad_a_j)
    mean_grad = np.dot(a, grad_a)
    grad_x = a * (grad_a - mean_grad)
    
    return obj, grad_x


def lse_optimize_lbfgs(f0, time_limit, verbose=False):
    """
    Optimize using L-BFGS-B with progressive log-sum-exp sharpening.
    Parameterizes a = softmax(x) to handle non-negativity and sum=1 constraints.
    """
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 1e-30)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    f = f / s
    
    # Initialize x from f: x_i = log(f_i)
    x0 = np.log(f + 1e-300)
    
    best_x = x0.copy()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    
    # Progressive beta schedule
    beta_schedule = [2, 4, 8, 16, 32, 64, 128, 256, 512, 1024]
    
    for beta in beta_schedule:
        if time.monotonic() - t0 > time_limit * 0.88:
            break
        
        # Time allocation per beta level
        remaining = time_limit - (time.monotonic() - t0)
        stages_left = len(beta_schedule) - beta_schedule.index(beta)
        stage_time = min(remaining / stages_left, 120.0)
        
        if stage_time < 5.0:
            break
        
        stage_deadline = time.monotonic() + stage_time
        
        def func_and_grad(x_eval):
            if time.monotonic() > stage_deadline:
                # Return something to stop
                return best_obj, np.zeros(n)
            obj, grad = lse_objective_and_grad(x_eval, beta, n)
            return obj, grad
        
        # L-BFGS-B optimization
        res = minimize(
            func_and_grad,
            x0,
            jac=True,
            method='L-BFGS-B',
            options={
                'maxiter': 500,
                'maxfun': 2000,
                'ftol': 1e-16,
                'gtol': 1e-12,
                'maxls': 50,
                'maxcor': 20,
            }
        )
        
        x0 = res.x
        # Evaluate actual objective
        a_new = np.exp(x0 - x0.max())
        a_new = a_new / a_new.sum()
        c_new = fast_convolve(a_new)
        obj_new = c_new.max()
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_f = a_new.copy()
            best_x = x0.copy()
        
        if verbose:
            print(f"    L-BFGS beta={beta}: obj={best_obj:.12f}, iters={res.nit}")
    
    return best_f, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c_vec = -np.ones(n)
    # Build constraint matrix using sparse structure
    # (f*g)[k] = sum_j f[j] * g[k-j] <= M
    # For efficiency with large n, use the banded structure
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        cols = np.arange(j, 2 * n - 1, n)  # columns where g[j] contributes
        rows = np.arange(0, n)  # rows k where f[j]*g[k-j] appears
        # Actually: (f*g)[k] = sum_j f[j] * g[k-j]
        # So for fixed j, g[j] appears in rows k = j, j+1, ..., j+n-1
        # A_ub[k, j] = f[k-j] for k in [j, j+n-1]
        A_ub[j:j+n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

            f = f_new
            m_exp = 1.0
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.12f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.04, 4.0)
        
        if no_improve > 250:
            break
    
    return best_f, best_obj


def coordinate_polish(f0, time_limit, verbose=False):
    """Fine coordinate descent: adjust pairs of coordinates to reduce max convolution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f0, fast_convolve(f0).max()
    f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        c = fast_convolve(f)
        
        # Cycle through coordinates
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            # Try spreading mass to neighbors
            for mult in [0.6, 0.7, 0.8, 0.9, 1.1, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                
                f_trial = f.copy()
                f_trial[j] = new_val
                # Redistribute the difference to nearby elements
                diff = old_val - new_val
                if abs(diff) > 1e-15:
                    # Spread to a local window
                    window = 5
                    lo = max(0, j - window)
                    hi = min(n - 1, j + window)
                    neighbors = np.arange(lo, hi + 1)
                    neighbors = neighbors[neighbors != j]
                    if len(neighbors) > 0:
                        f_trial[neighbors] += diff / len(neighbors)
                
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                f[j] = best_j_val
                diff = old_val - best_j_val
                if abs(diff) > 1e-15:
                    window = 5
                    lo = max(0, j - window)
                    hi = min(n - 1, j + window)
                    neighbors = np.arange(lo, hi + 1)
                    neighbors = neighbors[neighbors != j]
                    if len(neighbors) > 0:
                        f[neighbors] += diff / len(neighbors)
                f = np.maximum(f, 0)
                f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            # Try pair exchanges
            rng = np.random.default_rng(int(time.monotonic() * 1000))
            for _ in range(20):
                if time.monotonic() - t0 > time_limit:
                    break
                j1 = rng.integers(0, n)
                j2 = rng.integers(0, n)
                if j1 == j2:
                    continue
                for eps in [0.001, 0.005, 0.01]:
                    f_trial = f.copy()
                    f_trial[j1] = f[j1] + eps
                    f_trial[j2] = f[j2] - eps
                    f_trial = np.maximum(f_trial, 0)
                    f_trial = f_trial / f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-14:
                        f = f_trial
                        best_obj = val
                        best_f = f.copy()
                        improved = True
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    kernel = np.array([0.1, 0.2, 0.4, 0.2, 0.1])
    f_smooth = np.convolve(f_interp, kernel, mode='same')
    f_smooth = np.maximum(f_smooth, 0)
    ss = f_smooth.sum()
    if ss > 1e-15:
        f_smooth /= ss
    return f_smooth


def make_optimal_shape(n):
    """Raised cosine (Hann window) - known to be good for spectral flatness."""
    x = np.linspace(-1, 1, n)
    a = 0.5 * (1 + np.cos(np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_flat_spectrum(n, iterations=50):
    """
    Create a sequence with flat magnitude spectrum using iterative projection.
    Alternates between:
    1. Making spectrum flat (in frequency domain)
    2. Making sequence non-negative and normalized (in time domain)
    """
    # Start with raised cosine
    a = make_optimal_shape(n)
    fft_size = 1 << (2 * n - 2).bit_length()
    
    for _ in range(iterations):
        # Forward FFT
        A = np.fft.rfft(a, fft_size)
        # Make magnitude flat (keep phase, set magnitude to constant)
        mag = np.abs(A)
        mag[mag > 1e-15] = 1.0
        phase = np.angle(A)
        A_new = mag * np.exp(1j * phase)
        # Inverse FFT
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        # Project to non-negative, normalized
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new
    
    return a


def mirror_descent_optimize(f0, time_limit, verbose=False):
    """
    Mirror descent (multiplicative weights) with adaptive step size.
    Uses the subgradient of max(a*a) to update.
    """
    f = np.maximum(f0.astype(np.float64), 1e-10)
    f = f / f.sum()
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    lr = 0.02
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(f)
        M = c.max()
        
        # Subgradient: for the argmax of c, the subgradient is 2*correlation(onehot_argmax, f)
        # More generally, use a mixture of top peaks
        top_k = min(20, len(c))
        peak_idx = np.argpartition(c, -top_k)[-top_k:]
        peak_vals = c[peak_idx]
        # Weight by exponential of value
        w = np.exp(100 * (peak_vals - M))
        w = w / w.sum()
        
        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peak_idx)):
            if w[i] < 1e-12:
                continue
            k = int(peak_idx[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += w[i] * f[k - idx]
        sg *= 2.0
        
        # Multiplicative update with entropy regularization
        f = f * np.exp(-lr * sg)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()
        
        obj_new = fast_convolve(f).max()
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
            lr = min(lr * 1.01, 0.04)
        else:
            no_improve += 1
            lr *= 0.97
        
        if no_improve > 150:
            rng = np.random.default_rng(it * 31 + 17)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.001, 1e-10)
            f = f / f.sum()
            lr = 0.015
            no_improve = 0
        
        if lr < 1e-6:
            break
    
    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):

        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if p_score < best_score:
                best_score = p_score

    except Exception:
        pass

    n_parent = len(parent)

            t_alloc2 = min(rem * 0.5, 200)
