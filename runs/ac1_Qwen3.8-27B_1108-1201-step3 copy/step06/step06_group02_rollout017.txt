"""

- Pre-computed LP constraint structure for speed
- LP dual-based adaptive weighting for direction finding
- Spectral flattening between iterations

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_normalized(a):
    """Compute 2*n*max(conv)/sum^2 for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()


def build_lp_matrix(n):
    """Pre-compute the banded structure for conv(f,g) <= M constraints.
    Returns indices for efficient matrix construction."""
    rows = []
    cols = []
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        for k in range(k_start, k_end):
            i_f = k - j
            if 0 <= i_f < n:
                rows.append(k)
                cols.append(j)
    return np.array(rows, dtype=int), np.array(cols, dtype=int)


def solve_lp_direction(f, M, n, row_idx, col_idx):
    """Solve LP: max sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    vals = np.zeros(len(row_idx))
    for i in range(len(row_idx)):
        k = row_idx[i]
        j = col_idx[i]
        i_f = k - j
        vals[i] = f[i_f]
    from scipy.sparse import csr_matrix
    A_sparse = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, res
    return None, res


def spectral_flatten(a, strength=0.3):
    """Apply spectral flattening to reduce peak autocorrelation.
    Works by compressing the dynamic range of the power spectrum."""
    n = len(a)
    A = np.fft.fft(a)
    P = np.abs(A) ** 2
    P_mean = P.mean()
    # Compress: P_new = P^alpha for alpha < 1 flattens the spectrum
    alpha = 1.0 - strength * 0.3
    P_flat = P ** alpha
    # Preserve phase
    phase = np.angle(A)
    A_new = np.sqrt(P_flat) * np.exp(1j * phase)
    a_new = np.real(np.fft.ifft(A_new))
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def palindromic_project(a):
    """Project onto palindromic subspace: a[i] = a[n-1-i]."""
    n = len(a)
    a_pal = 0.5 * (a + a[::-1])
    a_pal = np.maximum(a_pal, 0.0)
    s = a_pal.sum()
    if s > 1e-15:
        a_pal = a_pal / s
    return a_pal

def multiplicative_descent(f0, time_limit, verbose=False):
    """Mirror descent with multi-peak weighting and adaptive step size."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    lr = 0.003
    no_improve = 0
    temperature = 0.001
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Multi-peak subgradient with softmax weighting
        k_actual = min(50, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-10:
                continue
            k = int(peaks[i])
            for j in range(max(0, k - n + 1), min(n, k + 1)):
                partner = k - j
                if 0 <= partner < n:
                    sg[j] += 2.0 * w[i] * f[partner]

        # Multiplicative (mirror) update
        f = f * np.exp(-lr * sg)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        co = fast_convolve(f).max()
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    MD iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        # Adaptive learning rate
        lr *= 0.9997
        temperature *= 1.005

        if no_improve > 300:
            # Restart with perturbation
            rng = np.random.default_rng(it * 41 + 7)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-15)
            f = f / f.sum()
            lr = 0.003
            no_improve = 0

        if lr < 1e-9:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.25):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine_mod(n, freq, amp):

            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return score

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768, 0.25)
        p_score = try_seq(parent, "default_start")

    # === Phase 1: Diverse initialization scan (60s) ===
    print("\n=== Phase 1: Initialization Scan ===")
    t_phase = time.monotonic()

    n_values = [256, 384, 512, 640, 768, 896, 1024]
    init_candidates = []

    for n_try in n_values:
        init_candidates.append((make_gaussian(n_try, 0.22), n_try, 'gauss'))
        init_candidates.append((make_gaussian(n_try, 0.35), n_try, 'gauss_wide'))
        init_candidates.append((make_bell(n_try, 0.55), n_try, 'bell'))
        init_candidates.append((make_cosine_mod(n_try, 1, 0.15), n_try, 'cos1'))
        init_candidates.append((make_cosine_mod(n_try, 2, 0.12), n_try, 'cos2'))
        init_candidates.append((make_multi_cosine(n_try, 4, 0.2, seed=n_try), n_try, 'multi'))
        init_candidates.append((np.ones(n_try) / n_try, n_try, 'flat'))

    if parent is not None:
        init_candidates.append((parent, len(parent), 'parent'))

    # Quick objective evaluation
    scored_inits = []
    for a_init, n_try, label in init_candidates:
        sc = objective_normalized(a_init)
        scored_inits.append((sc, a_init, n_try, label))

    scored_inits.sort(key=lambda x: x[0])
    print(f"  Top 5 initializations:")
    for sc, a, n_t, label in scored_inits[:5]:
        print(f"    {label} n={n_t}: obj={sc:.6f}")

    # Evaluate top candidates with full evaluate_sequence
    for sc, a_init, n_try, label in scored_inits[:8]:
        try_seq(a_init, f"init_{label}_{n_try}")

    print(f"  Phase 1 time: {time.monotonic() - t_phase:.1f}s")

    # === Phase 2: Main LP-FW on best initialization (1200s) ===
    print("\n=== Phase 2: LP-FW Optimization ===")
    rem = deadline - time.monotonic() - 400
    if rem > 60 and best_seq is not None:
        # Start from the best found
        start_f = best_seq / best_seq.sum()
        t_alloc = min(rem, 1100)
        print(f"  LP-FW [best]: n={len(start_f)}, alloc={t_alloc:.0f}s")
        a_opt, obj = lp_fw_optimize(start_f, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, "lpfw_main")

    # Also try LP-FW from second-best initialization
    rem = deadline - time.monotonic() - 300
    if rem > 60:
        # Find best init different from current best
        for sc, a_init, n_try, label in scored_inits:
            if label == 'parent' or len(a_init) == len(best_seq):
                continue
            t_alloc = min(rem, 300)
            if t_alloc < 30:
                break
            print(f"  LP-FW [{label}]: n={n_try}, alloc={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(a_init, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, f"lpfw_{label}")
            break  # Just try one more

    # === Phase 3: Multiplicative descent refinement (200s) ===
    print("\n=== Phase 3: Multiplicative Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(rem, 180)
            print(f"  MD: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = multiplicative_descent(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "md")

    # === Phase 4: Palindromic projection + re-optimize (100s) ===
    print("\n=== Phase 4: Palindromic Refinement ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            f_pal = palindromic_project(best_seq)
            sc_pal = try_seq(f_pal, "palindromic")
            if sc_pal < best_score + 0.01:
                t_alloc = min(rem, 80)
                print(f"  LP-FW [pal]: n={len(f_pal)}, alloc={t_alloc:.0f}s")
                a_opt, obj = lp_fw_optimize(f_pal, time_limit=t_alloc, verbose=False)
                try_seq(a_opt, "lpfw_pal")

    # === Phase 5: Multi-length exploration (150s) ===
    print("\n=== Phase 5: Multi-Length Exploration ===")
    if best_seq is not None:
        base_n = len(best_seq)
        target_ns = []
        for tn in [256, 384, 512, 640, 768, 896, 1024, 1280]:
            if tn != base_n:
                target_ns.append(tn)

        for target_n in target_ns[:4]:  # Try top 4 different lengths
            rem = deadline - time.monotonic() - 100
            if rem < 20:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")

            if sc < best_score + 0.03:
                t_alloc = min(rem * 0.5, 60)
                if t_alloc > 10:
                    print(f"  LP-FW n={target_n}: alloc={t_alloc:.0f}s")
                    a_opt, obj = lp_fw_optimize(f_interp, time_limit=t_alloc, verbose=False)
                    try_seq(a_opt, f"lpfw_n{target_n}")

    # === Phase 6: Perturbation + re-optimize (100s) ===
    print("\n=== Phase 6: Perturbation Search ===")
    if best_seq is not None:
        n_best = len(best_seq)
        perturbations = []

        # Frequency-domain perturbations
        for amp in [0.005, 0.01, 0.02, 0.03]:
            A = np.fft.fft(best_seq)
            noise = rng.standard_normal(len(A)) * amp
            A_pert = A + noise
            a_pert = np.real(np.fft.ifft(A_pert))
            a_pert = np.maximum(a_pert, 0)
            s = a_pert.sum()
            if s > 1e-10:
                perturbations.append((f'freq_{amp}', a_pert / s))

        # Spatial perturbations
        for shift in [1, 2, 3]:
            perturbations.append((f'shift{shift}', np.roll(best_seq, shift)))

        # Spectral flattening perturbations
        for strength in [0.2, 0.4, 0.6]:
            perturbations.append((f'flat{strength}', spectral_flatten(best_seq, strength)))

        # Blend with different templates
        for mix in [0.1, 0.2, 0.3]:
            perturbations.append((f'blend{mix}',
                                  (1 - mix) * best_seq + mix * make_gaussian(n_best, 0.3)))

        for label, f_pert in perturbations:
            rem = deadline - time.monotonic() - 50
            if rem < 10:
                break
            sc_pert = try_seq(f_pert, f"pert_{label}")
            if sc_pert < best_score + 0.02:
                t_alloc = min(rem * 0.3, 40)
                if t_alloc > 8:
                    a_opt, obj = lp_fw_optimize(f_pert, time_limit=t_alloc, verbose=False)
                    try_seq(a_opt, f"lpfw_pert_{label}")

    # === Phase 7: Coordinate refinement (80s) ===
    print("\n=== Phase 7: Coordinate Refinement ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_alloc = min(rem, 70)
            print(f"  Coord refine: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = coordinate_refine(best_seq, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, "coord_refine")

    # === Phase 8: Final LP-FW squeeze (30s) ===
    print("\n=== Phase 8: Final Squeeze ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            t_alloc = min(rem, 25)
            print(f"  Final LP-FW: alloc={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, "final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Total time: {total_time:.1f}s")
    return [float(x) for x in best_seq]
