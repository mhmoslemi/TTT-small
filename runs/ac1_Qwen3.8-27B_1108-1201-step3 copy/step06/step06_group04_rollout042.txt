    lr = lr_init

        conv = fast_convolve(a)
        # Compute softmax weights
        scaled = conv / T
        scaled -= scaled.max()  # numerical stability
        w = np.exp(scaled)
        p = w / w.sum()

        A = np.zeros(N); A[:n] = a

        # Multiplicative update (mirror descent on simplex)

        lr *= 0.9999

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """FFT-based linear auto-convolution."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed mirror descent with
adaptive temperature, momentum, periodic LP-Frank-Wolfe steps, multi-length
exploration, and coordinate descent polishing. Key innovations: FFT-based

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """FFT-based linear auto-convolution."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_grad(a, conv, T):
    """Compute the gradient of T*log(sum(exp(conv/T))) w.r.t. a using FFT cross-correlation."""
    n = len(a)
    scaled = conv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    p = w / w.sum()
    # grad[j] = 2 * sum_k p[k] * a[k-j] (valid range)
    # This is 2 * cross_correlation(p, a)[j]
    fft_size = 1 << (3 * n - 3).bit_length()
    P = np.zeros(fft_size)
    P[:2 * n - 1] = p
    A = np.zeros(fft_size)
    A[:n] = a
    grad = 2.0 * np.fft.irfft(np.conj(np.fft.rfft(P)) * np.fft.rfft(A), fft_size)[:n]
    return grad, p


def mirror_descent(a0, time_limit, lr_init=0.3, use_momentum=True, verbose=False):
    """
    Mirror descent (multiplicative weights) with smoothed objective.
    Works on the simplex. Uses adaptive temperature and optional momentum.
    """
    a = np.maximum(a0.astype(np.float64), 1e-30)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    momentum = np.zeros(n) if use_momentum else None
    mu = 0.85
    no_improve = 0
    it = 0
    initial_max = conv.max()

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        max_conv = conv.max()

        # Adaptive temperature: start large (smooth), decrease over time
        # T should be a fraction of the spread in conv values
        progress = min(1.0, it / 50000.0)
        T = max_conv * (0.3 * (1 - progress) + 0.01 * progress)
        T = max(T, 1e-10)

        grad, p = compute_smoothed_grad(a, conv, T)

        if use_momentum:
            momentum = mu * momentum + (1.0 - mu) * grad
            step = momentum
        else:
            step = grad

        # Multiplicative update in log-space
        a = a * np.exp(-lr * step)
        a = np.maximum(a, 1e-30)
        s = a.sum()
        a = a / s

        # Evaluate true objective
        true_obj = 2.0 * n * fast_convolve(a).max()
        if true_obj < best_obj - 1e-13:
            best_obj = true_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

            lr = lr_init * 0.01

            # Jitter to escape local minima

            noise = rng.standard_normal(n) * 0.005
            a = np.maximum(best_a + noise, 1e-30)
            a = a / a.sum()
            if use_momentum:
                momentum = np.zeros(n)
            no_improve = 0
            lr = max(lr, lr_init * 0.3)

            break

    return best_a, best_obj


def build_lp_matrix(f, n):
    """Build constraint matrix for LP: rows correspond to conv[k], columns to g[j]."""
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        A_ub[k_start:k_end + 1, j] = f[:k_end - k_start + 1]
    return A_ub


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    A_ub = build_lp_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_step(f, time_limit=60, verbose=False):
    """
    LP-based Frank-Wolfe optimization. Returns improved sequence.
    """
    f = np.maximum(f.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.2
            if m_exp > 20.0:
                break
            continue

        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            m_exp *= 1.1
            continue
        g = g / gs

        diff = np.max(np.abs(g - f))
        if diff < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.05, 10.0)
            continue

        # Line search: try several alphas
        best_alpha = 0.0
        best_val = best_obj
        for alpha_test in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            f_trial = f + alpha_test * (g - f)
            f_trial = np.maximum(f_trial, 0)
            st = f_trial.sum()
            if st < 1e-15:
                continue
            f_trial = f_trial / st
            obj_trial = 2.0 * n * fast_convolve(f_trial).max()
            if obj_trial < best_val:
                best_val = obj_trial
                best_alpha = alpha_test

        if best_alpha < 1e-8:
            no_improve += 1
            if no_improve > 20:
                break
            m_exp = min(m_exp * 1.03, 8.0)
            continue

        f = f + best_alpha * (g - f)
        f = np.maximum(f, 0)
        f = f / f.sum()

        conv = fast_convolve(f)
        M = conv.max()
        m_exp = 1.0

        if 2.0 * n * M < best_obj - 1e-14:
            best_obj = 2.0 * n * M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and it % 5 == 0:
            print(f"    LP-FW iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj

def make_gaussian(n, sigma_frac=0.25):

def make_sidon_like(n, seed=42):
    """
    Sidon-like set: select positions with distinct pairwise sums.
    Place mass at selected positions.
    """
    rng = np.random.default_rng(seed)
    # Greedily select positions with distinct pairwise sums
    used_sums = set()
    selected = []
    for _ in range(n):
        candidates = list(range(n))
        rng.shuffle(candidates)
        placed = False
        for c in candidates:
            new_sums = set()
            conflict = False
            for s in selected:
                pair_sum = c + s
                if pair_sum in used_sums:
                    conflict = True
                    break
                new_sums.add(pair_sum)
            if not conflict:
                selected.append(c)
                used_sums |= new_sums
                placed = True
                break
        if not placed:
            break

    if len(selected) < 2:
        selected = [0, n // 2]

    # Create sequence with mass at selected positions
    # Use varying weights to break symmetry
    weights = 1.0 / (1.0 + 0.1 * np.abs(np.arange(len(selected)) - len(selected) / 2.0))
    a = np.zeros(n)
    for i, pos in enumerate(selected):
        a[pos] = weights[i]
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        score, valid = trial_evaluate(a)
        if valid and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")

    # === Phase 1: Diverse initializations at multiple lengths ===
    print("=== Phase 1: Diverse Initializations ===")
    lengths = [512, 640, 768, 896, 1024]
    best_init = None
    best_init_obj = float('inf')

    for n in lengths:
        inits = [
            make_gaussian(n, 0.2),
            make_gaussian(n, 0.3),
            make_gaussian(n, 0.4),
            make_chirp_abs(n),
            make_two_peak(n, 0.25),
            make_two_peak(n, 0.35),
            np.ones(n) / n,
        ]
        if n == len(parent):
            inits.append(parent.copy())
        else:
            inits.append(interpolate_sequence(parent, n))

        for ic in inits:
            ic = np.maximum(ic, 0)
            s = ic.sum()
            if s < 1e-10:
                continue
            ic = ic / s
            obj = 2.0 * n * fast_convolve(ic).max()
            if obj < best_init_obj:
                best_init_obj = obj
                best_init = ic.copy()
            update_best(ic, f"init_n{n}")

    print(f"  Best init: normalized_obj={best_init_obj:.6f} (n={len(best_init)})")

    # === Phase 2: Mirror descent on best init ===
    print(f"\n=== Phase 2: Mirror Descent (n={len(best_init)}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 30:
        t_alloc = min(rem, 180)
        print(f"  MD: t={t_alloc:.0f}s")
        a_opt, _ = mirror_descent(best_init, time_limit=t_alloc, lr_init=0.3, use_momentum=True)
        update_best(a_opt, "md_main")

    # === Phase 3: LP-FW on best ===
    if best_seq is not None:
        print(f"\n=== Phase 3: LP-FW (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 500
        if rem > 20:
            t_alloc = min(rem, 100)
            print(f"  LP-FW: t={t_alloc:.0f}s")
            a_opt, _ = lp_fw_step(best_seq, time_limit=t_alloc)
            update_best(a_opt, "lpfw")

    # === Phase 4: Mirror descent on current best ===
    if best_seq is not None:
        print(f"\n=== Phase 4: MD on best (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 400
        if rem > 20:
            t_alloc = min(rem, 120)
            print(f"  MD: t={t_alloc:.0f}s")
            a_opt, _ = mirror_descent(best_seq, time_limit=t_alloc, lr_init=0.2, use_momentum=True)
            update_best(a_opt, "md_on_best")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length Exploration ===")
    base = best_seq if best_seq is not None else best_init
    for target_n in [640, 896, 1024, 1280]:
        if len(base) == target_n:
            continue
        rem = deadline - time.monotonic() - 300
        if rem < 40:
            break
        f_interp = interpolate_sequence(base, target_n)
        update_best(f_interp, f"interp_n{target_n}")

        # Short MD + LP-FW at this length
        t_md = min(rem * 0.4, 50)
        t_lp = min(rem * 0.3, 30)
        if t_md > 10:
            a_opt, _ = mirror_descent(f_interp, time_limit=t_md, lr_init=0.25)
            update_best(a_opt, f"md_n{target_n}")
            f_interp = a_opt
        rem = deadline - time.monotonic() - 250
        if rem > 15 and t_lp > 5:
            a_opt, _ = lp_fw_step(f_interp, time_limit=t_lp)
            update_best(a_opt, f"lpfw_n{target_n}")

    # === Phase 6: Alternating MD and LP-FW on best ===
    if best_seq is not None:
        print(f"\n=== Phase 6: Alternating MD/LP-FW (n={len(best_seq)}) ===")
        for round_i in range(4):
            rem = deadline - time.monotonic() - 100
            if rem < 30:
                break
            t_alloc = min(rem / (4 - round_i), 80)

            # MD step
            t_md = t_alloc * 0.6
            if t_md > 10:
                a_opt, _ = mirror_descent(best_seq, time_limit=t_md, lr_init=0.15, use_momentum=True)
                update_best(a_opt, f"alt_md_{round_i}")

            # LP-FW step
            rem = deadline - time.monotonic() - 80
            t_lp = min(rem * 0.4, 25)
            if rem > 15 and t_lp > 5:
                a_opt, _ = lp_fw_step(best_seq, time_limit=t_lp)
                update_best(a_opt, f"alt_lp_{round_i}")

    # === Phase 7: Peak rebalancing ===
    if best_seq is not None:
        print(f"\n=== Phase 7: Peak Rebalancing ===")
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_alloc = min(rem, 40)
            a_opt, _ = sweep_peak_rebalance(best_seq, time_limit=t_alloc)
            update_best(a_opt, "rebalance")

    # === Phase 8: Coordinate descent polish ===
    if best_seq is not None:
        print(f"\n=== Phase 8: CD Polish ===")
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_alloc = min(rem, 30)
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_alloc)
            update_best(a_opt, "cd_polish")

    # === Phase 9: Final short MD with very small lr ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            t_alloc = min(rem, 15)
            a_opt, _ = mirror_descent(best_seq, time_limit=t_alloc, lr_init=0.05, use_momentum=False)
            update_best(a_opt, "final_md")

    # Final verification
    if best_seq is None:
        best_seq = np.ones(768) / 768
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
