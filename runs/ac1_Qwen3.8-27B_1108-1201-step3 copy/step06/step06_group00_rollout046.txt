"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

2. L-BFGS in log-space with adaptive temperature annealing (main optimization)

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_local(a):
    """Compute objective 2n*max(conv)/(sum)^2 locally."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def compute_grad_a_fft(a, w):
    """Compute gradient of weighted sum of convolution values w.r.t. a.
    grad_a[i] = 2 * sum_k w[k] * a[k-i] where 0<=k-i<n
    """
    n = len(a)
    c_flip = w[::-1].copy()
    N = 1 << (3 * n - 2 - 1).bit_length()
    fft_a = rfft(a, N)
    fft_cf = rfft(c_flip, N)
    conv = irfft(fft_a * fft_cf, N)
    return 2.0 * conv[n - 1:2 * n - 1][::-1]


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-30))


def softmax_obj_grad(z, T, n):
    """Soft-max smoothed objective and gradient in log-space."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    grad_a = compute_grad_a_fft(a, w)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z


def lbgfs_single(a0, T, time_limit, maxiter=5000):
    """L-BFGS in log-space with soft-max objective at fixed temperature."""
    t0 = time.monotonic()
    n = len(a0)
    z0 = z_from_a(a0)
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        obj, _ = softmax_obj_grad(z, T, n)
        return obj
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = softmax_obj_grad(z, T, n)
        return g
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        final_obj = func(res.x)
        if np.isfinite(final_obj):
            return softmax_from_z(res.x)
    except Exception:
        pass
    return a0

def subgradient_optimize(a0, time_limit, lr_init=0.005, momentum=0.9,
                         temp_init=0.0005, temp_growth=1.02,
                         kick_interval=2000, kick_size=0.003, seed=42):
    """Projected subgradient with Nesterov momentum and FFT-based gradient."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    a = a / S
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    cur_temp = temp_init
    no_improve = 0
    it_count = 0
    
    while time.monotonic() - t0 < time_limit:
        it_count += 1
        b = fast_convolve(a)
        b_max = b.max()
        
        # Soft-max weights over peaks
        scaled = b / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
        else:
            w /= w_sum
        
        # FFT-based gradient
        g = compute_grad_a_fft(a, w)
        
        # Nesterov step
        v = momentum * v - cur_lr * g
        a_new = project_simplex(a + v)
        
        # Evaluate
        b_new = fast_convolve(a_new)
        obj_new = b_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Periodic adjustments
        if no_improve > kick_interval:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.7, lr_init * 0.1)
            no_improve = 0
        elif no_improve > kick_interval // 2:
            cur_lr *= 0.95
        
        cur_temp *= temp_growth
    
    return best_a, best_obj


def coordinate_polish(a0, time_limit, n_trials_per_elem=6):
    """Coordinate descent: try scaling each element by various multipliers."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    multipliers = np.array([0.4, 0.6, 0.8, 0.9, 1.1, 1.25, 1.5, 2.0])
    
    pass_count = 0
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale <= 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                if scale > 0:
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                    a = np.maximum(a, 0)
                    if best_j_obj < best_obj - 1e-15:
                        best_obj = best_j_obj
                        best_a = a.copy()
                        improved = True
        if not improved:
            break
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_structural_init(n, typ):

        a = np.exp(-0.5 * ((x - 0.5) / 0.5) ** 2)

            a[:taper] = fade
            a[-taper:] = fade[::-1]

        from scipy.special import gammaln
        log_a = 1.5 * np.log(np.maximum(x, 1e-10)) + 1.5 * np.log(np.maximum(1-x, 1e-10))
        a = np.exp(log_a - log_a.max())
    elif typ == 'sine':
        a = np.sin(np.pi * x)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")
    
    return [float(x) for x in best_seq]
