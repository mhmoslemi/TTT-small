import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size-1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A*A, fft_size)[:size]

def compute_score(a):
    s = a.sum()
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    return 2.0 * len(a_norm) * fast_convolve(a_norm).max()

def simplex_project(v):
    """Project v onto the simplex {x: x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1] if cond.any() else n-1
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def multi_peak_pressure(a, n_peaks=20, n_iter=10000, lr=0.01, time_limit=60):
    """
    Local optimizer that uses pressure from multiple peaks simultaneously.
    The idea: compute a "pressure field" from the top peaks of the autocorrelation,
    then move mass from high-pressure regions to low-pressure regions.
    """
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        
        # Get top peaks (excluding endpoints which are trivially small)
        conv_mid = conv[1:-1].copy()
        k_actual = min(n_peaks, len(conv_mid))
        peak_indices = np.argpartition(conv_mid, -k_actual)[-k_actual:] + 1
        peak_values = conv[peak_indices]
        
        # Softmax weights for peaks
        temp = 0.01 * max(peak_values.max(), 1e-10)
        w = np.exp((peak_values - peak_values.max()) / temp)
        w /= w.sum()
        
        # Compute gradient (subgradient of the smooth max)
        grad = np.zeros(n)
        for ii, k in enumerate(peak_indices):
            if w[ii] < 1e-12:
                continue
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)

            # But need to handle the case where j = k-j

                    grad[j] += w[ii] * a[j]
                else:
                    grad[j] += 2 * w[ii] * a[k-j]

        # But we're working with normalized a (sum=1), so we need to account for the simplex constraint
        
        # Project gradient onto tangent space of simplex
        grad_mean = grad.mean()
        grad_tangent = grad - grad_mean
        
        # Gradient step
        a_new = a - lr * grad_tangent
        a_new = simplex_project(a_new)
        
        score = compute_score(a_new)
        if score < best_score:

            best_a = a_new.copy()
            a = a_new

    cssv = np.cumsum(u)

            if 2 * j == k:
                grad[j] += w[ii] * a[j]
            else:
                grad[j] += 2 * w[ii] * a[k - j]

    result = np.maximum(v - theta, 0.0)
    return result

def get_gradient_smoothed(a, top_k=100, temperature=0.005):
    """Compute smoothed gradient of max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    conv_work = conv.copy()
    conv_work[0] = -np.inf
    conv_work[-1] = -np.inf
    
    k_act = min(top_k, n-1)
    peaks = np.argpartition(conv_work, -k_act)[-k_act:]
    pv = conv[peaks]
    
    w = np.exp((pv - pv.max()) / temperature)
    w /= w.sum()
    
    grad = np.zeros(n)
    for ii in range(len(peaks)):
        if w[ii] < 1e-14:
            continue
        k = int(peaks[ii])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je+1)

            grad[diag_j] += w[ii] * a[diag_j]

def lp_best_response(a_fixed, solver=None):
    """LP to find best response."""
    if solver is None:

    Uses backtracking via score comparison.
    """

def get_lp_solver():
    global _lp_solver_name
    if _lp_solver_name is not None:
        return _lp_solver_name
    for solver_name in [cp.ECOS, cp.SCS, cp.GLPK, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_name = solver_name
                return _lp_solver_name
        except Exception:
            continue
    _lp_solver_name = cp.ECOS
    return cp.ECOS

def lp_best_response(a_fixed, solver=None):
    """LP: given fixed a, find b >= 0, sum(b)=1 minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    constraints = [cp.sum(b_var) == 1]
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        # conv[k] = sum_{j=js}^{je} a[k-j] * b[j]
        coeff = a_fixed[k - je : k - js + 1][::-1]
        if coeff.max() > 0:
            constraints.append(coeff @ b_var <= M_var)

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return None, None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    symmetry exploitation, multi-start, and multi-length search.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = compute_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Using default parent: n=512, score={best_score:.7f}")

    # === Phase 1: Quick FP to improve initialization ===
    print("\n=== Phase 1: FP Initialization ===")
    n_parent = len(parent)
    rem = deadline - time.monotonic() - 1200
    if rem > 60:
        t_fp = min(rem * 0.7, 300.0)
        print(f"  FP n={n_parent}, budget={t_fp:.0f}s")
        a_fp, score_fp = fp_optimize(parent, t_fp, solver=solver, verbose=True)
        if score_fp < best_score:
            best_score = score_fp
            best_seq = a_fp.copy()
            print(f"  FP improved: {best_score:.7f}")
    else:
        print("  Skipping FP (not enough time)")

    # === Phase 2: High-precision TVD on parent length ===
    print(f"\n=== Phase 2: TVD Optimization (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 800
    if rem > 30:
        t_tvd = min(rem * 0.6, 400.0)
        print(f"  TVD n={len(best_seq)}, budget={t_tvd:.0f}s")
        a_tvd, score_tvd = tvd_optimize(best_seq, t_tvd, top_k=100, temp_init=0.002,
                                         lr_init=0.005, verbose=True, seed=seed)
        if score_tvd < best_score:
            best_score = score_tvd
            best_seq = a_tvd.copy()
            print(f"  TVD improved: {best_score:.7f}")

    # === Phase 3: Symmetry-aware refinement ===
    print(f"\n=== Phase 3: Symmetric TVD (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 30:
        t_sym = min(rem * 0.5, 200.0)
        print(f"  TVD-Sym n={len(best_seq)}, budget={t_sym:.0f}s")
        a_sym, score_sym = tvd_symmetric(best_seq, t_sym, verbose=True, seed=seed + 1)
        if score_sym < best_score:
            best_score = score_sym
            best_seq = a_sym.copy()
            print(f"  Sym improved: {best_score:.7f}")

    # === Phase 4: Multi-start perturbation ===
    print(f"\n=== Phase 4: Multi-Start (best={best_score:.7f}) ===")
    n_perturb = 4
    pert_scales = [0.003, 0.008, 0.015, 0.03]
    for i in range(n_perturb):
        rem = deadline - time.monotonic() - 200
        if rem < 30:
            break
        t_pert = min(rem * 0.35, 100.0)
        scale = pert_scales[min(i, len(pert_scales) - 1)]
        pert = rng.standard_normal(len(best_seq)) * scale
        a_pert = simplex_project(best_seq + pert)
        print(f"  Perturb {i+1}/{n_perturb} scale={scale}, budget={t_pert:.0f}s")
        a_r, s_r = tvd_optimize(a_pert, t_pert, top_k=80, temp_init=0.004,
                                  lr_init=0.004, verbose=False, seed=seed + 100 + i * 7)
        if s_r < best_score:
            best_score = s_r
            best_seq = a_r.copy()
            print(f"  Perturb {i+1} improved: {best_score:.7f}")

    # === Phase 5: Peak shaving ===
    print(f"\n=== Phase 5: Peak Shaving (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 150
    if rem > 20:
        t_ps = min(rem * 0.6, 120.0)
        print(f"  PeakShave budget={t_ps:.0f}s")
        a_ps, s_ps = peak_shave_refine(best_seq, t_ps, verbose=True)
        if s_ps < best_score:
            best_score = s_ps
            best_seq = a_ps.copy()
            print(f"  PeakShave improved: {best_score:.7f}")

    # === Phase 6: Try different lengths ===
    print(f"\n=== Phase 6: Multi-Length (best={best_score:.7f}) ===")
    n_best = len(best_seq)
    alt_lengths = []
    for factor in [0.75, 1.25, 1.5]:
        alt_n = int(n_best * factor)
        if 128 <= alt_n <= 4096 and alt_n != n_best:
            alt_lengths.append(alt_n)

    for alt_n in alt_lengths[:2]:
        rem = deadline - time.monotonic() - 100
        if rem < 30:
            break
        t_alt = min(rem * 0.5, 120.0)
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  Alt length n={alt_n}, budget={t_alt:.0f}s")
        a_alt, s_alt = tvd_optimize(init_alt, t_alt, top_k=80, temp_init=0.003,
                                      lr_init=0.004, verbose=False, seed=seed + 500)
        if s_alt < best_score:
            best_score = s_alt
            best_seq = a_alt.copy()
            print(f"  Alt n={alt_n} improved: {best_score:.7f}")

    # === Phase 7: Final micro-polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 15
    if rem > 15:
        t_final = min(rem * 0.7, 60.0)
        print(f"  Final TVD budget={t_final:.0f}s")
        a_fin, s_fin = tvd_optimize(best_seq, t_final, top_k=100, temp_init=0.001,
                                      lr_init=0.003, verbose=True, seed=seed + 999,
                                      restart_interval=600)
        if s_fin < best_score:
            best_score = s_fin
            best_seq = a_fin.copy()
            print(f"  Final improved: {best_score:.7f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        if fs < best_score:
            best_score = fs
        else:
            print(f"  WARNING: evaluate_sequence gives {fs:.7f} vs computed {best_score:.7f}")
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
