import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def obj_normed(f):
    """Objective for sum-normalized f: 2*n*max(conv(f,f))"""
    n = len(f)
    return 2.0 * n * fast_convolve(f).max()

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

def build_A_ub(f, n):
    """Build constraint matrix: A_ub[k,j] = f[k-j] if 0<=k-j<n else 0"""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # (f*g)[k] = sum_j f[j]*g[k-j], so for fixed j, g[k-j] appears at position k
        # A[k,j] = f[j] if k-j is in [0,n-1], i.e., j <= k <= j+n-1
        k_lo = j
        k_hi = min(j + n - 1, 2 * n - 2)
        A[k_lo:k_hi+1, j] = f[:k_hi - k_lo + 1]
    return A

def solve_lp(f, M, n):
    """Solve: max sum(g) s.t. A_ub @ g <= M, g >= 0"""
    A_ub = build_A_ub(f, n)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success:
        dual = res.ineqlin.marginals  # dual variables for A_ub constraints
        return res.x, dual
    return None, None

def bisect_M(f, M_lo, M_hi, n, max_iter=10):
    """Find the smallest M such that LP gives a non-trivial direction different from f."""
    for _ in range(max_iter):
        M_mid = 0.5 * (M_lo + M_hi)
        g, dual = solve_lp(f, M_mid, n)
        if g is None:
            M_lo = M_mid
            continue
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            M_lo = M_mid
            continue
        g = g / gs
        if np.max(np.abs(g - f)) > 1e-8:
            M_hi = M_mid
        else:
            M_lo = M_mid
    return M_hi, g, dual

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def obj_val(f):
    n = len(f)
    return 2.0 * n * fast_convolve(f).max()

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_grad(f, w):
    """grad[i] = 2 * sum_j w[i+j] * f[j] using FFT."""
    n = len(f)
    fr = f[::-1]
    full = np.fft.irfft(np.fft.rfft(fr, 3*n) * np.fft.rfft(w, 3*n), 3*n)
    # (fr * w)[n-1 + i] for i = 0..n-1
    return 2.0 * full[n-1 : 2*n-1]

def smoothed_gradient_descent(f0, time_limit, T_init=50.0, lr_init=0.05, 
                                beta=0.9, T_growth=1.002, max_no_improve=200):
    """Smoothed gradient descent with Nesterov momentum on the simplex."""
    f = np.maximum(f0, 0.0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = obj_val(f)
    v = np.zeros(n)
    T = T_init
    lr = lr_init
    no_improve = 0
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        
        # Compute softmax weights
        scaled = T * conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        
        # Compute gradient
        g = compute_grad(f, w)
        
        # Project gradient onto tangent space (sum = 0)
        g_proj = g - g.mean()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach combining:
1. LP-Frank-Wolfe with reduced constraints (top-K active lags) for fast direction finding
2. Smoothed gradient descent with multiplicative weights (mirror descent) for refinement
3. Dual-guided perturbation using LP dual variables for escaping local minima
4. Multi-length exploration with evolutionary local search
5. Adaptive temperature annealing for the smoothed max

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_val(f):
    n = len(f)
    return 2.0 * n * fast_convolve(f).max()


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad(f, w):
    """Compute grad[i] = 2 * sum_j w[i+j] * f[j] using FFT."""
    n = len(f)
    fr = f[::-1]
    fft_size = 3 * n
    F_fr = np.fft.rfft(fr, fft_size)
    F_w = np.fft.rfft(np.concatenate([w, np.zeros(fft_size - len(w))]), fft_size)
    full = np.fft.irfft(F_fr * F_w, fft_size)
    return 2.0 * full[n - 1:2 * n - 1]


def solve_lp_reduced(f, M, n, K=30):
    """LP with only top-K active constraints for speed."""
    conv = fast_convolve(f)
    if K >= len(conv):
        top_k = np.arange(len(conv))
    else:
        top_k = np.argpartition(conv, -K)[-K:]
    top_k = np.sort(top_k)

    A_ub = np.zeros((len(top_k), n))
    for idx, k in enumerate(top_k):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        length = je - js + 1
        A_ub[idx, js:je + 1] = f[k - js:k - js + length]

    c = -np.ones(n)
    b_ub = np.full(len(top_k), M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 5.0})
    if res.success:
        dual = res.ineqlin.marginals if hasattr(res.ineqlin, 'marginals') else None
        return res.x, dual
    return None, None


def solve_lp_full(f, M, n):
    """LP with all constraints."""
    conv = fast_convolve(f)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_lo = j
        k_hi = min(j + n - 1, 2 * n - 2)
        length = k_hi - k_lo + 1
        A_ub[k_lo:k_hi + 1, j] = f[:length]

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 10.0})
    if res.success:
        dual = res.ineqlin.marginals if hasattr(res.ineqlin, 'marginals') else None
        return res.x, dual
    return None, None

def smoothed_grad_optimize(f0, time_limit, T_init=30.0, lr_init=0.08,
                           beta=0.85, verbose=False):
    """Smoothed gradient descent with multiplicative weights and momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = obj_val(f)
    v = np.zeros(n)
    T = T_init
    lr = lr_init
    no_improve = 0
    it = 0

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        conv_max = conv.max()

        # Compute softmax weights with adaptive temperature
        scaled = T * conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= w_sum

        # Compute gradient
        g = compute_grad(f, w)

        # Project onto tangent space
        g_proj = g - g.mean()

        # Nesterov momentum
        v = beta * v - lr * g_proj

        # Try multiplicative weights update
        f_new = f * np.exp(v * 0.1)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        obj_new = obj_val(f_new)
        if obj_new < best_obj - 1e-13:
            f = f_new
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            lr = min(lr * 1.02, 0.4)
        else:
            # Try additive step
            f_add = f - 0.3 * lr * g_proj
            f_add = project_simplex(f_add)
            obj_add = obj_val(f_add)
            if obj_add < best_obj - 1e-13:
                f = f_add
                best_obj = obj_add
                best_f = f.copy()
                no_improve = 0
                v = np.zeros(n)
                lr = min(lr * 1.01, 0.4)
            else:
                no_improve += 1
                lr *= 0.97
                T *= 1.005

                if no_improve > 100:
                    v = np.zeros(n)
                    T *= 3.0
                    lr = lr_init
                    no_improve = 0
                    rng = np.random.default_rng(it * 31 + 17)
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.002)

        if verbose and it % 20 == 0:
            print(f"    Smoothed GD iter {it}: obj={best_obj:.10f}, T={T:.1f}, lr={lr:.4f}")

        if lr < 1e-9:
            break

    return best_f, best_obj


def multiplicative_refine(f0, time_limit, verbose=False):
    """Pure multiplicative weights mirror descent targeting top peaks."""
    f = np.maximum(f0.astype(np.float64), 1e-10)
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = obj_val(f)
    lr = 0.003
    temp = 0.001
    no_improve = 0
    it = 0

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)

        # Weight peaks by softmax
        k_top = min(20, len(conv))
        top_idx = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[top_idx]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(k_top)
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Subgradient
        sg = np.zeros(n)
        for i in range(k_top):
            if w[i] < 1e-12:
                continue
            k = int(top_idx[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        # Multiplicative update
        f = f * np.exp(-lr * sg)
        f = np.maximum(f, 1e-10)
        f = f / f.sum()

        co = obj_val(f)
        if co < best_obj - 1e-13:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        temp *= 1.003

        if no_improve > 120:
            rng = np.random.default_rng(it * 47 + 23)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.002, 1e-10)
            f = f / f.sum()
            lr = 0.003
            temp = 0.001
            no_improve = 0

        if verbose and it % 50 == 0:
            print(f"    MultRefine iter {it}: obj={best_obj:.10f}")

        if lr < 1e-10:
            break

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Aggressive coordinate descent targeting peak contributors."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = obj_val(f)
    no_improve = 0

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit and no_improve < 30:
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        top_k = min(8, len(conv))
        peak_indices = np.argsort(conv)[-top_k:]

        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[k - idx]
            total = products.sum()
            if total < 1e-15:
                continue

            order = np.argsort(products)[::-1]
            for oi in range(min(3, len(order))):
                j_m = int(idx[order[oi]])
                old_val = f[j_m]
                if old_val < 1e-10:
                    continue

                for mult in [0.4, 0.6, 0.8, 1.3, 1.6]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.8:
                        continue
                    trial = f.copy()
                    trial[j_m] = new_val
                    d = old_val - new_val
                    if abs(d) > 1e-14:
                        mask = np.ones(n, dtype=bool)
                        mask[js:je + 1] = False
                        if mask.sum() > 0:
                            w_mask = f[mask]
                            wsum = w_mask.sum()
                            if wsum > 1e-14:
                                trial[mask] += d * w_mask / wsum
                    trial = np.maximum(trial, 0)
                    ts = trial.sum()
                    if ts < 1e-14:
                        continue
                    trial = trial / ts
                    obj_t = obj_val(trial)
                    if obj_t < best_obj - 1e-14:
                        best_obj = obj_t
                        best_f = trial.copy()
                        f = trial
                        improved = True

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

        if verbose and no_improve == 0 and it % 5 == 0:
            print(f"    CoordDescent iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.25):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_decaying(n, rate=3.0):
    x = np.arange(n)
    a = np.exp(-rate * x / n)
    a = a / a.sum()
    return a


def make_bell(n, width=0.7):
    x = np.arange(n) - n / 2.0
    half_w = n * width / 2.0
    t = np.clip(np.abs(x) / half_w, 0, 1.0)
    a = (1.0 - t) ** 3
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_smooth_asymmetric(n, skew=0.3):
    """Smooth asymmetric sequence: Gaussian with skewed center."""
    center = n / 2.0 * (1.0 - skew)
    x = np.arange(n) - center
    sigma = n * 0.2
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def try_sequence(a, best_seq, best_score, label=""):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score
