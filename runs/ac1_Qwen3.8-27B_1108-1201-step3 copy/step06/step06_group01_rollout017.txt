import numpy as np

from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_correlate(w, a):
    """Cross-correlation: (cross_corr)[j] = sum_k w[k] * a[k-j] (valid range)."""
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(W * A, fft_size)[:n]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():

        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:

    _lp_solver = cp.ECOS

    if solver is None:

    M_var = cp.Variable()
    T = build_conv_matrix(a)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s

def subgradient_best_response(a, n_iters=200, lr_init=0.5):
    """
    Approximate best response using projected subgradient on the simplex.
    Solves: min_b max_k conv(a,b)[k] s.t. b>=0, sum(b)=1.

    b = np.ones(n) / n
    lr = lr_init
    t0 = time.monotonic()

        # Compute conv(a, b)
        T_b = fast_convolve(a) * 0 + np.convolve(a, b)  # conv(a,b)
        # Actually, let's compute cross-correlation properly
        # conv(a, b)[k] = sum_j a[j] * b[k-j]
        # We can compute this as: for each k, sum_j a[j]*b[k-j]
        # This is cross-correlation of a and b
        n_ab = len(a) + len(b) - 1
        fft_size = 1 << (n_ab - 1).bit_length()
        A = rfft(a, fft_size)
        B = rfft(b, fft_size)
        conv_ab = irfft(A * B, fft_size)[:n_ab]
        
        max_val = conv_ab.max()
        # Find active constraints
        threshold = max_val - 1e-10 * max_val
        active = np.where(conv_ab >= threshold)[0]
        
        # Subgradient w.r.t. b: for active k, the gradient is a[k-j] (shifted)
        # g[j] = sum_{k active} w_k * a[k-j]
        # Use uniform weights over active set
        w = np.zeros(n_ab)
        w[active] = 1.0 / len(active)
        
        # g = cross_correlate(w, a) but we need: g[j] = sum_k w[k] * a[k-j]
        # This is the cross-correlation of w and a
        g = np.zeros(n)
        for j in range(n):
            # g[j] = sum_k w[k] * a[k-j] where a is defined for 0..n-1
            k_min = j
            k_max = j + n - 1
            if k_max < n_ab:
                g[j] = np.dot(w[k_min:k_max+1], a[:min(n, k_max - k_min + 1)])
        
        # Actually let's do it via FFT for speed
        size = n + n_ab - 1
        fft_s = 1 << (size - 1).bit_length()
        W = rfft(w, fft_s)
        A2 = rfft(a, fft_s)
        g = irfft(W * A2, fft_s)[:n]
        g = np.maximum(g, 0)  # subgradient should be non-negative for our case
        
        # Projected subgradient step on simplex
        b_candidate = b - lr * g
        # Project onto simplex {x >= 0, sum(x) = 1}
        u = np.sort(b_candidate)[::-1]
        cssv = np.cumsum(u)
        rho = n - 1
        for i in range(n):
            if u[i] * (i + 1) > cssv[i] - 1:
                rho = i
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        b_new = np.maximum(b_candidate - theta, 0.0)
        
        b = b_new
        lr *= 0.995
    
    return b / b.sum()

def active_set_polish(a, time_limit, tol=1e-8):

    """
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Identify active set
    conv = fast_convolve(a)
    max_val = conv.max()
    # Active set: indices where conv is within tol of max
    active_mask = conv >= max_val - tol * max_val
    active_indices = np.where(active_mask)[0]
    
    n_active = len(active_indices)
    print(f"    Active set: {n_active} constraints (of {2*n-1})")
    
    # Do focused coordinate descent considering only active constraints
    for iteration in range(100):
        if time.monotonic() - t0 > time_limit:
            break
        
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_x = a[i]
            if old_x < 1e-15:
                continue
            
            conv_c = fast_convolve(a)
            
            # Only consider active indices for the objective
            # But we need to make sure we don't violate others significantly
            B_act = conv_c[active_indices].copy()
            
            c_act = np.zeros(n_active)
            B_full = conv_c.copy()
            c_full = np.zeros(2 * n - 1)
            
            for idx, k in enumerate(active_indices):
                ki = k - i
                if 0 <= ki < n:
                    if ki == i:
                        B_act[idx] = conv_c[k] - old_x * old_x
                    else:
                        B_act[idx] = conv_c[k] - 2.0 * old_x * a[ki]
                        c_act[idx] = 2.0 * a[ki]
                        c_full[k] = c_act[idx]
                        B_full[k] = B_act[idx]
            
            # Simplified: just use ternary search on the active max
            x_lo = 0.0
            x_hi = max(2.0 * old_x + 0.1, 1.0)
            
            def g_act(x):
                vals = B_act + c_act * x
                # Add x^2 for k=2i if it's in active set
                if 2 * i in active_indices:
                    idx2i = np.where(active_indices == 2 * i)[0]
                    if len(idx2i) > 0:
                        vals[idx2i[0]] += x * x
                return vals.max()
            
            for _ in range(40):
                m1 = x_lo + (x_hi - x_lo) / 3.0
                m2 = x_hi - (x_hi - x_lo) / 3.0
                if g_act(m1) < g_act(m2):
                    x_hi = m2
                else:
                    x_lo = m1
            
            x_opt = (x_lo + x_hi) / 2.0
            a[i] = x_opt
            
            # Full evaluation
            ss = a.sum()
            if ss < 0.01:
                a[i] = old_x
                continue
            mc = fast_convolve(a).max()
            score = 2.0 * n * mc / (ss * ss)
            
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                improved = True
            else:
                a[i] = old_x
        
        if not improved:
            # Update active set
            conv = fast_convolve(a)
            max_val = conv.max()
            active_mask = conv >= max_val - tol * max_val
            active_indices = np.where(active_mask)[0]
            
            # If no improvement for a while, break
            break
    
    return best_a, best_score


def pairwise_swap(a, time_limit, n_trials=10000):
    """Try pairwise swaps to reduce max correlation."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    trials = 0
    while time.monotonic() - t0 < time_limit and trials < n_trials:
        trials += 1
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        # Try moving mass from i to j or vice versa
        for direction in [1, -1]:
            for frac in [0.1, 0.25, 0.5]:
                amt = direction * frac * a[min(i,j)]
                if amt + a[max(i,j)] < -1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15 or a_trial[j] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 0.01:
                    continue
                mc = fast_convolve(a_trial).max()
                score = 2.0 * n * mc / (ss * ss)
                if score < best_score - 1e-12:
                    best_score = score
                    best_a = a_trial.copy()
                    a = a_trial
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining extrapolated FP, coordinate descent,
    active-set polishing, and pairwise swaps.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    t_start = time.monotonic()
    
    best_seq = None
    best_score = float('inf')
    
    # Find LP solver
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(768) / 768
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.6f}")
    
    # === Phase 1: Extrapolated FP on parent length ===
    n_parent = len(parent)
    print(f"\n=== Phase 1: Extrapolated FP (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t_fp = min(rem * 0.35, 500.0)
    if t_fp > 30:
        a_fp, score_fp, iters = extrapolated_fp(
            n_parent, t_fp, initial=parent, solver=solver, verbose=True
        )
        best_seq, best_score = check_and_update(a_fp, f"xFP_n{n_parent}", best_seq, best_score)
    
    # === Phase 2: Coordinate Descent ===
    print(f"\n=== Phase 2: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_cd = min(rem * 0.25, 300.0)
        if t_cd > 20:
            print(f"  CD budget: {t_cd:.0f}s")
            a_cd, score_cd = coordinate_descent(best_seq, t_cd, max_passes=300, verbose=True)
            best_seq, best_score = check_and_update(a_cd, "coord_desc", best_seq, best_score)
    
    # === Phase 3: Active set polish ===
    print(f"\n=== Phase 3: Active Set Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_as = min(rem * 0.15, 150.0)
        if t_as > 15:
            print(f"  AS budget: {t_as:.0f}s")
            a_as, score_as = active_set_polish(best_seq, t_as)
            best_seq, best_score = check_and_update(a_as, "active_set", best_seq, best_score)
    
    # === Phase 4: Pairwise swaps ===
    print(f"\n=== Phase 4: Pairwise Swaps (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_pw = min(rem * 0.1, 100.0)
        if t_pw > 10:
            print(f"  PW budget: {t_pw:.0f}s")
            a_pw, score_pw = pairwise_swap(best_seq, t_pw)
            best_seq, best_score = check_and_update(a_pw, "pairwise", best_seq, best_score)
    
    # === Phase 5: Second round of FP with refined start ===
    print(f"\n=== Phase 5: FP Round 2 (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_fp2 = min(rem * 0.15, 200.0)
        if t_fp2 > 30:
            a_fp2, score_fp2, iters2 = extrapolated_fp(
                len(best_seq), t_fp2, initial=best_seq, solver=solver, verbose=False
            )
            best_seq, best_score = check_and_update(a_fp2, f"xFP2_n{len(best_seq)}", best_seq, best_score)
            
            # Quick CD on new best
            rem = deadline - time.monotonic()
            if rem > 30:
                a_cd2, _ = coordinate_descent(best_seq, min(rem * 0.4, 80.0), max_passes=100, verbose=False)
                best_seq, best_score = check_and_update(a_cd2, "cd2", best_seq, best_score)
    
    # === Phase 6: Try alternative lengths ===
    print(f"\n=== Phase 6: Alternative Lengths (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = []
        for factor in [0.8, 1.2, 1.5, 0.6, 1.33]:
            alt_n = int(n_best * factor)
            if 256 <= alt_n <= 4096 and alt_n != n_best and alt_n not in alt_lengths:
                alt_lengths.append(alt_n)
        
        for alt_n in alt_lengths[:4]:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t_frac = min(rem * 0.3, 120.0)
            if t_frac < 20:
                break
            init_alt = interpolate_sequence(best_seq, alt_n)
            
            # Quick FP
            a_alt, score_alt, _ = extrapolated_fp(alt_n, t_frac * 0.6, initial=init_alt, solver=solver, verbose=False)
            best_seq_tmp, best_score_tmp = check_and_update(a_alt, f"xFP_alt_n{alt_n}", best_seq, best_score)
            
            # Quick CD
            rem = deadline - time.monotonic()
            if rem > 20:
                a_cd_alt, _ = coordinate_descent(a_alt, min(rem * 0.3, 40.0), max_passes=50, verbose=False)
                best_seq, best_score = check_and_update(a_cd_alt, f"cd_alt_n{alt_n}", best_seq, best_score)
    
    # === Phase 7: Final polish cycle ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        # CD
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            a_f1, _ = coordinate_descent(best_seq, min(rem * 0.4, 60.0), max_passes=100, verbose=False)
            best_seq, best_score = check_and_update(a_f1, "final_cd", best_seq, best_score)
        
        # Pairwise
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_f2, _ = pairwise_swap(best_seq, min(rem * 0.5, 30.0), n_trials=5000)
            best_seq, best_score = check_and_update(a_f2, "final_pw", best_seq, best_score)
        
        # One more mini-CD
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_f3, _ = coordinate_descent(best_seq, min(rem * 0.6, 20.0), max_passes=50, verbose=False)
            best_seq, best_score = check_and_update(a_f3, "final_cd2", best_seq, best_score)
    
    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    elapsed = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)}) [{elapsed:.0f}s elapsed]")
    return [float(x) for x in best_seq]
