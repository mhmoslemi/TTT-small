"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-n exploration with diverse initializations
2. Log-sum-exp smooth optimization with adaptive temperature
3. Softmax parameterization variant
4. Fine-grained coordinate descent polish
5. LP-based refinement

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_gradient(f, tau):
    """Compute gradient of log-sum-exp smooth approximation."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # Gradient: (2/tau) * sum_k w[k] * a[k-j] for each j
    # = (2/tau) * (a * w_rev)[2n-2-j] for j in [0,n-1]
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    h = conv_result[n - 1:2 * n - 1][::-1]
    
    grad = (2.0 / tau) * h
    return grad, conv, w

def adam_optimize(f0, time_limit, tau_init=None, lr_init=0.003, 
                  tau_decay=0.9998, restart_threshold=200, verbose=False):
    """Adam optimization with adaptive temperature and restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    conv = fast_convolve(f)
    if tau_init is None:
        tau = 0.1 * max(conv.max(), 1e-10)
    else:
        tau = tau_init
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    t = 0
    t0 = time.monotonic()
    
    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        grad, conv, w = smooth_gradient(f, tau)
        M = conv.max()
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)
        f = f_new
        
        conv_new = fast_convolve(f)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive temperature
        tau *= tau_decay
        tau = max(tau, 1e-8)
        
        # Adaptive learning rate
        if no_improve > 50:
            lr *= 0.9
        if no_improve > restart_threshold:
            # Restart from best with perturbation
            rng = np.random.default_rng(t * 7919 + 13)
            noise = 0.001 + 0.002 * min(no_improve / restart_threshold, 2.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = lr_init * 0.8
            tau = 0.05 * max(fast_convolve(f).max(), 1e-10)
            if verbose:
                print(f"      Restart at t={t}, n={n}, score={2*n*best_obj:.6f}")
        
        if verbose and t % 1000 == 0:
            print(f"      t={t}, best={2*n*best_obj:.6f}, tau={tau:.2e}, lr={lr:.5f}")
    
    return best_f, 2 * n * best_obj

def softmax_adam_optimize(c0, time_limit, tau_init=None, lr_init=0.01,

    """Adam in softmax parameterization space."""
    n = len(c0)

        e = np.exp(c_shifted)

    f = get_a(c)
    conv = fast_convolve(f)
    if tau_init is None:
        tau = 0.1 * max(conv.max(), 1e-10)
    else:
        tau = tau_init
    
    m = np.zeros(n)

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def lpfw_step(f, M, n, relaxation=1.0):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M*relaxation, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M * relaxation)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

        g_sum = g.sum()

def coordinate_descent(f0, time_limit, verbose=False):
    """Targeted coordinate descent on peak-contributing elements."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        improved = False
        
        # Find top-3 peaks
        top_k = min(3, 2 * n - 1)
        peak_indices = np.argsort(conv)[-top_k:]
        
        for pk in peak_indices:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if je < js:
                continue
            
            idx = np.arange(js, je + 1)
            products = f[idx] * f[pk - idx]
            j_max = idx[np.argmax(products)]
            
            # Try to reduce f[j_max] and redistribute
            old_val = f[j_max]
            if old_val < 1e-10:
                continue
            
            for factor in [0.5, 0.7, 0.85]:
                trial = f.copy()
                trial[j_max] *= factor
                diff = old_val - trial[j_max]
                
                # Redistribute to elements that don't contribute to top peaks
                contrib = np.zeros(n, dtype=bool)
                for pk2 in peak_indices:
                    js2 = max(0, int(pk2) - n + 1)
                    je2 = min(n - 1, int(pk2))
                    contrib[js2:je2+1] = True
                contrib[j_max] = True
                
                free = ~contrib
                if free.sum() > 0:
                    trial[free] += diff * f[free] / max(f[free].sum(), 1e-15)
                
                trial = np.maximum(trial, 0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial /= ss
                obj = fast_convolve(trial).max()
                if obj < best_obj - 1e-14:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    improved = True
                    break
        
        if not improved:
            no_improve += 1
            if no_improve > 10:
                rng = np.random.default_rng(int(time.monotonic() * 10000))
                f = project_simplex(best_f + rng.standard_normal(n) * 0.0005)
                no_improve = 0
        else:
            no_improve = 0
    
    return best_f, 2 * n * best_obj

def make_init_n(n, style, rng):
    """Create diverse initializations."""
    if style == 0:  # Asymmetric Gaussian
        center = 0.2 + 0.4 * rng.random()
        sigma = 0.15 + 0.2 * rng.random()
        x = np.arange(n) - n * center
        a = np.exp(-0.5 * (x / (n * sigma)) ** 2)
    elif style == 1:  # Exponential decay
        rate = 1.0 + 3.0 * rng.random()
        a = np.exp(-np.arange(n) / (n / rate))
    elif style == 2:  # Truncated Gaussian (symmetric)
        sigma = 0.2 + 0.3 * rng.random()
        x = np.arange(n) - n / 2.0
        a = np.exp(-0.5 * (x / (n * sigma)) ** 2)
    elif style == 3:  # Two bumps
        c1 = 0.1 + 0.2 * rng.random()
        c2 = 0.6 + 0.3 * rng.random()
        s1 = 0.05 + 0.1 * rng.random()
        s2 = 0.05 + 0.1 * rng.random()
        x = np.arange(n) / n
        a = np.exp(-((x - c1) / s1) ** 2) + np.exp(-((x - c2) / s2) ** 2)
    elif style == 4:  # Linear gradient
        a = np.linspace(0.1, 1.0, n)
    elif style == 5:  # Random smooth
        k = n // 4 + 1
        c = rng.standard_normal(k)
        a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, k), c - c.min() + 0.1)
    else:
        a = np.ones(n) / n
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_obj_grad(f, tau):
    """
    Compute smooth max-convolution objective and gradient.
    Returns (gradient, convolution, softmax_weights).
    """
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # grad[j] = (2/tau) * sum_k w[k] * f[k-j]
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    h = conv_result[n - 1:2 * n - 1][::-1]
    
    grad = (2.0 / tau) * h
    return grad, conv, w

def _solve_lp(f, M, n, relaxation=1.0):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M*relaxation, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M * relaxation)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def peak_mutation_search(f0, time_limit, verbose=False):

    Local search: identify peak-contributing pairs and try mutation moves
    that reduce the peak by redistributing mass.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)

        f = f / s

    best_f = f.copy()

    no_improve = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        # Find top peaks
        num_peaks = min(5, 2 * n - 1)
        peak_indices = np.argsort(conv)[-num_peaks:]

            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[pk - idx]
            
            # Try top-3 contributing pairs
            top_pairs = idx[np.argsort(products)[-3:]]
            
            for j_max in top_pairs:
                j_max = int(j_max)
                j_other = pk - j_max
                old_val = f[j_max]
                if old_val < 1e-12:
                    continue

                for frac in [0.3, 0.5, 0.7]:

                    contrib = np.zeros(n, dtype=bool)
                    for pk2 in peak_indices:
                        js2 = max(0, int(pk2) - n + 1)
                        je2 = min(n - 1, int(pk2))
                        contrib[js2:je2 + 1] = True
                    contrib[j_max] = True
                    contrib[j_other] = True

                    trial = f.copy()
                    move_amount = old_val * frac
                    trial[j_max] -= move_amount
                    trial[j_target] += move_amount
                    
                    # Also try symmetric move on j_other
                    if j_other != j_max and f[j_other] > 1e-12:
                        move_2 = f[j_other] * frac * 0.5
                        trial[j_other] -= move_2
                        # Redistribute to another free position
                        free_2 = free.copy()
                        free_2[j_target] = True
                        if free_2.sum() > 0:
                            free_2_idx = np.where(free_2)[0]
                            j_target2 = free_2_idx[rng.integers(len(free_2_idx))]
                            trial[j_target2] += move_2

                    trial = np.maximum(trial, 0)
                    ss = trial.sum()
                    if ss < 1e-15:
                        continue
                    trial /= ss
                    obj = fast_convolve(trial).max()
                    
                    if obj < best_obj - 1e-14:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break
            
            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve > 20:
                # More aggressive perturbation
                noise = 0.001 + 0.002 * min(no_improve / 50.0, 1.0)
                f = project_simplex(best_f + rng.standard_normal(n) * noise)
                no_improve = 0
        else:
            no_improve = 0

    return best_f, 2 * n * best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Fine-grained coordinate descent: try small adjustments to each element."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        # Try adjusting random elements
        num_trials = min(200, max(50, 50000 // n))
        for _ in range(num_trials):
            if time.monotonic() - t0 > time_limit:
                break
            
            j = rng.integers(n)
            if f[j] < 1e-12:
                continue
            
            # Try reducing f[j] and redistributing
            for factor in [0.7, 0.85, 0.95]:
                old = f[j]
                trial = f.copy()
                trial[j] = old * factor
                diff = old - trial[j]
                
                # Redistribute proportionally to other elements
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                if f[mask].sum() < 1e-15:
                    continue
                trial[mask] += diff * f[mask] / f[mask].sum()
                
                trial = np.maximum(trial, 0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial /= ss
                obj = fast_convolve(trial).max()
                
                if obj < best_obj - 1e-14:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    improved = True
                    break
            
            if improved:
                break

        if not improved:
            # Perturb
            noise = 0.0005 + 0.001 * (time.monotonic() - t0) / time_limit
            f = project_simplex(best_f + rng.standard_normal(n) * noise)

    return best_f, 2 * n * best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_diverse_init(n, style, rng):
    """Create diverse initializations of various types."""
    if style == 0:
        center = 0.15 + 0.4 * rng.random()
        sigma = 0.12 + 0.2 * rng.random()
        x = np.arange(n) - n * center
        a = np.exp(-0.5 * (x / (n * sigma)) ** 2)
    elif style == 1:
        rate = 1.0 + 4.0 * rng.random()
        a = np.exp(-np.arange(n) / (n / rate))
    elif style == 2:
        sigma = 0.2 + 0.25 * rng.random()
        x = np.arange(n) - n / 2.0
        a = np.exp(-0.5 * (x / (n * sigma)) ** 2)
    elif style == 3:
        c1 = 0.1 + 0.2 * rng.random()
        c2 = 0.5 + 0.35 * rng.random()
        s1 = 0.04 + 0.08 * rng.random()
        s2 = 0.04 + 0.08 * rng.random()
        w1 = 0.3 + 0.4 * rng.random()
        x = np.arange(n) / n
        a = w1 * np.exp(-((x - c1) / s1) ** 2) + (1 - w1) * np.exp(-((x - c2) / s2) ** 2)
    elif style == 4:
        a = np.linspace(0.05, 1.0, n)
    elif style == 5:
        k = max(3, n // 8)
        c = np.cumsum(rng.standard_normal(k))
        c = c - c.min() + 0.5
        a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, k), c)
    elif style == 6:
        # Sparse: a few random bumps
        num_bumps = 2 + rng.integers(3)
        a = np.zeros(n)
        for _ in range(num_bumps):
            c = rng.random()
            s = 0.02 + 0.08 * rng.random()
            amp = 0.5 + rng.random()
            x = np.arange(n) / n
            a += amp * np.exp(-((x - c) / s) ** 2)
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization pipeline:
    Phase 1: Diverse initialization across multiple n values (40s)
    Phase 2: Adam annealing optimization (700s)
    Phase 3: LP-FW refinement (350s)
    Phase 4: Peak mutation search (200s)
    Phase 5: Multi-n exploration (150s)
    Phase 6: Coordinate descent + final polish (remainder)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            old = best_score
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)}) [from {old:.6f}]")
        return score

    # === Load parent if available ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Diverse initialization (40s) ===
    print("=== Phase 1: Diverse Initialization ===")
    t1_start = time.monotonic()
    t1_budget = 40

    n_candidates = [512, 640, 700, 768, 832, 896, 960, 1024, 1152]
    init_results = []

    if parent is not None:
        init_results.append((parent, len(parent), best_score))

    for n_try in n_candidates:
        if time.monotonic() - t1_start > t1_budget:
            break
        for style in range(7):
            if time.monotonic() - t1_start > t1_budget:
                break
            a_init = make_diverse_init(n_try, style, rng)
            sc = try_seq(a_init, f"n{n_try}_s{style}")
            init_results.append((a_init, n_try, sc))

    # Sort by score, keep top candidates
    init_results.sort(key=lambda x: x[2])
    print(f"  Best init: score={init_results[0][2]:.6f}, n={init_results[0][1]}")

    # === Phase 2: Adam annealing (main workhorse, 700s) ===
    print("\n=== Phase 2: Adam Annealing Optimization ===")
    t2_budget = 700

    # Run on top 3 diverse candidates (different n if possible)
    used_ns = set()
    top_candidates = []
    for a_init, n_try, sc in init_results:
        if n_try not in used_ns:
            top_candidates.append((a_init, n_try, sc))
            used_ns.add(n_try)
        if len(top_candidates) >= 3:
            break

    for idx, (a_init, n_try, sc) in enumerate(top_candidates):
        if time.monotonic() > deadline - 500:
            break
        rem = min(deadline - time.monotonic() - 500 + idx * 200, 
                  t2_budget - idx * 200)
        if rem < 50:
            rem = 50 if time.monotonic() < deadline - 500 else 0
        if rem <= 0:
            break

        print(f"  [{idx+1}] n={n_try}, init={sc:.6f}, time={rem:.0f}s")
        a_opt, obj = adam_anneal_optimize(a_init, time_limit=rem,
                                           lr_init=0.005, tau_decay=0.999,
                                           restart_every=250, verbose=True)
        try_seq(a_opt, f"adam_n{n_try}")

        # Second pass with tighter tau from the result
        rem2 = min(deadline - time.monotonic() - 500, 150)
        if rem2 > 30 and best_seq is not None and len(best_seq) == n_try:
            tau_tight = 0.03 * max(fast_convolve(best_seq).max(), 1e-10)
            a_opt2, obj2 = adam_anneal_optimize(best_seq, time_limit=rem2,
                                                 tau_init=tau_tight, lr_init=0.002,
                                                 tau_decay=0.9995, verbose=False)
            try_seq(a_opt2, f"adam_fine_n{n_try}")

    # === Phase 3: LP-FW refinement (350s) ===
    print("\n=== Phase 3: LP-FW Refinement ===")
    if best_seq is not None:
        rem = min(deadline - time.monotonic(), 350)
        if rem > 60:
            print(f"  n={len(best_seq)}, time={rem:.0f}s")
            a_opt, obj = lpfw_optimize(best_seq, time_limit=rem, verbose=True)
            try_seq(a_opt, "lpfw")

            # Another LP-FW round from perturbed start
            rem2 = min(deadline - time.monotonic(), 100)
            if rem2 > 30:
                perturbed = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.004)
                a_opt2, _ = lpfw_optimize(perturbed, time_limit=rem2)
                try_seq(a_opt2, "lpfw_pert")

    # === Phase 4: Peak mutation search (200s) ===
    print("\n=== Phase 4: Peak Mutation Search ===")
    if best_seq is not None:
        rem = min(deadline - time.monotonic(), 200)
        if rem > 30:
            print(f"  n={len(best_seq)}, time={rem:.0f}s")
            a_opt, obj = peak_mutation_search(best_seq, time_limit=rem, verbose=True)
            try_seq(a_opt, "mutation")

    # === Phase 5: Multi-n exploration (150s) ===
    print("\n=== Phase 5: Multi-n Exploration ===")
    if best_seq is not None:
        base = best_seq
        base_n = len(base)
        explored_ns = {base_n}

        # Try neighboring n values
        for dn in [64, -64, 128, -128, 192, -192]:
            target_n = base_n + dn
            if target_n in explored_ns or target_n < 256 or target_n > 2048:
                continue
            explored_ns.add(target_n)

            rem = deadline - time.monotonic()
            if rem < 60:
                break

            f_interp = interpolate_sequence(base, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")

            t_alloc = min(rem * 0.3, 60)
            if t_alloc > 20:
                a_opt, obj = adam_anneal_optimize(f_interp, time_limit=t_alloc,
                                                   lr_init=0.004, tau_decay=0.999,
                                                   restart_every=200)
                try_seq(a_opt, f"adam_n{target_n}")

    # === Phase 6: Coordinate descent polish ===
    print("\n=== Phase 6: Coordinate Descent Polish ===")
    if best_seq is not None:
        rem = min(deadline - time.monotonic(), 120)
        if rem > 20:
            a_opt, obj = coordinate_descent(best_seq, time_limit=rem)
            try_seq(a_opt, "coord_cd")

    # === Phase 7: Final Adam polish ===
    if best_seq is not None:
        rem = min(deadline - time.monotonic(), 60)
        if rem > 15:
            tau_tight = 0.02 * max(fast_convolve(best_seq).max(), 1e-10)
            a_opt, obj = adam_anneal_optimize(best_seq, time_limit=rem,
                                               tau_init=tau_tight, lr_init=0.001,
                                               tau_decay=0.9999, restart_every=100)
            try_seq(a_opt, "final_polish")

    # === Phase 8: Final LP-FW ===
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 10:
            a_opt, obj = lpfw_optimize(best_seq, time_limit=rem)
            try_seq(a_opt, "final_lpfw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
