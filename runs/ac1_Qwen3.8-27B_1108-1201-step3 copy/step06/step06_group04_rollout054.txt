"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using hybrid LP-Frank-Wolfe with
chunk-swap local search, go-far line search, population-based exploration,
and multi-length optimization.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

def obj_normalized(f):
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()

def build_lp_matrix(f, n):
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        A_ub[k_start:k_end + 1, j] = f[:length]
    return A_ub

def solve_lp_direction(f, M, n):
    A_ub = build_lp_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def line_search_go_far(f, g, n, max_far=5.0):
    """Line search that allows going far (alpha > 1) to escape local minima."""
    def eval_at(alpha):
        ft = f + alpha * (g - f)
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-15:
            return float('inf')
        ft = ft / s
        return fast_convolve(ft).max()
    
    # Try a range of alphas including > 1
    best_alpha = 0.0
    best_val = eval_at(0.0)
    alphas = np.concatenate([
        np.arange(0.0, 1.01, 0.1),
        np.arange(1.1, max_far + 0.01, 0.2)
    ])
    for a in alphas:
        v = eval_at(a)
        if v < best_val:
            best_val = v
            best_alpha = a
    
    return best_alpha, best_val

def chunk_swap_search(f, n, time_limit, verbose=False):
    """
    Local search that swaps contributions between convolution lags.
    For a peak at lag k, we reduce a[j]*a[k-j] for dominant pairs
    by transferring mass to indices that don't contribute to that peak.
    """
    f = np.maximum(f.copy(), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, obj_normalized(f)
    f = f / s
    
    best_obj = obj_normalized(f)
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 50:
        conv = fast_convolve(f)
        M = conv.max()
        improved = False
        
        # Get top peaks
        top_k = min(8, len(conv))
        peak_indices = np.argsort(conv)[-top_k:]
        
        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            
            idx = np.arange(js, je + 1)
            products = f[idx] * f[k - idx]
            total = products.sum()
            if total < 1e-15:
                continue
            
            # Try reducing top contributors and redistributing
            order = np.argsort(products)[::-1]
            
            for oi in range(min(5, len(order))):
                j_max = int(idx[order[oi]])
                partner = k - j_max
                old_val = f[j_max]
                if old_val < 1e-12:
                    continue
                
                # Try various reductions
                for frac in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
                    reduction = old_val * frac * 0.3  # 30% of frac * old_val
                    if reduction < 1e-10:
                        continue
                    
                    trial = f.copy()
                    trial[j_max] -= reduction
                    if trial[j_max] < 0:
                        continue
                    
                    # Redistribute to indices far from peak k
                    # Prefer indices that don't contribute to any top peak
                    non_contrib = np.ones(n, dtype=bool)
                    for pk in peak_indices[:4]:
                        pk = int(pk)
                        ps = max(0, pk - n + 1)
                        pe = min(n - 1, pk)
                        non_contrib[ps:pe+1] = False
                    
                    if not non_contrib.any():
                        non_contrib = np.ones(n, dtype=bool)
                        non_contrib[js:je+1] = False
                    
                    if not non_contrib.any():
                        continue
                    
                    # Distribute proportionally to existing values
                    w = trial[non_contrib].copy()
                    wsum = w.sum()
                    if wsum < 1e-15:
                        w = np.ones(n)[non_contrib]
                        wsum = w.sum()
                    trial[non_contrib] += reduction * w / wsum
                    
                    trial = np.maximum(trial, 0)
                    tsum = trial.sum()
                    if tsum < 1e-15:
                        continue
                    trial = trial / tsum
                    
                    obj = obj_normalized(trial)
                    if obj < best_obj - 1e-12:
                        best_obj = obj
                        best_f = trial.copy()
                        f = trial
                        improved = True
            
            if time.monotonic() - t0 > time_limit:
                break
        
        if not improved:
            no_improve += 1
            if verbose and no_improve % 10 == 0:
                print(f"    Chunk swap: stuck at obj={best_obj:.10f}, no_improve={no_improve}")
        else:
            no_improve = 0
    
    return best_f, best_obj

def go_far_lpfw(f0, time_limit, verbose=False):
    """LP-FW with go-far line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.2
            if m_exp > 8.0:
                # Reset with perturbation
                rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue
        
        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.1
            continue
        g = g / g_sum
        
        diff = np.max(np.abs(g - f))
        if diff < 1e-10:
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                M = fast_convolve(f).max()
                m_exp = 1.0
                no_improve = 0
            continue
        
        # Go-far line search
        alpha, alpha_val = line_search_go_far(f, g, n, max_far=3.0)
        
        if alpha < 1e-8:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 5.0)
            continue
        
        f_new = f + alpha * (g - f)
        f_new = np.maximum(f_new, 0.0)
        sn = f_new.sum()
        if sn < 1e-15:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 5.0)
            continue
        f_new = f_new / sn
        f = f_new
        
        conv = fast_convolve(f)
        M = conv.max()
        m_exp = 1.0
        
        current_obj = 2.0 * n * M
        if current_obj < best_obj - 1e-14:
            best_obj = current_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and no_improve % 10 == 0:
            print(f"    GoFar LP-FW: obj={best_obj:.10f}, no_improve={no_improve}, m_exp={m_exp:.2f}")
        
        if no_improve > 80:
            break
    
    return best_f, best_obj

def momentum_lpfw(f0, time_limit, verbose=False):
    """LP-FW with momentum (Nesterov-style acceleration)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_obj = 2.0 * n * fast_convolve(f).max()
    best_f = f.copy()
    
    # Momentum state
    momentum = np.zeros(n)
    beta = 0.5  # momentum coefficient
    beta_decay = 0.999
    
    t0 = time.monotonic()
    no_improve = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        M = fast_convolve(f).max()
        
        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.2
            if m_exp > 8.0:
                f = project_simplex(best_f + np.random.default_rng(int(time.monotonic()*1000)%100000).standard_normal(n) * 0.003)
                m_exp = 1.0
                no_improve = 0
            continue
        
        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.1
            continue
        g = g / g_sum
        
        # Nesterov step: use momentum to extrapolate
        direction = g - f
        momentum = beta * momentum + direction
        f_momentum = f + momentum
        f_momentum = np.maximum(f_momentum, 0)
        sm = f_momentum.sum()
        if sm > 1e-15:
            f_momentum = f_momentum / sm
        
        # Try both the momentum step and the standard step
        obj_mom = obj_normalized(f_momentum)
        obj_std = obj_normalized(g)
        
        if obj_mom < obj_std:
            f_new = f_momentum
            current_obj = obj_mom
        else:
            f_new = g
            current_obj = obj_std
        
        # Also try intermediate
        for alpha in [0.3, 0.5, 0.7]:
            trial = f + alpha * (g - f)
            trial = np.maximum(trial, 0)
            ts = trial.sum()
            if ts > 1e-15:
                trial = trial / ts
                ot = obj_normalized(trial)
                if ot < current_obj:
                    f_new = trial
                    current_obj = ot
        
        f = f_new
        beta *= beta_decay
        
        if current_obj < best_obj - 1e-14:
            best_obj = current_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and no_improve % 10 == 0:
            print(f"    Momentum LP-FW: obj={best_obj:.10f}, no_improve={no_improve}")
        
        if no_improve > 100:
            break
    
    return best_f, best_obj

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def make_asymmetric_gaussian(n, center_frac, sigma_frac):
    x = np.arange(n)
    center = n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_power_law(n, decay, offset=0.5):
    x = np.arange(n) / n
    a = (1.0 - x + offset) ** decay
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_oscillating(n, freq, decay):
    x = np.arange(n) / n
    a = np.exp(-decay * x) * (0.5 + 0.5 * np.cos(2 * np.pi * freq * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using hybrid LP-Frank-Wolfe with
go-far line search, momentum acceleration, chunk-swap local search,
and multi-length population-based exploration.

import numpy as np
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_normalized(f):
    """Objective for sum-normalized sequence: 2*n*max(conv)."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def eval_alpha(f, g, alpha, n):
    """Evaluate objective at f + alpha*(g-f), normalized."""
    ft = f + alpha * (g - f)
    ft = np.maximum(ft, 0.0)
    s = ft.sum()
    if s < 1e-15:
        return float('inf')
    ft = ft / s
    return fast_convolve(ft).max()


def build_lp_matrix(f, n):
    """Build the constraint matrix for the LP."""
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        A_ub[k_start:k_end + 1, j] = f[:length]
    return A_ub


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    A_ub = build_lp_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def line_search_gofar(f, g, n, max_far=3.0):
    """Line search allowing alpha > 1 to escape local minima."""
    best_alpha = 0.0
    best_val = eval_alpha(f, g, 0.0, n)
    
    alphas = np.concatenate([
        np.arange(0.05, 1.01, 0.05),
        np.arange(1.1, max_far + 0.01, 0.2)
    ])
    for a in alphas:
        v = eval_alpha(f, g, a, n)
        if v < best_val:
            best_val = v
            best_alpha = a
    
    # Refine around best
    if 0 < best_alpha < max_far:
        span = 0.1
        for a in np.arange(max(0.01, best_alpha - span), min(max_far, best_alpha + span) + 0.01, 0.02):
            v = eval_alpha(f, g, a, n)
            if v < best_val:
                best_val = v
                best_alpha = a
    
    return best_alpha, best_val

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_asym Gaussian(n, center_frac, sigma_frac):
    """Asymmetric Gaussian initial sequence."""
    x = np.arange(n)
    center = n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:

def make_power_law(n, decay, floor=0.05):
    """Power-law decay initial sequence."""
    x = np.arange(n) / n
    a = (1.0 - x + floor) ** decay

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using hybrid LP-Frank-Wolfe with
go-far line search, momentum acceleration, chunk-swap local search,
and multi-length population-based exploration.

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_normalized(f):
    """Objective for sum-normalized sequence: 2*n*max(conv)."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def eval_alpha(f, g, alpha, n):
    """Evaluate objective at f + alpha*(g-f), normalized."""
    ft = f + alpha * (g - f)

    if s < 1e-15:
        return float('inf')

def build_lp_matrix(f, n):
    """Build the constraint matrix for the LP."""
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        A_ub[k_start:k_end + 1, j] = f[:length]
    return A_ub


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    A_ub = build_lp_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

        np.arange(0.05, 1.01, 0.05),
        np.arange(1.1, max_far + 0.01, 0.2)
    ])
    for a in alphas:
        v = eval_alpha(f, g, a, n)
        if v < best_val:
            best_val = v
            best_alpha = a
    
    # Refine around best
    if 0 < best_alpha < max_far:
        span = 0.1
        lo = max(0.01, best_alpha - span)
        hi = min(max_far, best_alpha + span)
        for a in np.arange(lo, hi + 0.01, 0.02):
            v = eval_alpha(f, g, a, n)
            if v < best_val:
                best_val = v
                best_alpha = a
    
    return best_alpha, best_val

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
