    rng = np.random.default_rng(42)
    phases = rng.uniform(0, 2*np.pi, n)

    F[0] = np.abs(F[0])  # DC component positive

        F_new[0] = np.abs(F_new[0])
        f_new = np.fft.ifft(F_new).real
        f_new = np.maximum(f_new, 0)
        f_new = f_new / f_new.sum()

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(n, f):

    A[k,j] = f[k-j] if 0 <= k-j <= n-1, else 0."""

        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral initialization (constant modulus iteration for flat autocorrelation)
2. LP-based Frank-Wolfe with quadratic line search and adaptive M reduction
3. Multi-length exploration with diverse initializations
4. Coordinate descent for final refinement

Key improvements over previous approach:
- Spectral method to initialize with near-flat autocorrelation
- Quadratic line search exploiting the structure of FW updates
- Binary search on M for tighter convergence in the LP phase
- More aggressive multi-length exploration
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(n, f):
    """Build (2n-1) x n matrix where A[k,j] = f[k-j] if 0<=k-j<=n-1."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    A = build_constraint_matrix(n, f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def quadratic_line_search(f, g, n, max_iters=80):
    """
    Find alpha in [0,1] that minimizes max_k ((1-alpha)*f + alpha*g) * ((1-alpha)*f + alpha*g)_k
    Uses coarse grid + ternary search.
    Returns (best_alpha, best_value).
    """
    d = g - f
    # Precompute coefficients
    c0 = fast_convolve(f)       # (f*f)_k
    c1 = np.zeros(2 * n - 1)    # (f*d)_k = cross-correlation
    c2 = fast_convolve(d)       # (d*d)_k
    
    # Compute (f*d)_k = sum_j f[j] * d[k-j]
    # This is a cross-correlation. Use FFT.
    fft_size = 1 << ((2 * n - 1) - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    D = np.fft.rfft(d, fft_size)
    # (f*d)_k = sum_j f[j] * d[k-j] = convolve(f, reverse(d))[k + n - 1]
    d_rev = d[::-1]
    Dr = np.fft.rfft(d_rev, fft_size)
    conv_fd = np.fft.irfft(F * Dr, fft_size)[:2 * n - 1]
    # conv_fd[k] = sum_j f[j] * d_rev[k-j] = sum_j f[j] * d[n-1-(k-j)] = sum_j f[j] * d[n-1-k+j]
    # We want (f*d)_k = sum_j f[j] * d[k-j]
    # Let's use a different approach:
    # (f*d)_k = correlation of f and d at lag k
    # In FFT: ifft(fft(f) * conj(fft(d)))... but that's for circular correlation.
    # For linear: use zero padding.
    
    # Actually, let me just compute it directly using the same FFT approach as fast_convolve
    # But for cross-convolution
    # (f*d)_k = sum_j f[j]*d[k-j] where 0<=j<n, 0<=k-j<n
    # = sum_j f[j]*d[k-j] * I[0<=k-j<n]
    
    # Using FFT: pad both to size >= 2n-1
    F = np.fft.rfft(f, fft_size)
    D = np.fft.rfft(d, fft_size)
    # ifft(F * D) gives circular convolution, which equals linear convolution if padded enough
    # linear_conv(f, d)[k] = sum_j f[j]*d[k-j] for 0<=k<=2n-2
    cross = np.fft.irfft(F * D, fft_size)[:2 * n - 1]
    c1 = cross
    
    # Now h_k(alpha) = c0[k] + 2*c1[k]*alpha + c2[k]*alpha^2
    # h(alpha) = max_k h_k(alpha)
    
    def h(alpha):
        vals = c0 + 2 * c1 * alpha + c2 * alpha * alpha
        return np.max(vals)
    
    # Coarse grid search
    grid_alphas = np.linspace(0, 1, 21)
    grid_vals = [h(a) for a in grid_alphas]
    best_idx = np.argmin(grid_vals)
    
    # Refine with ternary search in the best interval
    if best_idx == 0:
        lo, hi = 0.0, grid_alphas[1]
    elif best_idx == len(grid_alphas) - 1:
        lo, hi = grid_alphas[-2], 1.0
    else:
        lo, hi = grid_alphas[best_idx - 1], grid_alphas[best_idx + 1]
    
    for _ in range(40):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if h(m1) < h(m2):
            hi = m2
        else:
            lo = m1
    
    best_alpha = (lo + hi) / 2.0
    best_val = h(best_alpha)
    
    # Check endpoints
    v0 = h(0.0)
    v1 = h(1.0)
    if v0 < best_val:
        best_alpha = 0.0
        best_val = v0
    if v1 < best_val:
        best_alpha = 1.0
        best_val = v1
    
    return best_alpha, best_val


def lpfw_optimize(f0, time_limit, verbose=False):
    """
    LP-based Frank-Wolfe with quadratic line search.
    Minimizes max_k (f*f)_k subject to f >= 0, sum(f) = 1.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    c = fast_convolve(f)
    M = c.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        t_iter_start = time.monotonic()
        
        # Solve LP direction
        g = solve_lp_direction(f, M, n)
        if g is None:
            # Try with a larger M
            g = solve_lp_direction(f, M * 1.1, n)
            if g is None:
                no_improve = 9999
                break
            M = M * 1.1
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            no_improve += 100
            break
        g = g / g_sum
        
        # Check fixed point
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            continue
        else:
            no_improve = 0
        
        # Quadratic line search
        alpha, val = quadratic_line_search(f, g, n)
        
        if alpha < 1e-10:
            no_improve += 1
            if no_improve > 100:
                # Perturb and restart
                rng = np.random.default_rng(it * 13 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                c = fast_convolve(f)
                M = c.max()
                no_improve = 0
            continue
        
        # Update
        f_new = (1 - alpha) * f + alpha * g
        f_new = np.maximum(f_new, 0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            no_improve += 10
            continue
        f_new = f_new / s_new
        
        c_new = fast_convolve(f_new)
        M_new = c_new.max()
        
        if M_new < best_obj - 1e-14:
            best_obj = M_new
            best_f = f_new.copy()
            if verbose and it % 20 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LPFW iter {it}: obj={best_obj:.12f}, alpha={alpha:.4f}, "
                      f"gap={np.max(np.abs(g-f)):.2e}, time={elapsed:.0f}s/{time_limit:.0f}s")
        else:
            # Accept the update anyway (it might help escape local minima later)
            pass
        
        f = f_new
        M = c_new.max()
        
        # Periodic perturbation to escape local minima
        if it % 500 == 0 and it > 100:
            rng_pert = np.random.default_rng(it * 31 + 17)
            f_pert = project_simplex(f + rng_pert.standard_normal(n) * 0.001)
            c_pert = fast_convolve(f_pert)
            if c_pert.max() < M:
                f = f_pert
                M = c_pert.max()
    
    return best_f, best_obj


def lpfw_with_m_reduction(f0, time_limit, verbose=False):
    """
    LP-FW with binary search on M for faster convergence.
    At each iteration, binary search for the minimum M such that the LP
    gives sum(g) >= 1. This gives a tighter direction.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    c = fast_convolve(f)
    M = c.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Binary search on M to find the critical value
        lo = M * 0.5
        hi = M * 1.01
        
        # First check if even M works
        g_hi = solve_lp_direction(f, hi, n)
        if g_hi is None or g_hi.sum() < 1e-10:
            no_improve += 50
            if no_improve > 200:
                break
            continue
        
        # Binary search for the threshold
        if it < 5:
            # Use full binary search for first few iterations
            for _ in range(15):
                mid = (lo + hi) / 2
                g_mid = solve_lp_direction(f, mid, n)
                if g_mid is not None and g_mid.sum() >= 1.0 - 1e-8:
                    hi = mid
                else:
                    lo = mid
            M_crit = hi
        else:
            # After initial convergence, use M = M_crit directly
            M_crit = M
        
        # Solve at M_crit
        g = solve_lp_direction(f, M_crit, n)
        if g is None:
            g = solve_lp_direction(f, M, n)
            M_crit = M
        if g is None:
            no_improve += 50
            if no_improve > 200:
                break
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            no_improve += 50
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 100:
                rng = np.random.default_rng(it * 19 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                c = fast_convolve(f)
                M = c.max()
                no_improve = 0
            continue
        else:
            no_improve = 0
        
        # Quadratic line search
        alpha, val = quadratic_line_search(f, g, n)
        
        if alpha < 1e-10:
            no_improve += 1
            if no_improve > 100:
                break
            continue
        
        f_new = (1 - alpha) * f + alpha * g
        f_new = np.maximum(f_new, 0)
        f_new = f_new / f_new.sum()
        
        c_new = fast_convolve(f_new)
        M_new = c_new.max()
        
        if M_new < best_obj - 1e-14:
            best_obj = M_new
            best_f = f_new.copy()
            if verbose and it % 30 == 0:
                print(f"    LPFW-M iter {it}: obj={best_obj:.12f}, M_crit={M_crit:.8f}, "
                      f"time={time.monotonic()-t0:.0f}s")
        
        f = f_new
        M = c_new.max()
        
        # Update M_crit for next iteration
        if M_new < M_crit:
            M_crit = M_new
    
    return best_f, best_obj


def spectral_init(n, iters=200, seed=42):
    """
    Initialize using constant modulus iteration to get near-flat autocorrelation.
    """
    rng = np.random.default_rng(seed)
    # Start with random phases, flat magnitude
    F = np.exp(1j * rng.uniform(0, 2 * np.pi, n))
    F[0] = abs(F[0])  # DC positive
    
    f = np.fft.ifft(F).real
    f = np.maximum(f, 0)
    f = f / f.sum()
    
    for _ in range(iters):
        F = np.fft.fft(f)
        # Equalize the power spectrum
        target_power = np.mean(np.abs(F) ** 2)
        F_new = F / np.maximum(np.abs(F), 1e-15) * np.sqrt(target_power)
        F_new[0] = abs(F_new[0])
        f_new = np.fft.ifft(F_new).real
        f_new = np.maximum(f_new, 0)
        f_new = f_new / f_new.sum()
        f = f_new
    
    return f


def make_gaussian(n, sigma_frac=0.3, seed=42):
    """Gaussian initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_hann(n):
    """Hann window initial sequence."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    return f_interp


def coordinate_polish(f0, time_limit, verbose=False):
    """
    Fine coordinate descent: adjust individual coordinates to reduce max convolution.
    Uses binary search on each coordinate for precision.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f0, fast_convolve(f0).max()
    f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    sweep = 0
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False
        c = fast_convolve(f)
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-14:
                continue
            
            # Try to reduce the peak by adjusting f[j]
            # If f[j] contributes to the peak, reducing it helps
            # Find which lag k has the peak
            peak_k = np.argmax(c)
            
            # Does f[j] contribute to c[peak_k]?
            # c[peak_k] = sum_i f[i] * f[peak_k - i]
            # f[j] appears in c[peak_k] if 0 <= peak_k - j < n, contributing f[j] * f[peak_k - j]
            # and also if 0 <= j - peak_k < n (but peak_k >= 0 and j >= 0, so j - peak_k could be negative)
            # Actually c has indices 0..2n-2, and c[k] = sum_i f[i]*f[k-i] for valid i, k-i
            # f[j] appears in c[k] when j is a valid index: 0<=j<n and 0<=k-j<n
            # So f[j] contributes f[j]*f[k-j] to c[k] if 0 <= k-j <= n-1
            
            if 0 <= peak_k - j <= n - 1:
                # f[j] contributes to the peak
                # Try reducing f[j]
                for frac in [0.85, 0.9, 0.95, 1.05, 1.1]:
                    new_val = old_val * frac
                    if new_val < 0:
                        continue
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    s_trial = f_trial.sum()
                    if s_trial < 1e-15:
                        continue
                    f_trial = f_trial / s_trial
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-15:
                        best_obj = val
                        best_f = f_trial.copy()
                        f = f_trial
                        improved = True
                        break
        
        if not improved and sweep > 2:
            break
    
    return best_f, best_obj

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Linear cross-convolution (a*b)_k = sum_j a[j]*b[k-j] via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(n, f):
    """Build (2n-1) x n matrix where A[k,j] = f[k-j] if 0<=k-j<=n-1 else 0."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    A = build_constraint_matrix(n, f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def quadratic_line_search(f, g, n):
    """
    Find alpha in [0,1] minimizing max_k ((1-alpha)*f + alpha*g * (1-alpha)*f + alpha*g)_k.
    Uses the quadratic structure: h_k(alpha) = c0_k + 2*c1_k*alpha + c2_k*alpha^2.
    Returns (best_alpha, best_value).
    """
    d = g - f
    c0 = fast_convolve(f)
    c1 = cross_convolve(f, d)
    c2 = fast_convolve(d)

    # h(alpha) = max_k (c0 + 2*c1*alpha + c2*alpha^2)
    # Coarse grid search
    grid = np.linspace(0.0, 1.0, 25)
    vals = np.array([np.max(c0 + 2 * c1 * a + c2 * a * a) for a in grid])
    best_idx = np.argmin(vals)

    # Ternary search in best interval
    if best_idx == 0:
        lo, hi = 0.0, grid[1]
    elif best_idx == len(grid) - 1:
        lo, hi = grid[-2], 1.0
    else:
        lo, hi = grid[best_idx - 1], grid[best_idx + 1]

    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        v1 = np.max(c0 + 2 * c1 * m1 + c2 * m1 * m1)
        v2 = np.max(c0 + 2 * c1 * m2 + c2 * m2 * m2)
        if v1 < v2:
            hi = m2
        else:
            lo = m1

    best_alpha = (lo + hi) / 2.0
    best_val = np.max(c0 + 2 * c1 * best_alpha + c2 * best_alpha * best_alpha)

    # Check endpoints
    v0 = np.max(c0)
    v1 = np.max(c0 + 2 * c1 + c2)
    if v0 < best_val:
        best_alpha, best_val = 0.0, v0
    if v1 < best_val:
        best_alpha, best_val = 1.0, v1

    return best_alpha, best_val


def lpfw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with quadratic line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    c = fast_convolve(f)
    M = c.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = solve_lp_direction(f, M, n)
        if g is None:
            g = solve_lp_direction(f, M * 1.05, n)
            if g is None:
                no_improve += 100
                if no_improve > 300:
                    break
                M *= 1.1
                continue
            M *= 1.05

        g_sum = g.sum()
        if g_sum < 1e-15:
            no_improve += 100
            break
        g = g / g_sum

        gap = np.max(np.abs(g - f))
        if gap < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            continue
        else:
            no_improve = 0

        alpha, val = quadratic_line_search(f, g, n)

        if alpha < 1e-12:
            no_improve += 1
            if no_improve > 100:
                rng = np.random.default_rng(it * 13 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                c = fast_convolve(f)
                M = c.max()
                no_improve = 0
            continue

        f_new = (1.0 - alpha) * f + alpha * g
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            no_improve += 10
            continue
        f_new = f_new / s_new

        c_new = fast_convolve(f_new)
        M_new = c_new.max()

        if M_new < best_obj - 1e-14:
            best_obj = M_new
            best_f = f_new.copy()
            if verbose and it % 30 == 0:
                print(f"    LPFW iter {it}: obj={best_obj:.12f}, alpha={alpha:.4f}, "
                      f"gap={gap:.2e}, t={time.monotonic()-t0:.0f}s/{time_limit:.0f}s")

        f = f_new
        M = c_new.max()

        if it % 800 == 0 and it > 200:
            rng_p = np.random.default_rng(it * 31 + 17)
            f_p = project_simplex(f + rng_p.standard_normal(n) * 0.001)
            c_p = fast_convolve(f_p)
            if c_p.max() < M * 1.001:
                f = f_p
                M = c_p.max()

    return best_f, best_obj


def lpfw_m_reduction(f0, time_limit, verbose=False):
    """
    LP-FW with binary search on M for faster initial convergence, then
    switches to standard LP-FW with quadratic line search.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    c = fast_convolve(f)
    M = c.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    phase = 'binary'

    while time.monotonic() - t0 < time_limit:
        it += 1

        if phase == 'binary' and it <= 20:
            # Binary search phase: find tight M
            lo = M * 0.4
            hi = M * 1.01
            for _ in range(12):
                mid = (lo + hi) / 2.0
                g_mid = solve_lp_direction(f, mid, n)
                if g_mid is not None and g_mid.sum() >= 1.0 - 1e-6:
                    hi = mid
                else:
                    lo = mid
            M_crit = hi
            g = solve_lp_direction(f, M_crit, n)
            if g is None:
                g = solve_lp_direction(f, M, n)
                M_crit = M
            if g is None:
                no_improve += 50
                if no_improve > 200:
                    break
                continue
        else:
            if phase == 'binary' and it == 21:
                phase = 'refine'
            g = solve_lp_direction(f, M, n)
            if g is None:
                g = solve_lp_direction(f, M * 1.02, n)
                if g is None:
                    no_improve += 50
                    if no_improve > 200:
                        break
                    M *= 1.05
                    continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            no_improve += 50
            break
        g = g / g_sum

        gap = np.max(np.abs(g - f))
        if gap < 1e-11:
            no_improve += 1
            if no_improve > 80:
                rng = np.random.default_rng(it * 19 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                c = fast_convolve(f)
                M = c.max()
                no_improve = 0
            continue
        else:
            no_improve = 0

        alpha, val = quadratic_line_search(f, g, n)

        if alpha < 1e-12:
            no_improve += 1
            if no_improve > 80:
                break
            continue

        f_new = (1.0 - alpha) * f + alpha * g
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()

        c_new = fast_convolve(f_new)
        M_new = c_new.max()

        if M_new < best_obj - 1e-14:
            best_obj = M_new
            best_f = f_new.copy()
            if verbose and it % 20 == 0:
                print(f"    LPFW-M iter {it}: obj={best_obj:.12f}, t={time.monotonic()-t0:.0f}s/{time_limit:.0f}s")

        f = f_new
        M = c_new.max()

    return best_f, best_obj


def spectral_init(n, iters=200, seed=42):
    """Initialize using constant modulus iteration for near-flat autocorrelation."""
    rng = np.random.default_rng(seed)
    F = np.exp(1j * rng.uniform(0, 2 * np.pi, n))
    F[0] = abs(F[0])

    f = np.fft.ifft(F).real
    f = np.maximum(f, 0.0)
    f = f / f.sum()

    for _ in range(iters):
        F = np.fft.fft(f)
        abs_F = np.abs(F)
        target_power = np.mean(abs_F ** 2)
        F_new = F / np.maximum(abs_F, 1e-15) * np.sqrt(target_power)
        F_new[0] = abs(F_new[0])
        f_new = np.fft.ifft(F_new).real
        f_new = np.maximum(f_new, 0.0)
        ss = f_new.sum()
        if ss < 1e-15:
            break
        f_new = f_new / ss
        f = f_new

    return f


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_hann(n):
    """Hann window initial sequence."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6, power=4):
    """Bell-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0.0, 1.0)
    a = 1.0 - t ** power
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length with light smoothing."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    kernel = np.array([0.1, 0.2, 0.4, 0.2, 0.1])
    f_smooth = np.convolve(f_interp, kernel, mode='same')
    f_smooth = np.maximum(f_smooth, 0.0)
    ss = f_smooth.sum()
    if ss > 1e-15:
        f_smooth /= ss
    return f_smooth


def coordinate_polish(f0, time_limit):
    """Coordinate descent: adjust individual coordinates to reduce max convolution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f0, fast_convolve(f0).max()
    f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    sweep = 0
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False
        c = fast_convolve(f)
        peak_k = np.argmax(c)

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue

            # Check if f[j] contributes to the peak
            if 0 <= peak_k - j <= n - 1:
                for frac in [0.8, 0.85, 0.9, 0.95, 1.05, 1.1, 1.15]:
                    new_val = old_val * frac
                    if new_val < 0:
                        continue
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    s_trial = f_trial.sum()
                    if s_trial < 1e-15:
                        continue
                    f_trial = f_trial / s_trial
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-15:
                        best_obj = val
                        best_f = f_trial.copy()
                        f = f_trial
                        c = fast_convolve(f)
                        peak_k = np.argmax(c)
                        improved = True
                        break

        if not improved and sweep > 3:
            break

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Diverse initialization (spectral, Gaussian, Hann, parent, perturbations)
    Phase 2: LP-FW with quadratic line search (main optimizer)
    Phase 3: LP-FW with M-reduction (faster convergence variant)
    Phase 4: Multi-length exploration
    Phase 5: Final coordinate descent polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Helper to evaluate and track
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Diverse Initialization ===
    print("=== Phase 1: Initialization ===")
    t1 = time.monotonic()

    n_parent = len(parent) if parent is not None else 768
    init_candidates = []

    if parent is not None:
        init_candidates.append(("parent", parent.copy()))
        for trial in range(4):
            noise = rng.standard_normal(n_parent) * (0.01 + 0.01 * trial)
            a_pert = np.maximum(parent + noise, 0.0)
            a_pert /= a_pert.sum()
            init_candidates.append((f"parent_n{trial}", a_pert))

    # Spectral initialization (novel)
    a_spec = spectral_init(n_parent, iters=150, seed=seed)
    init_candidates.append(("spectral", a_spec))
    a_spec2 = spectral_init(n_parent, iters=150, seed=seed + 100)
    init_candidates.append(("spectral2", a_spec2))

    # Standard shapes
    init_candidates.append(("gauss", make_gaussian(n_parent, 0.25)))
    init_candidates.append(("gauss_wide", make_gaussian(n_parent, 0.4)))
    init_candidates.append(("hann", make_hann(n_parent)))
    init_candidates.append(("bell", make_bell(n_parent, 0.6, 4)))

    # Evaluate and rank
    init_scores = []
    for name, a in init_candidates:
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            continue
        a = a / s
        c = fast_convolve(a)
        obj = 2.0 * len(a) * c.max()
        init_scores.append((obj, name, a))
        try_seq(a, f"init_{name}")

    init_scores.sort(key=lambda x: x[0])
    print(f"  Top 3 inits: {[(n2, f'{o:.6f}') for o, n2, _ in init_scores[:3]]}")
    print(f"  Phase 1 time: {time.monotonic()-t1:.1f}s")

    # === Phase 2: Main LP-FW on best initialization ===
    print(f"\n=== Phase 2: LP-FW Main (n={len(init_scores[0][2])}) ===")
    t2 = time.monotonic()
    best_init = init_scores[0][2]

    rem = deadline - time.monotonic() - 500
    t_alloc = max(60, min(rem * 0.55, 700))
    print(f"  Time allocation: {t_alloc:.0f}s")

    a_opt, obj = lpfw_optimize(best_init, time_limit=t_alloc, verbose=True)
    try_seq(a_opt, "lpfw_main")

    # Try second-best init if significantly different
    if len(init_scores) > 1:
        rem = deadline - time.monotonic() - 400
        if rem > 100:
            t_alloc2 = min(rem * 0.3, 200)
            a_opt2, _ = lpfw_optimize(init_scores[1][2], time_limit=t_alloc2)
            try_seq(a_opt2, f"lpfw_{init_scores[1][1]}")

    print(f"  Phase 2 time: {time.monotonic()-t2:.1f}s")

    # === Phase 3: LP-FW with M-reduction ===
    print(f"\n=== Phase 3: LP-FW M-Reduction ===")
    t3 = time.monotonic()

    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 60:
            t_alloc = min(rem * 0.4, 300)
            print(f"  Time allocation: {t_alloc:.0f}s")
            a_opt3, _ = lpfw_m_reduction(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt3, "lpfw_mred")

    print(f"  Phase 3 time: {time.monotonic()-t3:.1f}s")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-Length Exploration ===")
    t4 = time.monotonic()
    base = best_seq if best_seq is not None else best_init
    base_n = len(base)

    target_lengths = []
    for factor in [0.75, 1.25, 1.5, 0.6]:
        n_try = int(round(base_n * factor / 32)) * 32
        n_try = max(64, min(2048, n_try))
        if n_try != base_n and n_try not in target_lengths:
            target_lengths.append(n_try)

    for n_abs in [512, 768, 1024, 1280, 1536, 2048]:
        if n_abs != base_n and n_abs not in target_lengths:
            target_lengths.append(n_abs)

    target_lengths.sort()

    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 150
        if rem < 80:
            break

        f_interp = interpolate_sequence(base, target_n)
        try_seq(f_interp, f"interp_n{target_n}")

        # LP-FW on interpolated sequence
        t_gd = min(rem * 0.25, 120)
        if t_gd > 30:
            a_opt_n, _ = lpfw_optimize(f_interp, time_limit=t_gd)
            try_seq(a_opt_n, f"lpfw_n{target_n}")

    print(f"  Phase 4 time: {time.monotonic()-t4:.1f}s")

    # === Phase 5: Diverse LP-FW restarts on best ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    t5 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(5):
            rem = deadline - time.monotonic() - 60
            if rem < 40:
                break
            t_alloc = min(rem / (5 - p), 100)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.003
                f_init = project_simplex(best_seq + noise)
            elif p == 2:
                rc = make_hann(n_best)
                f_init = 0.8 * best_seq + 0.2 * rc
                f_init = np.maximum(f_init, 0.0)
                f_init /= f_init.sum()
            elif p == 3:
                f_init = best_seq[::-1].copy()
            else:
                spec = spectral_init(n_best, iters=50, seed=seed + p * 7)
                f_init = 0.6 * best_seq + 0.4 * spec
                f_init = np.maximum(f_init, 0.0)
                f_init /= f_init.sum()

            a_opt_p, _ = lpfw_optimize(f_init, time_limit=t_alloc)
            try_seq(a_opt_p, f"restart_{p}")

    print(f"  Phase 5 time: {time.monotonic()-t5:.1f}s")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    t6 = time.monotonic()

    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_coord = min(rem * 0.5, 60)
            a_coord, _ = coordinate_polish(best_seq, time_limit=t_coord)
            try_seq(a_coord, "coord_polish")

        rem = deadline - time.monotonic() - 10
        if rem > 15:
            t_final = min(rem, 30)
            a_final, _ = lpfw_optimize(best_seq, time_limit=t_final)
            try_seq(a_final, "final_squeeze")

    print(f"  Phase 6 time: {time.monotonic()-t6:.1f}s")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
