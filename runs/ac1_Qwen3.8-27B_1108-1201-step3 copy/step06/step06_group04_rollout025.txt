Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using annealed smoothed gradient

gradient computation, and multi-scale restart strategy.

import time

def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max_gradient(a, w):
    """
    Compute gradient of L_t = (1/t)log(sum_k exp(t*b[k])) w.r.t. a.
    grad[m] = 2 * sum_k w[k] * a[k-m]
    where w[k] = exp(t*(b[k]-max_b)) / sum_j exp(t*(b[j]-max_b))
    """
    n = len(a)
    # Use FFT-based cross-correlation
    N = 1 << (3 * n).bit_length()
    A = np.fft.rfft(a, N)
    w_rev = w[::-1].copy()
    W_rev = np.fft.rfft(w_rev, N)
    conv = np.fft.irfft(A * W_rev, N)
    # cross_corr[m] = conv[(2n-2)-m] for m = 0,...,n-1
    grad = 2.0 * conv[n-1:2*n-1][::-1]
    return grad

def smooth_gradient_descent(a0, time_limit, t_start=1.0, t_end=200.0, 
                            t_growth=1.02, lr_start=0.1, lr_decay=0.9999,
                            verbose=False):
    """Annealed smoothed gradient descent."""
    a = a0.copy()
    n = len(a)
    
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    
    t = t_start
    lr = lr_start
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Compute softmax weights
        if t > 50:
            # Use top-k approximation for numerical stability
            k_top = min(50, len(b))
            top_idx = np.argpartition(b, -k_top)[-k_top:]
            b_shifted = b - max_b
            b_shifted[top_idx] = b[top_idx] - max_b  # already done
            # Only compute exp for top elements
            w = np.zeros(len(b))
            vals = b[top_idx] - max_b
            vals = vals - vals.max() if len(vals) > 0 else vals  # for stability, already shifted
            exp_vals = np.exp(t * vals)
            w[top_idx] = exp_vals
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.zeros(len(b))
                w[int(np.argmax(b))] = 1.0
            else:
                w /= w_sum
        else:
            b_shifted = b - max_b
            exp_vals = np.exp(t * b_shifted)
            w = exp_vals / exp_vals.sum()
        
        # Compute gradient
        grad = smooth_max_gradient(a, w)
        
        # Gradient descent step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # Check improvement
        obj = 2.0 * n * fast_convolve(a_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Anneal temperature
        if no_improve < 5:
            t = min(t * t_growth, t_end)
        else:
            t = max(t * 0.95, t_start)  # Cool back if not improving
        
        # Decay learning rate
        lr *= lr_decay
        if lr < 1e-8:
            break
        
        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(it)
            a = np.maximum(best_a + rng.standard_normal(n) * 0.005, 0)
            a = a / a.sum()
            t = t_start
            lr = lr_start
            no_improve = 0
        
        if verbose and it % 100 == 0:
            print(f"    SGD iter {it}: obj={best_obj:.10f}, t={t:.1f}, lr={lr:.2e}")
    
    return best_a, best_obj

- `lambda_k (b[k] - t) = 0` (complementary slackness)

From the second condition: `sum_k lambda_k * 2*a[k-m] = mu + nu[m]`.

For `a[m] > 0` (free variable): `nu[m] = 0`, so `sum_k lambda_k * 2*a[k-m] = mu`.

This means: at the optimum, for all free coordinates `m`, the weighted sum of `a[k-m]` over active peaks is constant.

This gives us an iterative algorithm:
1. Start with `a` and initialize `lambda` (e.g., uniform over top peaks).
2. Compute the "pressure" `P[m] = sum_k lambda_k * a[k-m]`.
3. Update `a` to make `P[m]` more uniform (for free coordinates).
4. Update `lambda` based on which peaks are active.
5. Repeat.

A practical version:
1. Compute `b = conv(a, a)`.
2. Identify active peaks: top `K` values or those above a threshold.
3. Set `lambda` proportional to the excess: `lambda[k] = max(b[k] - threshold, 0)^p` for some `p`.
4. Compute pressure: `P[m] = 2 * sum_k lambda[k] * a[k-m]`.
5. Update: `a[m] <- a[m] * (1 - eta * (P[m] - mean(P)))` or similar.
6. Normalize.

This is essentially a **primal-dual** or **mirror descent** approach.

For the active set equalization: suppose peaks at positions `k_1, ..., k_p` are active (equal to max). We want to find a perturbation `delta` such that:
- `db[k_i]/da * delta = b[k_i] - target` for all `i` (where `target` is the new target level)
- `sum(delta) = 0`

The Jacobian row for peak `k_i` is `J[i, m] = 2 * a[k_i - m]` (for valid `m`).

We have a `p x n` system (with `p << n`), and we want the minimum-norm solution:
`delta = J^T (J J^T + lambda*I)^{-1} (b_active - target)`

This is a ridge regression! And `J J^T` is `p x p` (small), so it's computationally cheap.

Let me implement this:

```python
def active_set_equalization(a, k_active, time_limit, verbose=False):
    """Newton-like step to equalize active peaks."""
    a = a.copy()
    n = len(a)
    
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        max_b = b.max()
        b_active = b[k_active]
        target = b_active.min()  # Lowest active peak is the target
        
        # Check if well-equalized
        if b_active.max() - target < 1e-10:
            break
        
        # Build Jacobian: J[i, m] = 2 * a[k_active[i] - m] for valid m
        J = np.zeros((len(k_active), n))
        for i, k in enumerate(k_active):
            for m in range(n):
                km = k - m
                if 0 <= km < n:
                    J[i, m] = 2.0 * a[km]
        
        # Solve: minimize ||J delta - (b_active - target)||^2 + lambda||delta||^2
        # subject to sum(delta) = 0
        # Use constrained least squares
        rhs = b_active - target
        # Add constraint row: sum(delta) = 0
        n_constraints = len(k_active) + 1
        A = np.zeros((n_constraints, n))
        A[:len(k_active), :] = J
        A[len(k_active), :] = 1.0  # sum constraint
        rhs_full = np.concatenate([rhs, [0.0]])
        
        # Regularized least squares: delta = A^T (A A^T + lam_reg * I)^{-1} rhs_full
        lam_reg = 1e-6
        ATA = A @ A.T + lam_reg * np.eye(n_constraints)
        # Solve ATA * y = rhs_full, then delta = A^T * y
        try:
            y = np.linalg.solve(ATA, rhs_full)
            delta = A.T @ y
        except:
            delta = np.zeros(n)
        
        # Clip delta to maintain non-negativity
        # Find step size alpha such that a + alpha*delta >= 0
        neg_mask = delta < 0
        if neg_mask.any():
            alpha_max = np.min(a[neg_mask] / (-delta[neg_mask]))
        else:
            alpha_max = 1.0
        alpha = min(1.0, alpha_max * 0.9)
        
        a_new = a + alpha * delta
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        obj = 2.0 * n * fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            alpha = alpha * 0.5
            a_new = a + alpha * delta
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            obj = 2.0 * n * fast_convolve(a_new).max()
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new
            else:
                break
    
    return best_a, best_obj
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using annealed smoothed gradient
descent with log-sum-exp potential, active-set Newton equalization, FFT-based
gradient computation, mirror descent with adaptive step, and multi-scale
restart strategy for escaping local minima.

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_pressure(a, w):
    """Compute P[m] = 2 * sum_k w[k] * a[k-m] for m=0,...,n-1 via FFT."""
    n = len(a)
    N = 1 << (3 * n).bit_length()
    A = np.fft.rfft(a, N)
    w_rev = w[::-1].copy()
    W_rev = np.fft.rfft(w_rev, N)
    conv = np.fft.irfft(A * W_rev, N)
    return 2.0 * conv[n-1:2*n-1][::-1]

def smoothed_gd(a0, time_limit, t_start=2.0, t_end=500.0,
                t_growth=1.04, lr_start=0.15, lr_decay=0.99995,
                verbose=False):
    a = a0.copy()
    n = len(a)
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    t = t_start
    lr = lr_start
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        max_b = b.max()
        if t > 100:
            k_top = min(100, len(b))
            top_idx = np.argpartition(b, -k_top)[-k_top:]
            w = np.zeros(len(b))
            vals = b[top_idx] - max_b
            exp_vals = np.exp(t * vals)
            w[top_idx] = exp_vals
            ws = w.sum()
            if ws < 1e-30:
                w = np.zeros(len(b))
                w[int(np.argmax(b))] = 1.0
                ws = 1.0
            w /= ws
        else:
            shifted = b - max_b
            exp_vals = np.exp(t * shifted)
            w = exp_vals / exp_vals.sum()
        grad = compute_pressure(a, w)
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        obj = 2.0 * n * fast_convolve(a_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        a = a_new
        if no_improve < 3:
            t = min(t * t_growth, t_end)
        elif no_improve < 15:

        else:

def active_set_newton(a0, time_limit, K=30, verbose=False):
    a = a0.copy()

    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        max_b = b.max()
        k_act = np.argpartition(b, -K)[-K:]
        k_act = k_act[np.argsort(b[k_act])[::-1]]
        b_act = b[k_act]
        spread = b_act.max() - b_act.min()
        if spread < 1e-12:
            break
        J = np.zeros((len(k_act), n))
        for i, k in enumerate(k_act):
            m_start = max(0, k - n + 1)

        rhs = b_act - b_act.min()
        A_mat = np.vstack([J, np.ones((1, n))])
        rhs_full = np.concatenate([rhs, [0.0]])

        else:
            alpha_max = 1.0
        alpha = min(1.0, alpha_max * 0.95)

def mirror_descent(a0, time_limit, lr_init=0.03, decay=0.9999,
                   temp_init=0.01, temp_growth=1.015, K=40, verbose=False):
    a = np.maximum(a0.copy(), 1e-15)
    a = a / a.sum()
    n = len(a)
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        max_b = b.max()
        k_act = min(K, len(b))
        peaks = np.argpartition(b, -k_act)[-k_act:]
        pv = b[peaks]
        shifted = pv - max_b
        w = np.exp(shifted / temp)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(len(peaks))
            w[np.argmax(pv - pv.min())] = 1.0
            ws = 1.0
        else:
            w /= ws
        P = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            kk = int(peaks[i])
            m_start = max(0, kk - n + 1)
            m_end = min(n - 1, kk)
            idx = np.arange(m_start, m_end + 1)
            P[idx] += 2.0 * w[i] * a[kk - idx]
        a = a * np.exp(-lr * P)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        obj = 2.0 * n * fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        lr *= decay
        temp *= temp_growth
        if no_improve > 200:
            rng = np.random.default_rng(it * 17 + 3)
            a = np.maximum(best_a + rng.standard_normal(n) * 0.004, 1e-15)
            a = a / a.sum()
            lr = lr_init * 0.8
            no_improve = 0
        if lr < 1e-9:
            break
        if verbose and it % 100 == 0:
            print(f"    MD iter {it}: obj={best_obj:.10f}, lr={lr:.2e}")
    return best_a, best_obj

def interp_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, f)
    result = np.maximum(result, 0)
    s = result.sum()
    if s > 1e-15:
        result /= s
    return result

        for oi in range(min(6, len(order))):
            j_max = int(idx[order[oi]])
            old_val = a[j_max]
            if old_val < 1e-12:
                continue
            for mult in [0.6, 0.75, 0.85, 0.92, 1.1, 1.25]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                trial = a.copy()
                trial[j_max] = new_val
                d = old_val - new_val
                if abs(d) > 1e-15:
                    non_peak = np.ones(n, dtype=bool)
                    non_peak[m_start:m_end+1] = False
                    if non_peak.sum() > 0:
                        w = a[non_peak]
                        wsum = w.sum()
                        if wsum > 1e-15:
                            trial[non_peak] += d * w / wsum
                trial = np.maximum(trial, 0)
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                trial = trial / ts
                obj = 2.0 * n * fast_convolve(trial).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = trial.copy()
                    a = trial
                    improved = True
        if not improved:
            no_improve += 1
            k2 = int(np.argmax(fast_convolve(a)))
            ms2 = max(0, k2 - n + 1)
            me2 = min(n - 1, k2)
            trial = a.copy()
            trial[ms2:me2+1] *= 0.95
            d2 = a.sum() - trial.sum()
            nc = np.ones(n, dtype=bool)
            nc[ms2:me2+1] = False
            if nc.sum() > 0:
                nw = a[nc].sum()
                if nw > 1e-15:
                    trial[nc] += d2 * a[nc] / nw
            trial = np.maximum(trial, 0)
            trial = trial / trial.sum()
            obj = 2.0 * n * fast_convolve(trial).max()
            if obj < best_obj:
                a = trial
                best_obj = obj
                best_a = trial.copy()
                no_improve = 0
            else:
                break
        else:
            no_improve = 0
        if verbose and no_improve == 0 and (no_improve == 0):
            pass
    return best_a, best_obj

def try_candidate(a, best_seq, best_score, label=""):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_seq = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768, 0.25)
        best_seq, best_score = try_candidate(parent, best_seq, best_score, "default")

    # Phase 1: Diverse initializations
    print("=== Phase 1: Initializations ===")
    n_parent = len(parent)
    init_list = [parent.copy()]
    for n_try in [640, 700, 768, 832, 896, 1024]:
        if n_try == n_parent:
            init_list.append(parent.copy())
        else:
            init_list.append(interp_sequence(parent, n_try))
        init_list.append(make_gaussian(n_try, 0.2))
        init_list.append(make_gaussian(n_try, 0.3))
        init_list.append(make_gaussian(n_try, 0.4))
        init_list.append(make_cosine(n_try))
        init_list.append(make_tukey(n_try, 0.4))
        init_list.append(make_tukey(n_try, 0.7))
    for ic in init_list:
        best_seq, best_score = try_candidate(ic, best_seq, best_score, f"init_n{len(ic)}")

    base = best_seq.copy()
    n_base = len(base)

    # Phase 2: Smoothed gradient descent
    print(f"\n=== Phase 2: Smoothed GD (n={n_base}) ===")
    rem = deadline - time.monotonic() - 800
    if rem > 60:
        t_alloc = min(rem, 350)
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = smoothed_gd(base, time_limit=t_alloc, verbose=True)
        best_seq, best_score = try_candidate(a_opt, best_seq, best_score, "sgd_main")

    # Phase 3: Active set Newton
    if best_seq is not None:
        print(f"\n=== Phase 3: Active Set Newton (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 500
        if rem > 30:
            t_alloc = min(rem, 150)
            print(f"  Allocating {t_alloc:.0f}s")
            a_opt, obj = active_set_newton(best_seq, time_limit=t_alloc, K=40, verbose=True)
            best_seq, best_score = try_candidate(a_opt, best_seq, best_score, "asn_main")

    # Phase 4: Mirror descent
    if best_seq is not None:
        print(f"\n=== Phase 4: Mirror Descent (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 350
        if rem > 30:
            t_alloc = min(rem, 120)
            print(f"  Allocating {t_alloc:.0f}s")
            a_opt, obj = mirror_descent(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = try_candidate(a_opt, best_seq, best_score, "md_main")

    # Phase 5: Multi-length exploration
    base = best_seq if best_seq is not None else np.ones(768) / 768
    print(f"\n=== Phase 5: Multi-length ===")
    for target_n in [700, 800, 896, 1024]:
        if len(base) == target_n:
            continue
        rem = deadline - time.monotonic() - 200
        if rem < 50:
            break
        f_int = interp_sequence(base, target_n)
        best_seq, best_score = try_candidate(f_int, best_seq, best_score, f"interp_n{target_n}")
        t_alloc = min(rem * 0.3, 80)
        if t_alloc < 20:
            continue
        print(f"  SGD n={target_n}: {t_alloc:.0f}s")
        a_opt, obj = smoothed_gd(f_int, time_limit=t_alloc * 0.6, verbose=False)
        best_seq, best_score = try_candidate(a_opt, best_seq, best_score, f"sgd_n{target_n}")
        t_alloc2 = min(rem * 0.2, 40)
        if t_alloc2 > 10:
            a_opt2, obj2 = active_set_newton(a_opt, time_limit=t_alloc2, K=30)
            best_seq, best_score = try_candidate(a_opt2, best_seq, best_score, f"asn_n{target_n}")

    # Phase 6: Deep optimization on best
    if best_seq is not None:
        print(f"\n=== Phase 6: Deep Opt (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 150
        if rem > 40:
            t1 = min(rem * 0.35, 100)
            t2 = min(rem * 0.35, 100)
            t3 = min(rem * 0.3, 80)
            print(f"  SGD: {t1:.0f}s, ASN: {t2:.0f}s, MD: {t3:.0f}s")
            a_opt, obj = smoothed_gd(best_seq, time_limit=t1, t_start=5.0, t_end=300.0, verbose=True)
            best_seq, best_score = try_candidate(a_opt, best_seq, best_score, "deep_sgd")
            if best_seq is not None:
                a_opt2, _ = active_set_newton(best_seq, time_limit=t2, K=50, verbose=True)
                best_seq, best_score = try_candidate(a_opt2, best_seq, best_score, "deep_asn")
            if best_seq is not None:
                a_opt3, _ = mirror_descent(best_seq, time_limit=t3, lr_init=0.02, verbose=False)
                best_seq, best_score = try_candidate(a_opt3, best_seq, best_score, "deep_md")

    # Phase 7: Perturbed restarts
    if best_seq is not None:
        print(f"\n=== Phase 7: Perturbed Restarts ===")
        n_best = len(best_seq)
        for p in range(4):
            rem = deadline - time.monotonic() - 60
            if rem < 30:
                break
            t_alloc = min(rem / (5 - p), 60)
            if t_alloc < 15:
                break
            if p == 0:
                noise = rng.standard_normal(n_best) * 0.002
                f_init = project_simplex(best_seq + noise)
            elif p == 1:
                f_init = 0.93 * best_seq + 0.07 * (np.ones(n_best) / n_best)
                f_init = f_init / f_init.sum()
            elif p == 2:
                f_init = 0.5 * best_seq + 0.5 * best_seq[::-1]
                f_init = f_init / f_init.sum()
            else:
                f_init = interp_sequence(best_seq, max(64, n_best - 32))
                f_init = interp_sequence(f_init, n_best)
            print(f"  Perturb {p}: {t_alloc:.0f}s")
            a_opt, _ = smoothed_gd(f_init, time_limit=t_alloc * 0.5, verbose=False)
            best_seq, best_score = try_candidate(a_opt, best_seq, best_score, f"pert_{p}_sgd")
            if best_seq is not None:
                a_opt2, _ = active_set_newton(best_seq, time_limit=t_alloc * 0.4, K=30)
                best_seq, best_score = try_candidate(a_opt2, best_seq, best_score, f"pert_{p}_asn")

    # Phase 8: Final coordinate descent polish
    if best_seq is not None:
        print(f"\n=== Phase 8: Final Polish ===")
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            t1 = min(rem * 0.5, 40)
            t2 = min(rem * 0.3, 25)
            t3 = min(rem * 0.2, 15)
            print(f"  CD: {t1:.0f}s, ASN: {t2:.0f}s, MD: {t3:.0f}s")
            a_opt, _ = coord_descent_polish(best_seq, time_limit=t1)
            best_seq, best_score = try_candidate(a_opt, best_seq, best_score, "pol_cd")
            if best_seq is not None:
                a_opt2, _ = active_set_newton(best_seq, time_limit=t2, K=60)
                best_seq, best_score = try_candidate(a_opt2, best_seq, best_score, "pol_asn")
            if best_seq is not None:
                a_opt3, _ = mirror_descent(best_seq, time_limit=t3, lr_init=0.01)
                best_seq, best_score = try_candidate(a_opt3, best_seq, best_score, "pol_md")

    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = min(best_score, final_score)
    print(f"\nFINAL: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
