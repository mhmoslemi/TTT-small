def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    s = a.sum()
    if s > 1e-15:

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def spectral_flatten(f0, time_limit, verbose=False, n_iter=500):
    """
    Spectral domain optimization: directly modify the power spectrum to be flatter.
    
    Key idea: conv(a,a) is the inverse DFT of |A|^2 (with appropriate normalization).
    We want |A|^2 to be as flat as possible.
    
    Algorithm:
    1. Start with sequence a
    2. Compute A = FFT(a)
    3. Modify |A| to be closer to uniform (while maintaining the constraint that IFFT is nonnegative)
    4. Take IFFT to get new a
    5. Project onto nonnegative
    6. Repeat
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        # Compute spectrum
        A = np.fft.rfft(f, fft_size)
        power = np.abs(A) ** 2
        
        # Target: uniform power spectrum
        # But we need the constraint that IFFT(g * A/|A|) is nonnegative
        # Simple approach: interpolate between current power and uniform
        uniform_power = power.sum() / len(power)
        
        # Blend: reduce peaks in power spectrum
        mix = 0.1  # how much to flatten
        new_power = (1 - mix) * power + mix * uniform_power
        
        # Reconstruct with same phase
        phase = np.angle(A)
        A_new = np.sqrt(new_power) * np.exp(1j * phase)
        
        # Make Hermitian
        A_full = np.zeros(fft_size, dtype=complex)
        n_rfft = len(A)
        A_full[:n_rfft] = A_new
        for i in range(1, fft_size - n_rfft + 1):
            A_full[fft_size - i] = np.conj(A_new[i - 1])
        
        # Inverse FFT
        a_new = np.fft.ifft(A_full).real[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        co = fast_convolve(a_new).max()
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = a_new.copy()
            f = a_new
        else:
            # If it got worse, try smaller mix
            for mix2 in [0.05, 0.02, 0.01]:
                new_power2 = (1 - mix2) * power + mix2 * uniform_power
                A_new2 = np.sqrt(new_power2) * np.exp(1j * phase)
                A_full2 = np.zeros(fft_size, dtype=complex)
                A_full2[:n_rfft] = A_new2
                for i in range(1, fft_size - n_rfft + 1):
                    A_full2[fft_size - i] = np.conj(A_new2[i - 1])
                a_try = np.fft.ifft(A_full2).real[:n]
                a_try = np.maximum(a_try, 0)
                s2 = a_try.sum()
                if s2 < 1e-15:
                    continue
                a_try = a_try / s2
                co2 = fast_convolve(a_try).max()
                if co2 < best_obj - 1e-14:
                    best_obj = co2
                    best_f = a_try.copy()
                    f = a_try
                    break
    
    return best_f, best_obj

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-strategy approach:
- Riesz product initialization (products of cosines for flat autocorrelation)
- Spectral domain flattening (direct power spectrum manipulation)
- Multiplicative weights with adaptive step size
- Peak-targeted local search
- LP-Frank-Wolfe for convex subproblems

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_normalized(a):
    """Compute 2*n*max(conv) for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()


def make_riesz_product(n, n_factors=6, rho_max=0.15, seed=42):
    """
    Riesz product: product of (1 + rho_m * cos(2*pi*freq_m*x + phase_m)).
    These have rapidly decaying autocorrelation.
    """
    rng = np.random.default_rng(seed)
    x = np.arange(n)
    a = np.ones(n, dtype=np.float64)
    
    # Use well-separated frequencies
    freqs = []
    base_freqs = [1, 2, 5, 11, 23, 47, 97, 193]
    shuffled = list(base_freqs)
    rng.shuffle(shuffled)
    
    for m in range(n_factors):
        freq = shuffled[m % len(shuffled)]
        rho = rng.uniform(0.03, rho_max)
        phase = rng.uniform(0, 2 * np.pi)
        factor = 1.0 + rho * np.cos(2.0 * np.pi * freq * x / n + phase)
        a = a * factor
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def make_gaussian(n, sigma_frac=0.3):

    x = np.arange(n) - n / 2.0

    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:

def make_bell(n, width_frac=0.6):
    """Bell-shaped initial sequence with flat top."""
    x = np.arange(n) - n / 2.0

    s = a.sum()

def make_cosine_mod(n, freq, amp):
    """Cosine-modulated uniform sequence."""
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)

    return a


def make_multi_cosine(n, n_freqs=3, max_amp=0.3, seed=0):
    """Multi-frequency cosine-modulated sequence."""
    rng = np.random.default_rng(seed)
    a = np.ones(n, dtype=np.float64)
    for m in range(1, n_freqs + 1):
        amp = rng.uniform(0.05, max_amp)
        phase = rng.uniform(0, 2 * np.pi)
        a = a + amp * np.cos(2.0 * np.pi * m * np.arange(n) / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def make_flat_spectrum_init(n, seed=42):
    """Initialize by creating a flat power spectrum with random phase."""
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    rng = np.random.default_rng(seed)
    
    n_freqs = fft_size // 2 + 1
    magnitude = np.ones(n_freqs) / np.sqrt(fft_size)
    phase = rng.uniform(0, 2 * np.pi, n_freqs)
    A = magnitude * np.exp(1j * phase)
    
    A_full = np.zeros(fft_size, dtype=complex)
    A_full[:n_freqs] = A
    for i in range(1, fft_size - n_freqs + 1):
        A_full[fft_size - i] = np.conj(A[i - 1])
    
    a = np.fft.ifft(A_full).real[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def spectral_flatten(f0, time_limit, verbose=False, n_iter=1000):
    """
    Spectral domain optimization: modify power spectrum to be flatter.
    conv(a,a) ~ IFFT(|FFT(a)|^2), so flattening |FFT(a)|^2 reduces max autocorrelation.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    
    mix = 0.08
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit - 2:
            break
        
        A = np.fft.rfft(f, fft_size)
        power = np.abs(A) ** 2
        n_rfft = len(A)
        uniform_power = power.sum() / n_rfft
        
        # Adaptive mix: increase flattening if progress
        new_power = (1.0 - mix) * power + mix * uniform_power
        
        phase = np.angle(A)
        A_new = np.sqrt(new_power) * np.exp(1j * phase)
        
        A_full = np.zeros(fft_size, dtype=complex)
        A_full[:n_rfft] = A_new
        for i in range(1, fft_size - n_rfft + 1):
            A_full[fft_size - i] = np.conj(A_new[i - 1])
        
        a_new = np.fft.ifft(A_full).real[:n]
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn
        
        co = fast_convolve(a_new).max()
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = a_new.copy()
            f = a_new
            if verbose and it % 100 == 0:
                print(f"        Spectral iter {it}: obj={best_obj:.10f}")
        else:
            # Try smaller mix
            found = False
            for mix2 in [mix * 0.5, mix * 0.25, mix * 0.1]:
                new_power2 = (1.0 - mix2) * power + mix2 * uniform_power
                A_new2 = np.sqrt(new_power2) * np.exp(1j * phase)
                A_full2 = np.zeros(fft_size, dtype=complex)
                A_full2[:n_rfft] = A_new2
                for i in range(1, fft_size - n_rfft + 1):
                    A_full2[fft_size - i] = np.conj(A_new2[i - 1])
                a_try = np.fft.ifft(A_full2).real[:n]
                a_try = np.maximum(a_try, 0)
                s2 = a_try.sum()
                if s2 < 1e-15:
                    continue
                a_try = a_try / s2
                co2 = fast_convolve(a_try).max()
                if co2 < best_obj - 1e-14:
                    best_obj = co2
                    best_f = a_try.copy()
                    f = a_try
                    mix = mix2
                    found = True
                    break
            if not found:
                mix = mix * 0.8
                if mix < 0.001:
                    break
    
    return best_f, best_obj


def multiplicative_descent(f0, time_limit, verbose=False, lr_init=0.01, peak_count=40):
    """
    Multiplicative weights (mirror descent) with smoothed subgradient
    and adaptive learning rate.
    """
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0
    temperature = 0.0008
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 3:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        
        # Smoothed subgradient using softmax over top peaks
        k_actual = min(peak_count, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        
        scaled = pv / max(temperature, 1e-12)
        scaled = scaled - scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += w[i] * f[k - idx]
        
        # Multiplicative update
        f_new = f * np.exp(-lr * sg)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()
        
        # Backtracking line search
        best_alpha = 1.0
        best_val = M
        for alpha in [1.0, 0.6, 0.3, 0.15]:
            ft = f + alpha * (f_new - f)
            ft = np.maximum(ft, 1e-15)
            ft = ft / ft.sum()
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        f = f + best_alpha * (f_new - f)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()
        
        co = fast_convolve(f).max()
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"        MD iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
        
        # Adaptive lr
        if no_improve % 50 == 0 and no_improve > 0:
            lr *= 0.92
        if no_improve > 150:
            rng = np.random.default_rng(it * 31 + 7)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-15)
            f = f / f.sum()
            lr = lr_init * 0.4
            no_improve = 0
        
        temperature *= 1.004
        if lr < 1e-10:
            break
    
    return best_f, best_obj


def peak_local_search(f0, time_limit, verbose=False):
    """
    Local search targeting the maximum peak specifically.
    Tries mass redistribution from peak contributors to non-contributors.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 3:
            break
        if no_improve > 200:
            break
        it += 1
        
        conv = fast_convolve(f)
        M = conv.max()
        peak_k = int(np.argmax(conv))
        
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contrib_idx = np.arange(js, je + 1)
        
        if len(contrib_idx) == 0:
            break
        
        products = f[contrib_idx] * f[peak_k - contrib_idx]
        max_prod_i = contrib_idx[np.argmax(products)]
        
        non_contrib_mask = np.ones(n, dtype=bool)
        non_contrib_mask[contrib_idx] = False
        nc_indices = np.where(non_contrib_mask)[0]
        
        best_trial = None
        best_trial_obj = M
        
        # Strategy 1: Move mass from top contributor to non-contributors
        for frac in [0.03, 0.05, 0.08, 0.12, 0.18, 0.25]:
            diff = f[max_prod_i] * frac
            if diff < 1e-15:
                continue
            trial = f.copy()
            trial[max_prod_i] -= diff
            if len(nc_indices) > 0:
                nc_vals = trial[nc_indices]
                nc_sum = nc_vals.sum()
                if nc_sum > 1e-15:
                    trial[nc_indices] += diff * nc_vals / nc_sum
                else:
                    trial[nc_indices] += diff / len(nc_indices)
            trial = np.maximum(trial, 0)
            trial = trial / trial.sum()
            obj = fast_convolve(trial).max()
            if obj < best_trial_obj - 1e-15:
                best_trial_obj = obj
                best_trial = trial
        
        # Strategy 2: Move mass from second contributor
        if len(contrib_idx) >= 2:
            sorted_prods = np.argsort(products)[::-1]
            second_i = contrib_idx[sorted_prods[1]]
            for frac in [0.05, 0.1, 0.15]:
                diff = f[second_i] * frac
                if diff < 1e-15:
                    continue
                trial = f.copy()
                trial[second_i] -= diff
                if len(nc_indices) > 0:
                    nc_vals = trial[nc_indices]
                    nc_sum = nc_vals.sum()
                    if nc_sum > 1e-15:
                        trial[nc_indices] += diff * nc_vals / nc_sum
                    else:
                        trial[nc_indices] += diff / len(nc_indices)
                trial = np.maximum(trial, 0)
                trial = trial / trial.sum()
                obj = fast_convolve(trial).max()
                if obj < best_trial_obj - 1e-15:
                    best_trial_obj = obj
                    best_trial = trial
        
        # Strategy 3: Simultaneous reduction of top 2-3 contributors
        if len(contrib_idx) >= 2:
            top_k = min(3, len(contrib_idx))
            top_indices = contrib_idx[np.argsort(products)[::-1][:top_k]]
            for total_frac in [0.06, 0.1, 0.15]:
                diff_each = f[top_indices[0]] * total_frac / top_k
                trial = f.copy()
                for ti in top_indices:
                    trial[ti] -= diff_each
                if len(nc_indices) > 0:
                    total_moved = diff_each * top_k
                    nc_vals = trial[nc_indices]
                    nc_sum = nc_vals.sum()
                    if nc_sum > 1e-15:
                        trial[nc_indices] += total_moved * nc_vals / nc_sum
                    else:
                        trial[nc_indices] += total_moved / len(nc_indices)
                trial = np.maximum(trial, 0)
                trial = trial / trial.sum()
                obj = fast_convolve(trial).max()
                if obj < best_trial_obj - 1e-15:
                    best_trial_obj = obj
                    best_trial = trial
        
        if best_trial is not None:
            f = best_trial
            best_obj = best_trial_obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"        PLS iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                # Try a different peak
                top_peaks = np.argsort(conv)[-3:]
                for pk in top_peaks:
                    pk = int(pk)
                    if pk == peak_k:
                        continue
                    js2 = max(0, pk - n + 1)
                    je2 = min(n - 1, pk)
                    cidx2 = np.arange(js2, je2 + 1)
                    if len(cidx2) == 0:
                        continue
                    prods2 = f[cidx2] * f[pk - cidx2]
                    src2 = cidx2[np.argmax(prods2)]
                    nc2 = np.ones(n, dtype=bool)
                    nc2[cidx2] = False
                    nc2_idx = np.where(nc2)[0]
                    for frac2 in [0.05, 0.1, 0.15]:
                        diff2 = f[src2] * frac2
                        trial = f.copy()
                        trial[src2] -= diff2
                        if len(nc2_idx) > 0:
                            trial[nc2_idx] += diff2 * trial[nc2_idx] / max(trial[nc2_idx].sum(), 1e-15)
                        trial = np.maximum(trial, 0)
                        trial = trial / trial.sum()
                        obj2 = fast_convolve(trial).max()
                        if obj2 < best_obj - 1e-14:
                            f = trial
                            best_obj = obj2
                            best_f = f.copy()
                            no_improve = 0
                            break
                    if no_improve == 0:
                        break
        
        if no_improve > 100 and no_improve % 50 == 0:
            rng = np.random.default_rng(it * 13 + 7)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.002, 0)
            f = f / f.sum()
            no_improve = 0
    
    return best_f, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-Frank-Wolfe with multi-level M relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_levels = [1.0, 1.01, 1.03, 1.06, 1.1, 1.2, 1.5, 2.0]
    m_idx = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3.0:
            break
        
        g = None
        for level_try in range(m_idx, len(m_levels)):
            M_try = conv.max() * m_levels[level_try]
            g = solve_lp_direction(f, M_try, n)
            if g is not None and g.sum() > 1e-12:
                m_idx = level_try
                break
            m_idx = level_try + 1
        
        if g is None or g.sum() < 1e-12:
            rng = np.random.default_rng(it * 13 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            m_idx = 0
            no_improve += 1
            if no_improve > 50:
                break
            continue
        
        g = g / g.sum()
        diff = np.max(np.abs(g - f))
        if diff < 1e-10:
            no_improve += 1
            m_idx = min(m_idx + 1, len(m_levels) - 1)
            if no_improve > 40:
                break
            continue
        no_improve = 0
        
        best_alpha = 0.0
        best_val = conv.max()
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha < 1e-10:
            no_improve += 1
            m_idx = min(m_idx + 1, len(m_levels) - 1)
            continue
        
        f = np.maximum(f + best_alpha * (g - f), 0.0)
        f = f / f.sum()
        conv = fast_convolve(f)
        m_idx = 0
        
        if conv.max() < best_obj - 1e-14:
            best_obj = conv.max()
            best_f = f.copy()
            if verbose and it % 20 == 0:
                print(f"        LP-FW iter {it}: obj={best_obj:.10f}")
    
    return best_f, best_obj


def subgradient_descent(f0, time_limit, verbose=False, lr_init=0.004):
    """Subgradient descent with Euclidean projection onto simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0
    smooth_param = 0.008
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 3:
            break
        it += 1
        
        conv = fast_convolve(f)
        M = conv.max()
        
        k_actual = min(30, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / max(smooth_param, 1e-12)
        scaled = scaled - scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += w[i] * f[k - idx]
        
        f_try = project_simplex(f - lr * sg)
        
        best_val = M
        best_alpha = 1.0
        for alpha in [1.0, 0.5, 0.25, 0.1]:
            ft = f + alpha * (f_try - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        f = f + best_alpha * (f_try - f)
        f = np.maximum(f, 0.0)
        f = f / f.sum()
        
        co = fast_convolve(f).max()
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"        SGD iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
        
        if no_improve % 40 == 0:
            lr *= 0.9
        if no_improve > 120:
            rng = np.random.default_rng(it * 23 + 11)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            lr = lr_init * 0.5
            no_improve = 0
        
        smooth_param *= 0.998
        if lr < 1e-10:
            break
    
    return best_f, best_obj


def evaluate_and_track(a, best_state):
    """Evaluate sequence and update best state dict."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_state['score']:
        best_state['score'] = score
        best_state['seq'] = a.copy()
    return score

        init_list.append((f"riesz_{n_try}", make_riesz_product(n_try, 5, 0.12, seed=n_try)))
        init_list.append((f"riesz2_{n_try}", make_riesz_product(n_try, 7, 0.1, seed=n_try + 500)))
        init_list.append((f"sigma_{n_try}", make_flat_spectrum_init(n_try, seed=n_try)))
    
    # Quick evaluation of all inits
    scored_inits = []
    for label, a_init in init_list:
        sc = objective_normalized(a_init)
        scored_inits.append((sc, label, a_init))
        # Also check actual score for promising ones
        if sc < 1.6:
            actual = track(a_init, f"init_{label}")
    
    # Sort by normalized objective
    scored_inits.sort(key=0)  # This won't work with lambda, use itemgetter
    scored_inits.sort(key=np.vectorize(np.float64)([x[0] for x in scored_inits]))
    
    # Manual sort since we can't use lambda
    scored_inits_sorted = []
    for item in scored_inits:
        inserted = False
        for i in range(len(scored_inits_sorted)):
            if item[0] < scored_inits_sorted[i][0]:
                scored_inits_sorted.insert(i, item)
                inserted = True
                break
        if not inserted:
            scored_inits_sorted.append(item)
    scored_inits = scored_inits_sorted
    
    print(f"  {len(init_list)} candidates generated in {time.monotonic()-t0:.1f}s")
    print(f"  Best init obj: {scored_inits[0][0]:.6f} ({scored_inits[0][1]})")
    
    # Phase 2: Spectral flattening on top candidates
    print("\n--- Phase 2: Spectral Flattening ---")
    rem = deadline - time.monotonic() - 350
    if rem > 60:
        for trial_idx in range(min(4, len(scored_inits))):
            rem = deadline - time.monotonic() - 300
            if rem < 30:
                break
            _, label, a_init = scored_inits[trial_idx]
            t_alloc = min(rem * 0.5, 70)
            if t_alloc < 15:
                break
            print(f"  Spectral [{label}]: n={len(a_init)}, alloc={t_alloc:.0f}s")
            a_opt, obj = spectral_flatten(a_init, time_limit=t_alloc, verbose=True)
            track(a_opt, f"spectral_{label}")
    
    # Phase 3: Multiplicative descent
    print("\n--- Phase 3: Multiplicative Descent ---")
    if best_state['seq'] is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 30:
            t_alloc = min(rem * 0.5, 100)
            print(f"  MD: n={len(best_state['seq'])}, alloc={t_alloc:.0f}s")
            a_opt, obj = multiplicative_descent(best_state['seq'], time_limit=t_alloc, verbose=True)
            track(a_opt, "md")
    
    # Phase 4: Peak local search
    print("\n--- Phase 4: Peak Local Search ---")
    if best_state['seq'] is not None:
        rem = deadline - time.monotonic() - 180
        if rem > 30:
            t_alloc = min(rem * 0.5, 80)
            print(f"  PLS: n={len(best_state['seq'])}, alloc={t_alloc:.0f}s")
            a_opt, obj = peak_local_search(best_state['seq'], time_limit=t_alloc, verbose=True)
            track(a_opt, "pls")
    
    # Phase 5: LP-Frank-Wolfe
    print("\n--- Phase 5: LP-Frank-Wolfe ---")
    if best_state['seq'] is not None:
        rem = deadline - time.monotonic() - 120
        if rem > 30:
            t_alloc = min(rem * 0.5, 60)
            print(f"  LP-FW: n={len(best_state['seq'])}, alloc={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_state['seq'], time_limit=t_alloc, verbose=True)
            track(a_opt, "lpfw")
    
    # Phase 6: Subgradient descent
    print("\n--- Phase 6: Subgradient Descent ---")
    if best_state['seq'] is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_alloc = min(rem * 0.5, 40)
            print(f"  SGD: n={len(best_state['seq'])}, alloc={t_alloc:.0f}s")
            a_opt, obj = subgradient_descent(best_state['seq'], time_limit=t_alloc, verbose=True)
            track(a_opt, "sgd")
    
    # Phase 7: Multi-length exploration
    print("\n--- Phase 7: Multi-Length ---")
    if best_state['seq'] is not None:
        base_n = len(best_state['seq'])
        target_lengths = [512, 640, 768, 896, 1024, 1152, 1280]
        target_lengths = [t for t in target_lengths if t != base_n][:4]
        
        for target_n in target_lengths:
            rem = deadline - time.monotonic() - 60
            if rem < 20:
                break
            f_interp = interpolate_sequence(best_state['seq'], target_n)
            sc = track(f_interp, f"interp_n{target_n}")
            
            if sc < best_state['score'] + 0.03:
                t_alloc = min(rem * 0.3, 30)
                if t_alloc > 10:
                    a_opt, _ = multiplicative_descent(f_interp, time_limit=t_alloc, lr_init=0.008)
                    track(a_opt, f"md_interp_n{target_n}")
                    if deadline - time.monotonic() > 15:
                        a_opt2, _ = spectral_flatten(f_interp, time_limit=15, n_iter=200)
                        track(a_opt2, f"spect_interp_n{target_n}")
    
    # Phase 8: Final alternating polish
    print("\n--- Phase 8: Final Polish ---")
    if best_state['seq'] is not None:
        polish_methods = [
            ("md_fine", "md"),
            ("pls_fine", "pls"),
            ("spect_fine", "spect"),
            ("lpfw_fine", "lpfw"),
            ("sgd_fine", "sgd"),
        ]
        
        for pname, mtype in polish_methods:
            rem = deadline - time.monotonic() - 5
            if rem < 8:
                break
            t_alloc = min(rem * 0.6, 25)
            if t_alloc < 5:
                break
            print(f"  Polish [{pname}]: alloc={t_alloc:.0f}s")
            if mtype == "md":
                a_opt, _ = multiplicative_descent(best_state['seq'], time_limit=t_alloc, lr_init=0.003, peak_count=30)
            elif mtype == "pls":
                a_opt, _ = peak_local_search(best_state['seq'], time_limit=t_alloc)
            elif mtype == "spect":
                a_opt, _ = spectral_flatten(best_state['seq'], time_limit=t_alloc, n_iter=300)
            elif mtype == "lpfw":
                a_opt, _ = lp_fw_optimize(best_state['seq'], time_limit=t_alloc)
            elif mtype == "sgd":
                a_opt, _ = subgradient_descent(best_state['seq'], time_limit=t_alloc, lr_init=0.002)
            else:
                continue
            track(a_opt, pname)
    
    # Phase 9: One final round on the absolute best
    print("\n--- Phase 9: Final Squeeze ---")
    if best_state['seq'] is not None:
        rem = deadline - time.monotonic() - 2
        if rem > 10:
            t_alloc = min(rem, 12)
            print(f"  Final MD squeeze: alloc={t_alloc:.0f}s")
            a_opt, _ = multiplicative_descent(best_state['seq'], time_limit=t_alloc, lr_init=0.002, peak_count=20)
            track(a_opt, "final_squeeze")
    
    # Fallback
    if best_state['seq'] is None:
        best_state['seq'] = np.ones(768) / 768
        best_state['score'] = float(evaluate_sequence(best_state['seq'].tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_state['seq'].tolist()))
    if np.isfinite(fs):
        best_state['score'] = fs
    
    print(f"\n===== FINAL: score={best_state['score']:.6f} (n={len(best_state['seq'])}) =====")
    return [float(x) for x in best_state['seq']]
