import numpy as np
from scipy.optimize import linprog, minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, non-negative)."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def obj_norm(a):
    """Objective for normalized sequence (sum=1). Returns 2n*max(conv)."""
    return 2.0 * len(a) * fast_convolve(a).max()

def gs_init(n, n_iters=50, seed=42):
    """Gerchberg-Saxton algorithm for flat-spectrum initialization."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    a = a / a.sum()

    # Actually, for a normalized sequence with sum=1, FFT(a)[0] = 1
    # The average |FFT|^2 = sum|a|^2 / n by Parseval (for DFT)
    # Let's just aim for flat magnitude
    
    for it in range(n_iters):
        F = np.fft.fft(a)
        # Set magnitude to target, keep phase
        phase = F / np.abs(F + 1e-30)
        F_new = target_mag * phase
        # But we need F_new[0] to be real and positive (since sum(a) > 0)
        F_new[0] = np.abs(F_new[0])
        a = np.fft.ifft(F_new).real
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s
    return a

def lp_fw_step(f, M):
    """Single LP-FW step: find g that maximizes sum(g) s.t. (f*g) <= M, g>=0."""
    n = len(f)
    c = -np.ones(n)

    # (f*g)[k] = sum_j f[j]*g[k-j] (with valid indices)

    # So row k of A_ub is: [f[k], f[k-1], ..., f[k-n+1]] (with padding)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):

        # with coefficient f[k-j]
        k_start = j
        k_end = j + n - 1
        A_ub[k_start:k_end+1, j] = f[n-1-j::-1]  # f[n-1-j], f[n-2-j], ..., f[0]

    # So for a fixed j, (f*g)[k] includes f[k-j]*g[j] when 0 <= k-j < n, i.e., j <= k <= j+n-1
    # The coefficient of g[j] in row k is f[k-j]
    # So A_ub[k, j] = f[k-j] when 0 <= k-j < n, else 0
    
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):

    b_ub = np.full(2 * n - 1, M)

    if res.success:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid multi-phase approach:
1. Gerchberg-Saxton spectral shaping for diverse initializations
2. LP-Frank-Wolfe with warm-starting for coarse optimization
3. Nesterov-accelerated subgradient with adaptive step size
4. Direct coordinate search and peak equalization for final polishing
5. Parametric Fourier optimization with scipy minimize
6. Multi-length exploration with adaptive time allocation

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, non-negative)."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_norm(a):
    """Objective for normalized sequence (sum=1). Returns 2n*max(conv)."""
    return 2.0 * len(a) * fast_convolve(a).max()


def gs_init(n, n_iters=80, seed=42):
    """Gerchberg-Saxton algorithm for flat-spectrum non-negative initialization."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    a = a / a.sum()
    for it in range(n_iters):
        F = np.fft.fft(a)
        mag_target = np.sqrt(np.sum(np.abs(F) ** 2) / n)
        phase = F / (np.abs(F) + 1e-30)
        F_new = mag_target * phase
        F_new[0] = np.abs(F_new[0])
        a = np.fft.ifft(F_new).real
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-like initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    """Bell-shaped initial sequence with flat top."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine_mod(n, freq, amp):
    """Cosine-modulated uniform sequence."""
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_multi_cosine(n, n_freqs=3, max_amp=0.3, seed=0):
    """Multi-frequency cosine-modulated sequence."""
    rng = np.random.default_rng(seed)
    a = np.ones(n)
    for m in range(1, n_freqs + 1):
        amp = rng.uniform(0.05, max_amp)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * m * np.arange(n) / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def build_lp_matrix(f):
    """Build the LP constraint matrix for LP-FW step."""
    n = len(f)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        A_ub[j:k_end, j] = f[:k_end - j]
    return A_ub


def lp_fw_optimize(f0, time_limit, verbose=False, warm_start=None):
    """LP-Frank-Wolfe with adaptive M levels and warm-starting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    A_ub = build_lp_matrix(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_scalars = [1.0, 1.02, 1.05, 1.1, 1.2, 1.5, 2.0, 3.0]
    m_idx = 0
    prev_g = None

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3.0:
            break

        g = None
        for level_try in range(m_idx, len(m_scalars)):
            M_try = conv.max() * m_scalars[level_try]
            b_ub = np.full(2 * n - 1, M_try)
            bounds = [(0, None)] * n
            try:
                if prev_g is not None:
                    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub,
                                  bounds=bounds, method='highs', options={"presolve": True})
                else:
                    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub,
                                  bounds=bounds, method='highs')
                if res.success:
                    g = res.x
                    m_idx = level_try
                    break
            except Exception:
                continue
            m_idx = level_try + 1

        if g is None or g.sum() < 1e-12:
            m_idx = min(m_idx + 1, len(m_scalars) - 1)
            no_improve += 1
            if no_improve > 40:
                break
            # Perturb and continue
            rng = np.random.default_rng(it * 17 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            conv = fast_convolve(f)
            A_ub = build_lp_matrix(f)
            m_idx = 0
            continue

        g = g / g.sum()
        prev_g = g

        # Check direction
        diff = np.max(np.abs(g - f))
        if diff < 1e-12:
            no_improve += 1
            m_idx = min(m_idx + 1, len(m_scalars) - 1)
            if no_improve > 30:
                break
            continue
        no_improve = 0

        # Line search (coarse + fine)
        best_alpha = 0.0
        best_val = conv.max()
        alphas = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine golden section if coarse found something
        if best_alpha > 0 and best_alpha < 1.0:
            idx = alphas.index(best_alpha)
            lo = alphas[max(0, idx - 1)]
            hi = alphas[min(len(alphas) - 1, idx + 1)]
            for _ in range(8):
                a1 = hi - 0.618 * (hi - lo)
                a2 = lo + 0.618 * (hi - lo)
                ft1 = np.maximum(f + a1 * (g - f), 0.0)
                ft2 = np.maximum(f + a2 * (g - f), 0.0)
                ft1 /= ft1.sum()
                ft2 /= ft2.sum()
                v1 = fast_convolve(ft1).max()
                v2 = fast_convolve(ft2).max()
                if v1 < v2:
                    hi = a2
                else:
                    lo = a1
                mid = (lo + hi) / 2.0
                ftm = np.maximum(f + mid * (g - f), 0.0)
                ftm /= ftm.sum()
                vm = fast_convolve(ftm).max()
                if vm < best_val:
                    best_val = vm
                    best_alpha = mid

        if best_alpha < 1e-10:
            no_improve += 1
            m_idx = min(m_idx + 1, len(m_scalars) - 1)
            continue

        f_new = np.maximum(f + best_alpha * (g - f), 0.0)
        f_new = f_new / f_new.sum()
        conv = fast_convolve(f_new)
        f = f_new
        A_ub = build_lp_matrix(f)
        m_idx = 0

        if conv.max() < best_obj - 1e-14:
            best_obj = conv.max()
            best_f = f.copy()
            if verbose and it % 10 == 0:
                print(f"    LP-FW iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def subgradient_nesterov(a0, time_limit, verbose=False, lr_init=0.02):
    """Nesterov-accelerated subgradient method with adaptive step size."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()

    y = a.copy()  # Nesterov lookahead
    v = np.zeros(n)  # velocity
    lr = lr_init
    it = 0
    no_improve = 0
    best_lr = lr

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute subgradient at y
        conv_y = fast_convolve(y)
        M = conv_y.max()

        # Weight peaks by exponential of (value / temperature)
        temp = max((M - np.median(conv_y)) * 0.5, M * 0.01)
        weights = np.exp((conv_y - M) / temp)
        w_sum = weights.sum()
        if w_sum < 1e-15:
            weights = np.zeros(len(conv_y))
            weights[np.argmax(conv_y)] = 1.0
        else:
            weights /= w_sum

        # Compute weighted subgradient
        sg = np.zeros(n)
        top_k = np.argpartition(weights, -min(30, 2 * n - 1))[-min(30, 2 * n - 1):]
        for k in top_k:
            w = weights[k]
            if w < 1e-10:
                continue
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += w * 2.0 * y[k - idx]

        # Nesterov update
        v = 0.9 * v - lr * sg
        a_new = project_simplex(y + v)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        # Nesterov momentum for y
        beta = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        beta = min(beta, 20.0)  # cap momentum
        y = project_simplex(a_new + (v / lr) * (1.0 / (beta + 1.0) - 1.0) * 0.1)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        new_obj = fast_convolve(a_new).max()

        if new_obj < best_obj - 1e-12:
            best_obj = new_obj
            best_a = a_new.copy()
            no_improve = 0
            best_lr = lr
        else:
            no_improve += 1
            lr *= 0.97

        # Adaptive restart
        if no_improve > 100:
            rng = np.random.default_rng(it * 31 + 7)
            a_new = np.maximum(best_a + rng.standard_normal(n) * 0.005 * best_a.max(), 1e-15)
            a_new = a_new / a_new.sum()
            y = a_new.copy()
            v = np.zeros(n)
            lr = max(best_lr, 0.005)
            no_improve = 0

        if lr < 1e-8:
            lr = lr_init * 0.1
            v = np.zeros(n)

        if verbose and it % 200 == 0:
            print(f"    SG-N iter {it}: obj={best_obj:.10f}, lr={lr:.6f}")

        a = a_new

    return best_a, best_obj


def coordinate_search(a0, time_limit, verbose=False):
    """Direct coordinate search targeting peak reduction."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if no_improve > 200:
            break

        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))

        # Find contributing indices
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contrib = np.arange(js, je + 1)
        if len(contrib) == 0:
            break

        products = a[contrib] * a[peak_k - contrib]
        total = products.sum()
        if total < 1e-15:
            break

        # Get top contributing pairs
        top_pairs = np.argsort(products)[::-1][:10]
        found_improvement = False

        for pi in top_pairs:
            if found_improvement:
                break
            i = contrib[pi]
            j = peak_k - i

            # Try reducing a[i] and spreading to non-contributing coords
            for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                delta = a[i] * frac
                if delta < 1e-15:
                    continue
                trial = a.copy()
                trial[i] -= delta
                # Spread to random non-contributing coordinates
                non_c = np.ones(n, dtype=bool)
                non_c[contrib] = False
                if non_c.sum() < 1:
                    continue
                nc_idx = np.where(non_c)[0]
                # Distribute proportionally to existing values
                nc_vals = trial[nc_idx]
                nc_sum = nc_vals.sum()
                if nc_sum < 1e-15:
                    trial[nc_idx] += delta / len(nc_idx)
                else:
                    trial[nc_idx] += delta * nc_vals / nc_sum
                trial = np.maximum(trial, 0)
                trial = trial / trial.sum()
                obj = fast_convolve(trial).max()
                if obj < best_obj - 1e-13:
                    a = trial
                    best_obj = obj
                    best_a = a.copy()
                    no_improve = 0
                    found_improvement = True
                    break

        if not found_improvement:
            no_improve += 1
            # Try a different peak
            if no_improve % 20 == 0:
                peaks = np.argsort(conv)[::-1][:5]
                for pk in peaks:
                    pk = int(pk)
                    if pk == peak_k:
                        continue
                    js2 = max(0, pk - n + 1)
                    je2 = min(n - 1, pk)
                    c2 = np.arange(js2, je2 + 1)
                    if len(c2) == 0:
                        continue
                    prods2 = a[c2] * a[pk - c2]
                    src = c2[np.argmax(prods2)]
                    for frac in [0.1, 0.2, 0.4]:
                        delta = a[src] * frac
                        trial = a.copy()
                        trial[src] -= delta
                        nc = np.ones(n, dtype=bool)
                        nc[c2] = False
                        nc_idx = np.where(nc)[0]
                        if len(nc_idx) < 1:
                            continue
                        nc_vals = trial[nc_idx]
                        nc_sum = nc_vals.sum()
                        if nc_sum > 1e-15:
                            trial[nc_idx] += delta * nc_vals / nc_sum
                        else:
                            trial[nc_idx] += delta / len(nc_idx)
                        trial = np.maximum(trial, 0)
                        trial = trial / trial.sum()
                        obj2 = fast_convolve(trial).max()
                        if obj2 < best_obj - 1e-13:
                            a = trial
                            best_obj = obj2
                            best_a = a.copy()
                            no_improve = 0
                            break

        if verbose and it % 50 == 0:
            print(f"    CS iter {it}: obj={best_obj:.10f}")

    return best_a, best_obj


def parametric_fourier_optimize(a0, time_limit, verbose=False):
    """Optimize using a Fourier parameterization of the sequence."""
    n = len(a0)
    best_a = a0.copy()
    best_obj = obj_norm(a0)
    t0 = time.monotonic()

    # Parameterize: a[k] = max(0, 1/n + sum_{m=1}^{M} r_m * cos(2*pi*m*k/n + phi_m))
    # Then normalize to sum=1
    # Parameters: [r_1, phi_1, r_2, phi_2, ...]
    M = min(8, n // 4)
    n_params = 2 * M

    def fourier_seq(params):
        a = np.full(n, 1.0 / n)
        for m in range(M):
            r = params[2 * m]
            phi = params[2 * m + 1]
            a += r * np.cos(2.0 * np.pi * m * np.arange(n) / n + phi)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            return np.ones(n) / n
        return a / s

    def fourier_obj(params):
        a = fourier_seq(params)
        return obj_norm(a)

    # Initialize params from current best
    F = np.fft.rfft(a0)
    init_params = np.zeros(n_params)
    for m in range(M):
        if m < len(F):
            init_params[2 * m] = 2.0 * np.abs(F[m]) / n
            init_params[2 * m + 1] = np.angle(F[m])

    # Optimize
    bounds = [(-0.5, 0.5)] * M + [(0, 2 * np.pi)] * M
    # Interleave properly
    bounds = []
    for m in range(M):
        bounds.append((-0.5, 0.5))
        bounds.append((-np.pi, np.pi))

    res = minimize(fourier_obj, init_params, method='Nelder-Mead',
                   options={'maxiter': 5000, 'xatol': 1e-10, 'fatol': 1e-12,
                            'adaptive': True})

    if res.fun < best_obj:
        best_obj = res.fun
        best_a = fourier_seq(res.x)
        if verbose:
            print(f"    Parametric Fourier: obj={best_obj:.10f}")

    # Try different M values
    for M_try in [3, 5, 12]:
        if time.monotonic() - t0 > time_limit * 0.7:
            break
        M = min(M_try, n // 4)
        n_params = 2 * M

        def fourier_seq2(params, M=M, n=n):
            a = np.full(n, 1.0 / n)
            for m in range(M):
                r = params[2 * m]
                phi = params[2 * m + 1]
                a += r * np.cos(2.0 * np.pi * m * np.arange(n) / n + phi)
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-15:
                return np.ones(n) / n
            return a / s

        def fourier_obj2(params, M=M, n=n):
            a = fourier_seq2(params, M, n)
            return obj_norm(a)

        F = np.fft.rfft(a0)
        init_params = np.zeros(n_params)
        for m in range(M):
            if m < len(F):
                init_params[2 * m] = 2.0 * np.abs(F[m]) / n
                init_params[2 * m + 1] = np.angle(F[m])

        bounds = []
        for m in range(M):
            bounds.append((-0.5, 0.5))
            bounds.append((-np.pi, np.pi))

        res = minimize(fourier_obj2, init_params, method='Nelder-Mead',
                       options={'maxiter': 3000, 'xatol': 1e-10, 'fatol': 1e-12,
                                'adaptive': True})
        if res.fun < best_obj:
            best_obj = res.fun
            best_a = fourier_seq2(res.x, M, n)
            if verbose:
                print(f"    Parametric M={M}: obj={best_obj:.10f}")

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining GP-W, subgradient, coordinate search,
    parametric optimization, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10 or len(a) == 0:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768)
        try_seq(parent, "default_gauss")

    # === Phase 1: Diverse initializations ===
    print("\n=== Phase 1: Diverse Initializations ===")
    t0 = time.monotonic()

    init_candidates = []
    n_values = [512, 640, 768, 896, 1024]

    for n_try in n_values:
        init_candidates.append((f'gs_{n_try}', n_try, gs_init(n_try, 60, seed=n_try)))
        init_candidates.append((f'gs2_{n_try}', n_try, gs_init(n_try, 100, seed=n_try + 99)))
        init_candidates.append((f'gauss_{n_try}', n_try, make_gaussian(n_try, 0.25)))
        init_candidates.append((f'bell_{n_try}', n_try, make_bell(n_try, 0.55)))
        init_candidates.append((f'cos_{n_try}', n_try, make_cosine_mod(n_try, 2, 0.25)))
        init_candidates.append((f'multi_{n_try}', n_try, make_multi_cosine(n_try, 4, 0.25, seed=n_try)))

    if parent is not None:
        init_candidates.append(('parent', len(parent), parent))

    # Score all initializations
    scored_inits = []
    for label, n_try, a_init in init_candidates:
        sc = obj_norm(a_init)
        scored_inits.append((sc, a_init, label))
        if sc < 1.8:
            try_seq(a_init, f"init_{label}")

    scored_inits.sort(key=lambda x: x[0])
    print(f"  Best init: {scored_inits[0][2]}, obj={scored_inits[0][0]:.6f}")
    print(f"  Phase 1 time: {time.monotonic()-t0:.1f}s")

    # === Phase 2: LP-FW on top candidates ===
    print("\n=== Phase 2: LP-FW Optimization ===")
    for trial_idx in range(min(4, len(scored_inits))):
        rem = deadline - time.monotonic() - 400
        if rem < 30:
            break
        sc_init, a_init, label = scored_inits[trial_idx]
        # Skip if too close to current best (already optimized)
        if best_seq is not None and len(a_init) == len(best_seq):
            if np.max(np.abs(a_init - best_seq)) < 0.01:
                continue
        t_alloc = min(rem * 0.5, 180)
        print(f"  LP-FW [{label}] n={len(a_init)}: alloc={t_alloc:.0f}s (init={sc_init:.6f})")
        a_opt, obj = lp_fw_optimize(a_init, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, f"lpfw_{label}")

    # === Phase 3: Subgradient on current best ===
    print("\n=== Phase 3: Nesterov Subgradient ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_alloc = min(rem, 300)
            print(f"  SG-N: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = subgradient_nesterov(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "sg_nesterov")

    # === Phase 4: Coordiant search ===
    print("\n=== Phase 4: Coordinate Search ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 20:
            t_alloc = min(rem, 200)
            print(f"  CS: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = coordinate_search(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "coord_search")

    # === Phase 5: Parametric Fourier ===
    print("\n=== Phase 5: Parametric Fourier ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_alloc = min(rem, 100)
            print(f"  Parametric: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = parametric_fourier_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "parametric")

    # === Phase 6: Multi-length exploration ===
    print("\n=== Phase 6: Multi-Length Exploration ===")
    if best_seq is not None:
        base_n = len(best_seq)
        for target_n in [1024, 896, 768, 640, 512]:
            if target_n == base_n:
                continue
            rem = deadline - time.monotonic() - 100
            if rem < 15:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")
            if sc < best_score + 0.03:
                t_alloc = min(rem * 0.3, 60)
                if t_alloc > 10:
                    a_opt, obj = lp_fw_optimize(f_interp, time_limit=t_alloc)
                    try_seq(a_opt, f"lpfw_n{target_n}")
                    a_opt, obj = coordinate_search(a_opt, time_limit=20)
                    try_seq(a_opt, f"cs_n{target_n}")

    # === Phase 7: Perturbation + re-optimization ===
    print("\n=== Phase 7: Perturbation Restart ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for perm_idx, amp in enumerate([0.005, 0.01, 0.02, 0.03]):
            rem = deadline - time.monotonic() - 60
            if rem < 15:
                break
            rng_p = np.random.default_rng(seed + 1000 + perm_idx * 7)
            noise = rng_p.standard_normal(n_best) * amp
            f_pert = np.maximum(best_seq + noise, 0)
            f_pert = f_pert / f_pert.sum()
            sc = try_seq(f_pert, f"pert_{amp}")
            if sc < best_score + 0.02:
                t_alloc = min(rem * 0.4, 50)
                a_opt, obj = subgradient_nesterov(f_pert, time_limit=t_alloc)
                try_seq(a_opt, f"sg_pert_{amp}")

    # === Phase 8: Final polish ===
    print("\n=== Phase 8: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t_alloc = min(rem * 0.5, 40)
            a_opt, obj = coordinate_search(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_cs")

        rem = deadline - time.monotonic() - 5
        if rem > 8:
            t_alloc = min(rem, 20)
            a_opt, obj = subgradient_nesterov(best_seq, time_limit=t_alloc, lr_init=0.005)
            try_seq(a_opt, "final_sg")

        rem = deadline - time.monotonic() - 3
        if rem > 5:
            t_alloc = min(rem, 10)
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_lpfw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"Total time used: {time.monotonic() - (deadline - max(1.0, float(budget_s) - 30.0)):.1f}s")
    return [float(x) for x in best_seq]
