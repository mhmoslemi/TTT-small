```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed gradient descent,
peak-targeted descent, and LP-Frank-Wolfe with diverse initializations
and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.special import logsumexp
import time


def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_raw(a):
    """Compute 2*n*max(conv)/sum^2 (the actual objective)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * n * c.max() / (s * s)


def objective_simplex(a):
    """Compute 2*n*max(conv) assuming a is on the simplex (sum=1)."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()


def smoothed_objective_and_grad(a, beta):
    """
    Compute smoothed objective and gradient for a on the simplex.
    Objective: 2n * logsumexp(beta * conv(a))
    Gradient: 2n * beta * 2 * corr(softmax(beta*conv), a)
    """
    n = len(a)
    c = fast_convolve(a)
    bc = beta * c
    bc_shifted = bc - bc.max()
    exp_bc = np.exp(bc_shifted)
    w = exp_bc / exp_bc.sum()
    
    # Smoothed objective
    obj = 2.0 * n * (logsumexp(bc) - 0)  # logsumexp(bc) = bc.max() + log(sum(exp(bc_shifted)))
    # Actually: logsumexp(bc) = bc.max() + log(sum(exp(bc - bc.max())))
    # So obj = 2*n * (bc.max() + log(w.sum() * exp(0))) = 2*n * bc.max() + 2*n * log(sum(exp_bc))
    # But w = exp(bc_shifted) / sum(exp(bc_shifted)), so logsumexp(bc) = bc.max() + log(sum(exp_bc))
    # Hmm, let me just compute it directly:
    # logsumexp(bc) = bc.max() + log(sum(exp(bc - bc.max())))
    # And sum(exp(bc - bc.max())) = sum(exp_bc)
    # So logsumexp(bc) = bc.max() + log(sum(exp_bc))
    
    # For the gradient, I need:
    # d/d a_j [2n * logsumexp(beta * c)] = 2n * beta * 2 * sum_k w_k * a_{k-j}
    # = 4n * beta * corr(w, a)[j]
    
    # Compute corr(w, a)[j] = sum_k w[k] * a[k-j]
    # Using FFT: IFFT(FFT(w, L) * conj(FFT(a, L)))[j]
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.fft(w, fft_size)
    A = np.fft.fft(a, fft_size)
    corr_full = np.fft.ifft(W * np.conj(A)).real
    grad = 4.0 * n * beta * corr_full[:n]
    
    return obj, grad
```

```python
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-stage optimization:
1. Diverse initializations (parent, Gaussian, cosine-modulated, two-bump)
2. Multi-stage smoothed gradient descent via L-BFGS-B with softmax parameterization
3. LP-Frank-Wolfe with adaptive M levels
4. Peak-targeted coordinate descent for fine-tuning
5. Multi-length exploration
6. Final polish

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.special import logsumexp
import time


def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def obj_and_grad_softmax(x, beta):
    """
    Objective and gradient in x-space (where a = softmax(x)).
    Objective: 2n * logsumexp(beta * conv(a))
    """
    n = len(x)
    a = softmax(x)
    c = fast_convolve(a)
    bc = beta * c
    bc_max = bc.max()
    log_sum_exp = bc_max + np.log(np.sum(np.exp(bc - bc_max)))
    
    # Objective: 2n * logsumexp
    obj = 2.0 * n * log_sum_exp
    
    # Gradient of logsumexp w.r.t. a: 2 * w * a_{k-j} pattern
    # w_k = exp(beta*c_k) / sum(exp(beta*c_k))
    w = np.exp(bc - bc_max)
    w /= w.sum()
    
    # da_j / dx_i = a_j * (delta_ij - a_i)
    # d obj / da_j = 2n * beta * 2 * sum_k w_k * a_{k-j}
    # d obj / dx_i = sum_j (d obj / da_j) * a_j * (delta_ij - a_i)
    #               = a_i * (d obj / da_i - sum_j a_j * d obj / da_j)
    
    # Compute g_j = d obj / da_j = 4n * beta * sum_k w_k * a_{k-j}
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.fft(w, fft_size)
    A = np.fft.fft(a, fft_size)
    corr = np.fft.ifft(W * np.conj(A)).real[:n]
    g = 4.0 * n * beta * corr  # g[j] = d obj / da_j
    
    # Center: g_centered[i] = a_i * (g[i] - sum_j a_j * g[j])
    g_dot_a = np.dot(a, g)
    grad = a * (g - g_dot_a)
    
    return obj, grad


def lbfgs_optimize(a0, time_limit, verbose=False):
    """
    Multi-stage L-BFGS-B optimization with increasing beta.
    """
    n = len(a0)
    a = project_simplex(a0.copy())
    
    # Convert to x-space: a = softmax(x), so x = log(a) (approximately)
    # Actually, any x that gives the right softmax works. Use x = log(a + eps)
    eps = 1e-15
    x = np.log(a + eps)
    # Subtract mean for numerical stability (softmax is shift-invariant)
    x -= x.mean()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    stages = [(10, 300), (30, 300), (100, 300), (300, 300), (1000, 300)]
    
    for beta, max_iter in stages:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func(x_in):
            obj, grad = obj_and_grad_softmax(x_in, beta)
            return obj, grad
        
        # Run L-BFGS-B
        res = minimize(func, x, method='L-BFGS-B', jac=True,
                      options={'maxiter': max_iter, 'maxfun': max_iter * 3,
                              'gtol': 1e-10, 'ftol': 1e-12})
        
        a_new = softmax(res.x)
        obj = fast_convolve(a_new).max()
        
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"      L-BFGS-B beta={beta}: obj={obj:.10f} (iters={res.nit})")
        
        # Update x for next stage
        x = np.log(a_new + eps) - np.log(a_new + eps).mean()
    
    return best_a, best_obj
```

```python
def fourier_optimize(n, M, time_limit, initial_params=None):
    """
    Optimize a sequence parameterized by a Fourier series + softmax.
    a_i = softmax(x_i) where x_i = c + sum_m (A_m cos(2πm i/n) + B_m sin(2πm i/n))
    """
    # Parameters: [c, A_1, B_1, ..., A_M, B_M]
    dim = 1 + 2 * M
    
    if initial_params is None:
        params = np.zeros(dim)
    else:
        params = initial_params.copy()
    
    i_vals = np.arange(n)
    cos_bases = []
    sin_bases = []
    for m in range(1, M + 1):
        cos_bases.append(np.cos(2 * np.pi * m * i_vals / n))
        sin_bases.append(np.sin(2 * np.pi * m * i_vals / n))
    cos_bases = np.array(cos_bases)  # shape (M, n)
    sin_bases = np.array(sin_bases)  # shape (M, n)
    
    def fun_and_grad(p):
        c = p[0]
        A = p[1:2*M:2]
        B = p[2:2*M+1:2]
        
        x = c + cos_bases @ A + sin_bases @ B
        a = softmax(x)
        
        # Objective: 2n * logsumexp(beta * conv(a))
        beta = 200  # high for sharp max approximation
        cov = fast_convolve(a)
        bc = beta * cov
        bc_max = bc.max()
        lse = bc_max + np.log(np.sum(np.exp(bc - bc_max)))
        obj = 2.0 * n * lse
        
        # Gradient
        w = np.exp(bc - bc_max)
        w /= w.sum()
        
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = np.fft.fft(w, fft_size)
        A_fft = np.fft.fft(a, fft_size)
        corr = np.fft.ifft(W * np.conj(A_fft)).real[:n]
        g_a = 4.0 * n * beta * corr  # d obj / da_j
        
        # da_j / dx_k = a_j * (delta_jk - a_k)
        # g_x_k = sum_j g_a_j * a_j * (delta_jk - a_k) = a_k * (g_a_k - dot(a, g_a))
        g_x = a * (g_a - np.dot(a, g_a))
        
        # dx_k / dp params:
        # x = c + cos @ A + sin @ B
        # dx/d(c) = 1
        # dx/d(A_m) = cos(m, :)  (row m of cos_bases)
        # dx/d(B_m) = sin(m, :)  (row m of sin_bases)
        grad = np.zeros(dim)
        grad[0] = np.sum(g_x)  # d obj / d c = sum_k g_x_k * 1
        for m in range(M):
            grad[1 + 2*m] = np.dot(g_x, cos_bases[m])  # d obj / d A_m
            grad[2 + 2*m] = np.dot(g_x, sin_bases[m])  # d obj / d B_m
        
        return obj, grad
    
    # Optimize with L-BFGS-B
    res = minimize(fun_and_grad, params, method='L-BFGS-B', jac=True,
                   options={'maxiter': 500, 'maxfun': 5000, 'gtol': 1e-12, 'ftol': 1e-14})
    
    # Extract optimal sequence
    p = res.x
    c = p[0]
    A = p[1:2*M:2]
    B = p[2:2*M+1:2]
    x = c + cos_bases @ A + sin_bases @ B
    a = softmax(x)
    
    return a, fast_convolve(a).max(), res
```
