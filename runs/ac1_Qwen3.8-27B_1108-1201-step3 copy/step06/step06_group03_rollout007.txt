"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using PAIRAFW (Pairwise Average
Frank-Wolfe), Nesterov-accelerated smoothed optimization, differential
evolution for global search, and coordinate descent refinement.
Explores multiple sequence lengths with diverse asymmetric initializations.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def pairafw_optimize(f0, time_limit, verbose=False):
    """
    Pairwise Average Frank-Wolfe method.
    At each step, compute the LP direction g, then find the coordinate j
    where (g_j - f_j) is largest (most improvement), and update only that
    coordinate by averaging f_j with g_j.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 50 == 0:
            rem = time_limit - (time.monotonic() - t0)
            if verbose:
                print(f"    PAIRAFW iter {it}: obj={best_obj:.10f}, remaining={rem:.0f}s")

        # Solve LP for direction
        g = solve_lp_direction(f, M, n)
        if g is None:
            # Try relaxing M
            g = solve_lp_direction(f, M * (1 + 0.01 * it), n)
            if g is None:
                # Random restart
                rng = np.random.default_rng(it * 17 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                continue

        if g_sum < 1e-15:
            rng = np.random.default_rng(it * 23 + 5)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            continue
        g = g / g_sum

        gap = g - f
        j = int(np.argmax(gap))

                    conv = fast_convolve(f)
                    M = conv.max()
                    no_improve = 0
                continue
            else:
                no_improve = 0

        f_new = f.copy()
        f_new[j] = 0.5 * (f[j] + g[j])
        f_new = np.maximum(f_new, 0.0)

def nesterov_smoothed(f0, time_limit, verbose=False, tau_init=10.0, lr_init=0.01):
    """
    Nesterov-accelerated gradient descent on the smoothed objective:
    LSE(tau * (a*a)) with simplex constraint.
    """
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    t0 = time.monotonic()
    tau = tau_init
    lr = lr_init
    T = 0
    y = f.copy()  # Nesterov momentum variable
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        T += 1
        # Nesterov momentum
        t1 = T + 1
        t2 = T + 2
        theta = (1 + np.sqrt(1 + 4 * t1 * t1)) / 2
        y_new = y + (t1 / t2) * (y - f)
        y_new = np.maximum(y_new, 1e-12)
        y_new = y_new / y_new.sum()
        y = y_new

        # Compute smoothed gradient at y
        conv_y = fast_convolve(y)
        M_y = conv_y.max()
        
        # LSE weights
        scaled = tau * conv_y
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(conv_y))
            w[int(np.argmax(conv_y))] = 1.0
        else:
            w = w / ws

        # Gradient of LSE(tau*c) w.r.t. a: tau * sum_k w_k * dc_k/da
        grad = np.zeros(n)
        for k in range(len(conv_y)):
            if w[k] < 1e-14:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = y[k - idx] + y[idx]
            # Handle the double-counting when k = 2*j
            # Actually for auto-convolution, dc_k/da_m = a_{k-m} for m in valid range
            # but when k-m = m, it's 2*a_m
            # More carefully: c_k = sum_j a_j * a_{k-j}
            # dc_k/da_m = a_{k-m} + a_m (for m != k-m)
            #            = 2*a_m (for m = k-m, i.e., k = 2m)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je >= js:
                idx = np.arange(js, je + 1)
                contrib = y[k - idx] + y[idx]
                grad[js:je + 1] += tau * w[k] * contrib

        # Gradient step with projection
        lr_t = lr * (T + 1) / (T + 1 + tau * 2 * n)  # Adaptive LR
        f_new = f - lr_t * grad
        f_new = np.maximum(f_new, 1e-12)
        f_new = f_new / f_new.sum()

        obj = fast_convolve(f_new).max()
        f = f_new
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt tau
        if T % 100 == 0:
            tau *= 1.05
            if no_improve > 50:
                rng = np.random.default_rng(T * 7 + 13)
                f = np.maximum(best_f + rng.standard_normal(n) * 0.004, 1e-12)
                f = f / f.sum()
                y = f.copy()
                no_improve = 0

        if verbose and T % 200 == 0:
            print(f"    Nesterov iter {T}: obj={best_obj:.10f}, tau={tau:.1f}")

    return best_f, best_obj

def differential_evolution_search(best_f, n, time_limit, pop_size=20, verbose=False):
    """
    Differential evolution for global exploration around best_f.
    """
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    # Initialize population
    population = [best_f.copy()]
    for i in range(pop_size - 1):
        scale = 0.01 + 0.02 * (i / max(1, pop_size - 1))
        mutant = np.maximum(best_f + rng.standard_normal(n) * scale, 0)
        mutant = mutant / mutant.sum()
        population.append(mutant)

    best_obj = fast_convolve(best_f).max()
    best_solution = best_f.copy()

    it = 0
    F = 0.8  # scale factor
    CR = 0.9  # crossover rate

    while time.monotonic() - t0 < time_limit:
        it += 1
        if verbose and it % 5 == 0:
            print(f"    DE iter {it}: best_obj={best_obj:.10f}")

        for i in range(pop_size):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Select three different random individuals
            idxs = [x for x in range(pop_size) if x != i]
            a, b, c = rng.choice(idxs, size=3, replace=False)
            
            # Mutation
            v = population[a] + F * (population[b] - population[c])
            v = np.maximum(v, 0)
            v = v / v.sum()
            
            # Crossover
            trial = population[i].copy()
            for j in range(n):
                if rng.random() < CR or j == int(rng.integers(0, n)):
                    trial[j] = v[j]
            trial = np.maximum(trial, 0)
            trial = trial / trial.sum()
            
            # Selection
            trial_obj = fast_convolve(trial).max()
            curr_obj = fast_convolve(population[i]).max()
            
            if trial_obj < curr_obj:
                population[i] = trial
                if trial_obj < best_obj - 1e-14:
                    best_obj = trial_obj
                    best_solution = trial.copy()

    return best_solution, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

        t_prev = max(T - 1, 1)
        theta_prev = (1 + np.sqrt(1 + 4 * t_prev * t_prev)) / 2
        theta_curr = (1 + np.sqrt(1 + 4 * T * T)) / 2
        beta = (1 - theta_prev) / theta_curr
        y = np.maximum(f + beta * (y - f), 1e-12)
        y = y / y.sum()

        # Compute gradient of LSE(tau * conv(y))
        conv_y = fast_convolve(y)
        scaled = tau * conv_y
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(conv_y))
            w[int(np.argmax(conv_y))] = 1.0
        else:
            w = w / ws

        # Gradient computation (only for significant weights)
        grad = np.zeros(n)
        significant = np.where(w > 1e-10)[0]
        for k in significant:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = y[k - idx] + y[idx]
            grad[js:je + 1] += tau * w[k] * contrib

        # Adaptive learning rate
        lr = 1.0 / (tau * T * 2.0)
        f_new = f - lr * grad
        f_new = np.maximum(f_new, 1e-12)
        f_new = f_new / f_new.sum()

        obj = fast_convolve(f_new).max()
        f = f_new

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt
        if T % 50 == 0:
            tau *= 1.03
        if no_improve > 40:
            rng = np.random.default_rng(T * 7 + 13)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-12)
            f = f / f.sum()
            y = f.copy()
            no_improve = 0

        if verbose and T % 500 == 0:
            print(f"    Nesterov iter {T}: obj={best_obj:.10f}, tau={tau:.1f}")

    return best_f, best_obj
