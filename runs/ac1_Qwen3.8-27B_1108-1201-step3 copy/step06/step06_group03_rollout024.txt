import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]
    return c


def compute_objective(a):
    """Compute the normalized objective: 2*n*max(conv) when sum=1."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    c = fast_convolve(a_norm)
    return 2.0 * n * c.max()


def project_simplex(v):

    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_at_peak(f, k):
    """Compute subgradient of (f*f)[k] with respect to f."""
    n = len(f)
    sg = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    if je < js:
        return sg
    idx = np.arange(js, je + 1)
    sg[js:je + 1] = f[k - idx] + f[idx]
    return sg


def smoothed_mirror_descent(f0, time_limit, verbose=False,
                            tau_init=10.0, tau_max=5000.0,
                            lr_init=0.02, lr_decay=0.9995,
                            tau_growth=1.005, restart_every=150,
                            top_frac=0.1):
    """
    Adaptive smoothed mirror descent with temperature annealing.
    Key improvement: temperature starts low (focus on worst peak) and grows
    to consider more peaks, enabling escape from local optima.
    """
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)
    len_c = len(fast_convolve(f))

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    tau = tau_init
    lr = lr_init
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Compute weighted subgradient using softmax over top peaks
        k_top = max(5, int(len_c * top_frac))
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        scaled = tau * pv - (tau * pv).max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            # Effectively temperature = 0, just use argmax
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            contrib = subgradient_at_peak(f, k)
            sg += w[i] * contrib

        # Multiplicative descent step
        f_new = f * np.exp(-lr * sg / max(sg.max(), 1e-10))
        f_new = np.maximum(f_new, 1e-15)
        s = f_new.sum()
        if s < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / s

        # Evaluate
        obj_new = fast_convolve(f_new).max()
        
        # Backtracking if needed
        if obj_new > M * 1.1:
            for damp in [0.5, 0.25, 0.1, 0.05, 0.01]:
                f_trial = f * (1.0 - damp) + f_new * damp
                f_trial = np.maximum(f_trial, 1e-15)
                f_trial = f_trial / f_trial.sum()
                obj_trial = fast_convolve(f_trial).max()
                if obj_trial < M:
                    f_new = f_trial
                    obj_new = obj_trial
                    break
            else:
                lr *= 0.5
                continue

        f = f_new

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive parameters
        tau = min(tau * tau_growth, tau_max)
        lr = max(lr * lr_decay, 1e-6)

        # Restart with perturbation
        if no_improve > restart_every:
            rng = np.random.default_rng(iter_count * 31 + 7)
            noise_scale = 0.002 + 0.003 * min(no_improve / restart_every, 3.0)
            f = np.maximum(best_f + rng.standard_normal(n) * noise_scale, 1e-15)
            f = f / f.sum()
            lr = lr_init * 0.5
            tau = max(tau_init, tau * 0.5)
            no_improve = 0

        if lr < 1e-7 and tau > tau_max * 0.9:
            break

    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, verbose=False, max_m_inflation=3.0):
    """
    Improved LP-Frank-Wolfe with:
    - Tight initial bound (no inflation)
    - Better line search (golden section)
    - Dual-based stopping criterion
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_inflate = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        M_current = fast_convolve(f).max()
        
        # Solve LP: max sum(g) s.t. (f*g)[k] <= M_current * m_inflate, g >= 0
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M_current * m_inflate)
        c_obj = -np.ones(n)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception:
            m_inflate *= 1.2
            if m_inflate > max_m_inflation:
                # Perturb and restart
                rng = np.random.default_rng(it * 17 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_inflate = 1.0
            continue

        if not res.success or res.x is None:
            m_inflate *= 1.1
            if m_inflate > max_m_inflation:
                rng = np.random.default_rng(it * 17 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                m_inflate = 1.0
            continue

        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            m_inflate *= 1.1
            continue
        
        g = g / g_sum

        # Check if we've converged
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                m_inflate = min(m_inflate * 1.05, max_m_inflation)
            if no_improve > 100:
                # Perturb
                rng = np.random.default_rng(it * 41 + 11)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                m_inflate = 1.0
                no_improve = 0
            continue
        else:
            no_improve = 0

        # Golden section line search
        gr = (np.sqrt(5) - 1) / 2  # golden ratio ~0.618
        lo = 0.0
        hi = 1.0
        c1 = lo + (1 - gr) * (hi - lo)
        c2 = lo + gr * (hi - lo)
        
        def eval_alpha(alpha):
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                return np.inf
            ft = ft / st
            return fast_convolve(ft).max()
        
        fc1 = eval_alpha(c1)
        fc2 = eval_alpha(c2)
        
        for _ in range(20):
            if fc1 < fc2:
                hi = c2
                c2 = c1
                fc2 = fc1
                c1 = lo + (1 - gr) * (hi - lo)
                fc1 = eval_alpha(c1)
            else:
                lo = c1
                c1 = c2
                fc1 = fc2
                c2 = lo + gr * (hi - lo)
                fc2 = eval_alpha(c2)
        
        best_alpha = (lo + hi) / 2
        f_new = f + best_alpha * (g - f)
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()
        
        M_new = fast_convolve(f_new).max()
        f = f_new
        m_inflate = 1.0  # Reset inflation
        
        if M_new < best_obj - 1e-14:
            best_obj = M_new
            best_f = f.copy()
            if verbose and it % 50 == 0:
                print(f"      LP-FW iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def peak_peeling_descent(f0, time_limit, verbose=False):
    """
    Iteratively identify the worst peak, compute the gradient, and take
    a multiplicative step to reduce it. Then check if other peaks have
    become worse. This "peels off" peaks one at a time.
    """
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.05
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Find top peaks
        k_worst = int(np.argmax(conv))
        
        # Also consider top-5 for stability
        top_k = np.argsort(conv)[-5:]
        
        # Compute combined gradient
        sg = np.zeros(n)
        for k in top_k:
            k = int(k)
            weight = conv[k] / conv[k_worst]  # weight by peak value
            sg += weight * subgradient_at_peak(f, k)
        
        # Normalize gradient
        sg_max = sg.max()
        if sg_max < 1e-15:
            break
        sg = sg / sg_max

        # Multiplicative step
        f_new = f * np.exp(-lr * sg)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()
        
        obj_new = fast_convolve(f_new).max()
        
        if obj_new < M * (1 - 1e-10):
            f = f_new
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"      PP iter {it}: obj={best_obj:.10f}")
            lr = min(lr * 1.01, 0.1)
        else:
            lr *= 0.7
            no_improve += 1
        
        if no_improve > 80:
            rng = np.random.default_rng(it * 23 + 5)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.002, 1e-15)
            f = f / f.sum()
            lr = 0.05
            no_improve = 0

    return best_f, best_obj


def make_asymmetric_init(n, variant=0, seed=42):
    """Generate various asymmetric initial sequences."""
    rng = np.random.default_rng(seed + variant * 137)
    x = np.arange(n) / (n - 1) if n > 1 else np.array([0.0])

    if variant == 0:
        a = np.exp(-3.0 * x)
    elif variant == 1:
        a = np.exp(-3.0 * (1 - x))
    elif variant == 2:
        a = np.exp(-2.0 * x) * (1.0 + 0.3 * np.sin(4 * np.pi * x))
        a = np.maximum(a, 0)
    elif variant == 3:
        a = 1.0 / (1.0 + np.exp(4.0 * (x - 0.4)))
    elif variant == 4:
        a = 1.0 / (1.0 + np.exp(4.0 * (x - 0.6)))
    elif variant == 5:
        # Two-scale: slow envelope * fast modulation
        envelope = np.exp(-1.5 * x)
        modulation = 0.7 + 0.3 * np.sin(20 * np.pi * x)
        a = envelope * modulation
    elif variant == 6:
        # Power law
        a = (x + 0.01) ** (-0.7)
    elif variant == 7:
        # Sigmoid front-loaded
        a = 1.0 / (1.0 + np.exp(6.0 * (x - 0.3)))
    elif variant == 8:
        # Random smooth
        a = np.ones(n)
        a += rng.standard_normal(n) * 0.3
        # Smooth
        from scipy.ndimage import uniform_filter1d
        a = uniform_filter1d(a, size=max(5, n // 50))
        a = np.maximum(a, 0)
    elif variant == 9:
        # Cosine ramp (front-heavy)
        a = np.cos(0.5 * np.pi * x) ** 1.5 + 0.05
    else:
        a = np.exp(-2.0 * x)

    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    if old_n == 1:
        return np.full(new_n, f[0] / new_n)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def multi_temperature_smd(f0, time_limit, verbose=False):
    """
    Run SMD at multiple temperatures and keep the best result.
    Low temperature: focus on single worst peak (fast local convergence)
    High temperature: consider many peaks (better global view)
    """
    n = len(f0)
    temps = [5.0, 20.0, 50.0, 200.0]
    best_f = f0.copy()
    best_obj = fast_convolve(f0).max()
    
    t0 = time.monotonic()
    for i, tau in enumerate(temps):
        rem = time_limit - (time.monotonic() - t0)
        if rem < 10:
            break
        t_alloc = rem / (len(temps) - i)
        f_opt, obj = smoothed_mirror_descent(
            best_f, time_limit=t_alloc, verbose=verbose,
            tau_init=tau, tau_max=tau * 20,
            lr_init=0.015, tau_growth=1.003,
            restart_every=100, top_frac=0.15
        )
        if obj < best_obj:
            best_obj = obj
            best_f = f_opt.copy()
            if verbose:
                print(f"      MT-SMD tau={tau}: obj={best_obj:.10f}")
    
    return best_f, best_obj

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]
    return c


def compute_obj_normalized(a):
    """Compute 2*n*max(conv) assuming sum(a)=1."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_n = a / s
    return 2.0 * n * fast_convolve(a_n).max()


def subgradient_at_peak(f, k):
    """Compute subgradient of (f*f)[k] w.r.t. f."""
    n = len(f)
    sg = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    if je >= js:
        idx = np.arange(js, je + 1)
        sg[js:je + 1] = f[k - idx] + f[idx]
    return sg


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_mirror_descent(f0, time_limit, verbose=False,
                            tau_init=10.0, tau_max=5000.0,
                            lr_init=0.02, lr_decay=0.9997,
                            tau_growth=1.004, restart_every=120,
                            top_frac=0.12):
    """Adaptive smoothed mirror descent with temperature annealing."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)
    len_c = 2 * n - 1

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    tau = tau_init
    lr = lr_init
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        conv = fast_convolve(f)
        M = conv.max()

        k_top = max(5, int(len_c * top_frac))
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        scaled = tau * pv
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            sg += w[i] * subgradient_at_peak(f, int(peaks[i]))

        sg_max = sg.max()
        if sg_max < 1e-15:
            break

        f_new = f * np.exp(-lr * sg / sg_max)
        f_new = np.maximum(f_new, 1e-15)
        s = f_new.sum()
        if s < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / s

        obj_new = fast_convolve(f_new).max()

        if obj_new > M * 1.05:
            improved = False
            for damp in [0.5, 0.25, 0.1, 0.05, 0.01]:
                f_trial = f * (1.0 - damp) + f_new * damp
                f_trial = np.maximum(f_trial, 1e-15)
                f_trial = f_trial / f_trial.sum()
                obj_trial = fast_convolve(f_trial).max()
                if obj_trial < M:
                    f_new = f_trial
                    obj_new = obj_trial
                    improved = True
                    break
            if not improved:
                lr *= 0.5
                continue

        f = f_new

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        tau = min(tau * tau_growth, tau_max)
        lr = max(lr * lr_decay, 1e-7)

        if no_improve > restart_every:
            rng = np.random.default_rng(iter_count * 31 + 7)
            noise_scale = 0.002 + 0.003 * min(float(no_improve) / restart_every, 2.0)
            f = np.maximum(best_f + rng.standard_normal(n) * noise_scale, 1e-15)
            f = f / f.sum()
            lr = max(lr_init * 0.4, 1e-6)
            tau = max(tau_init, tau * 0.4)
            no_improve = 0

    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, verbose=False, m_inflate_max=2.5):
    """Improved LP-Frank-Wolfe with golden-section line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_inflate = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        M_current = fast_convolve(f).max()

        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M_current * m_inflate)
        c_obj = -np.ones(n)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception:
            m_inflate *= 1.15
            if m_inflate > m_inflate_max:
                rng = np.random.default_rng(it * 17 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                m_inflate = 1.0
            continue

        if not res.success or res.x is None:
            m_inflate *= 1.1
            if m_inflate > m_inflate_max:
                rng = np.random.default_rng(it * 17 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                m_inflate = 1.0
            continue

        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            m_inflate *= 1.1
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 40:
                m_inflate = min(m_inflate * 1.05, m_inflate_max)
            if no_improve > 150:
                rng = np.random.default_rng(it * 41 + 11)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                m_inflate = 1.0
                no_improve = 0
            continue
        else:
            no_improve = 0

        # Golden section line search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo = 0.0
        hi = 1.0
        c1 = lo + (1.0 - gr) * (hi - lo)
        c2 = lo + gr * (hi - lo)

        def eval_alpha_at(alpha):
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                return np.inf
            ft = ft / st
            return fast_convolve(ft).max()

        fc1 = eval_alpha_at(c1)
        fc2 = eval_alpha_at(c2)

        for _ in range(25):
            if fc1 < fc2:
                hi = c2
                c2 = c1
                fc2 = fc1
                c1 = lo + (1.0 - gr) * (hi - lo)
                fc1 = eval_alpha_at(c1)
            else:
                lo = c1
                c1 = c2
                fc1 = fc2
                c2 = lo + gr * (hi - lo)
                fc2 = eval_alpha_at(c2)

        best_alpha = (lo + hi) / 2.0
        f_new = f + best_alpha * (g - f)
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()

        M_new = fast_convolve(f_new).max()
        f = f_new
        m_inflate = 1.0

        if M_new < best_obj - 1e-14:
            best_obj = M_new
            best_f = f.copy()
            if verbose and it % 100 == 0:
                print(f"      LP-FW iter {it}: obj_per_n={best_obj:.10f}")

    return best_f, best_obj


def peak_peeling_descent(f0, time_limit, verbose=False):
    """Iteratively target the worst convolution peak."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.05
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        top_k = np.argsort(conv)[-5:]
        sg = np.zeros(n)
        for k in top_k:
            k = int(k)
            weight = conv[k] / conv[k_worst] if conv[top_k[0]] > 1e-15 else 1.0
            k_worst = int(top_k[0])
            weight = conv[k] / max(conv[k_worst], 1e-15)
            sg += weight * subgradient_at_peak(f, k)

        sg_max = sg.max()
        if sg_max < 1e-15:
            break
        sg = sg / sg_max

        f_new = f * np.exp(-lr * sg)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        obj_new = fast_convolve(f_new).max()

        if obj_new < M * (1.0 - 1e-10):
            f = f_new
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 200 == 0:
                    print(f"      PP iter {it}: obj={best_obj:.10f}")
            lr = min(lr * 1.005, 0.12)
        else:
            lr *= 0.7
            no_improve += 1

        if no_improve > 100:
            rng = np.random.default_rng(it * 23 + 5)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.002, 1e-15)
            f = f / f.sum()
            lr = 0.04
            no_improve = 0

    return best_f, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    if old_n == 1:
        return np.full(new_n, f[0] / new_n)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def multi_temperature_smd(f0, time_limit, verbose=False):
    """Run SMD at multiple temperatures sequentially."""
    best_f = f0.copy()
    best_obj = fast_convolve(f0).max()
    temps = [5.0, 20.0, 80.0, 300.0]

    t0 = time.monotonic()
    for i, tau in enumerate(temps):
        rem = time_limit - (time.monotonic() - t0)
        if rem < 10:
            break
        t_alloc = rem / (len(temps) - i)
        f_opt, obj = smoothed_mirror_descent(
            best_f, time_limit=t_alloc, verbose=verbose,
            tau_init=tau, tau_max=tau * 15,
            lr_init=0.015, tau_growth=1.003,
            restart_every=80, top_frac=0.15
        )
        if obj < best_obj:
            best_obj = obj
            best_f = f_opt.copy()
            if verbose:
                print(f"      MT-SMD tau={tau}: obj={best_obj:.10f}")

    return best_f, best_obj

            if r == 0:
                noise = rng.standard_normal(n_best) * 0.004
                f_init = np.maximum(best_seq + noise, 1e-12)
                f_init = f_init / f_init.sum()
            elif r == 1:
                asym = make_asymmetric_init(n_best, variant=int(rng.integers(0, 10)), seed=seed + 500 + r)
                f_init = 0.85 * best_seq + 0.15 * asym
                f_init = f_init / f_init.sum()
            else:
                f_init = best_seq.copy()
                mask = rng.random(n_best) < 0.1
                f_init[mask] *= rng.random(int(mask.sum()) + 1) * 0.5
                f_init = np.maximum(f_init, 1e-12)
                f_init = f_init / f_init.sum()

            print(f"  Restart {r}: t={t_alloc:.0f}s")
            a_opt, obj = smoothed_mirror_descent(f_init, time_limit=t_alloc,
                                                 tau_init=8.0, tau_max=400.0,
                                                 lr_init=0.012, restart_every=60)
            score = float(evaluate_sequence(a_opt.tolist()))
            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"  NEW BEST restart {r}: {score:.6f}")

    # === Phase 8: Final LP-FW squeeze ===
    print("\n=== Phase 8: Final Squeeze ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 15:
            t_alloc = min(rem, 45)
            print(f"  Final LP-FW: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))
            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"  NEW BEST final: {score:.6f}")

    # === Phase 9: Final SMD polish (high temp) ===
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 10:
            t_alloc = min(rem - 5, 25)
            print(f"  Final SMD: t={t_alloc:.0f}s")
            a_opt, obj = smoothed_mirror_descent(best_seq, time_limit=t_alloc,
                                                 tau_init=40.0, tau_max=2000.0,
                                                 lr_init=0.004, tau_growth=1.001,
                                                 restart_every=40, top_frac=0.3)
            score = float(evaluate_sequence(a_opt.tolist()))
            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
