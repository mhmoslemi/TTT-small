"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Fictitious play with LP best responses to approximate the Nash equilibrium
   of the auto-correlation minmax game.
2. Multi-temperature log-sum-exp smoothed L-BFGS optimization for gradient-based
   refinement.
3. Adaptive subgradient descent with simplex projection for local polishing.
4. Multi-length exploration to find the globally best sequence length.
The key mathematical insight: the problem reduces to finding a probability
vector minimizing the peak auto-correlation, which is the value of a symmetric
zero-sum game solvable via fictitious play.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_autocorr(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute the objective score for a given sequence."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max() / (s * s)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T where T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver
    if '_lp_solver' in globals() and _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.GLPK, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver):
    """
    Given fixed a (non-negative, sum=1), find b (non-negative, sum=1)
    minimizing max_k conv(a, b)[k]. Returns (b, M) or (None, None).
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def fictitious_play(a_init, time_limit, solver, verbose=True):
    """
    Run fictitious play to approximate Nash equilibrium.
    Returns (best_a, best_score, n_iters).
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0)
    s = a.sum()
    a = a / s

    b = a.copy()
    a_avg = a.copy()
    b_avg = b.copy()
    best_a = a.copy()
    best_score = score_of(a)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        b_new, M_b = solve_lp_best_response(a_avg, solver)
        if b_new is not None:
            b = b_new
        a_new, M_a = solve_lp_best_response(b_avg, solver)
        if a_new is not None:
            a = a_new

        a_avg = (iters * a_avg + a) / (iters + 1)
        b_avg = (iters * b_avg + b) / (iters + 1)

        # Evaluate candidates
        for cand in [a, b, 0.5 * (a + b), a_avg]:
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = score_of(cand)
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()

        if verbose and iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP iter={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    total = exp_z.sum()
    if total < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / total


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def lse_objective_and_grad(z, T, n):
    """
    Compute LSE-smoothed objective and gradient in log-space.
    Objective: 2n * LSE(T * conv(softmax(z), softmax(z)))
    """
    a = softmax_from_z(z)
    c = fast_autocorr(a)
    Tc = T * c
    Tc_max = Tc.max()
    exp_Tc = np.exp(Tc - Tc_max)
    log_sum_exp = Tc_max + np.log(exp_Tc.sum())
    obj = 2.0 * n * log_sum_exp / T

    # Gradient wrt a: 2n * cross_corr(w, a) where w = softmax(T*c)
    w = exp_Tc / exp_Tc.sum()
    # cross_corr(w, a)[i] = sum_k w[k] * a[k-i] (valid range)
    # Using FFT: this is the cross-correlation
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # cross_corr[i] = sum_k w[k] * conj(a[k-i]) -- but for real sequences:
    # We need grad_a[i] = 2n * sum_k w[k] * a[k-i] (only valid k)
    # This is flip(w) convolved with a, restricted to [0, n-1]
    # equivalently: ifft(conj(W) * A)
    grad_a = irfft(np.conj(W) * A, fft_size)[:n]
    grad_a = 2.0 * n * grad_a

    # Gradient wrt z: a * (grad_a - a * dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def lbgfs_multi_temperature(a_init, time_limit):
    """
    Multi-temperature LSE + L-BFGS optimization.
    Starts with high T (smooth) and decreases to low T (sharp).
    Returns (best_a, best_score).
    """
    n = len(a_init)
    t0 = time.monotonic()
    z = z_from_a(a_init)
    best_a = a_init.copy()
    best_score = score_of(a_init)

    # Temperature schedule: start smooth, end sharp
    temps = [30, 60, 120, 250, 500, 1000, 2000]
    # Time allocation per temperature
    total_time = time_limit
    # Allocate more time to later (sharper) temperatures
    weights = [0.05, 0.07, 0.10, 0.13, 0.16, 0.20, 0.29]
    times = [w * total_time for w in weights]

    for idx, (T, t_alloc) in enumerate(zip(temps, times)):
        elapsed = time.monotonic() - t0
        if elapsed >= time_limit:
            break
        t_alloc = min(t_alloc, time_limit - elapsed)
        if t_alloc < 2:
            break

        print(f"    L-BFGS T={T}, budget={t_alloc:.0f}s")
        stopped = [False]
        local_t0 = time.monotonic()

        def func(z_in):
            if time.monotonic() - local_t0 > t_alloc:
                stopped[0] = True
                return 1e15
            obj, _ = lse_objective_and_grad(z_in, T, n)
            return obj

        def jac(z_in):
            if time.monotonic() - local_t0 > t_alloc:
                stopped[0] = True
                return np.zeros_like(z_in)
            _, g = lse_objective_and_grad(z_in, T, n)
            return g

        try:
            res = minimize(
                fun=func,
                x0=z.copy(),
                jac=jac,
                method='L-BFGS-B',
                options={'maxiter': 3000, 'ftol': 1e-15, 'gtol': 1e-9,
                         'maxfun': 20000, 'disp': False}
            )
            a_new = softmax_from_z(res.x)
            sc = score_of(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                z = res.x.copy()
            else:
                # Try with the LSE-optimized z but check the actual score
                z = res.x.copy()
                a_check = softmax_from_z(z)
                sc2 = score_of(a_check)
                if sc2 < best_score:
                    best_score = sc2
                    best_a = a_check.copy()
        except Exception as e:
            print(f"      L-BFGS error at T={T}: {e}")

        elapsed = time.monotonic() - t0
        print(f"    L-BFGS T={T} done ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def subgradient_descend(a_init, time_limit, lr_init=0.003, momentum=0.9):
    """
    Subgradient descent with momentum and simplex projection.
    Targets the top peaks of the auto-correlation.
    Returns (best_a, best_score).
    """
    n = len(a_init)
    t0 = time.monotonic()
    a = np.maximum(a_init.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_of(a)
    v = np.zeros(n)
    lr = lr_init
    no_improve = 0
    step_count = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        # Find top peaks (excluding the trivially small edges)
        effective_range = slice(n // 4, 3 * n // 4 + n // 4)
        c_eff = c.copy()
        c_eff[:n // 4] = -np.inf
        c_eff[3 * n // 4 + n // 4:] = -np.inf
        top_k = min(30, n // 8)
        peaks = np.argpartition(c_eff, -top_k)[-top_k:]
        peak_vals = c[peaks]

        # Temperature-weighted subgradient
        temp = 0.01 * max(peak_vals.max(), 1e-10)
        scaled = (peak_vals - peak_vals.max()) / temp
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        # Update with momentum
        v = momentum * v - lr * sg
        a_candidate = a + v

        # Project onto simplex
        u = np.sort(a_candidate)[::-1]
        cssv = np.cumsum(u)
        cond = u * np.arange(1, n + 1) > (cssv - 1)
        if cond.any():
            rho = np.nonzero(cond)[0][-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            a_new = np.maximum(a_candidate - theta, 0.0)
        else:
            a_new = np.ones(n) / n
        a_new = a_new / a_new.sum()

        sc = score_of(a_new)
        step_count += 1
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        # Decay learning rate
        lr *= 0.999
        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 1000000)
            a = best_a + rng.standard_normal(n) * 0.005
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 5, 0.003)
            no_improve = 0

        if step_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def peak_shave(a_init, time_limit):
    """
    Greedy peak shaving: identify the peak, reduce the contributing
    elements, redistribute mass.
    Returns (best_a, best_score).
    """
    n = len(a_init)
    t0 = time.monotonic()
    a = np.maximum(a_init.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_of(a)

    step = 0
    while time.monotonic() - t0 < time_limit:
        step += 1
        c = fast_autocorr(a)
        peak_k = int(np.argmax(c))
        peak_val = c[peak_k]
        target = best_score * s * s / (2.0 * n)
        if peak_val <= target * 1.001:
            break

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        n_top = min(5, len(contribs))
        top_contrib = np.argsort(contribs)[::-1][:n_top]

        improved = False
        for ci in top_contrib:
            i = int(idx[ci])
            j = int(peak_k - i)
            if a[i] < 1e-12:
                continue
            for factor in [0.97, 0.94, 0.90, 0.85, 0.80]:
                a_trial = a.copy()
                old_val = a_trial[i]
                a_trial[i] = old_val * factor
                diff = old_val - a_trial[i]
                if diff < 1e-15:
                    continue
                # Redistribute to other elements proportional to their values
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                sc = score_of(a_trial)
                if sc < best_score - 1e-13:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # Try reducing both i and j
        if not improved:
            for ci in top_contrib[:3]:
                i = int(idx[ci])
                j = int(peak_k - i)
                if i == j or a[j] < 1e-12:
                    continue
                for f1 in [0.95, 0.90]:
                    for f2 in [0.95, 0.90]:
                        a_trial = a.copy()
                        old_i = a_trial[i]
                        old_j = a_trial[j]
                        a_trial[i] = old_i * f1
                        a_trial[j] = old_j * f2
                        diff = (old_i - a_trial[i]) + (old_j - a_trial[j])
                        mask = np.ones(n, dtype=bool)
                        mask[i] = False
                        mask[j] = False
                        others = a_trial[mask]
                        others_sum = others.sum()
                        if others_sum < 1e-15:
                            continue
                        a_trial[mask] = others + diff * others / others_sum
                        a_trial = a_trial / a_trial.sum()
                        sc = score_of(a_trial)
                        if sc < best_score - 1e-13:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                break

    return best_a, best_score


def interpolate_to_length(a, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp = a_interp / ss
    return a_interp


def check_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        sc = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        # Fallback initialization
        n_init = 768
        a_init = np.ones(n_init) / n_init
        t = np.linspace(0, 2 * np.pi, n_init)
        a_init = a_init + 0.02 * np.cos(2 * t) + 0.01 * np.sin(3 * t)
        a_init = np.maximum(a_init, 0)
        a_init = a_init / a_init.sum()
        best_seq = a_init
        best_score = float(evaluate_sequence(a_init.tolist()))
        print(f"Fallback init: n={n_init}, score={best_score:.6f}")

    n_parent = len(parent) if parent is not None else len(best_seq)

    # === Phase 1: Fictitious Play ===
    print(f"\n=== Phase 1: Fictitious Play (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 60:
        t_fp = min(rem * 0.7, 500.0)
        print(f"  FP n={n_parent}, budget={t_fp:.0f}s")
        a_fp, score_fp, iters = fictitious_play(best_seq, t_fp, solver, verbose=True)
        best_seq, best_score = check_update(a_fp, f"FP_n{n_parent}", best_seq, best_score)

    # === Phase 2: Multi-temperature L-BFGS ===
    print(f"\n=== Phase 2: L-BFGS Multi-T (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 400
    if rem > 30 and best_seq is not None:
        t_lbfgs = min(rem * 0.7, 400.0)
        print(f"  L-BFGS n={len(best_seq)}, budget={t_lbfgs:.0f}s")
        a_lb, sc_lb = lbgfs_multi_temperature(best_seq, t_lbfgs)
        best_seq, best_score = check_update(a_lb, "lbgfs_multi", best_seq, best_score)

    # === Phase 3: Subgradient descent ===
    print(f"\n=== Phase 3: Subgradient (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 300
    if rem > 20 and best_seq is not None:
        t_sg = min(rem * 0.6, 200.0)
        print(f"  SubGrad n={len(best_seq)}, budget={t_sg:.0f}s")
        a_sg, sc_sg = subgradient_descend(best_seq, t_sg)
        best_seq, best_score = check_update(a_sg, "subgrad", best_seq, best_score)

    # === Phase 4: Peak shaving ===
    print(f"\n=== Phase 4: Peak Shave (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 15 and best_seq is not None:
        t_ps = min(rem * 0.6, 150.0)
        print(f"  PeakShave n={len(best_seq)}, budget={t_ps:.0f}s")
        a_ps, sc_ps = peak_shave(best_seq, t_ps)
        best_seq, best_score = check_update(a_ps, "peak_shave", best_seq, best_score)

    # === Phase 5: Second round FP + polish ===
    print(f"\n=== Phase 5: FP Round 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 150
    if rem > 40 and best_seq is not None:
        n_cur = len(best_seq)
        t_fp2 = min(rem * 0.5, 200.0)
        print(f"  FP2 n={n_cur}, budget={t_fp2:.0f}s")
        a_fp2, sc_fp2, it2 = fictitious_play(best_seq, t_fp2, solver, verbose=False)
        best_seq, best_score = check_update(a_fp2, "FP2", best_seq, best_score)

        # Quick L-BFGS polish
        rem = deadline - time.monotonic() - 80
        if rem > 15 and best_seq is not None:
            t_l2 = min(rem * 0.6, 100.0)
            a_l2, sc_l2 = lbgfs_multi_temperature(best_seq, t_l2)
            best_seq, best_score = check_update(a_l2, "lbgfs2", best_seq, best_score)

    # === Phase 6: Try alternative lengths ===
    print(f"\n=== Phase 6: Alternative Lengths (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 60
    if rem > 60 and best_seq is not None:
        n_cur = len(best_seq)
        alt_lengths = []
        for factor in [0.75, 1.25, 1.5]:
            alt_n = int(n_cur * factor)
            if 256 <= alt_n <= 4096 and alt_n != n_cur:
                alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:2]:
            rem = deadline - time.monotonic() - 30
            if rem < 40:
                break
            t_alloc = min(rem * 0.6, 120.0)
            init_alt = interpolate_to_length(best_seq, alt_n)
            print(f"  Alt length n={alt_n}, budget={t_alloc:.0f}s")

            # Quick FP
            t_fp_alt = t_alloc * 0.5
            a_alt, sc_alt, _ = fictitious_play(init_alt, t_fp_alt, solver, verbose=False)
            best_seq, best_score = check_update(a_alt, f"FP_alt_n{alt_n}", best_seq, best_score)

            # Quick L-BFGS
            rem = deadline - time.monotonic() - 20
            if rem > 10:
                t_lb_alt = min(rem * 0.7, 60.0)
                a_alt2, _ = lbgfs_multi_temperature(best_seq, t_lb_alt)
                best_seq, best_score = check_update(a_alt2, f"lbfgs_alt_n{alt_n}", best_seq, best_score)

    # === Phase 7: Final micro-polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            t_final = min(rem * 0.5, 20.0)
            a_fin, _ = peak_shave(best_seq, t_final)
            best_seq, best_score = check_update(a_fin, "final_ps", best_seq, best_score)

        rem = deadline - time.monotonic() - 2
        if rem > 3:
            t_final2 = min(rem * 0.6, 10.0)
            a_fin2, _ = subgradient_descend(best_seq, t_final2, lr_init=0.001, momentum=0.95)
            best_seq, best_score = check_update(a_fin2, "final_sg", best_seq, best_score)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
