import cvxpy as cp
import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def lp_refine(p, M, solver='ECOS'):
    """Solve LP: max sum(g) s.t. g >= 0, conv(p, g) <= M"""
    n = len(p)
    g = cp.Variable(n, nonneg=True)
    
    # conv(p, g) <= M for all k
    # conv(p,g)[k] = sum_i p[i] * g[k-i]
    # This is linear in g!
    # Build the convolution as: for each k, sum_i p[i] * g[k-i]
    # We can vectorize this using the structure
    
    # Actually, conv(p, g) = F^{-1}(F(p) * F(g)) but that's nonlinear in terms of cvxpy
    # Better: build the constraint matrix explicitly
    
    # conv(p,g)[k] = sum_{i: 0<=k-i<n} p[i] * g[k-i]
    # For a fixed k, this is a dot product of a fixed vector with g
    
    # Let's build the matrix: C[k, i] = p[i] if 0 <= k-i < n, else 0
    # Then C @ g gives conv(p, g)
    
    # Size of C is (2n-1) x n
    # C[k, i] = p[i] * delta(0 <= k-i < n)
    # C[k, i] = p[i] * delta(0 <= k-i) * delta(k-i < n)
    # C[k, i] = p[i] * delta(k-i >= 0) * delta(k-i < n)
    # C[k, i] = p[i] * delta(i <= k) * delta(i > k-n)
    # C[k, i] = p[i] * delta(max(0, k-n+1) <= i <= min(k, n-1))
    
    # For memory efficiency with n=1152, C is 2303x1152, which is about 2.6M entries
    # This is fine.
    
    # Actually, we can use a Toeplitz-like structure. Let's just build it.
    C = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        C[k, lo:hi+1] = p[lo:hi+1]
    
    constraints = [C @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=solver)
    
    if g.value is None:
        return None
    g_val = g.value
    g_val = np.maximum(g_val, 0)
    s = g_val.sum()
    if s < 1e-10:
        return None
    return g_val / s

                           solver='ECOS', max_iter=1000):

    p = p / p.sum()
    n = len(p)
    t0 = time.monotonic()
    best_p = p.copy()
    best_M = fast_convolve(p).max()
    t = t_initial
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = fast_convolve(p).max()
        
        # Solve LP
        g = lp_solve(p, M, solver)
        if g is None:
            t *= 0.5
            continue
        
        # Line search: find best alpha in [0, 1] for p_new = (1-alpha)*p + alpha*g
        best_alpha = 0
        best_obj = M
        for alpha in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            p_try = (1 - alpha) * p + alpha * g
            M_try = fast_convolve(p_try).max()
            if M_try < best_obj:
                best_obj = M_try
                best_alpha = alpha
        
        if best_alpha == 0:
            t *= 0.5
            if t < t_min:
                break
            continue
        
        p = (1 - best_alpha) * p + best_alpha * g
        p = p / p.sum()
        
        if best_obj < best_M - 1e-10:
            best_M = best_obj
            best_p = p.copy()
        
        # Adaptive step size
        if best_alpha < 0.05:
            t *= 0.9
        elif best_alpha > 0.5:
            t = min(t * 1.2, 0.5)
    
    return best_p, best_M

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(p, n_active=None, active_indices=None):
    """Build constraint matrix C such that C @ g = conv(p, g).
    If n_active is given, only build rows for active indices."""
    n = len(p)
    if active_indices is None:
        active_indices = np.arange(2 * n - 1)
    C = np.zeros((len(active_indices), n))
    for idx, k in enumerate(active_indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        C[idx, lo:hi+1] = p[lo:hi+1]
    return C, active_indices

def lp_solve_direction(p, M, solver='PDLP', max_constraints=None):
    """
    Solve LP: maximize sum(g) s.t. g >= 0, conv(p, g) <= M (for selected constraints).
    Returns g normalized to sum=1, or None if infeasible.
    """
    n = len(p)
    b = fast_convolve(p)
    
    # Select active constraints (top peaks)
    if max_constraints is not None and max_constraints < len(b):
        # Use top max_constraints peaks
        top_idx = np.argsort(b)[-max_constraints:]
        top_idx = np.sort(top_idx)
    else:
        top_idx = np.arange(len(b))
    
    C, active_idx = build_conv_matrix(p, active_indices=top_idx)
    
    g = cp.Variable(n, nonneg=True)
    constraints = [C @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver='ECOS', verbose=False)
        except Exception:
            return None
    
    if g.value is None:
        return None
    
    g_val = np.array(g.value).flatten()
    g_val = np.maximum(g_val, 0.0)
    s = g_val.sum()
    if s < 1e-10:
        return None
    return g_val / s

def lp_iterative_optimize(p0, time_limit, solver='PDLP', max_constraints=None,

    """

    p = p / p.sum()
    n = len(p)
    t0 = time.monotonic()
    best_p = p.copy()
    best_M = fast_convolve(p).max()
    
    stagnation = 0
    for it in range(10000):
        if time.monotonic() - t0 > time_limit:
            break

        g = lp_solve_direction(p, M, solver=solver, max_constraints=max_constraints)
        if g is None:
            break
        
        # Line search
        best_alpha = 0
        best_M_try = M
        for alpha in line_search_points:
            p_try = (1 - alpha) * p + alpha * g
            p_try = np.maximum(p_try, 0)
            s = p_try.sum()
            if s < 1e-10:
                continue
            p_try = p_try / s
            M_try = fast_convolve(p_try).max()
            if M_try < best_M_try - 1e-15:
                best_M_try = M_try
                best_alpha = alpha
        
        if best_alpha == 0:
            stagnation += 1
            if stagnation > 50:
                break
            continue
        
        stagnation = 0
        p = (1 - best_alpha) * p + best_alpha * g
        p = np.maximum(p, 0)
        p = p / p.sum()
        
        if best_M_try < best_M - 1e-12:
            best_M = best_M_try
            best_p = p.copy()
    
    return best_p, best_M

def subgradient_optimize(a0, time_limit, lr_init=0.005, momentum=0.9,
                         temp_init=0.0005, temp_growth=1.02,
                         kick_interval=2000, kick_size=0.003,
                         n_top=None, seed=42):
    """Projected subgradient method with momentum."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    a = a / S
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    cur_temp = temp_init
    no_improve = 0
    it_count = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()
        
        scaled = b / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
        else:
            w /= w_sum
        
        if n_top is not None and n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx]
            w_sel /= w_sel.sum()
            w_full = np.zeros(len(b))
            w_full[top_idx] = w_sel
            w = w_full
        
        g = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += w[k] * a[k - idx]
        
        v = momentum * v - cur_lr * g
        a_new = project_simplex(a + v)
        
        b_new = fast_convolve(a_new)
        obj_new = b_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        it_count += 1
        
        if no_improve > kick_interval:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.7, lr_init * 0.1)
            no_improve = 0
        elif no_improve > kick_interval // 2:
            cur_lr *= 0.95
        
        cur_temp *= temp_growth
        if cur_temp > 0.5:
            cur_temp = 0.5
    
    return best_a, best_obj

def coordinate_polish(a0, time_limit, multipliers=None):
    """Coordinate descent polish."""
    if multipliers is None:
        multipliers = np.array([0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0])
    
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                ss = a_trial.sum()
                if ss < 1e-10:
                    continue
                a_trial = a_trial / ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
    return best_a, best_obj

def make_gaussian_init(n, sigma_frac=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma_frac) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 1e-10)

def make_uniform_init(n):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
