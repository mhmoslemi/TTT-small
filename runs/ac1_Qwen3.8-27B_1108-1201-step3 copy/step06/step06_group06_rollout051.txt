```python
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Diverse initialization with asymmetric and spectral-based starting points
2. Adam smooth optimization with adaptive tau and aggressive restarts
3. L-BFGS on softmax-parameterized smoothed objective for fast convergence
4. Direct subgradient with Barzilai-Borwein step size
5. Multi-length exploration (512 to 4096)
6. Final fine-grained polish

Key innovations:
- L-BFGS phase exploits the smooth structure near the optimum
- Spectral equalization in frequency domain
- Asymmetric initialization based on anti-correlation at the center lag
- BB step size for adaptive direct subgradient

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_h_and_M(f, tau):
    """Compute h vector and M_smooth for the log-sum-exp smoothed gradient.
    h_i = sum_l w_{i+l} * f_l
    M_smooth = sum_k w_k * c_k
    where w_k = exp(c_k/tau) / sum(exp(c_k/tau))
    """
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    h = conv_result[n - 1:2 * n - 1][::-1]
    
    M_smooth = np.dot(w, conv)
    return h, M_smooth, conv, w


def adam_smooth_optimize(f0, time_limit, tau_init=None, lr_init=0.003, verbose=False):
    """Adam optimization on the log-sum-exp smoothed objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    conv = fast_convolve(f)
    if tau_init is None:
        tau = 0.08 * conv.max()
    else:
        tau = tau_init
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    lr = lr_init
    t = 0
    
    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        h, M_smooth, conv, w = compute_h_and_M(f, tau)
        M = conv.max()
        
        # Gradient: (2/tau) * h (for the simplex-embedded version)
        grad = (2.0 / tau) * h
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        
        tau *= 0.9997
        if tau < 1e-7:
            tau = 1e-7
        
        if no_improve > 100:
            lr *= 0.85
        if no_improve > 400:
            rng = np.random.default_rng(t * 13 + 7)
            noise_scale = 0.002 + 0.003 * min(no_improve / 1000.0, 3.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = lr_init * 0.7
            tau = 0.05 * fast_convolve(f).max()
            if verbose:
                print(f"    Adam restart at iter {t}: obj={2*n*best_obj:.8f}")
        
        if verbose and t % 1000 == 0:
            obj_score = 2 * n * best_obj
            print(f"    Adam iter {t}: obj={obj_score:.8f}, tau={tau:.2e}, lr={lr:.5f}")
    
    return best_f, 2 * n * best_obj


def lbfgs_smooth_optimize(f0, time_limit, verbose=False):
    """L-BFGS on softmax-parameterized smoothed objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    conv = fast_convolve(f)
    tau = 0.04 * conv.max()
    
    # Initialize x from a: x_j = log(a_j) (approximately)
    x = np.log(f + 1e-30)
    x -= x.mean()  # Center to reduce flat direction effect
    
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    
    def objective_and_grad(x):
        a = np.exp(x - x.max())
        a /= a.sum()
        conv_local = fast_convolve(a)
        scaled = conv_local / tau
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        val = conv_local.max() / tau + np.log(np.sum(np.exp(scaled)))
        val += 1e-12 * np.sum(x * x)
        
        # Gradient
        w_rev = w[::-1]
        total_len = n + len(w) - 1
        fft_size = 1 << (total_len - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        W = np.fft.rfft(w_rev, fft_size)
        conv_result = np.fft.irfft(A * W, fft_size)
        h = conv_result[n - 1:2 * n - 1][::-1]
        
        S_h = np.dot(h, a)
        grad = (2.0 / tau) * a * (h - S_h)
        grad += 2e-12 * x
        
        return val, grad
    
    # Run L-BFGS with time management
    max_iter = 0
    while time.monotonic() - t0 < time_limit:
        batch_time = min(60, time_limit - (time.monotonic() - t0))
        
        try:
            result = minimize(
                lambda x: objective_and_grad(x)[0],
                x,
                method='L-BFGS-B',
                jac=lambda x: objective_and_grad(x)[1],
                options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-10,
                         'maxfun': 500, 'iprint': -1}
            )
            x = result.x
            a = np.exp(x - x.max())
            a /= a.sum()
            obj = fast_convolve(a).max()
            
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = a.copy()
                if verbose:
                    print(f"    L-BFGS improved: obj={2*n*best_obj:.8f}")
            
            # Decrease tau for next round
            tau *= 0.7
            if tau < 1e-8:
                tau = 1e-8
                break
            
            # Re-center x
            x = np.log(a + 1e-30)
            x -= x.mean()
            
        except Exception as e:
            if verbose:
                print(f"    L-BFGS error: {e}")
            break
    
    return best_f, 2 * n * best_obj


def bb_subgradient_optimize(f0, time_limit, verbose=False):
    """Direct subgradient method with Barzilai-Borwein step size."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    t = 0
    
    # BB state
    prev_f = f.copy()
    prev_grad = None
    eta = 0.01  # initial step size
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Find peaks (within 1% of max)
        threshold = M * 0.99
        peak_mask = conv >= threshold
        peak_indices = np.where(peak_mask)[0]
        
        if len(peak_indices) == 0:
            peak_indices = [np.argmax(conv)]
        
        # Compute subgradient: average of subgradients at peaks
        # Subgradient at peak k: g_j = 2 * a_{k-j} (for valid j)
        # We compute: grad_j = (2/|P|) * sum_{k in P} a_{k-j}
        # This is equivalent to (2/|P|) * (a * w_peaks) where w_peaks is the indicator
        
        # Use soft weighting for better behavior
        scaled = conv[peak_indices] / (0.01 * M)
        scaled -= scaled.max()
        w_peaks = np.exp(scaled)
        w_peaks /= w_peaks.sum()
        
        # Construct full weight vector
        w_full = np.zeros(2 * n - 1)
        w_full[peak_indices] = w_peaks
        
        # Compute h (same as before but with w_full instead of softmax weights)
        w_rev = w_full[::-1]
        total_len = n + len(w_full) - 1
        fft_size = 1 << (total_len - 1).bit_length()
        A = np.fft.rfft(f, fft_size)
        W = np.fft.rfft(w_rev, fft_size)
        conv_result = np.fft.irfft(A * W, fft_size)
        h = conv_result[n - 1:2 * n - 1][::-1]
        
        grad = 2.0 * h
        
        # BB step size update
        if prev_grad is not None:
            s_vec = f - prev_f
            y_vec = grad - prev_grad
            sT_y = np.dot(s_vec, y_vec)
            yT_y = np.dot(y_vec, y_vec)
            sT_s = np.dot(s_vec, s_vec)
            
            if yT_y > 1e-14:
                if sT_y > 1e-10:
                    eta_new = sT_y / yT_y
                elif sT_s > 1e-14:
                    eta_new = sT_s / (np.linalg.norm(y_vec) * np.linalg.norm(s_vec) + 1e-30)
                    eta_new = sT_s / yT_y if yT_y > 0 else eta
                else:
                    eta_new = eta
                # Clip step size
                eta = np.clip(0.5 * eta + 0.5 * eta_new, 1e-6, 0.1)
            else:
                eta *= 0.95
        
        # Take step
        f_new = f - eta * grad
        f_new = project_simplex(f_new)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        prev_f = f.copy()
        prev_grad = grad.copy()
        f = f_new
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f.copy()
            if verbose and t % 100 == 0:
                print(f"    BB iter {t}: obj={2*n*best_obj:.8f}, eta={eta:.6f}")
        
        if t % 500 == 0:
            #偶尔 restart to avoid getting stuck
            rng = np.random.default_rng(t * 7)
            if fast_convolve(f).max() > best_obj * 1.01:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
    
    return best_f, 2 * n * best_obj


def mirror_descent_optimize(f0, time_limit, verbose=False):
    """Mirror descent (multiplicative weights) on the smoothed objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    t = 0
    tau = 0.06 * fast_convolve(f).max()
    eta = 0.1  # mirror descent step size
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        h, M_smooth, conv, w = compute_h_and_M(f, tau)
        M = conv.max()
        
        # Mirror descent: f_new = f * exp(-eta * grad) / Z
        # grad = (2/tau) * h (for the smoothed objective)
        grad = (2.0 / tau) * h
        log_update = -eta * grad
        log_update -= log_update.max()  # numerical stability
        update = np.exp(log_update)
        f_new = f * update
        f_new = np.maximum(f_new, 1e-30)
        f_new /= f_new.sum()
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        f = f_new
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f.copy()
            if verbose and t % 200 == 0:
                print(f"    Mirror iter {t}: obj={2*n*best_obj:.8f}")
        
        # Decay tau and eta
        tau *= 0.9995
        if tau < 1e-7:
            tau = 1e-7
        eta *= 0.999
        
        # Periodic restart
        if t % 2000 == 0:
            rng = np.random.default_rng(t * 11)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            tau = 0.04 * fast_convolve(f).max()
            eta = 0.08
    
    return best_f, 2 * n * best_obj


def spectral_equalization(f0, time_limit, verbose=False):
    """Frequency-domain spectral equalization: try to flatten |FFT(a)|^2."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    t = 0
    
    # FFT size
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Compute spectrum
        A = np.fft.rfft(f, fft_size)
        power = np.abs(A) ** 2
        
        # Find the minimum non-DC power
        if len(power) > 1:
            non_dc = power[1:]
            min_idx = np.argmin(non_dc) + 1
            min_power = non_dc.min()
            max_power = power[1:].max() if len(power) > 1 else 1.0
            
            # Target ratio: try to raise the minimum
            if min_power > 1e-30:
                ratio = min_power / max_power
                if ratio < 0.5:
                    # Need to boost the minimum frequency component
                    # Direction in time domain: exp(2*pi*i*min_idx*j/fft_size)
                    # But we need a real, non-negative direction
                    # Use cos(2*pi*min_idx*j/fft_size)
                    direction = np.cos(2 * np.pi * min_idx * np.arange(n) / fft_size)
                    
                    # Try adding this direction
                    for alpha in [0.001, 0.005, 0.01, 0.02]:
                        trial = f + alpha * direction
                        trial = np.maximum(trial, 0)
                        ts = trial.sum()
                        if ts < 1e-10:
                            continue
                        trial /= ts
                        obj = fast_convolve(trial).max()
                        if obj < best_obj - 1e-15:
                            f = trial
                            best_obj = obj
                            best_f = f.copy()
                            break
        
        if t % 100 == 0 and verbose:
            print(f"    Spectral iter {t}: obj={2*n*best_obj:.8f}, min/max={ratio:.4f}")
        
        if t % 500 == 0:
            # Add small noise to explore
            rng = np.random.default_rng(t * 3)
            f = project_simplex(f + rng.standard_normal(n) * 0.001)
    
    return best_f, 2 * n * best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_asymmetric_gaussian(n, center_frac=0.4, sigma_frac=0.3):
    """Asymmetric Gaussian initialization."""
    x = np.arange(n) - n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_ramp(n, power=1.0):
    """Ramp-shaped initialization: a_j = (j+1)^power / sum."""
    a = (np.arange(n) + 1) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_two_lobe(n, separation=0.3):
    """Two-lobe initialization with separation."""
    x = np.arange(n) / n
    a = np.exp(-20 * (x - 0.5 + separation / 2) ** 2) + np.exp(-20 * (x - 0.5 - separation / 2) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosmodulated(n, freq=3.0, depth=0.7):
    """Cosine-modulated initialization."""
    a = 1.0 + depth * np.cos(2 * np.pi * freq * np.arange(n) / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
