Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based flattening iterations.

Core idea: At each step, solve an LP to find the direction d that minimizes
max_k (a*d)[k] subject to d >= 0, sum(d) = 1. Moving toward d flattens the
cross-convolution, which reduces the auto-convolution peak after the update.

This is a principled subgradient/LP hybrid that should converge faster than

from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1

    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build sparse matrix T such that T @ d = a * d (linear convolution)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    T = csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return T

def solve_lp_direction(a):

    Returns (d, t_opt, success)."""
    n = len(a)
    T = build_conv_matrix(a)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based flattening iterations.

Core idea: At each step, solve an LP to find direction d that minimizes
max_k (a*d)[k] subject to d >= 0, sum(d) = 1. Moving toward d flattens the
cross-convolution, reducing the auto-convolution peak. Combined with line
search, diverse restarts, and coordinate descent polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
from scipy.sparse import csc_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def objective_value(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (S * S)


def build_conv_matrix(a):
    """Build dense matrix T such that T @ d = a * d (linear convolution, full mode).
    T has shape (2n-1, n). T[k, j] = a[j] if 0 <= k-j < n."""
    n = len(a)
    # Column j has nonzeros at rows j, j+1, ..., j+n-1 (clipped to [0, 2n-2])
    # All with value a[j]
    T = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_hi = min(j + n - 1, 2 * n - 2)
        T[j:k_hi + 1, j] = a[j]
    return T


def solve_lp_direction(a):
    """Solve: min t s.t. (a*d)[k] <= t for all k, d >= 0, sum(d) = 1.
    Returns (d, t_opt) or (None, None) on failure."""
    n = len(a)
    T = build_conv_matrix(a)
    
    # Variables: [d_0, ..., d_{n-1}, t]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # A_ub: T @ d - t <= 0  =>  [T | -1] @ [d; t] <= 0
    A_ub = np.hstack([T, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(d) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: d_i >= 0, t free
    bounds = [(0.0, None)] * n + [(-np.inf, np.inf)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        d = res.x[:n]
        d = np.maximum(d, 0)
        s = d.sum()
        if s < 1e-15:
            return None, None
        d = d / s
        return d, res.x[-1]
    return None, None


def line_search(a, d, n_fine=30):
    """Find best step t in [0,1] minimizing max(conv((1-t)a+td, (1-t)a+td))."""
    best_t = 0.0
    best_obj = objective_value(a)
    
    # Coarse grid
    ts_coarse = np.array([0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0])
    for t in ts_coarse:
        a_trial = (1 - t) * a + t * d
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 0.01:
            continue
        a_trial = a_trial / s
        obj = objective_value(a_trial)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    
    # Fine search around best
    if best_t > 0:
        lo = max(0.0, best_t - 0.15)
        hi = min(1.0, best_t + 0.15)
        ts_fine = np.linspace(lo, hi, n_fine)
        for t in ts_fine:
            a_trial = (1 - t) * a + t * d
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()

            a_trial = a_trial / s
            obj = objective_value(a_trial)
            if obj < best_obj:
                best_obj = obj
                best_t = t
    
    return best_t, best_obj


def lp_flatten(a0, time_limit, max_stall=30, verbose=False):
    """LP-based flattening iteration loop."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 0.01:
        return a0.copy(), objective_value(a0)
    a = a / S
    
    best_a = a.copy()
    best_obj = objective_value(a)
    no_improve = 0
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        
        # Solve LP for direction
        d, t_opt = solve_lp_direction(a)
        if d is None:
            if verbose:
                print(f"  LP failed at iter {iteration}")
            break
        
        # Check if direction differs from current
        diff = np.linalg.norm(d - a)
        if diff < 1e-10:
            if verbose:
                print(f"  Converged at iter {iteration} (||d-a||={diff:.2e})")
            break
        
        # Line search
        step, obj_new = line_search(a, d)
        
        if obj_new < best_obj - 1e-10:
            a_new = (1 - step) * a + step * d
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 0.01:
                no_improve += 1
            else:
                a_new = a_new / s
                a = a_new
                best_a = a.copy()
                best_obj = obj_new
                no_improve = 0
                if verbose and (iteration % 20 == 1):
                    print(f"  Iter {iteration}: obj={best_obj:.8f}, step={step:.4f}, "
                          f"t_opt={t_opt:.6f}, M={a.sum()*0+fast_convolve(a).max():.6f}")
        else:
            no_improve += 1
        
        if no_improve > max_stall:
            if verbose:
                print(f"  Stalled at iter {iteration}, obj={best_obj:.8f}")
            break
    
    return best_a, best_obj


def subgradient_step(a, n_top=50, temp=0.001):
    """Single subgradient step with temperature-weighted peaks."""
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()
    
    # Temperature-weighted distribution over peaks
    scaled = (b - b_max) / temp
    scaled = np.clip(scaled, -50, 0)
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(b))
        w[b.argmax()] = 1.0
    else:
        w /= w_sum
    
    # Subgradient: g[j] = 2 * sum_k w[k] * a[k-j] * I(0<=k-j<n)
    g = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-12:
            continue
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            g[j] += 2.0 * w[k] * a[k - j]
    
    return g, b_max


def subgradient_optimize(a0, time_limit, lr=0.001, momentum=0.9,
                         n_top=50, temp=0.001, kick_size=0.005,
                         max_stall=2000, seed=42):
    """Subgradient with momentum."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 0.01:
        return a0.copy(), objective_value(a0)
    a = a / S
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = objective_value(a)
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    
    while time.monotonic() - t0 < time_limit:
        g, b_max = subgradient_step(a, n_top=n_top, temp=cur_temp)
        
        # Nesterov momentum
        v = momentum * v - cur_lr * g
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        
        obj_new = objective_value(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            a = a_new  # still move
        
        # Adaptive
        if no_improve > 500:
            cur_lr *= 0.8
            cur_temp *= 1.1
        if no_improve > max_stall:
            # Random kick
            a = best_a + rng.standard_normal(n) * kick_size
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:

            v = np.zeros(n)
            cur_lr = lr
            no_improve = 0
    
    return best_a, best_obj


def coordinate_polish(a0, time_limit, n_multipliers=6):
    """Coordinate descent: scale individual elements."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_obj = objective_value(a)
    improved = True
    
    multipliers = np.array([0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0])
    
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j = old_val
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                ss = a_trial.sum()
                if ss < 0.01:
                    continue
                a_trial /= ss
                val = objective_value(a_trial)
                if val < best_j_obj - 1e-12:
                    best_j_obj = val
                    best_j = new_val
            if best_j != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j) / denom
                a[j] = best_j
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-12:
                    best_obj = best_j_obj
                    improved = True
    return a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
