import numpy as np
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_local(a):
    """Compute max of convolution (with sum(a)=1 normalization assumed)."""
    b = fast_convolve(a)
    return float(b.max())

    # = sum_j a[j] * w[m+j]

    # Actually, let me just do it directly for clarity, or use a different FFT trick

    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)

    r = np.nonzero(cond)[0][-1]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Subgradient method with Polyak step size, Nesterov momentum, and adaptive active-set
2. 2D coordinate descent on important coordinate pairs
3. L-BFGS refinement on smoothed objective (multiple temperatures)
4. Multi-length exploration with interpolation
5. Random restarts for escaping local minima
6. Final micro-polish with fine coordinate descent
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    r = np.nonzero(cond)[0][-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def subgrad_fft(a, w):
    """
    Compute subgradient of max_k f_k(a): g[m] = 2 * sum_j a[j]*w[j+m].
    Uses FFT cross-correlation.
    """
    n = len(a)
    N = 1 << (3 * n - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    corr = irfft(W * np.conj(A), N)[:n]
    return 2.0 * corr


def subgrad_direct(a, w):
    """Direct subgradient computation (efficient for sparse w)."""
    n = len(a)
    g = np.zeros(n)
    nonzero = np.where(w > 1e-12)[0]
    for k in nonzero:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi + 1] += 2.0 * w[k] * a[lo:hi + 1][::-1]
    return g


def eval_score(a):
    """Compute the true score 2n*max_b/(sum_a)^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def sg_optimize(a0, time_limit, lr_scale=0.5, momentum_init=0.9,
                n_top=30, eps_active=0.005, seed=42, verbose=False):
    """
    Subgradient optimization with Polyak step, Nesterov momentum, adaptive active set.
    Input: a0 normalized to sum=1. Returns (best_a, best_maxb).
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
        S = 1.0
    else:
        a = a / S
    v = np.zeros(n)
    best_a = a.copy()
    best_maxb = fast_convolve(a).max()
    f_lb = 1.0 / (2 * n - 1)
    momentum = momentum_init
    no_improve = 0
    it = 0
    kick_interval = 300
    kick_count = 0
    last_print = t0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()

        # Active set: peaks within eps_active of max
        threshold = max_b - max_b * eps_active
        active_mask = b >= threshold
        n_active = active_mask.sum()
        if n_active == 0:
            active_mask[b.argmax()] = True
            n_active = 1
        if n_active > n_top:
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            active_mask = np.zeros(len(b), dtype=bool)
            active_mask[top_idx] = True
            n_active = n_top

        w = np.zeros(len(b))
        w[active_mask] = 1.0 / n_active

        # Subgradient
        if n_active < 50:
            g = subgrad_direct(a, w)
        else:
            g = subgrad_fft(a, w)

        g_norm_sq = g.dot(g)
        if g_norm_sq < 1e-30:
            break

        # Polyak step size
        gap = max_b - f_lb
        alpha = lr_scale * gap / g_norm_sq
        alpha = min(alpha, 0.05)
        alpha = max(alpha, 1e-6)

        # Nesterov momentum
        v = momentum * v - alpha * g
        a_new = project_simplex(a + v)

        # Evaluate
        b_new = fast_convolve(a_new)
        max_b_new = b_new.max()

        if max_b_new < best_maxb - 1e-14:
            best_maxb = max_b_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        it += 1

        # Adaptive restart
        if no_improve > kick_interval:
            kick_size = 0.002 / (1.0 + kick_count * 0.5)
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            no_improve = 0
            kick_count += 1
            momentum = min(0.97, momentum + 0.01)
            if kick_count > 20:
                break

        # Log progress
        if verbose and time.monotonic() - last_print > 5.0:
            last_print = time.monotonic()
            print(f"    SG it={it}, max_b={max_b:.8f}, best={best_maxb:.8f}, "
                  f"score={2*n*best_maxb:.6f}, n_active={n_active}")

    return best_a, best_maxb


def cd_2d(a0, time_limit, n_top_coords=25, grid_size=8, seed=42):
    """
    2D coordinate descent: for pairs of top coordinates, try values on a grid.
    Input: a0 normalized to sum=1.
    """
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_maxb = fast_convolve(a).max()
    rng = np.random.default_rng(seed)

    # Indices of top coordinates
    top_idx = np.argsort(a)[::-1][:n_top_coords]

    improved = True
    passes = 0
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        passes += 1
        for i in range(len(top_idx)):
            for j in range(i + 1, len(top_idx)):
                if time.monotonic() - t0 > time_limit:
                    return best_a, best_maxb
                ii = top_idx[i]
                jj = top_idx[j]
                ai = a[ii]
                aj = a[jj]
                if ai < 1e-10 and aj < 1e-10:
                    continue
                # Rest sum
                rest_sum = 1.0 - ai - aj
                if rest_sum < 1e-10:
                    continue

                # Try grid of (x, y) values
                best_triplet = (ai, aj, best_maxb)
                x_candidates = np.linspace(0, ai * 3, grid_size)
                x_candidates = np.clip(x_candidates, 0, 1 - 1e-8)
                y_candidates = np.linspace(0, aj * 3, grid_size)
                y_candidates = np.clip(y_candidates, 0, 1 - 1e-8)

                for x in x_candidates:
                    for y in y_candidates:
                        s = x + y
                        if s >= 0.999 or s < 0:
                            continue
                        rest_scale = (1.0 - s) / rest_sum
                        if rest_scale < 0 or rest_scale > 10:
                            continue
                        a_trial = a.copy()
                        a_trial[ii] = x
                        a_trial[jj] = y
                        mask = np.ones(n, dtype=bool)
                        mask[ii] = False
                        mask[jj] = False
                        a_trial[mask] *= rest_scale
                        a_trial = a_trial / a_trial.sum()
                        mb = fast_convolve(a_trial).max()
                        if mb < best_triplet[2] - 1e-14:
                            best_triplet = (x, y, mb)

                if best_triplet[2] < best_maxb - 1e-14:
                    x_new, y_new, mb_new = best_triplet
                    rest_scale = (1.0 - x_new - y_new) / rest_sum
                    a[ii] = x_new
                    a[jj] = y_new
                    mask = np.ones(n, dtype=bool)
                    mask[ii] = False
                    mask[jj] = False
                    a[mask] *= rest_scale
                    a = a / a.sum()
                    best_maxb = mb_new
                    best_a = a.copy()
                    improved = True

    return best_a, best_maxb


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def soft_max_obj_grad(z, T, n):
    """Soft-max smoothed objective and gradient in log-space."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    w = exp_Tb / exp_Tb.sum()
    # Gradient with respect to a: grad_a[m] = 2 * sum_j a[j] * w[j+m]
    N = 1 << (3 * n - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    corr = irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * corr
    obj = (2.0 * n / T) * log_sum_exp
    # Convert to gradient in z-space: grad_z = a * (grad_a - a * sum(grad_a * a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z


def lbgfs_refine(a0, T, time_limit, maxiter=3000):
    """L-BFGS in log-space with soft-max smoothed objective."""
    t0 = time.monotonic()
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-300))
    obj0, _ = soft_max_obj_grad(z0, T, n)
    best_obj = obj0
    best_z = z0.copy()

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        o, _ = soft_max_obj_grad(z, T, n)
        return o

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = soft_max_obj_grad(z, T, n)
        return g

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 10, 'disp': False}
        )
        final_obj = func(res.x)
        if np.isfinite(final_obj) and final_obj < best_obj:
            best_z = res.x.copy()
            best_obj = final_obj
    except Exception:
        pass

    return softmax_from_z(best_z), best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 1e-12)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian_init(n, sigma_frac=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma_frac) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    taper = max(1, int(r * n // 2))
    t = np.linspace(0, 1, taper)
    a[:taper] = 0.5 * (1 + np.cos(np.pi * (2 * t / r - 1)))
    a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_skewed_init(n, alpha=2.0):
    """Beta-distribution shaped initialization."""
    x = np.linspace(0.01, 0.99, n)
    # Simple beta pdf approximation
    a = x ** (alpha - 1) * (1 - x) ** (alpha - 1)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

            a_opt, _ = sg_optimize(a_pert, time_limit=t_alloc,
                                    lr_scale=0.3, momentum_init=0.9,
                                    n_top=30, eps_active=0.003,
                                    seed=seed + attempt * 7919)
            try_seq(a_opt, f"restart_{attempt}")

    # === Phase 8: Final micro-polish ===
    print(f"\n=== Phase 8: Final polish (score={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        # Fine subgradient
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t_alloc = min(rem * 0.5, 60.0)
            a_fin, _ = sg_optimize(best_seq, time_limit=t_alloc,
                                    lr_scale=0.2, momentum_init=0.95,
                                    n_top=100, eps_active=0.001,
                                    seed=seed + 99999)
            try_seq(a_fin, "fin_sg")

        # Final 2D CD
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            t_cd = min(rem * 0.6, 30.0)
            a_cd_final, _ = cd_2d(best_seq, time_limit=t_cd,
                                    n_top_coords=15, grid_size=10,
                                    seed=seed + 88888)
            try_seq(a_cd_final, "fin_cd")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"Total time: {time.monotonic() - t_start:.1f}s")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
