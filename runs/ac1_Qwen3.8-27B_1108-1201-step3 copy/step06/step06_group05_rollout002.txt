                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                c = fast_convolve(f)
                M = float(c.max())
                A_prebuilt = build_constraint_matrix(f, n)
                continue
            M = M_try
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            continue
        g = g / g_sum
        
        # Check fixed point
        fw_gap = np.max(np.abs(g - f))
        if fw_gap < 1e-10:
            # Try support reduction
            prune_counter += 1
            if prune_counter > 20:
                # Prune small coordinates
                threshold = np.median(f[f > 0]) * 0.1
                mask = f > threshold
                if mask.sum() < n * 0.3:
                    mask = f > np.sort(f)[int(n * 0.3)]
                if mask.sum() > 10:
                    f_pruned = np.zeros(n)
                    f_pruned[mask] = f[mask]
                    f_pruned /= f_pruned.sum()
                    c_new = fast_convolve(f_pruned)
                    obj_new = float(c_new.max())
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_f = f_pruned.copy()
                        f = f_pruned.copy()
                        M = obj_new
                        A_prebuilt = build_constraint_matrix(f, n)
                        prune_counter = 0
                else:
                    # Random restart
                    rng = np.random.default_rng(it * 17 + 3)
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                    c = fast_convolve(f)
                    M = float(c.max())
                    A_prebuilt = build_constraint_matrix(f, n)
                    prune_counter = 0
            continue
        
        # Golden section line search
        def ls_eval(alpha):
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                return 1e10
            ft = ft / st
            return float(fast_convolve(ft).max())
        
        # We can't define a closure, so inline the search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo_a = 0.0
        hi_a = 1.0
        c_a = hi_a - gr * (hi_a - lo_a)
        d_a = lo_a + gr * (hi_a - lo_a)
        
        ft_c = f + c_a * (g - f)
        ft_c = np.maximum(ft_c, 0.0)
        ft_c = ft_c / ft_c.sum()
        fc = float(fast_convolve(ft_c).max())
        
        ft_d = f + d_a * (g - f)
        ft_d = np.maximum(ft_d, 0.0)
        ft_d = ft_d / ft_d.sum()
        fd = float(fast_convolve(ft_d).max())
        
        best_alpha = c_a if fc < fd else d_a
        best_val = min(fc, fd)
        
        for _ in range(30):
            if fc < fd:
                hi_a = d_a
                d_a = c_a
                fd = fc
                c_a = hi_a - gr * (hi_a - lo_a)
                ft_c = f + c_a * (g - f)
                ft_c = np.maximum(ft_c, 0.0)
                ft_c = ft_c / ft_c.sum()
                fc = float(fast_convolve(ft_c).max())
            else:
                lo_a = c_a
                c_a = d_a
                fc = fd
                d_a = lo_a + gr * (hi_a - lo_a)
                ft_d = f + d_a * (g - f)
                ft_d = np.maximum(ft_d, 0.0)
                ft_d = ft_d / ft_d.sum()
                fd = float(fast_convolve(ft_d).max())
            if fc < best_val:
                best_val = fc
                best_alpha = c_a
            if fd < best_val:
                best_val = fd
                best_alpha = d_a
        
        if best_val < M - 1e-14:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            c_new = fast_convolve(f_new)
            M_new = float(c_new.max())
            f = f_new
            M = M_new
            A_prebuilt = build_constraint_matrix(f, n)
            
            if M_new < best_obj - 1e-14:
                best_obj = M_new
                best_f = f.copy()
                prune_counter = 0
                if verbose and it % 50 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.12f}, gap={fw_gap:.2e}")
        else:
            prune_counter += 1
        
        if it % 200 == 0 and verbose:
            print(f"    LP-FW iter {it}: obj={best_obj:.12f}, time={elapsed:.0f}s")
    
    return best_f, best_obj


def subgradient_optimize(f0, time_limit, verbose=False):
    """
    Subgradient method with averaging for the max-convolution problem.
    Uses the exact subgradient of max(a*a).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = float(fast_convolve(f).max())
    
    t0 = time.monotonic()
    T = 0
    f_avg = f.copy()
    
    while time.monotonic() - t0 < time_limit:
        T += 1
        c = fast_convolve(f)
        M = float(c.max())
        
        # Find active set (positions within epsilon of max)
        eps = max(1e-8, M * 0.01)
        active = c >= (M - eps)
        
        # Subgradient: s_j = 2 * sum_{k in active} f[k-j] * [0 <= k-j < n]
        # This is 2 * (f * active_indicator)[reversed and truncated]
        # More precisely: s_j = 2 * sum_k active[k] * f[k-j] for valid k-j
        # = 2 * (convolve(f, active))[j + n - 1] ... let me think
        # conv(f, active)[m] = sum_j f[j] * active[m-j]
        # We want s_j/2 = sum_k active[k] * f[k-j] = sum_k f[k-j] * active[k]
        # Let i = k-j, so k = i+j: sum_i f[i] * active[i+j]
        # = conv(f, active_rev)[?] ... 
        # Actually: cross-correlation of f with active at lag j
        # = sum_i f[i] * active[i+j]
        # This is the (j)-th element of the cross-correlation
        # Using FFT: correlate(f, active) = ifft(conj(FFT(f)) * FFT(active))
        
        fft_size = 1 << (2 * n - 2).bit_length()  # need 2n-1 points
        F = np.fft.rfft(f, fft_size)
        A = np.fft.rfft(active.astype(float), fft_size)
        corr = np.fft.irfft(np.conj(F) * A, fft_size)
        # corr[j] = sum_i f[i] * active[(i+j) mod fft_size]
        # For j = 0..n-1, we want corr[j] (linear, not circular, since fft_size >= 2n-1)
        s_j = 2.0 * corr[:n]
        
        # Step size: Polya-Shor: lr = 2 * (f0-f*) / (L^2 * T)
        # Approximate: lr = c / sqrt(T)
        lr = 1.0 / np.sqrt(T)
        
        # Projected subgradient step
        f_new = project_simplex(f - lr * s_j)
        
        # Running average
        f_avg = f_avg * (T - 1) / T + f_new / T
        
        # Check progress
        obj_new = float(fast_convolve(f_new).max())
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
        
        obj_avg = float(fast_convolve(f_avg).max())
        if obj_avg < best_obj - 1e-12:
            best_obj = obj_avg
            best_f = f_avg.copy()
        
        f = f_new
        
        if verbose and T % 500 == 0:
            print(f"    SG iter {T}: obj={obj_new:.10f}, avg={obj_avg:.10f}, best={best_obj:.10f}")
    
    return best_f, best_obj


def peak_shaving(f0, time_limit, verbose=False):
    """
    Direct peak shaving: identify the top peaks and redistribute mass
    from contributing elements to reduce them.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = float(fast_convolve(f).max())
    
    t0 = time.monotonic()
    it = 0
    lr = 0.02
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(f)
        M = float(c.max())
        
        # Find top K peaks
        K = min(10, 2 * n - 1)
        peak_indices = np.argsort(c)[-K:]
        
        # For each peak, compute which f elements contribute most
        # and reduce them
        reduction = np.zeros(n)
        for k in peak_indices:
            k = int(k)
            weight = (c[k] - M * 0.8)  # excess above 80% of max
            if weight <= 0:
                weight = 0.1
            # Contributing elements: f[j] where j in [max(0,k-n+1), min(n-1,k)]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            # Contribution of f[j] to peak k is f[j] * f[k-j]
            for j in range(js, je + 1):
                contribution = f[j] * f[k - j]
                # Reduce f[j] proportionally to its contribution
                reduction[j] += weight * f[k - j]
        
        # Normalize reduction
        red_norm = np.linalg.norm(reduction)
        if red_norm < 1e-15:
            break
        reduction = reduction / red_norm
        
        # Apply with adaptive step
        f_new = f - lr * reduction
        f_new = np.maximum(f_new, 0.0)
        sn = f_new.sum()
        if sn < 1e-15:
            break
        f_new = f_new / sn
        
        obj_new = float(fast_convolve(f_new).max())
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            lr = min(lr * 1.01, 0.05)
        else:
            lr *= 0.95
            if lr < 1e-6:
                # Restart
                rng = np.random.default_rng(it * 11 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                lr = 0.02
        
        if verbose and it % 100 == 0:
            print(f"    PS iter {it}: obj={obj_new:.10f}, best={best_obj:.10f}")
    
    return best_f, best_obj


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - (n - 1) / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_raised_cosine(n):
    x = np.linspace(-1, 1, n)
    a = 0.5 * (1.0 + np.cos(np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_uniform(n):
    return np.ones(n) / n


def interpolate_sequence(f, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    return f_interp


def coordinate_descent_polish(f0, time_limit, verbose=False):
    """
    Coordinate descent: try adjusting each coordinate (or small groups)
    to reduce the max convolution.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f0, float('inf')
    f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = float(fast_convolve(f).max())
    t0 = time.monotonic()
    
    # Only polish coordinates with significant mass
    active_coords = np.where(f > f.max() * 0.01)[0]
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        for j in active_coords:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-10:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            # Try multipliers
            for mult in [0.6, 0.7, 0.8, 0.9, 1.1, 1.2, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                # Redistribute: reduce j, increase others proportionally
                f_trial = f.copy()
                f_trial[j] = new_val
                # Scale others to maintain sum = 1
                others_mask = np.ones(n, dtype=bool)
                others_mask[j] = False
                other_sum = f_trial[others_mask].sum()
                if other_sum < 1e-15:
                    continue
                scale = (1.0 - new_val) / other_sum
                f_trial[others_mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                
                val = float(fast_convolve(f_trial).max())
                if val < best_j_obj - 1e-13:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                f[j] = best_j_val
                others_mask = np.ones(n, dtype=bool)
                others_mask[j] = False
                other_sum = f[others_mask].sum()
                if other_sum > 1e-15:
                    scale = (1.0 - best_j_val) / other_sum
                    f[others_mask] *= scale
                f = np.maximum(f, 0)
                f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            # Try adding mass to currently-zero coordinates
            zero_coords = np.where(f < 1e-10)[0]
            if len(zero_coords) == 0:
                break
            any_improved = False
            for j in zero_coords[:50]:  # Limit to 50
                if time.monotonic() - t0 > time_limit:
                    break
                for frac in [0.001, 0.005, 0.01]:
                    f_trial = f.copy()
                    f_trial[j] = frac
                    others_mask = np.ones(n, dtype=bool)
                    others_mask[j] = False
                    f_trial[others_mask] *= (1.0 - frac)
                    ss = f_trial.sum()
                    if ss < 1e-15:
                        continue
                    f_trial = f_trial / ss
                    val = float(fast_convolve(f_trial).max())
                    if val < best_obj - 1e-13:
                        f = f_trial
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        any_improved = True
                        break
            if not any_improved:
                break
    
    return best_f, best_obj
