def simplex_project(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1] if cond.any() else n-1
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lse_smoothed_obj(a, T):
    """Log-sum-exp smoothed objective."""
    n = len(a)
    c = fast_convolve(a)
    max_c = c.max()
    exp_c = np.exp((c - max_c) / T)
    lse = max_c + T * np.log(exp_c.sum())
    return 2.0 * n * lse

    """Gradient of the LSE-smoothed objective."""
    n = len(a)

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_subgradient(a, c=None):
    """
    Compute a subgradient of max_k c[k] where c = conv(a,a).
    Uses FFT for O(n log n) computation.
    Returns (subgradient_vector, peak_value, c).
    """
    n = len(a)
    if c is None:
        c = fast_convolve(a)
    peak_val = c.max()
    tol = max(1e-14, 1e-10 * peak_val)
    peaks = (c >= peak_val - tol).astype(np.float64)
    s = peaks.sum()
    if s > 0:
        peaks /= s
    else:
        peaks = np.zeros(2 * n - 1)
        peaks[int(np.argmax(c))] = 1.0

    # sg[i] = 2 * sum_k peaks[k] * a[k-i] (with a[j]=0 for j outside [0,n-1])
    a_rev = a[::-1]
    size = (2 * n - 1) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(peaks, fft_size)
    AR = rfft(a_rev, fft_size)
    conv_wr = irfft(W * AR, fft_size)[:size]
    sg = 2.0 * conv_wr[n - 1:n - 1 + n]
    return sg, peak_val, c


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


def lse_objective(a, T):
    """Log-sum-exp smoothed objective: 2n * LSE(c/T) where c = conv(a,a)."""
    n = len(a)
    c = fast_convolve(a)
    max_c = c.max()
    exp_c = np.exp((c - max_c) / T)
    lse = max_c + T * np.log(exp_c.sum())
    return 2.0 * n * lse


def lse_gradient(a, T):
    """Gradient of the LSE-smoothed objective (projected onto simplex tangent)."""
    n = len(a)
    c = fast_convolve(a)
    max_c = c.max()
    exp_c = np.exp((c - max_c) / T)
    w = exp_c / exp_c.sum()

    # Gradient of 2n * LSE = 2n * sum_k w[k] * dc[k]/da
    # = 2n * 2 * sum_k w[k] * shift(a, k)
    # = 4n * conv(w, reverse(a))[n-1:n-1+n]
    a_rev = a[::-1]
    size = (2 * n - 1) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv_wr = irfft(W * AR, fft_size)[:size]
    sg = 4.0 * n * conv_wr[n - 1:n - 1 + n]

    # Project onto tangent space of simplex (remove component in direction of 1)
    sg = sg - np.mean(sg)
    obj = 2.0 * n * (max_c + T * np.log(exp_c.sum()))
    return obj, sg

            if time.monotonic() - t0 > time_limit:

                new_val = a[i] * factor

                score = 2.0 * n * fast_convolve(a_trial).max()

                for ratio in [0.01, 0.02, 0.05, 0.1]:
                    amt = a[top_i] * ratio
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[top_i] -= amt
                    a_trial[target_i] += amt
                    if a_trial[top_i] < 0 or a_trial[target_i] > 1:
                        continue
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break

        # Random perturbation if stuck for a while
        if not improved and rnd % 50 == 49:
            rng = np.random.default_rng(rnd * 7919 + n)
            i = rng.integers(n)
            for factor in [0.99, 0.98, 0.97, 1.01, 1.02, 1.03]:
                new_val = a[i] * factor
                diff = new_val - a[i]
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                osum = others.sum()
                if osum < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others - diff * others / osum
                if (a_trial >= 0).all():
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def safe_score(a):
    """Compute score directly (avoids calling evaluate_sequence for speed)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_norm = a / s
    n = len(a_norm)
    c = fast_convolve(a_norm)
    return 2.0 * n * c.max()


def full_pipeline(a0, time_limit):
    """Run the full optimization pipeline on a given starting point."""
    n = len(a0)
    a = simplex_project(a0.astype(np.float64))
    a = a / max(a.sum(), 1e-15)
    best_a = a.copy()
    best_score = safe_score(a)
    t0 = time.monotonic()

    # Phase A: L-BFGS with multiple temperatures
    temps = [5.0, 15.0, 50.0, 100.0, 200.0]
    remaining = time_limit - (time.monotonic() - t0)
    t_per_T = remaining * 0.25 / len(temps)

    for T in temps:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        t_alloc = min(t_per_T, rem * 0.9)
        a_lse, score_lse = optimize_lse_lbgfs(best_a, T, t_alloc)
        if score_lse < best_score:
            best_score = score_lse
            best_a = a_lse
            print(f"      LSE T={T:.0f}: score={best_score:.7f}")

    # Phase B: Subgradient momentum
    rem = time_limit - (time.monotonic() - t0)
    if rem > 10:
        t_alloc = rem * 0.3
        a_sg, score_sg = subgradient_momentum(best_a, t_alloc,
                                              lr_init=0.003, momentum=0.85)
        if score_sg < best_score:
            best_score = score_sg
            best_a = a_sg
            print(f"      SGD-momentum: score={best_score:.7f}")

    # Phase C: Coordinate descent
    rem = time_limit - (time.monotonic() - t0)
    if rem > 5:
        t_alloc = rem * 0.3
        a_cd, score_cd = coordinate_descent(best_a, t_alloc)
        if score_cd < best_score:
            best_score = score_cd
            best_a = a_cd
            print(f"      CoordDescent: score={best_score:.7f}")

    # Phase D: Another round of L-BFGS with the best found so far
    rem = time.limit = time_limit - (time.monotonic() - t0)
    if rem > 5:
        for T in [10.0, 30.0]:
            rem = time_limit - (time.monotonic() - t0)
            if rem < 3:
                break
            t_alloc = min(rem * 0.4, 30.0)
            a_lse2, score_lse2 = optimize_lse_lbgfs(best_a, T, t_alloc,
                                                     maxiter=5000)
            if score_lse2 < best_score:
                best_score = score_lse2
                best_a = a_lse2
                print(f"      LSE2 T={T:.0f}: score={best_score:.7f}")

    # Phase E: Final coordinate descent
    rem = time_limit - (time.monotonic() - t0)
    if rem > 3:
        a_cd2, score_cd2 = coordinate_descent(best_a, rem * 0.7)
        if score_cd2 < best_score:
            best_score = score_cd2
            best_a = a_cd2

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main entry point. Multi-phase optimization:
    1. Initialize from parent or structured start
    2. L-BFGS with LSE smoothing at multiple temperatures
    3. Subgradient momentum with exact objective
    4. Coordinate descent polishing
    5. Multi-length exploration
    6. Final verification
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = safe_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        # Create a structured initial sequence
        n_init = 768
        t = np.linspace(0, 2 * np.pi, n_init, endpoint=False)
        a_init = 1.0 + 0.3 * np.cos(t) + 0.1 * np.cos(2 * t + 1.0)
        a_init = np.maximum(a_init, 0.0)
        a_init = a_init / a_init.sum()
        best_seq = a_init
        best_score = safe_score(a_init)
        print(f"Initial (no parent): n={n_init}, score={best_score:.7f}")

    # === Phase 1: Main pipeline on parent length ===
    n_parent = len(best_seq)
    print(f"\n=== Phase 1: Main pipeline n={n_parent} ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_alloc = rem * 0.45
        a_new, score_new = full_pipeline(best_seq, t_alloc)
        if score_new < best_score:
            best_score = score_new
            best_seq = a_new
            print(f"  Phase 1 result: {best_score:.7f}")

    # === Phase 2: Try different lengths ===
    print(f"\n=== Phase 2: Multi-length exploration (best={best_score:.7f}) ===")
    n_best = len(best_seq)
    alt_lengths = []
    for factor in [0.875, 1.0, 1.125, 1.25, 1.5]:
        alt_n = int(n_best * factor)
        if 256 <= alt_n <= 2048 and alt_n not in alt_lengths:
            alt_lengths.append(alt_n)

    rem = deadline - time.monotonic()
    time_per_alt = rem * 0.25 / max(1, len(alt_lengths))

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        t_alloc = min(time_per_alt, rem * 0.85)
        if t_alloc < 10:
            continue

        print(f"  Trying n={alt_n} (budget={t_alloc:.0f}s)")
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt, score_alt = full_pipeline(init_alt, t_alloc)
        if score_alt < best_score:
            best_score = score_alt
            best_seq = a_alt
            n_best = alt_n
            print(f"  NEW BEST: n={alt_n}, score={best_score:.7f}")

    # === Phase 3: Random restarts at best length ===
    print(f"\n=== Phase 3: Random restarts (best={best_score:.7f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic()
    n_restarts = min(5, int(rem / 60))
    rng = np.random.default_rng(seed)

    for r in range(n_restarts):
        rem = deadline - time.monotonic()
        if rem < 15:
            break
        t_alloc = min(rem * 0.15, 60.0)

        # Perturb the best solution
        pert_scale = 0.01 * (r + 1)
        a_pert = best_seq + rng.standard_normal(n_best) * pert_scale
        a_pert = simplex_project(a_pert)
        a_pert = a_pert / a_pert.sum()

        a_new, score_new = full_pipeline(a_pert, t_alloc)
        if score_new < best_score:
            best_score = score_new
            best_seq = a_new
            print(f"  Restart {r}: NEW BEST {best_score:.7f}")

    # === Phase 4: Final intensive polish ===
    print(f"\n=== Phase 4: Final polish (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        # L-BFGS with very low temperature (almost exact)
        t_alloc = min(rem * 0.25, 40.0)
        a_polish, score_polish = optimize_lse_lbgfs(best_seq, 3.0, t_alloc,
                                                     maxiter=20000)
        if score_polish < best_score:
            best_score = score_polish
            best_seq = a_polish
            print(f"  Final LSE: {best_score:.7f}")

        # Subgradient with very small step
        rem = deadline - time.monotonic()
        if rem > 5:
            t_alloc = min(rem * 0.3, 30.0)
            a_sg, score_sg = subgradient_momentum(best_seq, t_alloc,
                                                  lr_init=0.001, momentum=0.8)
            if score_sg < best_score:
                best_score = score_sg
                best_seq = a_sg
                print(f"  Final SGD: {best_score:.7f}")

        # Coordinate descent
        rem = deadline - time.monotonic()
        if rem > 3:
            a_cd, score_cd = coordinate_descent(best_seq, rem * 0.6)
            if score_cd < best_score:
                best_score = score_cd
                best_seq = a_cd
                print(f"  Final CD: {best_score:.7f}")

    # === Phase 5: Verify with evaluate_sequence ===
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    # If something went wrong, fall back
    if not np.isfinite(best_score):
        if parent is not None:
            best_seq = parent
            best_score = float(evaluate_sequence(best_seq.tolist()))
        else:
            best_seq = np.ones(512) / 512
            best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL: n={len(best_seq)}, score={best_score:.7f}")
    return [float(x) for x in best_seq]
