"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_obj(a):
    """Compute 2*n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def coordinate_descent(a, time_limit=60.0, verbose=False):
    """Efficient coordinate descent with exact 1D optimization per coordinate."""
    a = a.astype(np.float64).copy()
    n = len(a)
    conv = fast_convolve(a)
    S = a.sum()
    best_obj = 2.0 * n * conv.max() / (S * S)
    best_a = a.copy()
    t0 = time.monotonic()
    sweep = 0
    no_improve_count = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False

        # Randomize coordinate order after first sweep for better mixing
        if sweep == 1:
            order = np.arange(n)
        else:
            order = np.random.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-15:
                # Try turning on a zero coordinate
                for trial_val in [1.0 / n, 2.0 / n, 5.0 / n]:
                    a[j] = trial_val
                    new_conv = fast_convolve(a)
                    new_S = a.sum()
                    obj = 2.0 * n * new_conv.max() / (new_S * new_S)
                    if obj < best_obj - 1e-13:
                        conv = new_conv
                        S = new_S
                        best_obj = obj
                        best_a = a.copy()
                        improved = True
                    else:
                        a[j] = 0.0
                continue

            # Affected lags for coordinate j
            k_min = j
            k_max = min(j + n - 1, 2 * n - 2)
            affected_ks = np.arange(k_min, k_max + 1)

            # Compute max of unaffected lags
            # Fast path: if n is large, the unaffected set is most of the array
            if k_min == 0 and k_max == 2 * n - 2:
                unaff_max = -1e30
            else:
                # Find max excluding affected range
                unaff_max = -1e30
                if k_min > 0:
                    unaff_max = max(unaff_max, conv[:k_min].max())
                if k_max < 2 * n - 2:
                    unaff_max = max(unaff_max, conv[k_max + 1:].max())

            # Current objective
            cur_max = conv.max()
            cur_obj = 2.0 * n * cur_max / (S * S)

            # Generate candidate t values (change to a[j])
            fracs = np.array([-0.9, -0.7, -0.5, -0.3, -0.2, -0.1, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0])
            candidates = old_val * fracs
            candidates = np.concatenate([candidates, [0.0]])
            # Boundary
            candidates = np.append(candidates, -old_val + 1e-12)
            # Candidates that zero out the top affected peaks
            top_affected_idx = affected_ks[np.argsort(conv[affected_ks])[-3:]]
            for k in top_affected_idx:
                if k == 2 * j:
                    # conv[k] + 2*old_val*t + t^2 = 0
                    disc = old_val * old_val - conv[k]
                    if disc >= 0:

                else:
                    partner = k - j
                    if 0 <= partner < n and a[partner] > 1e-15:
                        candidates = np.append(candidates, -conv[k] / (2.0 * a[partner]))

            best_t = 0.0
            best_trial_obj = cur_obj

            for t in candidates:

                if t < -old_val + 1e-11:
                    continue
                new_val = old_val + t
                if new_val < 0:
                    continue
                S_new = S + t
                if S_new < 1e-15:
                    continue

                # Compute new max: max of unaff_max and new affected values
                new_affected_max = -1e30
                for k in affected_ks:

                    else:

def subgradient_descent(a, time_limit=60.0, verbose=False):
    """Subgradient descent with Polyak step and running average."""
    a = a.astype(np.float64).copy()
    n = len(a)
    conv = fast_convolve(a)
    S = a.sum()
    best_obj = 2.0 * n * conv.max() / (S * S)
    best_a = a.copy()
    t0 = time.monotonic()

    # Best known upper bound on the normalized problem (sum=1): minimize n*max(conv)
    S_norm = 1.0
    a_norm = a / S
    conv_norm = fast_convolve(a_norm)
    best_norm_obj = n * conv_norm.max()  # This is best_obj / 2
    lower_bound = 1.0 / (2 * n) * 2 * n  # trivial: n * (1/n) = 1 (from Cauchy)
    # Actually lower bound for n*max(conv) with sum=1 is 1 (uniform gives exactly 1)
    # Wait: uniform gives max(conv) = 1/n, so n*max(conv) = 1. And J = 2*1 = 2.
    # So lower bound on best_norm_obj is 1.

    step_size = 0.01
    avg_a = a_norm.copy().astype(np.float64)
    count = 0
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1.0:
            break

        conv = fast_convolve(a_norm)
        M = conv.max()
        cur_obj = n * M
        S_check = a_norm.sum()

        # Polyak step size
        if best_norm_obj > 1.0:
            polyak_step = (cur_obj - 1.0) / (M + 1e-10)
            step = min(step_size, max(polyak_step, 0.001))
        else:
            step = step_size

        # Subgradient of max(conv) with respect to a_norm
        # d(conv[k])/da[i] = a[k-i] (if valid) + a[k-i] (if k-i != i, i.e., factor of 2)
        # Actually: conv[k] = sum_i a[i]*a[k-i]
        # d(conv[k])/da[j] = a[k-j] + a[k-j] (if k-j != j) or a[j] (if k-j = j, i.e., k=2j)
        # More precisely: d(conv[k])/da[j] = a[k-j] if 0<=k-j<n, plus a[k-j] if k-j != j and 0<=k-j<n
        # = 2*a[k-j] if k != 2j and 0<=k-j<n, or a[j] if k=2j

        # Find all peak lags (within epsilon of max)
        peak_tol = 1e-10
        peak_ks = np.where(conv >= M - peak_tol)[0]
        if len(peak_ks) == 0:
            peak_ks = np.array([np.argmax(conv)])

        # Weighted subgradient (uniform over peaks)
        w = 1.0 / len(peak_ks)
        sg = np.zeros(n)
        for k in peak_ks:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                if k == 2 * j:
                    sg[j] += w * a_norm[j]
                else:
                    sg[j] += w * 2.0 * a_norm[k - j]

        # Update: move in negative subgradient direction
        a_norm = a_norm - step * sg
        a_norm = np.maximum(a_norm, 0.0)
        S_new = a_norm.sum()
        if S_new < 1e-15:
            a_norm = a / S
        else:
            a_norm = a_norm / S_new

        # Track running average (for the averaged sequence)
        avg_a = avg_a * (count / (count + 1)) + a_norm * (1.0 / (count + 1))
        count += 1

        # Evaluate
        conv_check = fast_convolve(avg_a)
        avg_obj = n * conv_check.max()
        if avg_obj < best_norm_obj - 1e-10:
            best_norm_obj = avg_obj
            best_a = avg_a.copy()
            best_obj = 2.0 * best_norm_obj
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    SG iter {it}: obj={best_obj:.10f}, step={step:.6f}")
        else:
            no_improve += 1

        # Decay step size
        step_size *= 0.999
        if step_size < 1e-6:
            break

        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(it * 7 + 3)
            a_norm = np.maximum(best_a + rng.standard_normal(n) * 0.003, 0.0)
            a_norm = a_norm / a_norm.sum()
            step_size = 0.01
            no_improve = 0

    return best_a, best_obj


def lp_fw_optimize(a, time_limit=60.0, verbose=False):
    """LP-Frank-Wolfe with multi-level M relaxation."""
    a = a.astype(np.float64).copy()
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = n * M
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    m_levels = [1.0, 1.02, 1.05, 1.1, 1.2, 1.5, 2.0, 3.0]
    m_idx = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2.0:
            break

        conv = fast_convolve(a)
        M_current = conv.max()

        # Solve LP: max sum(g) s.t. (a*g)[k] <= M_level for all k, g >= 0
        g = None
        for level in range(m_idx, len(m_levels)):
            M_level = M_current * m_levels[level]
            c = -np.ones(n)
            A_ub = np.zeros((2 * n - 1, n))
            for j in range(n):
                A_ub[j:j + n, j] = a
            b_ub = np.full(2 * n - 1, M_level)
            bounds = [(0, None)] * n
            try:
                res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                              options={'time_limit': rem * 0.5})
                if res.success and res.x is not None and res.x.sum() > 1e-12:
                    g = res.x
                    m_idx = level
                    break
            except Exception:
                pass
            m_idx = level + 1

        if g is None or g.sum() < 1e-12:
            # Perturb and retry
            rng = np.random.default_rng(it * 13 + 7)
            a = np.maximum(a + rng.standard_normal(n) * 0.002, 0.0)
            a = a / a.sum()
            m_idx = 0
            no_improve += 1
            if no_improve > 40:
                break
            continue

        g = g / g.sum()

        # Line search
        best_alpha = 0.0
        best_val = n * M_current
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            at = np.maximum(a + alpha * (g - a), 0.0)
            st = at.sum()
            if st < 1e-15:
                continue
            at = at / st
            val = n * fast_convolve(at).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha < 1e-8:
            no_improve += 1
            m_idx = min(m_idx + 1, len(m_levels) - 1)
            if no_improve > 30:
                break
            continue

        no_improve = 0
        a_new = np.maximum(a + best_alpha * (g - a), 0.0)
        a_new = a_new / a_new.sum()
        a = a_new
        m_idx = 0

        obj = n * fast_convolve(a).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            if verbose and it % 10 == 0:
                print(f"    LP-FW iter {it}: obj={2*best_obj:.10f}")

    return best_a, 2.0 * best_obj


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine_mod(n, freq, amp):
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_multi_cosine(n, n_freqs=3, max_amp=0.3, seed=0):
    rng = np.random.default_rng(seed)
    a = np.ones(n)
    for m in range(1, n_freqs + 1):
        amp = rng.uniform(0.05, max_amp)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * m * np.arange(n) / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def perturb_sequence(a, seed=0, scale=0.005):
    rng = np.random.default_rng(seed)
    perturbed = a + rng.standard_normal(len(a)) * scale
    perturbed = np.maximum(perturbed, 0)
    s = perturbed.sum()
    if s > 1e-15:
        perturbed /= s
    return perturbed


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: coordinate descent, subgradient, LP-FW,
    multi-start, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    def track(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            old = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) [was {old:.8f}]")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = track(parent, "parent")
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768, 0.25)
        track(parent, "default")

    n_parent = len(parent)
    print(f"Starting with n={n_parent}, best_score={best_score:.8f}")

    # === Phase 1: Quick coordinate descent on parent ===
    print("\n=== Phase 1: Coordinate Descent on Parent ===")
    rem = deadline - time.monotonic() - 800
    t_alloc = max(30, min(rem, 200))
    print(f"  Allocating {t_alloc:.0f}s")
    a_opt, obj = coordinate_descent(parent, time_limit=t_alloc, verbose=True)
    track(a_opt, "cd_parent")

    # === Phase 2: LP-FW refinement ===
    print("\n=== Phase 2: LP-FW Refinement ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 600
        t_alloc = max(20, min(rem, 200))
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
        track(a_opt, "lpfw")

    # === Phase 3: More coordinate descent ===
    print("\n=== Phase 3: Coordinate Descent Round 2 ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        t_alloc = max(20, min(rem, 200))
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = coordinate_descent(best_seq, time_limit=t_alloc, verbose=True)
        track(a_opt, "cd2")

    # === Phase 4: Subgradient polishing ===
    print("\n=== Phase 4: Subgradient Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        t_alloc = max(15, min(rem, 120))
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = subgradient_descent(best_seq, time_limit=t_alloc, verbose=True)
        track(a_opt, "sgd")

    # === Phase 5: Multi-start with perturbations ===
    print("\n=== Phase 5: Multi-Start Perturbations ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        n_starts = min(8, max(3, int(rem / 40)))
        print(f"  Trying {n_starts} perturbations")

        for i in range(n_starts):
            rem = deadline - time.monotonic() - 20
            if rem < 20:
                break
            perturb_scale = 0.003 * (1 + i * 0.5)
            a_pert = perturb_sequence(best_seq, seed=seed + 100 + i, scale=perturb_scale)
            sc = track(a_pert, f"pert{i}")

            # Quick CD on perturbation
            t_alloc = max(10, min(rem - 5, 30))
            if t_alloc > 10:
                a_opt, obj = coordinate_descent(a_pert, time_limit=t_alloc)
                track(a_opt, f"pert_cd{i}")

    # === Phase 6: Multi-length exploration ===
    print("\n=== Phase 6: Multi-Length Exploration ===")
    if best_seq is not None:
        base_n = len(best_seq)
        target_lengths = [base_n + 64, base_n - 64, base_n + 128, base_n - 128,
                          1024, 896, 640]
        target_lengths = [t for t in target_lengths if 256 <= t <= 2048]
        # Remove duplicates and sort by proximity to base_n
        seen = set()
        unique_targets = []
        for t in sorted(target_lengths, key=lambda x: abs(x - base_n)):
            if t not in seen:
                seen.add(t)
                unique_targets.append(t)

        for target_n in unique_targets[:5]:
            rem = deadline - time.monotonic() - 50
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = track(f_interp, f"interp_n{target_n}")

            if sc < best_score + 0.01:
                t_alloc = max(10, min(rem - 5, 60))
                if t_alloc > 10:
                    a_opt, obj = coordinate_descent(f_interp, time_limit=t_alloc)
                    track(a_opt, f"cd_n{target_n}")

    # === Phase 7: Final LP-FW squeeze ===
    print("\n=== Phase 7: Final LP-FW ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_alloc = min(rem, 60)
            print(f"  Allocating {t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            track(a_opt, "lpfw_final")

    # === Phase 8: Final coordinate descent polish ===
    print("\n=== Phase 8: Final CD Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t_alloc = min(rem, 30)
            print(f"  Allocating {t_alloc:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_alloc)
            track(a_opt, "cd_final")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
            print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK: score={best_score:.8f}")

    print(f"Total time: {time.monotonic() - t_start:.1f}s")
    return [float(x) for x in best_seq]
