"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-stage approach:
1. Asymmetric initialization with diverse patterns
2. Smoothed Mirror Descent (SMD) with adaptive log-sum-exp temperature
3. L-BFGS optimization on log-parameters (softmax reparameterization)
4. Peak-targeted coordinate descent
5. LP-Frank-Wolfe refinement
6. Multi-length exploration with interpolation
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time

TOTAL_TIME = 1800

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_score_conv(conv, n):
    """Compute 2*n*max(conv) for normalized sequence (sum=1)."""
    return 2.0 * n * conv.max()

def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - np.max(x)
    e = np.exp(x_shifted)
    return e / e.sum()

def log_param_grad(a, tau):
    """
    Compute gradient of smooth_max(conv(softmax(x))) w.r.t. x,
    where a = softmax(x). Returns gradient in a-space (chain rule: 
    grad_x = a * (grad_a - sum(a * grad_a)) for softmax reparam).
    
    Actually, let me just compute the gradient w.r.t. a directly,
    and the caller will handle the reparameterization.
    """
    n = len(a)
    conv = fast_convolve(a)
    # Softmax weights on conv peaks
    v = tau * conv
    v_max = np.max(v)
    w = np.exp(v - v_max)
    w /= w.sum()
    
    # Gradient w.r.t. a
    grad_a = np.zeros(n)
    # For efficiency, only process non-negligible weights
    threshold = 1e-10
    for k in range(len(conv)):
        if w[k] < threshold:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[js:je+1] += w[k] * (a[k - idx] + a[idx])
    return grad_a

def smooth_mirror_descent(f0, time_limit, tau_init=1.0, tau_max=50.0,
                          lr_init=0.02, lr_decay=0.9995,
                          tau_growth=1.02, restart_every=150, verbose=False):
    """
    Smoothed mirror descent with adaptive temperature.
    Work on the simplex directly (a >= 0, sum(a) = 1).
    """
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    tau = tau_init
    lr = lr_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Compute gradient via log-sum-exp smoothing
        k_top = max(5, int(len(conv) * 0.1))
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = tau * conv[peaks]
        pv_shifted = pv - np.max(pv)
        w = np.exp(pv_shifted)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            idx = np.arange(js, je + 1)
            sg[js:je+1] += w[i] * (f[k - idx] + f[idx])
        
        # Multiplicative step
        f_new = f * np.exp(-lr * sg)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()
        
        # Line search
        obj_new = fast_convolve(f_new).max()
        if obj_new > M * 1.01:
            # Dampen
            found = False
            for damp in [0.5, 0.25, 0.1, 0.05]:
                f_trial = f * (1.0 - damp) + f_new * damp
                f_trial = np.maximum(f_trial, 1e-15)
                f_trial = f_trial / f_trial.sum()
                obj_trial = fast_convolve(f_trial).max()
                if obj_trial < M:
                    f_new = f_trial
                    obj_new = obj_trial
                    found = True
                    break
            if not found:
                lr *= 0.5
                continue
        
        f = f_new
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"      SMD iter {it}: obj={best_obj:.8f}, tau={tau:.2f}, lr={lr:.6f}")
        else:
            no_improve += 1
        
        # Adaptive parameters
        tau = min(tau * tau_growth, tau_max)
        lr *= lr_decay
        
        # Restart with perturbation
        if no_improve > restart_every:
            rng = np.random.default_rng(it * 7 + 13)
            noise_scale = 0.002 + 0.003 * (no_improve / restart_every)
            f = np.maximum(best_f + rng.standard_normal(n) * noise_scale, 1e-15)
            f = f / f.sum()
            lr = lr_init * 0.5
            tau = max(tau_init, tau * 0.5)
            no_improve = 0
        
        if lr < 1e-7:
            break
    
    return best_f, best_obj

def lbfgs_optimize(f0, time_limit, verbose=False):
    """
    L-BFGS optimization on log-parameters.
    a = softmax(x), minimize smooth_max(conv(a), tau).
    """
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    
    # Initialize x from log(f)
    x0 = np.log(f + 1e-15)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    
    # Multi-stage: increase tau over stages
    tau_schedule = [2.0, 5.0, 10.0, 20.0, 40.0, 80.0]
    
    for tau in tau_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        stage_time = min(60.0, time_limit - (time.monotonic() - t0))
        if stage_time < 5:
            break
        
        def objective(x):
            a = softmax(x)
            conv = fast_convolve(a)
            v = tau * conv
            v_max = np.max(v)
            return (v_max + np.log(np.exp(v - v_max).sum())) / tau
        
        def gradient(x):
            a = softmax(x)
            conv = fast_convolve(a)
            v = tau * conv
            v_max = np.max(v)
            w = np.exp(v - v_max)
            w /= w.sum()
            
            # Gradient w.r.t. a
            grad_a = np.zeros(n)
            threshold = 1e-8
            for k in range(len(conv)):
                if w[k] < threshold:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[js:je+1] += w[k] * (a[k - idx] + a[idx])
            
            # Chain rule: d/dx [f(softmax(x))] = softmax(x) * (grad_a - softmax(x) @ grad_a)
            # Actually: d(softmax(x))_i/dx_j = a_i * (delta_ij - a_j)
            # So grad_x_j = sum_i grad_a_i * a_i * (delta_ij - a_j)
            #             = a_j * (grad_a_j - sum_i grad_a_i * a_i)
            grad_x = a * (grad_a - np.dot(grad_a, a))
            return grad_x
        
        # Run L-BFGS-B for this stage
        result = minimize(objective, x0, method='L-BFGS-B', jac=gradient,
                         options={'maxiter': 200, 'maxfun': 500,
                                  'iprint': 0, 'gtol': 1e-8})
        
        # Update
        a_new = softmax(result.x)
        obj_new = fast_convolve(a_new).max()
        if obj_new < best_obj:
            best_obj = obj_new
            best_f = a_new.copy()
            if verbose:
                print(f"      L-BFGS tau={tau}: obj={best_obj:.8f}, iters={result.nit}")
        
        x0 = result.x
    
    return best_f, best_obj

def peak_balancing_descent(f0, time_limit, verbose=False):
    """
    Targeted peak balancing: iterate over top peaks, compute exact subgradient
    for each, and take multiplicative steps.
    """
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Get top 5 peaks
        k_top = min(5, len(conv))
        peaks = np.argsort(conv)[-k_top:]
        
        # Compute combined subgradient
        grad = np.zeros(n)
        for k in peaks:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)

    f_interp = np.maximum(f_interp, 0)

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-stage optimization:
1. Asymmetric initialization with diverse patterns
2. Smoothed Mirror Descent with adaptive log-sum-exp temperature
3. L-BFGS on log-parameters (softmax reparameterization) for faster convergence

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - np.max(x)
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def compute_grad_weights(conv, tau, top_frac=0.15):
    """Compute softmax weights on convolution values for gradient computation."""
    n_conv = len(conv)
    k_top = max(5, int(n_conv * top_frac))
    peaks = np.argpartition(conv, -k_top)[-k_top:]
    pv = tau * conv[peaks]
    pv_shifted = pv - np.max(pv)
    w = np.exp(pv_shifted)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    # Build full weight vector
    full_w = np.zeros(n_conv)
    full_w[peaks] = w
    return full_w


def conv_gradient_wrt_a(a, weights):
    """
    Compute gradient of sum_k weights[k] * conv_k w.r.t. a.
    conv_k = sum_i a_i * a_{k-i}
    d(conv_k)/da_j = a_{k-j} + a_{j-k} (with a_m=0 for m out of range)
    """
    n = len(a)
    grad = np.zeros(n)
    threshold = 1e-12
    for k in range(len(weights)):
        w = weights[k]
        if w < threshold:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        idx = np.arange(js, je + 1)
        grad[js:je + 1] += w * (a[k - idx] + a[idx])
    return grad


def smoothed_mirror_descent(f0, time_limit, tau_init=1.0, tau_max=80.0,
                            lr_init=0.015, lr_decay=0.9997,
                            tau_growth=1.015, restart_every=120, verbose=False):

    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    best_obj = conv.max()
    best_f = f.copy()

def lbfgs_optimize(f0, time_limit, verbose=False):
    """L-BFGS on log-parameters with multi-stage temperature."""
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    x0 = np.log(f + 1e-15)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()

    tau_stages = [3.0, 6.0, 12.0, 25.0, 50.0, 100.0]
    max_iter_per_stage = 300

    for tau in tau_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def obj_func(x):
            a = softmax(x)
            conv = fast_convolve(a)
            v = tau * conv
            v_max = np.max(v)
            return (v_max + np.log(np.sum(np.exp(v - v_max)))) / tau

        def grad_func(x):
            a = softmax(x)
            conv = fast_convolve(a)
            v = tau * conv
            v_max = np.max(v)
            w = np.exp(v - v_max)
            ws = w.sum()
            if ws < 1e-15:
                w = np.zeros(len(v))
                w[np.argmax(v)] = 1.0
            else:
                w /= ws
            grad_a = conv_gradient_wrt_a(a, w)
            grad_x = a * (grad_a - np.dot(grad_a, a))
            return grad_x

        result = minimize(obj_func, x0, method='L-BFGS-B', jac=grad_func,
                          options={'maxiter': max_iter_per_stage, 'maxfun': max_iter_per_stage * 2,
                                   'iprint': 0, 'gtol': 1e-7})

        a_new = softmax(result.x)
        obj_new = fast_convolve(a_new).max()
        if obj_new < best_obj:
            best_obj = obj_new
            best_f = a_new.copy()
            x0 = result.x
            if verbose:
                print(f"      L-BFGS tau={tau}: obj={best_obj:.8f}, nit={result.nit}")
        else:
            if verbose:
                print(f"      L-BFGS tau={tau}: no improvement (obj={obj_new:.8f} vs best={best_obj:.8f})")

    return best_f, best_obj

        if no_improve > 60:
            rng = np.random.default_rng(it * 41)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-15)
            f = f / f.sum()
            lr = 0.008
            no_improve = 0

    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        c_obj = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': max(1, time_limit - (time.monotonic() - t0))})

        if not res.success or res.x is None:
            no_improve += 1
            if no_improve > 15:
                break
            continue

        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            no_improve += 1
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        no_improve = 0

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-8:
            # Fine bisection
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(5):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new

            if M < best_obj - 1e-12:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 5 == 0:
                    print(f"      LP-FW it={it}: obj={best_obj:.8f}")
        else:
            no_improve += 1

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with random ordering."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    pass_count = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        rng = np.random.default_rng(pass_count * 1000 + 7)
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj

            old_val = f[j]
            if old_val < 1e-12:
                continue

            current_obj = fast_convolve(f).max()
            best_j_obj = current_obj
            best_j_val = old_val

            fracs = [-0.4, -0.25, -0.1, 0.1, 0.25, 0.4, 0.6]
            for frac in fracs:
                new_val = old_val * (1.0 + frac)
                if new_val < 0 or new_val > 0.7:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val and best_j_obj < best_obj - 1e-14:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                best_obj = best_j_obj
                best_f = f.copy()
                improved = True

        if not improved and pass_count > 2:
            # Small perturbation to escape
            rng2 = np.random.default_rng(pass_count * 311)
            f = np.maximum(best_f + rng2.standard_normal(n) * 0.001, 1e-15)
            f = f / f.sum()
            if pass_count > 5:
                break

    return best_f, best_obj


def make_init(n, variant, seed=42):
    """Generate diverse asymmetric initial sequences."""
    rng = np.random.default_rng(seed + variant * 100)

    if variant == 0:
        a = np.exp(-np.arange(n) / (n * 0.1))
    elif variant == 1:
        a = np.exp(-np.arange(n)[::-1] / (n * 0.1))
    elif variant == 2:
        a = np.exp(-np.arange(n) / (n * 0.2)) * (1.0 + 0.6 * np.sin(2 * np.pi * np.arange(n) / (n * 0.8)))
        a = np.maximum(a, 0)
    elif variant == 3:
        a = np.linspace(0.02, 1.0, n) ** 2
    elif variant == 4:
        x = np.arange(n) / n
        a = np.exp(-100 * (x - 0.2) ** 2) + 0.25 * np.exp(-30 * (x - 0.6) ** 2)
    elif variant == 5:
        a = (np.cos(np.pi * np.arange(n) / (2 * n)) ** 2 + 0.05)
    elif variant == 6:
        a = np.ones(n)
        a[:n // 4] *= 2.5
        a[n // 4:n // 2] *= 1.3
        a[n // 2:3 * n // 4] *= 0.5
        a[3 * n // 4:] *= 0.2
        a += rng.standard_normal(n) * 0.03
        a = np.maximum(a, 0)
    elif variant == 7:
        a = np.log1p(np.arange(n) + 1) ** 1.3
    elif variant == 8:
        a = (np.arange(n) + 1) ** (-0.8)
    elif variant == 9:
        a = np.zeros(n)
        split = n // 4
        a[:split] = np.exp(-np.arange(split) / (n * 0.04))
        a[split:] = 0.08 * np.exp(-np.arange(n - split) / (n * 0.4))
    elif variant == 10:
        x = np.arange(n) / n
        a = np.exp(-50 * (x - 0.1) ** 2) + 0.15 * np.exp(-15 * (x - 0.4) ** 2) + 0.05 * np.exp(-8 * (x - 0.7) ** 2)
    elif variant == 11:
        # Power-law with cutoff
        a = (1.0 + np.arange(n)) ** (-1.2)
        a[n // 2:] *= 0.5
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-stage optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses SMD, L-BFGS, peak-balanced descent, LP-FW, and coordinate descent
    across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Quick SMD on parent ===
    print("\n=== Phase 1: SMD on parent ===")
    if parent is not None:
        rem = deadline - time.monotonic() - 800
        t_alloc = max(30, min(rem, 120))
        print(f"  SMD on parent (n={len(parent)}): t={t_alloc:.0f}s")
        a_opt, obj = smoothed_mirror_descent(parent, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, "smd_parent")

    # === Phase 2: L-BFGS on best ===
    print("\n=== Phase 2: L-BFGS ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 600
        t_alloc = max(30, min(rem, 180))
        print(f"  L-BFGS (n={len(best_seq)}): t={t_alloc:.0f}s")
        a_opt, obj = lbfgs_optimize(best_seq, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, "lbfgs")

    # === Phase 3: Peak-balanced descent ===
    print("\n=== Phase 3: Peak-Balanced Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 500
        t_alloc = max(20, min(rem, 100))
        print(f"  PBD (n={len(best_seq)}): t={t_alloc:.0f}s")
        a_opt, obj = peak_balanced_descent(best_seq, time_limit=t_alloc, n_peaks=7, verbose=True)
        try_seq(a_opt, "pbd")

    # === Phase 4: LP-FW ===
    print("\n=== Phase 4: LP-FW ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        t_alloc = max(20, min(rem, 150))
        print(f"  LP-FW (n={len(best_seq)}): t={t_alloc:.0f}s")
        a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, "lpfw")

    # === Phase 5: Multi-length exploration ===
    print("\n=== Phase 5: Multi-Length Exploration ===")
    target_lengths = [896, 1024, 1152, 1280]
    if best_seq is not None:
        base = best_seq
        for target_n in target_lengths:
            rem = deadline - time.monotonic() - 300
            if rem < 60:
                break
            if len(base) == target_n:
                continue
            f_interp = interpolate_seq(base, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")
            if sc < best_score + 0.05:
                # Refine at this length
                t_smd = max(15, min(rem * 0.3, 60))
                print(f"  SMD n={target_n}: t={t_smd:.0f}s")
                a_opt, obj = smoothed_mirror_descent(f_interp, time_limit=t_smd, verbose=False)
                sc2 = try_seq(a_opt, f"smd_n{target_n}")
                if sc2 < best_score + 0.03:
                    rem2 = deadline - time.monotonic() - 250
                    if rem2 > 30:
                        t_lb = max(15, min(rem2 * 0.5, 80))
                        print(f"  L-BFGS n={target_n}: t={t_lb:.0f}s")
                        a_opt2, obj2 = lbfgs_optimize(a_opt, time_limit=t_lb, verbose=False)
                        try_seq(a_opt2, f"lbfgs_n{target_n}")

    # === Phase 6: SMD on diverse inits at best n ===
    print("\n=== Phase 6: Diverse Initializations ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for v in range(6):
            rem = deadline - time.monotonic() - 200
            if rem < 30:
                break
            a_init = make_init(n_best, variant=v + 12, seed=seed + v)
            sc = try_seq(a_init, f"init{v}_n{n_best}")
            if sc < best_score + 0.1:
                t_alloc = max(15, min(rem * 0.3, 60))
                print(f"  SMD init{v} n={n_best}: t={t_alloc:.0f}s")
                a_opt, obj = smoothed_mirror_descent(a_init, time_limit=t_alloc, verbose=False)
                try_seq(a_opt, f"opt_init{v}")

    # === Phase 7: Coordinate descent polish ===
    print("\n=== Phase 7: Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 30:
            t_alloc = min(rem, 100)
            print(f"  CD (n={len(best_seq)}): t={t_alloc:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, "cd")

    # === Phase 8: Final SMD squeeze ===
    print("\n=== Phase 8: Final Squeeze ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_alloc = min(rem, 60)
            print(f"  Final SMD (n={len(best_seq)}): t={t_alloc:.0f}s")
            a_opt, obj = smoothed_mirror_descent(best_seq, time_limit=t_alloc,
                                                 tau_init=10.0, tau_max=200.0,
                                                 lr_init=0.005, verbose=False)
            try_seq(a_opt, "final_smd")

    # === Phase 9: Final L-BFGS ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            t_alloc = min(rem, 40)
            print(f"  Final L-BFGS: t={t_alloc:.0f}s")
            a_opt, obj = lbfgs_optimize(best_seq, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, "final_lbfgs")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
