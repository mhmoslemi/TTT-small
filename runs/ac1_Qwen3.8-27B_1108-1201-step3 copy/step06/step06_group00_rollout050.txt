"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based projected steps
combined with subgradient polishing and coordinate descent.

The LP step finds the maximum-sum non-negative sequence g satisfying
conv(f,g) <= M (current peak), which provides a globally informed
direction for reducing the peak of the self-convolution.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_objective(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build the convolution matrix A where A[k,j] = f[k-j] if valid."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_step(f, M, timeout=30.0):
    """
    Solve: max sum(g) s.t. conv(f,g) <= M (elementwise), g >= 0.
    Returns the optimal g or None if failed.
    """
    n = len(f)
    A = build_conv_matrix(f)
    
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = M * np.ones(2 * n - 1)
    
    try:
        res = linprog(c, A_ub=(A, b_ub), b_ub=b_ub, 
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception as e:
        print(f"  LP failed: {e}")
    
    return None


def line_search(f, g, M, max_iter=60):
    """
    Find the largest t in (0,1] such that max(conv((1-t)f+tg, (1-t)f+tg)) < M.
    Uses bisection.
    """
    lo, hi = 0.0, 1.0
    
    # Check t=1
    b1 = fast_convolve(g)
    if b1.max() < M:
        return 1.0
    
    # Check t=0.5
    b05 = fast_convolve(0.5 * f + 0.5 * g)
    if b05.max() < M:
        lo = 0.5
    else:
        hi = 0.5
    
    for _ in range(max_iter):
        mid = (lo + hi) / 2
        f_new = (1 - mid) * f + mid * g
        b_new = fast_convolve(f_new)
        if b_new.max() < M:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-12:
            break
    
    return lo


def subgradient_optimize(a0, time_limit, lr_init=0.003, momentum=0.92,
                         temp_init=0.0003, temp_growth=1.015,
                         kick_interval=1500, kick_size=0.002,
                         n_top=None, seed=42):
    """
    Projected subgradient method with Nesterov momentum.
    Optimizes on the simplex (sum=1).
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    a = a / S
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    cur_temp = temp_init
    no_improve = 0
    it_count = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()
        
        # Temperature-weighted peaks
        scaled = b / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        if n_top is not None and n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx].copy()
            w_sel /= w_sel.sum()
            w_full = np.zeros(len(b))
            w_full[top_idx] = w_sel
            w = w_full
        
        # Subgradient computation
        g = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-14:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += w[k] * a[k - idx]
        
        # Nesterov step
        v = momentum * v - cur_lr * g
        a_new = project_simplex(a + v)
        
        b_new = fast_convolve(a_new)
        obj_new = b_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        it_count += 1
        
        # Periodic adjustments
        if no_improve > kick_interval:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.7, lr_init * 0.1)
            no_improve = 0
        elif no_improve > kick_interval // 2:
            cur_lr *= 0.97
        
        cur_temp *= temp_growth
        if cur_temp > 0.5:
            cur_temp = 0.5
    
    return best_a, best_obj


def coordinate_polish(a0, time_limit, n_trials_per_elem=8):
    """Coordinate descent polish."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    multipliers = np.array([0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0])
    
    improved_global = True
    while improved_global and time.monotonic() - t0 < time_limit:
        improved_global = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_global = True
    return best_a, best_obj


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def cross_corr_fft(w, a):
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:n]


def soft_max_obj_grad(z, T, n):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    grad_a = 2.0 * n * cross_corr_fft(w, a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z

        if time.monotonic() - t0 > time_limit:
            return 1e15
        obj, _ = soft_max_obj_grad(z, T, n)
        return obj
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = soft_max_obj_grad(z, T, n)
        return g
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        final_obj = func(res.x)
        if np.isfinite(final_obj) and final_obj < best_obj:
            best_z = res.x
            best_obj = final_obj
    except Exception:
        pass
    
    return softmax_from_z(best_z), best_obj


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
