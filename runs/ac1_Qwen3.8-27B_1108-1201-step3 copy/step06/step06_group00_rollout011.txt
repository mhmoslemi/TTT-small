import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_autoconv(a):
    """Full auto-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.maximum(v, 0.0)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1)
    return np.maximum(v - theta, 0.0)


def subgrad_direction(a, b, weights):
    """Compute subgradient direction: sum_k w[k] * grad(b[k]).
    grad(b[k])[i] = a[k-i]*I(0<=k-i<n) + a[i-k]*I(0<=i-k<n)
    """
    n = len(a)
    g = np.zeros(n)
    for k in range(len(weights)):
        wk = weights[k]
        if wk < 1e-14:
            continue
        # First term: a[k-i] for valid i, i.e., 0 <= k-i < n => k-n+1 <= i <= k
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            g[idx] += wk * a[k - idx]
        # Second term: a[i-k] for valid i, i.e., 0 <= i-k < n => k <= i <= k+n-1
        lo2 = max(0, k)
        hi2 = min(n - 1, k + n - 1)
        if lo2 <= hi2:
            idx = np.arange(lo2, hi2 + 1)
            g[idx] += wk * a[idx - k]
    return g


def apply_symmetry(a):
    """Enforce a[i] = a[n-1-i]."""
    n = len(a)
    b = 0.5 * (a + a[::-1])
    return b


def subgradient_optimize(a0, time_limit, lr_init=0.01, momentum=0.9,

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    a = a / S
    if use_symmetry:
        a = apply_symmetry(a)
        a = project_simplex(a)
    v = np.zeros(n)
    best_a = a.copy()
    best_bmax = fast_autoconv(a).max()
    cur_lr = lr_init
    cur_temp = temp_init
    no_improve = 0
    it_count = 0
    avg_a = a.copy()
    alpha_avg = 0.01

    while time.monotonic() - t0 < time_limit:
        b = fast_autoconv(a)
        b_max = b.max()

        # Compute temperature-weighted weights over active peaks
        scaled = b / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
        else:
            w /= w_sum

        if n_top is not None and n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx]
            ws = w_sel.sum()
            if ws > 1e-30:
                w_sel /= ws
            w_full = np.zeros(len(b))
            w_full[top_idx] = w_sel
            w = w_full

        g = subgrad_direction(a, b, w)

        # Nesterov step with backtracking
        v = momentum * v - cur_lr * g
        v_new = a + v
        if use_symmetry:
            v_new = apply_symmetry(v_new)
        a_new = project_simplex(v_new)

        b_new = fast_autoconv(a_new)
        obj_new = b_new.max()

        # Backtracking if no improvement
        retries = 0
        while obj_new >= b_max - 1e-15 and retries < 5:
            cur_lr *= 0.5
            v = momentum * v - cur_lr * g
            v_new = a + v
            if use_symmetry:
                v_new = apply_symmetry(v_new)
            a_new = project_simplex(v_new)
            b_new = fast_autoconv(a_new)
            obj_new = b_new.max()
            retries += 1

        a = a_new

        # Track best
        if obj_new < best_bmax - 1e-15:
            best_bmax = obj_new
            best_a = a.copy()
            no_improve = 0
            # Resume lr
            cur_lr = min(cur_lr * 1.05, lr_init)
        else:
            no_improve += 1

        # Running average (Polyak averaging)
        alpha_avg = alpha_avg * 0.999 + 0.001
        avg_a = (1 - alpha_avg) * avg_a + alpha_avg * a

        it_count += 1

        # Periodic adjustments
        if no_improve > kick_interval:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            if use_symmetry:
                a = apply_symmetry(a)
                a = project_simplex(a)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.5, lr_init * 0.1)
            no_improve = 0
        elif no_improve > kick_interval // 2:
            cur_lr *= 0.98

        cur_temp *= temp_growth
        if cur_temp > 0.5:
            cur_temp = 0.5

    # Check both best_a and avg_a
    avg_a = project_simplex(avg_a)
    if use_symmetry:
        avg_a = apply_symmetry(avg_a)
        avg_a = project_simplex(avg_a)
    avg_obj = fast_autoconv(avg_a).max()
    if avg_obj < best_bmax:
        best_a = avg_a
        best_bmax = avg_obj

    return best_a, best_bmax


def coordinate_descent(a0, time_limit, n_active=80, use_symmetry=True):
    """Coordinate descent: for each i, optimize a[i] given others fixed.
    b[k] is linear in a[i], so we minimize a convex piecewise-linear function.
    """
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_autoconv(a).max()
    improved_total = True

        improved_total = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            if old_val < 1e-14:
                x_max = 1.0
            else:
                x_max = 1.0 - (1.0 - old_val)  # = old_val, but we allow increase
                # Actually: sum(a) = 1, so a[i] <= 1 - sum_{j!=i} a[j] = 1 - (1 - a[i]) = a[i]
                # That's circular. The constraint is just a[i] >= 0 and the rest sum to 1-a[i].
                # We can set a[i] to any value in [0, 1] and scale the rest.
                # But that changes the other coordinates. Let's fix the sum of others.
                # sum_{j!=i} a[j] = 1 - old_val. If we set a[i] = x, we need sum_{j!=i} a[j] = 1-x.
                # So we scale: a_new[j] = a[j] * (1-x)/(1-old_val) for j != i.
                x_max = 1.0

            # b[k] = a[i]*(a[k-i] + a[i-k]) + c[k]
            # where c[k] = b[k] - a[i]*(a[k-i] + a[i-k])
            b = fast_autoconv(a)

            d = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                val = 0.0
                if 0 <= k - i < n:
                    val += a[k - i]
                if 0 <= i - k < n:
                    val += a[i - k]
                d[k] = val

            c = b - old_val * d

            # Find the x in [0, x_max] that minimizes max_k (d[k]*x + c[k])
            # subject to the constraint that when we rescale others, the actual b changes.
            # Wait - if we change a[i] and rescale others to maintain sum=1, the relationship
            # is NOT simply linear in x. Let me reconsider.
            #
            # If we set a[i] = x and scale others by (1-x)/(1-old_val):
            # a_new[j] = a[j] * (1-x)/(1-old_val) for j != i
            # a_new[i] = x
            #
            # Then b_new[k] = sum_{p,q: p+q=k} a_new[p]*a_new[q]
            # = a_new[i]*a_new[k-i] + a_new[k-i]*a_new[i] + a_new[i]*a_new[i] (if 2i=k)
            #   + sum_{p,q != i, p+q=k} a_new[p]*a_new[q]
            #
            # This is quadratic in x, not linear! So the simple 1D approach doesn't work
            # with rescaling.
            #
            # Alternative: DON'T rescale. Just change a[i] and keep sum != 1.
            # The objective is 2n*max(b)/S^2 where S = sum(a).
            # This is more complex. Let's use a simpler approach:
            # Just try a few candidate values for a[i] with rescaling.

            # Simple approach: try multipliers
            multipliers = [0.5, 0.7, 0.85, 0.95, 1.0, 1.05, 1.15, 1.3, 1.5, 2.0]
            best_x_obj = best_obj
            best_x_val = old_val

            for mult in multipliers:
                x = old_val * mult
                if x < 0 or x > 0.9999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - x) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[i] = x
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()  # ensure exact sum=1
                if use_symmetry:
                    a_trial = apply_symmetry(a_trial)
                    a_trial = a_trial / a_trial.sum()
                b_trial = fast_autoconv(a_trial)
                obj_trial = b_trial.max()
                if obj_trial < best_x_obj - 1e-15:
                    best_x_obj = obj_trial
                    best_x_val = x

            if best_x_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_x_val) / denom
                a[i] = best_x_val
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                a[mask] *= scale
                a = a / a.sum()
                if use_symmetry:
                    a = apply_symmetry(a)
                    a = a / a.sum()
                if best_x_obj < best_obj - 1e-15:
                    best_obj = best_x_obj
                    best_a = a.copy()
                    improved_total = True

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian_init(n, sigma_frac=0.4):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma_frac) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    taper = int(r * n // 2)
    if taper > 0:
        t = np.linspace(0, 1, taper)
        taper_vals = 0.5 * (1 + np.cos(np.pi * (2 * t / r - 1)))
        a[:taper] = taper_vals
        a[-taper:] = taper_vals[::-1]
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def flat_topped_init(n, w=0.6):
    """Flat-topped window with cosine taper at edges."""
    a = np.ones(n)
    taper = int((1 - w) * n // 2)
    if taper > 0:
        t = np.linspace(0, 1, taper)
        taper_vals = 0.5 * (1 + np.cos(np.pi * t))
        a[:taper] = taper_vals
        a[-taper:] = taper_vals[::-1]
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Subgradient from parent (symmetric)
    2. L-BFGS-style refinements via multiple temperatures  
    3. Diverse initializations
    4. Multi-length exploration
    5. Coordinate descent polish
    6. Final deep optimization
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1152) / 1152
        print("No parent, using uniform n=1152")

    n_parent = len(parent)

    # Make parent symmetric
    parent_sym = apply_symmetry(parent)
    parent_sym = parent_sym / parent_sym.sum()
    try_seq(parent_sym, "parent_sym")

    # === Phase 1: Subgradient from symmetric parent ===
    print(f"\n=== Phase 1: Subgradient from parent (n={n_parent}) ===")
    rem = deadline - time.monotonic() - 800
    if rem > 60:
        t_alloc = min(rem, 500.0)
        print(f"  Allocating {t_alloc:.0f}s")
        a_sg, _ = subgradient_optimize(
            parent_sym, time_limit=t_alloc,
            lr_init=0.005, momentum=0.91,
            temp_init=0.0005, temp_growth=1.012,
            kick_interval=1500, kick_size=0.002,
            n_top=max(20, n_parent // 8), seed=seed,
            use_symmetry=True
        )
        try_seq(a_sg, f"sg_parent_n{n_parent}")

    # === Phase 2: Coordinate descent polish ===
    print(f"\n=== Phase 2: CD polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 30:
            t_cd = min(rem, 200.0)
            print(f"  CD t={t_cd:.0f}s")
            a_cd, _ = coordinate_descent(best_seq, time_limit=t_cd, use_symmetry=True)
            try_seq(a_cd, "cd_polish")

    # === Phase 3: More subgradient with different params ===
    print(f"\n=== Phase 3: Additional subgradient variants ===")
    if best_seq is not None:
        for variant_name, kwargs in [
            ("focused", dict(temp_init=0.0001, temp_growth=1.005, lr_init=0.002, momentum=0.95, n_top=10)),
            ("broad", dict(temp_init=0.05, temp_growth=1.01, lr_init=0.003, momentum=0.9, n_top=max(40, n_parent//4))),
            ("medium", dict(temp_init=0.005, temp_growth=1.02, lr_init=0.004, momentum=0.93, n_top=max(15, n_parent//10))),
        ]:
            rem = deadline - time.monotonic() - 300
            if rem < 30:
                break
            t_alloc = min(rem * 0.3, 120.0)
            if t_alloc < 15:
                continue
            print(f"  Variant: {variant_name}, t={t_alloc:.0f}s")
            a_v, _ = subgradient_optimize(
                best_seq, time_limit=t_alloc,
                use_symmetry=True,
                seed=seed + hash(variant_name) % 10000,
                **kwargs
            )
            try_seq(a_v, f"sg_{variant_name}")

    # === Phase 4: Diverse initializations ===
    print(f"\n=== Phase 4: Diverse initializations ===")
    if best_seq is not None:
        n_base = len(best_seq)
    else:
        n_base = n_parent

    inits = {
        "gauss_0.35": make_gaussian_init(n_base, 0.35),
        "gauss_0.5": make_gaussian_init(n_base, 0.5),
        "cosine": make_cosine_init(n_base),
        "tukey_0.4": make_tukey_init(n_base, 0.4),
        "tukey_0.6": make_tukey_init(n_base, 0.6),
        "flattop_0.5": flat_topped_init(n_base, 0.5),
        "flattop_0.7": flat_topped_init(n_base, 0.7),
        "parent_perturb": project_simplex(apply_symmetry(best_seq + rng.standard_normal(n_base) * 0.01)) if best_seq is not None else parent_sym,
    }

    for name, init_a in inits.items():
        rem = deadline - time.monotonic() - 200
        if rem < 20:
            break
        t_alloc = min(rem * 0.2, 50.0)
        if t_alloc < 8:
            continue
        print(f"  {name}: t={t_alloc:.0f}s")
        a_opt, _ = subgradient_optimize(
            init_a, time_limit=t_alloc,
            lr_init=0.008, momentum=0.9,
            temp_init=0.002, temp_growth=1.03,
            kick_interval=800, kick_size=0.005,
            n_top=max(10, n_base // 15),
            seed=seed + hash(name) % 10000,
            use_symmetry=True
        )
        try_seq(a_opt, f"div_{name}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best_cur = len(best_seq)
        target_lengths = []
        for factor in [0.75, 0.875, 1.125, 1.25, 1.5]:
            tl = int(n_best_cur * factor)
            tl = max(512, min(3072, tl))
            if tl not in target_lengths and tl != n_best_cur:
                target_lengths.append(tl)
        target_lengths.sort(key=lambda x: abs(x - n_best_cur))

        for target_n in target_lengths[:3]:
            rem = deadline - time.monotonic() - 150
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            f_interp = apply_symmetry(f_interp)
            f_interp = f_interp / f_interp.sum()

            t_alloc = min(rem * 0.35, 100.0)
            if t_alloc < 10:
                continue
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = subgradient_optimize(
                f_interp, time_limit=t_alloc,
                lr_init=0.004, momentum=0.92,
                temp_init=0.001, temp_growth=1.015,
                kick_interval=1200, kick_size=0.002,
                n_top=max(15, target_n // 10),
                seed=seed + target_n % 10000,
                use_symmetry=True
            )
            try_seq(a_opt, f"len_n{target_n}")

            # Quick CD on new length
            rem = deadline - time.monotonic() - 120
            if rem > 20 and best_seq is not None and len(best_seq) == target_n:
                t_cd2 = min(rem * 0.3, 60.0)
                a_cd2, _ = coordinate_descent(best_seq, time_limit=t_cd2, use_symmetry=True)
                try_seq(a_cd2, f"cd_n{target_n}")

    # === Phase 6: Deep final optimization ===
    print(f"\n=== Phase 6: Deep final optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_f = len(best_seq)
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_deep = min(rem, 150.0)
            print(f"  Deep SG: n={n_f}, t={t_deep:.0f}s")
            a_deep, _ = subgradient_optimize(
                best_seq, time_limit=t_deep,
                lr_init=0.001, momentum=0.96,
                temp_init=0.0002, temp_growth=1.003,
                kick_interval=3000, kick_size=0.0005,
                n_top=max(30, n_f // 5),
                seed=seed + 99999,
                use_symmetry=True
            )
            try_seq(a_deep, "deep_sg")

    # === Phase 7: Final CD polish ===
    print(f"\n=== Phase 7: Final CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 20:
            t_final_cd = min(rem, 80.0)
            print(f"  Final CD: t={t_final_cd:.0f}s")
            a_fin, _ = coordinate_descent(best_seq, time_limit=t_final_cd, use_symmetry=True)
            try_seq(a_fin, "final_cd")

    # === Phase 8: Micro perturbation restarts ===
    print(f"\n=== Phase 8: Micro restarts (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_m = len(best_seq)
        for attempt in range(5):
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            noise_scale = 0.002 / (attempt + 1)
            a_pert = project_simplex(apply_symmetry(best_seq + rng.standard_normal(n_m) * noise_scale))
            t_alloc = min(rem * 0.4, 30.0)
            if t_alloc < 5:
                continue
            a_p, _ = subgradient_optimize(
                a_pert, time_limit=t_alloc,
                lr_init=0.001, momentum=0.95,
                temp_init=0.0003, temp_growth=1.005,
                kick_interval=500, kick_size=0.0005,
                n_top=max(10, n_m // 8),
                seed=seed + attempt * 31337,
                use_symmetry=True
            )
            try_seq(a_p, f"micro_{attempt}")

    # Fallback
    if best_seq is None:
        best_seq = parent_sym.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent_sym.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
