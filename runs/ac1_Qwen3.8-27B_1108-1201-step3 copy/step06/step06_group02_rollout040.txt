                obj = fast_convolve(trial).max()
                if obj < best_obj - 1e-14:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    no_improve = 0
                    break
            if no_improve == 0:
                break

        if no_improve > 0:
            # Try different peaks
            top_peaks = np.argsort(conv)[-5:]
            found = False
            for pk in top_peaks:
                pk = int(pk)
                if pk == peak_k:
                    continue
                js2 = max(0, pk - n + 1)
                je2 = min(n - 1, pk)
                cidx2 = np.arange(js2, je2 + 1)
                if len(cidx2) == 0:
                    continue
                prods2 = f[cidx2] * f[pk - cidx2]
                src2 = cidx2[np.argmax(prods2)]
                nc2 = np.ones(n, dtype=bool)
                nc2[cidx2] = False
                for frac2 in [0.15, 0.3, 0.5]:
                    diff2 = f[src2] * frac2
                    trial = f.copy()
                    trial[src2] -= diff2
                    nc2_vals = trial[nc2]
                    nc2_sum = nc2_vals.sum()
                    if nc2_sum < 1e-15:
                        continue
                    trial[nc2] += diff2 * nc2_vals / nc2_sum
                    trial = normalize(trial)
                    obj2 = fast_convolve(trial).max()
                    if obj2 < best_obj - 1e-14:
                        f = trial
                        best_obj = obj2
                        best_f = f.copy()
                        no_improve = 0
                        found = True
                        break
                if found:
                    break
            if not found:
                no_improve += 1

    return best_f, best_obj


def subgradient_descent(f0, time_limit, verbose=False, lr_init=0.01, top_k=20):
    """Subgradient-based multiplicative descent targeting top peaks."""
    f = np.maximum(f0.astype(np.float64).copy(), 1e-12)
    f = f / f.sum()
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2.0:
            break

        conv = fast_convolve(f)
        M = conv.max()

        # Get top-k peaks with weights
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        # Temperature-based weighting
        temp = 0.001
        scaled = (pv - pv.min()) / (pv.max() - pv.min() + 1e-15)
        w = np.exp(scaled / max(temp, 1e-4))
        w = w / w.sum()

        # Subgradient: for peak k, d(f*f)(k)/df[i] = f[k-i] (when valid)
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-10:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_range = np.arange(js, je + 1)
            partner = k - idx_range
            sg[idx_range] += w[idx] * f[partner]

        # Multiplicative update
        f = f * np.exp(-lr * sg)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        co = fast_convolve(f).max()
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    SGD iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        lr *= 0.999
        if no_improve > 100:
            rng = np.random.default_rng(it * 53 + 13)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-12)
            f = f / f.sum()
            lr = lr_init * 0.5
            no_improve = 0

        if lr < 1e-9:
