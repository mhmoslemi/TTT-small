    """Best response using constraint generation."""
    n = len(a_fixed)
    T = build_conv_matrix(a_fixed)
    
    # Start with the top 50 constraints
    c = T @ b_init
    active = np.argpartition(c, -50)[-50:]
    
    b_val = b_init.copy()
    for _ in range(20):  # Max 20 constraint additions
        # Solve LP with current active constraints
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        prob = cp.Problem(cp.Minimize(M_var), [
            T[active, :] @ b_var <= M_var,
            cp.sum(b_var) == 1
        ])
        try:
            prob.solve(solver=solver)
        except:
            prob.solve(solver=cp.ECOS)
        
        b_new = np.asarray(b_var.value, dtype=np.float64)
        b_new = np.maximum(b_new, 0)
        b_new = b_new / b_new.sum()
        M_new = float(M_var.value)
        
        # Check if all constraints are satisfied
        c_new = T @ b_new
        violated = c_new > M_new + inner_tol
        
        if not violated.any():
            return b_new, M_new
        
        # Add violated constraints
        new_active = np.unique(np.concatenate([active, np.where(violated)[0]]))
        active = new_active
        b_val = b_new
    
    return b_val, float((T @ b_val).max())

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Compute linear convolution conv(a, b) using FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix_rows(a_fixed, row_indices):
    """Build only the specified rows of the convolution matrix T.
    T[k, j] = a_fixed[k-j] if 0 <= k-j < len(a_fixed), else 0.
    Returns a (len(row_indices), len(a_fixed)) matrix.
    """
    n = len(a_fixed)
    m = len(row_indices)
    T_sub = np.zeros((m, n))
    for i, k in enumerate(row_indices):
        k = int(k)
        # T[k, j] = a_fixed[k-j] for j where 0 <= k-j < n
        # i.e., j ranges from max(0, k-n+1) to min(n-1, k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            T_sub[i, js:je + 1] = a_fixed[je - js::-1] if k >= n else a_fixed[k - js:k - je + 1][::-1]
            # Let me redo this more carefully
            # T[k,j] = a_fixed[k-j] for valid j
            # For j in [js, je]: k-j ranges from k-je to k-js
            # a_fixed[k-je], a_fixed[k-je+1], ..., a_fixed[k-js]
            # which is a_fixed[k-je : k-js+1]
            lo = k - je
            hi = k - js
            T_sub[i, js:je + 1] = a_fixed[lo:hi + 1]
    return T_sub


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

        b_new = np.asarray(b_var.value, dtype=np.float64)

def score_of(a):
    """Compute the score 2*n*max(conv(a,a)) for a normalized vector."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()


def fictitious_play_optimize(n, time_limit, initial=None, solver=None, verbose=True, ewma_beta=1.0):

    ewma_beta: if 1.0, use standard running average. If < 1.0, use EWMA.

        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)

        else:
            a = a / s

    b = a.copy()
    a_avg = a.copy().astype(np.float64)

    while time.monotonic() - t0 < time_limit:

        b_new, _ = best_response_lp(a, b_init=b, solver=solver)

            b = b_new

def damped_lp_refine(a0, time_limit, solver=None, max_iters=5000):

    a = a / s

    t0 = time.monotonic()
    b = a.copy()

    for it in range(max_iters):

        if b_new is None:
            continue
        b = b_new

        alpha = min(1.0, 10.0 / (it + 1))

        a_new = np.maximum(a_new, 0)

            best_a = a_new.copy()
        a = a_new

        if (it + 1) % 200 == 0:

def peak_shave_local(a0, time_limit):
    """Aggressive local peak shaving with redistribution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 200:
        c = fast_convolve(a)
        # Find peaks (exclude endpoints which are always small)
        c_mid = c.copy()
        c_mid[0] = -np.inf
        c_mid[-1] = -np.inf
        peak_k = int(np.argmax(c_mid))

        if c[peak_k] * 2 * n <= best_score + 1e-15:
            no_improve += 1
            continue

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_contrib = np.argsort(contribs)[::-1][:5]

        improved = False
        for ci in top_contrib:
            i = idx[ci]
            j = peak_k - i
            if i == j or a[j] < 1e-15 or a[i] < 1e-15:
                continue
            for f1 in [0.85, 0.75, 0.6]:
                for f2 in [0.85, 0.75, 0.6]:
                    a_trial = a.copy()
                    a_trial[i] *= f1
                    a_trial[j] *= f2
                    diff = a[i] * (1 - f1) + a[j] * (1 - f2)
                    if diff < 1e-15:
                        continue
                    # Redistribute to low-contribution positions
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    mask[j] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        continue
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    sc = score_of(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        no_improve = 0
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        # Also try single-element reduction
        if not improved:
            for ci in top_contrib[:3]:
                i = idx[ci]
                if a[i] < 1e-14:
                    continue
                for factor in [0.9, 0.8, 0.7]:
                    a_trial = a.copy()
                    a_trial[i] *= factor
                    diff = a[i] * (1 - factor)
                    if diff < 1e-15:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        continue
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    sc = score_of(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        no_improve = 0
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            no_improve += 1

    return best_a, best_score


def subgradient_refine(a0, time_limit, lr_init=0.005, top_k=50):
    """Subgradient method with momentum for fine-tuning."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_of(a)
    v = np.zeros(n)
    mom = 0.9
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    step_count = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        k_actual = min(top_k, len(c))
        peaks = np.argpartition(c, -k_actual)[-k_actual:]
        pv = c[peaks]
        temp = 0.003 * max(pv.max(), 1e-10)
        scaled = (pv - pv.max()) / temp
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for ii in range(len(peaks)):
            if w[ii] < 1e-13:
                continue
            k = int(peaks[ii])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[ii] * a[k - idx]

        v = mom * v - lr * sg
        a_candidate = a + v
        a_new = project_simplex(a_candidate)
        a_new = a_new / a_new.sum()
        sc = score_of(a_new)
        step_count += 1

        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.999
        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 1000000)
            a = best_a + rng.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 2, 0.003)
            no_improve = 0

        if step_count % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.6f}")

    return best_a, best_score


def lbgfs_polish(a0, time_limit, T=200, maxiter=5000):
    """L-BFGS polish with soft-max smoothing in log-space."""
    from scipy.optimize import minimize
    n = len(a0)
    a_safe = np.maximum(a0, 1e-300)
    z0 = np.log(a_safe)
    z0 = z0 - z0.sum()  # Center
    t0 = time.monotonic()
    stopped = [False]

    def func(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        z_shifted = z - z.max()
        a_local = np.exp(z_shifted)
        a_local = a_local / a_local.sum()
        c = fast_convolve(a_local)
        Tb = T * c
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        lse = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T) * lse

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros_like(z)
        z_shifted = z - z.max()
        a_local = np.exp(z_shifted)
        a_local = a_local / a_local.sum()
        c = fast_convolve(a_local)
        Tb = T * c
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        # Gradient of max_k c[k] wrt a: for peak k, grad_a[i] = a[k-i] + a[i-k]
        # Using weighted sum over peaks:
        grad_a = np.zeros(n)
        for ii in range(len(w)):
            if w[ii] < 1e-12:
                continue
            k = ii
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += w[ii] * a_local[k - idx]
            # Also the symmetric part
            idx2 = np.arange(max(0, n - 1 - k), min(n, n + 1 - k))
            # Actually, for conv(a,a)[k] = sum_j a[j]*a[k-j], d/d a[i] = a[k-i] + a[i-k] (if valid)
            # Let me just use the first part since conv is symmetric
        # Gradient in z-space: grad_z = a * (grad_a - a * dot(grad_a, a))
        S = np.dot(grad_a, a_local)
        grad_z = a_local * (grad_a - a_local * S)
        return grad_z

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        z_out = res.x
        a_out = np.exp(z_out - z_out.max())
        a_out = a_out / a_out.sum()
    except Exception:
        a_out = a0.copy()

    return a_out


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
