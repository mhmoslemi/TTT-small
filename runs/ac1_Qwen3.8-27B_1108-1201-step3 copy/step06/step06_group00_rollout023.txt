"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an LP-based "best response" 
iteration scheme with multiple refinements.

The core idea: at each iteration, solve an LP to find the best direction
g that keeps conv(a,g) bounded by the current max, then move toward g.
This is a game-theoretic approach where the LP finds the "exploit" of
the current sequence.

Additional techniques:
- Active set reduction for the LP (only enforce near-binding constraints)
- Multiple initializations (parent, structured, random)
- Subgradient polishing between LP iterations
- Coordinate descent micro-optimization
- Length exploration

import numpy as np
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_local(a):
    """Compute normalized objective 2n*max(conv)/(sum)^2 locally."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def build_lp_constraint_matrix(a, active_lags=None):
    """
    Build the sparse constraint matrix for the LP.
    Returns A (sparse) where A[k, i] gives the coefficient of g[i] in constraint k.
    The constraint is: A @ g <= M (for each active lag k)
    """
    n = len(a)
    if active_lags is None:
        active_lags = np.arange(2 * n - 1)
    
    rows = []
    cols = []
    vals = []
    
    for k in active_lags:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            rows.append(k)  # We'll remap later
            cols.append(i)
            vals.append(a[k - i])
    
    # Remap rows to be sequential
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    if active_lags is not None:
        A = A[active_lags, :]
    
    return A


def solve_best_response_lp(a, M, active_lags=None, timeout=30):
    """
    Solve: maximize sum(g) subject to conv(a,g) <= M, g >= 0.
    Returns (optimal_value, g_optimal).
    """
    n = len(a)
    
    # Build constraint matrix
    if active_lags is None:
        active_lags = np.arange(2 * n - 1)
    
    rows = []
    cols = []
    vals = []
    
    for idx, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            rows.append(idx)
            cols.append(i)
            vals.append(a[k - i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_lags), n))
    b = np.full(len(active_lags), M)
    c = -np.ones(n)  # minimize -sum(g)
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, 
                      method='highs', options={'time_limit': timeout})
        if res.success:
            return -res.fun, res.x
        else:
            return None, None
    except Exception:
        return None, None


def lp_iteration(a, n_active=None, t_adapt=True, timeout=30):
    """
    One iteration of the LP-based best response scheme.
    Returns (new_a, info_dict).
    """
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    
    # Select active lags
    if n_active is not None and n_active < 2 * n - 1:
        # Use lags near the max
        threshold = M - 0.01 * M  # 1% below max
        active_mask = conv >= threshold
        if active_mask.sum() < 5:
            # At least 5 active lags
            active_idx = np.argpartition(conv, -max(5, n_active // 10))[-max(5, n_active // 10):]
            active_mask = np.zeros(2*n-1, dtype=bool)
            active_mask[active_idx] = True
        active_lags = np.where(active_mask)[0]
    else:
        active_lags = None
    
    val, g = solve_best_response_lp(a, M, active_lags, timeout=timeout)
    
    if val is None or g is None:
        return a, {'success': False, 'M': M, 'val': None}
    
    # Check if we should update
    if val <= 1.0 + 1e-6:
        return a, {'success': False, 'M': M, 'val': val, 'converged': True}
    
    # Adaptive step size
    if t_adapt:
        t = min(0.3, 0.5 / (val - 1.0 + 1e-6))
    else:
        t = 0.1
    
    g_norm = g / g.sum()
    a_new = (1 - t) * a + t * g_norm
    a_new = np.maximum(a_new, 0)
    a_new = a_new / a_new.sum()
    
    return a_new, {'success': True, 'M': M, 'val': val, 't': t}


def lp_optimize(a0, time_limit, n_active=None, timeout=30, seed=42):
    """
    Run the LP-based best response optimization.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    no_improve = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        a_new, info = lp_iteration(a, n_active=n_active, timeout=timeout)
        iter_count += 1
        
        if not info['success']:
            if info.get('converged', False):
                # Try with fewer active constraints to escape
                a = project_simplex(a + rng.standard_normal(n) * 0.005)
                continue
            # LP failed, try subgradient step instead
            a = subgradient_step(a, lr=0.001)
            continue
        
        M_new = fast_convolve(a_new).max()
        
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        if no_improve > 50:
            # Restart with perturbation
            a = project_simplex(best_a + rng.standard_normal(n) * 0.01)
            no_improve = 0
        
        if iter_count % 50 == 0:
            print(f"    LP iter {iter_count}: M={best_M:.10f}, info={info}")
    
    return best_a, best_M


def subgradient_step(a, lr=0.001, n_top=50):
    """One subgradient step (for when LP fails)."""
    n = len(a)
    conv = fast_convolve(a)
    
    # Focus on top peaks
    if n_top < len(conv):
        top_idx = np.argpartition(conv, -n_top)[-n_top:]
    else:
        top_idx = np.arange(len(conv))
    
    # Uniform weight on top peaks
    g = np.zeros(n)
    for k in top_idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        g[idx] += conv[k] * a[k - idx]
    
    g /= len(top_idx)
    g /= (g.sum() + 1e-30)
    
    a_new = project_simplex(a - lr * g)
    return a_new


def subgradient_optimize(a0, time_limit, lr_init=0.005, momentum=0.9,
                         temp_init=0.0005, temp_growth=1.02,
                         kick_interval=2000, kick_size=0.003,
                         n_top=None, seed=42):
    """
    Projected subgradient method with Nesterov momentum (fallback method).
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    cur_temp = temp_init
    no_improve = 0
    it_count = 0
    
    while time.monotonic() - t0 < time_limit:
        for _ in range(30):
            if time.monotonic() - t0 >= time_limit:
                break
            it_count += 1
            b = fast_convolve(a)
            b_max = b.max()
            
            scaled = b / cur_temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.zeros(len(b))
                w[b.argmax()] = 1.0
            else:
                w /= w_sum
            
            if n_top is not None and n_top < len(b):
                top_idx = np.argpartition(b, -n_top)[-n_top:]
                w_sel = w[top_idx]
                w_sel /= w_sel.sum()
                w_full = np.zeros(len(b))
                w_full[top_idx] = w_sel
                w = w_full
            
            g = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-12:
                    continue
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                idx = np.arange(lo, hi + 1)
                g[idx] += w[k] * a[k - idx]
            
            v = momentum * v - cur_lr * g
            a_new = project_simplex(a + v)
            
            b_new = fast_convolve(a_new)
            obj_new = b_new.max()
            
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            a = a_new

def softmax_log_obj(z, T, n):
    """Soft-max smoothed objective in log-space."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    
    # Gradient
    w = exp_Tb / exp_Tb.sum()
    # cross_corr: grad_a[i] = sum_k w[k] * a[k-i] (valid range)
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad_a[idx] += w[k] * a[k - idx]
    
    S = np.dot(grad_a, a)
    grad_z = a * (2.0 * n * grad_a - a * 2.0 * n * S)
    
    return obj, grad_z


def lbgfs_refine(a0, T, time_limit, maxiter=3000):
    """L-BFGS in log-space with soft-max objective."""
    t0 = time.monotonic()
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-300))
    best_obj = softmax_log_obj(z0, T, n)[0]
    best_z = z0.copy()
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return softmax_log_obj(z, T, n)[0]
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return softmax_log_obj(z, T, n)[1]
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        final_obj = func(res.x)
        if np.isfinite(final_obj) and final_obj < best_obj:
            best_z = res.x
            best_obj = final_obj
    except Exception:
        pass
    
    a_out = np.exp(best_z - best_z.max())
    a_out = a_out / a_out.sum()
    return a_out, best_obj


def coordinate_polish(a0, time_limit):
    """Coordinate descent polish."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    multipliers = np.array([0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0])
    
    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian_init(n, sigma_frac=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma_frac) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    taper = max(1, int(r * n // 2))
    t = np.linspace(0, 1, taper)
    a[:taper] = 0.5 * (1 + np.cos(np.pi * (2 * t / r - 1)))
    a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-10)
    return a / a.sum()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based best-response iteration
combined with subgradient descent, L-BFGS soft-max polishing, and coordinate
descent. The LP finds the optimal "exploit direction" at each step, providing
stronger descent than pure subgradient methods. Active-set reduction speeds up
the LP, and periodic restarts with diverse perturbations escape local minima.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_local(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def build_lp_matrix(a, active_lags):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            rows.append(idx)
            cols.append(i)
            vals.append(a[k - i])
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_lags), n))
    return A


def solve_lp(a, M, active_lags, timeout=20):
    n = len(a)
    A = build_lp_matrix(a, active_lags)
    b = np.full(len(active_lags), M)
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                      method='highs', options={'time_limit': timeout})
        if res.success:
            return -res.fun, res.x
        return None, None
    except Exception:
        return None, None


def lp_single_step(a, n_active=None, timeout=20):
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    if n_active is not None and n_active < 2 * n - 1:
        threshold = M * 0.95
        active_mask = conv >= threshold
        if active_mask.sum() < 10:
            k_min = max(10, n_active // 5)
            active_idx = np.argpartition(conv, -k_min)[-k_min:]
            active_mask = np.zeros(2 * n - 1, dtype=bool)
            active_mask[active_idx] = True
        active_lags = np.where(active_mask)[0]
    else:
        active_lags = np.arange(2 * n - 1)
    val, g = solve_lp(a, M, active_lags, timeout=timeout)
    if val is None:
        return a, M, None
    return a, M, (val, g)


def lp_optimize(a0, time_limit, n_active=None, timeout=20, seed=42):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    no_improve = 0
    it = 0
    while time.monotonic() - t0 < time_limit:
        a, M, result = lp_single_step(a, n_active=n_active, timeout=timeout)
        it += 1
        if result is None:
            a = project_simplex(a + rng.standard_normal(n) * 0.005)
            no_improve += 1
            if no_improve > 20:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.01)
                no_improve = 0
            continue
        val, g = result
        if val <= 1.0 + 1e-5:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.008)
            no_improve += 1
            if no_improve > 30:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.02)
                no_improve = 0
            continue
        t = min(0.4, 0.6 / max(val - 1.0, 0.01))
        g_norm = g / g.sum()
        a_new = (1 - t) * a + t * g_norm
        a_new = np.maximum(a_new, 0.0)
        ss = a_new.sum()
        if ss < 1e-15:
            a_new = best_a.copy()
        else:
            a_new = a_new / ss
        M_new = fast_convolve(a_new).max()
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a_new.copy()
            no_improve = 0
            if it % 20 == 0:
                el = eval_local(best_a)
                print(f"    LP it={it}: M={best_M:.10f}, obj={el:.6f}")
        else:
            no_improve += 1
        a = a_new
        if no_improve > 60:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.015)
            no_improve = 0
    return best_a, best_M


def subgradient_optimize(a0, time_limit, lr_init=0.005, momentum=0.9,
                         temp_init=0.0005, temp_growth=1.02,
                         kick_interval=2000, kick_size=0.003,
                         n_top=None, seed=42):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    cur_temp = temp_init
    no_improve = 0
    it_count = 0
    while time.monotonic() - t0 < time_limit:
        for _ in range(30):
            if time.monotonic() - t0 >= time_limit:
                break
            it_count += 1
            b = fast_convolve(a)
            b_max = b.max()
            scaled = b / cur_temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.zeros(len(b))
                w[b.argmax()] = 1.0
            else:
                w /= w_sum
            if n_top is not None and n_top < len(b):
                top_idx = np.argpartition(b, -n_top)[-n_top:]
                w_sel = w[top_idx]
                ws = w_sel.sum()
                if ws > 1e-30:
                    w_sel /= ws
                else:
                    w_sel = np.ones(len(top_idx)) / len(top_idx)
                w_full = np.zeros(len(b))
                w_full[top_idx] = w_sel
                w = w_full
            g = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-12:
                    continue
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                idx = np.arange(lo, hi + 1)
                g[idx] += w[k] * a[k - idx]
            v = momentum * v - cur_lr * g
            a_new = project_simplex(a + v)
            b_new = fast_convolve(a_new)
            obj_new = b_new.max()
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            a = a_new
        if no_improve > kick_interval:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.7, lr_init * 0.1)
            no_improve = 0
        elif no_improve > kick_interval // 2:
            cur_lr *= 0.95
        cur_temp *= temp_growth
        if cur_temp > 0.5:
            cur_temp = 0.5
    return best_a, best_obj


def soft_max_obj_grad(z, T, n):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    lse = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * lse
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad_a[idx] += w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (2.0 * n * grad_a - 2.0 * n * S * a)
    return obj, grad_z


def lbgfs_refine(a0, T, time_limit, maxiter=3000):
    t0 = time.monotonic()
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-300))
    best_obj = soft_max_obj_grad(z0, T, n)[0]
    best_z = z0.copy()

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return soft_max_obj_grad(z, T, n)[0]

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return soft_max_obj_grad(z, T, n)[1]

    try:
        res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-14,
                                'gtol': 1e-10, 'maxfun': maxiter * 20})
        fo = func(res.x)
        if np.isfinite(fo) and fo < best_obj:
            best_z = res.x
            best_obj = fo
    except Exception:
        pass
    a_out = np.exp(best_z - best_z.max())
    a_out = a_out / a_out.sum()
    return a_out, best_obj


def coordinate_polish(a0, time_limit):
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    multipliers = np.array([0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0])
    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_obj = best_obj
            best_j_val = old_val
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian_init(n, sigma_frac=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma_frac) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    a = np.ones(n)
    taper = max(1, int(r * n // 2))
    t = np.linspace(0, 1, taper)
    a[:taper] = 0.5 * (1 + np.cos(np.pi * (2 * t / r - 1)))
    a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_flat_top_init(n, flat_frac=0.6):
    a = np.ones(n)
    taper = max(1, int((1 - flat_frac) * n // 2))
    if taper > 0 and taper < n // 2:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1 + np.cos(np.pi * (2 * t / (1 - flat_frac) - 1)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-10)
    return a / a.sum()

    print(f"\n=== P8: Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_cd, _ = coordinate_polish(best_seq, time_limit=min(rem, 25.0))
            try_seq(a_cd, "cd")
        rem = deadline - time.monotonic() - 5
        if rem > 8:
            a_fin, _ = lbgfs_refine(best_seq, T=100, time_limit=min(rem, 15.0),
                                    maxiter=1000)
            try_seq(a_fin, "final_lb")

    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")
    return [float(x) for x in best_seq]
