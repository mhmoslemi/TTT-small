import numpy as np

from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_gradient(a, w, n):
    """
    Compute gradient of (2n/tau)*log(sum exp(tau*b)) w.r.t. a,
    where b = conv(a,a) and w = softmax(tau*b).
    grad_i = 4n * sum_j a_j * w_{i+j}
    This is 4n * cross-correlation of w and a.
    """
    # We need result[i] = sum_{j=0}^{n-1} a[j] * w[i+j] for i=0,...,n-1
    # This is np.correlate(w, a, mode='valid')
    # But for large n, use FFT
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # Cross-correlation: ifft(conj(FFT(w)) * FFT(a)) gives correlate(w, a)
    corr = irfft(np.conj(W) * A, fft_size)[:n]
    return 4.0 * n * corr


def smooth_objective_and_grad(a, tau, n):
    """
    Compute the smoothed objective (2n/tau)*log(sum exp(tau*b))
    and its gradient w.r.t. a, where b = conv(a,a).
    """
    b = fast_convolve(a)
    max_b = b.max()
    scaled = tau * (b - max_b)
    exp_scaled = np.exp(scaled)
    log_sum_exp = max_b + np.log(exp_scaled.sum()) / tau
    obj = 2.0 * n * log_sum_exp
    w = exp_scaled / exp_scaled.sum()
    grad = compute_gradient(a, w, n)
    return obj, grad


def simplex_project(v):
    """Project vector v onto the probability simplex (sum=1, all >= 0)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = int(np.nonzero(cond)[0][-1])
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

        a = np.exp(z_shifted)
        a_sum = a.sum()
        a = a / a_sum
        _, grad_a = smooth_objective_and_grad(a, tau, n)
        # Gradient w.r.t. z: a * (grad_a - a * (a dot grad_a))
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        return grad_z

    # Initialize z from a0
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe / a_safe.sum())
    z0 = z0 - z0.max()  # For numerical stability

    try:
        res = minimize(
            fun=func,
            x0=z0,
            jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 20, 'disp': False, 'maxls': 50}
        )
        if stopped[0]:
            return a0
        z_out = res.x
        z_out = z_out - z_out.max()
        a_out = np.exp(z_out)
        a_out = a_out / a_out.sum()
        a_out = np.maximum(a_out, 0)
        return a_out
    except Exception:
        return a0


def coordinate_descent(a0, time_limit, n_steps=50000):
    """
    Coordinate descent: optimize one coordinate at a time.
    For coordinate i, find the best value given all others fixed.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    no_improve = 0
    step = 0

    while time.monotonic() - t0 < time_limit and no_improve < 200:
        if step % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD step={step} ({elapsed:.0f}s): score={best_score:.6f}")

        # Pick a coordinate to optimize
        i = rng.integers(0, n)
        old_val = a[i]
        a_sum_rest = s - old_val  # sum of all other elements

        if a_sum_rest < 1e-15:
            continue

        # The objective as a function of a[i] = x (others fixed, total sum = 1):
        # We need to find x that minimizes max_k conv(a,a)[k] / 1 (since sum=1)
        # conv(a,a)[k] = a[i]^2 * delta_{k,2i} + a[i]*(a[k-i] + a[i-k])_valid + rest_k
        # For a given x, the convolution changes linearly (plus quadratic at k=2i)

        # Binary search for optimal x in [0, 1]
        # Actually, let's try a few candidate values
        candidates = [0.0, old_val * 0.5, old_val * 0.7, old_val * 1.3,
                      1.0 / n, old_val * 0.1, old_val * 3.0]
        # Also try values near current
        if old_val > 0:
            candidates.extend([old_val * f for f in [0.3, 0.8, 1.2, 1.5]])

        best_x = old_val
        best_trial_score = best_score

        for x in candidates:
            x = max(0.0, min(x, 1.0))
            a_trial = a.copy()
            a_trial[i] = x
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_trial_score - 1e-14:
                best_trial_score = score
                best_x = x

        if best_x != old_val:
            a[i] = best_x
            a = a / a.sum()
            score = 2.0 * n * fast_convolve(a).max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        step += 1

    return best_a, best_score


def block_coordinate_descent(a0, time_limit, block_size=8):
    """
    Optimize blocks of consecutive coordinates simultaneously using
    local grid search / golden section.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)
    no_improve = 0
    step = 0

    while time.monotonic() - t0 < time_limit and no_improve < 300:
        step += 1
        if step % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    BCD step={step} ({elapsed:.0f}s): score={best_score:.6f}")

        # Pick a random block
        start = int(rng.integers(0, max(1, n - block_size)))
        end = min(start + block_size, n)
        block = slice(start, end)

        old_block = a[block].copy()
        old_sum = a_sum_rest = a.sum() - old_block.sum()

        if old_sum < 1e-15:
            continue

        # Try random perturbations of the block
        improved = False
        for _ in range(20):
            # Random direction in the block
            direction = rng.standard_normal(end - start)
            # Project to maintain non-negativity
            perturbation = 0.01 * old_block.sum() * direction
            new_block = old_block + perturbation
            if (new_block < -1e-12).any():
                continue
            new_block = np.maximum(new_block, 0)

            a_trial = a.copy()
            a_trial[block] = new_block
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()
                old_block = a[block].copy()
                improved = True

        if improved:
            no_improve = 0
        else:
            no_improve += 1

    return best_a, best_score


def peak_shave_local(a0, time_limit, n_steps=5000):
    """Aggressive local peak shaving with redistribution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(4242)
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit or no_improve > 200:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]

        # Find which pairs contribute to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_contrib = np.argsort(contribs)[::-1][:5]

        improved = False
        for ci in top_contrib:
            i = idx[ci]
            j = peak_k - int(ci)
            old_i = a[i]
            old_j = a[j]
            if old_i < 1e-14:
                continue

            # Try reducing i (and possibly j) and redistributing
            for f1 in [0.85, 0.7, 0.5]:
                for f2 in [1.0, 0.9, 0.8] if i != j else [1.0]:
                    a_trial = a.copy()
                    a_trial[i] = old_i * f1
                    if i != j:
                        a_trial[j] = old_j * f2
                    a_trial = np.maximum(a_trial, 0)
                    diff = a.sum() - a_trial.sum()
                    if diff > 1e-15:
                        # Redistribute to random positions (spread out)
                        n_redist = min(3, n)
                        redist_idx = rng.integers(0, n, size=n_redist)
                        per = diff / n_redist
                        a_trial[redist_idx] += per
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                if improved:
                    break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            # Try a random perturbation to escape
            if no_improve % 50 == 0:
                a = best_a + rng.standard_normal(n) * 0.005
                a = simplex_project(a)
                a = a / a.sum()

    return best_a, best_score


def subgradient_momentum(a0, time_limit, tau=50, lr_init=0.01, momentum=0.9):
    """Subgradient method with momentum, using smoothed max."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    v = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    step_count = 0

    while time.monotonic() - t0 < time_limit:
        step_count += 1

        # Compute gradient
        b = fast_convolve(a)
        max_b = b.max()
        scaled = tau * (b - max_b)
        exp_scaled = np.exp(np.clip(scaled, -500, 0))
        w = exp_scaled / exp_scaled.sum()
        grad = compute_gradient(a, w, n)

        # Gradient step with momentum
        v = momentum * v - lr * grad
        a_candidate = a + v

        # Project onto simplex
        a_new = simplex_project(a_candidate)
        a_new = a_new / a_new.sum()

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.999
        if lr < 1e-6:
            lr = 1e-6

        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 1000000)
            a = best_a + rng.standard_normal(n) * 0.01
            a = simplex_project(a)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 2, 0.005)
            no_improve = 0

        if step_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad step={step_count} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def diverse_initializations(n, rng):
    """Generate diverse initial sequences for multi-start."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine window
    t = np.linspace(0, n - 1, n)
    a = np.cos(np.pi * t / (2 * n)) ** 2
    a = np.maximum(a, 0)
    inits.append(a / a.sum())

    # Raised cosine
    a = 0.5 * (1 + np.cos(2 * np.pi * t / n))
    inits.append(a / a.sum())

    # Exponential decay
    a = np.exp(-t / (n / 3))
    inits.append(a / a.sum())

    # Chirp-like (quadratic phase in frequency -> linear FM in time)
    freq = np.linspace(0, 2, n)
    a = np.abs(np.exp(1j * np.pi * (t / n) ** 2 * 4))
    a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t / n)
    a = np.maximum(a, 0)
    if a.sum() > 0:
        inits.append(a / a.sum())

    # Random
    a = rng.random(n)
    inits.append(a / a.sum())

    # Alternating
    a = np.ones(n)
    a[::2] = 1.0
    a[1::2] = 0.3
    inits.append(a / a.sum())

    # Gaussian bump
    center = n / 2
    a = np.exp(-((t - center) ** 2) / (2 * (n / 6) ** 2))
    inits.append(a / a.sum())

    return inits


def propose_candidate():
    """
    Multi-phase optimization for minimum peak auto-correlation sequences:
    1. Load parent sequence
    2. L-BFGS in log-domain with adaptive temperature
    3. Subgradient with momentum
    4. Block coordinate descent
    5. Peak shaving local search
    6. Multi-length exploration
    7. Final polish with multiple methods
    """
    budget = 1800.0
    deadline = time.monotonic() + budget
    rng = np.random.default_rng(2024)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        # Fallback: start with uniform
        n_start = 768
        best_seq = np.ones(n_start) / n_start
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Fallback uniform: n={n_start}, score={best_score:.6f}")

    n_best = len(best_seq)

    # === Phase 1: L-BFGS in log-domain with multiple temperatures ===
    print(f"\n=== Phase 1: L-BFGS log-domain (best={best_score:.6f}) ===")
    elapsed = time.monotonic()
    rem = deadline - elapsed - 200
    if rem > 30:
        # Use multiple temperatures, starting warm and cooling
        temperatures = [200, 100, 50, 25, 10]
        time_per_T = min(rem / len(temperatures), 120.0)
        current = best_seq.copy()
        for T in temperatures:
            rem_T = deadline - time.monotonic() - 100
            if rem_T < 10:
                break
            t_alloc = min(time_per_T, rem_T * 0.7)
            print(f"  L-BFGS tau={T}, n={n_best}, budget={t_alloc:.0f}s")
            a_new = lbgfs_log_domain(current, T, t_alloc, maxiter=3000)
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            best_seq, best_score = check_and_update(a_new, f"lbgfs_T{T}", best_seq, best_score)
            # Update current to best so far (warm start for next temperature)
            if len(a_new) == n_best:
                current = a_new

    # === Phase 2: Subgradient with momentum ===
    print(f"\n=== Phase 2: Subgradient momentum (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 150
    if rem > 20:
        t_alloc = min(rem * 0.4, 150.0)
        print(f"  SubGrad n={n_best}, budget={t_alloc:.0f}s")
        a_sg = subgradient_momentum(best_seq, t_alloc, tau=50, lr_init=0.005, momentum=0.9)

        a_sg = np.maximum(a_sg, 0)
        a_sg = a_sg / a_sg.sum()
        best_seq, best_score = check_and_update(a_sg, "subgrad", best_seq, best_score)

        # Second subgradient pass with different params
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_alloc2 = min(rem * 0.5, 100.0)
            a_sg2 = subgradient_momentum(a_sg, t_alloc2, tau=100, lr_init=0.003, momentum=0.95)
            a_sg2 = np.maximum(a_sg2, 0)
            a_sg2 = a_sg2 / a_sg2.sum()
            best_seq, best_score = check_and_update(a_sg2, "subgrad2", best_seq, best_score)

    # === Phase 3: Block coordinate descent ===
    print(f"\n=== Phase 3: Block CD (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 100
    if rem > 15:
        t_alloc = min(rem * 0.4, 100.0)
        print(f"  BCD n={n_best}, budget={t_alloc:.0f}s")
        a_bcd = block_coordinate_descent(best_seq, t_alloc, block_size=8)
        a_bcd = np.maximum(a_bcd, 0)
        a_bcd = a_bcd / a_bcd.sum()
        best_seq, best_score = check_and_update(a_bcd, "bcd", best_seq, best_score)

    # === Phase 4: Peak shaving ===
    print(f"\n=== Phase 4: Peak shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 80
    if rem > 10:
        t_alloc = min(rem * 0.5, 80.0)
        print(f"  PeakShave n={n_best}, budget={t_alloc:.0f}s")
        a_ps = peak_shave_local(best_seq, t_alloc, n_steps=10000)
        a_ps = np.maximum(a_ps, 0)
        a_ps = a_ps / a_ps.sum()
        best_seq, best_score = check_and_update(a_ps, "peak_shave", best_seq, best_score)

    # === Phase 5: Another L-BFGS pass (refined) ===
    print(f"\n=== Phase 5: L-BFGS refined (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 60
    if rem > 15:
        for T in [30, 15]:
            rem_T = deadline - time.monotonic() - 30
            if rem_T < 10:
                break
            t_alloc = min(rem_T * 0.5, 60.0)
            print(f"  L-BFGS refined tau={T}, budget={t_alloc:.0f}s")
            a_new = lbgfs_log_domain(best_seq, T, t_alloc, maxiter=3000)
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            best_seq, best_score = check_and_update(a_new, f"lbgfs_ref_T{T}", best_seq, best_score)

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 40
    if rem > 60:
        alt_lengths = []
        for factor in [0.75, 1.25, 1.5, 0.5, 2.0]:
            alt_n = int(n_best * factor)
            if 128 <= alt_n <= 4096 and alt_n != n_best and alt_n not in alt_lengths:
                alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:4]:
            rem_all = deadline - time.monotonic() - 20
            if rem_all < 30:
                break
            t_alloc = min(rem_all * 0.35, 80.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            print(f"  Alt length n={alt_n}, budget={t_alloc:.0f}s")

            # Quick L-BFGS
            a_alt = lbgfs_log_domain(init_alt, 50, t_alloc * 0.6, maxiter=2000)
            a_alt = np.maximum(a_alt, 0)
            a_alt = a_alt / a_alt.sum()
            best_seq, best_score = check_and_update(a_alt, f"alt_n{alt_n}_lbgfs", best_seq, best_score)

            # Subgradient
            rem_sg = deadline - time.monotonic() - 15
            if rem_sg > 10 and len(a_alt) == alt_n:
                a_alt2 = subgradient_momentum(a_alt, min(rem_sg * 0.5, 30.0), tau=50, lr_init=0.003)
                a_alt2 = np.maximum(a_alt2, 0)
                a_alt2 = a_alt2 / a_alt2.sum()
                best_seq, best_score = check_and_update(a_alt2, f"alt_n{alt_n}_sg", best_seq, best_score)

    # === Phase 7: Final micro-polish ===
    print(f"\n=== Phase 7: Final polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 10
    if rem > 5:
        # Quick peak shave
        t_final = min(rem * 0.4, 20.0)
        a_fin1 = peak_shave_local(best_seq, t_final, n_steps=3000)
        a_fin1 = np.maximum(a_fin1, 0)
        a_fin1 = a_fin1 / a_fin1.sum()
        best_seq, best_score = check_and_update(a_fin1, "final_ps", best_seq, best_score)

        rem = deadline - time.monotonic() - 5
        if rem > 5:
            # Quick L-BFGS with very low temperature
            t_fin2 = min(rem * 0.6, 15.0)
            a_fin2 = lbgfs_log_domain(best_seq, 10, t_fin2, maxiter=2000)
            a_fin2 = np.maximum(a_fin2, 0)
            a_fin2 = a_fin2 / a_fin2.sum()
            best_seq, best_score = check_and_update(a_fin2, "final_lbgfs", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = np.ones(768) / 768
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
