def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize_improved(f0, time_limit, verbose=False, use_away=True):
    """
    LP-based Frank-Wolfe with away steps and golden-section line search.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    # Track vertices for away steps
    vertex = f.copy()  # The most recent FW vertex
    vertex_count = 1

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2.0:
            break
            
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp = min(m_exp * 1.3, 20.0)
            if m_exp > 20.0:
                # Restart
                rng = np.random.default_rng(it * 17 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp = min(m_exp * 1.1, 10.0)
            continue
        g = g / g_sum

        # Check convergence
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                m_exp = min(m_exp * 1.05, 5.0)
            if no_improve > 100:
                # Perturb and restart
                rng = np.random.default_rng(it * 31 + 7)
                noise_scale = 0.002 + 0.003 * min(1, no_improve / 200)
                f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
                vertex = g.copy()
                vertex_count = 1
            continue
        else:
            no_improve = 0

        # Golden section line search
        best_alpha = 0.0
        best_val = M
        
        # First, coarse search
        alphas_coarse = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
        for alpha in alphas_coarse:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0.05:
            # Fine golden section search around best_alpha
            lo = max(0.0, best_alpha - 0.15)
            hi = min(1.0, best_alpha + 0.15)
            gr = (np.sqrt(5) - 1) / 2  # golden ratio ~0.618
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            
            ft_c = np.maximum(f + c * (g - f), 0.0)
            ft_c = ft_c / ft_c.sum()
            val_c = fast_convolve(ft_c).max()
            
            ft_d = np.maximum(f + d * (g - f), 0.0)
            ft_d = ft_d / ft_d.sum()
            val_d = fast_convolve(ft_d).max()
            
            for _ in range(12):
                if val_c < val_d:
                    hi = d
                    d = c
                    val_d = val_c
                    c = hi - gr * (hi - lo)
                    ft_c = np.maximum(f + c * (g - f), 0.0)
                    ft_c = ft_c / ft_c.sum()
                    val_c = fast_convolve(ft_c).max()
                else:
                    lo = c
                    c = d
                    val_c = val_d
                    d = lo + gr * (hi - lo)
                    ft_d = np.maximum(f + d * (g - f), 0.0)
                    ft_d = ft_d / ft_d.sum()
                    val_d = fast_convolve(ft_d).max()
            
            best_alpha = (lo + hi) / 2.0
            ft = np.maximum(f + best_alpha * (g - f), 0.0)
            ft = ft / ft.sum()
            best_val = fast_convolve(ft).max()

        # Away step (if enabled)
        if use_away and vertex_count >= 2 and np.max(np.abs(vertex - g)) > 0.01:
            # Try away step: f - beta * (vertex - f)
            # This moves away from the old vertex
            best_away_val = best_val
            best_away_beta = 0.0
            for beta in [0.1, 0.2, 0.3, 0.5]:
                ft_away = f - beta * (vertex - f)
                ft_away = np.maximum(ft_away, 0.0)
                st = ft_away.sum()
                if st < 1e-15:
                    continue
                ft_away = ft_away / st
                val = fast_convolve(ft_away).max()
                if val < best_away_val:
                    best_away_val = val
                    best_away_beta = beta
            
            if best_away_beta > 0 and best_away_val < best_val:
                f_new = f - best_away_beta * (vertex - f)
                f_new = np.maximum(f_new, 0.0)
                f_new = f_new / f_new.sum()
                M = fast_convolve(f_new).max()
                f = f_new
                vertex = g.copy()
                vertex_count += 1
                m_exp = 1.0
                if M < best_obj - 1e-14:
                    best_obj = M
                    best_f = f.copy()
                    if verbose and it % 20 == 0:
                        print(f"      LP-FW(away) iter {it}: obj={best_obj:.10f}")
                continue

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv_new = fast_convolve(f_new)
            M = conv_new.max()
            f = f_new
            vertex = g.copy()
            vertex_count += 1
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 20 == 0:
                    print(f"      LP-FW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 5.0)

        if no_improve > 200:
            break

    return best_f, best_obj


def smoothed_mirror_descent_v2(f0, time_limit, verbose=False,
                                tau_init=5.0, tau_decay=1.005,
                                lr_init=0.02, lr_decay=0.9997,
                                restart_every=150, top_frac=0.2):
    """
    Improved smoothed mirror descent with better parameter scheduling.
    """
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    tau = tau_init
    lr = lr_init
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Smoothed gradient via top-k peaks with softmax
        len_c = len(conv)
        k_top = max(10, int(len_c * top_frac))
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        scaled = tau * pv
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute weighted subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            idx = np.arange(js, je + 1)
            contrib = f[k - idx] + f[idx]
            sg[js:je + 1] += w[i] * contrib

        # Multiplicative (mirror) descent step
        f_new = f * np.exp(-lr * sg)
        f_new = np.maximum(f_new, 1e-15)
        s = f_new.sum()
        if s < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / s

        # Line search with backtracking
        obj_new = fast_convolve(f_new).max()
        if obj_new > M * 1.02:
            for damp in [0.5, 0.25, 0.1, 0.05]:
                f_trial = f * (1.0 - damp) + f_new * damp
                f_trial = np.maximum(f_trial, 1e-15)
                f_trial = f_trial / f_trial.sum()
                obj_trial = fast_convolve(f_trial).max()
                if obj_trial < M:
                    f_new = f_trial
                    obj_new = obj_trial
                    break
            else:
                lr *= 0.5
                continue

        f = f_new

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
            if verbose and iter_count % 200 == 0:
                print(f"      SMD2 iter {iter_count}: obj={best_obj:.10f}, tau={tau:.1f}, lr={lr:.6f}")
        else:
            no_improve += 1

        # Adaptive parameters
        tau *= tau_decay
        lr = max(lr * lr_decay, 0.5 * lr)  # Slow decay

        # Periodic restart
        if no_improve > restart_every:
            rng = np.random.default_rng(iter_count * 7 + 13)
            noise_scale = 0.002 + 0.003 * (no_improve / restart_every)
            f = np.maximum(best_f + rng.standard_normal(n) * noise_scale, 1e-15)
            f = f / f.sum()
            lr = lr_init * 0.5
            no_improve = 0

        if lr < 1e-8:
            break

    return best_f, best_obj


def peak_targeted_polish(f0, time_limit, verbose=False):
    """
    Fine polishing: target the single worst peak with small multiplicative updates.
    """
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.005
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Find top peaks
        k_best = int(np.argmax(conv))
        # Get contributors
        js = max(0, k_best - n + 1)
        je = min(n - 1, k_best)
        idx = np.arange(js, je + 1)
        contrib = f[k_best - idx] + f[idx]
        
        # Also consider second peak
        conv_masked = conv.copy()
        conv_masked[k_best] = -np.inf
        k_second = int(np.argmax(conv_masked))
        
        js2 = max(0, k_second - n + 1)
        je2 = min(n - 1, k_second)
        idx2 = np.arange(js2, je2 + 1)
        contrib2 = f[k_second - idx2] + f[idx2]

        grad = np.zeros(n)
        grad[js:je + 1] += contrib
        grad[js2:je2 + 1] += 0.3 * contrib2

        # Multiplicative step (very small)
        f_new = f * np.exp(-lr * grad)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        obj_new = fast_convolve(f_new).max()
        if obj_new < M - 1e-15:
            f = f_new
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"      PTP iter {it}: obj={best_obj:.10f}")
            lr = min(lr * 1.01, 0.02)
        else:
            lr *= 0.7
            no_improve += 1

        if no_improve > 200:
            rng = np.random.default_rng(it * 41 + 19)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.002, 1e-15)
            f = f / f.sum()
            lr = 0.005
            no_improve = 0

        if lr < 1e-10:
            break

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, verbose=False):
    """Coordinate descent: optimize each coordinate via 1D search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    ss = f.sum()
    if ss < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / ss
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    # Precompute which convolution positions each coordinate affects
    # Coefficient j affects conv[k] for k in [j, j+n-1]
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        rng = np.random.default_rng(int((time.monotonic() - t0) * 1000) % 100000 + 1)
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj

            old_val = f[j]
            if old_val < 1e-14:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            # Try multiplicative perturbations
            for factor in [0.5, 0.7, 0.8, 1.2, 1.3, 1.5, 0.3, 2.0]:
                new_val = old_val * factor
                if new_val < 0 or new_val > 0.5:
                    continue
                # Renormalize keeping sum = 1
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                sst = f_trial.sum()
                if sst < 1e-15:
                    continue
                f_trial = f_trial / sst
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def truncate_sequence(f, new_n):
    """Truncate sequence to new_n elements (keep the front)."""
    if new_n >= len(f):
        return f[:new_n].copy() if new_n < len(f) else f.copy()
    return f[:new_n].copy()


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization with away-step LP-FW, diverse initialization,
    multi-length exploration, and coordinate descent polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 60.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Diverse Initializations ===
    print("=== Phase 1: Diverse Initializations ===")
    init_candidates = []
    
    # Exponential with various tau fractions
    test_lengths = [512, 640, 768, 900, 1024]
    
    for n_try in test_lengths:
        for tau_frac in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
            a_init = make_exponential_init(n_try, tau_frac)
            init_candidates.append((a_init, f"exp_t{tau_frac}_n{n_try}"))
        
        # Bimodal
        a_init = make_bimodal_init(n_try, 0.2, 0.6)
        init_candidates.append((a_init, f"bimodal_n{n_try}"))
        
        # Poly-exponential
        a_init = make_poly_exponential_init(n_try, 0.5)
        init_candidates.append((a_init, f"polyexp_n{n_try}"))

    # Parent
    if parent is not None:
        init_candidates.append((parent, "parent"))

    # Score all inits
    scored_inits = []
    for a_init, label in init_candidates:
        s = a_init.sum()
        if s < 1e-10:
            continue
        a_norm = a_init / s
        conv = fast_convolve(a_norm)
        obj = 2 * len(a_norm) * conv.max()
        scored_inits.append((obj, a_norm, label))

    scored_inits.sort(key=lambda x: x[0])
    print(f"  Total inits: {len(scored_inits)}")
    print(f"  Top 5: {[f'{l}:{o:.4f}' for o, _, l in scored_inits[:5]]}")

    # === Phase 2: LP-FW on top initializations ===
    print("\n=== Phase 2: LP-FW on Top Inits ===")
    top_k = min(4, len(scored_inits))
    for idx in range(top_k):
        rem = deadline - time.monotonic() - 300
        if rem < 30:
            break
        a_init, label = scored_inits[idx][1], scored_inits[idx][2]
        t_alloc = min(rem / max(1, top_k - idx), 250)
        if t_alloc < 20:
            t_alloc = 20
        print(f"  LP-FW on {label} (n={len(a_init)}): t={t_alloc:.0f}s")
        a_opt, obj = lp_fw_optimize_improved(a_init, time_limit=t_alloc, verbose=True, use_away=True)
        try_seq(a_opt, f"lpfw_{label}")

    # === Phase 3: SMD on best ===
    print("\n=== Phase 3: SMD on Best ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 30:
            t_alloc = min(rem, 150)
            print(f"  SMD on best (n={len(best_seq)}): t={t_alloc:.0f}s")
            a_opt, obj = smoothed_mirror_descent_v2(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "smd_best")

    # === Phase 4: Peak-targeted polish ===
    print("\n=== Phase 4: Peak-Targeted Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_alloc = min(rem, 100)
            print(f"  PTP (n={len(best_seq)}): t={t_alloc:.0f}s")
            a_opt, obj = peak_targeted_polish(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "ptp")

    # === Phase 5: Multi-length exploration ===
    print("\n=== Phase 5: Multi-Length Exploration ===")
    if best_seq is not None:
        base = best_seq
        for target_n in [640, 768, 850, 950, 1024, 1150, 1280]:
            rem = deadline - time.monotonic() - 150
            if rem < 30:
                break
            if len(base) == target_n:
                continue
            f_interp = interpolate_sequence(base, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")
            if sc < best_score + 0.01:
                t_alloc = min(rem * 0.3, 60)
                if t_alloc < 15:
                    t_alloc = 15
                print(f"  LP-FW n={target_n}: t={t_alloc:.0f}s")
                a_opt, obj = lp_fw_optimize_improved(f_interp, time_limit=t_alloc, use_away=True)
                try_seq(a_opt, f"lpfw_n{target_n}")

    # === Phase 6: Truncation exploration ===
    print("\n=== Phase 6: Truncation ===")
    if best_seq is not None and len(best_seq) > 512:
        for trunc_n in [len(best_seq) - 64, len(best_seq) - 32, len(best_seq) + 32, len(best_seq) + 64]:
            if trunc_n < 100:
                continue
            if trunc_n == len(best_seq):
                continue
            if trunc_n < len(best_seq):
                a_trunc = best_seq[:trunc_n].copy()
            else:
                # Extend with small values
                a_ext = np.zeros(trunc_n)
                a_ext[:len(best_seq)] = best_seq
                # Fill remaining with small tail
                tail_len = trunc_n - len(best_seq)
                a_ext[len(best_seq):] = best_seq[-1] * np.exp(-np.arange(tail_len) / max(1, tail_len * 0.5))
            
            a_trunc = np.maximum(a_trunc, 0)
            s = a_trunc.sum()
            if s < 1e-10:
                continue
            a_trunc = a_trunc / s
            sc = try_seq(a_trunc, f"trunc_n{trunc_n}")
    
    # === Phase 7: Random restarts ===
    print("\n=== Phase 7: Random Restarts ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for r in range(3):
            rem = deadline - time.monotonic() - 100
            if rem < 25:
                break
            t_alloc = min(rem / max(1, 3 - r), 80)
            if t_alloc < 15:
                t_alloc = 15

            if r == 0:
                noise = rng.standard_normal(n_best) * 0.004
                f_init = np.maximum(best_seq + noise, 1e-15)
                f_init = f_init / f_init.sum()
            elif r == 1:
                # Blend with exponential
                tau_frac = rng.uniform(0.3, 1.5)
                exp_init = make_exponential_init(n_best, tau_frac)
                f_init = 0.7 * best_seq + 0.3 * exp_init
                f_init = f_init / f_init.sum()
            else:
                # Mirror + noise
                f_init = 0.5 * best_seq + 0.5 * best_seq[::-1]
                f_init += rng.standard_normal(n_best) * 0.003
                f_init = np.maximum(f_init, 1e-15)
                f_init = f_init / f_init.sum()

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_objective_normalized(f):
    """Compute 2*n*max(conv(f,f)) assuming sum(f)=1."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def make_exponential_init(n, tau_frac):
    """Create exponential decay initialization with tau = tau_frac * n."""
    tau = tau_frac * n
    j = np.arange(n)
    a = np.exp(-j / tau)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_bimodal_init(n, frac1, frac2, weight2=0.6):
    """Create bimodal initialization."""
    j = np.arange(n)
    a = np.exp(-40 * (j / n - frac1) ** 2) + weight2 * np.exp(-40 * (j / n - frac2) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:

def make_poly_exponential_init(n, p, tau_frac=0.5):
    """Poly-exponential: (j+1)^p * exp(-j/tau)."""
    tau = tau_frac * n

    j = np.arange(n)

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, n):
    """Build the LP constraint matrix A where A[k,j] = f[k-j] if valid."""
    A = np.zeros((2 * n - 1, n))

    return A


def solve_lp_direction(f, M, n, A_cached=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c_obj = -np.ones(n)

        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
    else:

    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj

            old_val = f[j]
            if old_val < 1e-14:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for factor in [0.4, 0.6, 0.8, 1.2, 1.4, 1.8, 0.2, 2.5]:
                new_val = old_val * factor
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                sst = f_trial.sum()
                if sst < 1e-15:
                    continue
                f_trial = f_trial / sst
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def try_sequence(a, best_seq, best_score, label=""):
    """Try a sequence and return updated (best_seq, best_score)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score


def sort_by_obj(arr):
    """Sort list of (obj, data, label) tuples by obj."""
    return sorted(arr, key=lambda x: x[0])


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization: diverse asymmetric initialization ->
    LP-FW with away steps -> SMD -> peak polish -> multi-length -> CD.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 65.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Diverse Initializations ===
    print("=== Phase 1: Diverse Initializations ===")
    init_candidates = []

    test_lengths = [512, 640, 768, 896, 1024]

    for n_try in test_lengths:
        # Exponential with various tau
        for tau_frac in [0.25, 0.4, 0.55, 0.75, 1.0, 1.3, 1.8]:
            a_init = make_exponential_init(n_try, tau_frac)
            init_candidates.append((a_init, "exp_t" + str(tau_frac) + "_n" + str(n_try)))

        # Bimodal
        a_init = make_bimodal_init(n_try, 0.2, 0.6)
        init_candidates.append((a_init, "bimodal_n" + str(n_try)))

        # Poly-exponential
        a_init = make_poly_exponential_init(n_try, 0.3, 0.5)
        init_candidates.append((a_init, "polyexp_n" + str(n_try)))

        # Cosine window
        a_init = make_cosine_window_init(n_try, 0.25, 0.35)
        init_candidates.append((a_init, "coswin_n" + str(n_try)))

    # Parent
    if parent is not None:
        init_candidates.append((parent, "parent"))

    # Score all inits
    scored_inits = []
    for a_init, label in init_candidates:
        s = a_init.sum()
        if s < 1e-10:
            continue
        a_norm = a_init / s
        conv = fast_convolve(a_norm)
        obj = 2 * len(a_norm) * conv.max()
        scored_inits.append((obj, a_norm, label))

    scored_inits = sort_by_obj(scored_inits)
    print(f"  Total inits: {len(scored_inits)}")
    top_labels = [scored_inits[i][2] + ":" + format(scored_inits[i][0], '.4f') for i in range(min(5, len(scored_inits)))]
    print(f"  Top 5: {top_labels}")

    # === Phase 2: LP-FW on top initializations ===
    print("\n=== Phase 2: LP-FW on Top Inits ===")
    top_k = min(4, len(scored_inits))
    for idx in range(top_k):
        rem = deadline - time.monotonic() - 400
        if rem < 30:
            break
        a_init = scored_inits[idx][1]
        label = scored_inits[idx][2]
        t_alloc = min(rem / max(1, top_k - idx), 200)
        if t_alloc < 20:
            t_alloc = 20
        print(f"  LP-FW on {label} (n={len(a_init)}): t={t_alloc:.0f}s")
        a_opt, obj = lp_fw_optimize_improved(a_init, time_limit=t_alloc, verbose=True, use_away=True)
        best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "lpfw_" + label)

    # === Phase 3: SMD on best ===
    print("\n=== Phase 3: SMD on Best ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_alloc = min(rem, 150)
            print(f"  SMD on best (n={len(best_seq)}): t={t_alloc:.0f}s")
            a_opt, obj = smoothed_mirror_descent_v2(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "smd_best")

    # === Phase 4: LP-FW again on best (refined) ===
    print("\n=== Phase 4: LP-FW Refinement ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 30:
            t_alloc = min(rem, 150)
            print(f"  LP-FW refined (n={len(best_seq)}): t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize_improved(best_seq, time_limit=t_alloc, verbose=True, use_away=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "lpfw_refined")

    # === Phase 5: Multi-length exploration ===
    print("\n=== Phase 5: Multi-Length Exploration ===")
    if best_seq is not None:
        base = best_seq
        for target_n in [640, 768, 850, 950, 1024, 1150, 1280, 1400]:
            rem = deadline - time.monotonic() - 200
            if rem < 30:
                break
            if len(base) == target_n:
                continue
            f_interp = interpolate_sequence(base, target_n)
            best_seq, best_score = try_sequence(f_interp, best_seq, best_score, "interp_n" + str(target_n))
            if best_score < 999 and f_interp.sum() > 1e-10:
                sc = compute_objective_normalized(f_interp)
                if sc < best_score + 0.02:
                    t_alloc = min(rem * 0.35, 60)
                    if t_alloc < 15:
                        t_alloc = 15
                    print(f"  LP-FW n={target_n}: t={t_alloc:.0f}s")
                    a_opt, obj = lp_fw_optimize_improved(f_interp, time_limit=t_alloc, use_away=True)
                    best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "lpfw_n" + str(target_n))

    # === Phase 6: Truncation exploration ===
    print("\n=== Phase 6: Truncation ===")
    if best_seq is not None and len(best_seq) > 400:
        bl = len(best_seq)
        for trunc_n in [bl - 32, bl - 64, bl - 96, bl + 32, bl + 64, bl + 96]:
            rem = deadline - time.monotonic() - 150
            if rem < 20:
                break
            if trunc_n < 200 or trunc_n == bl:
                continue
            if trunc_n < bl:
                a_trunc = best_seq[:trunc_n].copy()
            else:
                a_ext = np.zeros(trunc_n)
                a_ext[:bl] = best_seq
                tail_len = trunc_n - bl
                decay = np.exp(-np.arange(tail_len) / max(1, tail_len * 0.4))
                a_ext[bl:] = best_seq[-1] * decay
            a_trunc = np.maximum(a_trunc, 0)
            s = a_trunc.sum()
            if s < 1e-10:
                continue
            a_trunc = a_trunc / s
            best_seq, best_score = try_sequence(a_trunc, best_seq, best_score, "trunc_n" + str(trunc_n))

    # === Phase 7: Random restarts with LP-FW ===
    print("\n=== Phase 7: Random Restarts ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for r in range(4):
            rem = deadline - time.monotonic() - 100
            if rem < 25:
                break
            t_alloc = min(rem / max(1, 4 - r), 60)
            if t_alloc < 15:
                t_alloc = 15

            if r == 0:
                noise = rng.standard_normal(n_best) * 0.003
                f_init = np.maximum(best_seq + noise, 1e-15)
                f_init = f_init / f_init.sum()
            elif r == 1:
                tau_frac = rng.uniform(0.3, 1.2)
                exp_init = make_exponential_init(n_best, tau_frac)
                f_init = 0.7 * best_seq + 0.3 * exp_init
                f_init = f_init / f_init.sum()
            elif r == 2:
                f_init = 0.6 * best_seq + 0.4 * best_seq[::-1]
                f_init += rng.standard_normal(n_best) * 0.003
                f_init = np.maximum(f_init, 1e-15)
                f_init = f_init / f_init.sum()
            else:
                # Larger perturbation + SMD first
                noise = rng.standard_normal(n_best) * 0.008
                f_init = np.maximum(best_seq + noise, 1e-15)
                f_init = f_init / f_init.sum()
                # Quick SMD to settle
                a_smd, _ = smoothed_mirror_descent_v2(f_init, time_limit=min(20, t_alloc * 0.3))
                f_init = a_smd

            print(f"  Restart {r}: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize_improved(f_init, time_limit=t_alloc, use_away=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "restart_" + str(r))

    # === Phase 8: Peak-targeted polish ===
    print("\n=== Phase 8: Peak-Targeted Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 20:
            t_alloc = min(rem, 60)
            print(f"  PTP (n={len(best_seq)}): t={t_alloc:.0f}s")
            a_opt, obj = peak_targeted_polish(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "ptp")

    # === Phase 9: Coordinate Descent ===
    print("\n=== Phase 9: Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_alloc = min(rem, 50)
            print(f"  CD: t={t_alloc:.0f}s")
            a_opt, obj = coordinate_descent_polish(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "cd")

    # === Phase 10: Final LP-FW squeeze ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t_alloc = min(rem, 30)
            print(f"  Final LP-FW: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize_improved(best_seq, time_limit=t_alloc, use_away=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
