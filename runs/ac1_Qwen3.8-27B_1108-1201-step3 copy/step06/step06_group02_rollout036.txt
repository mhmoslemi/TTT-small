def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-like initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    """Bell-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine(n, freq, amp):
    """Cosine-modulated uniform sequence."""
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_inverse_gaussian(n, sigma_frac=0.3):
    """Inverse Gaussian - concentrated at edges."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(0.5 * (x / sigma) ** 2)
    # Clip to avoid overflow
    a = np.minimum(a, 1e6)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_two_peak(n, separation=0.3):
    """Two-peaked sequence."""
    x = np.arange(n) / n
    a = np.exp(-20 * (x - 0.35) ** 2) + np.exp(-20 * (x - 0.65) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def spectral_flatten(f0, time_limit, verbose=False):
    """
    Spectral shaping: iteratively modify sequence to flatten its power spectrum.
    The idea: if |A(f)|^2 is flat, then the auto-convolution is as flat as possible.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    lr = 0.001
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > 100:
            break
        
        # Compute power spectrum
        fft_size = 1 << (2 * n - 1).bit_length()  # Wait, (2n-1).bit_length() might be off
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        A = np.fft.rfft(f, fft_size)
        power = np.abs(A) ** 2
        
        # Target: flat power spectrum
        # The mean power is sum(|A|^2) / fft_size (by Parseval)
        target = power.mean()
        
        # Error in time domain: inverse FFT of (power - target)
        # Actually, we want to modify f to make power flatter.
        # The gradient of log(power) with respect to f in the real domain...
        # This gets complicated. Let's use a simpler approach.
        
        # Alternative: compute the "excess" at each lag and use it as subgradient
        conv = fast_convolve(f)
        M = conv.max()
        
        # Weighted subgradient emphasizing peaks more
        # Use exponential weighting
        scaled = (conv - conv.min()) / (conv.max() - conv.min() + 1e-15)
        w = np.exp(5.0 * scaled)
        w /= w.sum()
        
        sg = np.zeros(n)
        # Only use top 20% of lags for speed
        threshold = np.percentile(conv, 80)
        active = np.where(conv >= threshold)[0]
        
        for k in active:
            if w[k] < 1e-10:
                continue
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            sg[js:je+1] += 2.0 * w[k] * f[k - np.arange(js, je + 1)]
        
        # Normalize subgradient
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        sg = sg / sg_norm
        
        # Step
        f_new = project_simplex(f - lr * sg * sg_norm)  # Scale back
        val = fast_convolve(f_new).max()
        
        if val < best_obj - 1e-14:
            best_obj = val
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.05, 0.01)
        else:
            no_improve += 1
            lr *= 0.8
    
    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- LP-Frank-Wolfe with adaptive M relaxation (workhorse optimizer)
- Nesterov-momentum subgradient descent for faster convergence
- Multi-length exploration (parent n + interpolated larger n)
- Spectral-aware initialization and diverse restarts
- Focused time allocation: LP-FW first, then subgradient, then polish

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_normalized(a):
    """Compute 2*n*max(conv) for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_sg_single_peak(f, k):
    """Compute subgradient of (f*f)[k] with respect to f."""
    n = len(f)
    sg = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    if js > je:
        return sg
    sg[js:je + 1] = 2.0 * f[k - np.arange(js, je + 1)]
    return sg


def compute_sg_weighted(f, peaks, weights):
    """Compute weighted subgradient over multiple peaks."""
    n = len(f)
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        sg[js:je + 1] += 2.0 * weights[i] * f[k - np.arange(js, je + 1)]
    return sg


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-Frank-Wolfe with adaptive M relaxation and line search."""
    f = normalize(f0)
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = [1.0, 1.01, 1.03, 1.05, 1.1, 1.2, 1.5, 2.0, 3.0, 5.0]
    m_idx = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1.0:
            break

        conv = fast_convolve(f)
        M_current = conv.max()

        g = None
        for level_try in range(m_idx, len(m_relax)):
            M_try = M_current * m_relax[level_try]
            g = solve_lp_direction(f, M_try, n)
            if g is not None and g.sum() > 1e-12:
                m_idx = level_try
                break
            m_idx = level_try + 1

        if g is None or g.sum() < 1e-12:
            rng = np.random.default_rng(it * 13 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            no_improve += 1
            if no_improve > 40:
                break
            continue

        g = g / g.sum()
        diff = np.max(np.abs(g - f))
        if diff < 1e-10:
            no_improve += 1
            m_idx = min(m_idx + 1, len(m_relax) - 1)
            if no_improve > 30:
                break
            continue
        no_improve = 0

        # Adaptive line search
        best_alpha = 0.0
        best_val = M_current
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha < 1e-10:
            no_improve += 1
            m_idx = min(m_idx + 1, len(m_relax) - 1)
            continue

        f_new = np.maximum(f + best_alpha * (g - f), 0.0)
        f_new = f_new / f_new.sum()
        f = f_new
        m_idx = 0

            best_obj = new_obj
            best_f = f.copy()
            if verbose and it % 5 == 0:
                print(f"    LP-FW iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def fast_subgradient(f0, time_limit, verbose=False, n_top=15):
    """
    Fast subgradient descent with top-peak weighting and adaptive step size.
    Uses multiple peaks weighted by proximity to max for smoother descent.
    """
    f = normalize(f0)
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    lr = 0.003
    it = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > 400:
            rng = np.random.default_rng(it * 7919 + 3)
            perturb = rng.standard_normal(n) * 0.004
            f = project_simplex(best_f + perturb)
            conv = fast_convolve(f)
            no_improve = 0
            lr = 0.003
            continue

        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Get top peaks
        k_max = min(n_top, len(conv))
        top_idx = np.argpartition(conv, -k_max)[-k_max:]

        # Weight: exponential based on how close to max
        top_vals = conv[top_idx]
        # Scale: peaks at 0.95*M get weight ~1, peaks at M get weight ~e
        scaled = (top_vals - 0.9 * M) / (0.05 * M + 1e-15)
        w = np.exp(np.maximum(scaled, 0))
        w /= w.sum()

        # Compute weighted subgradient
        sg = compute_sg_weighted(f, top_idx, w)

        # Line search with a few trial steps
        best_alpha = 0.0
        best_val = M
        for alpha in [lr, lr * 2.0, lr * 0.5, lr * 3.0, lr * 0.25]:
            f_trial = project_simplex(f - alpha * sg)
            val = fast_convolve(f_trial).max()
            if val < best_val - 1e-15:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-12:
            f = project_simplex(f - best_alpha * sg)
            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 1:
                    print(f"      FSG iter {it}: obj={best_obj:.9f}")
            else:
                no_improve += 1
                lr *= 0.95
        else:
            no_improve += 1
            lr *= 0.7

        # Periodically boost lr
        if it % 200 == 0 and no_improve > 50:
            lr = 0.005

    return best_f, best_obj


def momentum_subgradient(f0, time_limit, verbose=False):
    """Subgradient with momentum for faster convergence."""
    f = normalize(f0)
    n = len(f)
    v = np.zeros(n)  # momentum accumulator
    mu = 0.9  # momentum coefficient

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    lr = 0.001
    it = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > 500:
            rng = np.random.default_rng(it * 31 + 11)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            no_improve = 0
            lr = 0.001
            continue

        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Top peak subgradient
        peak_k = int(np.argmax(conv))
        sg = compute_sg_single_peak(f, peak_k)

        # Also add contribution from top-5 peaks
        top_idx = np.argpartition(conv, -5)[-5:]
        for k in top_idx:
            k = int(k)
            if k != peak_k and conv[k] > 0.98 * M:
                sg += 0.5 * compute_sg_single_peak(f, k)

        # Momentum update
        v = mu * v + (1 - mu) * sg
        direction = v

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [lr, lr * 2, lr * 0.5]:
            f_trial = project_simplex(f - alpha * direction)
            val = fast_convolve(f_trial).max()
            if val < best_val - 1e-15:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-12:
            f = project_simplex(f - best_alpha * direction)
            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 200 == 1:
                    print(f"      MOT iter {it}: obj={best_obj:.9f}")
            else:
                no_improve += 1
                lr *= 0.9
        else:
            no_improve += 1
            lr *= 0.8

    return best_f, best_obj


def make_gaussian(n, sigma_frac=0.25):
    x = np.arange(n) - n / 2.0

    return a


def make_bell(n, width_frac=0.5):

def make_periodic(n, block_len=8, seed=0):
    """Periodic block structure."""
    rng = np.random.default_rng(seed)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_alloc = min(100.0, rem)
            print(f"  LP-FW polish n={n_cur}: alloc={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lpfw_polish")
    else:
        rem = deadline - time.monotonic() - 60
        if rem > 10:
            t_alloc = min(60.0, rem)
            a_opt, obj = momentum_subgradient(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "momentum_polish")
    print(f"  Phase 4 time: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Final micro-optimizations ===
    print("\n=== Phase 5: Final micro-optimizations ===")
    t0 = time.monotonic()
    n_final = len(best_seq)

    # Small perturbations + short re-optimization
    perturbations = [
        ('shift1', np.roll(best_seq, 1)),
        ('shift_neg1', np.roll(best_seq, -1)),
        ('shift2', np.roll(best_seq, 2)),
        ('flip', best_seq[::-1]),
        ('blend_uni', 0.95 * best_seq + 0.05 * (np.ones(n_final) / n_final)),
        ('blend_uni2', 0.9 * best_seq + 0.1 * (np.ones(n_final) / n_final)),
    ]

    rng = np.random.default_rng(seed + 999)
    for amp in [0.002, 0.005, 0.01]:
        perturbations.append((f'noise_{amp}', np.maximum(best_seq + rng.standard_normal(n_final) * amp, 0)))

    for label, f_pert in perturbations:
        rem = deadline - time.monotonic() - 10
        if rem < 8:
            break
        f_pert = project_simplex(f_pert)
        t_alloc = min(12.0, rem * 0.5)
        if t_alloc < 3:
            break
        a_opt, obj = fast_subgradient(f_pert, time_limit=t_alloc)
        try_seq(a_opt, f"micro_{label}")
    print(f"  Phase 5 time: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Last resort LP-FW squeeze ===
    if best_seq is not None and len(best_seq) <= 896:
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            t_alloc = min(rem, 10.0)
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_squeeze")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
