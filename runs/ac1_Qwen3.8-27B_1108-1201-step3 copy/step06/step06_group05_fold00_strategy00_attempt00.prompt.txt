def evaluate_sequence(sequence: list[float]) -> float:
    """

    Returns np.inf if the input is invalid.

        if np.isnan(x) or np.isinf(x):

    sequence = [float(x) for x in sequence]

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_softmax(f, c, beta):
    """
    Compute gradient of the log-sum-exp smoothed max:
    LSE_beta(c) = (1/beta) * log(sum(exp(beta * c_k)))
    Gradient w.r.t. f_j: sum_k softmax(beta*c)_k * 2*f_{k-j} * [0<=k-j<n]
    """
    n = len(f)
    # Softmax weights over convolution coefficients
    scaled = beta * c
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()
    
    # g_j = 2 * sum_i f_i * w_{j+i}
    # This is a cross-correlation: g = 2 * correlate(w, f)
    # Using FFT: correlate(w, f)_j = sum_i f_i * w_{j+i}
    # = convolve(f, w_rev)[len(w)-1 + j]
    w_rev = w[::-1].copy()
    conv_len = n + len(w_rev) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(F * W, fft_size)[:conv_len]
    
    # g_j = 2 * conv[len(w_rev)-1 + j] = 2 * conv[2n-2+j]... 
    # Let me verify: conv[k] = sum_i f[i] * w_rev[k-i] = sum_i f[i] * w[2n-2-(k-i)] = sum_i f[i] * w[2n-2-k+i]
    # We want g_j/2 = sum_i f[i] * w[j+i], so 2n-2-k = j => k = 2n-2-j

    indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
    g = 2.0 * conv[indices]
    return g


def compute_gradient_pnorm(f, c, p):
    """Compute gradient of (1/p)*sum(c^p) where c = f*f."""
    n = len(f)

        w = np.ones(len(c))

    return g

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def multiplicative_peak_reduction(f0, time_limit, verbose=False):
    """
    Multiplicative weight update that specifically targets peaks.
    Uses the convolution structure to compute per-element peak contributions.
    """
    f = np.maximum(f0.astype(np.float64), 1e-10)
    f = f / f.sum()
    n = len(f)
    
    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(f)
        M = c.max()
        
        # Find the top peaks
        k_peaks = min(30, len(c))
        peak_idx = np.argpartition(c, -k_peaks)[-k_peaks:]
        
        # Weight peaks by their excess above median
        median_c = np.median(c)
        peak_vals = c[peak_idx]
        excess = np.maximum(peak_vals - median_c, 0)
        if excess.sum() < 1e-15:
            excess = np.ones(len(peak_idx))
        w_peaks = excess / excess.sum()
        
        # Compute subgradient: for each peak k, the contributing f elements are
        # f[j] for j in [max(0,k-n+1), min(n-1,k)]
        sg = np.zeros(n)
        for i in range(len(peak_idx)):
            if w_peaks[i] < 1e-12:
                continue
            k = int(peak_idx[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += w_peaks[i] * f[k - idx]
        
        # Multiplicative update
        f = f * np.exp(-lr * 2.0 * sg)
        f = np.maximum(f, 1e-12)
        f = f / f.sum()
        
        c_new = fast_convolve(f)
        obj_new = c_new.max()
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
            lr = min(lr * 1.02, 0.05)
        else:
            no_improve += 1
            lr *= 0.95
        
        if verbose and it % 50 == 0:
            print(f"    MPR iter {it}: obj={c_new.max():.10f}, best={best_obj:.10f}")

            lr = 0.008
            no_improve = 0

def coordinate_polish(f0, time_limit, verbose=False):
    """
    Fine coordinate descent: adjust individual coordinates to reduce max convolution.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f0, fast_convolve(f0).max()
    f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        c = fast_convolve(f)
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            # Try different multiplier values
            for mult in [0.7, 0.8, 0.9, 1.1, 1.2, 1.4]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.3:
                    continue
                
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic-like smoothing."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    # Light smoothing
    kernel = np.array([0.1, 0.2, 0.4, 0.2, 0.1])
    f_smooth = np.convolve(f_interp, kernel, mode='same')
    f_smooth = np.maximum(f_smooth, 0)
    ss = f_smooth.sum()
    if ss > 1e-15:
        f_smooth /= ss
    return f_smooth


def make_gaussian(n, sigma_frac=0.3):
    """Create a Gaussian initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6, power=4):
    """Create a bell-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_optimal_shape(n):
    """
    Create a sequence based on the theoretical optimal shape for minimizing
    max auto-convolution. Uses a raised-cosine-like profile.
    """
    x = np.linspace(-1, 1, n)
    # Raised cosine (Hann window)
    a = 0.5 * (1 + np.cos(np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a
