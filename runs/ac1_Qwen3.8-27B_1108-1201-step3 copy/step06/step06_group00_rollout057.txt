import numpy as np
from scipy.fft import rfft, irfft
from scipy import sparse
import cvxpy as cp
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(f):
    """Build the matrix C where C[k,j] = f[k-j] if 0<=k-j<n, else 0.
    Shape: (2n-1, n)"""
    n = len(f)
    C = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            C[k, j] = f[k - j]
    return C

def solve_lp(f, M):
    """Solve LP: maximize sum(g) subject to C@G <= M, G >= 0.
    Returns optimal g."""
    n = len(f)
    C = build_conv_matrix(f)
    G = cp.Variable(n)
    constraints = [G >= 0, C @ G <= M]
    objective = cp.Maximize(cp.sum(G))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, verbose=False)
    if G.value is not None:
        return np.maximum(G.value, 0)
    return np.zeros(n)

def objective_value(f):
    """Compute the objective 2n*max(conv)/(sum)^2."""
    n = len(f)
    S = f.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(f)
    return float(2.0 * n * b.max() / (S * S))
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an LP-based direction finding
method with line search, inspired by the state-of-the-art approach. The key
innovation is using a fast 1D line search (evaluating a precomputed quadratic
form) to determine the optimal mixing parameter at each iteration, combined
with periodic diversity injection and multi-scale exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Full linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(f):
    """Build matrix C where C[k,j] = f[k-j] if 0<=k-j<n, else 0. Shape (2n-1, n)."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for k in range(n):
        C[k, :k + 1] = f[k::-1]
    for k in range(n, 2 * n - 1):
        C[k, k - n + 1:] = f[n - 1:k - n::-1]
    return C


def solve_lp(f, M):
    """Solve: maximize sum(g) s.t. C@G <= M, G >= 0. Returns optimal g."""
    n = len(f)
    C = build_conv_matrix(f)
    c = -np.ones(n)
    A_ub = C
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    # Fallback: try CVXPY
    try:
        import cvxpy as cp
        G = cp.Variable(n)
        constraints = [G >= 0, C @ G <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(G)), constraints)
        prob.solve(solver=cp.SCS, verbose=False)
        if G.value is not None:
            return np.maximum(G.value, 0)
    except Exception:
        pass
    return np.zeros(n)


def objective_value(f):
    """Compute 2n*max(conv(f,f))/(sum(f))^2."""
    n = len(f)
    S = f.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(f)
    return float(2.0 * n * b.max() / (S * S))


def line_search(f, g, conv_ff, conv_fg, conv_gg, n):
    """Find best mixing t to minimize 2n*max((1-t)^2 ff + 2t(1-t) fg + t^2 gg)."""
    best_t = 0.0
    best_obj = 2.0 * n * conv_ff.max()
    
    # Grid search over t values
    t_values = [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.1, 0.12, 0.15, 0.2,
                0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0]
    
    for t in t_values:
        conv_new = (1 - t) ** 2 * conv_ff + 2 * t * (1 - t) * conv_fg + t ** 2 * conv_gg
        obj = 2.0 * n * conv_new.max()
        if obj < best_obj:
            best_obj = obj
            best_t = t
    
    # Refine around best_t with finer grid
    if 0 < best_t < 1:
        t_lo = max(0, best_t - 0.05)
        t_hi = min(1, best_t + 0.05)
        for t in np.linspace(t_lo, t_hi, 20):
            conv_new = (1 - t) ** 2 * conv_ff + 2 * t * (1 - t) * conv_fg + t ** 2 * conv_gg
            obj = 2.0 * n * conv_new.max()
            if obj < best_obj:
                best_obj = obj
                best_t = t
    
    return best_t, best_obj


def lp_iteration(f, n):
    """One LP-based optimization iteration. Returns (new_f, improvement, success)."""
    conv_ff = fast_convolve(f)
    M = conv_ff.max()
    
    if M < 1e-15:
        return f, 0.0, False
    
    g_0 = solve_lp(f, M)
    g_sum = g_0.sum()
    if g_sum < 1e-10:
        return f, 0.0, False
    
    g = g_0 / g_sum
    
    conv_fg = cross_convolve(f, g)
    conv_gg = fast_convolve(g)
    
    best_t, best_obj_before = line_search(f, g, conv_ff, conv_fg, conv_gg, n)
    
    if best_t == 0:
        return f, 0.0, False
    
    new_f = (1 - best_t) * f + best_t * g
    new_f = np.maximum(new_f, 0)
    ss = new_f.sum()
    if ss < 1e-10:
        return f, 0.0, False
    new_f = new_f / ss
    
    new_conv = fast_convolve(new_f)
    new_M = new_conv.max()
    old_obj = 2.0 * n * M
    new_obj = 2.0 * n * new_M
    
    improvement = old_obj - new_obj
    return new_f, improvement, True


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def coordinate_polish(a0, time_limit):
    """Coordinate descent polish on the simplex."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    multipliers = np.array([0.5, 0.7, 0.85, 1.15, 1.3, 1.5])
    
    for sweep in range(100):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 1.0:
                    continue
                rest_sum = 1.0 - old_val
                if rest_sum < 1e-12:
                    continue
                scale = (1.0 - new_val) / rest_sum
                if scale < 0:
                    continue

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an LP-based direction-finding
method (inspired by the state of the art) with adaptive line search, periodic
diversity injection, multi-length exploration, and coordinate descent polish.

Key ideas:
- At each iteration, solve an LP to find the best direction g that reduces
  the peak auto-convolution while preserving the sum constraint.
- Use a fast 1D line search (evaluating precomputed quadratic forms) to
  determine the optimal step size.
- Inject diversity periodically to escape local minima.
- Explore multiple sequence lengths for robustness.
- Final polish via coordinate descent.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Full linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(f):
    """Build matrix C where C[k,j] = f[k-j] if 0<=k-j<n, else 0. Shape (2n-1, n)."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for k in range(n):
        C[k, :k + 1] = f[k::-1]
    for k in range(n, 2 * n - 1):
        C[k, k - n + 1:] = f[n - 1:k - n::-1]
    return C


def solve_lp_scipy(f, M):
    """Solve: maximize sum(g) s.t. C@G <= M, G >= 0 using scipy linprog (HiGHS)."""
    n = len(f)
    C = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def solve_lp_cvxpy(f, M):
    """Solve the LP using CVXPY as fallback."""
    try:
        import cvxpy as cp
        n = len(f)
        C = build_conv_matrix(f)
        G = cp.Variable(n)
        constraints = [G >= 0, C @ G <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(G)), constraints)
        prob.solve(solver=cp.SCS, verbose=False)
        if G.value is not None and not np.any(np.isnan(G.value)):
            return np.maximum(G.value, 0.0)
    except Exception:
        pass
    return None


def solve_lp(f, M):
    """Unified LP solver with fallback."""
    result = solve_lp_scipy(f, M)
    if result is not None and result.sum() > 0:
        return result
    result = solve_lp_cvxpy(f, M)
    if result is not None and result.sum() > 0:
        return result
    return np.zeros(len(f))


def objective_value(f):
    """Compute 2n*max(conv(f,f))/(sum(f))^2."""
    n = len(f)
    S = f.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(f)
    return float(2.0 * n * b.max() / (S * S))


def line_search_t(conv_ff, conv_fg, conv_gg, n):
    """Find best mixing t to minimize 2n*max((1-t)^2*ff + 2t(1-t)*fg + t^2*gg)."""
    best_t = 0.0
    best_obj = 2.0 * n * conv_ff.max()

    t_values = [0.005, 0.01, 0.02, 0.03, 0.05, 0.07, 0.1, 0.13, 0.17, 0.22,
                0.28, 0.35, 0.42, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0]

    for t in t_values:
        conv_new = (1.0 - t) ** 2 * conv_ff + 2.0 * t * (1.0 - t) * conv_fg + t ** 2 * conv_gg
        obj = 2.0 * n * conv_new.max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_t = t

    # Local refinement around best_t
    if 0.0 < best_t < 1.0:
        t_lo = max(0.0, best_t * 0.5)
        t_hi = min(1.0, best_t * 2.0)
        t_fine = np.linspace(t_lo, t_hi, 30)
        for t in t_fine:
            conv_new = (1.0 - t) ** 2 * conv_ff + 2.0 * t * (1.0 - t) * conv_fg + t ** 2 * conv_gg
            obj = 2.0 * n * conv_new.max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_t = t

    return best_t, best_obj


def lp_step(f, n):
    """
    One LP-based optimization step.
    Returns (new_f, improvement, success).
    """
    conv_ff = fast_convolve(f)
    M = conv_ff.max()

    if M < 1e-15:
        return f, 0.0, False

    g_0 = solve_lp(f, M)
    g_sum = g_0.sum()
    if g_sum < 1e-12:
        return f, 0.0, False

    g = g_0 / g_sum

    conv_fg = cross_convolve(f, g)
    conv_gg = fast_convolve(g)

    best_t, _ = line_search_t(conv_ff, conv_fg, conv_gg, n)

    if best_t < 1e-6:
        return f, 0.0, False

    new_f = (1.0 - best_t) * f + best_t * g
    new_f = np.maximum(new_f, 0.0)
    ss = new_f.sum()
    if ss < 1e-12:
        return f, 0.0, False
    new_f = new_f / ss

    new_conv = fast_convolve(new_f)
    old_obj = 2.0 * n * M
    new_obj = 2.0 * n * new_conv.max()
    improvement = old_obj - new_obj

    return new_f, improvement, True


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving relative shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def coordinate_polish(a0, time_limit):
    """Coordinate descent polish on the simplex."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    multipliers = np.array([0.4, 0.6, 0.8, 0.9, 1.1, 1.25, 1.5, 2.0])

    for sweep in range(200):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-14:
                continue
            rest_sum = 1.0 - old_val
            if rest_sum < 1e-12:
                continue
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0.0 or new_val >= 0.999:
                    continue
                scale = (1.0 - new_val) / rest_sum
                if scale < 0.0 or scale > 10.0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_obj - 1e-15:
                    best_obj = val
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improved = True
                    break
        if not improved:
            break

    return best_a, best_obj


def subgradient_refine(f, time_limit, lr=0.005, momentum=0.9, seed=42):
    """
    Subgradient method on the simplex for final refinement.
    Uses temperature-weighted peak selection.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(f)
    a = f.copy().astype(np.float64)
    a = a / a.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_val = fast_convolve(a).max()
    cur_lr = lr
    cur_temp = 0.001
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()

        # Temperature-weighted subgradient
        scaled = b / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
        else:
            w /= w_sum

        # Compute subgradient via cross-correlation
        g = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-14:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += w[k] * a[k - idx]

        # Nesterov step
        v = momentum * v - cur_lr * g
        a_new = project_simplex(a + v)

        b_new = fast_convolve(a_new)
        obj_new = b_new.max()

        if obj_new < best_val - 1e-15:
            best_val = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        it += 1

        if no_improve > 500:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.8, 0.0005)
            no_improve = 0
        elif no_improve > 200:
            cur_lr *= 0.95

        cur_temp *= 1.005
        if cur_temp > 0.5:
            cur_temp = 0.5

    return best_a, best_val
