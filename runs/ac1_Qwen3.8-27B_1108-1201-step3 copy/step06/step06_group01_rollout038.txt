Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using aggressive L-BFGS with
softmax-smoothed objective, adaptive temperature annealing, dual-guided
subgradient, and peak shaving. Exploits the structure of the auto-correlation
to guide optimization toward flatter correlation profiles.

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    """Compute the objective value (scale-invariant)."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s * s)

def cross_corr(w, a):
    """Compute cross-correlation (w ⋆ a)[i] = sum_j w[i+j]*a[j] via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    w_flip = w[::-1]
    W = rfft(w_flip, fft_size)
    A = rfft(a, fft_size)
    cross = irfft(W * A, fft_size)[:size]
    return cross[size - 1 - np.arange(n)]

def smooth_obj_grad(z, T, n):
    """
    Smoothed objective and gradient in log-space.
    z is the log-domain variable, a = softmax(z).
    Objective: 2n/T * log(sum_k exp((a*a)[k]/T))
    Gradient: computed via chain rule through softmax.
    """
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    b = fast_convolve(a)
    max_b = b.max()
    log_sum_exp = max_b / T + np.log(np.exp(b / T - max_b / T).sum() / len(b))
    obj = 2.0 * n * log_sum_exp
    
    #gradient
    w = np.exp(b / T - max_b / T)
    w = w / w.sum()
    
    # grad_a[i] = (2n/T) * sum_k w[k] * 2*a[k-i] = (4n/T) * (w ⋆ a)[i]
    grad_a = (4.0 * n / T) * cross_corr(w, a)
    
    # grad_z via softmax Jacobian
    s = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * s)
    
    return obj, grad_z

def subgradient_momentum(a0, time_limit, lr_init=0.01, n_restarts=5):
    """
    Subgradient method with momentum and periodic restarts.
    Uses softmax-weighted subgradient for smooth approximation.
    """
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    v = np.zeros(n)
    mom = 0.9
    lr = lr_init
    t0 = time.monotonic()
    iters = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        b = fast_convolve(a)
        
        # Softmax weighting with adaptive temperature
        peak = b.max()
        temp = 0.02 * peak
        w = np.exp((b - peak) / temp)
        w = w / w.sum()
        
        # Subgradient
        sg = (4.0 * n) * cross_corr(w, a)
        
        # Momentum update
        v = mom * v - lr * sg
        
        # Project onto simplex
        a_cand = a + v
        # Simplex projection
        u = np.sort(a_cand)[::-1]
        cssv = np.cumsum(u)
        cond = u * np.arange(1, n + 1) > (cssv - 1)
        rho = np.nonzero(cond)[0][-1] if cond.any() else n - 1
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        a_new = np.maximum(a_cand - theta, 0.0)
        a_new = a_new / a_new.sum()
        
        score = compute_score(a_new)
        
        if score < best_score - 1e-10:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            a = a_new
            no_improve += 1
        
        # Adaptive learning rate
        lr *= 0.999
        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(iters)
            a = best_a + rng.standard_normal(n) * 0.005
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 2, 0.005)
            no_improve = 0
            mom = 0.85
        
        if iters % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.8f} lr={lr:.6f} iters={iters}")
    
    return best_a, best_score

def peak_shave(a0, time_limit):
    """
    Targeted peak shaving: identify the dominant peak in auto-correlation
    and try to reduce it via local modifications.
    """
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        target = peak_val  # We want to reduce this
        
        if peak_val * 2.0 * n < best_score * (1 - 1e-6) / (a.sum()**2):
            break
        
        # Find top contributors to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Try reducing the top contributor
        improved = False
        top_k = np.argsort(contribs)[::-1][:5]
        
        for ci in top_k:
            i = int(idx[ci])
            j = peak_k - i
            if a[i] < 1e-12:
                continue
            
            for factor in [0.98, 0.95, 0.9, 0.85, 0.8]:
                a_trial = a.copy()
                old_val = a_trial[i]
                new_val = old_val * factor
                diff = old_val - new_val
                if diff < 1e-15:
                    continue
                a_trial[i] = new_val
                # Redistribute to maintain sum
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                
                score = compute_score(a_trial)
                if score < best_score - 1e-10:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            
            if improved:
                break
        
        # Try reducing both i and j simultaneously
        if not improved:
            for ci in top_k[:3]:
                i = int(idx[ci])
                j = peak_k - i

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using aggressive L-BFGS with
softmax-smoothed objective and adaptive temperature annealing, combined with
dual-guided subgradient with momentum, Fourier-informed initialization,
peak shaving, and multi-length exploration. Key innovations over previous:
(1) much finer temperature schedule, (2) global softmax-weighted subgradient
instead of top-k, (3) Fourier-domain initialization, (4) adaptive step sizes.

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the objective 2*n*max_b/(sum_a)^2 (scale-invariant)."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s * s)


def cross_corr(w, a):
    """Compute (w *correlate* a)[i] = sum_j w[i+j]*a[j] via FFT.
    This is the gradient building block: d(a*a)[k]/da[i] = 2*a[k-i]."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    w_flip = w[::-1]
    W = rfft(w_flip, fft_size)
    A = rfft(a, fft_size)
    cross = irfft(W * A, fft_size)[:size]
    return cross[size - 1 - np.arange(n)]


def smooth_obj_grad(z, T, n):
    """
    Softmax-smoothed objective and its gradient in log-space.
    a = softmax(z). Objective approximates 2n*max_k(a*a)[k].
    """
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()

    b = fast_convolve(a)
    max_b = b.max()
    log_sum_exp = max_b / T + np.log(np.exp(b / T - max_b / T).sum() / len(b))
    obj = 2.0 * n * log_sum_exp

    w = np.exp(b / T - max_b / T)
    w = w / w.sum()

    # grad_a = (4n/T) * (w correlate a)
    grad_a = (4.0 * n / T) * cross_corr(w, a)

    # Gradient through softmax: grad_z = a*(grad_a - a*(grad_a . a))
    s = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * s)

    return obj, grad_z


class _LbfgsCtx:
    """Context object for L-BFGS callbacks (avoids closures)."""
    def __init__(self, T, n, deadline):
        self.T = T
        self.n = n
        self.deadline = deadline
        self.stopped = False

    def func(self, z):
        if time.monotonic() > self.deadline:
            self.stopped = True
            return 1e15
        obj, _ = smooth_obj_grad(z, self.T, self.n)
        return obj

    def jac(self, z):
        if time.monotonic() > self.deadline:
            self.stopped = True
            return np.zeros_like(z)
        _, g = smooth_obj_grad(z, self.T, self.n)
        return g


def lbgfs_anneal(a0, time_limit, T_schedule=None, verbose=True):
    """
    L-BFGS in log-space with temperature annealing.
    At each T, run L-BFGS to convergence, then decrease T.
    """
    from scipy.optimize import minimize

    n = len(a0)
    t0 = time.monotonic()

    if T_schedule is None:
        T_schedule = [5000, 2000, 1000, 500, 250, 125, 60, 30, 15, 7, 4, 2, 1, 0.5, 0.2, 0.1, 0.05]

    best_a = a0.copy()
    best_score = compute_score(a0)
    current_a = a0.copy()

    for T in T_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break

        z0 = np.log(np.maximum(current_a, 1e-300))
        ctx = _LbfgsCtx(T, n, t0 + time_limit)

        maxiter = max(100, min(10000, int(rem * 30)))

        try:
            res = minimize(
                fun=ctx.func, x0=z0, jac=ctx.jac, method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-15,
                         'maxfun': maxiter * 20, 'disp': False}
            )
            z_final = res.x
            z_shifted = z_final - z_final.max()
            a_new = np.exp(z_shifted)
            a_new = a_new / a_new.sum()
            a_new = np.maximum(a_new, 0)

            score = compute_score(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
            current_a = a_new.copy()

            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    L-BFGS T={T}: best={best_score:.8f} (it={res.nit}, {elapsed:.0f}s)")
        except Exception as e:
            if verbose:
                print(f"    L-BFGS T={T} error: {e}")

    return best_a, best_score


def subgradient_momentum(a0, time_limit, lr_init=0.008):
    """
    Subgradient with Nesterov momentum and adaptive temperature.
    Uses global softmax weighting over all lags.
    """
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    v = np.zeros(n)
    mom = 0.91
    lr = lr_init
    t0 = time.monotonic()
    iters = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        b = fast_convolve(a)
        peak = b.max()

        # Adaptive temperature: starts wider, narrows over time
        frac = min(1.0, iters / 5000.0)
        temp = (0.05 * (1 - frac) + 0.005 * frac) * peak
        w = np.exp((b - peak) / temp)
        w = w / w.sum()

        # Subgradient: (4n) * (w correlate a)
        sg = (4.0 * n) * cross_corr(w, a)

        # Nesterov-style: look-ahead
        a_look = a - mom * v
        # Re-evaluate gradient at look-ahead (approximate: just use current)
        v_new = mom * v - lr * sg
        a_new = a - v_new

        # Project onto simplex {x >= 0, sum(x) = 1}
        u = np.sort(a_new)[::-1]
        cssv = np.cumsum(u)
        cond = u * np.arange(1, n + 1) > (cssv - 1)
        if cond.any():
            rho = np.nonzero(cond)[0][-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
        else:
            theta = 0.0
        a_new = np.maximum(a_new - theta, 0.0)
        a_new = a_new / a_new.sum()

        v = v_new
        score = compute_score(a_new)

        if score < best_score - 1e-10:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            a = a_new
            no_improve += 1

        # Adaptive learning rate
        lr *= 0.9992
        if no_improve > 300:
            rng = np.random.default_rng(iters * 7919)
            a = best_a + rng.standard_normal(n) * 0.003
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 3.0, 0.004)
            mom = 0.85
            no_improve = 0

        if iters % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.8f} lr={lr:.6f}")

def peak_shave(a0, time_limit):
    """
    Targeted peak shaving: identify dominant peaks and reduce via
    local coefficient modifications with mass redistribution.
    """
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    step = 0
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]

        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = np.argsort(contribs)[::-1][:8]

        improved = False

        # Strategy 1: Reduce single element, redistribute mass
        for ci in top_idx:
            i = int(idx[ci])
            if a[i] < 1e-12:
                continue
            for factor in [0.99, 0.97, 0.95, 0.92, 0.88, 0.84]:
                a_trial = a.copy()
                old_val = a_trial[i]
                diff = old_val * (1.0 - factor)
                if diff < 1e-15:
                    continue
                a_trial[i] = old_val - diff
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                os = others.sum()
                if os < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / os
                a_trial = a_trial / a_trial.sum()
                score = compute_score(a_trial)
                if score < best_score - 1e-10:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stagnation = 0
                    break
            if improved:
                break

        # Strategy 2: Reduce paired elements (i and peak_k-i)
        if not improved:
            for ci in top_idx[:4]:
                i = int(idx[ci])
                j = peak_k - i
                if i == j or j < 0 or j >= n or a[j] < 1e-12:
                    continue
                for f1, f2 in [(0.97, 0.97), (0.95, 0.95), (0.97, 0.93),
                                (0.93, 0.97), (0.94, 0.94), (0.9, 0.9)]:
                    a_trial = a.copy()
                    a_trial[i] *= f1
                    a_trial[j] *= f2
                    a_trial = np.maximum(a_trial, 0)
                    ts = a_trial.sum()
                    if ts < 1e-15:
                        continue
                    a_trial = a_trial / ts
                    score = compute_score(a_trial)
                    if score < best_score - 1e-10:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stagnation = 0
                        break
                if improved:
                    break

        # Strategy 3: Add mass to "quiet" regions (away from peak)
        if not improved:
            # Find lags with low correlation values
            b_sorted = np.argsort(b)
            quiet_lags = b_sorted[:5]
            # Try adding mass at positions that don't contribute to any peak
            rng = np.random.default_rng(step * 31)
            for _ in range(3):
                # Pick a random quiet position
                pos = int(rng.integers(0, n))
                a_trial = a.copy()
                # Reduce a high-position element and add to pos
                hi_pos = int(np.argmax(a))
                if hi_pos == pos:
                    continue
                transfer = a[hi_pos] * 0.05
                a_trial[hi_pos] -= transfer
                a_trial[pos] += transfer
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = compute_score(a_trial)
                if score < best_score - 1e-10:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stagnation += 1
            if stagnation > 30:
                break
            # Perturb slightly
            rng = np.random.default_rng(step * 104729)
            a = a + rng.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_score


def fourier_init(n, seed=42):
    """
    Generate a non-negative sequence with approximately flat power spectrum.
    Start from random phases, take IFFT, square, normalize.
    """
    rng = np.random.default_rng(seed)
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.ones(n) * np.exp(1j * phases)
    # Make it Hermitian-symmetric for real output
    if n % 2 == 0:
        A_sym = np.zeros(n, dtype=complex)
        A_sym[0] = 1.0
        A_sym[n // 2] = 1.0
        for k in range(1, n // 2):
            A_sym[k] = np.exp(1j * phases[k])
            A_sym[n - k] = np.conj(A_sym[k])
    else:
        A_sym = np.zeros(n, dtype=complex)
        A_sym[0] = 1.0
        for k in range(1, (n + 1) // 2):
            A_sym[k] = np.exp(1j * phases[k])
            A_sym[n - k] = np.conj(A_sym[k])

    x = np.real(np.fft.ifft(A_sym))
    x = x ** 2  # Ensure non-negative
    x = np.maximum(x, 0)
    xs = x.sum()
    if xs < 1e-15:
        return np.ones(n) / n
    return x / xs


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving structure."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def dual_guided_refine(a0, time_limit):
    """
    Dual-guided refinement: use the auto-correlation profile as "weights"
    to identify where the sequence structure is suboptimal, then apply
    targeted L-BFGS with the dual variables as initialization guidance.
    """
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for iteration in range(50):
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 10:
            break

        b = fast_convolve(a)
        # Normalize correlation to get "dual variables"
        b_norm = b / b.sum()

        # Identify "hot" regions: positions that contribute to peaks
        # For each peak lag k, the contributing positions are j and k-j
        # Build a "pressure" map
        pressure = np.zeros(n)
        for k in range(len(b)):
            weight = b[k] / (b.sum() + 1e-15)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                pressure[j] += weight * a[j] * a[k - j] / (a[j] + 1e-15)

        # "Cool" regions: positions with low pressure
        # Try to shift mass from hot to cool regions
        hot_threshold = np.percentile(pressure, 75)
        cool_threshold = np.percentile(pressure, 25)
        hot_mask = pressure > hot_threshold
        cool_mask = pressure < cool_threshold

        if hot_mask.sum() == 0 or cool_mask.sum() == 0:
            continue

        # Small shift
        shift_amount = a[hot_mask].sum() * 0.02
        a_trial = a.copy()
        a_trial[hot_mask] -= shift_amount * a_trial[hot_mask] / a_trial[hot_mask].sum()
        cool_sum = a_trial[cool_mask].sum()
        if cool_sum > 1e-15:
            a_trial[cool_mask] += shift_amount * a_trial[cool_mask] / cool_sum
        a_trial = np.maximum(a_trial, 0)
        a_trial = a_trial / a_trial.sum()

        score = compute_score(a_trial)
        if score < best_score - 1e-10:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
        else:
            # Try smaller shift
            shift_amount = a[hot_mask].sum() * 0.005
            a_trial = a.copy()
            a_trial[hot_mask] -= shift_amount * a_trial[hot_mask] / (a_trial[hot_mask].sum() + 1e-15)
            cool_sum = a_trial[cool_mask].sum()
            if cool_sum > 1e-15:
                a_trial[cool_mask] += shift_amount * a_trial[cool_mask] / cool_sum
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = compute_score(a_trial)
            if score < best_score - 1e-10:
                best_score = score
                best_a = a_trial.copy()
                a = a_trial

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    Phases:
    1. Initialize from parent or Fourier-based random
    2. L-BFGS with aggressive temperature annealing (main optimization)
    3. Dual-guided refinement
    4. Subgradient with momentum (escape local minima)
    5. Fine L-BFGS polish
    6. Peak shaving
    7. Multi-length exploration
    8. Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')

    # === Get parent sequence ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        # Fourier-based initialization
        n_init = 768
        best_seq = fourier_init(n_init, seed=seed)
        best_score = compute_score(best_seq)
        print(f"Fourier init: n={n_init}, score={best_score:.8f}")

    n_best = len(best_seq)

    # === Phase 1: L-BFGS with aggressive annealing ===
    print(f"\n=== Phase 1: L-BFGS Coarse Annealing (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_p1 = min(rem * 0.40, 700.0)
    print(f"  Budget: {t_p1:.0f}s, n={n_best}")

    T_coarse = [5000, 2000, 1000, 500, 250, 125, 60, 30, 15, 7, 4, 2, 1, 0.5, 0.2, 0.1]
    a_p1, score_p1 = lbgfs_anneal(best_seq, t_p1, T_schedule=T_coarse, verbose=True)
    if score_p1 < best_score:
        best_score = score_p1
        best_seq = a_p1.copy()
    print(f"  Phase 1 result: {best_score:.8f}")

    # === Phase 2: Dual-guided refinement ===
    print(f"\n=== Phase 2: Dual-Guided (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_p2 = min(rem * 0.10, 150.0)
    if t_p2 > 10:
        a_p2, score_p2 = dual_guided_refine(best_seq, t_p2)
        if score_p2 < best_score:
            best_score = score_p2
            best_seq = a_p2.copy()
        print(f"  Phase 2 result: {best_score:.8f}")

    # === Phase 3: Subgradient with momentum ===
    print(f"\n=== Phase 3: Subgradient (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_p3 = min(rem * 0.15, 250.0)
    if t_p3 > 10:
        a_p3, score_p3 = subgradient_momentum(best_seq, t_p3, lr_init=0.007)
        if score_p3 < best_score:
            best_score = score_p3
            best_seq = a_p3.copy()
        print(f"  Phase 3 result: {best_score:.8f}")

    # === Phase 4: Fine L-BFGS ===
    print(f"\n=== Phase 4: L-BFGS Fine (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_p4 = min(rem * 0.20, 350.0)
    if t_p4 > 10:
        T_fine = [50, 25, 10, 5, 2, 1, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
        a_p4, score_p4 = lbgfs_anneal(best_seq, t_p4, T_schedule=T_fine, verbose=True)
        if score_p4 < best_score:
            best_score = score_p4
            best_seq = a_p4.copy()
        print(f"  Phase 4 result: {best_score:.8f}")

    # === Phase 5: Peak shaving ===
    print(f"\n=== Phase 5: Peak Shaving (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_p5 = min(rem * 0.12, 150.0)
    if t_p5 > 5:
        a_p5, score_p5 = peak_shave(best_seq, t_p5)
        if score_p5 < best_score:
            best_score = score_p5
            best_seq = a_p5.copy()
        print(f"  Phase 5 result: {best_score:.8f}")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        alt_lengths = []
        for factor in [1.25, 1.5, 0.8, 2.0]:
            alt_n = int(n_best * factor)
            if 256 <= alt_n <= 4096 and alt_n != n_best:
                alt_lengths.append(alt_n)

        t_per_alt = min(rem * 0.20, 120.0)
        for alt_n in alt_lengths[:2]:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_alloc = min(t_per_alt, rem * 0.6)
            init_alt = interpolate_sequence(best_seq, alt_n)
            T_alt = [200, 100, 50, 20, 10, 5, 2, 1, 0.5, 0.2]
            a_alt, score_alt = lbgfs_anneal(init_alt, t_alloc, T_schedule=T_alt, verbose=False)
            if score_alt < best_score:
                best_score = score_alt
                best_seq = a_alt.copy()
                n_best = alt_n
                print(f"  New best at n={alt_n}: {best_score:.8f}")

    # === Phase 7: Final ultra-fine L-BFGS ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t_p7 = min(rem * 0.6, 120.0)
        T_ultra = [20, 10, 5, 2, 1, 0.5, 0.2, 0.1, 0.05]
        a_p7, score_p7 = lbgfs_anneal(best_seq, t_p7, T_schedule=T_ultra, verbose=True)
        if score_p7 < best_score:
            best_score = score_p7
            best_seq = a_p7.copy()

    # Final peak shave
    rem = deadline - time.monotonic()
    if rem > 10:
        a_ps, score_ps = peak_shave(best_seq, min(rem * 0.5, 30.0))
        if score_ps < best_score:
            best_score = score_ps
            best_seq = a_ps.copy()

    # === Verify and return ===
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if not np.isfinite(final_score):
        if parent is not None:
            best_seq = parent
        else:
            best_seq = np.ones(256) / 256
        final_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL: score={final_score:.8f} n={len(best_seq)}")
    return [float(x) for x in best_seq]
