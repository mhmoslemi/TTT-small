import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring faster ones."""
    global _lp_solver
    if '_lp_solver' in globals():
        return _lp_solver
    best = None
    best_time = float('inf')
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK, cp.CLARABEL]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            t0 = time.monotonic()
            prob.solve(solver=solver_name)
            t1 = time.monotonic() - t0
            if prob.status in ('optimal', 'optimal_inaccurate') and t1 < best_time:
                best = solver_name
                best_time = t1
        except Exception:
            continue
    if best is None:
        best = cp.ECOS
    _lp_solver = best
    return best


def solve_lp_best_response(a, solver=None):
    """
    Given fixed a (non-negative, sum=1), find b (non-negative, sum=1)
    minimizing max_k conv(a, b)[k]. Returns (b, M).
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def lbgfs_refine(a0, time_limit, T=100, maxiter=5000):
    """L-BFGS refinement in log-space with soft-max smoothing."""
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    stopped = [False]

    def func(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        a = _softmax(z)
        b = fast_convolve(a)
        Tb = T * b
        m = Tb.max()
        log_sum_exp = m + np.log(np.exp(Tb - m).sum())
        return (2.0 * n / T) * log_sum_exp

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros_like(z)
        a = _softmax(z)
        b = fast_convolve(a)
        Tb = T * b
        m = Tb.max()
        w = np.exp(Tb - m)
        w = w / w.sum()
        # Gradient of soft-max smoothed objective w.r.t. a
        # grad_a = 2n * cross_corr(w, a) essentially
        grad_a = _cross_corr_grad(w, a)
        # Convert to z-gradient
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        return grad_z

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        a_out = _softmax(res.x)
    except Exception:
        a_out = a0.copy()

    return a_out


def _softmax(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def _cross_corr_grad(w, a):
    """
    Compute gradient of max_k (a*a)[k] w.r.t. a, weighted by soft-max weights w.
    grad_i = 2 * sum_k w[k] * a[k-i] (with boundary handling)
    This is computed via FFT.
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = irfft(rfft(w, fft_size))[:size] if False else None
    # More directly: grad_i = 2 * sum_k w[k] * a[k-i] for valid k,i
    # This is the cross-correlation (w * a) computed via FFT
    # (w ⋆ a)[i] = sum_k w[k] * a[k-i]
    # In convolution terms: sum_k w[k] * a[k-i] = (reverse(w) * a)[n-1-i]... 
    # Actually, let's just compute it directly via the relation:
    # sum_k w[k] * a[k-i] is the i-th element of the cross-correlation of w and a.
    # Using FFT: cross_corr(w, a)[i] = ifft(conj(fft(w)) * fft(a))[i]
    W_fft = rfft(w, fft_size)
    A_fft = rfft(a, fft_size)
    # Cross-correlation: (w ⋆ a)[i] = sum_k w[k] a[k-i]
    # This equals ifft(conj(W) * A) in the right range
    xc = irfft(np.conj(W_fft) * A_fft, fft_size)
    # The cross-correlation (w ⋆ a)[i] = sum_k w[k]*a[k-i]
    # In linear terms, this is the convolution of flip(w) with a, shifted
    # Let's just verify: for i=0, sum_k w[k]*a[k] 
    # ifft(conj(W)*A)[m] = sum_n w[n] a[n+m-N] (circular)
    # For linear cross-correlation, we need:
    # (w ⋆ a)[i] = sum_k w[k+n-1-i] * a[k] for appropriate range
    # This is getting confusing. Let me just compute it directly for small n
    # or use a different FFT formulation.
    
    # Actually, the simplest correct formulation:
    # grad_i = 4n * sum_k w[k] * a[k-i] (with 0<=k-i<n)
    # = 4n * (w * a)_anti[i] where * denotes the appropriate correlation
    # 
    # Let's use: flip w to get w_rev[m] = w[n-1-m] (but w has length 2n-1)
    # Then conv(w_rev, a)[n-1+i] = sum_j w_rev[n-1+i-j] * a[j] = sum_j w[j-i] * a[j]
    # = sum_k w[k+i-n+1] * a[k+n-1-i]... this is getting too complicated.
    #
    # Simplest: just compute sum_k w[k] * a[k-i] directly for each i.
    # For n up to ~2000, this is O(n^2) = 4M ops, which is fine.
    
    result = np.zeros(n)
    w_len = len(w)
    for i in range(n):
        # sum_k w[k] * a[k-i] for 0 <= k < w_len and 0 <= k-i < n
        k_min = max(0, i)
        k_max = min(w_len, i + n)
        if k_min < k_max:
            k = np.arange(k_min, k_max)
            result[i] = np.dot(w[k], a[k - i])
    return 2.0 * result

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def nesterov_smoothed(a0, time_limit, T=50, max_iter=50000):
    """Nesterov accelerated gradient on soft-smoothed objective in log-space."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z_prev = z.copy()
    best_z = z.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    t0 = time.monotonic()
    
    # Nesterov parameters
    k = 1
    t = 1.0
    
    while time.monotonic() - t0 < time_limit and k < max_iter:
        # Extrapolation
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_next
        z_extrap = z + beta * (z - z_prev)
        z_prev = z.copy()
        t = t_next

        a_extrap = _softmax(z_extrap)
        c = fast_convolve(a_extrap)
        Tc = T * c
        m = Tc.max()
        w = np.exp(Tc - m)
        w = w / w.sum()
        
        # Gradient in log-space
        grad_a = _cross_corr_grad(w, a_extrap)
        S = np.dot(grad_a, a_extrap)
        grad_z = a_extrap * (grad_a - S)
        
        # Step size: 2/(k*T) style
        lr = 2.0 / (k * T + 1)
        z = z_extrap - lr * grad_z
        k += 1
        
        a = _softmax(z)
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_z = z.copy()

        if k % 5000 == 0:
            print(f"    Nesterov iter={k}: score={best_score:.6f}")

    a_best = _softmax(best_z)
    return a_best

            best_seq, best_score = check_and_update(a_ps, "peak_shave", best_seq, best_score)

        # Second L-BFGS after peak shaving
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            t_alloc = min(rem * 0.4, 30.0)
            a_lb2 = lbgfs_refine(best_seq, t_alloc, T=500)
            best_seq, best_score = check_and_update(a_lb2, "lbgfs_post_ps", best_seq, best_score)

    # === Phase 6: Try alternative lengths ===
    print(f"\n=== Phase 6: Alternative Lengths (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = [n_best - 64, n_best + 64, int(n_best * 0.85), int(n_best * 1.15)]
        alt_lengths = [x for x in alt_lengths if 128 <= x <= 4096 and x != n_best]
        # Remove duplicates
        seen = set()
        alt_lengths = [x for x in alt_lengths if not (x in seen or seen.add(x))]

        for alt_n in alt_lengths[:3]:
            rem = deadline - time.monotonic() - 30
            if rem < 20:
                break
            t_alloc = min(rem * 0.3, 100.0)
            print(f"  Trying n={alt_n}, budget={t_alloc:.0f}s")
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _, _ = fictitious_play_accelerated(
                alt_n, t_alloc, initial=init_alt, solver=solver
            )
            best_seq, best_score = check_and_update(a_alt, f"FP_alt_n{alt_n}", best_seq, best_score)

            if best_seq is not None and len(best_seq) == alt_n:
                rem = deadline - time.monotonic() - 20
                if rem > 10:
                    t_ps = min(rem * 0.4, 30.0)
                    a_ps_alt, _ = peak_shave_aggressive(best_seq, t_ps)
                    best_seq, best_score = check_and_update(a_ps_alt, f"ps_alt_n{alt_n}", best_seq, best_score)

    # === Phase 7: Final micro-polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t_final = min(rem * 0.5, 20.0)
            a_final = lbgfs_refine(best_seq, t_final, T=1000, maxiter=3000)
            best_seq, best_score = check_and_update(a_final, "final_lbgfs", best_seq, best_score)

        rem = deadline - time.monotonic() - 5
        if rem > 8:
            a_ps_final, _ = peak_shave_aggressive(best_seq, min(rem * 0.5, 10.0))
            best_seq, best_score = check_and_update(a_ps_final, "final_ps", best_seq, best_score)

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
