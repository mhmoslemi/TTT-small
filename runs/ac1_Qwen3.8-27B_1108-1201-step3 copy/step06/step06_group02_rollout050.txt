    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth projected gradient descent with adaptive softmax temperature
2. Targeted peak-shaving coordinate moves
3. Multi-start with diverse initializations and lengths
4. Final polish with LP-based direction finding
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_softmax_weights(conv, temperature):
    """Compute softmax weights over convolution values."""
    shifted = conv - conv.max()
    w = np.exp(shifted / temperature)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= s
    return w


def compute_gradient(f, w, n):
    """Compute gradient of 2n * sum_k w[k] * conv_k.
    Result: 4n * cross_correlation(w, f)
    where cross_correlation[i] = sum_k w[k] * f[k-i] for valid k-i.
    """
    # cross_correlation[i] = sum_k w[k] * f[k-i]
    # = sum_j f[j] * w[i+j] (substituting j = k-i, so k = i+j)
    # This is the cross-correlation of f and w.
    # Using FFT: pad both to length >= 2n-1
    fft_size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w, fft_size)
    # Cross-correlation = IFFT(F * conj(W))
    # But we need a specific convention. Let me use:
    # (f cross w)[i] = sum_k f[k] * w[k-i]
    # This is IFFT(conj(F) * W) shifted... 
    # Actually, let me just compute it directly for clarity since n is moderate.
    
    # Alternative: use the fact that
    # cross[i] = sum_j f[j] * w[i+j] for i+j < len(w)
    # Pad w to be long enough
    # cross[i] = (f * w_rev)[len(w)-1-i] where w_rev = w[::-1]
    # (f * w_rev)[m] = sum_j f[j] * w_rev[m-j] = sum_j f[j] * w[len(w)-1-m+j]
    # Set len(w)-1-m+j = i+j => m = len(w)-1-i
    w_rev = w[::-1]
    fft_size2 = 1 << (len(f) + len(w_rev) - 1).bit_length()
    F2 = np.fft.rfft(f, fft_size2)
    WR = np.fft.rfft(w_rev, fft_size2)
    conv_result = np.fft.irfft(F2 * WR, fft_size2)
    
    cross = np.zeros(n)
    for i in range(n):
        m = len(w) - 1 - i  # = 2n-2-i
        if 0 <= m < len(conv_result):
            cross[i] = conv_result[m]
    return 4.0 * n * cross


def smooth_gradient_descent(f0, time_limit, verbose=False):
    """Projected gradient descent with smooth max (log-sum-exp) approximation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    
    # Parameters
    temperature_frac = 0.15  # temperature = frac * range of conv
    lr = 0.05
    lr_decay = 0.9998
    no_improve = 0
    it = 0
    min_temp = 1e-8
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Adaptive temperature
        temp_range = max(M - conv.min(), 1e-10)
        temperature = max(temperature_frac * temp_range, min_temp)
        
        # Compute softmax weights
        w = compute_softmax_weights(conv, temperature)
        
        # Compute gradient
        grad = compute_gradient(f, w, n)
        
        # Adaptive learning rate
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-30:
            break
        step_lr = min(lr, 1.0 / (grad_norm_sq + 1e-10))
        
        # Gradient step + projection
        f_new = f - step_lr * grad
        f_new = project_simplex(f_new)
        
        # Evaluate
        new_conv = fast_convolve(f_new)
        new_obj = new_conv.max()
        
        if new_obj < best_obj - 1e-13:
            best_obj = new_obj
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 100 == 0:
                print(f"    SGD iter {it}: obj={best_obj:.10f}, temp={temperature:.2e}, lr={step_lr:.2e}")
        else:
            # Try smaller step
            if step_lr > 1e-10:
                f_small = f - 0.5 * step_lr * grad
                f_small = project_simplex(f_small)
                new_conv2 = fast_convolve(f_small)
                if new_conv2.max() < best_obj - 1e-13:
                    best_obj = new_conv2.max()
                    best_f = f_small.copy()
                    f = f_small
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        
        lr *= lr_decay
        
        if no_improve > 300:
            # Perturb to escape local minimum
            rng = np.random.default_rng(it * 13 + 7)
            amp = 0.003 + 0.002 * (no_improve // 100)
            f = np.maximum(best_f + rng.standard_normal(n) * amp, 0)
            f = f / f.sum()
            lr = 0.05
            temperature_frac = max(0.05, temperature_frac * 0.8)
            no_improve = 0
            if temperature_frac < 0.02:
                break
    
    return best_f, best_obj


def targeted_peak_shaving(f0, time_limit, verbose=False):
    """Targeted coordinate moves to reduce specific peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        no_improve += 1
        if no_improve > 200:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        
        # Get top peaks (weighted)
        k_top = min(50, len(conv))
        top_idx = np.argpartition(conv, -k_top)[-k_top:]
        top_idx = top_idx[np.argsort(conv[top_idx])[::-1]]
        
        found_improvement = False
        
        for pk in top_idx[:10]:
            pk = int(pk)
            if conv[pk] < M * 0.8:
                break
            
            # Find contributing indices
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            contrib = np.arange(js, je + 1)
            if len(contrib) == 0:
                continue
            
            products = f[contrib] * f[pk - contrib]
            
            # Find the most contributing pair
            max_idx = np.argmax(products)
            i = contrib[max_idx]
            j = pk - i
            
            # Try reducing the larger of f[i], f[j] and moving mass elsewhere
            if f[i] >= f[j]:
                src = i
            else:
                src = j
            
            # Find target indices (those that don't contribute to this peak)
            # i.e., indices t such that pk-t is out of [0, n-1] or f[pk-t] is small
            for frac in [0.1, 0.2, 0.3, 0.5]:
                delta = f[src] * frac
                if delta < 1e-15:
                    continue
                
                trial = f.copy()
                trial[src] -= delta
                
                # Distribute to random non-contributing indices
                # Find indices where the contribution to peak pk would be small
                all_idx = np.arange(n)
                contrib_set = set(contrib.tolist())
                non_contrib = [t for t in all_idx if t not in contrib_set and t != src]
                
                if len(non_contrib) < 2:
                    non_contrib = [t for t in all_idx if t != src]
                
                if not non_contrib:
                    continue
                
                # Distribute proportionally to current values (or randomly)
                nc_arr = np.array(non_contrib)
                weights = trial[nc_arr] + 1e-10
                weights /= weights.sum()
                
                trial[nc_arr] += delta * weights
                trial = np.maximum(trial, 0)
                trial /= trial.sum()
                
                obj = fast_convolve(trial).max()
                if obj < best_obj - 1e-13:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    no_improve = 0
                    found_improvement = True
                    break
            
            if found_improvement:
                break
        
        if not found_improvement and no_improve > 50:
            # Try a random perturbation
            if no_improve % 50 == 0:
                amp = 0.002
                f = np.maximum(best_f + rng.standard_normal(n) * amp, 0)
                f = f / f.sum()
    
    return best_f, best_obj


def alternating_minimization(f0, time_limit, verbose=False):
    """
    Alternating optimization: fix the "shape" parameters and optimize 
    the amplitude, then fix amplitude and adjust shape.
    Uses a parametric family: a_i = base_i * (1 + eps * mod_i)
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    
    # Parametric perturbations: a_i = f_i * (1 + sum_m c_m * phi_m(i))
    # where phi_m are basis functions (Fourier modes)
    n_modes = min(20, n // 4)
    x = np.arange(n)
    
    # Fourier basis
    phi = np.zeros((n_modes, n))
    for m in range(n_modes):
        phi[m] = np.cos(2 * np.pi * (m + 1) * x / n)
    
    # Also add sine modes
    n_phi = 2 * n_modes
    phi_full = np.zeros((n_phi, n))
    phi_full[:n_modes] = phi
    for m in range(n_modes):
        phi_full[n_modes + m] = np.sin(2 * np.pi * (m + 1) * x / n)
    
    # Optimize coefficients c
    c = np.zeros(n_phi)
    lr_c = 0.01
    
    for it in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Current f
        f_new = f * (1.0 + phi_full.dot(c))
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s
        
        conv = fast_convolve(f_new)
        M = conv.max()
        
        if M < best_obj - 1e-13:
            best_obj = M
            best_f = f_new.copy()
            f = f_new
            if verbose and it % 100 == 0:
                print(f"    AM iter {it}: obj={best_obj:.10f}")
        
        # Gradient of M with respect to c (using softmax approximation)
        temp = max(M * 0.1, 1e-8)
        w = compute_softmax_weights(conv, temp)
        
        # d(conv_k)/dc_m = 2 * sum_j f_new[j] * phi_m[j] * f_new[k-j] * (appropriate terms)
        # This is getting complex. Let's use finite differences for a few modes.
        
        # Actually, let's just do coordinate descent on c
        best_c = c.copy()
        best_M = M
        for m in range(n_phi):
            for delta in [0.01, -0.01, 0.005, -0.005]:
                c_trial = c.copy()
                c_trial[m] += delta
                f_trial = f * (1.0 + phi_full.dot(c_trial))
                f_trial = np.maximum(f_trial, 0)
                s_t = f_trial.sum()
                if s_t < 1e-15:
                    continue
                f_trial = f_trial / s_t
                M_t = fast_convolve(f_trial).max()
                if M_t < best_M:
                    best_M = M_t
                    best_c = c_trial.copy()
        c = best_c
    
    return best_f, best_obj


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine_mod(n, freq, amp):
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_multi_cosine(n, n_freqs=3, max_amp=0.3, seed=0):
    rng = np.random.default_rng(seed)
    a = np.ones(n)
    for m in range(1, n_freqs + 1):
        amp = rng.uniform(0.05, max_amp)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * m * np.arange(n) / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_ramp(n, gamma=1.0):
    """Linearly ramped sequence: a_i = (i+1)^gamma / sum((j+1)^gamma)"""
    a = (np.arange(n) + 1) ** gamma
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_v_shape(n, gamma=1.0):
    """V-shaped: low in middle, high at edges"""
    x = np.abs(np.arange(n) - n / 2.0) / (n / 2.0)
    a = (x + 0.1) ** gamma
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_inverted_gaussian(n, sigma_frac=0.3):
    """Inverse Gaussian: high at edges, low in middle"""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = 1.0 / (np.exp(-0.5 * (x / sigma) ** 2) + 0.1)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a
