3. p≠i, q=i: a_{k-i} * a_i = a_{k-i} * x (same as above if k-i≠i)
4. p=i, q=i: a_i^2 = x^2 (only if k=2i)

So: conv[k] = T_k + x*a_{k-i}*[0≤k-i<n and k-i≠i] + a_{k-i}*x*[0≤k-i<n and k-i≠i] + x^2*[k=2i]
            = T_k + 2*x*a_{k-i}*[0≤k-i<n and k-i≠i] + x^2*[k=2i]

Great, so for each k, conv[k] is a quadratic in x:
conv[k] = T_k + B_k * x + C_k * x^2

where B_k = 2*a_{k-i} if 0≤k-i<n and k≠2i, else 0
      C_k = 1 if k=2i, else 0

The constraint conv[k] ≤ t becomes:
C_k * x^2 + B_k * x + (T_k - t) ≤ 0

For C_k = 0 (k≠2i): this is linear in x: B_k * x ≤ t - T_k, so x ≤ (t - T_k) / B_k (if B_k > 0)
For C_k = 1 (k=2i): this is quadratic: x^2 + B_{2i} * x + (T_{2i} - t) ≤ 0

Also, we need sum(a) = 1: sum_{j≠i} a_j + x = 1, so x = 1 - sum_{j≠i} a_j. This is a hard constraint, not an inequality.

Hmm, so with the sum constraint, x is determined! We can't freely choose x. We need sum(a) = 1.

So the coordinate descent approach is: for each i, we can change a_i but must compensate by changing another coordinate to maintain sum=1.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using improved optimization:
- Min-max LP direction (directly minimizes peak)
- Multi-peak targeting
- Pairwise swap local search
- Momentum-enhanced Frank-Wolfe
- Multi-length exploration
"""
import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(f, n):
    """Build matrix F where (F @ g)[k] = conv(f, g)[k]."""
    F = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        F[k_start:k_end + 1, j] = f[:length]
    return F

def solve_minmax_lp(f, n, F_cache=None, active_peaks=None):
    """Solve: min t s.t. (F g)_k <= t for k in active_set, sum(g)=1, g>=0."""
    if F_cache is None:
        F_cache = build_conv_matrix(f, n)
    
    if active_peaks is None:
        active_peaks = np.arange(2 * n - 1)
    
    m = len(active_peaks)
    # Variables: g_0, ..., g_{n-1}, t
    # Objective: minimize t -> c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Constraints: (F g)_k - t <= 0 for k in active_set
    # F[active_peaks] @ g - t <= 0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = F_cache[active_peaks]
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(m)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1)
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

def make_gaussian(n, sigma_frac=0.25):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_ramp(n, p=2.0):
    """Power-law ramp: a[i] ∝ (i+1)^p, normalized."""
    x = np.arange(n) + 1
    a = x ** p
    a = a / a.sum()
    return a

def make_hann(n):
    a = 0.5 * (1 - np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    a = a / a.sum()
    return a

def make_bell(n, width=0.7):
    x = np.arange(n) - n / 2.0
    half_w = n * width / 2.0
    t = np.clip(np.abs(x) / half_w, 0, 1.0)
    a = (1.0 - t) ** 3
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def interp_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def lpfw_minmax(f0, time_limit, verbose=False, mom=0.7):
    """LP-FW with min-max LP direction, momentum, and multi-peak targeting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    F_cache = build_conv_matrix(f, n)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    
    v = np.zeros(n)  # momentum
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    peak_k = 5  # number of active peaks
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        # Find active peaks
        top_indices = np.argpartition(conv, -peak_k)[-peak_k:]
        
        # Solve min-max LP
        g = solve_minmax_lp(f, n, F_cache, top_indices)
        
        if g is None:
            # Fallback: use subgradient direction
            k_peak = int(np.argmax(conv))
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            sg = np.zeros(n)
            for j in range(js, je + 1):
                sg[j] = 2.0 * f[k_peak - j] - f[j]
            g = np.maximum(np.ones(n) / n - sg, 0)
            g_sum = g.sum()
            if g_sum > 1e-15:
                g = g / g_sum
            else:
                g = np.ones(n) / n
        else:
            g = np.maximum(g, 0)
            g_sum = g.sum()
            if g_sum < 1e-15:
                g = np.ones(n) / n
            else:
                g = g / g_sum
        
        # Momentum update
        v = mom * v + (1 - mom) * (g - f)
        
        # Line search
        best_alpha = 0.0
        best_val = 2.0 * n * conv.max()
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            trial = f + alpha * v
            trial = np.maximum(trial, 0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            trial = trial / ts
            tv = fast_convolve(trial).max()
            t_obj = 2.0 * n * tv
            if t_obj < best_val:
                best_val = t_obj
                best_alpha = alpha
        
        if best_alpha < 1e-8:
            no_improve += 1
        else:
            f = f + best_alpha * v
            f = np.maximum(f, 0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            if 2.0 * n * M < best_obj - 1e-14:
                best_obj = 2.0 * n * M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        # Update F_cache periodically
        if it % 10 == 0:
            F_cache = build_conv_matrix(f, n)
        
        if no_improve > 100:
            # Restart with perturbation
            rng = np.random.default_rng(it * 7 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            v = np.zeros(n)
            no_improve = 0
            F_cache = build_conv_matrix(f, n)
        
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP-MM it {it}: obj={best_obj:.10f}, t={elapsed:.1f}s")
        
        if no_improve > 300:
            break
    
    return best_f, best_obj

def swap_local_search(f0, time_limit, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 500:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        k_peak = int(np.argmax(conv))
        
        # Contributors to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        contrib = np.arange(js, je + 1)
        products = f[contrib] * f[k_peak - contrib]
        
        # Sort by contribution (highest first)
        order = np.argsort(products)[::-1]
        top_contrib = contrib[order[:min(6, len(contrib))]]

        all_idx = np.arange(n)
        non_contrib = np.setdiff1d(all_idx, contrib)
        # Sort non-contributors by value (lowest first - they have room to grow)
        non_contrib_sorted = non_contrib[np.argsort(f[non_contrib])]

                        f = trial

                        trial[ni] -= delta

                        if new_obj < best_obj - 1e-13:
                            best_obj = new_obj
                            best_f = trial.copy()
                            f = trial

                            conv = fast_convolve(f)
                            break
                    if found_improve:
                        break
                if found_improve:
                    break
