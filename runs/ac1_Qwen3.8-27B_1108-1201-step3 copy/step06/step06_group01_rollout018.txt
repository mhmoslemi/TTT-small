```python
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * n * corr


def nesterov_optimize(a0, time_limit, T_init=0.02, T_final=0.001, 
                      lr_init=0.005, momentum=0.9, n_restarts=3):
    """
    Nesterov-accelerated gradient descent in log-space with temperature annealing.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    s = a.sum()
    a = a / s
    z = np.log(a)
    v = np.zeros(n)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    step = 0
    no_improve = 0
    max_no_improve = 500
    
    lr = lr_init
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        
        # Anneal temperature
        T = T_init * (T_final / T_init) ** progress
        
        # Nesterov lookahead
        z_look = z + momentum * v
        a_look = np.exp(z_look - z_look.max())
        a_look = a_look / a_look.sum()
        
        # Compute gradient at lookahead point
        b = fast_convolve(a_look)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        w = exp_b / exp_b.sum()
        
        grad_a = compute_grad(a_look, w, n, fft_size)
        
        # Transform to z-space gradient
        S = np.dot(grad_a, a_look)
        grad_z = a_look * (grad_a - S)
        
        # Update velocity and position
        v = momentum * v - lr * grad_z
        z = z + v
        
        # Implicitly, a = softmax(z)
        a = np.exp(z - z.max())
        a = a / a.sum()
        
        # Evaluate
        score = 2.0 * n * fast_convolve(a).max()
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive learning rate
        if no_improve > 200:
            lr *= 0.95
            if no_improve > max_no_improve:
                # Restart from best with perturbation
                z = np.log(np.maximum(best_a, 1e-300))
                z += np.random.standard_normal(n) * 0.01
                v = np.zeros(n)
                lr = lr_init * 0.5
                no_improve = 0
        
        # Decay learning rate slowly
        lr = max(lr_init * 0.01, lr_init * (1.0 - progress) ** 0.5)
        
        if step % 500 == 0:
            print(f"    Nesterov step={step} T={T:.5f} lr={lr:.6f} score={best_score:.6f}")
    
    return best_a, best_score
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam-optimized log-space gradient
descent with adaptive soft-max temperature annealing, diverse multi-start
initialization, population-based exploration, L-BFGS polishing, and
peak-targeted local search. Key improvements over the previous fictitious-play
approach: (1) faster per-iteration gradient methods replace expensive LP solves,
(2) continuous temperature annealing provides smoother optimization landscape,
(3) population diversity prevents premature convergence, (4) multi-scale
exploration across sequence lengths.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_grad_a(a, w, n, fft_size):
    """
    Compute gradient of 2n * soft_max(conv(a,a), T) w.r.t. a.
    grad_a[i] = 2n * sum_{j=0}^{n-1} w[j+i] * a[j]
    Uses FFT-based cross-correlation.
    """
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * n * corr


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    return np.log(np.maximum(a, 1e-300))


def get_score(a):
    """Compute the objective score for a sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def adam_logspace(a0, time_limit, T_init=0.05, T_final=0.002, lr=0.01,
                  beta1=0.9, beta2=0.999, eps=1e-8, schedule='linear',
                  verbose=False, verbose_interval=2000):
    """
    Adam optimizer in log-space with temperature annealing for soft-max smoothing.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    s = a.sum()
    a = a / s
    z = z_from_a(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    no_improve = 0
    max_no_improve = 3000
    last_lr = lr

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        # Temperature schedule
        if schedule == 'linear':
            T = T_init + (T_final - T_init) * progress
        elif schedule == 'exp':
            T = T_init * (T_final / T_init) ** progress
        elif schedule == 'hold_anneal':
            if progress < 0.3:
                T = T_init
            else:
                p2 = (progress - 0.3) / 0.7
                T = T_init * (T_final / T_init) ** p2
        else:
            T = T_init * (T_final / T_init) ** progress

        # Current a from z
        a_curr = softmax_from_z(z)

        # Compute soft-max and weights
        b = fast_convolve(a_curr)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        w = exp_b / exp_b.sum()

        # Gradient in a-space
        grad_a = compute_grad_a(a_curr, w, n, fft_size)

        # Transform to z-space: grad_z = a * (grad_a - a^T * grad_a)
        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Evaluate
        a_new = softmax_from_z(z)
        conv_new = fast_convolve(a_new)
        score = 2.0 * n * conv_new.max()

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive restart
        if no_improve > max_no_improve:
            z = z_from_a(best_a) + np.random.standard_normal(n) * 0.005
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = last_lr * 0.7

        # Slow LR decay
        last_lr = lr * (1.0 - 0.0001)
        lr = max(0.0001, last_lr)

        if verbose and t % verbose_interval == 0:
            print(f"      Adam t={t} T={T:.5f} lr={lr:.6f} score={best_score:.6f}")

    return best_a, best_score


def lbgfs_logspace(a0, time_limit, T=100, maxiter=5000):
    """L-BFGS optimization in log-space with soft-max smoothing."""
    n = len(a0)
    z0 = z_from_a(a0)
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    stopped = [False]

    def soft_max_obj(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        a = softmax_from_z(z)
        b = fast_convolve(a)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        return 2.0 * n * (T * (max_b / T + np.log(exp_b.sum())))

    def soft_max_grad(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros_like(z)
        a = softmax_from_z(z)
        b = fast_convolve(a)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        w = exp_b / exp_b.sum()
        grad_a = compute_grad_a(a, w, n, fft_size)
        S = np.dot(grad_a, a)
        return a * (grad_a - S)

    try:
        res = minimize(
            fun=soft_max_obj,
            x0=z0,
            jac=soft_max_grad,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
    except Exception:
        a_out = a0.copy()

    return a_out


def peak_shave(a0, time_limit, n_trials=5000):
    """
    Peak-targeted local search: identify the autocorrelation peak and
    try to reduce it by adjusting contributing elements.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Find peak (prefer interior peaks)
        b_search = b.copy()
        b_search[0] = -np.inf
        b_search[-1] = -np.inf
        peak_k = int(np.argmax(b_search))
        peak_val = b[peak_k]
        target = best_score / (2.0 * n)
        if peak_val <= target * 1.001:
            continue

        # Elements contributing to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = np.argsort(contribs)[::-1][:5]

        improved = False
        for ci in top_idx:
            i = idx[ci]
            j = int(peak_k - i)
            if i == j or a[j] < 1e-15 or a[i] < 1e-15:
                continue

            for f1 in [0.95, 0.9, 0.85, 0.8]:
                for f2 in [1.0, 1.02, 1.05]:
                    a_trial = a.copy()
                    a_trial[i] *= f1
                    a_trial[j] *= f2
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try single-element reduction
            for ci in top_idx[:3]:
                i = int(idx[ci])
                if a[i] < 1e-15:
                    continue
                for factor in [0.9, 0.8, 0.7]:
                    a_trial = a.copy()
                    diff = a_trial[i] * (1.0 - factor)
                    a_trial[i] *= factor
                    # Redistribute to maintain sum
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-15:
                        continue
                    a_trial[mask] += diff * a_trial[mask] / others_sum
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved and trial > 100:
            # Stuck - try random perturbation
            if trial % 50 == 0:
                a = best_a + np.random.standard_normal(n) * 0.005
                a = np.maximum(a, 0)
                a = a / a.sum()

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_sweeps=200):
    """
    Coordinate descent: for each element, try multiplicative updates
    and keep the best.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        # Random order
        order = rng.permutation(n)
        sweep_improved = False

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue

            best_local = a[i]
            best_local_score = best_score

            for factor in [0.7, 0.8, 0.9, 1.1, 1.2, 1.3]:
                a_trial = a.copy()
                a_trial[i] = a[i] * factor
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_local_score:
                    best_local_score = score
                    best_local = a[i] * factor

            if best_local != a[i]:
                a[i] = best_local
                a = a / a.sum()
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_a = a.copy()
                    sweep_improved = True

        if not sweep_improved:
            break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def diverse_initializations(parent, n, rng):
    """Generate diverse starting points for multi-start optimization."""
    candidates = []

    # Parent-based (if available and right length)
    if parent is not None:
        if len(parent) == n:
            p = np.maximum(parent, 0)
            p = p / p.sum()
            candidates.append(('parent', p.copy()))
            # Perturbed parent
            for scale in [0.01, 0.03, 0.05, 0.1]:
                pert = p + rng.standard_normal(n) * scale
                pert = np.maximum(pert, 0)
                pert = pert / pert.sum()
                candidates.append((f'parent_pert_{scale}', pert))
            # Structured perturbation
            for freq in [2, 3, 5, 8]:
                t = np.linspace(0, 2 * np.pi * freq, n)
                pert = p + 0.03 * np.cos(t)
                pert = np.maximum(pert, 0)
                sp = pert.sum()
                if sp > 1e-10:
                    pert = pert / sp
                    candidates.append((f'parent_cos_{freq}', pert))
        else:
            interp = interpolate_sequence(parent, n)
            candidates.append(('parent_interp', interp))

    # Uniform
    candidates.append(('uniform', np.ones(n) / n))

    # Triangular (concave)
    tri = np.linspace(0, 1, n)
    tri += np.linspace(1, 0, n)
    tri = np.maximum(tri, 0.01)
    tri = tri / tri.sum()
    candidates.append(('triangular', tri))

    # Bimodal
    mid = n // 2
    bi = np.zeros(n)
    bi[max(0, mid - n // 6):mid + n // 6] = 1.0
    bi[max(0, mid - n // 4):mid + n // 4] = 0.5
    bi = np.maximum(bi, 0.001)
    bi = bi / bi.sum()
    candidates.append(('bimodal', bi))

    # Gaussian-like
    x = np.linspace(-3, 3, n)
    gau = np.exp(-x ** 2 / 4)
    gau = np.maximum(gau, 1e-6)
    gau = gau / gau.sum()
    candidates.append(('gaussian', gau))

    # Random
    rand = np.abs(rng.standard_normal(n)) + 0.01
    rand = rand / rand.sum()
    candidates.append(('random', rand))

    # Exponential decay
    expo = np.exp(-np.linspace(0, 3, n))
    expo = expo / expo.sum()
    candidates.append(('expo', expo))

    return candidates


def evaluate_and_track(a, label, best_seq, best_score, print_flag=True):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if print_flag:
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining Adam log-space gradient descent with
    temperature annealing, diverse multi-start, L-BFGS polishing, coordinate
    descent, and peak-targeted local search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_default = 768
        uniform = np.ones(n_default) / n_default
        best_seq = uniform
        best_score = float(evaluate_sequence(uniform.tolist()))
        print(f"Default: n={n_default}, score={best_score:.6f}")

    n_parent = len(best_seq)
    print(f"Starting best score: {best_score:.6f}")

    # === Phase 1: Multi-start Adam at parent length (40% time) ===
    print(f"\n=== Phase 1: Multi-Start Adam (n={n_parent}) ===")
    t_p1 = min(720.0, (deadline - time.monotonic()) * 0.40)
    t_p1_start = time.monotonic()

    # Generate diverse initializations
    inits = diverse_initializations(parent if parent is not None else best_seq,
                                    n_parent, rng)
    # Sort by initial score (quick eval)
    inits_scored = []
    for label, init_a in inits:
        s = 2.0 * n_parent * fast_convolve(init_a).max()
        inits_scored.append((s, label, init_a))
    inits_scored.sort(key=lambda x: x[0])

    # Run Adam on top candidates
    n_starts = min(6, len(inits_scored))
    time_per_start = t_p1 / n_starts

    for idx in range(n_starts):
        rem = deadline - time.monotonic() - 300
        if rem < 60:
            break
        t_alloc = min(time_per_start, rem * 0.9)
        if t_alloc < 30:
            continue

        s_init, label, init_a = inits_scored[idx]
        print(f"  Start {idx+1}/{n_starts}: {label} (init={s_init:.6f}), budget={t_alloc:.0f}s")

        # Try multiple temperature schedules
        for sched, T_i, T_f in [('exp', 0.05, 0.002), ('hold_anneal', 0.03, 0.001),
                                 ('linear', 0.04, 0.003)]:
            rem = deadline - time.monotonic() - 250
            if rem < 20:
                break
            t_sub = min(t_alloc / 3.0, rem * 0.8)
            if t_sub < 15:
                continue

            a_res, score_res = adam_logspace(
                init_a, t_sub, T_init=T_i, T_final=T_f,
                lr=0.008, schedule=sched, verbose=False
            )
            best_seq, best_score = evaluate_and_track(a_res, f"adam_{label}_{sched}", best_seq, best_score)

        print(f"    After Adam starts: best={best_score:.6f}")

    # === Phase 2: Focused refinement on best (25% time) ===
    print(f"\n=== Phase 2: Focused Refinement (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    t_p2 = min(450.0, (deadline - time.monotonic()) * 0.25)
    t_p2_start = time.monotonic()

    # Long Adam run with careful annealing
    t_sub = t_p2 * 0.5
    a_res, score_res = adam_logspace(
        best_seq, t_sub, T_init=0.02, T_final=0.001,
        lr=0.005, schedule='hold_anneal', verbose=True, verbose_interval=3000
    )
    best_seq, best_score = evaluate_and_track(a_res, "adam_focused", best_seq, best_score)
    print(f"  After focused Adam: {best_score:.6f}")

    # L-BFGS polish at multiple temperatures
    rem = deadline - time.monotonic() - 200
    if rem > 30:
        for T in [200, 50, 10]:
            rem = deadline - time.monotonic() - 150
            if rem < 15:
                break
            t_sub = min(rem * 0.3, 60.0)
            a_lb = lbgfs_logspace(best_seq, t_sub, T=T, maxiter=3000)
            best_seq, best_score = evaluate_and_track(a_lb, f"lbgfs_T{T}", best_seq, best_score)
            print(f"  L-BFGS T={T}: {best_score:.6f}")

    # === Phase 3: Local polishing (20% time) ===
    print(f"\n=== Phase 3: Local Polishing (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 150

    if rem > 30:
        t_sub = min(rem * 0.4, 100.0)
        a_ps, _ = peak_shave(best_seq, t_sub, n_trials=3000)
        best_seq, best_score = evaluate_and_track(a_ps, "peak_shave", best_seq, best_score)
        print(f"  After peak shave: {best_score:.6f}")

    rem = deadline - time.monotonic() - 120
    if rem > 20:
        t_sub = min(rem * 0.5, 80.0)
        a_cd, _ = coordinate_descent(best_seq, t_sub, n_sweeps=100)
        best_seq, best_score = evaluate_and_track(a_cd, "coord_desc", best_seq, best_score)
        print(f"  After coord descent: {best_score:.6f}")

    # More peak shaving after coord descent
    rem = deadline - time.monotonic() - 100
    if rem > 15:
        t_sub = min(rem * 0.5, 50.0)
        a_ps2, _ = peak_shave(best_seq, t_sub, n_trials=2000)
        best_seq, best_score = evaluate_and_track(a_ps2, "peak_shave_2", best_seq, best_score)

    # === Phase 4: Try different lengths (10% time) ===
    print(f"\n=== Phase 4: Multi-Length Exploration (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 80

    if rem > 60:
        alt_lengths = []
        for factor in [1.25, 1.5, 0.75, 2.0, 1.1]:
            alt_n = int(n_best * factor)
            if 256 <= alt_n <= 4096 and alt_n != n_best:
                if alt_n not in alt_lengths:
                    alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:3]:
            rem = deadline - time.monotonic() - 40
            if rem < 40:
                break
            t_sub = min(rem * 0.4, 80.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, score_alt = adam_logspace(
                init_alt, t_sub, T_init=0.04, T_final=0.002,
                lr=0.007, schedule='hold_anneal', verbose=False
            )
            best_seq, best_score = evaluate_and_track(a_alt, f"alt_n{alt_n}", best_seq, best_score)
            print(f"  Alt n={alt_n}: {best_score:.6f}")

    # === Phase 5: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 10

    if rem > 20:
        # Final Adam with very low temperature
        t_sub = min(rem * 0.4, 60.0)
        a_fin, _ = adam_logspace(
            best_seq, t_sub, T_init=0.01, T_final=0.0005,
            lr=0.003, schedule='exp', verbose=False
        )
        best_seq, best_score = evaluate_and_track(a_fin, "final_adam", best_seq, best_score)

    rem = deadline - time.monotonic() - 5
    if rem > 10:
        t_sub = min(rem * 0.5, 30.0)
        a_ps_fin, _ = peak_shave(best_seq, t_sub, n_trials=1000)
        best_seq, best_score = evaluate_and_track(a_ps_fin, "final_ps", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        # Fallback
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.6f}")

    return [float(x) for x in best_seq]
```
