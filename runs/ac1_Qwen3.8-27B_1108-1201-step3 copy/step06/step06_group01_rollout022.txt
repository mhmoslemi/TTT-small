def cross_corr_compute(a, p, n):

    Compute (Q(p)*a)_i = sum_j a[j] * p[i+j] for i = 0, ..., n-1
    where a has length n and p has length 2n-1.
    """

    a_pad = np.zeros(2 * n - 1)
    a_pad[:n] = a

    # This is the "valid" cross-correlation of p and a_pad
    
    # Using FFT:
    fft_size = 1
    while fft_size < n + (2*n-1) - 1:
        fft_size *= 2
    
    A = rfft(a_pad, fft_size)
    P = rfft(p, fft_size)
    
    # Cross-correlation: (p cross a_pad)[k] = sum_j p[j+k] * a_pad[j]
    # Using FFT: IFFT(conj(FFT(p)) * FFT(a_pad))

    # For k = n-1: sum_{j=0}^{n-1} p[j+n-1] * a[j]

"""

Uses FFT-based convolution and cross-correlation for efficiency.

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr(a, p, n):
    """
    Compute (Q(p)*a)_i = sum_j a[j] * p[i+j] for i = 0, ..., n-1.
    a has length n, p has length 2n-1.
    """
    a_pad = np.zeros(2 * n - 1)
    a_pad[:n] = a
    
    fft_size = 1
    while fft_size < n + (2 * n - 1):
        fft_size *= 2
    
    A = rfft(a_pad, fft_size)
    P = rfft(p, fft_size)
    
    cross = irfft(np.conj(P) * A, fft_size)
    return cross[:n]


def project_to_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


def score_of(a):
    """Compute the score 2n * max(conv(a,a)) for a probability vector."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * c.max(), c


def lse_gradient_descent(a0, time_budget, T_start=100.0, T_end=1.0, verbose=True):
    """
    Log-sum-exp smoothed gradient descent for minimizing 2n*max(conv(a,a)).
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score, _ = score_of(a)
    
    t0 = time.monotonic()
    T = T_start
    lr = 0.01
    # Adam optimizer
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_step = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_budget:
        # Compute convolution
        c = fast_conv(a)
        
        # Compute softmax weights
        c_shifted = c - c.max()
        exp_Tc = np.exp(T * c_shifted)
        w_sum = exp_Tc.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[c.argmax()] = 1.0
        else:
            w = exp_Tc / w_sum
        
        # Compute gradient: grad[j] = 4n * sum_m a[m] * w[j+m]
        grad = 4.0 * n * cross_corr(a, w, n)
        
        # Adam update
        t_step += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_step)
        v_hat = v / (1 - beta2 ** t_step)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project onto simplex
        a_new = project_to_simplex(a_new)
        
        # Evaluate
        score, _ = score_of(a_new)
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        
        # Anneal temperature
        progress = min(1.0, (time.monotonic() - t0) / time_budget)
        T = T_start * (T_end / T_start) ** progress
        
        # Adaptive learning rate
        if no_improve > 100:
            lr *= 0.5
            no_improve = 0
            # Perturb if stuck
            if lr < 1e-6:
                a = best_a + np.random.default_rng(t_step).standard_normal(n) * 0.01
                a = np.maximum(a, 0)
                ss = a.sum()
                if ss > 1e-15:
                    a = a / ss
                else:
                    a = best_a.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                t_step = 0
                lr = 0.01
                no_improve = 0
        
        if verbose and t_step % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LSE-GD [{elapsed:.0f}s] T={T:.2f} lr={lr:.6f} score={best_score:.6f}")
    
    return best_a, best_score

def interpolate(a, new_n):
    """Interpolate a probability vector to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, a_norm)
    result = np.maximum(result, 0)
    rs = result.sum()
    if rs > 1e-15:
        result = result / rs
    return result
