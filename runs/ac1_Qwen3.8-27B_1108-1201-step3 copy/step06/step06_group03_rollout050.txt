import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def obj_value(a):
    """Compute the objective 2*n*max(conv(a,a))/(sum(a))^2"""
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fast_convolve(a)
    return 2 * len(a) * c.max() / (s * s)

def obj_max_conv(a):
    """Compute max(conv(a,a)) assuming sum(a)=1"""
    return fast_convolve(a).max()

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient(f, c):
    """Compute gradient of max(c) w.r.t. f (exact, at the argmax)."""
    n = len(f)
    k = int(np.argmax(c))
    grad = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    idx = np.arange(js, je + 1)
    grad[js:je + 1] = f[k - idx] + f[idx]
    return grad

def compute_smoothed_gradient(f, c, tau=10.0):
    """Compute smoothed gradient using softmax weights over top peaks."""
    n = len(f)
    len_c = len(c)
    k_top = max(5, int(len_c * 0.1))
    peaks = np.argpartition(c, -k_top)[-k_top:]
    pv = c[peaks]
    scaled = tau * (pv - pv.max())
    w = np.exp(scaled)
    w /= w.sum()
    
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        idx = np.arange(js, je + 1)
        grad[js:je + 1] += w[i] * (f[k - idx] + f[idx])
    return grad

def multiplicative_descent(f0, time_limit, lr=0.005, tau=5.0, verbose=False):
    """Multiplicative (mirror) descent with smoothed gradient."""
    f = np.maximum(f0.astype(np.float64), 1e-15)

    n = len(f)
    
    best_f = f.copy()
    best_obj = f - 0 # placeholder

            lr = 0.005

def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c_obj = -np.ones(n)
    # Build constraint matrix efficiently
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def lp_frank_wolfe(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    c = fast_convolve(f)
    M = c.max()
    best_f = f.copy()
    best_obj = M
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        g = lp_direction(f, M, n)
        if g is None:
            # Relax the constraint
            M *= 1.05
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            continue
        g = g / g_sum
        
        diff = np.max(np.abs(g - f))
        if diff < 1e-10:
            no_improve += 1
            if no_improve > 100:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = obj_max_conv(ft)
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            f = f_new
            M = obj_max_conv(f)
            no_improve = 0
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.8f}")
        else:
            no_improve += 1
    
    return best_f, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_max_conv(a):
    """Compute max(conv(a,a)) assuming sum(a)=1."""
    return fast_convolve(a).max()


def obj_value(a):
    """Compute the full objective 2*n*max(conv(a,a))/(sum(a))^2."""
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fast_convolve(a)
    return 2 * len(a) * c.max() / (s * s)


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_gradient(f, c, tau=10.0, top_frac=0.15):
    """Compute smoothed gradient using weighted peaks."""
    n = len(f)
    len_c = len(c)
    k_top = max(5, int(len_c * top_frac))
    peaks = np.argpartition(c, -k_top)[-k_top:]
    pv = c[peaks]
    scaled = tau * (pv - pv.max())
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        idx = np.arange(js, je + 1)
        grad[js:je + 1] += w[i] * (f[k - idx] + f[idx])
    return grad

def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))

    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

            for damp in [0.5, 0.25, 0.1]:
                f_trial = (1 - damp) * f + damp * f_new
                f_trial = np.maximum(f_trial, 1e-15)
                f_trial = f_trial / f_trial.sum()
                if obj_max_conv(f_trial) < M:
                    f = f_trial
                    accepted = True
                    break
            if not accepted:
                lr *= 0.75
                no_improve += 1

        if no_improve > 100:
            rng = np.random.default_rng(it * 29 + 7)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-15)
            f = f / f.sum()
            lr = 0.01
            no_improve = 0

        if lr < 1e-8:
            break

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with multiplicative perturbations."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = obj_max_conv(f)
    t0 = time.monotonic()

    multipliers = [0.2, 0.4, 0.6, 0.8, 1.2, 1.4, 1.6, 1.8, 2.0, 0.1]

    while time.monotonic() - t0 < time_limit:
        improved = False
        rng = np.random.default_rng(int((time.monotonic() - t0) * 1000) % 1000000 + 1)
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj

            old_val = f[j]
            if old_val < 1e-14:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                f_trial = f_trial / f_trial.sum()
                val = obj_max_conv(f_trial)
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                f[j] = best_j_val
                f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            # Try with different random ordering one more time
            rng2 = np.random.default_rng(int((time.monotonic() - t0) * 777) % 1000000 + 2)
            order2 = rng2.permutation(n)
            any_improved = False
            for j in order2:
                if time.monotonic() - t0 > time_limit:
                    break
                old_val = f[j]
                if old_val < 1e-14:
                    continue
                best_j_val = old_val
                best_j_obj = best_obj
                for mult in [0.3, 0.5, 0.7, 1.3, 1.5, 2.0]:
                    new_val = old_val * mult
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    f_trial = f_trial / f_trial.sum()
                    val = obj_max_conv(f_trial)
                    if val < best_j_obj - 1e-14:
                        best_j_obj = val
                        best_j_val = new_val
                if best_j_val != old_val:
                    f[j] = best_j_val
                    f = f / f.sum()
                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        any_improved = True
            if not any_improved:
                break

    return best_f, best_obj


def make_initializations(n, seed=42):
    """Generate diverse asymmetric initializations."""
    inits = []
    rng = np.random.default_rng(seed)

    # Exponential decay (various rates)
    for alpha in [0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = np.exp(-np.arange(n) * alpha / n)
        a = a / a.sum()
        inits.append(a)

    # Power law
    for beta in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        a = (np.arange(n) + 1) ** (-beta)
        a = a / a.sum()
        inits.append(a)

    # Cosine window (front-heavy)
    for arr in [0.3, 0.5, 0.8, 1.0, 1.5]:
        x = np.arange(n) / n
        a = np.cos(np.pi * x * arr / 2) ** 2
        a = np.maximum(a, 0)
        if a.sum() > 1e-10:
            a = a / a.sum()
            inits.append(a)

    # Sine-modulated exponential
    for freq in [1, 2, 3]:
        for damp in [1.0, 2.0, 3.0]:
            x = np.arange(n) / n
            a = np.exp(-damp * x) * (1.0 + 0.3 * np.sin(2 * np.pi * freq * x))
            a = np.maximum(a, 0)
            if a.sum() > 1e-10:
                a = a / a.sum()
                inits.append(a)

    # Random structured
    for _ in range(5):
        a = rng.random(n)
        a = a ** rng.uniform(1.0, 3.0)
        a = a / a.sum()
        inits.append(a)

    # Two-bump asymmetric
    for pos1, pos2, ratio in [(0.1, 0.5, 0.5), (0.2, 0.7, 0.3), (0.15, 0.4, 0.7)]:
        x = np.arange(n) / n
        a = np.exp(-80 * (x - pos1) ** 2) + ratio * np.exp(-80 * (x - pos2) ** 2)
        a = a / a.sum()
        inits.append(a)

    # Quadratic / polynomial
    for c1 in [0.0, 0.2, 0.5, 0.8]:
        x = np.arange(n) / n
        a = (1.0 - c1 * x) ** 2 + 0.05 * x ** 2
        a = a / a.sum()
        inits.append(a)

    # "Bump" concentrated at start
    for width in [0.1, 0.2, 0.3, 0.5]:
        x = np.arange(n) / n
        a = np.exp(-50 * (x / width) ** 2) + 0.01
        a = a / a.sum()
        inits.append(a)

    return inits


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def simulated_annealing(f0, time_limit, verbose=False):
    """Simulated annealing on the sequence."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = obj_max_conv(f)

    t0 = time.monotonic()
    temperature = 0.01
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rng = np.random.default_rng(it * 101 + 17)

        # Random perturbation
        f_new = f.copy()
        n_perturb = rng.integers(max(1, n // 100), max(2, n // 20))
        indices = rng.choice(n, size=n_perturb, replace=False)
        for idx in indices:
            f_new[idx] *= rng.uniform(0.5, 1.5)
        f_new = np.maximum(f_new, 0.0)
        ss = f_new.sum()
        if ss < 1e-15:
            continue
        f_new = f_new / ss

        new_obj = obj_max_conv(f_new)
        diff = new_obj - obj_max_conv(f)

        if diff < 0 or rng.random() < np.exp(-diff / (temperature + 1e-10)):
            f = f_new
            if new_obj < best_obj - 1e-15:
                best_obj = new_obj
                best_f = f.copy()
                if verbose and it % 50 == 0:
                    print(f"      SA iter {it}: obj={best_obj:.10f}, T={temperature:.5f}")

        temperature *= 0.999
        if temperature < 0.001:
            temperature = 0.01

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Combines LP-Frank-Wolfe, multiplicative descent, peak shaving,
    coordinate descent, and simulated annealing across multiple lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def record(a, label):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Optimize the parent with multiple methods ===
    print("\n=== Phase 1: Optimize Parent ===")
    if parent is not None:
        n_p = len(parent)

        # LP-FW (most direct method for peak reduction)
        rem = deadline - time.monotonic() - 900
        if rem > 60:
            t_alloc = min(rem, 200)
            print(f"  [1a] LP-FW on parent (n={n_p}): t={t_alloc:.0f}s")
            a_opt, obj = lp_frank_wolfe(parent, time_limit=t_alloc, verbose=True)
            record(a_opt, "lpfw_parent")

        # Multiplicative descent
        rem = deadline - time.monotonic() - 800
        if rem > 40:
            t_alloc = min(rem, 120)
            print(f"  [1b] MD on best: t={t_alloc:.0f}s")
            if best_seq is not None:
                a_opt, obj = multiplicative_descent(best_seq, time_limit=t_alloc, verbose=True)
                record(a_opt, "md_parent")

        # Peak shaving
        rem = deadline - time.monotonic() - 700
        if rem > 30:
            t_alloc = min(rem, 80)
            print(f"  [1c] Peak shaving: t={t_alloc:.0f}s")
            if best_seq is not None:
                a_opt, obj = peak_shaving(best_seq, time_limit=t_alloc, verbose=True)
                record(a_opt, "peakshave_parent")

        # Coordinate descent
        rem = deadline - time.monotonic() - 600
        if rem > 30:
            t_alloc = min(rem, 60)
            print(f"  [1d] CD: t={t_alloc:.0f}s")
            if best_seq is not None:
                a_opt, obj = coordinate_descent(best_seq, time_limit=t_alloc)
                record(a_opt, "cd_parent")

    # === Phase 2: Explore different lengths with new initializations ===
    print("\n=== Phase 2: New Initializations at Various Lengths ===")
    test_lengths = [384, 512, 640, 768, 896, 1024, 1152]

    for n_try in test_lengths:
        rem = deadline - time.monotonic() - 250
        if rem < 60:
            break

        inits = make_initializations(n_try, seed=seed + n_try)

        # Score and rank initializations
        scored = []
        for i, a in enumerate(inits):
            obj = obj_max_conv(a)
            scored.append((obj, a, i))
        scored.sort()

        # Try top 2 with MD
        for idx in range(min(2, len(scored))):
            rem = deadline - time.monotonic() - 150
            if rem < 25:
                break
            obj_init, a_init, init_idx = scored[idx]
            t_alloc = min(rem * 0.4, 60)
            if t_alloc < 15:
                t_alloc = 15
            print(f"  [2] n={n_try}, init#{init_idx} (obj={obj_init:.8f}): MD t={t_alloc:.0f}s")
            a_opt, obj = multiplicative_descent(a_init, time_limit=t_alloc)
            record(a_opt, f"md_n{n_try}_{init_idx}")

            # If promising, follow with LP-FW
            rem = deadline - time.monotonic() - 100
            if rem > 30:
                t_alloc2 = min(rem * 0.5, 40)
                if t_alloc2 > 10:
                    a_opt2, obj2 = lp_frank_wolfe(a_opt, time_limit=t_alloc2)
                    record(a_opt2, f"lpfw_n{n_try}_{init_idx}")

    # === Phase 3: Interpolate best to other lengths ===
    print("\n=== Phase 3: Multi-Length Interpolation ===")
    if best_seq is not None:
        for target_n in [512, 640, 768, 896, 1024, 1280, 1536]:
            rem = deadline - time.monotonic() - 150
            if rem < 30:
                break

            if len(best_seq) == target_n:
                continue

            f_interp = interpolate_sequence(best_seq, target_n)
            record(f_interp, f"interp_n{target_n}")

            rem = deadline - time.monotonic() - 100
            if rem > 25:
                t_alloc = min(rem * 0.4, 40)
                if t_alloc > 10:
                    a_opt, obj = multiplicative_descent(f_interp, time_limit=t_alloc)
                    record(a_opt, f"md_interp_n{target_n}")

    # === Phase 4: Simulated annealing on best ===
    print("\n=== Phase 4: Simulated Annealing ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 30:
            t_alloc = min(rem, 60)
            print(f"  SA on best (n={len(best_seq)}): t={t_alloc:.0f}s")
            a_opt, obj = simulated_annealing(best_seq, time_limit=t_alloc)
            record(a_opt, "sa")

    # === Phase 5: Final squeeze ===
    print("\n=== Phase 5: Final Squeeze ===")
    if best_seq is not None:
        # LP-FW with strong focus
        rem = deadline - time.monotonic() - 40
        if rem > 25:
            t_alloc = min(rem, 50)
            print(f"  Final LP-FW: t={t_alloc:.0f}s")
            a_opt, obj = lp_frank_wolfe(best_seq, time_limit=t_alloc, verbose=True)
            record(a_opt, "final_lpfw")

        # MD with very low lr
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_alloc = min(rem, 30)
            print(f"  Final MD (low lr): t={t_alloc:.0f}s")
            a_opt, obj = multiplicative_descent(best_seq, time_limit=t_alloc, lr=0.002, tau_init=20.0)
            record(a_opt, "final_md")

        # Coordinate descent polish
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            t_alloc = min(rem, 15)
            print(f"  Final CD: t={t_alloc:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_alloc)
            record(a_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
