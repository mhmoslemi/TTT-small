```python
LP-guided optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Key innovation: Use the LP duality to compute optimal subgradient weights
at each step, giving the "best" direction for reducing the peak of the
auto-convolution. Combined with adaptive step sizes, multi-length
exploration, and final L-BFGS/coordinate-descent polishing.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import cvxpy as cp

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_F_matrix(a):
    """Build the (2n-1) x n matrix F where F[k,j] = a[k-j] if valid."""
    n = len(a)
    F = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            F[k, j] = a[k - j]
    return F

def solve_lp_direction(a, M):
    """Solve LP: max sum(g) s.t. F@g <= M*1, g >= 0. Returns normalized g."""
    n = len(a)
    F = build_F_matrix(a)
    g = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(g))
    constraints = [F @ g <= M * np.ones(2*n-1)]
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iter=100000)
        if g.value is not None and np.isfinite(g.value).all():
            gv = np.maximum(g.value, 0)
            s = gv.sum()
            if s > 1e-10:
                return gv / s
    except Exception:
        pass
    return None
```

```python
"""
LP-guided optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Key innovations over previous approach:
- Solves the LP dual to compute OPTIMAL subgradient weights at each step
  (rather than heuristic temperature-weighted peaks)
- QP-regularized LP for stability
- Adaptive step size with backtracking
- Multi-phase strategy: LP-guided -> subgradient -> L-BFGS -> multi-length -> polish
- Exploits the KKT structure of the piecewise-linear objective
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_F_matrix(a):
    """Build (2n-1) x n matrix F where F[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    F = np.zeros((2 * n - 1, n))
    for j in range(n):
        # F[k,j] = a[k-j] for k = j, j+1, ..., j+n-1
        F[j:j+n, j] = a
    return F


def lp_guided_step(a, M, reg=0.0):
    """
    Solve: max sum(g) - reg*sum(g^2) s.t. conv(a,g) <= M, g >= 0.
    Returns normalized g* or None.
    """
    n = len(a)
    a_safe = a + 1e-12  # ensure positive
    F = build_F_matrix(a_safe)
    g = cp.Variable(n, nonneg=True)
    
    if reg > 0:
        objective = cp.Maximize(cp.sum(g) - reg * cp.sum_squares(g))
    else:
        objective = cp.Maximize(cp.sum(g))
    
    constraints = [F @ g <= M * np.ones(2 * n - 1)]
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, max_iter=50000, reltol=1e-7, abstol=1e-7)
        if g.value is not None:
            gv = np.maximum(g.value, 0.0)
            s = gv.sum()
            if s > 1e-10 and np.isfinite(gv).all():
                return gv / s
    except Exception:
        pass
    
    # Fallback: try SCS
    try:
        prob.solve(solver=cp.SCS, max_iters=100000, eps=1e-7)
        if g.value is not None:
            gv = np.maximum(g.value, 0.0)
            s = gv.sum()
            if s > 1e-10 and np.isfinite(gv).all():
                return gv / s
    except Exception:
        pass
    
    return None


def compute_subgradient_weights_from_lp(a, M):
    """
    Compute optimal peak weights from the LP dual.
    Returns weight vector w of length (2n-1) summing to 1.
    """
    n = len(a)
    a_safe = a + 1e-12
    F = build_F_matrix(a_safe)
    
    y = cp.Variable(2 * n - 1, nonneg=True)
    objective = cp.Minimize(cp.sum(y))
    constraints = [F.T @ y >= (1.0 / M) * np.ones(n)]
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, max_iter=50000, reltol=1e-7, abstol=1e-7)
        if y.value is not None:
            yv = np.maximum(y.value, 0.0)
            s = yv.sum()
            if s > 1e-10:
                return yv / s
    except Exception:
        pass
    
    return None


def subgradient_step_with_weights(a, w, lr):
    """
    One subgradient step using given peak weights w.
    g[j] = sum_k w[k] * a[k-j] for valid (k,j).
    """
    n = len(a)
    b = fast_convolve(a)
    
    # Compute subgradient via convolution
    # g = conv(a, w_reversed) restricted to [0, n)
    # Actually: g[j] = sum_k w[k] * a[k-j] where 0<=k-j<n
    # This is the cross-correlation of a and w
    # = conv(a, flip(w))[j] for j in [0, n)
    w_flipped = w[::-1]
    fft_size = 1 << (n + len(w) - 1 - 1).bit_length()
    A = rfft(a, fft_size)
    Wf = rfft(w_flipped, fft_size)
    g_raw = irfft(A * Wf, fft_size)
    g = g_raw[:n]
    
    return g


def subgradient_optimize_lp_weights(a0, time_limit, lr_init=0.003, momentum=0.9,
                                     kick_interval=2000, kick_size=0.003,
                                     lp_interval=50, seed=42):
    """
    Subgradient with periodic LP weight updates.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_conv = fast_convolve(a).max()
    cur_lr = lr_init
    no_improve = 0
    it = 0
    
    # Initial weights: uniform over all peaks
    b = fast_convolve(a)
    w_current = np.ones(len(b)) / len(b)
    
    while time.monotonic() - t0 < time_limit:
        # Periodically update weights via LP
        if it % lp_interval == 0:
            M = b.max()
            w_lp = compute_subgradient_weights_from_lp(a, M)
            if w_lp is not None:
                w_current = w_lp
            
            # Also compute from primal LP for direction
            g_star = lp_guided_step(a, M, reg=0.01)
            if g_star is not None:
                # Blend current a with g_star
                alpha = 0.1
                a_blend = (1 - alpha) * a + alpha * g_star
                a_blend = project_simplex(a_blend)
                b_blend = fast_convolve(a_blend)
                if b_blend.max() < best_conv:
                    best_conv = b_blend.max()
                    best_a = a_blend.copy()
                    a = a_blend.copy()
                    no_improve = 0
                it += 1
                continue
        
        # Subgradient step
        g = subgradient_step_with_weights(a, w_current, cur_lr)
        v = momentum * v - cur_lr * g
        a_new = project_simplex(a + v)
        
        b_new = fast_convolve(a_new)
        obj_new = b_new.max()
        
        if obj_new < best_conv - 1e-15:
            best_conv = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        b = b_new
        it += 1
        
        # Adaptive restart
        if no_improve > kick_interval:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.7, lr_init * 0.1)
            no_improve = 0
        elif no_improve > kick_interval // 2:
            cur_lr *= 0.95
    
    return best_a, best_conv

    cur_lr = lr_init
    cur_temp = temp_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()
        
        scaled = b / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
        else:
            w /= w_sum
        
        if n_top is not None and n_top < len(b):

            w_sel = w[top_idx]
            w_sel /= w_sel.sum()
            w_full = np.zeros(len(b))

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def soft_max_obj_grad(z, T, n):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    
    # Gradient of cross-correlation
    w_flipped = w[::-1]
    fft_size = 1 << (n + len(w) - 1 - 1).bit_length()
    A = rfft(a, fft_size)
    Wf = rfft(w_flipped, fft_size)
    grad_a = 2.0 * n * irfft(A * Wf, fft_size)[:n]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z


def lbgfs_refine(a0, T, time_limit, maxiter=5000):
    t0 = time.monotonic()
    n = len(a0)
    z0 = z_from_a(a0)
    best_obj = soft_max_obj_grad(z0, T, n)[0]
    best_z = z0.copy()
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        obj, _ = soft_max_obj_grad(z, T, n)
        return obj
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = soft_max_obj_grad(z, T, n)
        return g
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        final_obj = func(res.x)
        if np.isfinite(final_obj) and final_obj < best_obj:
            best_z = res.x
            best_obj = final_obj
    except Exception:
        pass
    
    return softmax_from_z(best_z), best_obj


def coordinate_polish(a0, time_limit):
    """Coordinate descent with re-normalization."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    multipliers = np.array([0.5, 0.7, 0.85, 1.15, 1.3, 1.5])
    
    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.9999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian_init(n, sigma=0.3):

    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_two_hump_init(n, sep=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-20 * (x - 0.25 + sep) ** 2) + np.exp(-20 * (x - 0.75 - sep) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_arcosine_init(n):
    """Arc-cosine shaped (more peaky at edges)."""
    x = np.linspace(0.01, 0.99, n)
    a = 1.0 / np.sqrt(x * (1 - x))
    a = np.maximum(a, 0)
    a = a ** 0.5  # soften
    return a / a.sum()


def eval_local(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def lp_guided_main_loop(a0, time_limit, seed=42):
    """
    Main LP-guided optimization loop with adaptive step size.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_conv = fast_convolve(a).max()
    best_obj = 2.0 * n * best_conv
    
    step_size = 0.15
    lp_count = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        M = b.max()
        
        # Solve LP to get optimal direction
        g_star = lp_guided_step(a, M, reg=0.005)
        lp_count += 1
        
        if g_star is None:
            # Fallback: uniform direction
            g_star = np.ones(n) / n
        
        # Adaptive step: try to mix
        for alpha in [step_size, step_size * 0.5, step_size * 2.0, 0.05]:
            a_trial = (1 - alpha) * a + alpha * g_star
            a_trial = project_simplex(a_trial)
            b_trial = fast_convolve(a_trial)
            if b_trial.max() < best_conv:
                best_conv = b_trial.max()
                best_a = a_trial.copy()
                a = a_trial.copy()
                no_improve = 0
                # Increase step if it worked well
                if b_trial.max() < best_conv * 0.999:
                    step_size = min(step_size * 1.1, 0.4)
                break
        else:
            no_improve += 1
            step_size = max(step_size * 0.7, 0.01)
        
        # Periodic perturbation
        if no_improve > 20:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.01)
            step_size = 0.1
            no_improve = 0
        
        # Print progress
        if lp_count % 50 == 0:
            cur_obj = 2.0 * n * best_conv
            elapsed = time.monotonic() - t0
            print(f"    LP#{lp_count}: obj={cur_obj:.6f}, conv={best_conv:.8f}, "
                  f"t={elapsed:.0f}s, step={step_size:.3f}")
    
    return best_a, best_conv


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining LP-guided steps, subgradient,
    L-BFGS, multi-length exploration, and coordinate polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  *** NEW BEST [{label}]: {score:.6f} (n={len(a_norm)}) ***")
        return score
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            try_seq(parent, "parent")
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(1024) / 1024
        print("No parent, using uniform n=1024")
    
    n_parent = len(parent)
    
    # === Phase 1: LP-guided optimization (600s) ===
    print(f"\n=== Phase 1: LP-guided optimization (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem, 600.0)
    if t_alloc > 60:
        print(f"  Allocating {t_alloc:.0f}s")
        a_lp, conv_lp = lp_guided_main_loop(parent, time_limit=t_alloc, seed=seed)
        try_seq(a_lp, f"lp_guided_n{n_parent}")
    
    # === Phase 2: Subgradient with LP-optimized weights (250s) ===
    print(f"\n=== Phase 2: Subgradient + LP weights (score={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t_alloc = min(rem, 250.0)
        if t_alloc > 30:
            print(f"  Allocating {t_alloc:.0f}s, n={n_best}")
            a_sg, _ = subgradient_optimize_lp_weights(
                best_seq, time_limit=t_alloc,
                lr_init=0.002, momentum=0.9,
                lp_interval=30, kick_interval=2000,
                kick_size=0.002, seed=seed + 1
            )
            try_seq(a_sg, f"sg_lp_n{n_best}")
    
    # === Phase 3: Standard subgradient for refinement (200s) ===
    print(f"\n=== Phase 3: Fine subgradient (score={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t_alloc = min(rem, 200.0)
        if t_alloc > 30:
            print(f"  Allocating {t_alloc:.0f}s")
            a_sg2, _ = temperature_weighted_subgradient(
                best_seq, time_limit=t_alloc,
                lr_init=0.001, momentum=0.93,
                temp_init=0.0002, temp_growth=1.01,
                kick_interval=3000, kick_size=0.001,
                n_top=max(20, n_best // 8),
                seed=seed + 2
            )
            try_seq(a_sg2, f"sg_fine_n{n_best}")
    
    # === Phase 4: L-BFGS multi-temperature (250s) ===
    print(f"\n=== Phase 4: L-BFGS (score={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for T in [30, 80, 150, 300]:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            t_alloc = min(rem * 0.25, 60.0)
            if t_alloc < 10:
                continue
            print(f"  L-BFGS T={T}, t={t_alloc:.0f}s")
            try:
                a_lb, _ = lbgfs_refine(best_seq, T=T, time_limit=t_alloc, maxiter=3000)
                try_seq(a_lb, f"lbgfs_T{T}")
            except Exception as e:
                print(f"    L-BFGS failed: {e}")
    
    # === Phase 5: Multi-length exploration (200s) ===
    print(f"\n=== Phase 5: Multi-length (score={best_score:.6f}) ===")
    if best_seq is not None:
        target_lengths = []
        for factor in [0.75, 1.25, 1.5, 2.0]:
            tl = int(n_parent * factor)
            tl = max(512, min(3072, tl))
            if tl not in target_lengths and abs(tl - n_parent) > 50:
                target_lengths.append(tl)
        target_lengths.sort(key=lambda x: abs(x - n_parent))
        
        for target_n in target_lengths[:3]:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.4, 50.0)
            if t_alloc < 10:
                continue
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = temperature_weighted_subgradient(
                f_interp, time_limit=t_alloc,
                lr_init=0.003, momentum=0.9,
                temp_init=0.001, temp_growth=1.02,
                kick_interval=1500, kick_size=0.003,
                n_top=max(10, target_n // 15),
                seed=seed + target_n
            )
            try_seq(a_opt, f"len_n{target_n}")
            
            # Quick LP step on best at this length
            if best_seq is not None and len(best_seq) == target_n:
                rem = deadline - time.monotonic()
                if rem > 20:
                    t_lp = min(rem * 0.3, 30.0)
                    a_lp2, _ = lp_guided_main_loop(best_seq, time_limit=t_lp, seed=seed+99)
                    try_seq(a_lp2, f"lp_n{target_n}")
    
    # === Phase 6: Diverse structural initializations (150s) ===
    print(f"\n=== Phase 6: Diverse inits (score={best_score:.6f}) ===")
    if best_seq is not None:
        n_base = len(best_seq)
    else:
        n_base = n_parent
    
    inits = {
        "gauss_0.2": make_gaussian_init(n_base, 0.2),
        "gauss_0.4": make_gaussian_init(n_base, 0.4),
        "cosine": make_cosine_init(n_base),
        "two_hump": make_two_hump_init(n_base),
        "arcosine": make_arcosine_init(n_base),
    }
    
    for name, init_a in inits.items():
        rem = deadline - time.monotonic()
        if rem < 15:
            break
        t_alloc = min(rem * 0.2, 30.0)
        if t_alloc < 5:
            continue
        print(f"  {name}")
        a_opt, _ = temperature_weighted_subgradient(
            init_a, time_limit=t_alloc,
            lr_init=0.005, momentum=0.9,
            temp_init=0.002, temp_growth=1.03,
            kick_interval=1000, kick_size=0.005,
            n_top=max(10, n_base // 20),
            seed=seed + hash(name) % 10000
        )
        try_seq(a_opt, f"div_{name}")
    
    # === Phase 7: Coordinate descent polish (100s) ===
    print(f"\n=== Phase 7: CD polish (score={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_cd = min(rem, 80.0)
        if t_cd > 10:
            print(f"  t={t_cd:.0f}s")
            a_cd, _ = coordinate_polish(best_seq, time_limit=t_cd)
            try_seq(a_cd, "cd_polish")
    
    # === Phase 8: Final micro-optimization (remaining) ===
    print(f"\n=== Phase 8: Final micro (score={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        if rem > 15:
            # Very fine subgradient
            t_micro = min(rem * 0.5, 40.0)
            a_micro, _ = temperature_weighted_subgradient(
                best_seq, time_limit=t_micro,
                lr_init=0.0003, momentum=0.97,
                temp_init=0.0001, temp_growth=1.003,
                kick_interval=5000, kick_size=0.0005,
                n_top=max(30, n_best // 5),
                seed=seed + 777777
            )
            try_seq(a_micro, "micro_sg")
            
            # Another LP round
            rem = deadline - time.monotonic()
            if rem > 20:
                t_final_lp = min(rem * 0.5, 30.0)
                a_flp, _ = lp_guided_main_loop(best_seq, time_limit=t_final_lp, seed=seed + 55555)
                try_seq(a_flp, "final_lp")
            
            # Final L-BFGS
            rem = deadline - time.monotonic()
            if rem > 10:
                t_fin = min(rem, 20.0)
                a_fin, _ = lbgfs_refine(best_seq, T=100, time_limit=t_fin, maxiter=2000)
                try_seq(a_fin, "final_lbgfs")
    
    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")
    
    return [float(x) for x in best_seq]
```
