    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(a, peak_indices):
    """Build sparse matrix for (a*g)_k constraints at given peak indices."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(peak_indices):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(idx)
            cols.append(i)
            vals.append(a[k - i])
    A_lp = csr_matrix((vals, (rows, cols)), shape=(len(peak_indices), n))
    return A_lp

def lp_direction(a, M, top_k=50, method='highs'):
    """
    Find LP direction: max sum(g) s.t. g>=0, (a*g)_k <= M for top-k peaks.
    Returns g normalized to sum=1, or None if infeasible.
    """
    n = len(a)
    conv = fast_convolve(a)
    K = min(top_k, len(conv))
    if K < len(conv):
        peak_idx = np.argpartition(conv, -K)[-K:]
    else:
        peak_idx = np.arange(len(conv))
    
    A_lp = build_conv_matrix(a, peak_idx)
    c = -np.ones(n)
    
    res = linprog(c, A_ub=A_lp, b_ub=np.full(len(peak_idx), M), 
                  bounds=[(0, None)] * n, method=method, options={'presolve': True})
    
    if res.status == 0:
        gs = res.x.sum()
        if gs > 1e-10:
            return res.x / gs
    return None

def make_bell(n, width_frac=0.6, power=2):

    kernel = np.ones(kernel_size) / kernel_size
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def symmetrize(a):
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def coordinate_descent_polish(a0, time_limit, verbose=False, label=""):
    """Coordinate descent: adjust each coordinate with optimal redistribution."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            best_j_obj = best_obj
            best_j_val = old_val
            for mult in [0.8, 0.9, 0.95, 1.05, 1.1, 1.25]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial = trial / ss
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        if not improved:
            break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase LP-based optimization for minimizing max autocorrelation.
    Phases: diverse init -> main LP -> multi-length exploration -> fine polish -> CD.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    n_base = 768
    if parent is not None:
        n_base = len(parent)

    # === Phase 1: Diverse Initialization ===
    print("=== Phase 1: Initialization ===")
    init_candidates = []
    for n_try in [n_base]:
        for sigma in [0.2, 0.25, 0.3, 0.35, 0.4, 0.45, 0.5]:
            init_candidates.append((make_gaussian(n_try, sigma), f"gauss_s{sigma}"))
        for w in [0.4, 0.5, 0.6, 0.7, 0.8]:
            for pwr in [1, 2, 3, 4, 6]:
                init_candidates.append((make_bell(n_try, w, pwr), f"bell_w{w}_p{pwr}"))
        init_candidates.append((make_bimodal(n_try), "bimodal"))
        for sd in [1, 2, 3, 4, 5]:
            init_candidates.append((make_smooth_random(n_try, seed=sd), f"rand_{sd}"))

    if parent is not None:
        init_candidates.append((parent, "parent"))
        init_candidates.append((symmetrize(parent), "parent_sym"))

    best_init = None
    best_init_obj = float('inf')
    for a_init, lbl in init_candidates:
        s = a_init.sum()
        if s < 1e-10:
            continue
        a_norm = a_init / s
        obj = fast_convolve(a_norm).max()
        c_obj = 2 * len(a_norm) * obj
        if c_obj < best_init_obj:
            best_init_obj = c_obj
            best_init = a_norm
        try_seq(a_norm, f"init_{lbl}")

    if best_init is None:
        best_init = make_gaussian(n_base, 0.3)

    print(f"  Best init: obj={best_init_obj:.6f}")

    # === Phase 2: Main LP from best init ===
    print(f"\n=== Phase 2: LP Main (from best init) ===")
    remaining = deadline - time.monotonic()
    t_main = remaining * 0.35
    print(f"  Time: {t_main:.0f}s")
    a_opt, obj = lp_optimize(best_init, time_limit=t_main, top_k=80, verbose=True, label="LP_init")
    try_seq(a_opt, "LP_init")

    # === Phase 3: LP from parent ===
    if parent is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            t_par = remaining * 0.30
            print(f"\n=== Phase 3: LP from parent (t={t_par:.0f}s) ===")
            p_interp = interpolate_sequence(parent, len(best_seq) if best_seq is not None else n_base)
            a_opt, obj = lp_optimize(p_interp, time_limit=t_par, top_k=80, verbose=True, label="LP_parent")
            try_seq(a_opt, "LP_parent")

    # === Phase 4: LP from symmetrized parent ===
    if parent is not None:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            t_sym = remaining * 0.20
            print(f"\n=== Phase 4: LP from sym parent (t={t_sym:.0f}s) ===")
            p_sym = symmetrize(parent)
            a_opt, obj = lp_optimize(p_sym, time_limit=t_sym, top_k=60, verbose=True, label="LP_sym")
            try_seq(a_opt, "LP_sym")

    # === Phase 5: Try different lengths ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            print(f"\n=== Phase 5: Multi-length exploration ===")
            t_per = min(remaining * 0.10, 45)
            target_n = len(best_seq)
            for n_try in [target_n - 64, target_n + 64, target_n + 128]:
                if n_try < 100 or n_try > 2000:
                    continue
                remaining = deadline - time.monotonic()
                if remaining < 30:
                    break
                a_resized = interpolate_sequence(best_seq, n_try)
                t_alloc = min(t_per, remaining - 20)
                if t_alloc < 15:
                    break
                a_opt, obj = lp_optimize(a_resized, time_limit=t_alloc, top_k=50,
                                         verbose=False, label=f"LP_n{n_try}")
                try_seq(a_opt, f"LP_n{n_try}")

    # === Phase 6: Fine LP polish ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            t_polish = min(remaining * 0.30, 120)
            print(f"\n=== Phase 6: Fine LP Polish (t={t_polish:.0f}s) ===")
            a_opt, obj = fine_polish(best_seq, time_limit=t_polish, verbose=True, label="polish")
            try_seq(a_opt, "polish")
            # Also try on symmetrized
            a_s = symmetrize(a_opt)
            try_seq(a_s, "polish_sym")
            remaining = deadline - time.monotonic()
            if remaining > 20 and best_seq is not None:
                a_opt2, obj2 = fine_polish(best_seq, time_limit=min(remaining - 5, 60),
                                           verbose=False, label="polish2")
                try_seq(a_opt2, "polish2")

    # === Phase 7: Coordinate Descent ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            t_cd = min(remaining * 0.40, 120)
            print(f"\n=== Phase 7: Coordinate Descent (t={t_cd:.0f}s) ===")
            a_opt, obj = coordinate_descent_polish(best_seq, time_limit=t_cd,
                                                   verbose=True, label="CD")
            try_seq(a_opt, "CD")

    # === Phase 8: Final ultra-fine LP ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            t_final = min(remaining - 5, 60)
            print(f"\n=== Phase 8: Final LP (t={t_final:.0f}s) ===")
            a_opt, obj = lp_optimize(best_seq, time_limit=t_final, top_k=5,
                                     verbose=True, label="final")
            try_seq(a_opt, "final")

    # === Phase 9: Symmetrization check ===
    if best_seq is not None:
        a_s = symmetrize(best_seq)
        try_seq(a_s, "final_sym")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
