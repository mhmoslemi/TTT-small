"""

with adaptive relaxation, multi-length exploration, and gradient-based refinement.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build the convolution constraint matrix for the LP.
    A[k, j] = f[k-j] if 0<=k-j<n, else 0.
    Returns a dense matrix (for n<=1536 this is manageable).
    """
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = f[k - j]
    return A


def build_conv_matrix_fast(f, n):
    """Build convolution matrix using FFT-based approach (faster for large n)."""
    A = np.zeros((2 * n - 1, n))
    # Column j has f[0], f[1], ..., f[n-1-j] in rows 0..n-1-j
    # and f[j+1], ..., f[n-1] in rows n+j..2n-2+j... 
    # Actually let's just do it directly but vectorized
    for j in range(n):
        # Column j: A[k, j] = f[k-j] for k=j..j+n-1, but k must be in 0..2n-2
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        # f[k-j] for k in k_start..k_end means f[0]..f[k_end-j]
        A[k_start:k_end + 1, j] = f[0:k_end - k_start + 1]
    return A


def solve_fw_lp(f, M, n):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns the optimal g (sum <= 1) or None if infeasible.
    """
    c = -np.ones(n)
    A_ub = build_conv_matrix_fast(f, n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if res.success:
            g = res.x
            # The optimal sum(g) should be <= 1 (since sum(1) = n and 1 is feasible when M >= max(f*g for g=1/n))
            # Actually the optimal is to maximize sum(g), bounded by 1 if M is large enough
            return g
        else:
            return None
    except Exception:
        return None


def fw_iterate(f, M, n, alpha=0.5):
    """One Frank-Wolfe iteration. Returns (new_f, improved, lp_sum)."""
    g = solve_fw_lp(f, M, n)
    if g is None:
        return None, False, 0.0
    g_sum = g.sum()
    if g_sum < 1e-15:
        return None, False, 0.0
    g = g / g_sum  # project to simplex
    if np.max(np.abs(g - f)) < 1e-14:
        return f, False, g_sum
    # Armijo backtracking line search
    f0 = f
    c0 = fast_convolve(f0).max()
    # Direction
    d = g - f0
    # Descent direction for the max objective (using current max as proxy)
    # We want to reduce max_k (f_new * f_new)[k]
    # Use the gradient at the current max as a proxy for directional derivative
    k_star = int(np.argmax(fast_convolve(f0)))
    # grad of c[k_star] at f0
    js = max(0, k_star - n + 1)
    je = min(n - 1, k_star)
    grad = np.zeros(n)
    for j in range(js, je + 1):
        other = k_star - j
        if other != j:
            grad[j] = f0[other]
            grad[other] += f0[j]
        else:
            grad[j] = 2.0 * f0[j]
    # Directional derivative
    dd = grad @ d
    if dd >= 0:
        # Not a descent direction for this particular peak; try smaller step
        alpha_new = alpha * 0.5
        f_new = np.maximum(f0 + alpha_new * d, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            return None, False, g_sum
        f_new = f_new / s
        c_new = fast_convolve(f_new).max()
        improved = c_new < c0 - 1e-14
        return f_new, improved, g_sum
    
    # Armijo backtracking
    c = 1e-4
    rho = 0.5
    alpha_cur = min(1.0, alpha)
    best_f = f0
    best_c = c0
    for _ in range(15):
        f_try = np.maximum(f0 + alpha_cur * d, 0.0)
        s_try = f_try.sum()
        if s_try < 1e-15:
            alpha_cur *= rho
            continue
        f_try = f_try / s_try
        c_try = fast_convolve(f_try).max()
        if c_try <= c0 + c * alpha_cur * dd:
            best_f = f_try
            best_c = c_try
            break
        elif c_try < best_c:
            best_f = f_try
            best_c = c_try
        alpha_cur *= rho
    improved = best_c < c0 - 1e-14
    return best_f, improved, g_sum


def frank_wolfe_optimize(f0, time_limit, m_relax_init=1.0, verbose=False):
    """Frank-Wolfe optimization with adaptive M relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M_current = conv.max()
    best_obj = M_current
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    m_relax = m_relax_init
    step_size = 0.5

    while time.monotonic() - t0 < time_limit:
        it += 1
        M_target = M_current * m_relax

        new_f, improved, lp_sum = fw_iterate(f, M_target, n, alpha=step_size)
        
        if new_f is not None:
            f = new_f
            conv = fast_convolve(f)
            new_M = conv.max()
            
            if new_M < best_obj - 1e-14:
                best_obj = new_M
                best_f = f.copy()
                no_improve = 0
                # Tighten relaxation
                m_relax = max(1.0, m_relax * 0.97)
                if verbose and it % 10 == 0:
                    print(f"    FW iter {it}: obj={2*n*best_obj:.8f}, M={M_current:.6f}, relax={m_relax:.4f}")
            else:
                no_improve += 1
            
            M_current = new_M
        else:
            no_improve += 1
            # LP infeasible or no progress, relax more
            m_relax *= 1.05
            if m_relax > 3.0:
                # Too relaxed, perturb and restart
                rng = np.random.default_rng(it * 7 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M_current = conv.max()
                m_relax = 1.0
                no_improve = 0

        # Adaptive learning rate
        if no_improve > 30:
            step_size = max(0.1, step_size * 0.9)
        if no_improve > 100:
            # Bigger perturbation
            rng = np.random.default_rng(it * 13 + 5)
            noise = 0.005 + 0.005 * min(no_improve / 200.0, 2.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise)
            M_current = fast_convolve(f).max()
            m_relax = 1.0
            no_improve = 0
            step_size = 0.5

        if no_improve > 300:
            break

    return best_f, 2 * n * best_obj


def subgradient_refine(f0, time_limit, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)

        f = f / s

    best_f = f.copy()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Find active set (peaks within 1% of max)
        threshold = M * 0.99
        active = np.where(conv >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])
        
        # Compute averaged subgradient over active set
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                other = k - j
                if other != j:
                    grad[j] += f[other]
                    grad[other] += f[j]
                else:
                    grad[j] += 2.0 * f[j]
        grad /= len(active)
        
        # FW step: find min grad[j] (the best coordinate to move mass to)
        j_best = int(np.argmin(grad))
        
        # Also try the LP approach but only with active constraints
        # For now, use a blended approach: mix FW vertex direction with subgradient
        # Direction: move mass towards j_best
        direction = np.zeros(n)
        direction[j_best] = 1.0
        direction -= f  # to maintain sum = 1
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
            f_try = np.maximum(f + alpha * direction, 0.0)
            s_try = f_try.sum()
            if s_try < 1e-15:
                continue
            f_try = f_try / s_try
            val = fast_convolve(f_try).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha < 1e-8:
            # Try random perturbations
            rng = np.random.default_rng(it * 3 + 1)
            for trial in range(5):
                f_try = project_simplex(f + rng.standard_normal(n) * 0.002)
                val = fast_convolve(f_try).max()
                if val < best_val:
                    best_val = val
                    best_alpha = 1.0
                    direction = f_try - f
                    f_try = f_try / f_try.sum()
        
        if best_alpha > 1e-8 and best_val < M - 1e-14:
            f = np.maximum(f + best_alpha * direction, 0.0)
            f = f / f.sum()
            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_f = f.copy()
                if verbose and it % 50 == 0:
                    print(f"    SubGrad iter {it}: obj={2*n*best_obj:.8f}")

    return best_f, 2 * n * best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, center=0.5, sigma=0.25):
    x = (np.arange(n) - n * center) / (n * sigma)
    a = np.exp(-0.5 * x ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width=0.6, power=4):
    x = (np.arange(n) - n / 2.0) / (n * width / 2.0)
    a = np.maximum(1.0 - np.abs(x) ** power, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_diffset_like(n, rng):
    """Create a difference-set-like initialization with good autocorrelation properties."""
    a = np.zeros(n)
    # Start with a random sparse set
    k = max(3, n // 10)
    positions = rng.choice(n, size=k, replace=False)
    a[positions] = 1.0
    # Smooth it out
    from scipy.ndimage import gaussian_filter1d
    a = gaussian_filter1d(a, sigma=1.5)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def adam_refine(f0, time_limit, tau_init=None, lr=0.003, verbose=False):
    """Adam optimization on log-sum-exp smooth approximation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    if tau_init is None:
        tau = 0.06 * conv.max()
    else:
        tau = tau_init

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t = 0
    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)
        scaled = conv / tau
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        # Gradient: h[j] = sum_l w[j+l] * f[l] (with appropriate indexing)
        # Use FFT
        w_rev = w[::-1]
        total_len = n + len(w) - 1
        fft_size = 1 << (total_len - 1).bit_length()
        A = np.fft.rfft(f, fft_size)
        W = np.fft.rfft(w_rev, fft_size)
        conv_result = np.fft.irfft(A * W, fft_size)
        h = conv_result[n - 1:2 * n - 1][::-1]
        grad = (2.0 / tau) * h

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        tau *= 0.9998
        tau = max(tau, 1e-8)

        if no_improve > 80:
            lr *= 0.8
        if no_improve > 300:
            rng = np.random.default_rng(t * 11 + 3)
            noise = 0.002 + 0.003 * min(no_improve / 500.0, 3.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = 0.003
            tau = 0.04 * fast_convolve(f).max()

    return best_f, 2 * n * best_obj

            break
        try_seq(a_init, f"init_{i}")

    work_seq = best_seq if best_seq is not None else parent
    work_n = len(work_seq)
    print(f"  Best init: n={work_n}, score={best_score:.6f}")
    t_phase1_done = time.monotonic()
    print(f"  Phase 1 took {t_phase1_done - t_phase1:.1f}s")

    # === Phase 2: Adam optimization on best ===
    print(f"\n=== Phase 2: Adam Optimization (n={work_n}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem, 500)
    if t2 > 60:
        print(f"  Time allocated: {t2:.0f}s")
        a_opt, obj = adam_refine(work_seq, time_limit=t2 * 0.6, lr=0.004, verbose=True)
        try_seq(a_opt, "adam_main")

        if time.monotonic() < deadline - 700:
            rem2 = min(deadline - time.monotonic() - 700, 200)
            a_opt2, _ = adam_refine(best_seq, time_limit=rem2,
                                     tau_init=0.03 * fast_convolve(best_seq).max(),
                                     lr=0.002, verbose=False)
            try_seq(a_opt2, "adam_fine")

    # === Phase 3: Frank-Wolfe ===
    print(f"\n=== Phase 3: Frank-Wolfe ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 60:
            t3 = min(rem, 450)
            print(f"  Time allocated: {t3:.0f}s")
            a_opt, obj = frank_wolfe_optimize(best_seq, time_limit=t3 * 0.7, verbose=True)
            try_seq(a_opt, "fw_main")

            if time.monotonic() < deadline - 400:
                rem2 = min(deadline - time.monotonic() - 400, 150)
                a_opt2, _ = frank_wolfe_optimize(best_seq, time_limit=rem2,
                                                  m_relax_init=1.05, verbose=False)
                try_seq(a_opt2, "fw_relaxed")

    # === Phase 4: Subgradient refinement ===
    print(f"\n=== Phase 4: Subgradient Refinement ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 350
        if rem > 30:
            t4 = min(rem, 200)
            print(f"  Time allocated: {t4:.0f}s")
            a_opt, obj = subgradient_refine(best_seq, time_limit=t4, verbose=True)
            try_seq(a_opt, "subgrad")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length Exploration ===")
    if best_seq is not None:
        base = best_seq
        for target_n in [1024, 1280, 640, 1536]:
            rem = deadline - time.monotonic() - 200
            if rem < 40:
                break
            f_interp = interpolate_sequence(base, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")

            # Quick Adam on interpolated
            t5 = min(rem * 0.3, 80)
            if t5 > 20:
                a_opt, _ = adam_refine(f_interp, time_limit=t5, lr=0.003)
                try_seq(a_opt, f"adam_n{target_n}")

            # Quick FW
            rem2 = deadline - time.monotonic() - 150
            if rem2 > 30:
                t5b = min(rem2, 60)
                a_opt2, _ = frank_wolfe_optimize(f_interp, time_limit=t5b)
                try_seq(a_opt2, f"fw_n{target_n}")

    # === Phase 6: Coordinate descent ===
    print(f"\n=== Phase 6: Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t6 = min(rem, 150)
            print(f"  Time allocated: {t6:.0f}s")
            a_opt, obj = coordinate_descent_refine(best_seq, time_limit=t6, verbose=True)
            try_seq(a_opt, "coord_descent")

    # === Phase 7: Final Adam polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 20:
            t7 = min(rem, 80)
            a_opt, _ = adam_refine(best_seq, time_limit=t7,
                                    tau_init=0.015 * fast_convolve(best_seq).max(),
                                    lr=0.001, verbose=False)
            try_seq(a_opt, "final_polish")

    # === Phase 8: Final FW squeeze ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t8 = min(rem, 40)
            a_opt, _ = frank_wolfe_optimize(best_seq, time_limit=t8, verbose=False)
            try_seq(a_opt, "final_fw")

    # === Phase 9: Final coordinate descent ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt, _ = coordinate_descent_refine(best_seq, time_limit=min(rem, 20))
            try_seq(a_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = min(best_score, final_score)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
