import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_of(a):
    """Compute the objective score."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    max_c = fast_convolve(a_norm).max()
    return float(2.0 * n * max_c)

def project_simplex(v):
    """Project v onto the probability simplex (non-negative, sum=1)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def freq_domain_iterate(a0, n_iters, time_limit, target_flat=None):
    """
    Gerchberg-Saxton style iterative method.
    Alternates between:
    - Time domain: project to non-negative, normalize
    - Frequency domain: shape the power spectrum to reduce auto-corr peak
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    
    # Target: we want the auto-correlation to be as flat as possible
    # The ideal would be all equal to 1/(2n-1) for normalized sum=1
    # But that's not achievable. Instead, we iteratively reduce the peak.
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute current auto-correlation
        ac = fast_convolve(a)
        peak = ac.max()
        
        # Compute the "excess" at the peak locations
        # We want to reduce the peak by shaping the spectrum
        
        # Method: In frequency domain, reduce the components that contribute most to the peak
        # The peak of auto-corr is at lag k*, which means:
        # ac[k] = sum_i a[i] * a[i+k] (for k >= 0)
        
        # Alternative: use the fact that ac = IFFT(|FFT(a)|^2)
        # We want to modify |FFT(a)|^2 to reduce the peak of ac
        
        # Simple approach: reduce the magnitude at frequencies where the phase alignment
        # causes constructive interference at the peak lag
        
        # Even simpler: damp the overall magnitude at high frequencies
        # (high freq oscillations help flatten the auto-corr)
        
        fft_size = 2 * n - 1
        A = fft(a, 2 * fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Shape: boost high frequencies (they create oscillations that flatten auto-corr)
        freqs = np.fft.fftfreq(2 * fft_size)
        # Create a shaping function that boosts higher frequencies
        f_norm = np.abs(freqs) / max(np.abs(freqs).max(), 1e-10)
        
        # Apply mild shaping
        shape = 1.0 + 0.05 * f_norm**2
        mag_new = mag * shape
        
        # Reconstruct
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, 2 * fft_size))[:n]
        
        # Project to simplex
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a
        else:
            a_new = a_new / s
        
        sc = score_of(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # If it got worse, try a smaller step
            a_new = 0.5 * a + 0.5 * (np.maximum(np.real(ifft(A_new, 2 * fft_size))[:n], 0))
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
            sc = score_of(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
    
    return best_a, best_score

            contribs = a[idx] * a[peak_k - idx]

                targets = [t for t in targets if t != top_i]

                    a_trial[targets] += diff * np.ones(len(targets)) / len(targets)
                a_trial = np.maximum(a_trial, 0)

                if s > 1e-15:
                    a_trial = a_trial / s
                sc = score_of(a_trial)

                    best_score = sc

            phase = rng.uniform(0, 2 * np.pi)

def simulated_annealing(a0, time_limit, T_init=0.01, T_min=0.0001, cooling=0.999):
    """
    Simulated annealing with structured moves.
    Accepts worse solutions with probability exp(-delta/T).
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_of(a)
    current_score = best_score
    T = T_init
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    step = 0
    
    while time.monotonic() - t0 < time_limit and T > T_min:
        step += 1
        # Choose a random move
        move_type = rng.integers(0, 5)
        a_trial = a.copy()
        
        if move_type == 0:
            # Single element adjustment
            i = rng.integers(n)
            delta = a[i] * rng.uniform(-0.2, 0.2)
            a_trial[i] = max(0, a[i] + delta)
            # Redistribute
            diff = a[i] - a_trial[i]
            if abs(diff) > 1e-12:
                j = rng.integers(n)
                if j != i:
                    a_trial[j] = max(0, a_trial[j] + diff)
        
        elif move_type == 1:
            # Pair swap
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j:
                a_trial[i], a_trial[j] = a[j], a[i]
        
        elif move_type == 2:
            # Neighborhood shuffle
            center = rng.integers(n)
            r = rng.integers(2, min(6, n // 4))
            lo = max(0, center - r)
            hi = min(n, center + r + 1)
            a_trial[lo:hi] = a[lo:hi] * rng.dirichlet(np.ones(hi - lo))
            s_r = a_trial[lo:hi].sum()
            if s_r > 1e-15:
                a_trial[lo:hi] *= (a[lo:hi].sum() / s_r)
        
        elif move_type == 3:
            # Sinusoidal bump
            center = rng.integers(n)
            width = rng.integers(3, min(20, n // 5))
            amp = a.mean() * rng.uniform(-0.1, 0.1)
            mask = (np.arange(n) >= center - width) & (np.arange(n) < center + width)
            a_trial[mask] += amp
        
        else:
            # Global perturbation
            noise = rng.standard_normal(n) * a.mean() * 0.02
            a_trial = a + noise
        
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-15:
            T *= cooling
            continue
        a_trial = a_trial / s
        new_score = score_of(a_trial)
        
        delta = new_score - current_score
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_trial
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        
        T *= cooling
    
    return best_a, best_score

        return obj

        return g

            method='L-BFGS-B',

                     'maxfun': maxiter * 50, 'disp': False}

        a_out = np.exp(z_out - z_out.max())
        a_out = a_out / a_out.sum()

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fictitious play with LP-based best responses (game-theoretic approach)
2. Multi-temperature L-BFGS with soft-max smoothing in log-space
3. Diverse coordinate descent with multiple move types

5. Multi-length exploration and cross-pollination
6. Restart with perturbation to escape local minima

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute the objective score for a normalized non-negative sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    max_c = fast_convolve(a_norm).max()
    return float(2.0 * n * max_c)


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    if '_lp_solver_cache' in globals():
        return _lp_solver_cache
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.GLPK, cp.ECOS, cp.PDLP, cp.SCS, cp.CBC, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                globals()['_lp_solver_cache'] = solver_name
                return solver_name
        except Exception:
            continue
    globals()['_lp_solver_cache'] = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """
    Given fixed a (non-negative, sum=1), find b (non-negative, sum=1)
    minimizing max_k conv(a, b)[k]. Returns (b, M).
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.CBC]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def fictitious_play_optimize(n, time_limit, initial=None, solver=None, verbose=True):
    """
    Run fictitious play (alternating best responses) for the auto-correlation
    minimization game with exponential moving average.
    """
    if solver is None:
        solver = get_lp_solver()
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
        t = np.linspace(0, 2 * np.pi, n)
        a = a + 0.01 * np.cos(t)
        a = np.maximum(a, 0)
        a = a / a.sum()
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    b = a.copy()
    a_avg = a.copy().astype(np.float64)
    b_avg = b.copy().astype(np.float64)
    best_score = float('inf')
    best_a = a.copy()
    iters = 0
    ema_alpha = 0.1  # EMA factor for running average

    while time.monotonic() - t0 < time_limit:
        iters += 1
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        # Exponential moving average (better than uniform for convergence)
        alpha = ema_alpha / (1 + 0.01 * iters)
        a_avg = (1 - alpha) * a_avg + alpha * a
        b_avg = (1 - alpha) * b_avg + alpha * b

        candidates = [a, b, 0.5 * (a + b), a_avg, 0.7 * a_avg + 0.3 * b_avg]

        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"  FP n={n} iter={iters} ({elapsed:.0f}s): best={best_score:.6f}")

def lbgfs_refine(a0, time_limit, T=100, maxiter=5000):
    """L-BFGS refinement in log-space with soft-max smoothing."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-20)
    z0 = np.log(a_safe / a_safe.sum())
    t0 = time.monotonic()

    def func_and_grad(z):
        z_shifted = z - z.max()
        a = np.exp(z_shifted)
        a = a / a.sum()

        b = fast_convolve(a)
        max_b = b.max()
        Tb = T * (b - max_b)
        exp_Tb = np.exp(Tb)
        log_sum_exp = max_b + np.log(exp_Tb.sum()) / T
        obj = (2.0 * n) * log_sum_exp

        w = exp_Tb / exp_Tb.sum()

        # Gradient: grad_a[m] = 4n * sum_j w[j+m] * a[j] (cross-correlation)
        N_fft = 4 * n + 4
        W_p = fft(w, N_fft)
        A_p = fft(a, N_fft)
        cross_corr = np.real(ifft(np.conj(W_p) * A_p, N_fft))[:n]
        grad_a = 4.0 * n * cross_corr

        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)

        return obj, grad_z

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        obj, _ = func_and_grad(z)
        return obj

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = func_and_grad(z)
        return g

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        z_out = res.x
        a_out = np.exp(z_out - z_out.max())
        a_out = a_out / a_out.sum()
    except Exception:
        a_out = a0.copy()

    return np.maximum(a_out, 0)

def simulated_annealing(a0, time_limit, seed=123):
    """Simulated annealing with structured moves."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_of(a)
    current_score = best_score
    T = 0.005
    T_min = 0.00001
    cooling = 0.9995
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    step = 0

    while time.monotonic() - t0 < time_limit and T > T_min:
        step += 1
        move_type = int(rng.integers(0, 6))
        a_trial = a.copy()

        if move_type == 0:
            # Single element nudge
            i = int(rng.integers(n))
            delta = a[i] * rng.uniform(-0.15, 0.15)
            a_trial[i] = max(0, a[i] + delta)
            diff = a[i] - a_trial[i]
            if abs(diff) > 1e-12:
                j = int(rng.integers(n))
                if j != i:
                    a_trial[j] = max(0, a_trial[j] + diff)

        elif move_type == 1:
            # Pair swap
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                a_trial[i], a_trial[j] = a[j], a[i]

        elif move_type == 2:
            # Neighborhood reshuffle
            center = int(rng.integers(n))
            r = int(rng.integers(2, min(6, n // 4)))
            lo = max(0, center - r)
            hi = min(n, center + r + 1)
            region_sum = a[lo:hi].sum()
            if region_sum > 1e-12:
                new_region = rng.dirichlet(np.ones(hi - lo) * 2) * region_sum
                a_trial[lo:hi] = new_region

        elif move_type == 3:
            # Sinusoidal bump
            center = int(rng.integers(n))
            width = int(rng.integers(3, min(15, n // 5)))
            amp = a.mean() * rng.uniform(-0.08, 0.08)
            positions = np.arange(n)
            mask = (positions >= center - width) & (positions < center + width)
            a_trial[mask] += amp

        elif move_type == 4:
            # Pair value blend
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                t = rng.uniform(0.3, 0.7)
                old_i, old_j = a[i], a[j]
                a_trial[i] = t * old_i + (1 - t) * old_j
                a_trial[j] = (1 - t) * old_i + t * old_j

        else:
            # Small global noise
            a_trial = a + rng.standard_normal(n) * a.mean() * 0.01

        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-15:
            T *= cooling
            continue
        a_trial = a_trial / s
        new_score = score_of(a_trial)

        delta = new_score - current_score
        if delta < 0 or rng.random() < np.exp(-abs(delta) / max(T, 1e-10)):
            a = a_trial
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

        T *= cooling
        if step % 2000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    SA step={step} ({elapsed:.0f}s): best={best_score:.6f} T={T:.6f}")

    return best_a, best_score

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_convolve(a)
        max_ac = ac.max()
        mean_ac = ac.mean()
        # Target: reduce the peak-to-mean ratio
        target_ratio = 1.0 - 0.002 * (it % 100)  # Small reduction target

        # Add a sinusoidal component to reduce the peak
        # The peak is at lag k*, so we add a "negative" contribution there
        peak_k = int(np.argmax(ac))
        # Try different frequencies
        for freq in [0.5, 1.0, 1.5, 2.0, 3.0]:
            phase = 2 * np.pi * freq * peak_k / n  # Phase to create dip at peak
            amp = a.mean() * 0.005
            pert = amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
            a_trial = a + pert
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s > 1e-15:
                a_trial = a_trial / s
            sc = score_of(a_trial)
            if sc < best_score - 1e-14:
                a = a_trial
                best_a = a.copy()
                best_score = sc
                break

        # Also try reducing high-magnitude frequency components
        if it % 10 == 0:
            N_fft = 2 * n
            A = fft(a, N_fft)
            mag = np.abs(A)
            phase = np.angle(A)
            # Slightly damp the top 5% magnitude components
            threshold = np.sort(mag)[::-1][max(0, len(mag) // 20)]
            damp_mask = mag > threshold
            mag[damp_mask] *= 0.995
            A_new = mag * np.exp(1j * phase)
            a_trial = np.real(ifft(A_new, N_fft))[:n]
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s > 1e-15:
                a_trial = a_trial / s
            sc = score_of(a_trial)
            if sc < best_score - 1e-14:
                a = a_trial
                best_a = a.copy()
                best_score = sc

    return best_a, best_score


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining LP fictitious play, L-BFGS,
    diverse coordinate descent, simulated annealing, and frequency-domain methods.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Get LP solver
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score

    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = score_of(best_seq)
        print(f"Default start: n=768, score={best_score:.6f}")

    n_parent = len(best_seq)

    # === Phase 1: Extended Fictitious Play (main optimization) ===
    print(f"\n=== Phase 1: Extended FP (best={best_score:.6f}) ===")
    t0_p1 = time.monotonic()
    fp_time = min(500.0, (deadline - time.monotonic()) * 0.35)

    # Run FP at parent length with longer budget
    print(f"  FP at n={n_parent}, budget={fp_time:.0f}s")
    a_fp, s_fp, iters = fictitious_play_optimize(
        n_parent, fp_time, initial=best_seq, solver=solver, verbose=True
    )
    best_seq, best_score = check_and_update(a_fp, f"FP_n{n_parent}_it{iters}", best_seq, best_score)

    # Damped LP refinement
    rem = deadline - time.monotonic() - (deadline - t0_p1) * 0.5
    if rem > 30:
        t_dl = min(rem * 0.6, 150.0)
        print(f"  Damped LP, budget={t_dl:.0f}s")
        a_dl, s_dl = damped_lp_refine(best_seq, t_dl, solver=solver)
        best_seq, best_score = check_and_update(a_dl, "damped_lp", best_seq, best_score)

    # === Phase 2: Multi-temperature L-BFGS ===
    print(f"\n=== Phase 2: L-BFGS multi-T (best={best_score:.6f}) ===")
    for T in [20, 50, 100, 200, 400, 800]:
        rem = deadline - time.monotonic() - 400
        if rem < 15:
            break
        t_alloc = min(50.0, rem * 0.25)
        print(f"  L-BFGS T={T}, budget={t_alloc:.0f}s")
        a_lb = lbgfs_refine(best_seq, time_limit=t_alloc, T=T, maxiter=3000)
        best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T}", best_seq, best_score)

    # === Phase 3: Diverse coordinate descent (long) ===
    print(f"\n=== Phase 3: Coord Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 300
    if rem > 30:
        t_cd = min(150.0, rem * 0.5)
        print(f"  Coord descent, budget={t_cd:.0f}s")
        a_cd, s_cd = diverse_coord_descent(best_seq, time_limit=t_cd, max_steps=15000, seed=42)
        best_seq, best_score = check_and_update(a_cd, "coord_descent", best_seq, best_score)

    # === Phase 4: Simulated annealing ===
    print(f"\n=== Phase 4: Simulated Annealing (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 30:
        t_sa = min(120.0, rem * 0.5)
        print(f"  SA, budget={t_sa:.0f}s")
        a_sa, s_sa = simulated_annealing(best_seq, time_limit=t_sa, seed=123)

    # === Phase 5: Post-SA refinement ===
    print(f"\n=== Phase 5: Post-SA refinement (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 150
    if rem > 20:
        # L-BFGS after SA
        t_lb = min(60.0, rem * 0.3)
        a_lb = lbgfs_refine(best_seq, time_limit=t_lb, T=100, maxiter=2000)
        best_seq, best_score = check_and_update(a_lb, "post_sa_lbgfs", best_seq, best_score)

    print(f"\n=== Phase 6: Nearby lengths (best={best_score:.6f}) ===")

        a_alt2 = lbgfs_refine(a_alt1, time_limit=t_alloc * 0.5, T=100, maxiter=1000)

    print(f"\n=== Phase 7: Restart (best={best_score:.6f}) ===")

        a_pert = best_seq + rng.standard_normal(n_best) * noise_level
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        t_re = min(rem * 0.4, 50.0)
        a_r1, _, _ = fictitious_play_optimize(n_best, t_re * 0.5, initial=a_pert, solver=solver, verbose=False)
        a_r2 = lbgfs_refine(a_r1, time_limit=t_re * 0.5, T=100, maxiter=1000)

    print(f"\n=== Phase 9: Final Polish (best={best_score:.6f}) ===")

    if rem > 10:
        t_final = min(30.0, rem * 0.6)
        a_final = lbgfs_refine(best_seq, time_limit=t_final, T=150, maxiter=3000)
        best_seq, best_score = check_and_update(a_final, "final_lbgfs", best_seq, best_score)

    rem = deadline - time.monotonic() - 3
    if rem > 8:
        a_final2, _ = diverse_coord_descent(best_seq, time_limit=rem * 0.7, max_steps=3000, seed=200)
        best_seq, best_score = check_and_update(a_final2, "final_coord", best_seq, best_score)

    # Fallback
    if best_seq is None:

        best_score = float(evaluate_sequence(best_seq.tolist()))

    return [float(x) for x in best_seq]
