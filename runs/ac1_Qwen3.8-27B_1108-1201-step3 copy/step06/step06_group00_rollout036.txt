def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

import numpy as np
from scipy.fft import rfft, irfft, ifft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def evaluate_local(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def compute_subgradient_direct(a, b, top_k, temperature):
    """Compute subgradient using temperature-weighted top-k peaks."""
    n = len(a)
    size = len(b)
    scaled = b / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(size)
        w[b.argmax()] = 1.0
    else:
        w /= w_sum
    if top_k is not None and top_k < size:
        top_idx = np.argpartition(b, -top_k)[-top_k:]
        w_sel = w[top_idx]
        s = w_sel.sum()
        if s < 1e-30:
            w = np.zeros(size)
            w[b.argmax()] = 1.0
        else:
            w_full = np.zeros(size)
            w_full[top_idx] = w_sel / s
            w = w_full
    g = np.zeros(n)
    for k in range(size):
        if w[k] < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        m_vals = np.arange(lo, hi + 1)
        g[m_vals] += 2.0 * w[k] * a[k - m_vals]
    return g


def subgradient_optimize(a0, time_limit, lr_init=0.003, momentum=0.92,
                         temp_init=0.01, temp_min=0.001, temp_max=0.3,
                         cooling_rate=0.9997, kick_interval=2000,
                         kick_size=0.003, n_top=50, seed=42):
    """Subgradient with Nesterov momentum and adaptive temperature."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    a = a / S
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    cur_temp = temp_init
    no_improve = 0
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()
        g = compute_subgradient_direct(a, b, n_top, cur_temp)
        v = momentum * v - cur_lr * g
        a_new = project_simplex(a + v)
        b_new = fast_convolve(a_new)
        obj_new = b_new.max()
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        a = a_new
        it_count += 1
        # Adaptive temperature
        cur_temp *= cooling_rate
        if cur_temp < temp_min:
            cur_temp = temp_min
        if no_improve > kick_interval:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.7, lr_init * 0.1)
            cur_temp = min(cur_temp * 2.0, temp_max)
            no_improve = 0
        elif no_improve > kick_interval // 2:
            cur_lr *= 0.95
    return best_a, best_obj


def spectral_balance(a0, n_iters=50, alpha=0.3, seed=42):
    """Spectral balancing: iteratively flatten the power spectrum."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    a = a / S
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    for it in range(n_iters):
        A = rfft(a, fft_size)
        mag = np.abs(A)
        phase = A / np.maximum(mag, 1e-15)
        target = mag.mean()
        mag_new = alpha * mag + (1.0 - alpha) * target
        mag_new[0] = mag[0]
        A_new = mag_new * phase
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
        a = a_new
    return best_a, best_obj


def softmax_obj_grad(z, T, n):
    """Soft-max smoothed objective and gradient in log-space."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        a = np.ones(n) / n
    else:
        a = exp_z / s
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    # Gradient w.r.t. a: 2n * sum_k w[k] * 2*a[k-i] = 4n * cross_corr(w, a)
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        m_vals = np.arange(lo, hi + 1)
        grad_a[m_vals] += 4.0 * n * w[k] * a[k - m_vals]
    # Chain rule: grad_z = a * (grad_a - a * <grad_a, a>)
    dot = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * dot)
    return obj, grad_z


def lbgfs_refine(a0, T_list, time_limit, maxiter_per_T=2000, seed=42):
    """L-BFGS with deterministic annealing over temperature list."""
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    a = a / S
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    for T in T_list:
        if time.monotonic() - t0 > time_limit:
            break
        z0 = np.log(np.maximum(a, 1e-300))
        t_phase = time.monotonic()
        phase_limit = min(time_limit - (t_phase - t0), 60.0)

        def func(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            return softmax_obj_grad(z, T, n)[0]

        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            return softmax_obj_grad(z, T, n)[1]

        try:
            res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                           options={'maxiter': maxiter_per_T, 'ftol': 1e-14,
                                    'gtol': 1e-10, 'maxfun': maxiter_per_T * 10,
                                    'disp': False})
            a_new = np.exp(res.x - res.x.max())
            a_new = a_new / a_new.sum()
            obj = fast_convolve(a_new).max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new
        except Exception:
            pass
    return best_a, best_obj

def coordinate_polish(a0, time_limit, n_passes_max=10):
    """Coordinate descent: try scaling each element."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    multipliers = np.array([0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0])
    for pass_num in range(n_passes_max):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_obj = best_obj
            best_j_val = old_val
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.9999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        if not improved:
            break
    return best_a, best_obj


def multiplicative_update(a0, time_limit, lr_init=0.01, n_top=30, seed=42):
    """Multiplicative (entropic) update optimization."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    cur_temp = 0.01
    it = 0
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        g = compute_subgradient_direct(a, b, n_top, cur_temp)
        # Multiplicative update
        log_update = -cur_lr * g
        log_update -= log_update.max()  # numerical stability
        a_new = a * np.exp(log_update)
        s = a_new.sum()
        if s < 1e-30:
            a_new = a
        else:
            a_new = a_new / s
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
        a = a_new
        it += 1
        cur_lr *= 0.9999
        cur_temp *= 0.9995
        if it % 2000 == 0:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            cur_lr = lr_init * 0.5
    return best_a, best_obj


def make_spectral_init(n, seed=42):
    """Spectral initialization with random phases for flat auto-correlation."""
    rng = np.random.default_rng(seed)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    phases[0] = 0.0
    if fft_size % 2 == 0:
        phases[fft_size // 2] = 0.0
    P = np.exp(1j * phases)
    P_full = np.zeros(fft_size, dtype=complex)
    P_full[:fft_size // 2 + 1] = P
    P_full[fft_size // 2 + 1:] = np.conj(P[fft_size // 2 - 1:0:-1])
    a_full = np.real(ifft(P_full))
    a = a_full[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = a_full[n:n + n]
        a = np.maximum(a, 0.0)
        s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def make_gaussian_init(n, sigma_frac=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma_frac) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_triangular_init(n):
    x = np.linspace(0, 1, n)
    a = 1.0 - 2.0 * np.abs(x - 0.5)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_ramp_init(n):
    a = np.linspace(0.01, 1.0, n)
    return a / a.sum()


def make_exponential_init(n):
    x = np.linspace(0, 1, n)
    a = np.exp(-3.0 * x)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_uniform_init(n):
    return np.ones(n) / n


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses subgradient, spectral balancing, L-BFGS, line search, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1152) / 1152
        print("No parent, using uniform n=1152")

    n_parent = len(parent)

    # === Phase 1: Diverse initializations with short optimization ===
    print(f"\n=== Phase 1: Diverse initializations ===")
    n_base = n_parent

    initializations = {}
    initializations["parent"] = parent.copy()
    initializations["spectral"] = make_spectral_init(n_base, seed)
    initializations["gauss_0.25"] = make_gaussian_init(n_base, 0.25)
    initializations["gauss_0.35"] = make_gaussian_init(n_base, 0.35)
    initializations["gauss_0.45"] = make_gaussian_init(n_base, 0.45)
    initializations["triangular"] = make_triangular_init(n_base)
    initializations["ramp"] = make_ramp_init(n_base)
    initializations["exponential"] = make_exponential_init(n_base)
    initializations["uniform"] = make_uniform_init(n_base)
    initializations["random_gauss"] = np.maximum(rng.standard_normal(n_base), 0.0)
    initializations["random_gauss"] /= initializations["random_gauss"].sum()
    initializations["random_exp"] = rng.exponential(1.0, n_base)
    initializations["random_exp"] /= initializations["random_exp"].sum()

    # Spectral balance of parent
    try:
        sb_a, _ = spectral_balance(parent, n_iters=20, alpha=0.5, seed=seed)
        initializations["spectral_bal"] = sb_a
    except Exception:
        pass

    for name, init_a in initializations.items():
        rem = deadline - time.monotonic() - 600
        if rem < 15:
            break
        t_alloc = min(rem * 0.08, 30.0)
        if t_alloc < 5:
            continue
        print(f"  Init {name} (t={t_alloc:.0f}s)")
        a_opt, _ = subgradient_optimize(
            init_a, time_limit=t_alloc,
            lr_init=0.01, momentum=0.9,
            temp_init=0.05, temp_min=0.005, temp_max=0.2,
            cooling_rate=0.998, kick_interval=1000, kick_size=0.005,
            n_top=max(20, n_base // 20), seed=seed + hash(name) % 10000
        )
        try_seq(a_opt, f"init_{name}")

    # === Phase 2: Extended subgradient on best ===
    print(f"\n=== Phase 2: Extended subgradient (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_alloc = min(rem, 250.0)
            print(f"  n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = subgradient_optimize(
                best_seq, time_limit=t_alloc,
                lr_init=0.005, momentum=0.93,
                temp_init=0.02, temp_min=0.001, temp_max=0.3,
                cooling_rate=0.9995, kick_interval=3000, kick_size=0.002,
                n_top=max(30, n_best // 10), seed=seed + 111
            )
            try_seq(a_opt, "ext_sg")

    # === Phase 3: Spectral balancing on best ===
    print(f"\n=== Phase 3: Spectral balancing (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 10:
            print(f"  Spectral balance...")
            a_sb, _ = spectral_balance(best_seq, n_iters=100, alpha=0.3, seed=seed + 222)
            try_seq(a_sb, "spectral_bal")
            # Also try different alpha
            a_sb2, _ = spectral_balance(best_seq, n_iters=50, alpha=0.1, seed=seed + 333)
            try_seq(a_sb2, "spectral_bal_slow")

    # === Phase 4: L-BFGS with deterministic annealing ===
    print(f"\n=== Phase 4: L-BFGS annealing (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 250
        if rem > 20:
            T_list = [200, 100, 50, 20, 10, 5, 2]
            t_alloc = min(rem * 0.8, 120.0)
            print(f"  L-BFGS annealing, n={n_best}, T_list={T_list}, t={t_alloc:.0f}s")
            a_lb, _ = lbgfs_refine(best_seq, T_list=T_list, time_limit=t_alloc,
                                   maxiter_per_T=1500, seed=seed + 444)
            try_seq(a_lb, "lbgfs_anneal")

    # === Phase 5: Multiplicative update optimization ===
    print(f"\n=== Phase 5: Multiplicative updates (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 15:
            t_alloc = min(rem * 0.5, 80.0)
            print(f"  Multiplicative, t={t_alloc:.0f}s")
            a_mu, _ = multiplicative_update(best_seq, time_limit=t_alloc,
                                            lr_init=0.005, n_top=30, seed=seed + 555)
            try_seq(a_mu, "multiplicative")

    # === Phase 6: Line-search subgradient ===
    print(f"\n=== Phase 6: Line-search subgradient (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 10:
            t_alloc = min(rem * 0.6, 60.0)
            print(f"  Linesearch, t={t_alloc:.0f}s")
            a_ls, _ = linesearch_subgradient(best_seq, time_limit=t_alloc,
                                             top_k=1, n_lr=8, seed=seed + 666)
            try_seq(a_ls, "linesearch")

    # === Phase 7: More L-BFGS at lower temperatures ===
    print(f"\n=== Phase 7: L-BFGS fine-tune (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            T_list = [10, 5, 3, 2, 1]
            t_alloc = min(rem * 0.7, 60.0)
            print(f"  L-BFGS fine T={T_list}, t={t_alloc:.0f}s")
            a_lb2, _ = lbgfs_refine(best_seq, T_list=T_list, time_limit=t_alloc,
                                    maxiter_per_T=1000, seed=seed + 777)
            try_seq(a_lb2, "lbgfs_fine")

    # === Phase 8: Coordinate descent ===
    print(f"\n=== Phase 8: Coordinate descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 10:
            t_cd = min(rem * 0.8, 40.0)
            print(f"  CD polish, t={t_cd:.0f}s")
            a_cd, _ = coordinate_polish(best_seq, time_limit=t_cd, n_passes_max=5)
            try_seq(a_cd, "cd_polish")

    # === Phase 9: Final micro-optimization ===
    print(f"\n=== Phase 9: Final micro-opt (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            # Very targeted subgradient
            t_alloc = min(rem * 0.4, 20.0)
            a_micro, _ = subgradient_optimize(
                best_seq, time_limit=t_alloc,
                lr_init=0.0005, momentum=0.97,
                temp_init=0.001, temp_min=0.0003, temp_max=0.01,
                cooling_rate=0.9999, kick_interval=10000, kick_size=0.0005,
                n_top=max(10, n_best // 5), seed=seed + 888
            )
            try_seq(a_micro, "micro_sg")

            # Linesearch final
            rem = deadline - time.monotonic() - 10
            if rem > 8:
                t_ls = min(rem * 0.5, 15.0)
                a_ls2, _ = linesearch_subgradient(best_seq, time_limit=t_ls,
                                                  top_k=1, n_lr=10, seed=seed + 999)
                try_seq(a_ls2, "linesearch_final")

            # Another spectral balance pass
            rem = deadline - time.monotonic() - 5
            if rem > 5:
                a_sb3, _ = spectral_balance(best_seq, n_iters=30, alpha=0.15, seed=seed + 1000)
                try_seq(a_sb3, "spectral_final")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'=' * 50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'=' * 50}")

    return [float(x) for x in best_seq]
