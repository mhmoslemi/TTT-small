def fast_convolve(a):

    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])

    return np.maximum(v - theta, 0.0)


def compute_lse_gradient(c, a, tau):
    """
    Compute gradient of tau*log(sum(exp(c[m]/tau))) with respect to a.
    Returns (smooth_max_value, gradient).
    """
    c_scaled = c / tau
    c_max = c_scaled.max()
    exps = np.exp(c_scaled - c_max)
    w = exps / exps.sum()
    obj = tau * (c_max + np.log(exps.sum()))
    # gradient[j] = sum_m w[m] * 2*a[m-j] for m-j in [0,n-1]
    # = 2 * correlate(w, a, 'valid')[j]
    g = 2.0 * np.correlate(w, a, mode='valid')
    return obj, g


def make_ramp(n, power=1.0):
    """Linear/power ramp: (1 - j/(n-1))^power."""
    j = np.arange(n, dtype=np.float64)
    a = (1.0 - j / (n - 1)) ** power
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_half_cosine(n):
    """Half cosine: cos(pi*j/(2*(n-1)))."""
    j = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * j / (2.0 * (n - 1)))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_gaussian(n, sigma_frac=0.3, center_frac=0.4):
    """Gaussian with center not at midpoint (asymmetric)."""
    j = np.arange(n, dtype=np.float64)
    center = n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * ((j - center) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_exponential(n, rate=0.5):
    """Exponential decay."""
    j = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6, center_frac=0.45):
    """Flat-top bell shape, slightly off-center."""
    j = np.arange(n, dtype=np.float64)
    center = n * center_frac
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(j - center) / half_width, 0, 1.0)
    a = (1.0 - t ** 4)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def lse_gradient_descent(a0, time_limit, tau_init=0.005, tau_min=0.0001,
                         lr_init=0.05, verbose=False, label=""):
    """
    LSE-based gradient descent on the simplex with decreasing temperature.
    Minimizes max(conv(a,a)) subject to a>=0, sum(a)=1.
    """
    a = np.maximum(a0.astype(np.float64), 1e-15).copy()
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    tau = tau_init
    lr = lr_init
    no_improve = 0
    it = 0
    best_lr = lr

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(a)
        M = c.max()

        if M < best_obj - 1e-14:
            best_obj = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Compute LSE gradient
        _, g = compute_lse_gradient(c, a, tau)

        # Backtracking line search
        improved = False
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            trial_lr = lr * factor
            if trial_lr < 1e-12:
                break
            candidate = a - trial_lr * g
            # Handle potential negative values from large steps
            candidate = np.maximum(candidate, 0.0)
            candidate = project_simplex(candidate)
            c_new = fast_convolve(candidate)
            M_new = c_new.max()
            if M_new < M - 1e-15:
                a = candidate
                lr = trial_lr
                improved = True
                if M_new < best_obj - 1e-14:
                    best_obj = M_new
                    best_a = a.copy()
                    no_improve = 0
                break
        if not improved:
            # Take a tiny step in the best direction
            trial_lr = lr * 0.01
            if trial_lr > 1e-14:
                candidate = a - trial_lr * g
                candidate = np.maximum(candidate, 0.0)
                candidate = project_simplex(candidate)
                c_new = fast_convolve(candidate)
                M_new = c_new.max()
                if M_new < M:
                    a = candidate
                    lr = trial_lr
                else:
                    lr *= 0.3

        # Anneal temperature (decrease for sharper approximation)
        # Use a schedule that decreases gradually
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_init * (tau_min / tau_init) ** (progress ** 0.5)
        tau = max(tau, tau_min)

        # Restart if stuck
        if no_improve > 200:
            rng = np.random.default_rng(it * 7919 + 13)
            noise_scale = 0.005 + 0.01 * min(no_improve / 1000.0, 1.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            lr = lr_init * 0.5
            tau = max(tau * 2, tau_init * 0.3)
            no_improve = 0

        if verbose and it % 500 == 0:
            print(f"    {label} iter {it}: obj={best_obj:.10f}, tau={tau:.6f}, "
                  f"lr={lr:.6f}, no_improve={no_improve}")

        if lr < 1e-12 and no_improve > 500:
            break

    return best_a, best_obj


def lse_with_adaptive_peaks(a0, time_limit, verbose=False, label=""):
    """
    LSE gradient descent with adaptive peak selection:
    Only include peaks above a threshold in the LSE, reducing noise from small peaks.
    Also tries multiple temperature values and picks the best.
    """
    a = np.maximum(a0.astype(np.float64), 1e-15).copy()
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    tau = 0.003
    lr = 0.03
    no_improve = 0
    it = 0
    threshold_frac = 0.9  # Only consider peaks above 90% of max

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(a)
        M = c.max()

        if M < best_obj - 1e-14:
            best_obj = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Thresholded LSE: only include significant peaks
        threshold = M * threshold_frac
        mask = c >= threshold
        if mask.sum() < 3:
            # Ensure at least 3 peaks
            idx = np.argpartition(c, -3)[-3:]
            mask[idx] = True

        c_masked = np.where(mask, c, -np.inf).astype(np.float64)
        # For LSE, replace -inf with a very negative number
        c_masked = np.where(mask, c, M - 10.0 * tau)

        c_scaled = c_masked / tau
        c_max = c_scaled.max()
        exps = np.exp(c_scaled - c_max)
        w = exps / exps.sum()

        g = 2.0 * np.correlate(w, a, mode='valid')

        # Line search
        improved = False
        for factor in [1.0, 0.5, 0.25, 0.1]:
            trial_lr = lr * factor
            if trial_lr < 1e-12:
                break
            candidate = a - trial_lr * g
            candidate = np.maximum(candidate, 0.0)
            candidate = project_simplex(candidate)
            M_new = fast_convolve(candidate).max()
            if M_new < M - 1e-15:
                a = candidate
                lr = trial_lr
                improved = True
                if M_new < best_obj - 1e-14:
                    best_obj = M_new
                    best_a = a.copy()
                    no_improve = 0
                break
        if not improved:
            lr *= 0.5

        # Anneal
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = 0.003 * (0.0003 / 0.003) ** (progress ** 0.7)
        threshold_frac = 0.9 + 0.05 * progress  # Tighten threshold over time

        # Restart
        if no_improve > 150:
            rng = np.random.default_rng(it * 104729 + 3)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            lr = 0.03
            tau = 0.003
            no_improve = 0

        if verbose and it % 500 == 0:
            print(f"    {label} iter {it}: obj={best_obj:.10f}, tau={tau:.6f}, "
                  f"thr={threshold_frac:.3f}, no_imp={no_improve}")

    return best_a, best_obj


def frank_wolfe_lse(a0, time_limit, tau_init=0.003, verbose=False, label=""):
    """
    Frank-Wolfe (conditional gradient) method with LSE objective.
    Different geometry than projected gradient: moves toward a single vertex.
    """
    a = np.maximum(a0.astype(np.float64), 1e-15).copy()
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    tau = tau_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(a)
        M = c.max()

        if M < best_obj - 1e-14:
            best_obj = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        _, g = compute_lse_gradient(c, a, tau)

        # FW step: find vertex that minimizes <g, e_j> = g[j]
        j_min = np.argmin(g)
        e_j = np.zeros(n)
        e_j[j_min] = 1.0

        # Line search for step size
        best_t = 0.0
        best_obj_t = M
        for t in [2.0 / (it + 2), 0.5, 0.3, 0.2, 0.1, 0.05]:
            candidate = (1.0 - t) * a + t * e_j
            c_new = fast_convolve(candidate)
            M_new = c_new.max()
            if M_new < best_obj_t:
                best_obj_t = M_new
                best_t = t

        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * e_j
            a = np.maximum(a, 0.0)
            a = a / a.sum()

        # Anneal
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_init * (0.0003 / tau_init) ** (progress ** 0.5)

        if no_improve > 200:
            rng = np.random.default_rng(it * 31 + 7)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            tau = tau_init
            no_improve = 0

        if verbose and it % 1000 == 0:
            print(f"    {label} FW iter {it}: obj={best_obj:.10f}, tau={tau:.6f}")

    return best_a, best_obj


def coordinate_ascent_peaks(a0, time_limit, verbose=False, label=""):
    """
    Targeted coordinate moves: find the peak, identify contributing pairs,
    and make small adjustments to reduce the peak.
    Uses a "push down the peak" strategy.
    """
    a = np.maximum(a0.astype(np.float64), 1e-15).copy()
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(a)
        M = c.max()

        if M < best_obj - 1e-14:
            best_obj = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 50:
            break

        # Find top peaks
        K = min(5, len(c))
        peak_indices = np.argsort(c)[-K:]

        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = a[idx] * a[k - idx]
            # Find the pair with maximum product
            j_max = idx[np.argmax(products)]

            # Try reducing a[j_max] and redistributing to minimize impact
            old_val = a[j_max]
            if old_val < 1e-10:
                continue

            best_trial_obj = M
            best_trial = None

            for frac in [0.7, 0.8, 0.9, 1.1, 1.2, 1.4]:
                new_val = old_val * frac
                if new_val < 0:
                    continue
                trial = a.copy()
                trial[j_max] = new_val
                diff = old_val - new_val

                # Redistribute to positions that DON'T contribute to this peak
                non_contrib = np.ones(n, dtype=bool)
                non_contrib[js:je + 1] = False
                if non_contrib.sum() > 0:
                    nc = a[non_contrib]
                    nc_sum = nc.sum()
                    if nc_sum > 1e-15:
                        trial[non_contrib] += diff * nc / nc_sum
                        trial = np.maximum(trial, 0.0)
                        ss = trial.sum()
                        if ss > 1e-15:
                            trial /= ss
                            obj = fast_convolve(trial).max()
                            if obj < best_trial_obj - 1e-14:
                                best_trial_obj = obj
                                best_trial = trial

            if best_trial is not None:
                a = best_trial
                if best_trial_obj < best_obj - 1e-14:
                    best_obj = best_trial_obj
                    best_a = a.copy()
                    no_improve = 0
                    if verbose and it % 5 == 0:
                        print(f"    {label} CA iter {it}: obj={best_obj:.10f}")

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, maintaining sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization:
    - LSE gradient descent with decreasing temperature (main)
    - Thresholded LSE (for focused peak reduction)
    - Frank-Wolfe (different geometry)
    - Coordinate ascent (local polish)
    Multiple initializations and sequence lengths, NO symmetrization.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Quick initialization exploration ===
    print("=== Phase 1: Initialization Exploration ===")
    init_pool = []

    lengths_to_try = [768, 1000, 1200, 1536]

    if parent is not None:
        init_pool.append((parent, "parent_n768"))
        # Interpolate parent to other lengths
        for n_try in [1000, 1200, 1536]:
            init_pool.append((interpolate_sequence(parent, n_try), f"parent_interp_n{n_try}"))

    for n_try in lengths_to_try:
        init_pool.append((make_ramp(n_try, 1.0), f"ramp1_n{n_try}"))
        init_pool.append((make_ramp(n_try, 2.0), f"ramp2_n{n_try}"))
        init_pool.append((make_ramp(n_try, 0.5), f"ramp05_n{n_try}"))
        init_pool.append((make_half_cosine(n_try), f"cos_n{n_try}"))
        init_pool.append((make_gaussian(n_try, 0.25, 0.35), f"gauss_n{n_try}"))
        init_pool.append((make_exponential(n_try, 1.0), f"exp1_n{n_try}"))
        init_pool.append((make_exponential(n_try, 2.0), f"exp2_n{n_try}"))
        init_pool.append((make_bell(n_try, 0.5, 0.4), f"bell_n{n_try}"))

    # Quick evaluate all
    quick_best = None
    quick_best_obj = float('inf')
    for a_init, label in init_pool:
        s = a_init.sum()
        if s < 1e-10:
            continue
        a_norm = a_init / s
        score = try_seq(a_norm, f"init_{label}")
        obj = 2 * len(a_norm) * fast_convolve(a_norm).max()
        if obj < quick_best_obj:
            quick_best_obj = obj
            quick_best = (a_norm, label, len(a_norm))

    if quick_best is None:
        quick_best = (make_ramp(768, 1.0), "default", 768)

    print(f"  Best init: {quick_best[1]} with obj={quick_best_obj:.6f}")

    # === Phase 2: Main LSE Gradient Descent on best init ===
    print("\n=== Phase 2: Main LSE GD ===")
    remaining = deadline - time.monotonic()
    t_md = remaining * 0.35
    print(f"  Time: {t_md:.0f}s")
    a_opt, obj = lse_gradient_descent(quick_best[0], time_limit=t_md,
                                      tau_init=0.005, tau_min=0.0002,
                                      lr_init=0.04, verbose=True, label="main")
    try_seq(a_opt, "LSE_main")

    # === Phase 3: Thresholded LSE on best ===
    print("\n=== Phase 3: Thresholded LSE ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_t = remaining * 0.20
        print(f"  Time: {t_t:.0f}s")
        a_opt, obj = lse_with_adaptive_peaks(best_seq, time_limit=t_t,
                                             verbose=True, label="thresh")
        try_seq(a_opt, "thresh_LSE")

    # === Phase 4: LSE on second-best init (different shape) ===
    print("\n=== Phase 4: LSE on diverse inits ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        # Try a few different inits with LSE
        diverse_inits = []
        for a_init, label in init_pool:
            if label != quick_best[1]:
                diverse_inits.append((a_init, label))
                if len(diverse_inits) >= 4:
                    break

        t_per = remaining * 0.15 / len(diverse_inits) if diverse_inits else 0
        for a_init, label in diverse_inits:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            t_alloc = min(t_per, remaining - 20)
            if t_alloc < 15:
                break
            print(f"  Trying {label} with {t_alloc:.0f}s")
            a_opt, obj = lse_gradient_descent(a_init, time_limit=t_alloc,
                                              tau_init=0.004, tau_min=0.0003,
                                              lr_init=0.03, verbose=False,
                                              label=label)
            try_seq(a_opt, f"LSE_{label}")

    # === Phase 5: Frank-Wolfe on best ===
    print("\n=== Phase 5: Frank-Wolfe ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_fw = remaining * 0.10
        print(f"  Time: {t_fw:.0f}s")
        a_opt, obj = frank_wolfe_lse(best_seq, time_limit=t_fw,
                                     tau_init=0.003, verbose=True, label="FW")
        try_seq(a_opt, "FW")

    # === Phase 6: Coordinate Ascent polish ===
    print("\n=== Phase 6: Coordinate Ascent ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_ca = remaining * 0.10
        print(f"  Time: {t_ca:.0f}s")
        a_opt, obj = coordinate_ascent_peaks(best_seq, time_limit=t_ca,
                                             verbose=True, label="CA")
        try_seq(a_opt, "CA")

    # === Phase 7: Final LSE polish with very low temperature ===
    print("\n=== Phase 7: Final Polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            t_polish = min(remaining - 15, 120)
            print(f"  Polish time: {t_polish:.0f}s")
            a_opt, obj = lse_gradient_descent(best_seq, time_limit=t_polish,
                                              tau_init=0.001, tau_min=0.00005,
                                              lr_init=0.02, verbose=True,
                                              label="polish")
            try_seq(a_opt, "polish_LSE")

            # Also try with slightly different params
            remaining = deadline - time.monotonic()
            if remaining > 30:
                t2 = min(remaining - 15, 60)
                a_opt, obj = lse_with_adaptive_peaks(best_seq, time_limit=t2,
                                                     verbose=False, label="polish2")
                try_seq(a_opt, "polish_thresh")

    # === Phase 8: Try larger sequence from best ===
    print("\n=== Phase 8: Larger Sequence ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            # Interpolate to larger size and optimize
            for n_large in [2000, 2500]:
                remaining = deadline - time.monotonic()
                if remaining < 40:
                    break
                a_large = interpolate_sequence(best_seq, n_large)
                t_alloc = min(remaining - 20, 80)
                if t_alloc < 20:
                    break
                print(f"  Trying n={n_large} with {t_alloc:.0f}s")
                a_opt, obj = lse_gradient_descent(a_large, time_limit=t_alloc,
                                                  tau_init=0.003, tau_min=0.0002,
                                                  lr_init=0.02, verbose=False,
                                                  label=f"large_n{n_large}")
                try_seq(a_opt, f"large_n{n_large}")

    # Fallback
    if best_seq is None:
        best_seq = make_ramp(768, 1.0)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
