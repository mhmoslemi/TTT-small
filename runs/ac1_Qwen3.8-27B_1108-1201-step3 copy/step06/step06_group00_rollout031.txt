```python
def lp_step(f, M_max=None):
    """
    Given f (probability vector), find g >= 0, sum(g)=1 that minimizes max(f*g).
    Returns g.
    """

    # c = [0, ..., 0, 1] (minimize t)
    
    # Constraints: (f*g)[k] - t <= 0 for each k
    # (f*g)[k] = sum_i f[k-i] * g_i (with f_j = 0 outside [0,n-1])

    # For row k: sum_i A[k,i] * g_i - t <= 0

        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based best-response dynamics.

The key idea: at each iteration, solve an LP that finds the best "response"
sequence g (minimizing max cross-correlation with current f), then blend
towards it. This is a form of fictitious play for the auto-correlation minimization game.

Combined with subgradient warm-up, coordinate descent polishing, and
multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_local(a):
    """Compute objective 2n*max(conv)/(sum)^2 locally."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def build_lp_matrices(f):
    """Build LP constraint matrices for the best-response problem."""
    n = len(f)
    # A[k,i] = f[k-i] if 0 <= k-i <= n-1, else 0
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = f
    return A


def lp_best_response(f, time_per_lp=2.0):
    """
    Find g >= 0, sum(g)=1 minimizing max_k (f*g)[k].
    Returns (g, t_opt) where t_opt = min max_k (f*g)[k].
    """
    n = len(f)
    t_start = time.monotonic()
    
    A = build_lp_matrices(f)
    
    # Variables: [g_0, ..., g_{n-1}, t]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.hstack([np.ones(n), [1.0]])
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    elapsed = time.monotonic() - t_start
    if res.success:
        g = res.x[:n]
        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum > 1e-10:
            g /= g_sum
        return g, res.x[n], elapsed
    else:
        return f.copy(), float('inf'), elapsed


def subgradient_optimize(a0, time_limit, lr_init=0.005, momentum=0.9,
                         temp_init=0.0005, temp_growth=1.02,
                         kick_interval=2000, kick_size=0.003,
                         n_top=None, seed=42):
    """Projected subgradient with Nesterov momentum."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    a = a / S
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    cur_temp = temp_init
    no_improve = 0
    it_count = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()
        
        scaled = b / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
        else:
            w /= w_sum
        
        if n_top is not None and n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx]
            w_sel /= w_sel.sum()
            w_full = np.zeros(len(b))
            w_full[top_idx] = w_sel
            w = w_full
        
        # Compute subgradient via FFT cross-correlation
        # g[i] = sum_k w[k] * a[k-i] (valid range)
        # This is the cross-correlation of w and a
        fft_size = 1 << (len(w) + n - 1 - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        g = irfft(np.conj(W) * A, fft_size)[:n]
        
        v = momentum * v - cur_lr * g
        a_new = project_simplex(a + v)
        
        b_new = fast_convolve(a_new)
        obj_new = b_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        it_count += 1
        
        if no_improve > kick_interval:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.7, lr_init * 0.1)
            no_improve = 0
        elif no_improve > kick_interval // 2:
            cur_lr *= 0.95
        
        cur_temp *= temp_growth
        if cur_temp > 0.5:
            cur_temp = 0.5
    
    return best_a, best_obj


def coordinate_polish(a0, time_limit):
    """Coordinate descent polish."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    multipliers = np.array([0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0])
    
    improved_global = True
    while improved_global and time.monotonic() - t0 < time_limit:
        improved_global = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
```

**Alternative: Multiplicative weight / mirror descent**

The problem min_g max_k (f*g)[k] subject to g>=0, sum(g)=1 can be solved approximately by mirror descent:

Initialize g uniformly.
For each iteration:
  - Compute (f*g)[k] for all k
  - Find the "hottest" lag k* (where (f*g)[k*] is maximum)
  - Update: g[i] *= exp(-eta * A[k*, i]) where A[k*, i] = f[k*-i]
  - Renormalize: g /= sum(g)

This is a multiplicative weight update that increases the "cost" of positions that contribute to the peak lag.

This is much faster than LP (just one FFT per iteration) and might work well in practice.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based best-response dynamics
with constraint generation, combined with subgradient warm-up, mirror descent
fine-tuning, and coordinate descent polishing.

The key innovation is the LP-based approach: at each iteration, solve the LP
that finds the best response sequence g (minimizing peak cross-correlation
with current f), then blend towards g. This is a principled descent direction
that exploits the bilinear structure of the cross-correlation.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross(f, g):
    n = len(f)
    m = len(g)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_local(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def lp_best_response(f, initial_active=None, max_cg_iters=30, max_constr=300):
    """
    LP best response via constraint generation.
    Finds g >= 0, sum(g)=1 minimizing max_k (f*g)[k].
    """
    n = len(f)
    conv_ff = fast_convolve(f)
    
    if initial_active is None:
        top_k = min(max_constr, 2 * n - 1)
        initial_active = set(np.argpartition(conv_ff, -top_k)[-top_k:].tolist())
    
    S = set(initial_active)
    g = f.copy()
    t_opt = conv_ff.max()
    
    for cg_it in range(max_cg_iters):
        active = sorted(S)
        m = len(active)
        A_sub = np.zeros((m, n))
        for row, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A_sub[row, lo:hi+1] = f[k - np.arange(lo, hi+1)]
        
        c = np.zeros(n + 1)
        c[n] = 1.0
        
        A_ub = np.hstack([A_sub, -np.ones((m, 1))])
        b_ub = np.zeros(m)
        A_eq = np.hstack([np.ones(n), [1.0]])
        b_eq = np.array([1.0])
        bounds = [(0, None)] * (n + 1)
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            break
        
        if not res.success:
            break
        
        g_new = res.x[:n].copy()
        g_new = np.maximum(g_new, 0)
        gs = g_new.sum()
        if gs < 1e-15:
            break
        g_new /= gs
        t_new = res.x[n]
        
        # Check violations
        cross = fast_cross(f, g_new)
        violations = cross - t_new
        violated_idx = np.where(violations > 1e-9)[0]
        
        g = g_new
        t_opt = min(t_opt, t_new)
        
        if len(violated_idx) == 0:
            break
        
        # Add most violated
        n_add = min(20, len(violated_idx))
        most_v = violated_idx[np.argsort(violations[violated_idx])[-n_add:]]
        S.update(most_v.tolist())
        S = set(list(S)[-max_constr:])  # Keep at most max_constr
        # Actually, we should keep the most important ones
        # Let me use a different strategy: keep all, but if too many, keep top by violation
    
    return g, t_opt


def subgradient_optimize(a0, time_limit, lr_init=0.005, momentum=0.9,
                         temp_init=0.0005, temp_growth=1.02,
                         kick_interval=2000, kick_size=0.003,
                         n_top=None, seed=42):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    cur_temp = temp_init
    no_improve = 0
    it_count = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        scaled = b / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
        else:
            w /= w_sum
        
        if n_top is not None and n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx]
            w_sel /= w_sel.sum()
            w_full = np.zeros(len(b))
            w_full[top_idx] = w_sel
            w = w_full
        
        fft_size = 1 << (len(w) + n - 1 - 1).bit_length()
        W = rfft(w, fft_size)
        A_ff = rfft(a, fft_size)
        grad = irfft(np.conj(W) * A_ff, fft_size)[:n]
        
        v = momentum * v - cur_lr * grad
        a_new = project_simplex(a + v)
        
        b_new = fast_convolve(a_new)
        obj_new = b_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        it_count += 1
        
        if no_improve > kick_interval:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.7, lr_init * 0.1)
            no_improve = 0
        elif no_improve > kick_interval // 2:
            cur_lr *= 0.95
        
        cur_temp *= temp_growth
        if cur_temp > 0.5:
            cur_temp = 0.5
    
    return best_a, best_obj


def mirror_descent_refine(f, time_limit, eta_init=0.1, eta_decay=0.999, seed=42):
    """
    Mirror descent (multiplicative weight) on the full problem.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(f)
    a = f.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    eta = eta_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()
        
        # Use top peaks
        top_k = max(5, n // 10)
        top_idx = np.argpartition(b, -top_k)[-top_k:]
        top_vals = b[top_idx]
        top_vals = top_vals / top_vals.max()
        w = np.zeros(len(b))
        w[top_idx] = top_vals
        w /= w.sum()
        
        # Subgradient: grad[i] = sum_k w[k] * a[k-i]
        fft_size = 1 << (len(w) + n - 1 - 1).bit_length()
        W = rfft(w, fft_size)
        A_ff = rfft(a, fft_size)
        grad = irfft(np.conj(W) * A_ff, fft_size)[:n]
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        obj_new = fast_convolve(a_new).max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                # Random restart perturbation
                a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
                no_improve = 0
        
        eta *= eta_decay
    
    return best_a, best_obj


def coordinate_polish(a0, time_limit):
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    multipliers = np.array([0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0])
    
    improved_global = True
    while improved_global and time.monotonic() - t0 < time_limit:
        improved_global = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                new_arr = a.copy()
                new_arr[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                new_arr[mask] *= scale
                a = new_arr / new_arr.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_global = True
    return best_a, best_obj


def lp_main_loop(f, time_limit, initial_step=0.3):

        if elapsed_so_far > time_limit * 0.95:

            if new_obj < best_obj - 1e-14:

            step *= 0.7
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based best-response dynamics
with constraint generation, subgradient warm-up, mirror descent refinement,
and coordinate descent polishing.

Key innovation: At each iteration, solve an LP to find the "best response"
sequence g (minimizing peak cross-correlation with current f), then blend
toward g. Constraint generation keeps the LP tractable.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross(f, g):
    n = len(f)
    size = n + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_local(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def lp_best_response(f, max_cg_iters=20, max_constr=250):
    n = len(f)
    conv_ff = fast_convolve(f)
    top_k = min(max_constr, 2 * n - 1)
    initial_active = set(np.argpartition(conv_ff, -top_k)[-top_k:].tolist())
    S_active = set(initial_active)
    g = f.copy()
    t_opt = conv_ff.max()
    for cg_it in range(max_cg_iters):
        active = sorted(S_active)
        m = len(active)
        A_sub = np.zeros((m, n))
        for row, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A_sub[row, lo:hi + 1] = f[k - np.arange(lo, hi + 1)]
        c = np.zeros(n + 1)
        c[n] = 1.0
        A_ub = np.hstack([A_sub, -np.ones((m, 1))])
        b_ub = np.zeros(m)
        A_eq = np.hstack([np.ones(n), [1.0]])
        b_eq = np.array([1.0])
        bounds = [(0, None)] * (n + 1)
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            break
        if not res.success:
            break
        g_new = res.x[:n].copy()
        g_new = np.maximum(g_new, 0)
        gs = g_new.sum()
        if gs < 1e-15:
            break
        g_new /= gs
        t_new = res.x[n]
        cross = fast_cross(f, g_new)
        violations = cross - t_new
        violated_idx = np.where(violations > 1e-9)[0]
        g = g_new
        t_opt = min(t_opt, t_new)
        if len(violated_idx) == 0:
            break
        n_add = min(15, len(violated_idx))
        most_v = violated_idx[np.argsort(violations[violated_idx])[-n_add:]]
        S_active.update(most_v.tolist())
        if len(S_active) > max_constr:
            cross_vals = cross
            keep = np.argsort(cross_vals)[-max_constr:]
            S_active = set(keep.tolist())
    return g, t_opt


def subgradient_optimize(a0, time_limit, lr_init=0.005, momentum=0.9,
                         temp_init=0.0005, temp_growth=1.02,
                         kick_interval=2000, kick_size=0.003,
                         n_top=None, seed=42):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    cur_temp = temp_init
    no_improve = 0
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        scaled = b / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
        else:
            w /= w_sum
        if n_top is not None and n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx]
            w_sel /= w_sel.sum()
            w_full = np.zeros(len(b))
            w_full[top_idx] = w_sel
            w = w_full
        fft_size = 1 << (len(w) + n - 1 - 1).bit_length()
        W = rfft(w, fft_size)
        A_ff = rfft(a, fft_size)
        grad = irfft(np.conj(W) * A_ff, fft_size)[:n]
        v = momentum * v - cur_lr * grad
        a_new = project_simplex(a + v)
        b_new = fast_convolve(a_new)
        obj_new = b_new.max()
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        a = a_new
        if no_improve > kick_interval:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.7, lr_init * 0.1)
            no_improve = 0
        elif no_improve > kick_interval // 2:
            cur_lr *= 0.95
        cur_temp *= temp_growth
        if cur_temp > 0.5:
            cur_temp = 0.5
    return best_a, best_obj


def mirror_descent_refine(f, time_limit, eta_init=0.05, eta_decay=0.9995, seed=42):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(f)
    a = f.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    eta = eta_init
    no_improve = 0
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        top_k = max(10, n // 8)
        top_idx = np.argpartition(b, -top_k)[-top_k:]
        top_vals = b[top_idx]
        top_vals = top_vals / (top_vals.max() + 1e-30)
        w = np.zeros(len(b))
        w[top_idx] = top_vals
        w /= w.sum()
        fft_size = 1 << (len(w) + n - 1 - 1).bit_length()
        W = rfft(w, fft_size)
        A_ff = rfft(a, fft_size)
        grad = irfft(np.conj(W) * A_ff, fft_size)[:n]
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        obj_new = fast_convolve(a_new).max()
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                no_improve = 0
        eta *= eta_decay
    return best_a, best_obj


def coordinate_polish(a0, time_limit):
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    multipliers = np.array([0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0])
    improved_global = True
    while improved_global and time.monotonic() - t0 < time_limit:
        improved_global = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_obj = best_obj
            best_j_val = old_val
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                new_arr = a.copy()
                new_arr[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                new_arr[mask] *= scale
                a = new_arr / new_arr.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_global = True
    return best_a, best_obj


def lp_main_loop(f, time_limit, initial_step=0.25, max_no_improve=80):
    t0 = time.monotonic()
    n = len(f)
    f = f.copy().astype(np.float64)
    f = f / f.sum()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    step = initial_step
    no_improve = 0
    it = 0
    rng = np.random.default_rng(12345)
    print(f"  LP loop: n={n}, initial obj={best_obj:.10f}, score={2*n*best_obj:.6f}")
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit * 0.97:
            break
        g, t_opt = lp_best_response(f)
        if time.monotonic() - t0 > time_limit * 0.99:
            break
        accepted = False
        for s in [step, step * 0.5, step * 0.2, 0.05, 0.01, 0.005]:
            f_new = (1 - s) * f + s * g
            f_new = project_simplex(f_new)
            new_obj = fast_convolve(f_new).max()
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_f = f_new.copy()
                f = f_new
                accepted = True
                no_improve = 0
                break
        it += 1
        if it % 10 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"  LP it={it}: score={score:.6f}, obj={best_obj:.8f}, "
                  f"step={step:.4f}, t_opt={t_opt:.6f}, t={elapsed:.0f}s")
        if not accepted:
            no_improve += 1
            step *= 0.6
            if no_improve > max_no_improve:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                step = max(0.1, initial_step * 0.5)
                no_improve = 0
        else:
            step = min(step * 1.1, 0.5)
    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    if parent is None:
        parent = np.ones(1152) / 1152
        print("No parent, using uniform n=1152")
    n_parent = len(parent)

    # Phase 1: Subgradient warm-up
    print(f"\n=== Phase 1: Subgradient warm-up ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 10:
        a_sg, _ = subgradient_optimize(parent, time_limit=t_alloc,
                                       lr_init=0.003, momentum=0.92,
                                       temp_init=0.0003, temp_growth=1.02,
                                       kick_interval=1000, kick_size=0.003,
                                       n_top=max(20, n_parent // 10), seed=seed)
        try_seq(a_sg, "sg_warmup")

    # Phase 2: LP-based optimization (MAIN)
    print(f"\n=== Phase 2: LP optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_alloc = min(rem * 0.45, 650.0)
        if t_alloc > 30:
            print(f"  Allocating {t_alloc:.0f}s")
            a_lp, _ = lp_main_loop(best_seq, time_limit=t_alloc, initial_step=0.25)
            try_seq(a_lp, "lp_main")

    # Phase 3: Mirror descent refinement
    print(f"\n=== Phase 3: Mirror descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_alloc = min(rem * 0.15, 180.0)
        if t_alloc > 15:
            a_md, _ = mirror_descent_refine(best_seq, time_limit=t_alloc,
                                            eta_init=0.03, eta_decay=0.9998, seed=seed + 1)
            try_seq(a_md, "mirror")

    # Phase 4: LP refinement (second round)
    print(f"\n=== Phase 4: LP refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_alloc = min(rem * 0.2, 250.0)
        if t_alloc > 20:
            print(f"  Allocating {t_alloc:.0f}s")
            a_lp2, _ = lp_main_loop(best_seq, time_limit=t_alloc, initial_step=0.1)
            try_seq(a_lp2, "lp_refine")

    # Phase 5: Coordinate descent
    print(f"\n=== Phase 5: CD polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_alloc = min(rem * 0.12, 100.0)
        if t_alloc > 10:
            a_cd, _ = coordinate_polish(best_seq, time_limit=t_alloc)
            try_seq(a_cd, "cd")

    # Phase 6: Final micro subgradient
    print(f"\n=== Phase 6: Final micro (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 15:
            t_alloc = min(rem * 0.4, 60.0)
            a_fin, _ = subgradient_optimize(best_seq, time_limit=t_alloc,
                                            lr_init=0.0005, momentum=0.95,
                                            temp_init=0.0001, temp_growth=1.005,
                                            kick_interval=3000, kick_size=0.001,
                                            n_top=max(20, len(best_seq) // 5),
                                            seed=seed + 99999)
            try_seq(a_fin, "final_sg")

    # Phase 7: Length exploration
    print(f"\n=== Phase 7: Length exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for factor in [1.15, 0.85, 1.3]:
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            new_n = int(n_best * factor)
            new_n = max(256, min(4096, new_n))
            if new_n == n_best:
                continue
            old_x = np.linspace(0, 1, n_best)
            new_x = np.linspace(0, 1, new_n)
            f_interp = np.interp(new_x, old_x, best_seq)
            f_interp = np.maximum(f_interp, 0)
            f_interp = f_interp / f_interp.sum()
            t_alloc = min(rem * 0.3, 50.0)
            if t_alloc < 8:
                continue
            print(f"  n={new_n}, t={t_alloc:.0f}s")
            a_new, _ = subgradient_optimize(f_interp, time_limit=t_alloc,
                                            lr_init=0.001, momentum=0.9,
                                            temp_init=0.0005, temp_growth=1.02,
                                            kick_interval=800, kick_size=0.002,
                                            n_top=max(10, new_n // 15),
                                            seed=seed + new_n)
            try_seq(a_new, f"len_{new_n}")

    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")
    return [float(x) for x in best_seq]
```
