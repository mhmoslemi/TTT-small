```python
import time
import cvxpy as cp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def cross_convolve(a, g):
    """Linear cross-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    G = np.fft.rfft(g, fft_size)
    return np.fft.irfft(A * G, fft_size)[:size]

def lp_find_direction(a, M):
    """Solve LP to find direction g maximizing sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    g = cp.Variable(n)
    
    # Build constraint: cross_convolve(a, g) <= M
    # (a*g)_k = sum_i a[k-i] * g[i] for valid i
    # This is A @ g where A[k,i] = a[k-i] if valid
    # Instead of building the full matrix, use the FFT-based check
    # But for cvxpy, we need explicit constraints
    
    # Build the matrix A (2n-1) x n
    A = np.zeros((2*n-1, n))
    for i in range(n):
        # g[i] contributes to (a*g)_k for k = i, i+1, ..., i+n-1
        # The coefficient is a[k-i]
        for k in range(i, min(i+n, 2*n-1)):
            A[k, i] = a[k-i]
    
    constraints = [g >= 0, A @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCS, verbose=False)
    except:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except:
            return None
    
    if g.value is None:
        return None
    
    g_val = np.maximum(np.array(g.value).flatten(), 0)
    if g_val.sum() < 1e-10:
        return None
    return g_val
```

- Use the LP direction as a "hint" for the gradient method

Actually, the LP approach is already quite powerful on its own. Let me focus on making it work well.

**Key improvement: Adaptive M**
Instead of using M = max(a*a) each time, we can use a slightly relaxed M:
- M_relaxed = M * (1 + epsilon) for small epsilon
- This gives more freedom in the LP, potentially finding better directions
- But risks increasing the peak

Or: use M = max(a*a) but with a slightly tighter bound:
- M_tight = M * (1 - delta) for small delta
- This forces the LP to find a direction that definitely reduces the peak
- But might make the LP infeasible

I think using M = max(a*a) is the safest and should work well.

- Maximize sum(g) - lambda * ||g - uniform||^2 (to prefer "flat" directions)
- Or: maximize sum(g) subject to g being close to uniform

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding 
with cross-convolution constraints, combined with accelerated mirror descent
and fine coordinate descent for final polish. The LP approach finds optimal
descent directions that guarantee non-increasing peak convolution, providing
a more structured optimization than pure gradient methods.
"""

import numpy as np
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(a, g):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(g)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    G = np.fft.rfft(g, fft_size)
    return np.fft.irfft(A * G, fft_size)[:size]


def build_cross_conv_matrix(a):
    """Build sparse-ish matrix A such that A @ g = cross_convolve(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for i in range(n):
        k_start = i
        k_end = min(i + n, m)
        A[k_start:k_end, i] = a[:k_end - i]
    return A


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, peak_indices, weights):
    """Compute weighted subgradient of max_k (a*a)_k."""
    n = len(a)
    g = np.zeros(n)
    for idx in range(len(peak_indices)):
        k = int(peak_indices[idx])
        w_k = weights[idx]
        if w_k < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        g[js:je + 1] += 2.0 * w_k * a[k - np.arange(js, je + 1)]
    return g


def lp_refine(a, time_limit, verbose=True, label="LP"):
    """
    LP-based refinement: iteratively find direction g via LP, then update.
    LP: max sum(g) s.t. cross_convolve(a, g) <= M, g >= 0
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    # Pre-build initial matrix
    A = build_cross_conv_matrix(a)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        # Solve LP
        g_var = cp.Variable(n)
        constraints = [g_var >= 0, A @ g_var <= M]
        objective = cp.Maximize(cp.sum(g_var))
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False, max_iters=10000)
        except:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except:
                if verbose:
                    print(f"  {label}: LP solve failed at iter {it}")
                no_improve += 1
                if no_improve > 20:
                    break
                continue
        
        if g_var.value is None:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        
        g_val = np.maximum(np.array(g_var.value).flatten(), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-10:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        
        # Rescale g to have same sum as a
        g_val = g_val / g_sum
        
        # Line search for step size t
        best_t = 0.0
        best_new_obj = best_obj + 1e-10
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            trial = (1 - t) * a + t * g_val
            trial = np.maximum(trial, 0.0)
            trial_sum = trial.sum()
            if trial_sum < 1e-10:
                continue
            trial = trial / trial_sum
            trial_conv = fast_convolve(trial)
            trial_obj = trial_conv.max()
            if trial_obj < best_new_obj:
                best_new_obj = trial_obj
                best_t = t
            if trial_obj < M * 0.999:
                break
        
        if best_t > 0 and best_new_obj < M - 1e-14:
            a = (1 - best_t) * a + best_t * g_val
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            
            if best_new_obj < best_obj - 1e-14:
                best_obj = best_new_obj
                best_a = a.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"    {label} iter {it}: obj={best_obj:.10f}, t={best_t}, elapsed={elapsed:.0f}s")
            else:
                no_improve += 1
        else:
            no_improve += 1
            if verbose and it % 100 == 0:
                print(f"    {label} iter {it}: no improvement, M={M:.10f}, elapsed={time.monotonic()-t0:.0f}s")
        
        # Rebuild matrix since a changed
        if best_t > 0:
            A = build_cross_conv_matrix(a)
        
        # Check for convergence
        if best_obj < 1/(2*n) * 1.01:
            break
        
        if no_improve > 50:
            # Try symmetrization
            a_sym = 0.5 * (a + a[::-1])
            a_sym = np.maximum(a_sym, 0.0)
            a_sym = a_sym / a_sym.sum()
            sym_obj = fast_convolve(a_sym).max()
            if sym_obj < best_obj:
                best_obj = sym_obj
                best_a = a_sym.copy()
                a = a_sym.copy()
                A = build_cross_conv_matrix(a)
                no_improve = 0
            else:
                # Add noise and restart
                rng = np.random.default_rng(it * 7919)
                a = np.maximum(a + rng.standard_normal(n) * 0.001, 1e-15)
                a = a / a.sum()
                A = build_cross_conv_matrix(a)
                no_improve = 0
        
        if no_improve > 200:
            break
    
    return best_a, best_obj


def mirror_descent(a0, time_limit, top_k=30, temp_init=0.0005, lr_init=0.01,
                   lr_decay=0.9999, verbose=False, label="MD"):
    """
    Fast multiplicative weight mirror descent on the simplex.
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        K = min(top_k, len(conv))
        if K < len(conv):
            peak_indices = np.argpartition(conv, -K)[-K:]
        else:
            peak_indices = np.arange(len(conv))
        
        pv = conv[peak_indices]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peak_indices))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        g = compute_subgradient(a, peak_indices, w)
        
        # Backtracking line search
        best_lr = lr * 0.1
        best_trial_obj = M * 1.01
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            if trial_lr < 1e-10:
                continue
            log_update = -trial_lr * g
            log_update = np.clip(log_update, -50, 50)
            a_trial = a * np.exp(log_update)
            a_trial = np.maximum(a_trial, 1e-15)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial = a_trial / ss
            trial_conv = fast_convolve(a_trial)
            trial_obj = trial_conv.max()
            if trial_obj <= best_trial_obj:
                best_lr = trial_lr
                best_trial_obj = trial_obj
                if trial_obj < M * 0.999:
                    break
        
        log_update = -best_lr * g
        log_update = np.clip(log_update, -50, 50)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        conv = fast_convolve(a)
        new_obj = conv.max()
        
        if new_obj < best_obj - 1e-13:
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    {label} MD iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
        
        temp = min(temp * 1.002, 0.05)
        if no_improve > 100:
            lr *= 0.5
            no_improve = 0
        lr = max(lr * lr_decay, 1e-8)
        
        if no_improve > 500:
            rng = np.random.default_rng(it * 7919 + 13)
            noise_scale = 0.001 + 0.002 * min(no_improve / 5000, 1.0)
            a = np.maximum(best_a + rng.standard_normal(n) * noise_scale, 1e-15)
            a = a / a.sum()
            lr = lr_init * 0.3
            no_improve = 0
        
        if it % 500 == 0:
            a_sym = 0.5 * (a + a[::-1])
            a_sym = np.maximum(a_sym, 1e-15)
            a_sym = a_sym / a_sym.sum()
            sym_obj = fast_convolve(a_sym).max()
            if sym_obj < best_obj:
                best_obj = sym_obj
                best_a = a_sym.copy()
                a = a_sym.copy()
        
        if lr < 1e-9:
            break
    
    return best_a, best_obj


def fine_coordinate_descent(a0, time_limit, verbose=False, label="CD"):
    """Fine coordinate descent: adjust one coordinate at a time."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0.0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial = trial / ss
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_obj


def symmetrize(a):
    """Symmetrize a sequence."""
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym /= ss
    return a_sym


def propose_candidate(seed=42, budget_s=1800, **kwargs):

    except Exception:

    # === Phase 4: Coordinate descent polish ===
    print(f"\n=== Phase 4: Coordinate Descent ===")
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding
(scipy HiGHS solver), accelerated mirror descent with adaptive temperature,
and multi-pass coordinate descent. The LP approach provides guaranteed
non-increasing peak via cross-convolution constraints, while mirror descent
and coordinate descent provide fine-grained local optimization.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_cross_conv_matrix(a):
    """Build matrix A such that A @ g = cross_convolve(a, g, mode='full')."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for i in range(n):
        k_end = min(i + n, m)
        A[i:k_end, i] = a[:k_end - i]
    return A


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, peak_indices, weights):
    """Compute weighted subgradient of max_k (a*a)_k."""
    n = len(a)
    g = np.zeros(n)
    for idx in range(len(peak_indices)):
        k = int(peak_indices[idx])
        w_k = weights[idx]
        if w_k < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        g[js:je + 1] += 2.0 * w_k * a[k - np.arange(js, je + 1)]
    return g
