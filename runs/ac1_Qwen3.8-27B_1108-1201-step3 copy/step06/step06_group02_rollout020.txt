def soft_max_descent(f0, time_limit, tau_init=0.001, tau_decay=0.999):
    f = f0.copy()
    f = f / f.sum()
    n = len(f)
    tau = tau_init
    lr = 0.01
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        # Softmax weights
        scaled = conv / tau
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        # Gradient
        grad = np.zeros(n)
        for k in range(2*n-1):
            if w[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            # d(conv_k)/da_i for i in idx: a_{k-i}
            partners = k - idx
            # Only count valid partners
            valid = (partners >= 0) & (partners < n)
            grad[idx[valid]] += 2.0 * w[k] * f[partners[valid]]
        
        # Gradient step with projection
        f_new = f - lr * grad
        f_new = np.maximum(f_new, 0)
        f_new = f_new / f_new.sum()
        
        # Check improvement
        new_obj = fast_convolve(f_new).max()
        if new_obj < best_obj:
            best_obj = new_obj
            best_f = f_new.copy()
            f = f_new
            lr = min(lr * 1.01, 0.1)
        else:
            lr *= 0.95
            if lr < 1e-8:
                break
        
        tau *= tau_decay
        tau = max(tau, 1e-6)
    
    return best_f, best_obj

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_normalized(a):
    """Compute 2*n*max(conv)/sum^2 for any a (handles normalization)."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    an = a / s
    conv = fast_convolve(an)
    return 2.0 * len(an) * conv.max()


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

    max_val = peak_vals.max()
    # Weight: equal weight for all peaks at max, small weight for near-max

    weights = np.zeros(len(peaks))

        if peak_vals[i] >= max_val - tol:
            weights[i] = 1.0
    if weights.sum() < 1e-15:
        weights[np.argmax(peak_vals)] = 1.0
    weights /= weights.sum()

    grad = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        partners = k - idx
        valid = (partners >= 0) & (partners < n)
        grad[idx[valid]] += 2.0 * weights[i] * f[partners[valid]]
    return grad


def soft_max_descent(f0, time_limit, verbose=False):
    """Smoothed subgradient descent using log-sum-exp approximation of max."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()

    # Adaptive tau: start small, decrease over time
    tau_init = best_obj * 0.05
    tau = tau_init
    lr = 0.003
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1.0:
            break

        conv = fast_convolve(f)
        M = conv.max()

        # Softmax weights
        scaled = conv / tau
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= w_sum

        # Only use top contributing weights for efficiency
        threshold = w.max() * 0.01
        active_mask = w > threshold
        if active_mask.sum() == 0:
            active_mask[int(np.argmax(conv))] = True
        w_active = np.where(active_mask, w, 0.0)
        w_active /= w_active.sum()

        # Gradient
        grad = np.zeros(n)
        active_k = np.where(active_mask)[0]
        for k in active_k:
            k = int(k)
            wk = w_active[k]
            if wk < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            partners = k - idx
            valid = (partners >= 0) & (partners < n)
            grad[idx[valid]] += 2.0 * wk * f[partners[valid]]

        # Line search with backtracking
        best_alpha = 0
        best_val = M
        alphas = [0.001, 0.003, 0.01, 0.03, 0.1, 0.3, 0.5]
        for alpha in alphas:
            f_trial = f - alpha * grad
            f_trial = np.maximum(f_trial, 0.0)
            ts = f_trial.sum()
            if ts < 1e-15:
                continue
            f_trial = f_trial / ts
            val = fast_convolve(f_trial).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            f = f - best_alpha * grad
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            new_obj = fast_convolve(f).max()
            if new_obj < best_obj - 1e-15:
                best_obj = new_obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.02, 0.1)
            else:
                no_improve += 1
                lr *= 0.9
        else:
            no_improve += 1
            lr *= 0.8

        # Decay tau
        tau *= 0.9995
        tau = max(tau, best_obj * 0.001)

        if verbose and it % 50 == 0 and no_improve == 0:
            print(f"    SoftMax iter {it}: obj={best_obj:.10f}, tau={tau:.6e}, lr={lr:.6e}")

        if no_improve > 300:
            # Restart with perturbation
            rng = np.random.default_rng(it * 17 + 3)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.001, 0.0)
            f = f / f.sum()
            lr = 0.003
            tau = tau_init * 0.5
            no_improve = 0

        if lr < 1e-10:
            break

    return best_f, best_obj


def pairwise_descent(f0, time_limit, verbose=False):
    """Pairwise coordinate descent: redistribute mass between pairs to reduce peak."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if no_improve > 200:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2.0:
            break

        conv = fast_convolve(f)
        M = conv.max()

        # Find top peaks
        top_k = min(10, 2 * n - 1)
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]

        improved = False
        for pk in peak_indices:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            contrib_idx = np.arange(js, je + 1)
            if len(contrib_idx) == 0:
                continue

            products = f[contrib_idx] * f[pk - contrib_idx]
            # Find top contributing pairs
            top_pairs = np.argsort(products)[-5:]

            for ti in top_pairs:
                i = contrib_idx[ti]
                j = pk - ti

                if i == j:
                    # Single coordinate: try moving mass away
                    best_trial = None
                    best_trial_obj = M
                    for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                        diff = f[i] * frac
                        if diff < 1e-15:
                            continue
                        trial = f.copy()
                        trial[i] -= diff
                        # Spread to random non-contributing positions
                        non_contrib = np.ones(n, dtype=bool)
                        non_contrib[contrib_idx] = False
                        if non_contrib.sum() < 2:
                            continue
                        nc_idx = np.where(non_contrib)[0]
                        # Spread proportionally to existing values
                        nc_vals = trial[nc_idx]
                        nc_sum = nc_vals.sum()
                        if nc_sum < 1e-15:
                            continue
                        trial[nc_idx] += diff * nc_vals / nc_sum
                        trial = np.maximum(trial, 0.0)
                        trial = trial / trial.sum()
                        obj = fast_convolve(trial).max()
                        if obj < best_trial_obj - 1e-15:
                            best_trial_obj = obj
                            best_trial = trial
                    if best_trial is not None:
                        f = best_trial
                        if best_trial_obj < best_obj - 1e-14:
                            best_obj = best_trial_obj
                            best_f = f.copy()
                            no_improve = 0
                            improved = True
                        break

                else:
                    # Pair (i, j): try to reduce the product f[i]*f[j]
                    # Either decrease one and increase the other, or decrease both
                    best_trial = None
                    best_trial_obj = M
                    for delta in [0.1, 0.2, 0.3, 0.5]:
                        # Transfer from larger to smaller
                        if f[i] >= f[j]:
                            src, dst = i, j
                        else:
                            src, dst = j, i
                        transfer = f[src] * delta
                        if transfer < 1e-15:
                            continue
                        trial = f.copy()
                        trial[src] -= transfer
                        trial[dst] += transfer
                        trial = np.maximum(trial, 0.0)
                        trial = trial / trial.sum()
                        obj = fast_convolve(trial).max()
                        if obj < best_trial_obj - 1e-15:
                            best_trial_obj = obj
                            best_trial = trial
                    if best_trial is not None:
                        f = best_trial
                        if best_trial_obj < best_obj - 1e-14:
                            best_obj = best_trial_obj
                            best_f = f.copy()
                            no_improve = 0
                            improved = True
                        break

        if not improved:
            no_improve += 1

        if verbose and it % 100 == 0:
            print(f"    Pairwise iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-Frank-Wolfe with multi-level M relaxation and careful line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_levels = [1.0, 1.01, 1.05, 1.1, 1.2, 1.5, 2.0, 3.0]
    m_idx = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3.0:
            break

        g = None
        for level_try in range(m_idx, len(m_levels)):
            M_try = conv.max() * m_levels[level_try]
            g = solve_lp_direction(f, M_try, n)
            if g is not None and g.sum() > 1e-12:
                m_idx = level_try
                break
            m_idx = level_try + 1

        if g is None or g.sum() < 1e-12:
            rng = np.random.default_rng(it * 13 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(f)
            m_idx = 0
            no_improve += 1
            if no_improve > 100:
                break
            continue

        g = g / g.sum()
        diff = np.max(np.abs(g - f))
        if diff < 1e-12:
            no_improve += 1
            m_idx = min(m_idx + 1, len(m_levels) - 1)
            if no_improve > 80:
                break
            continue
        no_improve = 0

        # Fine line search
        best_alpha = 0.0
        best_val = conv.max()
        for alpha in [0.01, 0.03, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Golden section refinement
        if best_alpha > 0 and best_alpha < 1.0:
            grid = [0.01, 0.03, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
            idx = grid.index(best_alpha)
            lo = grid[max(0, idx - 1)]
            hi = grid[min(len(grid) - 1, idx + 1)]
            for _ in range(15):
                a1 = hi - 0.618 * (hi - lo)
                a2 = lo + 0.618 * (hi - lo)
                ft1 = np.maximum(f + a1 * (g - f), 0.0)
                ft2 = np.maximum(f + a2 * (g - f), 0.0)
                ft1 = ft1 / ft1.sum()
                ft2 = ft2 / ft2.sum()
                v1 = fast_convolve(ft1).max()
                v2 = fast_convolve(ft2).max()
                if v1 < v2:
                    hi = a2
                else:
                    lo = a1
                mid = (lo + hi) / 2.0
                ftm = np.maximum(f + mid * (g - f), 0.0)
                ftm = ftm / ftm.sum()
                vm = fast_convolve(ftm).max()
                if vm < best_val:
                    best_val = vm
                    best_alpha = mid

        if best_alpha < 1e-12:
            no_improve += 1
            m_idx = min(m_idx + 1, len(m_levels) - 1)
            continue

        f = np.maximum(f + best_alpha * (g - f), 0.0)
        f = f / f.sum()
        conv = fast_convolve(f)
        M = conv.max()
        m_idx = 0

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            if verbose and it % 20 == 0:
                print(f"    LP-FW iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-like initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    """Bell-shaped with flat top."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_spectral_flat(n, seed=42):
    """Initialize from a near-flat power spectrum."""
    rng = np.random.default_rng(seed)
    # Create a power spectrum that's close to flat but with some randomness
    freqs = np.fft.rfftfreq(n)
    # Start with flat, add small perturbations
    power = np.ones(n // 2 + 1)
    power += rng.standard_normal(n // 2 + 1) * 0.1
    power = np.maximum(power, 0.01)
    # Compute sequence as inverse FFT of sqrt(power) with random phases
    phases = rng.standard_normal(n // 2 + 1)
    spectrum = np.sqrt(power) * np.exp(1j * phases)
    # Reconstruct
    a = np.fft.irfft(spectrum, n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def make_multi_cosine(n, n_freqs=4, max_amp=0.25, seed=0):
    """Multi-frequency cosine-modulated sequence."""
    rng = np.random.default_rng(seed)
    a = np.ones(n)
    for m in range(1, n_freqs + 1):
        amp = rng.uniform(0.05, max_amp)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * m * np.arange(n) / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_wigner_n(n, seed=0):
    """Wigner-like sequence: alternating structure."""
    rng = np.random.default_rng(seed)
    a = np.ones(n)
    # Add alternating pattern
    a[::2] *= rng.uniform(0.7, 1.3)
    a[1::2] *= rng.uniform(0.7, 1.3)
    # Add envelope
    x = np.arange(n) / (n - 1) - 0.5
    a *= 1.0 - 0.3 * x ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def random_perturb_search(f0, time_limit, verbose=False):
    """Random perturbation with local refinement."""
    f = f0.copy()
    f = f / f.sum()
    n = len(f)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    noise_scale = 0.002
    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5.0:
            break

        # Generate perturbation
        f_pert = np.maximum(best_f + rng.standard_normal(n) * noise_scale, 0.0)
        f_pert = f_pert / f_pert.sum()
        obj = fast_convolve(f_pert).max()

        if obj < best_obj - 1e-12:
            # Do a few steps of subgradient descent from here
            for _ in range(10):
                conv = fast_convolve(f_pert)
                grad = get_subgradient(f_pert, conv, top_k=20)
                for lr in [0.001, 0.003, 0.01]:
                    fl = np.maximum(f_pert - lr * grad, 0.0)
                    fl = fl / fl.sum()
                    ol = fast_convolve(fl).max()
                    if ol < obj - 1e-14:
                        f_pert = fl
                        obj = ol
                        break
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f_pert.copy()
                noise_scale = max(noise_scale * 0.95, 1e-4)
                if verbose:
                    print(f"    RandPerturb: obj={best_obj:.10f}, scale={noise_scale:.6e}")
        else:
            noise_scale = min(noise_scale * 1.02, 0.01)

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: LP-FW, soft-max descent, pairwise descent,
    spectral initialization, multi-length exploration, random perturbation search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768, 0.25)
        try_seq(parent, "default_parent")

    # === Phase 1: Diverse initializations ===
    print("\n=== Phase 1: Diverse Initializations ===")
    t0 = time.monotonic()

    init_candidates = []
    n_values = [640, 768, 896, 1024]

    for n_try in n_values:
        init_candidates.append(('gauss', n_try, make_gaussian(n_try, 0.25)))
        init_candidates.append(('gauss2', n_try, make_gaussian(n_try, 0.15)))
        init_candidates.append(('bell', n_try, make_bell(n_try, 0.5)))
        init_candidates.append(('cos', n_try, make_multi_cosine(n_try, 4, 0.25, seed=n_try)))
        init_candidates.append(('cos5', n_try, make_multi_cosine(n_try, 5, 0.2, seed=n_try + 7)))
        init_candidates.append(('spectral', n_try, make_spectral_flat(n_try, seed=n_try)))
        init_candidates.append(('wigner', n_try, make_wigner_n(n_try, seed=n_try)))
        init_candidates.append(('flat', n_try, np.ones(n_try) / n_try))

    if parent is not None:
        init_candidates.append(('parent', len(parent), parent))

    # Evaluate and rank
    scored_inits = []
    for label, n_try, a_init in init_candidates:
        sc = objective_normalized(a_init)
        scored_inits.append((sc, a_init, label, n_try))
        if sc < 1.6:
            try_seq(a_init, f"init_{label}_{n_try}")
    scored_inits.sort(key=lambda x: x[0])

    print(f"  Top 5 inits:")
    for sc, a, label, n_try in scored_inits[:5]:
        print(f"    {label}_{n_try}: obj={sc:.6f}")
    print(f"  Phase 1 time: {time.monotonic()-t0:.1f}s")

    # === Phase 2: LP-FW on best initializations ===
    print("\n=== Phase 2: LP-FW Optimization ===")
    for trial_idx in range(min(3, len(scored_inits))):
        rem = deadline - time.monotonic() - 600
        if rem < 60:
            break
        sc_init, a_init, label, n_try = scored_inits[trial_idx]
        t_alloc = min(rem * 0.5, 250)
        print(f"  LP-FW [{label}_{n_try}]: alloc={t_alloc:.0f}s (init_obj={sc_init:.6f})")
        a_opt, obj = lp_fw_optimize(a_init, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, f"lpfw_{label}_{n_try}")

    # === Phase 3: Soft-max descent ===
    print("\n=== Phase 3: Soft-Max Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_alloc = min(rem, 300)
            print(f"  SoftMax: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = soft_max_descent(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "softmax")

        # Also try on the second-best init
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            for trial_idx in range(len(scored_inits)):
                if scored_inits[trial_idx][2] == 'parent' and scored_inits[trial_idx][3] == len(best_seq):
                    continue
                rem = deadline - time.monotonic() - 250
                if rem < 20:
                    break
                sc_init, a_init, label, n_try = scored_inits[trial_idx]
                t_alloc = min(rem * 0.4, 120)
                if t_alloc < 15:
                    break
                print(f"  SoftMax [{label}_{n_try}]: alloc={t_alloc:.0f}s")
                a_opt, obj = soft_max_descent(a_init, time_limit=t_alloc, verbose=True)
                try_seq(a_opt, f"softmax_{label}_{n_try}")
                break

    # === Phase 4: Pairwise descent ===
    print("\n=== Phase 4: Pairwise Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 20:
            t_alloc = min(rem, 200)
            print(f"  Pairwise: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = pairwise_descent(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "pairwise")

    # === Phase 5: LP-FW again on current best ===
    print("\n=== Phase 5: LP-FW Refinement ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(rem, 200)
            print(f"  LP-FW refine: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lpfw_refine")

    # === Phase 6: Multi-length exploration ===
    print("\n=== Phase 6: Multi-Length Exploration ===")
    if best_seq is not None:
        base_n = len(best_seq)
        for target_n in [1024, 896, 640, 512, 768]:
            if target_n == base_n:
                continue
            rem = deadline - time.monotonic() - 150
            if rem < 20:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")

            if sc < best_score + 0.02:
                t_alloc = min(rem * 0.3, 80)
                if t_alloc > 10:
                    print(f"  SoftMax n={target_n}: alloc={t_alloc:.0f}s")
                    a_opt, obj = soft_max_descent(f_interp, time_limit=t_alloc, verbose=False)
                    try_seq(a_opt, f"softmax_interp_n{target_n}")

    # === Phase 7: Random perturbation search ===
    print("\n=== Phase 7: Random Perturbation Search ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_alloc = min(rem, 120)
            print(f"  RandPerturb: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = random_perturb_search(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "randperturb")

    # === Phase 8: Final LP-FW squeeze ===
    print("\n=== Phase 8: Final LP-FW ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 20:
            t_alloc = min(rem, 100)
            print(f"  Final LP-FW: alloc={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "final_lpfw")

    # === Phase 9: Final soft-max polish ===
    print("\n=== Phase 9: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_alloc = min(rem, 60)
            a_opt, obj = soft_max_descent(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "final_softmax")

        rem = deadline - time.monotonic() - 10
        if rem > 5:
            t_alloc = min(rem, 20)
            a_opt, obj = pairwise_descent(best_seq, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, "final_pairwise")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    elapsed = time.monotonic() - (deadline - max(1.0, float(budget_s) - 25.0))
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"Total time: {elapsed:.1f}s")
    return [float(x) for x in best_seq]
