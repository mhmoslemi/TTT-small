def fast_convolve(a):

    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_lp_grad(f, p):
    """Compute gradient of sum(conv(f,f)^p) w.r.t. f."""
    n = len(f)
    conv = fast_convolve(f)
    # w[k] = p * conv[k]^(p-1)
    # Avoid issues with conv[k] = 0
    w = p * np.power(np.maximum(conv, 1e-30), p - 1)
    # grad[i] = 2 * sum_k w[k] * f[k-i]
    # Using the formula: h[j] = sum_k w[j+k] * f[k]
    w_rev = w[::-1]

    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    grad = 2.0 * conv_result[n - 1:2 * n - 1][::-1]

        # Nesterov lookahead
        f_y = f + momentum * lr * m_hat_prev if t > 1 else f.copy()
        # Recompute gradient at lookahead point (more expensive but better convergence)
        if t % 2 == 0:  # Do lookahead every other step for efficiency
            grad_y, conv_y = compute_lp_grad(f_y, p)
        else:
            grad_y = grad

        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * grad_y * grad_y
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        m_hat_prev = m_hat

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        p = min(p * p_growth, p_max)

        # Adaptive learning rate
        if no_improve > 100:
            lr *= 0.9
        if no_improve > 500:
            # Perturb from best
            rng = np.random.default_rng(t * 13 + 7)
            noise_scale = 0.002 + 0.004 * min(no_improve / 1000.0, 3.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = max(lr_init * 0.5, lr)
            p = min(p, 30)

        if verbose and t % 500 == 0:
            print(f"      LpAdam iter {t}: score={2*n*best_obj:.6f}, p={p:.0f}, lr={lr:.5f}, no_imp={no_improve}")

    return best_f, 2 * n * best_obj


def subgradient_polish(f0, time_limit, verbose=False):
    """
    Subgradient method on the true max-convolution with averaging.
    Uses Polyak step size.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    f_star_conv = conv.max()
    best_obj = f_star_conv
    best_f = f.copy()
    t0 = time.monotonic()

    # Running average
    avg_f = np.zeros(n)
    total_t = 0.0
    it = 0
    gap = None

    while time.monotonic() - t0 < time_limit:
        it += 1
        subgrad, conv, M = compute_subgrad(f)
        total_t += 1.0

        # Polyak step: lr = (f(x) - f*) / ||grad||^2
        # We don't know f*, so use a heuristic
        # Lower bound: 1/(2n-1) * n / sum(a)^2 ... just use 0
        grad_norm_sq = np.dot(subgrad, subgrad)
        if grad_norm_sq < 1e-30:
            break
        lr = max(0.001, min(0.1, (M - 0.3 / (2 * n)) / grad_norm_sq))

        f_new = f - lr * subgrad
        f_new = project_simplex(f_new)

        # Update running average
        avg_f = avg_f + (f_new - avg_f) / total_t

        conv_avg = fast_convolve(avg_f)
        obj_avg = conv_avg.max()
        if obj_avg < best_obj - 1e-14:
            best_obj = obj_avg
            best_f = avg_f.copy()

        f = f_new

        if verbose and it % 200 == 0:
            print(f"      Subgrad iter {it}: score={2*n*best_obj:.6f}")

    return best_f, 2 * n * best_obj


def mirror_descent(f0, time_limit, verbose=False):
    """
    Mirror descent with KL divergence (multiplicative weights) on Lp objective.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    p = 30
    lr = 0.03

    while time.monotonic() - t0 < time_limit:
        it += 1
        grad, conv = compute_lp_grad(f, p)

        # Multiplicative weights: f_new[i] = f[i] * exp(-lr * grad[i] / max_grad)
        max_grad = np.abs(grad).max()
        if max_grad < 1e-15:
            break
        scaled = -lr * grad / max_grad
        # Clip to avoid overflow
        scaled = np.clip(scaled, -50, 50)
        f_new = f * np.exp(scaled)
        f_new = np.maximum(f_new, 0.0)
        ss = f_new.sum()
        if ss < 1e-15:
            f_new = f.copy()
        else:
            f_new /= ss

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()

        f = f_new
        p *= 1.005
        if p > 200:
            p = 200

        if it % 200 == 0:
            lr *= 0.9
            if lr < 0.002:
                lr = 0.002

        if verbose and it % 500 == 0:
            print(f"      Mirror iter {it}: score={2*n*best_obj:.6f}, p={p:.0f}")

    return best_f, 2 * n * best_obj


def coordinate_descent_polish(f0, time_limit, verbose=False):
    """
    Coordinate descent: for each coordinate, try to reduce the max convolution
    by adjusting that coordinate and redistributing.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Find peak positions
        peak_thresh = M * 0.99
        peaks = np.where(conv >= peak_thresh)[0]

        improved = False

        # Try perturbing random coordinates that contribute to peaks
        n_trials = min(50, n)
        for _ in range(n_trials):
            # Pick a random coordinate that contributes to a peak
            if len(peaks) == 0:
                break
            pk = peaks[rng.integers(len(peaks))]
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if je <= js:
                continue
            i = rng.integers(js, je + 1)
            j = pk - i
            if j < 0 or j >= n:
                continue

            # Try reducing f[i] and redistributing
            old_val = f[i]
            if old_val < 1e-12:
                continue

            for factor in [0.7, 0.85, 0.5]:
                trial = f.copy()
                trial[i] = old_val * factor
                diff = old_val - trial[i]

                # Redistribute to a "safe" location (far from peaks)
                # Find position with minimum contribution to peaks
                safe_scores = np.zeros(n)
                for pk2 in peaks[:5]:
                    js2 = max(0, int(pk2) - n + 1)
                    je2 = min(n - 1, int(pk2))
                    safe_scores[js2:je2 + 1] += 1
                # Prefer positions with low safe_score and low current value
                weights = 1.0 / (1.0 + safe_scores) * (f + 1e-10)
                total_w = weights.sum()
                if total_w < 1e-15:
                    continue
                trial += diff * weights / total_w
                trial = np.maximum(trial, 0)
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                trial /= ts

                obj = fast_convolve(trial).max()
                if obj < best_obj - 1e-14:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    improved = True
                    break

        if not improved and it % 10 == 0:
            # Larger random perturbation
            noise_scale = 0.001 + 0.002 * min(it / 100.0, 5.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)

        if verbose and it % 50 == 0:
            print(f"      CD iter {it}: score={2*n*best_obj:.6f}")

    return best_f, 2 * n * best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, center_frac=0.35, sigma_frac=0.3):
    """Gaussian initialization."""
    x = np.arange(n) - n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 1e-15)
    s = a.sum()
    a /= s
    return a


def make_raised_cosine(n, width_frac=0.8):
    """Raised cosine window."""
    center = n / 2.0
    half_width = n * width_frac / 2.0
    x = np.abs(np.arange(n) - center) / half_width
    a = np.where(x <= 1.0, 0.5 * (1.0 + np.cos(np.pi * x)), 0.0)
    a = np.maximum(a, 1e-15)
    s = a.sum()
    a /= s
    return a


def make_bell_flat(n, flat_frac=0.3):
    """Flat-topped bell (super-Gaussian)."""
    center = n / 2.0
    x = np.abs(np.arange(n) - center) / (n * 0.4)
    a = np.power(np.maximum(1.0 - x, 0), 4)
    a = np.maximum(a, 1e-15)
    s = a.sum()
    a /= s
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Diverse initialization and quick evaluation (30s)
    Phase 2: Lp-Adam Nesterov main optimization (900s)
    Phase 3: Subgradient polishing (150s)
    Phase 4: Mirror descent alternative (150s)
    Phase 5: Multi-n exploration (300s)
    Phase 6: Final coordinate descent + Lp-Adam polish (170s + buffer)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a_norm = a / s
        try:
            score = float(evaluate_sequence(a_norm.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  *** NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
        return score

    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")
    t_phase1 = time.monotonic()

    # Try parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"  Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # Generate diverse initializations for multiple n values
    inits = []
    for n_try in [512, 768, 1024, 2048]:
        inits.append(('gauss_c35_s30', make_gaussian(n_try, 0.35, 0.30)))
        inits.append(('gauss_c50_s25', make_gaussian(n_try, 0.50, 0.25)))
        inits.append(('raised_cos', make_raised_cosine(n_try, 0.8)))
        inits.append(('bell_flat', make_bell_flat(n_try, 0.3)))
        inits.append(('uniform', np.ones(n_try) / n_try))

    # Exponential decay
    for n_try in [768, 1024]:
        a_exp = np.exp(-np.arange(n_try) / (n_try * 0.35))
        a_exp /= a_exp.sum()
        inits.append((f'exp_n{n_try}', a_exp))

    for i, (lbl, a_init) in enumerate(inits):
        if time.monotonic() > deadline - 1400:
            break
        sc = try_seq(a_init, f"init_{lbl}_n{len(a_init)}")

    print(f"  Phase 1 done in {time.monotonic()-t_phase1:.1f}s, best={best_score:.6f}")

    if best_seq is None:
        best_seq = make_gaussian(768, 0.35, 0.30)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # === Phase 2: Main Lp-Adam optimization ===
    print(f"\n=== Phase 2: Lp-Adam Nesterov (n={len(best_seq)}) ===")
    work = best_seq.copy()
    rem = deadline - time.monotonic()
    t2 = min(rem, 800)
    if t2 > 60:
        print(f"  Allocated: {t2:.0f}s")
        a_opt, obj = lp_adam_nesterov(work, time_limit=t2 * 0.7,
                                      p_init=10, p_max=200, lr_init=0.005,
                                      momentum=0.85, verbose=True)
        try_seq(a_opt, "lp_adam_main")

        # Second pass with different params
        if time.monotonic() < deadline - 500:
            rem2 = deadline - time.monotonic() - 500
            t2b = min(rem2, 250)
            a_opt2, obj2 = lp_adam_nesterov(best_seq, time_limit=t2b,
                                            p_init=30, p_max=500, lr_init=0.002,
                                            momentum=0.9, verbose=False)
            try_seq(a_opt2, "lp_adam_fine")

    # === Phase 3: Subgradient polishing ===
    print(f"\n=== Phase 3: Subgradient Polishing ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 350
        if rem > 30:
            t3 = min(rem, 150)
            print(f"  Allocated: {t3:.0f}s")
            a_opt, obj = subgradient_polish(best_seq, time_limit=t3, verbose=True)
            try_seq(a_opt, "subgrad")

    # === Phase 4: Mirror descent ===
    print(f"\n=== Phase 4: Mirror Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 30:
            t4 = min(rem, 150)
            print(f"  Allocated: {t4:.0f}s")
            a_opt, obj = mirror_descent(best_seq, time_limit=t4, verbose=True)
            try_seq(a_opt, "mirror")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    if best_seq is not None:
        base = best_seq
        for target_n in [1024, 2048, 512, 1536]:
            if target_n == len(base):
                continue
            rem = deadline - time.monotonic() - 100
            if rem < 60:
                break
            f_interp = interpolate_sequence(base, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")

            t5 = min(rem * 0.35, 100)
            if t5 > 20:
                a_opt, obj = lp_adam_nesterov(f_interp, time_limit=t5,
                                              p_init=20, p_max=200, lr_init=0.003,
                                              momentum=0.85, verbose=False)
                try_seq(a_opt, f"lp_adam_n{target_n}")

    # === Phase 6: Final polishing ===
    print(f"\n=== Phase 6: Final Polishing ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 30:
            t6 = min(rem, 80)
            print(f"  CD allocated: {t6:.0f}s")
            a_opt, obj = coordinate_descent_polish(best_seq, time_limit=t6, verbose=False)
            try_seq(a_opt, "cd_polish")

        # Final Lp-Adam with very fine p
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t7 = min(rem, 50)
            a_opt, obj = lp_adam_nesterov(best_seq, time_limit=t7,
                                          p_init=100, p_max=500, lr_init=0.0005,
                                          momentum=0.95, verbose=False)
            try_seq(a_opt, "final_lp_adam")

        # Final subgradient
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t8 = min(rem, 30)
            a_opt, obj = subgradient_polish(best_seq, time_limit=t8, verbose=False)
            try_seq(a_opt, "final_subgrad")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
