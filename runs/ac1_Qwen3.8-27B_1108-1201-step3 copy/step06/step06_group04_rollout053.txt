"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using subgradient Nesterov acceleration,
adaptive peak-targeted coordinate descent, and multi-length exploration.
Key ideas: direct subgradient (no LP), Nesterov momentum, pair-swap CD,
and diverse asymmetric initializations.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_norm(f):
    """Objective for sum-normalized sequence: 2*n*max(conv)."""
    return 2.0 * len(f) * fast_convolve(f).max()


def get_subgrad(a, peak_indices):
    """Compute subgradient contribution from given peak indices."""
    n = len(a)
    g = np.zeros(n)
    for k in peak_indices:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += 2.0 * a[k - idx]
    return g

def pair_cd(f0, time_limit, verbose=False):

    Pair-swap coordinate descent targeting peak contributors.
    For the current peak, try adjusting pairs of elements.
    """
    f = f0.astype(np.float64).copy()

    s = f.sum()
    f = f / s

        improved = False

        peak_indices = peak_indices[peak_indices >= np.argmax(conv) * 0.5]  # sanity

            if no_improve > 200:

            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)

            products = f[idx] * f[k - idx]
            total = products.sum()
            if total < 1e-15:

            top_count = min(6, len(idx))
            order = np.argsort(products)[-top_count:]

                j2 = k - j

                    trial[j] = new_v

                    if cnt > 0:

                        f = trial
                        improved = True

                js = max(0, k - n + 1)

                trial = f.copy()

                non_c = np.ones(n, dtype=bool)
                non_c[js:je + 1] = False
                if non_c.sum() > 0:
                    nc_w = f[non_c]
                    nw = nc_w.sum()
                    if nw > 1e-15:
                        trial[non_c] += d * nc_w / nw
                trial = np.maximum(trial, 0)
                trial = trial / trial.sum()
                obj = obj_norm(trial)
                if obj < best_obj:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    no_improve = 0
                else:
                    # Random perturbation
                    rng = np.random.default_rng(it * 17 + 5)
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                    no_improve = 0
        else:
            no_improve = 0

        if verbose and it % 500 == 0:
            print(f"    Pair CD iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def mirror_descent(f0, time_limit, verbose=False):
    """Multiplicative weight mirror descent with peak targeting."""
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)
    best_obj = obj_norm(f)
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.003
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit and no_improve < 200:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Peak weighting
        n_peaks = min(20, len(conv))
        peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]
        pv = conv[peaks]
        # Soft: use log-sum-exp approximation
        max_pv = pv.max()
        w = np.exp((pv - max_pv) / 0.05)
        w /= w.sum()

        # Gradient estimate
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-10:
                continue
            kk = int(peaks[i])
            js = max(0, kk - n + 1)
            je = min(n - 1, kk)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[kk - idx]

        # Multiplicative update
        f = f * np.exp(-lr * sg)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        obj = obj_norm(f)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        if no_improve > 50:
            rng = np.random.default_rng(it * 23 + 1)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.004, 1e-15)
            f = f / f.sum()
            lr = max(0.003 * 0.7, 1e-5)
            no_improve = 0

        if lr < 1e-7:
            break

        if verbose and it % 2000 == 0:
            print(f"    MD iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def interpolate_seq(f, new_n):
    """Linear interpolation of sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, f)
    fi = np.maximum(fi, 0)
    s = fi.sum()
    if s > 1e-15:
        fi /= s
    return fi


def make_gauss(n, sigma_frac=0.25):
    """Gaussian initialization."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cos_modulated(n, sigma_frac=0.25, freq=2.0, amp=0.1):
    """Gaussian with cosine modulation."""
    x = np.arange(n)
    xc = (x - n / 2.0) / (n * sigma_frac)
    a = np.exp(-0.5 * xc ** 2) * (1.0 + amp * np.cos(2.0 * np.pi * freq * x / n))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_asym_gauss(n, center_frac=0.4, sigma_frac=0.25):
    """Asymmetric Gaussian (off-center)."""
    x = np.arange(n)
    center = n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_flat_top(n, edge_frac=0.2, slope=0.3):
    """Flat-top with smooth edges."""
    x = np.arange(n) / (n - 1)
    a = np.ones(n)
    left = edge_frac
    right = 1.0 - edge_frac
    for i in range(n):
        if x[i] < left:
            a[i] = 1.0 - slope * (1.0 - x[i] / left) * (1.0 - x[i] / left) if left > 0 else 0
        elif x[i] > right:
            a[i] = 1.0 - slope * ((x[i] - right) / edge_frac) ** 2 if edge_frac > 0 else 0
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def optimize_sequence(f0, time_limit, verbose=False):
    """
    Run all local optimizers in sequence, returning the best.
    """
    f = f0.astype(np.float64).copy()
    f = np.maximum(f, 0)
    s = f.sum()
    f = f / s
    n = len(f)
    best_obj = obj_norm(f)
    best_f = f.copy()

    t0 = time.monotonic()

    # Phase 1: Nesterov subgradient
    rem = time_limit - (time.monotonic() - t0) - 60
    if rem > 30:
        t1 = min(rem * 0.4, 200)
        print(f"    NS: t={t1:.0f}s")
        f_opt, obj = nesterov_subgrad(f, time_limit=t1, verbose=verbose)
        if obj < best_obj:
            best_obj = obj
            best_f = f_opt

    # Phase 2: Pair CD
    rem = time_limit - (time.monotonic() - t0) - 60
    if rem > 20:
        t2 = min(rem * 0.35, 120)
        print(f"    PairCD: t={t2:.0f}s")
        f_opt, obj = pair_cd(best_f, time_limit=t2, verbose=verbose)
        if obj < best_obj:
            best_obj = obj
            best_f = f_opt

    # Phase 3: Mirror descent
    rem = time_limit - (time.monotonic() - t0) - 30
    if rem > 15:
        t3 = min(rem * 0.3, 80)
        print(f"    MD: t={t3:.0f}s")
        f_opt, obj = mirror_descent(best_f, time_limit=t3, verbose=verbose)
        if obj < best_obj:
            best_obj = obj
            best_f = f_opt

    # Phase 4: Final pair CD polish
    rem = time_limit - (time.monotonic() - t0) - 5
    if rem > 10:
        t4 = min(rem * 0.6, 60)
        print(f"    FinalCD: t={t4:.0f}s")
        f_opt, obj = pair_cd(best_f, time_limit=t4, verbose=False)
        if obj < best_obj:
            best_obj = obj
            best_f = f_opt

    return best_f, best_obj


def try_candidate(a, best_seq, best_score, label=""):
    """Evaluate and track best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization: Nesterov subgradient, pair-swap coordinate
    descent, mirror descent, with diverse initializations and multi-length search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_gauss(768, 0.25)
        best_seq, best_score = try_candidate(parent, best_seq, best_score, "default")
        print(f"  Default init score: {best_score:.6f}")

    n_parent = len(parent)

    # === Phase 1: Diverse initialization ===
    print("\n=== Phase 1: Initialization ===")
    init_list = []

    # Parent and interpolations
    init_list.append(parent)
    for n_try in range(max(128, n_parent - 64), n_parent + 65, 32):
        if n_try != n_parent:
            init_list.append(interpolate_seq(parent, n_try))

    # Various shapes at parent length and nearby
    for n_try in [n_parent, max(128, n_parent - 100), min(2048, n_parent + 100)]:
        for sf in [0.15, 0.20, 0.25, 0.30, 0.35, 0.40]:
            init_list.append(make_gauss(n_try, sf))
        for cm in [0.3, 0.4, 0.5, 0.6, 0.7]:
            init_list.append(make_asym_gauss(n_try, cm, 0.25))
        for fmod in [1.0, 2.0, 3.0, 5.0]:
            for amp in [0.05, 0.1, 0.2]:
                init_list.append(make_cos_modulated(n_try, 0.25, fmod, amp))
        init_list.append(make_flat_top(n_try, 0.15, 0.3))
        init_list.append(make_flat_top(n_try, 0.3, 0.5))

    # Evaluate all initializations
    best_init = None
    best_init_obj = float('inf')
    for ic in init_list:
        ic = np.maximum(ic, 0)
        s = ic.sum()
        if s < 1e-10:
            continue
        a_norm = ic / s
        obj = obj_norm(a_norm)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_norm.copy()
        best_seq, best_score = try_candidate(a_norm, best_seq, best_score,
                                              f"init_n{len(a_norm)}")

    print(f"  Best init: obj_norm={best_init_obj:.6f} (n={len(best_init)})")

    # === Phase 2: Main optimization on best init ===
    print(f"\n=== Phase 2: Main optimization (n={len(best_init)}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 60:
        t_alloc = min(rem, 250)
        print(f"  Main opt: t={t_alloc:.0f}s")
        f_opt, obj = optimize_sequence(best_init, time_limit=t_alloc, verbose=True)
        best_seq, best_score = try_candidate(f_opt, best_seq, best_score, "main_opt")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length ===")
    base = best_seq if best_seq is not None else best_init
    lengths_to_try = sorted(set([
        max(128, len(base) - 128),
        max(128, len(base) - 64),
        len(base),
        len(base) + 64,
        len(base) + 128,
        len(base) + 256,
    ]))

    for target_n in lengths_to_try:
        if target_n == len(base) and len(base) == len(best_init):
            continue  # Already done
        rem = deadline - time.monotonic() - 200
        if rem < 60:
            break
        f_interp = interpolate_seq(base, target_n)
        best_seq, best_score = try_candidate(f_interp, best_seq, best_score,
                                              f"interp_n{target_n}")
        t_alloc = min(rem * 0.3, 100)
        if t_alloc < 25:
            continue
        print(f"  Opt n={target_n}: t={t_alloc:.0f}s")
        f_opt, obj = optimize_sequence(f_interp, time_limit=t_alloc, verbose=False)
        best_seq, best_score = try_candidate(f_opt, best_seq, best_score,
                                              f"opt_n{target_n}")

    # === Phase 4: Deep optimization on best ===
    if best_seq is not None:
        print(f"\n=== Phase 4: Deep opt (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 150
        if rem > 50:
            t_alloc = min(rem, 200)
            print(f"  Deep: t={t_alloc:.0f}s")
            f_opt, obj = optimize_sequence(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = try_candidate(f_opt, best_seq, best_score, "deep")

    # === Phase 5: Perturbed restarts ===
    if best_seq is not None:
        print(f"\n=== Phase 5: Perturbed restarts ===")
        n_best = len(best_seq)
        for variant in range(4):
            rem = deadline - time.monotonic() - 80
            if rem < 40:
                break
            t_alloc = min(rem / (4 - variant), 100)

            if variant == 0:
                noise = rng.standard_normal(n_best) * 0.003
                f_init = project_simplex(best_seq + noise)
            elif variant == 1:
                # Blend with uniform
                f_init = 0.9 * best_seq + 0.1 * (np.ones(n_best) / n_best)
                f_init = f_init / f_init.sum()
            elif variant == 2:
                # Reverse blend
                f_init = 0.5 * best_seq + 0.5 * best_seq[::-1]
                f_init = f_init / f_init.sum()
            else:
                # Sharper peak
                f_init = best_seq ** 1.5
                f_init = np.maximum(f_init, 0)
                f_init = f_init / f_init.sum()

            print(f"  Variant {variant}: t={t_alloc:.0f}s")
            f_opt, obj = optimize_sequence(f_init, time_limit=t_alloc, verbose=False)
            best_seq, best_score = try_candidate(f_opt, best_seq, best_score,
                                                  f"perturb_{variant}")

    # === Phase 6: Final fine-tuning ===
    if best_seq is not None:
        print(f"\n=== Phase 6: Final polish ===")
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            t1 = min(rem * 0.5, 80)
            print(f"  Final PairCD: t={t1:.0f}s")
            f_opt, obj = pair_cd(best_seq, time_limit=t1, verbose=True)
            best_seq, best_score = try_candidate(f_opt, best_seq, best_score, "fin_cd")

            rem2 = deadline - time.monotonic() - 5
            if rem2 > 10 and best_seq is not None:
                t2 = min(rem2, 40)
                print(f"  Final NS: t={t2:.0f}s")
                f_opt, obj = nesterov_subgrad(best_seq, time_limit=t2)
                best_seq, best_score = try_candidate(f_opt, best_seq, best_score, "fin_ns")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
