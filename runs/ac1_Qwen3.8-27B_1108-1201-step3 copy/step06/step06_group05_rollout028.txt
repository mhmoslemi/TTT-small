import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (max(size, 1) - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(x):
    """Stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def lse_objective_and_grad(x, beta):
    """
    Compute LSE_beta(conv(softmax(x), softmax(x))) and its gradient w.r.t. x.
    """
    a = softmax(x)
    n = len(a)
    c = fast_convolve(a)

    # LSE computation
    scaled = beta * c
    max_scaled = scaled.max()
    exp_scaled = np.exp(scaled - max_scaled)
    lse_val = (max_scaled + np.log(exp_scaled.sum())) / beta
    w = exp_scaled / exp_scaled.sum()

    # Gradient w.r.t. a: grad_a[j] = 2 * sum_k w[k] * a[k-j]
    # = 2 * cross_correlation(w, a)[j]
    # Using: cross_corr(w,a)[j] = (a_rev * w)[n-1+j] where a_rev = a[::-1]
    a_rev = a[::-1].copy()
    conv_len = n + len(w) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    F = np.fft.rfft(a_rev, fft_size)
    W = np.fft.rfft(w, fft_size)
    conv = np.fft.irfft(F * W, fft_size)[:conv_len]
    grad_a = 2.0 * conv[n - 1:2 * n - 1]

    # Gradient w.r.t. x: grad_x = a * (grad_a - dot(a, grad_a))
    dot_ag = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot_ag)

    return lse_val, grad_x

def lbfgsb_optimize(a_init, n, beta_schedule, time_limit, verbose=False):
    """
    Run L-BFGS-B with LSE smoothing, annealing beta.
    a_init: initial normalized sequence (sum=1, non-negative)
    """
    n = len(a_init)
    # Initial x: log(a) centered
    a_safe = np.maximum(a_init, 1e-30)
    x = np.log(a_safe / a_safe.sum())
    x = x - x.mean()  # center for numerical stability

    t0 = time.monotonic()
    best_x = x.copy()
    best_lse = np.inf

    for beta in beta_schedule:
        if time.monotonic() - t0 > time_limit * 0.95:
            break

        stage_start = time.monotonic()
        stage_time = time_limit / len(beta_schedule)

        def func_grad(x_eval):
            val, grad = lse_objective_and_grad(x_eval, beta)
            return val, grad

        try:
            res = minimize(func_grad, x, method='L-BFGS-B',
                           options={'maxiter': 300, 'ftol': 1e-14,
                                   'gtol': 1e-10, 'maxfun': 500})
        except Exception as e:
            if verbose:
                print(f"    L-BFGS-B error at beta={beta}: {e}")
            break

        if res.success:
            x = res.x
            lse_final, _ = lse_objective_and_grad(x, beta)
            # Use the same beta to compare
            if lse_final < best_lse:
                best_lse = lse_final
                best_x = x.copy()
            if verbose:
                a_check = softmax(x)
                obj_true = 2 * n * fast_convolve(a_check).max()
                print(f"    beta={beta}: LSE={lse_final:.10f}, true_obj={obj_true:.10f}, "
                      f"time={time.monotonic()-t0:.1f}s")
        else:
            # Check if we still improved
            lse_final, _ = lse_objective_and_grad(x, beta)
            if lse_final < best_lse:
                best_lse = lse_final
                best_x = x.copy()
            if verbose:
                print(f"    beta={beta}: not converged, LSE={lse_final:.10f}")

        # Time check within stage
        if time.monotonic() - stage_start > stage_time * 1.5:
            if verbose:
                print(f"    beta={beta}: time exceeded stage budget")

    a_best = softmax(best_x)
    a_best = np.maximum(a_best, 0)
    a_best = a_best / a_best.sum()
    obj = 2 * n * fast_convolve(a_best).max()
    return a_best, obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def subgradient_refine(f0, time_limit, verbose=False):
    """
    Subgradient method with iterate averaging and adaptive step size.
    Focuses on the top peaks to get a good descent direction.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    c = fast_convolve(f)
    best_obj = c.max()

    t0 = time.monotonic()
    it = 0
    lr = 0.05
    avg_f = np.zeros(n)
    avg_count = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(f)
        M = c.max()

        # Top-k peaks for subgradient
        k_top = min(20, 2 * n - 1)
        peak_idx = np.argpartition(c, -k_top)[-k_top:]
        peak_vals = c[peak_idx]
        # Weight by exponential of excess
        excess = peak_vals - peak_vals.min()
        w_peaks = np.exp(excess * 10)
        w_peaks /= w_peaks.sum()

        # Compute subgradient direction
        w_full = np.zeros(2 * n - 1)
        w_full[peak_idx] = w_peaks
        a_rev = f[::-1].copy()
        conv_len = n + (2 * n - 1) - 1
        fft_size = 1 << (conv_len - 1).bit_length()
        F = np.fft.rfft(a_rev, fft_size)
        W = np.fft.rfft(w_full, fft_size)
        conv = np.fft.irfft(F * W, fft_size)[:conv_len]
        grad = 2.0 * conv[n - 1:2 * n - 1]

        # Projected gradient step
        f_new = f - lr * grad
        f_new = project_simplex(f_new)

        # Polyak-like step size update
        # Lower bound on optimal: 1/(2n) (from average argument)
        # But we use empirical lower bound
        obj_new = fast_convolve(f_new).max()

        if obj_new < best_obj:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            lr = min(lr * 1.01, 0.1)
        else:
            no_improve += 1
            lr *= 0.98

        # Iterate averaging (Helinski)
        avg_f += f_new
        avg_count += 1
        avg_f_norm = avg_f / avg_count
        c_avg = fast_convolve(avg_f_norm)
        if c_avg.max() < best_obj:
            best_obj = c_avg.max()
            best_f = avg_f_norm.copy()

        if verbose and it % 100 == 0:
            print(f"    SubGrad iter {it}: obj={obj_new:.10f}, best={best_obj:.10f}")

        if no_improve > 300:
            break

    return best_f, best_obj


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_raised_cosine(n):

    a = 0.5 * (1 + np.cos(np.pi * x))

    a /= a.sum()
    return a

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    return f_interp
