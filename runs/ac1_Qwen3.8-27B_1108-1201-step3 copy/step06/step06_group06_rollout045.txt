        # B @ lam = mu * 1_S, sum(lam) = 1

        dim_K = len(K)

                if 0 <= i_val < n:
                    B[ii, kk] = 2.0 * a[i_val]

        A_sys2[:dim_S, :dim_K] = B
        A_sys2[:dim_S, dim_K] = -1.0
        A_sys2[dim_S, :dim_K] = 1.0
        b_sys2 = np.zeros(dim_S + 1)
        b_sys2[dim_S] = 1.0
        
        try:
            if dim_S + 1 >= dim_K + 1:
                lam_sol = np.linalg.lstsq(A_sys2, b_sys2, rcond=None)[0]
            else:
                lam_sol = np.linalg.lstsq(A_sys2.T, b_sys2.T, rcond=None)[0]
        except:
            lam = np.ones(dim_K) / dim_K
            continue
        
        lam = np.maximum(lam_sol[:dim_K], 0.0)
        lam_sum = lam.sum()
        if lam_sum < 1e-15:
            lam = np.ones(dim_K) / dim_K
        else:
            lam = lam / lam_sum
        
        # Check convergence
        if it > 0 and abs(M - best_M) < 1e-14:
            break
    
    return best_a
