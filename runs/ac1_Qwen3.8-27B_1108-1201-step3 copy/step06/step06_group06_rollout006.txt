"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines: (1) Nesterov-accelerated smooth optimization with cosine annealing,
(2) Direct subgradient with Polyak step size, (3) Mirror-descent (multiplicative),
(4) Peak-pair coordinate descent, (5) Multi-length spectral exploration.
Key insight: linear auto-convolution equals DFT-based power spectrum of zero-padded
sequence, enabling frequency-domain guided optimization.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_objective_and_grad(a, tau):
    """Compute log-sum-exp smooth approximation of max(conv) and its gradient."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    w = w / w_sum
    obj = tau * (np.log(w_sum) + scaled.max())

    # grad[j] = (2/tau) * sum_l w[j+l] * a[l]
    # This is a cross-correlation of a with w
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    h = conv_result[n - 1:2 * n - 1][::-1]
    grad = (2.0 / tau) * h
    return obj, grad, conv


def subgradient_max(a):
    """Compute subgradient of max(conv(a,a)). Returns (grad, max_val)."""
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    # Use a small band around max for stability
    threshold = M - 1e-6 * M
    active = (conv >= threshold).astype(float)
    count = active.sum()
    if count < 1:
        active = np.zeros(len(conv))
        active[np.argmax(conv)] = 1.0
        count = 1.0

    # grad[j] = (2/count) * sum_k active[k] * a[k-j] (valid k-j in [0,n-1])
    # Use FFT: cross-correlation
    active_rev = active[::-1]
    total_len = n + len(active) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    Act = np.fft.rfft(active_rev, fft_size)
    conv_result = np.fft.irfft(A * Act, fft_size)
    h = conv_result[n - 1:2 * n - 1][::-1]
    grad = (2.0 / count) * h
    return grad, M, active


def nesterov_smooth_optimize(f0, time_limit, tau_init=None, lr_init=0.005, verbose=False):
    """Nesterov-accelerated Adam on smooth surrogate with cosine temperature annealing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M0 = conv.max()
    if tau_init is None:
        tau = 0.06 * M0
    else:
        tau = tau_init
    tau_min = 0.01 * M0

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    t = 0

    best_obj = M0
    best_f = f.copy()
    no_improve = 0
    t0 = time.monotonic()
    total_time = time_limit

    # Nesterov momentum tracking
    y = f.copy()
    mu = 0.9  # Nesterov momentum

    while time.monotonic() - t0 < total_time:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / total_time, 1.0)

        # Cosine annealing for temperature
        tau_curr = tau_min + (tau - tau_min) * 0.5 * (1 + np.cos(np.pi * progress))
        # Learning rate: warmup then cosine decay
        if t < 50:
            lr_curr = lr * (t / 50)
        else:
            lr_curr = lr * 0.5 * (1 + np.cos(np.pi * max(0, (progress - 0.05) / 0.95)))

        # Nesterov lookahead
        y = f + mu * (f - (f - lr_curr * m / (np.sqrt(v) + eps) + f))  # simplified
        # Actually, let's do standard Nesterov:
        # y = f + mu * velocity (where velocity tracks recent changes)
        # Simplified: just use the momentum in Adam which already provides acceleration

        obj, grad, conv = smooth_objective_and_grad(f, tau_curr)
        M = conv.max()

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - lr_curr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        # Restart logic
        if no_improve > 200:
            rng = np.random.default_rng(t * 31 + 7)
            noise_scale = 0.003 + 0.005 * min(no_improve / 500.0, 2.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = lr_init * (0.5 + 0.3 * rng.random())
            tau = 0.04 * fast_convolve(f).max()
            if verbose:
                print(f"    Nesterov restart at t={t}: obj={2*n*best_obj:.8f}")

        if verbose and t % 1000 == 0:
            print(f"    Nesterov t={t}: obj={2*n*best_obj:.8f}, tau={tau_curr:.2e}, lr={lr_curr:.5e}")

    return best_f, 2 * n * best_obj


def direct_subgradient_optimize(f0, time_limit, verbose=False):
    """Direct subgradient method with Polyak-style step size for max(conv)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    t = 0

    # Lower bound: 1/(2n-1) (by averaging)
    # Actually a better lower bound: by Cauchy, sum_k (conv[k]) = (sum a)^2 = 1
    # So max >= 1/(2n-1). But we can do better.
    # For non-negative a: (conv[n-1]) = sum a[j]*a[n-1-j] >= (sum a[j]^2) by... no.
    # Use: max >= 1/(2n-1) as baseline.
    lower_bound = 1.0 / (2 * n - 1) * 0.95  # slightly optimistic

    gap = best_obj - lower_bound
    step_scale = 1.0
    momentum = np.zeros(n)
    beta_m = 0.7

    while time.monotonic() - t0 < time_limit:
        t += 1
        grad, M, active = subgradient_max(f)

        # Polyak step size: step = gap / ||grad||^2
        grad_norm_sq = np.sum(grad ** 2)
        if grad_norm_sq > 1e-15:
            step = (gap * step_scale) / grad_norm_sq
        else:
            step = 0.01 / n

        # Clamp step
        step = min(step, 5.0 / n)
        step = max(step, 1e-8)

        # Momentum
        momentum = beta_m * momentum + grad
        f_new = f - step * momentum
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            gap = best_obj - lower_bound
        else:
            step_scale *= 0.95

        f = f_new

        if verbose and t % 500 == 0:
            print(f"    Subgrad t={t}: obj={2*n*best_obj:.8f}, gap={gap:.6f}")

        # Convergence check
        if gap < 1e-7:
            break

    return best_f, 2 * n * best_obj


def mirror_descent_optimize(f0, time_limit, verbose=False):
    """Mirror descent (multiplicative weights) on the smooth objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M0 = conv.max()
    best_obj = M0
    best_f = f.copy()
    t0 = time.monotonic()
    t = 0
    lr = 0.05
    tau = 0.05 * M0

    while time.monotonic() - t0 < time_limit:
        t += 1
        obj, grad, conv = smooth_objective_and_grad(f, tau)
        M = conv.max()

        # Multiplicative update: a[j] *= exp(-lr * grad[j])
        # Then renormalize
        scaled_grad = lr * grad
        scaled_grad -= scaled_grad.mean()  # Center to preserve sum approximately
        f_new = f * np.exp(-scaled_grad)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()

        f = f_new

        # Adaptive learning rate
        if t % 100 == 0:
            lr *= 0.97
            tau *= 0.99

        if verbose and t % 500 == 0:
            print(f"    Mirror t={t}: obj={2*n*best_obj:.8f}, lr={lr:.4f}")

        if lr < 1e-6:
            break

    return best_f, 2 * n * best_obj


def peak_pair_cd(f0, time_limit, verbose=False):
    """Aggressive peak-pair coordinate descent with mass redistribution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    stalling = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        # Top peaks
        n_peaks = min(8, 2 * n - 1)
        peak_indices = np.argsort(conv)[-n_peaks:]

        for pk in peak_indices:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[pk - idx]
            j1 = int(idx[np.argmax(products)])
            j2 = pk - j1

            if j1 == j2:
                # Diagonal: reduce f[j1]
                old = f[j1]
                if old < 1e-12:
                    continue
                # Find where to put the mass: elements NOT contributing to top peaks
                safe_mask = np.ones(n, dtype=bool)
                for pk2 in peak_indices[:4]:
                    pk2 = int(pk2)
                    s2 = max(0, pk2 - n + 1)
                    e2 = min(n - 1, pk2)
                    safe_mask[s2:e2 + 1] = False
                safe_mask[j1] = False
                if safe_mask.sum() < 2:
                    continue
                weights = f[safe_mask]
                if weights.sum() < 1e-15:
                    continue
                weights = weights / weights.sum()

                for frac in [0.3, 0.5, 0.7]:
                    trial = f.copy()
                    removed = old * frac
                    trial[j1] = old - removed
                    trial[safe_mask] += removed * weights
                    trial = np.maximum(trial, 0)
                    trial /= trial.sum()
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break
            else:
                # Off-diagonal: reduce product f[j1]*f[j2]
                old1, old2 = f[j1], f[j2]
                if old1 < 1e-12 or old2 < 1e-12:
                    continue
                # Try reducing the larger one and moving mass to safe locations
                safe_mask = np.ones(n, dtype=bool)
                for pk2 in peak_indices[:4]:
                    pk2 = int(pk2)
                    s2 = max(0, pk2 - n + 1)
                    e2 = min(n - 1, pk2)
                    safe_mask[s2:e2 + 1] = False
                safe_mask[j1] = False
                safe_mask[j2] = False
                if safe_mask.sum() < 2:
                    continue
                weights = f[safe_mask]
                if weights.sum() < 1e-15:
                    continue
                weights = weights / weights.sum()

                target = j1 if old1 > old2 else j2
                other = j2 if target == j1 else j1
                old_target = f[target]

                for frac in [0.2, 0.4, 0.6]:
                    trial = f.copy()
                    removed = old_target * frac
                    trial[target] = old_target - removed
                    trial[safe_mask] += removed * weights
                    trial = np.maximum(trial, 0)
                    trial /= trial.sum()
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break

            if improved:
                break

        if not improved:
            stalling += 1
            if stalling > 20:
                # Perturb and try again
                rng = np.random.default_rng(int(time.monotonic() * 10000))
                noise = 0.002 + 0.003 * min(stalling / 50.0, 3.0)
                f = project_simplex(best_f + rng.standard_normal(n) * noise)
                stalling = 0
        else:
            stalling = 0

        if verbose and int(time.monotonic() - t0) % 20 == 0:
            print(f"    CD: obj={2*n*best_obj:.8f}, stall={stalling}")

    return best_f, 2 * n * best_obj


def spectral_init(n, mode=0):
    """Generate initializations using spectral design ideas."""
    if mode == 0:
        # Flat-top window (Kaiser-like)
        x = np.arange(n) / (n - 1)
        a = np.sinc((2 * x - 1) * 2.5)
    elif mode == 1:
        # Raised cosine
        x = np.arange(n) / n
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif mode == 2:
        # Gaussian
        x = (np.arange(n) - n / 2) / (n * 0.25)
        a = np.exp(-0.5 * x ** 2)
    elif mode == 3:
        # Triangular
        x = np.abs(np.arange(n) / n - 0.5) * 2
        a = np.maximum(0, 1 - x)
    elif mode == 4:
        # Cosine bell
        x = np.arange(n) / n
        a = np.cos(np.pi * (x - 0.5)) ** 2
        a = np.maximum(a, 0)
    elif mode == 5:
        # Asymmetric exponential
        a = np.exp(-np.arange(n) / (n * 0.3))
    elif mode == 6:
        # Double peak
        x = np.arange(n)
        c1, c2 = n * 0.3, n * 0.7
        a = np.exp(-0.5 * ((x - c1) / (n * 0.15)) ** 2) + np.exp(-0.5 * ((x - c2) / (n * 0.15)) ** 2)
    elif mode == 7:
        # Watermel1on: flat center, tapered edges
        x = np.arange(n) / n
        edge = 0.2
        center_mask = (x > edge) & (x < 1 - edge)
        a = np.ones(n)
        a[:int(edge * n)] = np.linspace(0.1, 1.0, int(edge * n))
        a[int((1 - edge) * n):] = np.linspace(1.0, 0.1, n - int((1 - edge) * n))
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def hopkins_restart(f, n_restarts=3, noise_scales=None, time_limit=10, verbose=False):
    """Try small perturbations and keep the best."""
    f = np.maximum(f.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    if noise_scales is None:
        noise_scales = [0.001, 0.003, 0.005, 0.01, 0.015]

    rng = np.random.default_rng(int(time.monotonic() * 100000))

    for ns in noise_scales:
        if time.monotonic() - t0 > time_limit:
            break
        for _ in range(n_restarts):
            if time.monotonic() - t0 > time_limit:
                break
            perturbed = project_simplex(f + rng.standard_normal(n) * ns)
            obj = fast_convolve(perturbed).max()
            if obj < best_obj:
                best_obj = obj
                best = perturbed.copy()

    return best, 2 * n * best_obj

                a_pert, obj_pert = hopkins_restart(best_seq, n_restarts=5,
                                                   noise_scales=[0.002, 0.005, 0.008],
                                                   time_limit=5)
                try_seq(a_pert, f"hopkins_{attempt}")
                if time.monotonic() > deadline - 30:
                    break
                # Quick Nesterov on the perturbed
                t_quick = min(time.monotonic() and (deadline - time.monotonic() - 20), 40)
                if t_quick > 10:
                    a_opt, obj = nesterov_smooth_optimize(best_seq, time_limit=t_quick,
                                                          lr_init=0.002, verbose=False)
                    try_seq(a_opt, f"quick_nest_{attempt}")

    # === Phase 7: Multi-length Exploration ===
    print(f"\n=== Phase 7: Multi-Length Exploration ===")
    if best_seq is not None:
        base_n = len(best_seq)
        for target_n in [base_n + 128, base_n - 128, base_n + 256, base_n // 2 + 64]:
            if target_n < 64 or target_n > 2048:
                continue
            if target_n == base_n:
                continue
            rem = deadline - time.monotonic() - 50
            if rem < 20:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = try_seq(f_interp, f"expl_n{target_n}")

            t_exp = min(rem * 0.4, 60)
            if t_exp > 15:
                a_opt, obj = nesterov_smooth_optimize(f_interp, time_limit=t_exp,
                                                      lr_init=0.003, verbose=False)
                try_seq(a_opt, f"expl_opt_n{target_n}")

    # === Phase 8: Final Polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 20:
            t8 = min(rem, 80)
            print(f"  Allocating {t8:.0f}s for final polish")
            # Very small learning rate Nesterov
            a_opt, obj = nesterov_smooth_optimize(
                best_seq, time_limit=t8 * 0.6, lr_init=0.0005, verbose=False)
            try_seq(a_opt, "final_polish_nest")

            rem2 = deadline - time.monotonic()
            if rem2 > 10:
                t8b = min(rem2, t8 * 0.4)
                a_opt2, obj2 = peak_pair_cd(best_seq, time_limit=t8b, verbose=False)
                try_seq(a_opt2, "final_polish_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
