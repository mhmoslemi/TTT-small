import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def make_gaussian(n, sigma_frac=0.3):
    """Create a Gaussian initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    return normalize_nonneg(a)


def make_hann(n):
    """Hann window (raised cosine)."""
    a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    return normalize_nonneg(a)


def make_bell(n, width_frac=0.6, power=4):
    """Bell-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** power
    return normalize_nonneg(a)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    kernel = np.array([0.1, 0.2, 0.4, 0.2, 0.1])
    f_smooth = np.convolve(f_interp, kernel, mode='same')
    return normalize_nonneg(f_smooth)


def peak_equalization_step(f, peak_count=5, eta_frac=0.05, verbose=False):
    """
    Newton-like step that equalizes the top peaks and reduces them.
    Returns (new_f, new_obj, improved).
    """
    n = len(f)
    c = fast_convolve(f)
    M = c.max()
    
    # Identify active peaks
    k_peaks = min(peak_count, len(c))
    peak_idx = np.argpartition(c, -k_peaks)[-k_peaks:]
    peak_idx = peak_idx[np.argsort(c[peak_idx])[::-1]]
    
    # Only use peaks that are close to the max
    tol = 0.1 * M
    active_peaks = peak_idx[c[peak_idx] >= M - tol]
    m = len(active_peaks)
    if m < 1:
        return f.copy(), M, False
    
    m = min(m, 20)  # Cap to avoid ill-conditioned systems
    active_peaks = active_peaks[:m]
    
    eta = eta_frac * M
    
    # Build linear system: A delta = b
    # Rows 0..m-1: 2 * sum_j f[j] * delta[k_i - j] = -eta
    # Row m: sum_j delta[j] = 0
    A = np.zeros((m + 1, n))
    b = np.zeros(m + 1)
    
    for i, k in enumerate(active_peaks):
        b[i] = -eta
        # A[i, j] = 2 * f[k - j] if 0 <= k - j < n
        js = max(0, k - (n - 1))
        je = min(n - 1, k)
        if js <= je:
            A[i, js:je+1] = 2.0 * f[k - np.arange(js, je + 1)]
    A[m, :] = 1.0
    b[m] = 0.0
    
    # Minimum norm solution: delta = A^T (A A^T)^{-1} b
    try:
        AAT = A @ A.T
        # Add small regularization for numerical stability
        AAT_reg = AAT + 1e-10 * np.eye(m + 1)
        Ab = np.linalg.solve(AAT_reg, b)
        delta = A.T @ Ab
    except np.linalg.LinAlgError:
        return f.copy(), M, False
    
    # Backtracking line search
    best_f = f.copy()
    best_obj = M
    alpha = 1.0
    for _ in range(20):
        f_trial = f + alpha * delta
        f_trial = normalize_nonneg(f_trial)
        c_trial = fast_convolve(f_trial)
        obj_trial = c_trial.max()
        
        if obj_trial < best_obj - 1e-15:
            best_f = f_trial
            best_obj = obj_trial
            if verbose:
                print(f"      PE: alpha={alpha:.4f}, obj={obj_trial:.10f} (target reduction {eta:.2e})")
            break
        alpha *= 0.5
        if alpha < 1e-8:
            break
    
    return best_f, best_obj, best_obj < M - 1e-15

def freq_domain_shaping(f0, time_limit, verbose=False):
    """
    Frequency-domain iterative shaping: modify the power spectrum to reduce
    auto-convolution peaks. Uses Gerchberg-Saxton-like iterations.
    """
    f = normalize_nonneg(f0.astype(np.float64))
    n = len(f)
    L = 1 << (2 * n - 1).bit_length()
    
    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    gamma = 0.1
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute spectrum
        a_pad = np.zeros(L)
        a_pad[:n] = f
        A = np.fft.fft(a_pad)
        power = np.abs(A) ** 2
        
        # Compute current auto-convolution
        c = np.fft.ifft(A * A).real[:2*n-1]
        M = c.max()
        
        # Identify peaks
        k_peaks = min(10, len(c))
        peak_idx = np.argpartition(c, -k_peaks)[-k_peaks:]
        
        # Compute "peak indicator" in frequency domain
        p = np.zeros(L)
        for k in peak_idx:
            p[int(k)] = c[int(k)] - M * 0.8  # Excess above 80% of max
        P = np.fft.fft(p)
        
        # Reduce spectral energy at peak-contributing frequencies
        power_reduction = gamma * np.abs(P) ** 2 / (L * L)
        power_new = np.maximum(power - power_reduction, 1e-20 * power.max())
        
        # Phase preservation
        phase = np.angle(A)
        A_new = np.sqrt(power_new) * np.exp(1j * phase)
        
        # Inverse FFT
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = normalize_nonneg(a_new)
        
        # Evaluate
        c_new = fast_convolve(a_new)
        obj_new = c_new.max()
        
        if obj_new < best_obj - 1e-14:
            f = a_new
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
            gamma = min(gamma * 1.05, 0.5)
        else:
            no_improve += 1
            gamma *= 0.9
            if no_improve > 50:
                gamma = 0.05
                if no_improve > 150:
                    break
        
        if verbose and it % 50 == 0:
            print(f"    FDS iter {it}: obj={c_new.max():.10f}, best={best_obj:.10f}")
    
    return best_f, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c_obj = -np.ones(n)
    # Build sparse-like constraint matrix
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """
    LP-based Frank-Wolfe optimization with adaptive M and golden section line search.
    """
    f = normalize_nonneg(f0.astype(np.float64))
    n = len(f)
    
    c = fast_convolve(f)
    M = c.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_multiplier = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        g = solve_lp_direction(f, M * m_multiplier, n)
        if g is None:
            m_multiplier *= 1.15
            if m_multiplier > 8.0:
                rng = np.random.default_rng(it * 19 + 3)
                f = normalize_nonneg(best_f + rng.standard_normal(n) * 0.003 / np.sqrt(n))
                c = fast_convolve(f)
                M = c.max()
                m_multiplier = 1.0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            m_multiplier *= 1.1
            continue
        g = g / g_sum
        
        # Check fixed point
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 100:
                m_multiplier = min(m_multiplier * 1.08, 5.0)
            if no_improve > 300:
                rng = np.random.default_rng(it * 23 + 11)
                f = normalize_nonneg(best_f + rng.standard_normal(n) * 0.002 / np.sqrt(n))
                c = fast_convolve(f)
                M = c.max()
                m_multiplier = 1.0
                no_improve = 0
            continue
        else:
            if np.max(np.abs(g - f)) < 1e-9:
                no_improve += 1
            else:
                no_improve = 0
        
        # Golden section line search
        inv_phi2 = (np.sqrt(5.0) - 1.0) / 2.0
        a_lo = 0.0
        a_hi = 1.0
        c_alpha = a_hi - inv_phi2 * (a_hi - a_lo)
        d_alpha = a_lo + inv_phi2 * (a_hi - a_lo)
        
        ft_c = normalize_nonneg(f + c_alpha * (g - f))
        ft_d = normalize_nonneg(f + d_alpha * (g - f))
        val_c = fast_convolve(ft_c).max()
        val_d = fast_convolve(ft_d).max()
        
        for _ in range(25):
            if val_c < val_d:
                a_hi = d_alpha
                d_alpha = c_alpha
                val_d = val_c
                c_alpha = a_hi - inv_phi2 * (a_hi - a_lo)
                ft_c = normalize_nonneg(f + c_alpha * (g - f))
                val_c = fast_convolve(ft_c).max()
            else:
                a_lo = c_alpha
                c_alpha = d_alpha
                val_c = val_d
                d_alpha = a_lo + inv_phi2 * (a_hi - a_lo)
                ft_d = normalize_nonneg(f + d_alpha * (g - f))
                val_d = fast_convolve(ft_d).max()
        
        best_alpha = 0.5 * (a_lo + a_hi)
        f_new = normalize_nonneg(f + best_alpha * (g - f))
        c_new = fast_convolve(f_new)
        M_new = c_new.max()
        
        if M_new < best_obj - 1e-15:
            f = f_new
            M = M_new
            best_obj = M_new
            best_f = f.copy()
            m_multiplier = 1.0
            if verbose and it % 20 == 0:
                print(f"    LP-FW iter {it}: obj={best_obj:.12f}")
        else:
            no_improve += 1
            if M_new < M + 1e-10:
                f = f_new
                M = M_new
            m_multiplier = min(m_multiplier * 1.03, 4.0)
        
        if no_improve > 250:
            break
    
    return best_f, best_obj


def subgradient_optimize(f0, time_limit, verbose=False):
    """
    Nesterov-accelerated subgradient method for the max convolution problem.
    """
    f = normalize_nonneg(f0.astype(np.float64))
    n = len(f)
    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    y = f.copy()
    momentum = 0.0
    lr = 0.01
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        momentum = (1 + momentum) / 2
        
        c_y = fast_convolve(y)
        k_max = c_y.argmax()
        
        # Subgradient: sg[j] = 2 * f[k_max - j] if 0 <= k_max - j < n
        sg = np.zeros(n)
        js = max(0, k_max - (n - 1))
        je = min(n - 1, k_max)
        if js <= je:
            sg[js:je+1] = 2.0 * f[k_max - np.arange(js, je + 1)]
        
        # Step
        f_new = y - lr * sg
        f_new = project_simplex(f_new)
        
        # Nesterov update
        y = f_new + ((momentum - 1) / momentum) * (f_new - f)
        y = project_simplex(y)
        f = f_new
        
        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            lr = min(lr * 1.01, 0.05)
        else:
            no_improve += 1
            if no_improve > 50:
                lr *= 0.9
        
        # Decay learning rate
        lr = max(lr * (1 - 1e-6 * it), 1e-6)
        
        if no_improve > 200:
            rng = np.random.default_rng(it * 41 + 7)
            f = normalize_nonneg(best_f + rng.standard_normal(n) * 0.001 / np.sqrt(n))
            y = f.copy()
            lr = 0.005
            no_improve = 0
        
        if verbose and it % 200 == 0:
            print(f"    SG iter {it}: obj={obj:.10f}, best={best_obj:.10f}")
    
    return best_f, best_obj


def multiplicative_update(f0, time_limit, verbose=False):
    """
    Multiplicative weight update targeting top peaks with adaptive rates.
    """
    f = normalize_nonneg(f0.astype(np.float64))
    n = len(f)
    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    lr = 0.005
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(f)
        M = c.max()
        
        # Top peaks with weights
        k_peaks = min(20, len(c))
        peak_idx = np.argpartition(c, -k_peaks)[-k_peaks:]
        peak_vals = c[peak_idx]
        excess = np.maximum(peak_vals - 0.9 * M, 0)
        if excess.sum() < 1e-15:
            excess = np.ones(len(peak_idx))
        w_peaks = excess / excess.sum()
        
        # Subgradient
        sg = np.zeros(n)
        for i in range(len(peak_idx)):
            if w_peaks[i] < 1e-12:
                continue
            k = int(peak_idx[i])
            js = max(0, k - (n - 1))
            je = min(n - 1, k)
            sg[js:je+1] += w_peaks[i] * 2.0 * f[k - np.arange(js, je + 1)]
        
        # Multiplicative update
        f = f * np.exp(-lr * sg)
        f = normalize_nonneg(f)
        
        c_new = fast_convolve(f)
        obj_new = c_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
            lr = min(lr * 1.02, 0.03)
        else:
            no_improve += 1
            lr *= 0.97
        
        if no_improve > 150:
            rng = np.random.default_rng(it * 31 + 17)
            f = normalize_nonneg(best_f + rng.standard_normal(n) * 0.002 / np.sqrt(n))
            lr = 0.004
            no_improve = 0
        
        if verbose and it % 100 == 0:
            print(f"    MU iter {it}: obj={c_new.max():.10f}, best={best_obj:.10f}")
        
        if lr < 1e-7:
            break
    
    return best_f, best_obj


def try_sequence(a, label, state):
    """Evaluate a sequence and update state if better."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < state['best_score']:
        state['best_score'] = score
        state['best_seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Diverse initialization
    Phase 2: Peak equalization (main new method)
    Phase 3: LP-FW for global refinement
    Phase 4: Multiplicative update + subgradient
    Phase 5: Multi-length exploration
    Phase 6: Final deep refinement
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    state = {'best_score': float('inf'), 'best_seq': None}
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            if p_score < state['best_score']:
                state['best_score'] = p_score
                state['best_seq'] = parent.copy()
    except Exception:
        pass
    
    if parent is None:
        parent = make_gaussian(768, 0.28)
        try_sequence(parent, "default_parent", state)
    
    n_parent = len(parent)
    
    # === Phase 1: Initialization Screening ===
    print("=== Phase 1: Initialization ===")
    t1 = time.monotonic()
    
    inits = []
    inits.append(("parent", parent.copy()))
    inits.append(("gauss_25", make_gaussian(n_parent, 0.25)))
    inits.append(("gauss_30", make_gaussian(n_parent, 0.30)))
    inits.append(("gauss_35", make_gaussian(n_parent, 0.35)))
    inits.append(("hann", make_hann(n_parent)))
    inits.append(("bell_5", make_bell(n_parent, 0.6, 5)))
    inits.append(("bell_7", make_bell(n_parent, 0.7, 7)))
    
    # Random perturbations
    for i in range(5):
        noise = rng.standard_normal(n_parent) * 0.02 / np.sqrt(n_parent)
        inits.append((f"pert{i}", normalize_nonneg(parent + noise)))
    
    best_init = None
    best_init_obj = float('inf')
    for name, a in inits:
        a = normalize_nonneg(a)
        c = fast_convolve(a)
        obj = 2 * n_parent * c.max()
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a.copy()
        try_sequence(a, f"init_{name}", state)
    
    print(f"  Best init obj: {best_init_obj:.8f}")
    print(f"  Phase 1 time: {time.monotonic()-t1:.1f}s")
    
    # === Phase 2: Peak Equalization ===
    print("\n=== Phase 2: Peak Equalization ===")
    rem = deadline - time.monotonic() - 800
    t_alloc = max(60, min(rem, 400))
    print(f"  Allocation: {t_alloc:.0f}s")
    
    a_pe, obj_pe = peak_equalization_optimize(best_init, time_limit=t_alloc, verbose=True)
    try_sequence(a_pe, "peak_eq", state)
    
    # Also try from parent if different
    if not np.allclose(best_init, parent, atol=1e-6):
        rem = deadline - time.monotonic() - 700
        if rem > 60:
            t_alloc2 = max(30, min(rem * 0.3, 150))
            a_pe2, _ = peak_equalization_optimize(parent, time_limit=t_alloc2)
            try_sequence(a_pe2, "peak_eq_parent", state)
    
    # Frequency domain shaping
    rem = deadline - time.monotonic() - 600
    if rem > 40:
        t_fds = min(rem * 0.2, 100)
        print(f"\n  Freq shaping: {t_fds:.0f}s")
        a_fs, _ = freq_domain_shaping(best_init, time_limit=t_fds)
        try_sequence(a_fs, "freq_shape", state)
    
    # === Phase 3: LP-FW ===
    print("\n=== Phase 3: LP-FW ===")
    if state['best_seq'] is not None:
        rem = deadline - time.monotonic() - 600
        if rem > 80:
            t_lpfw = max(60, min(rem * 0.5, 350))
            print(f"  Allocation: {t_lpfw:.0f}s")
            a_lw, _ = lp_fw_optimize(state['best_seq'], time_limit=t_lpfw, verbose=True)
            try_sequence(a_lw, "lpfw", state)
    
    # === Phase 4: Multiplicative + Subgradient ===
    print("\n=== Phase 4: Multiplicative + Subgradient ===")
    if state['best_seq'] is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 40:
            t_mu = max(20, min(rem * 0.3, 120))
            a_mu, _ = multiplicative_update(state['best_seq'], time_limit=t_mu)
            try_sequence(a_mu, "mult_up", state)
        
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_sg = max(20, min(rem * 0.3, 100))
            a_sg, _ = subgradient_optimize(state['best_seq'], time_limit=t_sg)
            try_sequence(a_sg, "subgrad", state)
    
    # === Phase 5: Multi-length Exploration ===
    print("\n=== Phase 5: Multi-Length Exploration ===")
    base = state['best_seq'] if state['best_seq'] is not None else best_init
    
    target_lengths = [1024, 896, 640]
    # Add some lengths near parent
    if n_parent not in target_lengths:
        target_lengths.append(n_parent + 64)
    
    target_lengths = [t for t in target_lengths if 256 <= t <= 2048]
    
    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 200
        if rem < 80:
            break
        
        f_interp = interpolate_sequence(base, target_n)
        try_sequence(f_interp, f"interp_n{target_n}", state)
        
        # Peak equalization on interpolated
        t_pe = min(rem * 0.3, 100)
        if t_pe > 20:
            a_opt, _ = peak_equalization_optimize(f_interp, time_limit=t_pe)
            try_sequence(a_opt, f"pe_n{target_n}", state)
        
        # LP-FW
        rem2 = deadline - time.monotonic() - 150
        if rem2 > 50:
            t_lw = min(rem2 * 0.5, 100)
            a_opt2, _ = lp_fw_optimize(a_opt, time_limit=t_lw)
            try_sequence(a_opt2, f"lpfw_n{target_n}", state)
    
    # === Phase 6: Deep Refinement ===
    print("\n=== Phase 6: Deep Refinement ===")
    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])
        
        # Multiple LP-FW passes with different restarts
        for p in range(3):
            rem = deadline - time.monotonic() - 60
            if rem < 40:
                break
            t_alloc = min(rem / (3 - p + 0.5), 150)
            
            if p == 0:
                f_init = state['best_seq'].copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.002 / np.sqrt(n_best)
                f_init = normalize_nonneg(state['best_seq'] + noise)
            else:
                # Blend with Hann
                h = make_hann(n_best)
                f_init = 0.8 * state['best_seq'] + 0.2 * h
                f_init = f_init / f_init.sum()
            
            print(f"  Deep LP-FW variant {p}: t={t_alloc:.0f}s")
            a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc, verbose=(p == 0))
            try_sequence(a_opt, f"deep_{p}", state)
        
        # Final peak equalization squeeze
        rem = deadline - time.monotonic() - 30
        if rem > 30:
            t_pe = min(rem, 80)
            a_opt, _ = peak_equalization_optimize(state['best_seq'], time_limit=t_pe, verbose=True)
            try_sequence(a_opt, "final_pe", state)
        
        # One more LP-FW
        rem = deadline - time.monotonic() - 10
        if rem > 20:
            a_opt, _ = lp_fw_optimize(state['best_seq'], time_limit=min(rem, 40))
            try_sequence(a_opt, "final_lpfw", state)
    
    # Fallback
    if state['best_seq'] is None:
        state['best_seq'] = np.ones(384) / 384
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(state['best_seq'].tolist()))
    if np.isfinite(fs):
        state['best_score'] = fs
    
    print(f"\nFINAL SCORE: {state['best_score']:.10f} (n={len(state['best_seq'])})")
    return [float(x) for x in state['best_seq']]
