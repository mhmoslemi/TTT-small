- Accelerated proximal gradient (FISTA) for faster convergence
- Softmax-weighted multi-peak subgradient computation
- Backtracking line search for adaptive step sizing
- Multi-length sweep with interpolation
- Targeted coordinate descent for final polishing
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_objective(a):
    """Compute 2*n*max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2 * n * conv.max() / (s * s))


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, top_k=8, temperature=0.01):
    """Compute weighted subgradient of max_k conv(a,a)[k] w.r.t. a."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    conv_sorted = np.sort(conv)[::-1]
    
    if conv_sorted[0] < 1e-15:
        return np.zeros(n)
    
    # Softmax weights over top peaks
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    # Subgradient: sg[m] = 2 * sum_j w_j * a[k_j - m]
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        # a[k-m] is valid when 0 <= k-m < n, i.e., k-n+1 <= m <= k
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je >= js:
            m_idx = np.arange(js, je + 1)
            a_vals = a[k - m_idx]
            sg[m_idx] += 2.0 * w[i] * a_vals
    
    return sg


def fista_optimize(a0, time_limit, top_k=8, temperature=0.01, verbose=False):
    """FISTA-accelerated proximal gradient for minimizing max conv(a,a)."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = compute_objective(a)
    
    t0 = time.monotonic()
    t_iter = 1.0  # FISTA momentum counter
    x_old = a.copy()
    eta = 1.0 / (2.0 * n ** 0.4)  # Initial step size
    no_improve = 0
    temp = temperature
    
    while time.monotonic() - t0 < time_limit:
        # FISTA momentum step
        momentum = (t_iter - 1) / (t_iter + 2)
        y = a + momentum * (a - x_old)
        y = np.maximum(y, 0)
        y = y / y.sum()
        
        # Compute subgradient at y
        conv_y = fast_convolve(y)
        sg = compute_subgradient(y, conv_y, top_k=top_k, temperature=temp)
        
        # Proximal step: project(y - eta * sg) onto simplex
        x_new = project_simplex(y - eta * sg)
        
        # Evaluate
        obj_new = fast_convolve(x_new).max()
        
        # Backtracking if objective increased too much
        if obj_new > best_obj * 1.05:
            eta *= 0.7
            x_new = project_simplex(y - eta * sg)
            obj_new = fast_convolve(x_new).max()
        
        a = x_new
        t_iter += 1
        temp *= 1.002  # Slowly increase temperature (consider more peaks)
        
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
            # Gradually increase step size if improving
            eta = min(eta * 1.02, 2.0 / (n ** 0.3))
            if verbose and int(t_iter) % 100 == 0:
                print(f"    FISTA t={int(t_iter)}: obj={best_obj:.10f}, eta={eta:.6f}")
        else:
            no_improve += 1
            if no_improve > 50:
                eta = max(eta * 0.8, 1e-6)
        
        # Periodic perturbation to escape local minima
        if no_improve > 200 and int(t_iter) % 50 == 0:
            rng = np.random.default_rng(int(t_iter) * 7 + 3)
            noise = rng.standard_normal(n) * 0.002
            a = project_simplex(best_a + noise)
            x_old = a.copy()
            t_iter = 1.0
            eta = 1.0 / (2.0 * n ** 0.4)
            no_improve = 0
        
        x_old = a.copy()
        
        if eta < 1e-8:
            break
    
    return best_a, best_obj


def multiplicative_descent(a0, time_limit, top_k=10, temp_init=0.005, verbose=False):
    """Multiplicative weight descent with adaptive step size."""
    a = np.maximum(a0.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = compute_objective(a)
    
    t0 = time.monotonic()
    eta = 0.5
    temp = temp_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        sg = compute_subgradient(a, conv, top_k=top_k, temperature=temp)
        
        # Multiplicative update
        a_new = a * np.exp(-eta * sg)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        obj_new = fast_convolve(a_new).max()
        
        if obj_new < M:
            a = a_new
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a.copy()
                no_improve = 0
                eta = min(eta * 1.01, 2.0)
            else:
                no_improve += 1
        else:
            eta *= 0.7
            no_improve += 1
        
        temp *= 1.003
        
        if no_improve > 100:
            rng = np.random.default_rng(it * 13 + 5)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
            eta = 0.5
            no_improve = 0
        
        if verbose and it % 100 == 0:
            print(f"    MD it={it}: obj={best_obj:.10f}, eta={eta:.4f}")
    
    return best_a, best_obj


def coordinate_descent_polish(a0, time_limit, verbose=False):
    """Targeted coordinate descent: adjust individual coordinates to reduce peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = compute_objective(a)
    t0 = time.monotonic()
    
    sweep = 0
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))
        
        # Find coordinates most contributing to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        if je < js:
            break
        
        idx = np.arange(js, je + 1)
        products = a[idx] * a[peak_k - idx]
        
        # Sort by contribution (largest first)
        order = np.argsort(products)[::-1]
        
        improved = False
        for oi in order[:min(20, len(order))]:
            j = int(idx[oi])
            if a[j] < 1e-10:
                continue
            
            # Try reducing a[j] and redistributing
            old_val = a[j]
            best_j_obj = best_obj
            best_j_val = old_val
            
            for frac in [0.5, 0.7, 0.85, 0.95, 1.1, 1.2, 1.5]:
                new_val = old_val * frac
                if new_val < 0 or new_val > 0.5:
                    continue
                trial = a.copy()
                trial[j] = new_val
                diff = old_val - new_val
                if abs(diff) > 1e-15:
                    # Redistribute to smallest elements
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    small_idx = np.argsort(a[mask])[:max(1, n // 4)]
                    mask_abs = np.zeros(n, dtype=bool)
                    mask_abs[small_idx + (small_idx >= j)] = True
                    mask_abs[j] = False
                    if mask_abs.sum() > 0:
                        trial[mask_abs] += diff * a[mask_abs] / a[mask_abs].sum()
                    else:
                        trial[mask] += diff * a[mask] / a[mask].sum()
                
                trial = np.maximum(trial, 0)
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                trial = trial / ts
                obj = fast_convolve(trial).max()
                
                if obj < best_j_obj - 1e-12:
                    best_j_obj = obj
                    best_j_val = new_val
                    best_trial = trial.copy()
            
            if best_j_val != old_val:
                a = best_trial
                if best_j_obj < best_obj - 1e-12:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            # Try peak shaving: scale down all contributors to peak
            trial = a.copy()
            trial[idx] *= 0.8
            diff = a.sum() - trial.sum()
            mask = np.ones(n, dtype=bool)
            mask[idx] = False
            if mask.sum() > 0:
                trial[mask] += diff * a[mask] / a[mask].sum()
            trial = np.maximum(trial, 0)
            trial = trial / trial.sum()
            obj = fast_convolve(trial).max()
            if obj < best_obj:
                a = trial
                best_obj = obj
                best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    from scipy.interpolate import CubicSpline
    old_x = np.linspace(0, 1, old_n)
    cs = CubicSpline(old_x, f, bc_type='natural')
    new_x = np.linspace(0, 1, new_n)
    f_interp = cs(new_x)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.3, center=0.5):
    """Gaussian initial sequence."""
    x = np.linspace(0, 1, n)
    sigma = sigma_frac
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width=0.7):
    """Bell-shaped initial sequence."""
    x = np.linspace(0, 1, n)
    t = np.abs(x - 0.5) * 2
    a = np.maximum(0, 1 - (t / width) ** 4)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_asymmetric(n, decay=3.0):
    """Asymmetric exponential initial sequence."""
    x = np.linspace(0, 1, n)
    a = np.exp(-decay * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_two_bump(n, pos1=0.3, pos2=0.7, sigma=0.1):
    """Two-bump initial sequence."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - pos1) / sigma) ** 2) + np.exp(-0.5 * ((x - pos2) / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a
