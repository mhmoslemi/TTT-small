   Let's check: $a^T Q_k a = (a*a)(k)$. For $k = 0$: $a^T Q_0 a = a_0^2$, so $Q_0$ has a 1 in the (0,0) position and 0 elsewhere. This is PSD.
   
   For $k = n-1$: $a^T Q_{n-1} a = \sum_i a_i a_{n-1-i} = \langle a, \text{rev}(a) \rangle$. The matrix $Q_{n-1}$ is the "anti-identity" matrix (reversal matrix). Its eigenvalues are $\pm 1$, so it's NOT PSD! This means the constraint $(a*a)(n-1) \leq \mu$ is NOT convex in general.

   Hmm, so the level set approach doesn't directly work as a convex problem.

   However, for the specific case where the maximum is at $k = n-1$ (which is true for "nice" sequences like the uniform one), we can use a different approach.

   Actually, let me reconsider. The problem is:
   Minimize $\max_k a^T Q_k a$ subject to $a \geq 0$, $\mathbf{1}^T a = 1$.

   The matrices $Q_k$ are NOT all PSD, so this is not a convex problem in general.

   But we can still use gradient-based methods. Let me focus on making the gradient-based approach more effective.

   **Key improvements I'll implement:**

   1. **Exact gradient computation** for the soft-max approximation using FFT-based convolution.
   2. **Adaptive step size with backtracking line search** to ensure progress.
   3. **Conjugate gradient direction** for faster convergence.
   4. **Temperature schedule** that starts moderate and decreases adaptively.
   5. **Multiple diverse starting points** including:
      - The parent sequence
      - Smooth random sequences
      - Sequences with specific structures (e.g., linearly varying amplitude)
      - Sequences from known good constructions (e.g., Barker-like)
   6. **Periodic "reheating"** to escape local minima.
   7. **Length exploration** with a focus on finding the optimal length.

   Let me also think about specific sequence structures that might be good:
   
   - **Linear ramp**: $a_i = (i+1)/\sum(j+1)$. This concentrates mass on one side.
   - **Quadratic**: $a_i = (i+1)^2 / \sum(j+1)^2$.
   - **Exponential**: $a_i = \exp(\alpha i) / \sum \exp(\alpha j)$ for various $\alpha$.
   - **Sine**: $a_i = \sin^2(\pi (i+0.5)/n) / \sum \sin^2(\pi (j+0.5)/n)$.
   - **Rectangular with taper**: Flat middle with tapered edges.

   The exponential ramp is particularly interesting because it creates a "triangle-like" auto-convolution with a shifted peak.

   Let me now code up the solution:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

- Conjugate gradient with backtracking line search on the simplex

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    next_pow2 = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, next_pow2)
    conv = np.fft.irfft(A * A, next_pow2)[:size]
    return np.maximum(conv, 0)  # Ensure non-negative (numerical)


def compute_gradient(a, conv, temp):
    """
    Compute gradient of soft-max approximation of max(conv).
    soft_max_tau(conv) = tau * log(sum(exp(conv_k / tau)))
    d(soft_max)/d(a_j) = sum_k w_k * 2 * a_{k-j}
    where w_k = exp(conv_k/tau) / sum exp(conv_j/tau)
    
    This equals 2 * correlate(w, a, 'valid')
    """
    n = len(a)
    # Compute softmax weights
    shifted = conv / temp
    shifted -= shifted.max()
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    
    # Gradient: 2 * correlate(w, a, 'valid')
    # correlate(w, a, 'valid')[j] = sum_i w[j+i] * a[i] for i=0..n-1
    # This gives sum_k w[k] * a[k-j] for k=j..j+n-1
    grad = 2.0 * np.correlate(w, a, mode='valid')[:n]
    return grad


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def line_search(f_cur, a, grad, temp, max_iter=30):
    """
    Backtracking line search on the simplex.
    Find step size alpha such that f(a - alpha * projected_grad) < f_cur - c * alpha * ||grad||^2
    """
    n = len(a)
    # Projected gradient direction
    # The projected gradient is: project(a) - project(a - grad)
    # But since a is already on the simplex, we just need to move in the -grad direction
    # and project back.
    
    # Normalize gradient for step size
    grad_norm = np.linalg.norm(grad)
    if grad_norm < 1e-15:
        return 0.0, a
    
    # Initial step size
    alpha = 1.0 / (grad_norm + 1e-10)
    c = 1e-4  # Armijo parameter
    
    for _ in range(max_iter):
        a_new = project_simplex(a - alpha * grad)
        conv_new = fast_convolve(a_new)
        f_new = temp * np.log(np.sum(np.exp((conv_new - conv_new.max()) / temp)) + 1e-300) + conv_new.max()
        # Actually, let's just use the soft-max value directly
        f_new = soft_max_value(conv_new, temp)
        
        if f_new <= f_cur - c * alpha * grad_norm * grad_norm:
            return alpha, a_new
        
        alpha *= 0.5
    
    return alpha, a


def soft_max_value(conv, temp):
    """Compute soft-max value."""
    shifted = conv / temp
    shifted -= shifted.max()
    return conv.max() + temp * np.log(np.sum(np.exp(shifted)))


def conjugate_gradient_optimize(f0, time_limit, temp_init=0.01, temp_min=0.001,
                                temp_decay=0.999, max_no_improve=200, verbose=False):
    """
    Conjugate gradient method with simplex projection for minimizing
    the soft-max approximation of max(auto-convolution).
    """
    a = f0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(f0)) / len(f0)
    else:
        a = a / s
    n = len(a)
    
    temp = temp_init
    best_obj = float('inf')
    best_a = a.copy()
    best_conv_max = fast_convolve(a).max()
    
    # Conjugate gradient state
    d = np.zeros(n)
    y_prev = np.zeros(n)
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        conv = fast_convolve(a)
        cur_conv_max = conv.max()
        f_cur = soft_max_value(conv, temp)
        
        # Compute gradient
        grad = compute_gradient(a, conv, temp)
        
        # Project gradient onto tangent space of simplex
        # The tangent space is {v : sum(v) = 0}
        grad_tangent = grad - np.mean(grad)
        
        # Fletcher-Reeves conjugate gradient
        if it > 1:
            y = grad_tangent - y_prev
            beta = np.dot(grad_tangent, grad_tangent) / (np.dot(y, y) + 1e-30)
            beta = max(0, min(beta, 5.0))  # Clip beta for stability
            d = -grad_tangent + beta * d
        else:
            d = -grad_tangent
        
        y_prev = grad_tangent.copy()
        
        # Line search
        # Find step in direction d
        # Parameterize: a_new = project_simplex(a + alpha * d)
        best_alpha = 0.0
        best_f = f_cur
        best_a_new = a.copy()
        
        # Try a range of step sizes
        alpha_init = 1.0 / (np.linalg.norm(d) + 1e-10)
        
        # Adaptive step sizes
        alphas = [alpha_init * 0.01, alpha_init * 0.05, alpha_init * 0.1, 
                  alpha_init * 0.5, alpha_init, alpha_init * 2.0, alpha_init * 5.0]
        
        for alpha in alphas:
            if alpha < 0:
                continue
            a_trial = project_simplex(a + alpha * d)
            conv_trial = fast_convolve(a_trial)
            f_trial = soft_max_value(conv_trial, temp)
            
            if f_trial < best_f:
                best_f = f_trial
                best_alpha = alpha
                best_a_new = a_trial
        
        if best_alpha > 0:
            a = best_a_new
            # Reset CG direction occasionally
            if it % 50 == 0:
                d = np.zeros(n)
        else:
            # No improvement, try steepest descent
            a_steep = project_simplex(a - alpha_init * 0.5 * grad)
            conv_steep = fast_convolve(a_steep)
            f_steep = soft_max_value(conv_steep, temp)
            if f_steep < f_cur:
                a = a_steep
            d = np.zeros(n)
        
        # Update best
        conv = fast_convolve(a)
        cur_max = conv.max()
        if cur_max < best_conv_max - 1e-12:
            best_conv_max = cur_max
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    CG iter {it}: max_conv={best_conv_max:.10f}, obj={2*len(a)*best_conv_max:.6f}")
        else:
            no_improve += 1
        
        # Temperature decay
        temp = max(temp * temp_decay, temp_min)
        
        # Reheating
        if no_improve > max_no_improve:
            rng = np.random.default_rng(it * 31 + 7)
            noise = rng.standard_normal(n) * 0.01
            a = project_simplex(best_a + noise)
            temp = min(temp_init * 1.5, temp_init * 3)
            d = np.zeros(n)
            no_improve = 0
        
        # Check time
        if time.monotonic() - t0 > time_limit:
            break
    
    return best_a, best_conv_max


def adam_optimize_v2(f0, time_limit, lr=0.005, beta1=0.9, beta2=0.999,
                     eps=1e-8, temp_init=0.008, temp_decay=0.9995,

    """Improved Adam with exact gradient and simplex projection."""
    a = f0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:

    v = np.zeros(n)
    t = 0
    temp = temp_init
    
    best_conv_max = fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(a)
        
        # Exact gradient via soft-max
        if top_k is not None and top_k < len(conv):
            # Use top-k for efficiency
            idx = np.argpartition(conv, -top_k)[-top_k:]
            w = np.zeros(len(conv))
            conv_top = conv[idx]
            shifted = conv_top / temp
            shifted -= shifted.max()
            w_top = np.exp(shifted)
            w_top /= w_top.sum()
            w[idx] = w_top
        else:
            grad = compute_gradient(a, conv, temp)
            # Normalize gradient (remove the factor of 2 since we accounted for it)
            grad = grad / 2.0  # We'll handle the scaling in the update
        
        # Actually, let me just compute it directly
        grad = compute_gradient(a, conv, temp)
        
        # Project gradient onto tangent space
        grad_tangent = grad - np.mean(grad)
        
        m = beta1 * m + (1 - beta1) * grad_tangent
        v = beta2 * v + (1 - beta2) * grad_tangent * grad_tangent
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new)
        
        temp = max(temp * temp_decay, 1e-7)
        cur_lr = max(lr * (0.99995 ** t), lr * 0.05)
        
        conv = fast_convolve(a)
        cur_max = conv.max()
        
        if cur_max < best_conv_max - 1e-12:

            best_a = a.copy()

        else:
            no_improve += 1
        
        if no_improve > 300:
            rng = np.random.default_rng(t * 17 + 3)

            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            temp = max(temp_init * 0.7, 0.003)
            cur_lr = lr
            no_improve = 0
    
    return best_a, best_conv_max


def make_exponential_init(n, alpha=0.5):
    """Exponential ramp initialization."""
    x = np.arange(n, dtype=np.float64)
    a = np.exp(alpha * x / n)
    return a / a.sum()


def make_gaussian_init(n, center=0.5, sigma=0.15):
    """Gaussian initialization."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_bell_curve_init(n, power=1.0):
    """Bell curve: sin^power(pi * x)."""
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_bimodal_init(n, sep=0.3):
    """Two Gaussian bumps."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - (0.5 - sep)) / 0.08) ** 2) + \
        np.exp(-0.5 * ((x - (0.5 + sep)) / 0.08) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def mirror_descent_v2(f0, time_limit, lr_init=0.02, lr_decay=0.999,
                      temp_init=0.008, temp_decay=0.9995, verbose=False):
    """Mirror descent with exact gradient."""
    a = f0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(f0)) / len(f0)
    else:
        a = a / s
    n = len(a)
    
    temp = temp_init
    cur_lr = lr_init
    
    best_conv_max = fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    t = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        conv = fast_convolve(a)
        grad = compute_gradient(a, conv, temp)
        grad_tangent = grad - np.mean(grad)
        
        # Multiplicative update with gradient
        log_update = -cur_lr * grad_tangent
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new /= a_new_sum
        a = a_new
        
        temp = max(temp * temp_decay, 1e-7)
        cur_lr = max(cur_lr * lr_decay, lr_init * 0.05)
        
        conv = fast_convolve(a)
        cur_max = conv.max()
        
        if cur_max < best_conv_max - 1e-12:
            best_conv_max = cur_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            rng = np.random.default_rng(t * 23 + 11)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            temp = max(temp_init * 0.6, 0.003)
            cur_lr = lr_init
            no_improve = 0
    
    return best_a, best_conv_max

"""

- Multiple optimization kernels (projected gradient with momentum, Adam, mirror descent)

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    next_pow2 = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, next_pow2)
    conv = np.fft.irfft(A * A, next_pow2)[:size]
    return np.maximum(conv, 0.0)


def compute_smgrad(a, conv, temp, top_k=None):
    """
    Exact gradient of soft-max objective:
    dF/da_j = 2 * sum_k w_k * a_{k-j}
    computed as 2 * correlate(w, a, 'valid')
    """
    n = len(a)
    c_len = len(conv)
    
    if top_k is not None and top_k < c_len:
        idx = np.argpartition(conv, -top_k)[-top_k:]
        w = np.zeros(c_len)
        cv = conv[idx]
        sh = cv / temp
        sh -= sh.max()
        wt = np.exp(sh)
        ws = wt.sum()
        if ws > 1e-300:
            w[idx] = wt / ws
        else:
            w[idx[np.argmax(cv)]] = 1.0
    else:
        w = np.zeros(c_len)
        sh = conv / temp
        sh -= sh.max()
        wt = np.exp(sh)
        ws = wt.sum()
        if ws > 1e-300:
            w = wt / ws
        else:
            w[np.argmax(conv)] = 1.0
    
    grad = 2.0 * np.correlate(w, a, mode='valid')[:n]
    return grad


def soft_max_val(conv, temp):
    """Soft-max value: tau * log(sum(exp(c_k/tau))) with numerical stability."""
    sh = conv / temp
    sh -= sh.max()
    return conv.max() + temp * np.log(np.sum(np.exp(sh)))


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

            conv = fast_convolve(a_new)
            grad = compute_smgrad(a_new, conv, 0.002)
            grad_tan = grad - np.mean(grad)
            lr_step = 0.002 / (np.linalg.norm(grad_tan) + 1e-10)
            a_new = project_simplex(a_new - lr_step * grad_tan)
        
        conv_new = fast_convolve(a_new)
        new_max = conv_new.max()
        
        # Accept/reject
        if new_max < cur_max or rng.random() < np.exp(-(new_max - cur_max) / t):
            a = a_new
            cur_max = new_max
            
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
        
        t = max(t * temp_decay, 1e-6)
    
    return best_a, best_max


def make_exponential_init(n, alpha=0.5):
    """Exponential ramp."""
    x = np.arange(n, dtype=np.float64)
    a = np.exp(alpha * x / (n - 1 + 1e-10))
    return a / a.sum()


def make_gaussian_init(n, center=0.5, sigma=0.15):
    """Gaussian bump."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_bell_init(n, power=1.0):
    """sin^power(pi*x) bell curve."""
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_bimodal_init(n, sep=0.25):
    """Two Gaussian bumps."""
    x = np.linspace(0, 1, n)
    a = (np.exp(-0.5 * ((x - (0.5 - sep)) / 0.08) ** 2) +
         np.exp(-0.5 * ((x - (0.5 + sep)) / 0.08) ** 2))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_smooth_random(n, rng, nc=8):
    """Sum of random Gaussians."""
    x = np.linspace(0, 1, n)
    a = np.zeros(n)
    for _ in range(nc):
        c = rng.uniform(0.1, 0.9)
        s = rng.uniform(0.05, 0.3)
        amp = rng.uniform(0.5, 2.0)
        a += amp * np.exp(-0.5 * ((x - c) / s) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, f)
    fi = np.maximum(fi, 0)
    ss = fi.sum()
    if ss > 1e-15:
        fi /= ss
    return fi


def active_set_polish(a0, time_limit, verbose=False):
    """
    Identify active constraints (conv values near max) and use
    scipy L-BFGS-B to minimize the max while keeping others in check.
    """
    a = a0.astype(np.float64).copy()
    a /= a.sum()
    n = len(a)
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_max = fast_convolve(a).max()
    
    for phase in range(10):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        
        # Active set: top values
        sorted_idx = np.argsort(conv)[::-1]
        active_count = max(2, min(len(conv) // 4, 20))
        active = sorted_idx[:active_count]
        
        # Define objective: maximize the minimum margin
        # i.e., minimize max_k (conv_k) subject to sum=1, a>=0
        # Use soft huber loss on the top values
        def obj(x):
            aa = np.maximum(x, 0.0)
            aa_s = aa.sum()
            if aa_s < 1e-15:
                return 1e10
            aa = aa / aa_s
            cc = fast_convolve(aa)
            # Soft max with very small temp
            sh = cc / 0.001
            sh -= sh.max()
            return cc.max() + 0.001 * np.log(np.sum(np.exp(sh)))
        
        # Use scipy L-BFGS-B
        rem = time_limit - (time.monotonic() - t0)
        max_iter = max(20, min(200, int(rem * 10)))
        
        try:
            res = scipy_minimize(obj, a, method='L-BFGS-B',
                                  bounds=[(0.0, 1.0)] * n,
                                  options={'maxiter': max_iter, 'ftol': 1e-15,
                                           'gtol': 1e-12, 'maxls': 20})
            a_new = np.maximum(res.x, 0.0)
            a_new_s = a_new.sum()
            if a_new_s > 1e-15:
                a_new /= a_new_s
                conv_new = fast_convolve(a_new)
                if conv_new.max() < best_max:
                    best_max = conv_new.max()
                    best_a = a_new.copy()
                    a = a_new.copy()
                    if verbose:
                        print(f"        Active-set phase {phase}: max={best_max:.12f}")
        except Exception:
            pass
    
    return best_a, best_max


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization:
    1. Exact gradient based methods (projected gradient, Adam, mirror descent)
    2. Simulated annealing for exploration
    3. Active set polishing with scipy
    4. Length exploration
    5. Diverse structural initializations
    """
    try:
        deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    except Exception:
        deadline = time.monotonic() + 1780.0
    
    best_seq = None
    best_score = float('inf')
    rng = np.random.default_rng(seed)
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return None
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return None
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return None
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={ps:.10f}")
            if np.isfinite(ps):
                best_score = ps
                best_seq = parent.copy()
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(512) / 512
    n_parent = len(parent)
    
    # === Phase 1: Projected gradient on parent ===
    print(f"\n=== Phase 1: Projected gradient (n={n_parent}) ===")
    t1_end = time.monotonic() + 200
    
    a_opt, _ = projected_gradient_optimize(parent, time_limit=min(80, max(10, t1_end - time.monotonic())),
                                             lr_init=0.008, temp_init=0.004, temp_decay=0.9994)
    try_seq(a_opt, "p1_pg")
    
    rem = t1_end - time.monotonic()
    if rem > 30:
        a_opt, _ = adam_exact(parent, time_limit=min(rem * 0.9, 80),
                               lr=0.004, temp_init=0.005, temp_decay=0.9994)
        try_seq(a_opt, "p1_adam")
    
    # === Phase 2: Diverse initializations ===
    print(f"\n=== Phase 2: Diverse inits (best={best_score:.8f}) ===")
    t2_end = time.monotonic() + 250
    
    inits = [
        ('exp_03', make_exponential_init(n_parent, 0.3)),
        ('exp_07', make_exponential_init(n_parent, 0.7)),
        ('exp_10', make_exponential_init(n_parent, 1.0)),
        ('gauss_02', make_gaussian_init(n_parent, 0.5, 0.2)),
        ('gauss_035', make_gaussian_init(n_parent, 0.5, 0.35)),
        ('bell_05', make_bell_init(n_parent, 0.5)),
        ('bell_2', make_bell_init(n_parent, 2.0)),
        ('bimodal', make_bimodal_init(n_parent, 0.2)),
        ('smooth_rnd', make_smooth_random(n_parent, rng, 10)),
    ]
    
    for name, init in inits:
        rem = t2_end - time.monotonic()
        if rem < 15:
            break
        t_alloc = min(rem * 0.55, 25)
        
        # Quick eval
        cv = fast_convolve(init)
        obj_est = 2 * n_parent * cv.max()
        if obj_est < best_score * 1.1:  # Only optimize if promising
            a_opt, _ = projected_gradient_optimize(init, time_limit=t_alloc,
                                                    lr_init=0.006, temp_init=0.004,
                                                    temp_decay=0.9993)
            try_seq(a_opt, f"p2_{name}")
        else:
            print(f"  Skip {name}: est_obj={obj_est:.4f} >= {best_score:.4f}")
    
    # === Phase 3: Deep optimization on best ===
    print(f"\n=== Phase 3: Deep optimization (best={best_score:.8f}) ===")
    t3_end = time.monotonic() + 350
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Adam with various hyperparams
        configs = [
            {'lr': 0.003, 'temp_init': 0.003, 'temp_decay': 0.9995},
            {'lr': 0.005, 'temp_init': 0.006, 'temp_decay': 0.9993},
            {'lr': 0.002, 'temp_init': 0.002, 'temp_decay': 0.9996},
        ]
        
        for ci, cfg in enumerate(configs):
            rem = t3_end - time.monotonic()
            if rem < 20:
                break
            a_opt, _ = adam_exact(best_seq, time_limit=min(rem * 0.6, 80), **cfg)
            try_seq(a_opt, f"p3_adam{ci}")
        
        # Mirror descent
        rem = t3_end - time.monotonic()
        if rem > 25:
            a_opt, _ = mirror_descent(best_seq, time_limit=min(rem * 0.5, 60),
                                       lr_init=0.012, temp_init=0.004, temp_decay=0.9993)
            try_seq(a_opt, "p3_mirror")
        
        # Annealing
        rem = t3_end - time.monotonic()
        if rem > 20:
            a_opt, _ = anneal_refine(best_seq, time_limit=min(rem * 0.6, 50))
            try_seq(a_opt, "p3_anneal")
    
    # === Phase 4: Length exploration ===
    print(f"\n=== Phase 4: Length exploration (best={best_score:.8f}) ===")
    t4_end = time.monotonic() + 200
    
    if best_seq is not None:
        n_base = len(best_seq)
        target_lengths = []
        for f in [0.7, 0.8, 0.9, 1.1, 1.25, 1.5, 2.0]:
            tl = int(n_base * f)
            if 128 <= tl <= 4096 and tl != n_base:
                target_lengths.append(tl)
        for tl in [256, 384, 512, 640, 768, 896, 1024, 1280, 1536, 2048]:
            if 128 <= tl <= 4096 and tl != n_base:
                target_lengths.append(tl)
        
        target_lengths = sorted(set(target_lengths), key=lambda x: abs(x - n_base))[:6]
        
        for tl in target_lengths:
            rem = t4_end - time.monotonic()
            if rem < 20:
                break
            fi = interpolate_seq(best_seq, tl)
            t_alloc = min(rem * 0.5, 35)
            a_opt, _ = projected_gradient_optimize(fi, time_limit=t_alloc,
                                                    lr_init=0.006, temp_init=0.004,
                                                    temp_decay=0.9993)
            try_seq(a_opt, f"p4_n{tl}")
    
    # === Phase 5: Active set polishing ===
    print(f"\n=== Phase 5: Active set polish (best={best_score:.8f}) ===")
    t5_end = time.monotonic() + 200
    
    if best_seq is not None:
        rem = t5_end - time.monotonic()
        if rem > 30:
            a_opt, _ = active_set_polish(best_seq, time_limit=min(rem * 0.7, 120), verbose=True)
            try_seq(a_opt, "p5_active")
        
        # Additional annealing + gradient
        rem = t5_end - time.monotonic()
        if rem > 20 and best_seq is not None:
            a_opt, _ = anneal_refine(best_seq, time_limit=min(rem * 0.6, 60))
            try_seq(a_opt, "p5_anneal2")
    
    # === Phase 6: Final fine-tuning ===
    print(f"\n=== Phase 6: Final fine-tune (best={best_score:.8f}) ===")
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Very fine Adam
        rem = deadline - time.monotonic() - 15
        if rem > 20:
            a_opt, _ = adam_exact(best_seq, time_limit=min(rem * 0.5, 60),
                                   lr=0.001, temp_init=0.002, temp_decay=0.9997,
                                   temp_min=0.0005)
            try_seq(a_opt, "p6_fine_adam")
        
        # Fine projected gradient
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            a_opt, _ = projected_gradient_optimize(best_seq, time_limit=min(rem * 0.7, 40),
                                                    lr_init=0.002, temp_init=0.002,
                                                    temp_decay=0.9997, temp_min=0.0005)
            try_seq(a_opt, "p6_fine_pg")
        
        # Tiny perturbations
        rem = deadline - time.monotonic() - 10
        for pi in range(5):
            rem = deadline - time.monotonic() - 8
            if rem < 8:
                break
            noise = rng.standard_normal(n_best) * 0.002
            fp = project_simplex(best_seq + noise)
            a_opt, _ = projected_gradient_optimize(fp, time_limit=min(rem * 0.4, 15),
                                                    lr_init=0.003, temp_init=0.002,
                                                    temp_decay=0.9995)
            try_seq(a_opt, f"p6_pert{pi}")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        try:
            best_score = float(evaluate_sequence(best_seq.tolist()))
        except Exception:
            best_score = 2.0
    
    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass
    
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
