LP-based multiplicative weight optimization for minimizing peak autocorrelation
of non-negative sequences. Uses the "Hedge" algorithm framework: at each step,
solve an LP to find the best direction, then take a convex combination step.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
   fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def normalize(a):
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s

def lp_direction(f, M):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k."""
    n = len(f)
    nc = 2 * n - 1
    
    # Build constraint matrix A (dense)
    A = np.zeros((nc, n))
    for k in range(nc):
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        for j in range(js, je + 1):
            A[k, j] = f[k - j]
    
    b = np.full(nc, M)
    c = -np.ones(n)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return np.ones(n) / n  # fallback

def lp_optimize(a0, time_limit, verbose=True):
    """Main LP-based optimization loop."""
    a = normalize(a0.copy())
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        M = b.max()
        
        if M < best_obj:
            best_obj = M
            best_a = a.copy()
        
        g = lp_direction(a, M)
        g = normalize(g)
        
        # Line search
        best_t = 0
        best_new_obj = best_obj
        for t in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
            a_new = (1 - t) * a + t * g
            a_new = np.maximum(a_new, 0)
            a_new = normalize(a_new)
            obj = fast_convolve(a_new).max()
            if obj < best_new_obj:
                best_new_obj = obj
                best_t = t
        
        if best_t == 0:
            # Try smaller steps
            for t in [0.001, 0.005]:
                a_new = (1 - t) * a + t * g
                a_new = np.maximum(a_new, 0)
                a_new = normalize(a_new)
                obj = fast_convolve(a_new).max()
                if obj < best_new_obj:
                    best_new_obj = obj
                    best_t = t

    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def normalize(a):
    """Normalize to probability vector, ensuring non-negativity."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def precompute_lp_indices(n):
    """Precompute sparse structure for LP constraint matrix."""
    nc = 2 * n - 1
    rows = []
    cols = []
    f_indices = []
    for k in range(nc):
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
            f_indices.append(k - j)
    return np.array(rows), np.array(cols), np.array(f_indices)


def build_A_sparse(f, precomputed):
    """Build the LP constraint matrix from precomputed structure."""
    rows, cols, f_indices = precomputed
    n = len(f)
    nc = 2 * n - 1
    vals = f[f_indices]
    return csr_matrix((vals, (rows, cols)), shape=(nc, n))


def lp_direction(f, M, precomputed):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k."""
    n = len(f)
    A = build_A_sparse(f, precomputed)
    nc = 2 * n - 1
    b = np.full(nc, M)
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
        if res.success:
            g = res.x
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return f.copy()


def lp_optimize(a0, time_limit):
    """Main LP-based multiplicative weight optimization."""
    a = normalize(a0.astype(np.float64).copy())
    n = len(a)
    precomputed = precompute_lp_indices(n)
    best_a = a.copy()
    best_obj = float(fast_convolve(a).max())
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        M = float(b.max())
        if M < best_obj - 1e-15:
            best_obj = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            break

        g = lp_direction(a, M, precomputed)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Line search for step size
        best_t = 0.0
        best_new = best_obj
        for t in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            a_new = normalize((1.0 - t) * a + t * g)
            obj = float(fast_convolve(a_new).max())
            if obj < best_new:
                best_new = obj
                best_t = t

        if best_t == 0.0:
            # Try even smaller steps
            for t in [0.001, 0.003, 0.005]:
                a_new = normalize((1.0 - t) * a + t * g)
                obj = float(fast_convolve(a_new).max())
                if obj < best_new:
                    best_new = obj
                    best_t = t

        if best_t == 0.0:
            break

        a = normalize((1.0 - best_t) * a + best_t * g)

        if it % 100 == 0 or it < 5:
            elapsed = time.monotonic() - t0
            print(f"  LP iter {it}: obj={best_obj:.10f}, t={best_t:.4f}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def cross_corr_fft(w, a):
    """Cross-correlation via FFT for gradient computation."""
    n = len(a)
    size = n + len(w) - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:n]


def softmax_from_z(z):
    """Convert log-space z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def z_from_a(a):
    """Convert probability vector to log-space."""
    return np.log(np.maximum(a, 1e-300))


def soft_max_obj_and_grad(z, T):
    """Compute soft-max smoothed objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    n = len(a)
    b = fast_convolve(a)

    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp

    w = exp_Tb / exp_Tb.sum()
    grad_a = 2.0 * n * cross_corr_fft(w, a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)

    return float(obj), grad_z


def lbfgs_optimize_z(z0, T, time_limit, maxiter=5000):
    """Run L-BFGS in log-space with time limit."""
    t0 = time.monotonic()

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return soft_max_obj_and_grad(z, T)[0]

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return soft_max_obj_and_grad(z, T)[1]

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        return res.x
    except Exception:
        return z0

def greedy_subgradient_optimize(a0, time_limit, top_k=50):
    """Greedy subgradient method with top-k peak weighting."""
    a = normalize(a0.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_obj = float(fast_convolve(a).max())
    t0 = time.monotonic()
    v = np.zeros(n)
    cur_lr = 0.01
    cur_temp = 0.0005
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        v = 0.9 * v - cur_lr * sg
        a = project_simplex(a + v)
        co = float(fast_convolve(a).max())

        if co < best_obj - 1e-14:
            best_obj = co
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9995
        cur_temp *= 1.005

        if no_improve > 500:
            break

    return best_a, best_obj


def multiplicative_weight_optimize(a0, time_limit, top_k=20):
    """Pure multiplicative weight update (no LP, greedy subgradient)."""
    a = normalize(a0.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_obj = float(fast_convolve(a).max())
    t0 = time.monotonic()
    z = np.zeros(n)  # log-weights
    eta = 0.001

    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = float(b.max())
        if M < best_obj - 1e-15:
            best_obj = M
            best_a = a.copy()

        # Compute weighted subgradient from top peaks
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        # Soft weights
        temp = max(0.0001, (pv.max() - pv.min()) / 5.0)
        scaled = (pv - pv.min()) / temp
        w = np.exp(scaled)
        w /= w.sum()

        # Subgradient: g[j] = sum_k w_k * a[k-j] (valid)
        g = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-10:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w[i] * a[k - idx]

        # Multiplicative weight update in log-space
        z = z - eta * g
        a = softmax_from_z(z)
        a = normalize(a)

        if (it + 1) % 5000 == 0:
            eta *= 0.9

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: LP-based Hedge method, L-BFGS polishing,
    multiplicative weight updates, and multi-restart.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)
    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        parent = normalize(parent)
        p_score = try_seq(parent, "parent")
        print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        parent = np.ones(768) / 768

    n_parent = len(parent)

    # === Phase 1: LP-based optimization (main workhorse) ===
    rem = deadline - time.monotonic()
    t1 = rem * 0.55
    print(f"\nPhase 1: LP-based Hedge, n={n_parent}, budget={t1:.0f}s")
    a_lp, obj_lp = lp_optimize(parent, t1)
    try_seq(a_lp, f"lp_main")

    # === Phase 2: L-BFGS polishing on best ===
    base = best_seq if best_seq is not None else a_lp
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.25, 300.0)
    if t2 > 20:
        print(f"\nPhase 2: L-BFGS polish, n={len(base)}, budget={t2:.0f}s")
        a_pol, _ = lbfgs_polish(base, t2)
        try_seq(a_pol, "lbfgs_polish")

    # === Phase 3: Subgradient / multiplicative weight ===
    base = best_seq if best_seq is not None else a_lp
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.2, 200.0)
    if t3 > 20:
        print(f"\nPhase 3: Subgradient, n={len(base)}, budget={t3:.0f}s")
        a_sg, _ = greedy_subgradient_optimize(base, t3, top_k=max(20, len(base) // 10))
        try_seq(a_sg, "subgrad")

    # === Phase 4: Multi-restart with different initializations ===
    rem = deadline - time.monotonic()
    print(f"\nPhase 4: Multi-restart, budget={rem:.0f}s")
    if rem > 60:
        # Try different starting shapes
        n = len(best_seq) if best_seq is not None else n_parent
        starts = []

        # U-shaped
        x = np.linspace(0, 1, n)
        u_shape = (x - 0.5) ** 2 + 0.1
        u_shape = np.maximum(u_shape, 0.01)
        starts.append((normalize(u_shape), "u_shape"))

        # V-shaped (high in middle)
        v_shape = 1.0 - (x - 0.5) ** 2
        v_shape = np.maximum(v_shape, 0.01)
        starts.append((normalize(v_shape), "v_shape"))

        # Sawtooth
        saw = np.array([1.0 if (i % 4) < 2 else 0.3 for i in range(n)])
        starts.append((normalize(saw), "sawtooth"))

        # Ramp
        ramp = np.linspace(0.1, 1.0, n)
        starts.append((normalize(ramp), "ramp"))

        # Random
        rand = rng.uniform(0.1, 1.0, n)
        starts.append((normalize(rand), "random"))

        # Perturbation of best
        if best_seq is not None:
            pert = best_seq + rng.standard_normal(n) * 0.01
            starts.append((normalize(pert), "perturb"))

        for start_a, label in starts:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_alloc = min(rem * 0.3, 80.0)
            if t_alloc < 10:
                break
            print(f"  Restart [{label}], t={t_alloc:.0f}s")
            a_r, _ = lp_optimize(start_a, t_alloc)
            try_seq(a_r, f"restart_{label}")

    # === Phase 5: Final polish on best ===
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 15:
            print(f"\nPhase 5: Final polish, budget={rem:.0f}s")
            t5 = min(rem * 0.5, 60.0)
            a_final, _ = lbfgs_polish(best_seq, t5)
            try_seq(a_final, "final_polish")

            # Try subgradient on best
            rem = deadline - time.monotonic()
            if rem > 10:
                t5b = min(rem * 0.5, 30.0)
                a_sg2, _ = greedy_subgradient_optimize(best_seq, t5b, top_k=30)
                try_seq(a_sg2, "final_subgrad")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
