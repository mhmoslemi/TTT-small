"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using exact coordinate descent with
1D quadratic subproblem solving, Frank-Wolfe escaping, and multi-scale
refinement. Key innovation: each coordinate update exactly minimizes the
max-convolution objective in 1D by exploiting the quadratic structure of
conv_k as a function of a single coordinate (under simplex-prescaling),
avoiding soft-max approximation errors entirely.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_conv_quadratic_coeffs(a, j):
    """
    For coordinate j, compute coefficients such that if we set a_j = x and
    rescale others to maintain sum=1, then conv_k(x) = A_k*x^2 + B_k*x + C_k.
    
    Returns arrays A, B, C of length 2*n-1.
    """
    n = len(a)
    s = 1.0 - a[j]  # sum of all other elements
    if s < 1e-15:
        return None, None, None

    conv_full = fast_convolve(a)
    
    # For each k, determine if j is in valid range [lo_k, hi_k]
    # lo_k = max(0, k - (n-1)), hi_k = min(n-1, k)
    ks = np.arange(2 * n - 1)
    lo = np.maximum(0, ks - (n - 1))
    hi = np.minimum(n - 1, ks)
    
    # j is in range for k if lo_k <= j <= hi_k
    j_in_range = (lo <= j) & (j <= hi)
    
    # k-j must be in [0, n-1] for the cross-term to exist
    k_minus_j = ks - j
    k_minus_j_valid = (k_minus_j >= 0) & (k_minus_j < n)
    
    # E_k = 1 if k=2j and j in range
    E = np.zeros(2 * n - 1)
    if 0 <= 2 * j < 2 * n - 1:
        E[2 * j] = 1.0 if j_in_range[2 * j] else 0.0
    
    # F_k = 2*a_{k-j} if j in range and k-j valid and k != 2j
    F = np.zeros(2 * n - 1)
    mask = j_in_range & k_minus_j_valid & (ks != 2 * j)
    F[mask] = 2.0 * a[k_minus_j[mask].astype(int)]
    
    # G_k = conv_k - terms involving a_j
    # Terms involving a_j: if j in range and k != 2j: 2*a_j*a_{k-j}
    #                      if k = 2j and j in range: a_j^2
    G = conv_full.copy()
    if j_in_range.any():
        # Subtract 2*a_j*a_{k-j} where applicable
        mask_cross = j_in_range & k_minus_j_valid & (ks != 2 * j)
        G[mask_cross] -= 2.0 * a[j] * a[k_minus_j[mask_cross].astype(int)]
        # Subtract a_j^2 for k=2j
        if 0 <= 2 * j < 2 * n - 1 and j_in_range[2 * j]:
            G[2 * j] -= a[j] * a[j]
    
    # Compute A, B, C
    s2 = s * s
    A = E - F / s + G / s2
    B = F / s - 2.0 * G / s2
    C = G / s2
    
    return A, B, C


def eval_quadratic_max(A, B, C, x):
    """Evaluate max_k (A_k*x^2 + B_k*x + C_k)."""
    vals = A * x * x + B * x + C
    return np.max(vals)


def solve_1d_subproblem(A, B, C, x_current):
    """
    Solve min_{x in [0,1]} max_k (A_k*x^2 + B_k*x + C_k)
    using iterative active-set refinement.
    """
    x = np.clip(x_current, 0.0, 1.0)
    
    for _ in range(30):
        # Evaluate at current x
        vals = A * x * x + B * x + C
        f_val = np.max(vals)
        
        # Find active set (top few that are close to max)
        threshold = f_val * 0.999
        active = np.where(vals >= threshold)[0]
        
        if len(active) == 0:
            break
        
        # Try vertex of each active quadratic
        best_x = x
        best_f = f_val
        
        # Check boundaries
        for x_trial in [0.0, 1.0]:
            ft = eval_quadratic_max(A, B, C, x_trial)
            if ft < best_f - 1e-15:
                best_f = ft
                best_x = x_trial
        
        # Check vertices of active quadratics
        for k in active[:20]:  # Limit to top 20 for speed
            if A[k] > 1e-15:
                xv = -B[k] / (2.0 * A[k])
                xv = np.clip(xv, 0.0, 1.0)
                ft = eval_quadratic_max(A, B, C, xv)
                if ft < best_f - 1e-15:
                    best_f = ft
                    best_x = xv
        
        # Check pairwise intersections of top active quadratics
        if len(active) >= 2:
            top = active[:10]
            for i in range(len(top)):
                for jj in range(i + 1, len(top)):
                    k1, k2 = top[i], top[jj]
                    # A1*x^2 + B1*x + C1 = A2*x^2 + B2*x + C2
                    # (A1-A2)*x^2 + (B1-B2)*x + (C1-C2) = 0
                    da = A[k1] - A[k2]
                    db = B[k1] - B[k2]
                    dc = C[k1] - C[k2]
                    if abs(da) > 1e-15:
                        disc = db * db - 4.0 * da * dc
                        if disc >= 0:
                            sqrt_disc = np.sqrt(disc)
                            x1 = (-db + sqrt_disc) / (2.0 * da)
                            x2 = (-db - sqrt_disc) / (2.0 * da)
                            for xc in [x1, x2]:
                                if 0.0 <= xc <= 1.0:
                                    ft = eval_quadratic_max(A, B, C, xc)
                                    if ft < best_f - 1e-15:
                                        best_f = ft
                                        best_x = xc
                    elif abs(db) > 1e-15:
                        xc = -dc / db
                        if 0.0 <= xc <= 1.0:
                            ft = eval_quadratic_max(A, B, C, xc)
                            if ft < best_f - 1e-15:
                                best_f = ft
                                best_x = xc
        
        if abs(best_x - x) < 1e-10:
            break
        x = best_x
    
    return x, best_f


def coordinate_descent(a, n_sweeps=50, random_order=False, seed=42, time_limit=60.0):
    """
    Coordinate descent with exact 1D subproblem solving.
    a is normalized to sum=1.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_conv_max = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    order = np.arange(n)
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        if random_order:
            order = rng.permutation(n)
        
        improved = False
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            s = 1.0 - a[j]
            if s < 1e-12:
                continue
            
            A, B, C = compute_conv_quadratic_coeffs(a, j)
            if A is None:
                continue
            
            x_new, f_new = solve_1d_subproblem(A, B, C, a[j])
            
            if abs(x_new - a[j]) > 1e-12:
                # Apply update: set a_j = x_new, rescale others
                a_new = a.copy()
                a_new[j] = x_new
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_new[mask] = a[mask] * (1.0 - x_new) / s
                a_new = a_new / a_new.sum()
                
                conv_new = fast_convolve(a_new)
                conv_max = conv_new.max()
                
                if conv_max < best_conv_max - 1e-14:
                    best_conv_max = conv_max
                    best_a = a_new.copy()
                a = a_new
                improved = True
    
        if not improved:
            break
    
    return best_a, best_conv_max


def frank_wolfe(a_init, n_iters=200, time_limit=30.0, top_k=20):
    """
    Frank-Wolfe (conditional gradient) method for min max_k conv_k(a)
    subject to a in simplex.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_conv_max = fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        f_val = conv.max()
        
        # Find active peaks
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        # Soft-max weights
        scaled = pv / 0.01
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        # Compute subgradient
        g = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            other_idx = k - idx
            other_idx = other_idx[(other_idx >= 0) & (other_idx < n)]
            valid_idx = idx[(k - idx >= 0) & (k - idx < n)]
            g[valid_idx] += 2.0 * w[i] * a[k - valid_idx]
        
        # FW step: s = e_{argmin g}
        min_idx = np.argmin(g)
        s = np.zeros(n)
        min_val = a[min_val] if False else 0.0  # s is just e_{min_idx}
        s = np.zeros(n)
        s[min_idx] = 1.0
        
        # Line search: minimize f((1-t)*a + t*s) for t in [0,1]
        # f(t) = max_k conv_k((1-t)*a + t*s)
        best_t = 2.0 / (it + 2.0)  # Default FW step size
        best_f = f_val
        
        # Try a few t values
        for t_trial in np.linspace(0.1, 0.9, 17):
            a_trial = (1.0 - t_trial) * a + t_trial * s
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = a_trial / a_trial.sum()
            ft = fast_convolve(a_trial).max()
            if ft < best_f:
                best_f = ft
                best_t = t_trial
        
        a = (1.0 - best_t) * a + best_t * s
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        
        conv = fast_convolve(a)
        f_val = conv.max()
        if f_val < best_conv_max:
            best_conv_max = f_val
            best_a = a.copy()
    
    return best_a, best_conv_max


def mirror_descent(a_init, n_iters=500, time_limit=30.0, base_lr=0.01, top_k=10):
    """
    Multiplicative weight / mirror descent with entropy regularization.
    Update: a_i <- a_i * exp(-lr * g_i) / Z
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_conv_max = fast_convolve(a).max()
    t0 = time.monotonic()
    
    lr = base_lr
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        f_val = conv.max()
        
        # Subgradient with softmax weights on peaks
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / 0.005
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        g = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            other = k - idx
            mask = (other >= 0) & (other < n)
            valid_idx = idx[mask]
            g[valid_idx] += 2.0 * w[i] * a[other[mask]]
        
        # Normalize gradient
        g_max = np.max(np.abs(g))
        if g_max > 1e-15:
            g = g / g_max
        
        # Multiplicative update
        a_new = a * np.exp(-lr * g)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        # Check improvement
        conv_new = fast_convolve(a_new)
        f_new = conv_new.max()
        
        if f_new < f_val:
            a = a_new
            if f_new < best_conv_max:
                best_conv_max = f_new
                best_a = a.copy()
            lr = min(lr * 1.01, 0.5)
        else:
            lr *= 0.9
            if lr < 1e-5:
                # Reset with perturbation
                rng = np.random.default_rng(it)
                a = best_a + rng.standard_normal(n) * 0.001
                a = np.maximum(a, 0)
                a = a / a.sum()
                lr = base_lr
        
        # Decay lr slowly
        lr = max(lr * 0.999, 1e-5)
    
    return best_a, best_conv_max


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian_init(n, sigma_frac=0.3):
    """Gaussian-shaped initialization."""
    x = np.linspace(-1, 1, n)
    sigma = sigma_frac
    a = np.exp(-x**2 / (2 * sigma**2))
    a = a / a.sum()
    return a


def make_cosine_init(n):
    """Cosine window initialization."""
    x = np.linspace(0, np.pi, n)
    a = np.cos(x / 2) ** 2
    a = a / a.sum()
    return a


def make_symmetric_init(n, base):
    """Make a sequence symmetric (palindrome)."""
    a = base.copy()
    a_sym = (a + a[::-1]) / 2.0
    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Try diverse initializations at parent length
    2. Coordinate descent (main workhorse)
    3. Frank-Wolfe for escaping
    4. Mirror descent for fine-tuning
    5. Multi-length exploration
    6. Final polishing
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        # Normalize for evaluation (score is scale-invariant)
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(512) / 512
    
    n_parent = len(parent)
    
    # === Phase 1: Diverse initializations at parent length ===
    print("\n=== Phase 1: Diverse initializations ===")
    t1_budget = min(60.0, (deadline - time.monotonic()) * 0.1)
    
    candidates = []
    candidates.append(("parent", parent.copy()))
    candidates.append(("gauss02", make_gaussian_init(n_parent, 0.2)))
    candidates.append(("gauss03", make_gaussian_init(n_parent, 0.3)))
    candidates.append(("gauss04", make_gaussian_init(n_parent, 0.4)))
    candidates.append(("cosine", make_cosine_init(n_parent)))
    candidates.append(("sym_parent", make_symmetric_init(n_parent, parent)))
    candidates.append(("rand", rng.dirichlet(np.ones(n_parent))))
    candidates.append(("rand2", rng.dirichlet(np.ones(n_parent) * 0.5)))
    
    for name, cand in candidates:
        if time.monotonic() > deadline - t1_budget:
            break
        try_seq(cand, f"init_{name}")
    
    base = best_seq if best_seq is not None else parent
    
    # === Phase 2: Coordinate descent (main phase) ===
    print(f"\n=== Phase 2: Coordinate descent (best={best_score:.8f}) ===")
    
    remaining = deadline - time.monotonic()
    cd_budget = min(remaining * 0.4, 600.0)
    
    # First pass: many sweeps on best
    a_cd, _ = coordinate_descent(base, n_sweeps=200, random_order=False,
                                  seed=42, time_limit=cd_budget * 0.5)
    try_seq(a_cd, "cd_pass1")
    
    # Second pass: random order for exploration
    if time.monotonic() < deadline - 200:
        a_cd2, _ = coordinate_descent(base, n_sweeps=100, random_order=True,
                                       seed=123, time_limit=cd_budget * 0.3)
        try_seq(a_cd2, "cd_pass2_random")
    
    # Continue from best with more sweeps
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_cd3, _ = coordinate_descent(best_seq, n_sweeps=200, random_order=False,
                                       seed=7, time_limit=cd_budget * 0.2)
        try_seq(a_cd3, "cd_pass3_refine")
    
    # === Phase 3: Frank-Wolfe ===
    print(f"\n=== Phase 3: Frank-Wolfe (best={best_score:.8f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        fw_budget = min(60.0, (deadline - time.monotonic()) * 0.15)
        a_fw, _ = frank_wolfe(best_seq, n_iters=300, time_limit=fw_budget, top_k=20)
        try_seq(a_fw, "fw")
        
        # Coordinate descent after FW
        if time.monotonic() < deadline - 150:
            a_cd4, _ = coordinate_descent(a_fw, n_sweeps=100, random_order=False,
                                           seed=99, time_limit=30.0)
            try_seq(a_cd4, "cd_after_fw")
    
    # === Phase 4: Mirror descent ===
    print(f"\n=== Phase 4: Mirror descent (best={best_score:.8f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        md_budget = min(60.0, (deadline - time.monotonic()) * 0.15)
        a_md, _ = mirror_descent(best_seq, n_iters=1000, time_limit=md_budget,
                                  base_lr=0.02, top_k=10)
        try_seq(a_md, "mirror")
        
        # CD after mirror
        if time.monotonic() < deadline - 100:
            a_cd5, _ = coordinate_descent(a_md, n_sweeps=100, random_order=False,
                                           seed=55, time_limit=30.0)
            try_seq(a_cd5, "cd_after_md")
    
    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length exploration (best={best_score:.8f}) ===")
    
    base_for_len = best_seq if best_seq is not None else parent
    target_lengths = sorted(set([
        max(256, n_parent // 2),
        n_parent,
        n_parent * 2,
        min(4096, n_parent * 4),
    ]))
    
    # Remove duplicates and cap
    target_lengths = [t for t in target_lengths if 64 <= t <= 8192]
    
    for target_n in target_lengths:
        if time.monotonic() > deadline - 100:
            break
        if target_n == n_parent and best_seq is not None:
            continue  # Already optimized at this length
        
        f_interp = interpolate_sequence(base_for_len, target_n)
        
        # Quick CD
        t_alloc = min(60.0, (deadline - time.monotonic()) * 0.2)
        if t_alloc < 15:
            continue
        
        print(f"  CD at n={target_n}, budget={t_alloc:.0f}s")
        a_len, _ = coordinate_descent(f_interp, n_sweeps=100, random_order=False,
                                       seed=target_n % 1000, time_limit=t_alloc)
        try_seq(a_len, f"cd_n{target_n}")
        
        if best_seq is not None and time.monotonic() < deadline - 60:
            # Further refine
            a_len2, _ = coordinate_descent(best_seq, n_sweeps=50, random_order=False,
                                            seed=target_n % 1000 + 1, time_limit=30.0)
            try_seq(a_len2, f"cd_n{target_n}_refine")
    
    # === Phase 6: Symmetrization attempts ===
    print(f"\n=== Phase 6: Symmetrization (best={best_score:.8f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 80:
        a_sym = make_symmetric_init(len(best_seq), best_seq)
        a_sym_cd, _ = coordinate_descent(a_sym, n_sweeps=150, random_order=False,
                                          seed=777, time_limit=45.0)
        try_seq(a_sym_cd, "sym_cd")
    
    # === Phase 7: Final polishing ===
    print(f"\n=== Phase 7: Final polishing (best={best_score:.8f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic() - 20
        t_final = min(remaining * 0.6, 90.0)
        
        # Long CD run
        a_final, _ = coordinate_descent(best_seq, n_sweeps=500, random_order=False,
                                         seed=2024, time_limit=t_final)
        try_seq(a_final, "final_cd")
        
        # Random perturbation + CD
        if time.monotonic() < deadline - 20:
            for attempt in range(3):
                if time.monotonic() > deadline - 15:
                    break
                noise = rng.standard_normal(len(best_seq)) * 0.005
                a_pert = best_seq + noise
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
                a_pert_cd, _ = coordinate_descent(a_pert, n_sweeps=50,
                                                   random_order=False,
                                                   seed=attempt + 100,
                                                   time_limit=15.0)
                try_seq(a_pert_cd, f"perturb_cd_{attempt}")
    
    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        # Use parent as fallback
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
