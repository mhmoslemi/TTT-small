"""
Optimizer minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def power_sum_obj(c, p):
    """Compute (sum c^p)^(1/p) and gradient contribution."""
    c_clipped = np.maximum(c, 0)
    vals = c_clipped ** p
    s = vals.sum()
    if s < 1e-300:
        return 0.0, np.zeros_like(c)
    obj = s ** (1.0 / p)
    # d/dc_k of (sum c^p)^(1/p) = c_k^(p-1) / (sum c^p)^((p-1)/p)
    grad_c = c_clipped ** (p - 1) / (s ** ((p - 1.0) / p))
    return obj, grad_c

def power_sum_grad(a, p):
    """Compute gradient of (sum conv(a,a)^p)^(1/p) w.r.t. a."""
    n = len(a)
    c = fast_convolve(a)
    vals = np.maximum(c, 0) ** p
    s = vals.sum()
    if s < 1e-300:
        return np.zeros(n), 0.0
    obj = s ** (1.0 / p)
    # d/dc_k = c_k^(p-1) / s^((p-1)/p)
    w = np.maximum(c, 0) ** (p - 1) / (s ** ((p - 1.0) / p))
    # d/d a_i of c_k = 2 * a[k-i] (if valid)
    # So grad_i = 2 * sum_k w[k] * a[k-i]
    # This is 2 * (w * a_reversed) correlation
    N = 1 << (3 * n - 1).bit_length()
    A_fft = np.fft.rfft(a, N)
    W_fft = np.fft.rfft(w, N)
    # We need: sum_k w[k] * a[k-i] for i = 0..n-1
    # a[k-i] is valid when 0 <= k-i <= n-1, i.e., i <= k <= i+n-1
    # This is cross-correlation of w and a
    # corr[i] = sum_k w[k] * a[k-i]
    # Using FFT: corr = ifft(FFT(w) * conj(FFT(a)))
    corr = np.fft.irfft(W_fft * np.conj(A_fft), N)[:n]
    grad = 2.0 * corr
    return grad, obj

def nesterov_smooth_optimize(f0, time_limit, p_start=30, p_end=200, verbose=True):
    """Optimize using power-sum smooth with Nesterov acceleration."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    y = f.copy()  # Nesterov momentum variable
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = elapsed / time_limit
        
        # Anneal p
        p = p_start * (p_end / p_start) ** progress
        # Anneal learning rate
        lr = 2.0 / (it + 10)  # Nesterov standard
        lr = min(lr, 0.05)
        
        grad, smooth_obj = power_sum_grad(y, p)
        
        # Project gradient to maintain sum=1
        grad_centered = grad - grad.mean()
        
        # Nesterov update
        y = f - lr * grad_centered
        y = project_simplex(y)
        
        f_new = y - (lr / 2.0) * (grad_centered - 0)  # simplified Nesterov
        # Actually, standard Nesterov: f_{k+1} = y_k - lr * grad(y_k)
        # y_{k+1} = f_{k+1} + (t_{k+1}-1)/t_{k+2} * (f_{k+1} - f_k)
        
        # Let me use a simpler approach: momentum + projection
        # f_new = project(f - lr * grad_centered)
        f_new = project_simplex(f - lr * grad_centered)
        
        # Accept if it improves the true objective
        cur_obj = fast_convolve(f_new).max()
        if cur_obj <= best_obj + 1e-15:
            f = f_new
            if cur_obj < best_obj - 1e-14:
                best_obj = cur_obj
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 200 == 0:
                    print(f"    NS iter {it}: obj={best_obj:.12f}, p={p:.1f}, lr={lr:.6f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            # Reduce learning rate
            lr *= 0.9
        
        if no_improve > 2000:
            # Restart with perturbation
            rng_p = np.random.default_rng(it * 7919 + 13)
            noise = rng_p.standard_normal(n) * 0.002
            f = project_simplex(best_f + noise)
            y = f.copy()
            no_improve = 0
    
    return best_f, best_obj

def peak_cd(f0, time_limit, verbose=True):
    """Peak-targeted coordinate descent.
    At each step, find the coordinate whose adjustment most reduces the peak."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    pass_count = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        
        # Get current convolution
        c = fast_convolve(f)
        M = c.max()
        peak_positions = np.where(c >= M * 0.999)[0]
        
        # Try each coordinate (in random order for diversity in later passes)
        if pass_count == 1:
            order = np.arange(n)
        else:
            order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            # Try reducing f[j] and redistributing to others proportionally
            # or try increasing f[j] and reducing others
            # Use a local search: try several fractions
            best_frac = 1.0
            best_local = M
            
            # Try: f[j] -> frac * old_val, others scaled to maintain sum=1
            for frac in [0.3, 0.5, 0.7, 0.85, 0.95, 1.1, 1.3, 1.5]:
                new_val = frac * old_val
                if new_val < 0 or new_val > 1.0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_local - 1e-14:
                    best_local = val
                    best_frac = frac
            
            if best_frac != 1.0:
                new_val = best_frac * old_val
                denom = 1.0 - old_val
                scale = (1.0 - new_val) / denom
                f[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = f / f.sum()
                if best_local < best_obj - 1e-14:
                    best_obj = best_local
                    best_f = f.copy()
                    improved = True
        
        if not improved and pass_count > 3:
            break
        if verbose and pass_count % 5 == 0:
            print(f"    PeakCD pass {pass_count}: obj={best_obj:.12f}")
    
    return best_f, best_obj

def pair_cd(f0, time_limit, verbose=True):
    """Pair coordinate descent with targeted peak reduction."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(67890)
    passes = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        n_pairs = min(500 * (n // 500 + 1), 5000)
        
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            
            oi, oj = f[i], f[j]
            total = oi + oj
            if total < 1e-12:
                continue
            
            # Try transferring delta from i to j (and vice versa)
            best_delta = 0.0
            best_val = fast_convolve(f).max()
            
            for delta in [0.1 * total, 0.3 * total, 0.5 * total, 0.7 * total, 0.9 * total,
                        -0.1 * total, -0.3 * total, -0.5 * total, -0.7 * total, -0.9 * total]:
                new_i = oi + delta
                new_j = oj - delta
                if new_i < 0 or new_j < 0:
                    continue
                f_trial = f.copy()
                f_trial[i] = new_i
                f_trial[j] = new_j
                val = fast_convolve(f_trial).max()
                if val < best_val - 1e-14:
                    best_val = val
                    best_delta = delta
            
            if best_delta != 0:
                f[i] += best_delta
                f[j] -= best_delta
                if best_val < best_obj - 1e-14:
                    best_obj = best_val
                    best_f = f.copy()
                    improved = True
        
        if not improved and passes > 5:
            break
    
    return best_f, best_obj

def pressure_balance(f0, time_limit, verbose=True):
    """Iterative method that balances the 'pressure' from active peaks.
    The idea: at the optimum, the subgradient (from active peaks) should be
    (after centering) orthogonal to the feasible directions.
    We iterate: compute subgradient, take a step to reduce its variance."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(f)
        M = c.max()
        
        # Identify active set: positions where c is close to M
        threshold = M * 0.995
        active = c >= threshold
        K = active.sum()
        
        if K == 0:
            break
        
        # Compute subgradient: g[i] = 2 * sum_{k in active, a[k-i] valid} a[k-i] / K
        # This is 2/K * (indicator_active * a_reversed) correlation
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / K
        
        N = 1 << (3 * n - 1).bit_length()
        A_fft = np.fft.rfft(f, N)
        W_fft = np.fft.rfft(w, N)
        corr = np.fft.irfft(W_fft * np.conj(A_fft), N)[:n]
        g = 2.0 * corr
        
        # Center the gradient (to maintain sum=1)
        g_centered = g - g.mean()
        
        # Line search: find best alpha
        best_alpha = 0.0
        best_val = M
        
        for alpha in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3]:
            f_trial = project_simplex(f - alpha * g_centered)
            val = fast_convolve(f_trial).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            f = project_simplex(f - best_alpha * g_centered)
            if best_val < best_obj - 1e-15:
                best_obj = best_val
                best_f = f.copy()
                if verbose and it % 100 == 0:
                    print(f"    PressBal iter {it}: obj={best_obj:.12f}, K={K}, alpha={best_alpha}")
        else:
            # No improvement, try smaller steps or break
            if it > 100:
                break
    
    return best_f, best_obj

def interpolate_seq(f, new_n):
    """Linear interpolation to new length, normalized to sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def _smooth_logsumexp_optimize(f0, time_limit, T_start=0.3, T_end=0.005, verbose=False):
    """Log-sum-exp smooth optimization with Adam (similar to previous but different params)."""
    import time as _time
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t_adam = 0
    alpha = 0.005
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    T = T_start
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        T = T_start * (T_end / T_start) ** (elapsed / time_limit)
        T = max(T, T_end)
        lr = alpha * (0.5 ** (elapsed / (time_limit * 0.7)))
        
        # Compute gradient of logsumexp(conv/T)
        conv = fast_convolve(f)
        scaled = conv / T
        scaled -= scaled.max()
        e = np.exp(scaled)
        w = e / e.sum()
        
        N = 1 << (3 * n - 1).bit_length()
        A_fft = np.fft.rfft(f, N)
        W_fft = np.fft.rfft(w, N)
        corr = np.fft.irfft(A_fft * np.conj(W_fft), N)[:n]
        grad = (2.0 / T) * corr
        grad_centered = grad - grad.mean()
        
        t_adam += 1
        m_state = beta1 * m_state + (1 - beta1) * grad_centered
        v_state = beta2 * v_state + (1 - beta2) * grad_centered ** 2
        m_hat = m_state / (1 - beta1 ** t_adam)
        v_hat = v_state / (1 - beta2 ** t_adam)
        
        f_new = project_simplex(f - lr * m_hat / (np.sqrt(v_hat) + eps))
        
        cur_obj = fast_convolve(f_new).max()
        if cur_obj < best_obj + 1e-15:
            f = f_new
            if cur_obj < best_obj - 1e-14:
                best_obj = cur_obj
                best_f = f.copy()
                no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 2000:
            rng_p = np.random.default_rng(it * 104729 + 3)
            f = project_simplex(best_f + rng_p.standard_normal(n) * 0.003)
            m_state = np.zeros(n)
            v_state = np.zeros(n)
            t_adam = 0
            no_improve = 0
    
    return best_f, best_obj
