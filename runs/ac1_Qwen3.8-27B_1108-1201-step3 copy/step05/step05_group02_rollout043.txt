                  method='highs', options=options)

        duals = res.ineqlin.marginals if hasattr(res.ineqlin, 'marginals') else None

"""
Hybrid optimizer combining LP-Frank-Wolfe, ITRLS (Iteratively Reweighted
Least Squares), smoothed-max FISTA, and peak-directed coordinate descent
to minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.

Key innovations over previous approach:
- ITRLS: exploits the Hankel structure of the reweighted QP via FFT
- Smoothed max with adaptive temperature for FISTA-style acceleration
- Peak-directed coordinate descent that targets top-k peaks simultaneously
- Adaptive time allocation between methods based on convergence signals

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution constraint matrix A where (A@g)[k] = (f*g)[k]."""
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_direction(f, M, n, time_limit=30):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {'time_limit': time_limit, 'presolve': True}
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive M relaxation and precise line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break

        g = solve_lp_direction(f, M * m_relax, n, time_limit=min(30, rem * 0.5))
        if g is None:
            m_relax = min(m_relax * 1.1, 10.0)
            if m_relax > 10.0:
                m_relax = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 50:
                break
            m_relax = min(m_relax * 1.02, 5.0)
            continue

        # Coarse line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.02, 0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 0.9, 1.0]
        for alpha in alphas:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine bisection around best
        if best_alpha > 0:
            if best_alpha >= 1.0:
                lo, hi = 0.8, 1.0
            elif best_alpha <= 0.02:
                lo, hi = 0.0, 0.08
            else:
                lo = max(0.0, best_alpha - 0.12)
                hi = min(1.0, best_alpha + 0.12)
            for _ in range(20):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-14:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_relax = 1.0
            f = f_new

            if M < best_obj - 1e-16:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    LP-FW iter {it} ({time.monotonic()-t0:.0f}s): M={best_obj:.12f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.03, 5.0)

        if no_improve > 60:
            rng = np.random.default_rng(it * 104729 + 7)
            scale = 0.002 + 0.003 * min(no_improve // 60, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0
            no_improve = 0

    return best_f, best_obj


def itr

ls_optimize(f0, time_limit, n_inner=50, verbose=False):
    """
    ITRLS: Iteratively Reweighted Least Squares for the min-max convolution problem.
    At each outer iteration:
      1. Compute weights w[k] = 1/(conv[k] + eps)
      2. Solve QP: min a^T Q a s.t. a>=0, sum(a)=1 where Q[i,j]=w[i+j]/2
      3. Update a
    The QP is solved via projected gradient with the Hankel structure exploited via FFT.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    f = f / s
    n = len(f)
    L = 2 * n - 1

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    # FFT setup for Q*a computation
    fft_len = 1 << (2 * L - 1).bit_length()

    def q_matvec(w, a):
        """Compute Q @ a where Q[i,j] = w[i+j]/2 for valid indices."""
        # (Q a)[i] = 0.5 * sum_j w[i+j] * a[j]
        # This is 0.5 * cross-correlation of w and a
        W = np.zeros(fft_len)
        W[:L] = w
        A = np.zeros(fft_len)
        A[:n] = a
        # corr[i] = sum_j W[i+j] * A[j] (for valid indices, no wrap-around)
        # Using
        FFT: ifft(conj(FFT(W)) * FFT(A)) gives circular corr
        W_fft = np.fft.rfft(W, fft_len)
        A_fft = np.fft.rfft(A, fft_len)
        corr = np.fft.irfft(np.conj(W_fft) * A_fft, fft_len)
        # corr[i] = sum_j W[j+i mod fft_len] * A[j]
        # For i < n and our padding, this equals sum_j W[i+j] * A[j] (no wrap)
        return 0.5 * corr[:n]

    outer = 0
    while time.monotonic() - t0 < time_limit:
        outer += 1
        conv = fast_convolve(f)
        eps = 1e-10
        w = 1.0 / (conv + eps)
        w = w / w.max()  # normalize for numerical stability

        # Solve QP via projected gradient
        a_inner = f.copy()
        # Estimate step size
        step = 1.0  # will adjust

        for inner in range(n_inner):
            grad = 2.0 * q_matvec(w, a_inner)
            # Line search: try reducing step
            for trial_step in [step, step * 0.5, step * 0.25, step * 0.1]:
                a_trial = project_simplex(a_inner - trial_step * grad)
                val = fast_convolve(a_trial).max()
                if val <= conv.max() + 1e-12:
                    a_inner = a_trial
                    break
            else:
                step *= 0.5
                continue

            a_inner = project_simplex(a_inner)
            conv_new = fast_convolve(a_inner)
            if conv_new.max() < best_obj - 1e-15:
                best_obj = conv_new.max()
                best_f = a_inner.copy()
                if verbose and outer % 5 == 0:
                    print(f"    ITRLS outer {outer} (inner {inner}): M={best_obj:.12f}")

        f = a_inner

        if time.monotonic() - t0 > time_limit:
            break

    return best_f, best_obj


def smoothed_max_fista(f0, time_limit, beta_init=50.0, beta_growth=1.02, verbose=False):
    """
    FISTA for the smoothed max: (1/beta)*log(sum_k exp(beta * conv[k])).
    As beta increases, this approximates the true max.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    f = f / s
    n = len(f)

    best_f = f.copy()
    conv0 = fast_convolve(f)
    best_obj = conv0.max()

    t0 = time.monotonic()
    x = f.copy()
    x_prev = f.copy()
    t_mom = 1.0
    beta = beta_init
    lr = 1.0  # will adapt

    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = x + ((t_mom - 1.0) / t_new) * (x - x_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute smoothed gradient
        conv_y = fast_convolve(y)
        # Softmax weights
        shifted = beta * conv_y - beta * conv_y.max()
        p = np.exp(shifted)
        ps = p.sum()
        if ps < 1e-300:
            p = np.zeros(len(conv_y))
            p[np.argmax(conv_y)] = 1.0
        else:
            p = p / ps

        # Gradient of smoothed max = sum_k p_k * grad(conv[k])
        # grad(conv[k])[i] = 2*y[k-i] (if 0<=k-i<n)
        sg = np.zeros(n)
        # Use FFT: sg[i] = 2 * sum_k p[k] * y[k-i] (for valid k-i)
        # = 2 * corr(p, y)[i]
        fft_len = 1 << (2 * (2 * n - 1) - 1).bit_length()
        P = np.zeros(fft_len)
        P[:len(conv_y)] = p
        Y = np.zeros(fft_len)
        Y[:n] = y
        P_fft = np.fft.rfft(P, fft_len)
        Y_fft = np.fft.rfft(Y, fft_len)
        corr = np.fft.irfft(np.conj(P_fft) * Y_fft, fft_len)
        sg = 2.0 * corr[:n]

        # Gradient step with line search
        best_alpha = lr
        best_val = conv_y.max()
        for alpha in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            x_trial = project_simplex(y - alpha * sg)
            val = fast_convolve(x_trial).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
            if alpha < lr * 0.05:
                break

        x_new = project_simplex(y - best_alpha * sg)

        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_f = x_new.copy()
            no_improve = 0
            lr = min(lr * 1.01, 1.0)
            if verbose and it % 200 == 0:
                print(f"    FISTA iter {it} ({time.monotonic()-t0:.0f}s): M={best_obj:.12f} beta={beta:.1f}")
        else:
            no_improve += 1
            lr *= 0.98

        x_prev = x
        x = x_new
        t_mom = t_new
        beta *= beta_growth

        if no_improve > 80:
            rng = np.random.default_rng(it * 31 + 17)
            x = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            x_prev = x.copy()
            t_mom = 1.0
            lr = 0.5
            no_improve = 0

    return best_f, best_obj


def peak_cd(f0, time_limit, verbose=False):
    """
    Peak-directed coordinate descent: for each top peak, find the best
    single-coordinate adjustment that reduces the peak, then apply it.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        M = conv.max()

        # Find top 5 peaks
        k_peaks = min(5, 2 * n - 1)
        peak_idxs = np.argpartition(conv, -k_peaks)[-k_peaks:]
        peak_idxs = peak_idxs[np.argsort(conv[peak_idxs])[::-1]]

        improved = False
        for p in peak_idxs:
            if time.monotonic() - t0 > time_limit:
                break

            # conv[p] = sum_j f[j]*f[p-j]
            # Try adjusting each coordinate near the peak
            # The most impactful adjustment: reduce f[j] where f[j]*f[p-j] is largest
            js = range(max(0, p - n + 1), min(n - 1, p) + 1)
            js_arr = np.array(list(js))
            contribs = f[js_arr] * f[p - js_arr]
            top_contribs = np.argsort(contribs)[::-1][:10]

            for tc in top_contribs:
                j = js_arr[tc]
                partner = p - j

                # Try reducing f[j] by various amounts and adding to a "safe" position
                # Safe position: where adding mass causes least convolution increase
                # Approximate: position where sum_k f[k]*f[k-pos] is smallest
                # For speed, just try a few candidates
                old_val = f[j]
                if old_val < 1e-10:
                    continue

                for frac in [0.05, 0.1, 0.2, 0.3]:
                    delta = old_val * frac
                    if delta < 1e-12:
                        continue

                    # Try adding delta to a few candidate positions
                    # Candidates: positions with low "convolution sensitivity"
                    # Approximate sensitivity: for position q, adding d increases
                    # conv[k] by ~d*(f[k-q] + f[q-k+...]) for nearby k
                    # Simple heuristic: add to position with lowest current value
                    safe_candidates = np.argsort(f)[:5]

                    for q in safe_candidates:
                        if q == j:
                            continue
                        f_trial = f.copy()
                        f_trial[j] -= delta
                        f_trial[q] += delta

                        # Renormalize
                        st = f_trial.sum()
                        if st < 1e-15:
                            continue
                        f_trial = f_trial / st

                        # Check if the peak at p decreased
                        conv_trial = fast_convolve(f_trial)
                        new_p_val = conv_trial[p]
                        new_max = conv_trial.max()

                        if new_max < M - 1e-12:
                            f = f_trial
                            M = new_max
                            improved = True
                            if M < best_obj - 1e-15:
                                best_obj = M
                                best_f = f.copy()
                                if verbose and it % 20 == 0:
                                    print(f"    PeakCD iter {it} ({time.monotonic()-t0:.0f}s): M={best_obj:.12f}")
                            break
                    if improved and M < best_obj:
                        break

        if not improved:
            # Try small random perturbations
            rng = np.random.default_rng(it * 7 + 3)
            f_pert = project_simplex(f + rng.standard_normal(n) * 0.001)
            val = fast_convolve(f_pert).max()
            if val < M - 1e-14:
                f = f_pert
                if val < best_obj - 1e-15:
                    best_obj = val
                    best_f = f.copy()
            elif it > 100:
                break

    return best_f, best_obj


def micro_search(f0, time_limit, verbose=False):
    """
    Very fine-grained search: tiny perturbations with optional quick polish.
    Exploits the fact that near the optimum, the landscape is smooth.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    # Try multiple perturbation scales
    scales = [0.0005, 0.001, 0.002, 0.003, 0.005, 0.008, 0.012, 0.02]
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        for scale in scales:
            if time.monotonic() - t0 > time_limit:
                break

            # Generate perturbation
            perturbation = rng.standard_normal(n) * scale
            f_cand = project_simplex(best_f + perturbation)

            val = fast_convolve(f_cand).max()
            if val < best_obj - 1e-15:
                best_obj = val
                best_f = f_cand.copy()
                if verbose:
                    print(f"    Micro iter {it}: scale={scale}, M={best_obj:.12f}")

                # Quick local polish: try a few more tiny steps
                for _ in range(3):
                    if time.monotonic() - t0 > time_limit:
                        break
                    pert2 = rng.standard_normal(n) * scale * 0.3
                    f_cand2 = project_simplex(best_f + pert2)
                    val2 = fast_convolve(f_cand2).max()
                    if val2 < best_obj - 1e-16:
                        best_obj = val2
                        best_f = f_cand2.copy()

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
