"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast incremental convolution
updates, coordinate descent with multi-probe search, multiplicative subgradient
descent, and structured initialization. Key improvement: O(n) per coordinate
update via rank-1 convolution modification, enabling thousands of passes.
"""

import numpy as np
import time

def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_score(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))

def normalize_to_sum(a, target=1.0):
    """Normalize to sum=target."""
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a * (target / s)

def project_simplex(v):
    """Euclidean projection onto {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def update_conv_inplace(b, a, j, delta):
    """Update b = conv(a,a) in place after changing a[j] by delta.
    b[k] += 2*delta*a[k-j] for k in [j, j+n-1], with correction at k=2j.
    """
    n = len(a)
    # b[j:j+n] += 2*delta*a[0:n]
    b[j:j+n] += 2.0 * delta * a[:n]
    # Correction for k=2j: we added 2*delta*a[j], but should add 2*delta*a[j] + delta^2
    # Actually the formula b[k] += 2*delta*a[k-j] gives b[2j] += 2*delta*a[j]
    # The correct change is (a[j]+delta)^2 - a[j]^2 = 2*a[j]*delta + delta^2
    # So we need an extra delta^2 at position 2j
    if 2 * j < len(b):
        b[2*j] += delta * delta

def coord_descent_fast(a, time_limit, n_probes=5, verbose=False):
    """Fast coordinate descent with incremental convolution update."""
    n = len(a)
    a = a.copy().astype(np.float64)
    b = fast_convolve(a).copy()
    sum_a = a.sum()
    if sum_a < 1e-15:
        return a, float('inf')
    score = 2.0 * n * b.max() / (sum_a * sum_a)
    best_a = a.copy()
    best_score = score
    t0 = time.monotonic()
    pass_count = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        order = rng.permutation(n) if pass_count > 1 else np.arange(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-12:
                # Try small injection
                candidates = [0.001 / n, 0.01 / n, 0.05 / n]
            else:
                # Multi-probe: try several multipliers
                candidates = [
                    old_val * 0.1,
                    old_val * 0.3,
                    old_val * 0.7,
                    old_val * 1.3,
                    old_val * 1.7,
                    0.0
                ]

            best_local_val = old_val
            best_local_score = score

            for new_val in candidates:
                new_val = max(0.0, new_val)
                delta = new_val - old_val
                if abs(delta) < 1e-15:
                    continue

                # Update b in place
                old_b_slice = b[j:j+n].copy()
                b[j:j+n] += 2.0 * delta * a[:n]
                if 2 * j < len(b):
                    b[2*j] += delta * delta

                new_sum = sum_a + delta
                if new_sum < 1e-12:
                    b[j:j+n] = old_b_slice
                    if 2*j < len(b):
                        # Undo delta^2
                        b[2*j] -= delta * delta
                    continue

                new_score = 2.0 * n * b.max() / (new_sum * new_sum)

                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local_val = new_val

                # Revert b
                b[j:j+n] = old_b_slice
                if 2*j < len(b):
                    b[2*j] -= delta * delta

            # Apply best
            if best_local_val != old_val:
                delta = best_local_val - old_val
                b[j:j+n] += 2.0 * delta * a[:n]
                if 2 * j < len(b):
                    b[2*j] += delta * delta
                a[j] = best_local_val
                sum_a += delta
                score = best_local_score
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    improved = True

        if verbose and pass_count % 10 == 0:
            print(f"    CD pass {pass_count}: score={best_score:.10f}")

        if not improved and pass_count > 5:
            break
        if pass_count > 5000:
            break

    return best_a, best_score


def multi_peak_subgradient(a, time_limit, top_k=20, lr=0.01, decay=0.999, verbose=False):
    """Subgradient method with multi-peak weighting and multiplicative updates."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s  # normalize to sum=1
    b = fast_convolve(a)
    score = 2.0 * n * b.max()
    best_a = a.copy()
    best_score = score
    t0 = time.monotonic()
    step = 0
    cur_lr = lr

    while time.monotonic() - t0 < time_limit:
        step += 1
        # Find top-k peaks
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]

        # Weight by importance (exponential softmax)
        temp = max(1e-8, pv.max() - pv.min()) * 0.3
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(k_actual):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            # Subgradient of b_k w.r.t. a: 2*w_i * a[k-idx] for idx != k-idx, plus correction
            contrib = 2.0 * w[i] * a[k - idx]
            # Handle the case where idx == k-idx (i.e., 2*idx == k)
            for t in range(len(idx)):
                if 2 * idx[t] == k:
                    contrib[t] += w[i] * a[idx[t]]  # extra term for the square
            sg[idx] += contrib

        a = np.maximum(a * np.exp(-cur_lr * sg), 1e-15)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast incremental convolution
updates (O(n) per coordinate change instead of O(n log n) FFT), aggressive
coordinate descent with multi-probe search, multi-peak subgradient descent,
pair-swap local search, and structured initialization with dimension sweeping.
Key innovation: rank-1 convolution modification enables ~10x faster iteration.

import numpy as np
import time
from scipy.optimize import minimize_scalar


def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize_to_sum(a, target=1.0):
    """Normalize to sum=target, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a * (target / s)


def project_simplex(v):
    """Euclidean projection onto {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def coord_descent_fast(a, time_limit, n_probes=6, verbose=False):
    """Fast coordinate descent with O(n) incremental convolution updates."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s  # normalize to sum=1
    b = fast_convolve(a).copy()
    sum_a = 1.0
    score = 2.0 * n * b.max()
    best_a = a.copy()
    best_score = score
    t0 = time.monotonic()
    pass_count = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        order = rng.permutation(n) if pass_count > 1 else np.arange(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-14:
                candidates = [1e-5, 5e-5, 1e-4]
            else:
                candidates = [
                    old_val * 0.05,
                    old_val * 0.2,
                    old_val * 0.5,
                    old_val * 0.8,
                    old_val * 1.25,
                    old_val * 1.6,
                    0.0
                ]

            best_local_val = old_val
            best_local_score = score

            for new_val in candidates:
                new_val = max(0.0, min(new_val, 0.99))
                delta = new_val - old_val
                if abs(delta) < 1e-15:
                    continue

                # Save and update b
                saved = b[j:j+n].copy()
                b[j:j+n] += 2.0 * delta * a[:n]
                if 2 * j < len(b):
                    b[2*j] += delta * delta

                new_sum = sum_a + delta
                if new_sum < 1e-10:
                    b[j:j+n] = saved
                    if 2*j < len(b):
                        b[2*j] -= delta * delta
                    continue

                new_score = 2.0 * n * b.max() / (new_sum * new_sum)

                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local_val = new_val

                # Revert
                b[j:j+n] = saved
                if 2*j < len(b):
                    b[2*j] -= delta * delta

            # Apply best
            if abs(best_local_val - old_val) > 1e-15:
                delta = best_local_val - old_val
                b[j:j+n] += 2.0 * delta * a[:n]
                if 2 * j < len(b):
                    b[2*j] += delta * delta
                a[j] = best_local_val
                sum_a += delta
                score = best_local_score
                if score < best_score - 1e-14:
                    best_score = score
                    best_a = a.copy()
                    improved = True

        if verbose and pass_count % 20 == 0:
            print(f"    CD pass {pass_count}: score={best_score:.10f}")

        if not improved and pass_count > 10:
            break
        if pass_count > 5000:
            break

    # Final normalization
    best_a = normalize_to_sum(best_a)
    return best_a, best_score


def multi_peak_subgradient(a, time_limit, top_k=30, lr0=0.05, decay=0.9995, verbose=False):
    """Multi-peak weighted subgradient with multiplicative (mirror) updates."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = normalize_to_sum(a)
    b = fast_convolve(a)
    score = 2.0 * n * b.max()
    best_a = a.copy()
    best_score = score
    t0 = time.monotonic()
    step = 0
    cur_lr = lr0

    while time.monotonic() - t0 < time_limit:
        step += 1
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]

        # Softmax weights
        temp = max(1e-8, pv.max() - pv.min()) * 0.4
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(k_actual):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = 2.0 * w[i] * a[k - idx]
            sg[idx] += contrib

        # Multiplicative update
        a = np.maximum(a * np.exp(-cur_lr * sg), 1e-15)
        a = a / a.sum()

        b = fast_convolve(a)
        sc = 2.0 * n * b.max()
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()

        cur_lr *= decay

        if step % 3000 == 0 and cur_lr < lr0 * 0.05:
            cur_lr = lr0 * 0.3
            noise = np.random.standard_normal(n) * 0.005
            a = project_simplex(best_a + noise)
            b = fast_convolve(a)

        if verbose and step % 1000 == 0:
            print(f"    SG step {step}: score={best_score:.8f}, lr={cur_lr:.6f}")

    best_a = normalize_to_sum(best_a)
    return best_a, best_score


def pair_swap_search(a, time_limit, n_trials=10000, verbose=False):
    """Local search: transfer mass between pairs with incremental conv update."""
    n = len(a)

    accepts = 0

    while time.monotonic() - t0 < time_limit:
        batch = min(100, n_trials)
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            trial += 1

            i, j = rng.integers(0, n, 2)
            if i == j:
                continue

            a_j = a[j]
            if a_j < 1e-12:
                continue

            delta = a_j * f

            saved_j = b[j:j+n].copy()

            b[j:j+n] -= 2.0 * delta * a[:n]

            a[j] -= delta

            if 2*i < len(b):

def golden_section_coord(a, j, b, sum_a, n):
    """Golden-section search on coordinate j using incremental updates."""
    lo = 0.0
    hi = min(0.99, max(a[j] * 4.0, 0.1))
    if hi <= lo:
        hi = 0.5
    best_val = a[j]
    best_sc = 2.0 * n * b.max() / (sum_a * sum_a)

    for _ in range(25):
        if time.monotonic() > 1e18:
            break
        m1 = lo + 0.382 * (hi - lo)
        m2 = lo + 0.618 * (hi - lo)

        for test_val in [m1, m2]:
            delta = test_val - a[j]
            if abs(delta) < 1e-14:
                continue
            saved = b[j:j+n].copy()
            b[j:j+n] += 2.0 * delta * a[:n]
            if 2*j < len(b):
                b[2*j] += delta * delta
            new_sum = sum_a + delta
            if new_sum > 1e-10:
                sc = 2.0 * n * b.max() / (new_sum * new_sum)
                if sc < best_sc:
                    best_sc = sc
                    best_val = test_val
            b[j:j+n] = saved
            if 2*j < len(b):
                b[2*j] -= delta * delta

        # Narrow range
        sc_m1 = _eval_coord(b, a, j, m1, sum_a, n)
        sc_m2 = _eval_coord(b, a, j, m2, sum_a, n)
        if sc_m1 < sc_m2:
            hi = m2
        else:
            lo = m1

    return best_val, best_sc


def _eval_coord(b, a, j, test_val, sum_a, n):
    """Evaluate score at a specific coordinate value (non-destructive)."""
    delta = test_val - a[j]
    if abs(delta) < 1e-14:
        return 2.0 * n * b.max() / (sum_a * sum_a)
    b_copy_slice = b[j:j+n].copy()
    b[j:j+n] += 2.0 * delta * a[:n]
    if 2*j < len(b):
        b[2*j] += delta * delta

    b[j:j+n] = b_copy_slice
    if 2*j < len(b):
        b[2*j] -= delta * delta
    return sc

        print(f"    #{i+1} [{label}]: {sc:.8f}")
        try_seq(a, label)

    # === Phase 1: Optimize diverse high-quality starts ===
    print(f"\n=== Phase 1: Optimize diverse starts (best={best_score:.8f}) ===")

    # Select diverse starts
    selected = []
    used_types = set()
    for sc, a, label in start_scores:
        if len(selected) >= 8:
            break
        # Extract base type
        parts = label.split('_n')[0]
        base_type = parts.split('_')[0] + '_' + parts.split('_')[1] if '_' in parts else parts
        if base_type not in used_types:
            selected.append((a, label))
            used_types.add(base_type)

    # Ensure we include the best few
    for sc, a, label in start_scores[:3]:
        if not any(s[1] == label for s in selected):
            selected.append((a, label))

    # Include parent if good
    if parent is not None:
        if not any(label == "parent" for _, label in selected):
            selected.append((parent, "parent"))

    # Remove duplicates by label
    seen_labels = set()
    selected_unique = []
    for a, label in selected:
        if label not in seen_labels:
            selected_unique.append((a, label))
            seen_labels.add(label)
    selected = selected_unique[:8]

    n_selected = len(selected)
    per_budget = min((deadline - time.monotonic() - 200) / max(1, n_selected), 200)

    for a_init, label in selected:
        rem = deadline - time.monotonic() - 150
        if rem < 30:
            break
        t_alloc = min(per_budget, rem)
        print(f"  Optimizing [{label}]: t={t_alloc:.0f}s")
        a_opt, sc_opt = optimize_sequence(a_init, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, label + "_opt")

    # === Phase 2: Dimension exploration ===
    print(f"\n=== Phase 2: Dimension exploration (best={best_score:.8f}) ===")
    if best_a is not None:
        best_n = len(best_a)
        target_ns = [384, 512, 640, 768, 896, 1024, 1152, 1280, 1536, 2048]
        for target_n in target_ns:
            if target_n == best_n:
                continue
            rem = deadline - time.monotonic() - 120
            if rem < 40:
                break
            t_alloc = min(rem * 0.4, 100)
            a_interp = interpolate_seq(best_a, target_n)
            a_opt, sc = optimize_sequence(a_interp, time_limit=t_alloc)
            try_seq(a_opt, f"dim_{target_n}")

    # === Phase 3: Aggressive refinement on best ===
    print(f"\n=== Phase 3: Aggressive refinement (best={best_score:.8f}) ===")
    if best_a is not None:
        # Long CD
        rem = deadline - time.monotonic() - 60
        if rem > 30:
            t_cd = min(rem * 0.5, 200)
            print(f"  Final long CD: t={t_cd:.0f}s")
            a_final, sc = coord_descent_fast(best_a, time_limit=t_cd, verbose=True)
            try_seq(a_final, "final_cd")

        # Pair swap
        rem = deadline - time.monotonic() - 40
        if rem > 20:
            t_ps = min(rem * 0.6, 120)
            a_ps, sc = pair_swap_search(best_a, time_limit=t_ps)
            try_seq(a_ps, "final_ps")

        # Perturbation + re-optimize
        rem = deadline - time.monotonic() - 20
        n_best = len(best_a)
        n_perturb = min(5, int(rem // 15)) if rem > 30 else 0
        for trial in range(n_perturb):
            rem = deadline - time.monotonic() - 15
            if rem < 15:
                break
            scale = 0.003 * (1 + trial * 0.5)
            noise = rng.standard_normal(n_best) * scale
            a_pert = project_simplex(best_a + noise)
            a_opt, sc = optimize_sequence(a_pert, time_limit=min(rem, 50))
            try_seq(a_opt, f"perturb_{trial}_s{scale:.4f}")

        # Different init types at best dimension
        rem = deadline - time.monotonic() - 15
        if rem > 30:
            for init_fn, lbl in [(lambda: geometric_init(n_best, 0.9995), "geom_refine"),
                                  (lambda: cosine_init(n_best), "cos_refine"),
                                  (lambda: gaussian_init(n_best, 0.35), "gauss_refine")]:
                rem = deadline - time.monotonic() - 10
                if rem < 15:
                    break
                a_init = init_fn()
                a_opt, sc = optimize_sequence(a_init, time_limit=min(rem, 40))
                try_seq(a_opt, lbl)

    # === Phase 4: Ultra-precise final polish ===
    print(f"\n=== Phase 4: Final polish (best={best_score:.8f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_final, sc = coord_descent_fast(best_a, time_limit=min(rem, 60), n_probes=8)
            try_seq(a_final, "ultra_final")

    # === Final verification ===
    if best_a is None:
        best_a = geometric_init(768, 0.999)
        best_score = compute_score(best_a)

    # Verify with actual evaluate_sequence
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_a]
    final_score = float(evaluate_sequence(final_list))
    if np.isfinite(final_score):
        best_score = final_score
        best_a = np.array(final_list)

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_a)})")
    return [float(x) for x in best_a]
