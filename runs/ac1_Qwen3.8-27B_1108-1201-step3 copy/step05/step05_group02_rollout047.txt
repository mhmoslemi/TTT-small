"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Minimax LP-based Frank-Wolfe (new LP formulation)
2. Smooth log-sum-exp gradient descent with momentum
3. Targeted peak shaving
4. Multi-length exploration with structured initializations
5. Final polishing

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_minimax_lp(f, n):
    """
    Solve: minimize t
    subject to: conv(f, g)[k] <= t for all k
                sum(g) = 1
                g >= 0
    Returns g (the optimal direction).
    """
    # Build constraint matrix
    # Variables: [t, g_0, ..., g_{n-1}], total n+1 vars
    c = np.zeros(n + 1)
    c[0] = 1.0  # minimize t
    
    # Inequality constraints: -t + sum_j F[k,j] g[j] <= 0
    # F[k,j] = f[k-j] if 0 <= k-j < n, else 0
    n_cons = 2 * n - 1
    A_ub = np.zeros((n_cons, n + 1))
    A_ub[:, 0] = -1.0
    
    for j in range(n):
        # F[k,j] = f[k-j] for k-j in [0, n-1], i.e., k in [j, j+n-1]
        k_start = j
        k_end = j + n  # exclusive
        A_ub[k_start:k_end, j + 1] = f
    
    b_ub = np.zeros(n_cons)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(None, None)] + [(0, None)] * n  # t free, g >= 0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'time_limit': 30, 'presolve': True})
    
    if res.success and res.x is not None:
        g = res.x[1:]
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs > 1e-15:
            g /= gs
        return g
    return None


def solve_standard_lp(f, M, n):
    """
    Solve: maximize sum(g)
    subject to: conv(f, g)[k] <= M for all k
                g >= 0
    """
    c = -np.ones(n)
    n_cons = 2 * n - 1
    A_ub = np.zeros((n_cons, n))
    
    for j in range(n):
        k_start = j
        k_end = j + n
        A_ub[k_start:k_end, j] = f
    
    b_ub = np.full(n_cons, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'time_limit': 30, 'presolve': True})
    
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0)
        gs = g.sum()
        if gs > 1e-15:
            g /= gs
        return g
    return None


def lse_smooth_optimize(f0, time_limit, T_init=50.0, T_growth=1.005,
                        lr_init=0.01, lr_decay=0.999, mom=0.9,
                        verbose=False):
    """
    Smooth optimization using log-sum-exp approximation of max.
    Minimizes (1/T) log(sum exp(T * conv(a,a)[k])) on the simplex.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    v = np.zeros(n)  # momentum
    T = T_init
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        conv = fast_convolve(f)
        Tc = T * conv
        Tc_max = Tc.max()
        # log-sum-exp
        lse_val = Tc_max / T + np.log(np.sum(np.exp(Tc - Tc_max))) / T
        
        # Gradient of LSE
        w = np.exp(Tc - Tc_max)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        
        # Compute gradient: sum_k w[k] * 2 * a[shifted]
        grad = np.zeros(n)
        # For efficiency, only process significant weights
        threshold = 1e-10
        active = w > threshold
        for k in range(len(conv)):
            if not active[k]:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                grad[js:je+1] += 2.0 * w[k] * f[k - js:je+1][::-1]
        
        # Momentum update
        v = mom * v - lr * grad
        f_new = project_simplex(f + v)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"      LSE iter {it}: obj={best_obj:.12f}, T={T:.1f}")
        else:
            no_improve += 1
        
        f = f_new
        T *= T_growth
        lr *= lr_decay
        
        if no_improve > 200:
            # Restart with smaller T
            T = T_init
            lr = lr_init * 0.5
            v = np.zeros(n)
            f = project_simplex(best_f + np.random.default_rng(it * 13).standard_normal(n) * 0.001)
            no_improve = 0
    
    return best_f, best_obj

def peak_shave(f0, time_limit, n_peaks=5, verbose=False):
    """
    Targeted peak shaving: identify the top peaks of the convolution
    and use coordinate-like updates to reduce them.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        
        # Get top peaks
        k_peaks = np.argsort(conv)[-n_peaks:][::-1]
        
        improved = False
        for k in k_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            
            # The peak at k is sum_i f[i]*f[k-i]
            # To reduce it, we want to reduce the "contributing" entries
            # Strategy: for each pair (i, k-i) contributing to the peak,
            # try to reduce f[i] or f[k-i] and redistribute
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            
            # Find the largest contributor
            contribs = f[js:je+1] * f[k-js:k-je+1][::-1]
            if contribs.max() < 1e-15:
                continue
            
            j_idx = np.argmax(contribs) + js
            other_idx = k - j_idx
            
            # Try reducing f[j_idx] and spreading to non-contributing positions
            # Find positions that don't contribute to any top peak
            peak_positions = set()
            for kp in k_peaks:
                s2 = max(0, kp - n + 1)
                e2 = min(n - 1, kp)
                peak_positions.update(range(s2, e2 + 1))
            
            # Try moving mass from j_idx to positions not in any peak
            safe_positions = [j for j in range(n) if j not in peak_positions and f[j] < 0.5]
            
            if not safe_positions:
                # Try all positions except the peak contributors
                safe_positions = [j for j in range(n) if j != j_idx and j != other_idx]
            
            for target in safe_positions[:10]:
                for frac in [0.1, 0.2, 0.3, 0.5]:
                    delta = f[j_idx] * frac
                    if delta < 1e-10:
                        continue
                    f_trial = f.copy()
                    f_trial[j_idx] -= delta
                    f_trial[target] += delta
                    f_trial = np.maximum(f_trial, 0)
                    f_trial = f_trial / f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_f = f_trial.copy()
                        f = f_trial
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            # Try small random perturbations
            rng = np.random.default_rng(int(time.monotonic() * 1000))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def generate_structured_init(n, kind='raised_cosine', rng=None):
    """Generate a structured initial sequence."""
    if kind == 'uniform':
        f = np.ones(n) / n
    elif kind == 'raised_cosine':
        j = np.arange(n)
        f = 0.5 * (1 + np.cos(2 * np.pi * j / (n - 1)))
    elif kind == 'gaussian':
        j = np.arange(n)
        sigma = n / 4.0
        f = np.exp(-(j - (n-1)/2)**2 / (2 * sigma**2))
    elif kind == 'zero_boundary':
        f = np.zeros(n)
        f[1:-1] = 1.0 / (n - 2)
    elif kind == 'triangular':
        j = np.arange(n)
        mid = (n - 1) / 2.0
        f = 1.0 - np.abs(j - mid) / (mid + 1)
    else:
        f = np.ones(n) / n
    
    f = np.maximum(f, 0)
    s = f.sum()
    if s > 1e-15:
        f = f / s
    return f
