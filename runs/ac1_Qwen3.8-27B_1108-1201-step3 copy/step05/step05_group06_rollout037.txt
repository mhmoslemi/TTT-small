import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, k_star):
    """Compute subgradient of max conv at position k_star."""
    n = len(a)
    g = np.zeros(n)
    # g[m] = 2*a[k_star - m] for m in [max(0, k_star-n+1), min(n-1, k_star)]
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    for m in range(lo, hi + 1):
        g[m] = 2.0 * a[k_star - m]
    return g

def subgradient_optimize(a0, time_limit, verbose=False):

    n = len(a)

    t0 = time.monotonic()

    v = np.zeros(n)
    beta = 0.9
    t_iter = 0
    
    while time.monotonic() - t0 < time_limit:

        elapsed = time.monotonic() - t0
        
        # Compute convolution and find max

        k_star = int(np.argmax(conv))
        cur_max = conv[k_star]

        if g_norm < 1e-15:

        # Learning rate schedule
        lr = 0.01 / (1.0 + 0.01 * t_iter)

        a_new = a + v
        a_new = project_simplex(a_new)

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines: subgradient descent with line search, log-sum-exp smooth Adam,
adaptive temperature, diverse asymmetric initializations (geometric, tilted,
bump-based), peak deflation, and coordinate descent polishing.
Exploits the non-convex structure by targeting asymmetric sequences where
the convolution peak is away from center, and uses the special structure of
the subgradient (reversed/shifted copy of a) for efficient peak reduction.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, k_star):
    """Subgradient of max conv at peak position k_star."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    if hi >= lo:
        g[lo:hi + 1] = 2.0 * a[k_star - hi:k_star - lo + 1]
    return g


def subgradient_optimize(a0, time_limit, verbose=False):
    """Subgradient descent with adaptive step size and momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    momentum = np.zeros(n)
    beta = 0.85
    t_iter = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t_iter += 1
        elapsed = time.monotonic() - t0
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        cur_max = conv[k_star]
        g = compute_subgradient(a, k_star)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        base_lr = 0.5 / (1.0 + 0.05 * t_iter)
        best_lr = base_lr
        best_val = cur_max
        for factor in [0.05, 0.1, 0.2, 0.4, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0]:
            lr_try = base_lr * factor
            a_try = project_simplex(a - lr_try * g)
            val = fast_convolve(a_try).max()
            if val < best_val:
                best_val = val
                best_lr = lr_try
        momentum = beta * momentum - best_lr * g
        a = project_simplex(a + momentum)
        new_max = fast_convolve(a).max()
        if new_max < best_obj - 1e-14:
            best_obj = new_max
            best_a = a.copy()
            no_improve = 0
            if verbose and t_iter % 500 == 0:
                print(f"    Subgrad it={t_iter}: obj={2*n*best_obj:.8f}, k*={k_star}")
        else:
            no_improve += 1
        if no_improve > 200:
            a = best_a.copy()
            momentum = np.zeros(n)
            no_improve = 0
    return best_a, best_obj


def smooth_grad(a, T):
    """Gradient of logsumexp(conv(a,a)/T)*T via FFT."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    e = np.exp(scaled)
    w = e / e.sum()
    lse_val = T * (scaled.max() + np.log(e.sum()))
    N = 1 << (3 * n - 1).bit_length()
    A_fft = np.fft.rfft(a, N)
    W_fft = np.fft.rfft(w, N)
    corr = np.fft.irfft(A_fft * np.conj(W_fft), N)[:n]
    grad = (2.0 / T) * corr
    return grad, lse_val, conv, w


def smooth_optimize(a0, time_limit, T_start=0.3, T_end=0.005, verbose=False):
    """Log-sum-exp smooth optimization with Adam and adaptive restarts."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t_adam = 0
    alpha = 0.012
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    total_time = time_limit

    while time.monotonic() - t0 < total_time:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / total_time, 1.0)
        T = T_start * (T_end / T_start) ** (progress ** 0.7)
        T = max(T, T_end)
        lr = alpha * (0.4 ** (progress))
        grad, lse_val, conv, w = smooth_grad(a, T)
        t_adam += 1
        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad ** 2
        m_hat = m_state / (1 - beta1 ** t_adam)
        v_hat = v_state / (1 - beta2 ** t_adam)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a)
        cur_obj = fast_convolve(a).max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    Smooth it={it}: obj={2*n*best_obj:.8f}, T={T:.5f}")
        else:
            no_improve += 1
        if no_improve > 2000:
            rng_r = np.random.default_rng(int(elapsed * 1000) % 999983)
            scale = 0.005 + 0.005 * min(no_improve // 500, 4)
            a = project_simplex(best_a + rng_r.standard_normal(n) * scale)
            m_state = np.zeros(n)
            v_state = np.zeros(n)
            t_adam = 0
            no_improve = 0
    return best_a, best_obj


def lp_direction(a, M, n):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = a
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def coordinate_descent(a0, time_limit, verbose=False):
    """Coordinate descent with golden section on each coordinate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    GR = 0.6180339887
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            cur_max = fast_convolve(a).max()
            lo = 0.0
            hi = 1.0 - old_val + 1e-6
            c = hi - GR * (hi - lo)
            d = lo + GR * (hi - lo)
            for _ in range(10):
                fc = _eval_coord(a, j, c, n)
                fd = _eval_coord(a, j, d, n)
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - GR * (hi - lo)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + GR * (hi - lo)
            opt_val = (lo + hi) / 2.0
            opt_obj = _eval_coord(a, j, opt_val, n)
            if opt_obj < best_obj - 1e-13:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - opt_val) / denom
                    a[j] = opt_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                best_obj = opt_obj
                best_a = a.copy()
                improved = True
        if not improved and passes > 2:
            break
    return best_a, best_obj


def _eval_coord(a, j, x, n):
    """Evaluate max conv when a[j]=x, others scaled to maintain sum=1."""
    old_val = a[j]
    denom = 1.0 - old_val
    if denom < 1e-10:
        return 1e10
    if x < -1e-10 or x > 1.0 + 1e-6:
        return 1e10
    scale = (1.0 - x) / denom
    a_trial = a.copy()
    a_trial[j] = max(0.0, x)
    mask = np.ones(n, dtype=bool)
    mask[j] = False
    a_trial[mask] *= scale
    return fast_convolve(a_trial).max()


def make_geometric(n, r):
    """Geometric sequence normalized to sum=1."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_tilted(n, tilt_center, width):
    """Gaussian bump at tilt_center, normalized."""
    j = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((j - tilt_center) / width) ** 2)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_power_law(n, alpha, pos=0.0):
    """Power law decay from position pos."""
    j = np.arange(n, dtype=np.float64)
    d = np.abs(j - pos) + 1
    a = d ** (-alpha)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_two_bump(n, c1, c2, w1, w2):
    """Two Gaussian bumps."""
    j = np.arange(n, dtype=np.float64)
    a = 0.5 * np.exp(-0.5 * ((j - c1) / w1) ** 2) + 0.5 * np.exp(-0.5 * ((j - c2) / w2) ** 2)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Linear interpolation to new length, normalized to sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
