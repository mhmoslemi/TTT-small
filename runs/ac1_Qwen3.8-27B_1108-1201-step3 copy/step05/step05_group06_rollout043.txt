"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key insight: the infinite sequence b[k]=C(2k,k)/4^k has perfectly flat self-convolution.
Truncating and normalizing gives cost ~ pi/2. We use this as a starting point and
refine with smooth log-sum-exp optimization, coordinate descent, and tail-aware
modifications to beat the pi/2 barrier.
"""

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def flat_conv_seq(n):
    """Generate the truncated flat-convolution sequence: a[k] = C(2k,k)/4^k normalized."""
    k = np.arange(n, dtype=np.float64)
    # Use log to avoid overflow for large k
    # log(C(2k,k)/4^k) = log(C(2k,k)) - k*log(4)
    # log(C(2k,k)) = log( (2k)! / (k!)^2 ) = sum_{i=1}^{k} log(2k-i+1) - sum_{i=1}^{k} log(i)
    # = sum_{i=1}^{k} log((k+i)/i) = sum_{i=1}^{k} (log(k+i) - log(i))
    # For large k, use Stirling or direct computation
    # For n up to 4096, direct computation is fine
    log_vals = np.zeros(n)
    running_log = 0.0
    for i in range(n):
        if i == 0:
            log_vals[0] = 0.0
        else:
            # C(2i,i)/4^i = C(2i-2,i-1) * (2i-1)/(2i) / ... actually let's use recurrence
            # C(2i,i)/4^i = C(2i-2,i-1)/4^(i-1) * (2i-1)/(2i) * (2i-1)/(i) ... 
            # Actually: C(2i,i) = C(2i-1,i) * 2i/(2i-i) ... 
            # Simpler: C(2i,i)/4^i = C(2i-2,i-1)/4^(i-1) * (2i)(2i-1)/(i*4)
            #         = prev * (2i)(2i-1)/(4i)
            #         = prev * (2i-1)/2
            # Wait: C(2i,i) = C(2i-1,i) * 2i/(2i-1+i-i) ... let me just use the ratio directly
            # C(2i,i)/C(2i-2,i-1) = (2i)!/(i!^2) * ((i-1)!^2)/(2i-2)! = (2i)(2i-1)/(i^2)
            # So C(2i,i)/4^i = [C(2i-2,i-1)/4^(i-1)] * (2i)(2i-1)/(4*i^2)
            #                = prev * (2i-1)/(2i)
            # Wait: (2i)(2i-1)/(4*i^2) = (2i-1)/(2i)
            running_log += math.log((2 * i + 1) / (2 * i + 2))
            log_vals[i] = running_log
    # Exponentiate (should be fine since values are <= 1)
    vals = np.exp(log_vals)
    vals = np.maximum(vals, 0)
    s = vals.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return vals / s


def flat_conv_seq_direct(n):
    """Direct computation of C(2k,k)/4^k for small n."""
    vals = np.zeros(n)
    for k in range(n):
        # C(2k,k)/4^k
        if k == 0:
            vals[k] = 1.0
        else:
            vals[k] = vals[k-1] * (2*k - 1) / (2*k)
    s = vals.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return vals / s


def make_geometric(n, r):
    """Geometric sequence normalized to sum=1."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_cosine(n, power=2.0):
    """Cosine window raised to a power."""
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** power + 0.005
    return a / a.sum()

    s = a.sum()

    k = np.arange(1, n + 1, dtype=np.float64)
    a = k ** (-alpha)
    s = a.sum()
    if s < 1e-15:

    return a / s


def smooth_grad(f, T):
    """Compute smooth gradient of logsumexp(conv(f,f)/T)."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / T
    scaled -= scaled.max()
    e = np.exp(scaled)
    w = e / e.sum()
    lse_val = T * (scaled.max() + np.log(e.sum()))
    
    # grad_i = (2/T) * sum_j f[j] * w[i+j] for i=0..n-1 (cross-correlation)
    N = 1 << (3 * n - 1).bit_length()
    A_fft = np.fft.rfft(f, N)
    W_fft = np.fft.rfft(w, N)
    corr = np.fft.irfft(A_fft * np.conj(W_fft), N)[:n]
    grad = (2.0 / T) * corr
    return grad, lse_val, conv, w


def smooth_optimize(f0, time_limit, T_start=0.1, T_end=0.003, verbose=False):
    """Optimize using log-sum-exp smooth approximation with Adam."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t_adam = 0
    
    alpha = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / time_limit, 1.0)
        T = T_start * (T_end / T_start) ** progress
        T = max(T, T_end)
        lr = alpha * (0.3 ** (progress * 0.7))
        
        grad, lse_val, conv, w = smooth_grad(f, T)
        
        t_adam += 1
        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad ** 2
        m_hat = m_state / (1 - beta1 ** t_adam)
        v_hat = v_state / (1 - beta2 ** t_adam)
        
        f = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f)
        
        cur_obj = fast_convolve(f).max()
        
        if cur_obj < best_obj - 1e-12:

        if no_improve > 5000:

            f = project_simplex(best_f + rng_p.standard_normal(n) * scale)
            m_state = np.zeros(n)
            v_state = np.zeros(n)
            t_adam = 0
            no_improve = 0
    
    return best_f, best_obj


def single_cd(f0, time_limit, verbose=False):
    """Single coordinate descent with golden section search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    ORDER = 0.6180339887
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        cur_conv = fast_convolve(f)
        cur_max = cur_conv.max()
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue
            
            # Try several candidate values
            candidates = [0.0, old_val * 0.5, old_val * 0.7, old_val * 0.85,
                         old_val * 0.95, old_val * 1.05, old_val * 1.15,
                         old_val * 1.3, 0.999]
            best_cand = old_val
            best_local = cur_max
            
            for c in candidates:
                if c < 0 or c > 0.9999:
                    continue
                scale = (1.0 - c) / denom
                f_trial = f.copy()
                f_trial[j] = c
                f_trial[j] = c  # set j
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                
                # Compute new max efficiently: only positions involving j change
                # C[k] involves j if k-j is in [0,n-1], i.e., j <= k <= j+n-1
                # Also if k-j < 0 or k-j >= n, no contribution
                # Old contribution of j to C[k]: a[j]*a[k-j] (if valid)
                # New contribution: c * a[k-j]*scale (if valid)
                # Delta = (c*scale - a[j]) * a[k-j] for valid k-j
                # Actually it's more complex since other elements also scale
                # Let's just compute the full conv for simplicity (n is not too large)
                val = fast_convolve(f_trial).max()
                if val < best_local - 1e-14:
                    best_local = val
                    best_cand = c
            
            if abs(best_cand - old_val) > 1e-10:
                scale = (1.0 - best_cand) / denom
                f[j] = best_cand
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = f / f.sum()
                if best_local < best_obj - 1e-13:
                    best_obj = best_local
                    best_f = f.copy()
                    improved = True
                cur_max = best_local
        
        if not improved and passes > 5:
            break
    
    return best_f, best_obj

                    best_f = f.copy()

def lpfw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Build LP: max sum(g) s.t. (f*g)[k] <= M*m_exp for all k, g >= 0
        M_try = M * m_exp
        c = -np.ones(n)
        # (f*g)[k] = sum_j f[j]*g[k-j] for valid indices
        # This is a convolution constraint, not a simple matrix multiplication
        # A_ub[k, j] = f[j] if k-j is in [0, n-1], i.e., j <= k <= j+n-1
        # So A_ub[k, j] = f[j] if max(0, k-n+1) <= j <= min(k, n-1)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            # g[j] contributes to (f*g)[k] at positions k = j, j+1, ..., j+n-1
            # The coefficient is f[k-j]
            for k in range(j, min(j + n, 2 * n - 1)):
                A_ub[k, j] = f[k - j]
        
        b_ub = np.full(2 * n - 1, M_try)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 5})
            if res.success and res.x is not None:
                g = res.x
            else:
                g = None
        except Exception:
            g = None
        
        if g is None:
            m_exp *= 1.2
            if m_exp > 10.0:
                m_exp = 1.0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 80:
                break
            m_exp = min(m_exp * 1.05, 5.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            # Refine
            lo = max(0.0, best_alpha - 0.2)
            hi = min(1.0, best_alpha + 0.2)
            for _ in range(15):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
            
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 30 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 5.0)
        
        if no_improve > 150:
            rng_p = np.random.default_rng(it * 31 + 7)
            scale = 0.003 + 0.003 * min(no_improve // 100, 5)
            noise = rng_p.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    
    return best_f, best_obj


def interpolate_seq(f, new_n):
    """Linear interpolation to new length, normalized to sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage optimizer using flat-convolution sequence initialization.
    The key insight: b[k]=C(2k,k)/4^k has perfectly flat self-convolution.
    Truncating gives cost ~ pi/2. We refine with smooth optimization and
    coordinate descent to beat this barrier.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
    except Exception:
        pass
    
    def _try(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            old_score = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)}) [was {old_score:.6f}]")
    
    # === Phase 0: Diverse initializations ===
    print("=== Phase 0: Initializations ===")
    
    # Flat convolution sequences at various lengths
    for n_try in [256, 384, 512, 768, 1024, 1536, 2048]:
        f_flat = flat_conv_seq_direct(n_try)
        _try(f_flat, f"flat_n{n_try}")
    
    # Bimodal variants
    for n_try in [512, 768, 1024]:
        f_bi = make_bimodal_flat(n_try)
        _try(f_bi, f"bimodal_n{n_try}")
    
    # Power law variants
    for n_try in [512, 768, 1024]:
        for alpha in [0.3, 0.5, 0.7, 1.0]:
            f_pl = make_power_law(n_try, alpha)
            _try(f_pl, f"pow_{alpha}_n{n_try}")
    
    # Geometric
    for n_try in [512, 768, 1024]:
        for r in [0.97, 0.98, 0.99, 0.995]:
            g = make_geometric(n_try, r)
            _try(g, f"geo_{r}_n{n_try}")
    
    # Cosine
    for n_try in [512, 768, 1024]:
        for pw in [1.0, 1.5, 2.0, 3.0]:
            g = make_cosine(n_try, power=pw)
            _try(g, f"cos_{pw}_n{n_try}")
    
    print(f"  After Phase 0: best={best_score:.6f} (n={len(best_seq) if best_seq is not None else 0})")
    
    # === Phase 1: Smooth optimization on best initializations ===
    print(f"\n=== Phase 1: Smooth Optimization ===")
    
    # Primary: optimize the flat convolution sequence at the best length
    base = best_seq.copy()
    n_best_init = len(base)
    
    rem = deadline - time.monotonic() - 800
    if rem > 30:
        t_alloc = min(rem, 400)
        print(f"  Smooth opt n={n_best_init}, t={t_alloc:.0f}s")
        a_opt, _ = smooth_optimize(base, time_limit=t_alloc, T_start=0.05, T_end=0.003, verbose=True)
        _try(a_opt, "smooth_primary")
    
    # Also try flat convolution at different lengths
    for n_try in [768, 1024, 1536, 2048]:
        rem = deadline - time.monotonic() - 500
        if rem < 30:
            break
        f_flat = flat_conv_seq_direct(n_try)
        t_alloc = min(rem, 200)
        a_opt, _ = smooth_optimize(f_flat, time_limit=t_alloc, T_start=0.05, T_end=0.005)
        _try(a_opt, f"smooth_flat_n{n_try}")
    
    # Mixed: 80% flat + 20% uniform
    rem = deadline - time.monotonic() - 400
    if rem > 20:
        n_cur = len(best_seq) if best_seq is not None else 768
        f_mix = 0.8 * best_seq + 0.2 * (np.ones(n_cur) / n_cur)
        f_mix = project_simplex(f_mix)
        t_alloc = min(rem, 150)
        a_opt, _ = smooth_optimize(f_mix, time_limit=t_alloc, T_start=0.03, T_end=0.003)
        _try(a_opt, "smooth_mixed")
    
    # === Phase 2: LP-FW ===
    print(f"\n=== Phase 2: LP-FW (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 350
    if rem > 30:
        t_alloc = min(rem, 200)
        print(f"  LP-FW n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = lpfw_optimize(best_seq, time_limit=t_alloc, verbose=True)
        _try(a_opt, "lpfw_primary")
    
    # LP-FW with perturbation
    rem = deadline - time.monotonic() - 250
    if rem > 20:
        noise = rng.standard_normal(len(best_seq)) * 0.004
        f_pert = project_simplex(best_seq + noise)
        t_alloc = min(rem, 120)
        a_opt, _ = lpfw_optimize(f_pert, time_limit=t_alloc)
        _try(a_opt, "lpfw_pert")
    
    # === Phase 3: Coordinate descent ===
    print(f"\n=== Phase 3: Coordinate Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 15:
        t_cd = min(rem * 0.5, 100)
        print(f"  Single CD t={t_cd:.0f}s")
        a_opt, _ = single_cd(best_seq, time_limit=t_cd)
        _try(a_opt, "cd_single")
    
    rem = deadline - time.monotonic() - 150
    if rem > 15:
        t_pw = min(rem * 0.5, 80)
        print(f"  Pairwise CD t={t_pw:.0f}s")
        a_opt, _ = pairwise_cd(best_seq, time_limit=t_pw)
        _try(a_opt, "cd_pair")
    
    # === Phase 4: Longer sequences ===
    print(f"\n=== Phase 4: Longer Sequences (best={best_score:.6f}) ===")
    for target_n in [2048, 3072, 4096]:
        rem = deadline - time.monotonic() - 100
        if rem < 30:
            break
        
        # Start from flat convolution at this length
        f_start = flat_conv_seq_direct(target_n)
        
        t1 = min(rem * 0.5, 120)
        a_opt, _ = smooth_optimize(f_start, time_limit=t1, T_start=0.04, T_end=0.004)
        _try(a_opt, f"smooth_long_n{target_n}")
        
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            # Also try interpolating the current best
            f_interp = interpolate_seq(best_seq, target_n)
            t2 = min(rem * 0.6, 80)
            a_opt, _ = smooth_optimize(f_interp, time_limit=t2, T_start=0.03, T_end=0.003)
            _try(a_opt, f"interp_n{target_n}")
    
    # === Phase 5: Final polishing ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Fine smooth optimization
        rem = deadline - time.monotonic() - 60
        if rem > 10:
            a_opt, _ = smooth_optimize(best_seq, time_limit=min(rem, 80),
                                       T_start=0.02, T_end=0.002, verbose=True)
            _try(a_opt, "smooth_fine")
        
        # LP-FW final
        rem = deadline - time.monotonic() - 40
        if rem > 10:
            a_opt, _ = lpfw_optimize(best_seq, time_limit=min(rem, 50))
            _try(a_opt, "lpfw_final")
        
        # Coordinate descent final
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            a_opt, _ = single_cd(best_seq, time_limit=min(rem, 30))
            _try(a_opt, "cd_final")
        
        # Perturbations
        for pi in range(5):
            rem = deadline - time.monotonic() - 15
            if rem < 8:
                break
            noise = rng.standard_normal(n_best) * 0.002
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = smooth_optimize(f_pert, time_limit=min(rem, 20),
                                       T_start=0.015, T_end=0.002)
            _try(a_opt, f"perturb_{pi}")
        
        # Tail extension: try adding a small bump at the end
        rem = deadline - time.monotonic() - 10
        if rem > 8 and n_best >= 128:
            f_ext = best_seq.copy()
            # Create a slightly longer sequence with adjusted tail
            new_n = n_best + 64
            f_long = interpolate_seq(f_ext, new_n)
            # Boost the tail
            tail_start = new_n - 64
            f_long[tail_start:] += 0.001 * np.linspace(1, 0, 64)
            f_long = project_simplex(f_long)
            a_opt, _ = smooth_optimize(f_long, time_limit=min(rem, 15),
                                       T_start=0.01, T_end=0.002)
            _try(a_opt, "tail_ext")
    
    # Fallback
    if best_seq is None:
        best_seq = flat_conv_seq_direct(512)
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
