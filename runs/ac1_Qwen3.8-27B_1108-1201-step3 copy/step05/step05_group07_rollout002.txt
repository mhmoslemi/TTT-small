import cvxpy as cp

def solve_lp(a, M, active_set=None):
    n = len(a)
    g = cp.Variable(n)
    
    # Objective: maximize sum(g)
    objective = cp.Maximize(cp.sum(g))
    
    # Constraints
    constraints = [g >= 0]
    
    # Convolution constraints: conv(a, g)[k] <= M
    # conv(a, g)[k] = sum_i a[i] * g[k-i]
    # For each k, this is a linear function of g
    if active_set is None:
        active_set = range(2*n - 1)
    
    for k in active_set:
        # conv(a,g)[k] = sum_{i} a[i] * g[k-i] where 0 <= i < n and 0 <= k-i < n
        # = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i] * g[k-i]
        expr = 0
        for i in range(max(0, k-n+1), min(n, k+1)):
            expr += a[i] * g[k-i]
        constraints.append(expr <= M)
    
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.GLPK)  # or another solver
    return g.value

        return np.ones(n) / n

def lbgfs_optimize(z0, T, time_limit, maxiter=3000):
    """L-BFGS in log-space with time limit enforcement."""
    t0 = time.monotonic()
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        obj, _ = soft_max_obj_grad(z, T)
        return obj
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = soft_max_obj_grad(z, T)
        return g
    
    try:
        res = minimize(
            fun=func,
            x0=z0,
            jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 25, 'disp': False}
        )
        return res.x, res.fun
    except Exception:
        return z0, func(z0)

        coeffs = a[k - idx]  # a[k-0], a[k-1], ..., a[k-(je-js)]
        # g[0], g[1], ..., g[je-js] with coefficients coeffs
        # So the constraint is: coeffs[0]*g[js] + coeffs[1]*g[js+1] + ... + coeffs[-1]*g[je] <= M
        # Wait, I need to be more careful.
        # conv(a,g)[k] = sum_i a[i]*g[k-i] where 0<=i<n and 0<=k-i<n
        # Let j = k-i, so i = k-j. Constraint: 0<=k-j<n means k-n+1<=j<=k, and 0<=j<n.
        # So j ranges from max(0, k-n+1) to min(n-1, k).
        # The term is a[k-j]*g[j].
        expr = cp.sum(a[k - idx] * g[idx])
        constraints.append(expr <= M)
    
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return None, None
    
    if g.value is None:
        return None, None
    
    return np.maximum(g.value, 0), prob.value


def lp_based_optimize(a0, time_limit=300.0, max_iters=200, verbose=True):
    """
    LP-based active set optimization.
    At each step, find a descent direction via LP and update.
    """
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Step size schedule
    base_t = 0.1
    min_t = 0.001
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        M = b.max()
        cur_obj = 2.0 * n * M
        
        # Find active set: all k where b[k] > 0.9 * M
        active_threshold = 0.85 * M
        active_set = list(np.nonzero(b >= active_threshold)[0])
        
        # Also make sure we include the argmax
        if (b.argmax() not in active_set):
            active_set.append(b.argmax())
        
        # Solve LP
        g, lp_val = solve_lp_active_set(a, M, active_set)
        
        if g is None or lp_val is None:
            # Try with all constraints
            g, lp_val = solve_lp_active_set(a, M, list(range(2 * n - 1)))
        
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        
        if lp_val <= 1.0 + 1e-8:
            # No improvement possible with current active set
            if verbose:
                pass  # print(f"  LP iteration {it}: no improvement (lp_val={lp_val:.6f})")
            # Try with a larger active set or break
            # Expand active set slightly
            active_threshold = 0.7 * M
            expanded_set = list(np.nonzero(b >= active_threshold)[0])
            g, lp_val = solve_lp_active_set(a, M, expanded_set)
            if g is None or g.sum() <= 1.0 + 1e-8:
                break
        
        # Normalize g and update
        g_norm = g / g.sum()
        
        # Line search: try different step sizes
        best_t_step = base_t
        best_obj_step = cur_obj
        for t_candidate in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            a_trial = (1 - t_candidate) * a + t_candidate * g_norm
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            obj_trial = 2.0 * n * fast_convolve(a_trial).max()
            if obj_trial < best_obj_step:
                best_obj_step = obj_trial
                best_t_step = t_candidate
        
        a = (1 - best_t_step) * a + best_t_step * g_norm
        a = np.maximum(a, 0)
        a = a / a.sum()
        
        cur_obj = 2.0 * n * fast_convolve(a).max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            if verbose and (it % 20 == 0 or it < 5):
                print(f"  LP iter {it}: obj={cur_obj:.8f} (improved)")
        else:
            # Revert if no improvement
            a = best_a.copy()
            # Reduce step size
            base_t = max(min_t, base_t * 0.5)
        
        # Gradually increase step size
        base_t = min(0.3, base_t * 1.05)
    
    return best_a, best_obj


def subgradient_polish(a0, time_limit=60.0, lr=0.005, mom=0.9, top_k=30):
    """Subgradient method with softmax-weighted active peaks."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = 0.001
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]
        
        v = mom * v - cur_lr * sg
        a = project_simplex(a + v)
        b = fast_convolve(a)
        co = 2.0 * n * b.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.999
        cur_temp *= 1.01
        
        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 1000000)
            a = project_simplex(best_a + rng.standard_normal(n)
