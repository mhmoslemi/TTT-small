def fast_convolve(a):

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)

    grad = np.zeros(n)
    for k in peaks:
        # dconv[k]/da[j] = 2*a[k-j] if 0<=k-j<n
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * a[k - idx]
    # Normalize the subgradient (average over peak positions for stability)
    grad /= len(peaks)
    return grad, conv, max_val

def softmax_gradient_descent(a0, time_limit, verbose=False):
    """Optimize using smooth gradient (softmax approximation of max)."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    # Adam optimizer state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    
    tau = 20.0  # Start with moderate smoothness
    lr = 0.02
    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        
        # Adaptive tau: increase over time to focus on peaks
        progress = (time.monotonic() - t0) / time_limit
        tau = 20.0 + progress * 180.0  # 20 to 200
        
        # Adaptive learning rate
        if no_improve > 200:
            lr = max(0.001, lr * 0.95)
        elif no_improve > 50:
            lr = max(0.002, lr * 0.98)
        else:
            lr = min(0.05, lr * 1.002)
        
        # Compute smooth objective and gradient
        smooth_obj, grad, conv = smooth_objective(a, tau)
        
        # Center gradient for simplex constraint
        grad_centered = grad - grad.mean()
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_centered
        v = beta2 * v + (1 - beta2) * grad_centered**2
        m_hat = m / (1 - beta1**iter_count)
        v_hat = v / (1 - beta2**iter_count)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        a_new = project_simplex(a - step, target_sum=1.0)
        
        # Evaluate exact objective
        obj_new = fast_convolve(a_new).max()
        
        # Line search: try larger steps
        if obj_new < best_obj:
            for scale in [1.5, 2.0, 3.0]:
                a_try = project_simplex(a - scale * step, target_sum=1.0)
                obj_try = fast_convolve(a_try).max()
                if obj_try < obj_new:
                    a_new = a_try
                    obj_new = obj_try
                else:
                    break
        
        a = a_new
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD iter {iter_count}: obj={best_obj:.10f}, tau={tau:.1f}, lr={lr:.5f}, no_imp={no_improve} ({elapsed:.0f}s)")
        
        # Restart if stuck
        if no_improve > 400:
            rng_r = np.random.default_rng(iter_count * 17 + 3)
            noise_scale = 0.003 + 0.003 * min(no_improve // 400, 3)
            a = project_simplex(best_a + rng_r.standard_normal(n) * noise_scale, target_sum=1.0)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = 0.015
            tau = 30.0
    
    return best_a, best_obj

def subgradient_descent(a0, time_limit, verbose=False):
    """Exact subgradient descent with backtracking line search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    momentum = 0.8
    lr = 0.03
    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        
        # Adaptive parameters
        if no_improve > 300:
            momentum = min(0.95, momentum + 0.003)
            lr = max(0.002, lr * 0.96)
        elif no_improve > 100:
            lr = max(0.003, lr * 0.99)
        else:
            lr = min(0.06, lr * 1.003)
        
        grad, conv, max_val = exact_subgradient(a)
        grad_centered = grad - grad.mean()
        
        v = momentum * v - lr * grad_centered
        a_new = project_simplex(a + v, target_sum=1.0)
        obj_new = fast_convolve(a_new).max()
        
        # Line search
        if obj_new < best_obj:
            a_ext = project_simplex(a + 2.0 * v, target_sum=1.0)
            obj_ext = fast_convolve(a_ext).max()
            if obj_ext < obj_new:
                a_new = a_ext
                obj_new = obj_ext
        
        a = a_new
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGrad iter {iter_count}: obj={best_obj:.10f}, lr={lr:.5f}, no_imp={no_improve} ({elapsed:.0f}s)")
        
        if no_improve > 500:
            rng_r = np.random.default_rng(iter_count * 23 + 11)
            noise_scale = 0.004
            a = project_simplex(best_a + rng_r.standard_normal(n) * noise_scale, target_sum=1.0)
            v = np.zeros(n)
            no_improve = 0
            lr = 0.025
            momentum = 0.8
    
    return best_a, best_obj

    a = np.maximum(a0.astype(np.float64), 0.0)

    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:

    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    pass_count = 0
    
    # Precompute for speed
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        
        for j in range(n):

            for mult in multipliers:
