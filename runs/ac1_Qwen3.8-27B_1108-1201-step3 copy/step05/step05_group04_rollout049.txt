Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-guided iterative refinement,
Adam optimization, and coordinate descent polishing.
Primary method: iterative LP to find optimal blend direction.

import time

from scipy.sparse import csr_matrix

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(f):
    """Build the convolution matrix F where (F @ g)[k] = conv(f, g)[k]."""
    n = len(f)
    m = 2 * n - 1
    F = np.zeros((m, n))
    for k in range(m):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            F[k, j] = f[k - j]
    return F

    n = len(f)
    F = build_conv_matrix(f)  # (2n-1) x n
    
    # LP: min t s.t. F@g <= t, sum(g) = 1, g >= 0
    # Variables: g[0..n-1], t
    # Minimize t (index n in the variable vector)
    
    c = np.zeros(n + 1)
    c[n] = 1  # minimize t
    
    # Inequality constraints: F@g - t <= 0
    A_ub = np.hstack([F, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)

    A_eq = np.zeros((1, n + 1))

    return res.x[:n]

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build matrix F where (F @ g)[k] = sum_j f[k-j]*g[j] (valid indices only)."""
    n = len(f)
    m = 2 * n - 1
    F = np.zeros((m, n))
    for k in range(m):
        start = max(0, k - n + 1)
        end = min(n - 1, k)
        F[k, start:end+1] = f[k - np.arange(start, end + 1)]
    return F


def lp_max_sum_direction(f, M):
    """Solve: max sum(g) s.t. F@g <= M, g >= 0. Returns g or None."""
    n = len(f)
    F = build_conv_matrix(f)
    c = -np.ones(n)
    res = scipy.optimize.linprog(
        c, A_ub=F, b_ub=np.full(2*n-1, M),

        # Efficient computation using the relationship:
        # If we define w_padded of length n with w_padded[m] = sum of w[k] for k in some range...
        # Actually, let me just use the straightforward approach:
        # sg[j] = 2 * sum_{m=0}^{n-1} w[j+m] * f[m]  (for j+m < 2n-1)
        # This is a "partial correlation" that can be computed with sliding dot products.
        # For efficiency, I'll use the FFT-based approach:
        # Create a vector v of length 2n-1: v[k] = w[k]
        # sg[j]/2 = sum_m v[j+m] * f[m] = (correlation of v and f at shift j)
        # = np.correlate(v, f, mode='valid')[j] ... let me check.
        # np.correlate(v, f, 'valid') computes sum_m v[m+lag] * f[m] for lag=0..len(v)-len(f)
        # If len(v) = 2n-1 and len(f) = n, then 'valid' gives n-... hmm.
        # Actually 'valid' gives min(len(v),len(f)) - max(len(v),len(f)) + 1 = n - (2n-1) + 1 = 2-n.
        # That's negative for n>2. So 'valid' doesn't work directly.
        
        # Let me use 'full' mode: np.correlate(v, f, 'full') has length (2n-1)+(n-1) = 3n-2.
        # The element at index (2n-1-n+1+j) = (n+j) ... this is getting confusing.
        
        # Simplest correct approach for reasonable n:
        # sg[j] = 2 * sum_{m=0}^{n-1} w[j+m] * f[m]  (indices must be < 2n-1)
        
        # For n up to ~2000, this is O(n^2) per iteration which might be slow.
        # Let me use the FFT approach:
        # Define: extend f to length 2n-1 (pad with zeros), call it f_pad.
        # Then sum_m w[j+m]*f[m] = cross-correlation(w, f) at lag j.
        # cross-correlation can be computed as: ifft(conj(fft(w_pad)) * fft(f_pad))
        # where we pad both to the same length.
        
        # Actually, the simplest efficient approach:
        # reverse_f = f[::-1]
        # conv_w_revf = np.convolve(w, reverse_f, 'full')  # length (2n-1)+(n-1) = 3n-2
        # Hmm, this is getting complicated. Let me just use the direct computation for now.
        # For n=768, it's 768*768/2 ≈ 295k operations per iteration, which is fast.
        
        sg = np.zeros(n)
        for j in range(n):
            k_end = min(j + n, 2 * n - 1)
            sg[j] = 2.0 * np.dot(w[j:k_end], f[:k_end - j])
        
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f_new)

        temp = max(temp * temp_decay, 1e-7)
        cur_lr = lr * (0.9999 ** t)
        cur_lr = max(cur_lr, lr * 0.1)

        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            rng = np.random.default_rng(t * 17 + 3)
            noise_scale = 0.003 + 0.002 * min(no_improve // 100, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            temp = max(temp_init * 0.7, 0.003)
            cur_lr = lr
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with local search per coordinate."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_conv = fast_convolve(f)
    best_obj = best_conv.max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_conv.max()

            multipliers = [0.5, 0.7, 0.8, 1.2, 1.5, 2.0]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                f_trial = np.maximum(f_trial, 0)
                fss = f_trial.sum()
                if fss < 1e-10:
                    continue
                f_trial = f_trial / fss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                f[j] = best_j_val
                f = np.maximum(f, 0)
                f = f / f.sum()
                best_conv = fast_convolve(f)
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 2:
            break

    return best_f, best_obj


def make_gaussian_init(n, center=0.5, sigma=0.15):
    """Create a Gaussian-like initialization."""
    x = np.linspace(0, 1, n)
    f = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_symmetric_init(n, shape='bump'):
    """Create a symmetric initialization."""
    if shape == 'bump':
        x = np.linspace(0, 1, n)
        f = np.sin(np.pi * x) ** 2
    elif shape == 'triangle':
        x = np.linspace(0, 1, n)
        f = 1 - np.abs(2 * x - 1)
    else:
        f = np.ones(n)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
