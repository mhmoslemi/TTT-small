            # Infeasible, try looser
            m_factor = min(m_factor + 0.1, 5.0)
            M_try = M * (1.0 + 0.05 * m_factor)
            g = solve_lp_direction(f, M_try, n)
            if g is None:
                m_factor = min(m_factor + 0.2, 10.0)
                M_try = M * (1.0 + 0.1 * m_factor)
                g = solve_lp_direction(f, M_try, n)
                if g is None:
                    continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > max_no_improve:
                break
            m_factor = min(m_factor * 1.1, 5.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Refine with bisection
        if best_alpha > 0.01:
            lo = max(0.0, best_alpha - 0.04)
            hi = min(1.0, best_alpha + 0.04)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv_new = fast_convolve(f_new)
            M_new = conv_new.max()
            f = f_new
            M = M_new
            m_factor = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_factor = min(m_factor * 1.1, 5.0)

        if no_improve > max_no_improve:
            # Restart with perturbation
            scale = 0.002 + 0.003 * min(no_improve // max_no_improve, 5)
            rng = np.random.default_rng(it * 131 + 17)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_factor = 1.0
            no_improve = 0

    return best_f, best_obj


def smooth_obj_grad(t, beta):
    """Compute smooth log-sum-exp objective and gradient in t-space."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    obj = conv_max + np.log(exp_shifted.sum()) / beta

    # Gradient via FFT cross-correlation
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R
    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)
    return obj, grad_t


def lbfqg_optimize(f0, time_limit, verbose=False):
    """L-BFGS on annealed log-sum-exp smoothed objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    eps = 1e-15
    t = np.log(f + eps)
    betas = [5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0, 640.0]
    t0_time = time.monotonic()
    total_time = time_limit

    for bi, beta in enumerate(betas):
        remaining = total_time - (time.monotonic() - t0_time)
        if remaining < 20:
            break
        try:
            res = minimize(
                smooth_obj_grad, t, args=(beta,),
                jac=True, method='L-BFGS-B',
                options={'maxiter': 500, 'gtol': 1e-10, 'maxls': 50, 'maxfun': 5000}
            )
            t = res.x
            a = softmax_fn(t)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 1e-15:
                a = a / a_sum
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = a.copy()
                if verbose:
                    print(f"    L-BFGS beta={beta}: obj={best_obj:.10f}")
        except Exception:
            break

    return best_f, best_obj


def nesterov_smooth(f0, time_limit, verbose=False):
    """Nesterov-accelerated gradient on smoothed objective for escaping local optima."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    beta = 20.0
    lr = 0.01
    momentum = 0.9
    v = np.zeros(n)
    y = f.copy()  # Nesterov lookahead
    no_improve = 0
    step = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        # Nesterov step
        y_new = y + momentum * (y - f)
        # Gradient at y
        conv_y = fast_convolve(y)
        conv_max = conv_y.max()
        shifted = beta * (conv_y - conv_max)
        exp_s = np.exp(shifted)
        w = exp_s / exp_s.sum()

        N = 1 << (3 * n - 2).bit_length()
        A = np.fft.rfft(y, N)
        W = np.fft.rfft(w, N)
        R = np.fft.irfft(W * np.conj(A), N)[:n]
        grad = 2.0 * R

        f_new = project_simplex(y_new - lr * grad)
        
        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
            if verbose and step % 200 == 0:
                print(f"    Nesterov step {step}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        y = f_new
        # Decay learning rate
        lr *= 0.9995
        beta *= 1.001

        if no_improve > 200:
            rng = np.random.default_rng(step * 7 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            y = f.copy()
            v = np.zeros(n)
            lr = 0.005
            beta = 20.0
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / f.sum()
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-13:
                continue
            best_j_val = old_val
            best_j_obj = best_obj
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        if not improved:
            break
    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.002, mom=0.94, top_k=50, temp=5e-5):
    """Subgradient polishing with softmax-weighted peaks and momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        cur_temp *= 1.003

        if no_improve > 400:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_structured_initial(n, seed=42):
    """Create structured initial points with various shapes."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    candidates = []
    # Cosine window
    candidates.append(0.5 * (1 + np.cos(2 * np.pi * x)))
    # Triangular
    candidates.append(1 - np.abs(2 * x - 1))
    # Gaussian center
    candidates.append(np.exp(-4 * (x - 0.5) ** 2))
    # Sinc-like
    t = (x - 0.5) * n * 0.5
    sinc = np.sinc(t / (n * 0.1))
    candidates.append(np.maximum(sinc, 0))
    # Ramp up, flat, ramp down
    ramp = np.ones(n)
    edge = max(1, n // 8)
    ramp[:edge] = np.linspace(0.3, 1, edge)
    ramp[-edge:] = np.linspace(1, 0.3, edge)
    candidates.append(ramp)
    # Random smooth
    noise = rng.standard_normal(n)
    # Simple low-pass via repeated averaging
    for _ in range(3):
        noise = (noise + np.roll(noise, 1) + np.roll(noise, -1)) / 3
    candidates.append(np.maximum(noise - noise.min() + 0.1, 0))
    result = []
    for c in candidates:
        c = np.maximum(c, 0)
        s = c.sum()
        if s > 1e-10:
            result.append(c / s)
    return result


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: LP-FW, L-BFGS, Nesterov, multi-length,
    coordinate descent, and subgradient polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512

    # === Phase 1: LP-FW on parent (350s) ===
    print("\n=== Phase 1: LP-FW on parent ===")
    t1 = min(350, deadline - time.monotonic() - 900)
    if t1 > 30:
        a_opt, _ = lp_fw_optimize(parent, time_limit=t1, verbose=True, m_shrink=0.993)
        try_seq(a_opt, "lpfw_parent")

    # Try reversed
    t1r = min(80, deadline - time.monotonic() - 800)
    if t1r > 20:
        a_opt, _ = lp_fw_optimize(parent[::-1].copy(), time_limit=t1r)
        try_seq(a_opt, "lpfw_reversed")

    # === Phase 2: L-BFGS (300s) ===
    print(f"\n=== Phase 2: L-BFGS (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    t2 = min(300, deadline - time.monotonic() - 700)
    if t2 > 40:
        a_opt, _ = lbfqg_optimize(base, time_limit=t2, verbose=True)
        try_seq(a_opt, "lbfqg_main")
        # Perturbed
        t2b = min(100, deadline - time.monotonic() - 550)
        if t2b > 20:
            noise = rng.standard_normal(len(base)) * 0.008
            f_pert = project_simplex(base + noise)
            a_opt, _ = lbfqg_optimize(f_pert, time_limit=t2b)
            try_seq(a_opt, "lbfqg_perturbed")

    # === Phase 3: Multi-length exploration (350s) ===
    print(f"\n=== Phase 3: Multi-length (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    target_lengths = [384, 512, 768, 1024, 1280, 1536]
    for target_n in target_lengths:
        if target_n == len(base):
            continue
        rem = deadline - time.monotonic() - 400
        if rem < 30:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.35, 70)
        print(f"  Trying n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc, m_shrink=0.995)
        try_seq(a_opt, f"len_{target_n}")

    # Also try structured inits at some lengths
    for target_n in [512, 768, 1024]:
        rem = deadline - time.monotonic() - 350
        if rem < 25:
            break
        inits = make_structured_initial(target_n, seed=seed + target_n)
        for ii, init in enumerate(inits[:2]):
            rem2 = deadline - time.monotonic() - 300
            if rem2 < 15:
                break
            t_alloc = min(rem2 * 0.4, 40)
            a_opt, _ = lp_fw_optimize(init, time_limit=t_alloc)
            try_seq(a_opt, f"struct_{target_n}_{ii}")

    # === Phase 4: Nesterov on best (150s) ===
    print(f"\n=== Phase 4: Nesterov (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    t4 = min(150, deadline - time.monotonic() - 300)
    if t4 > 20:
        a_opt, _ = nesterov_smooth(base, time_limit=t4, verbose=True)
        try_seq(a_opt, "nesterov_main")

    # === Phase 5: Deep LP-FW on best (300s) ===
    print(f"\n=== Phase 5: Deep LP-FW (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(rem * 0.6, 180)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True,
                                      m_shrink=0.992, max_no_improve=100)
            try_seq(a_opt, "deep_lpfw")

    # === Phase 6: Deep L-BFGS (150s) ===
    print(f"\n=== Phase 6: Deep L-BFGS (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_alloc = min(rem * 0.7, 120)
            a_opt, _ = lbfqg_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_lbfqg")

    # === Phase 7: Coordinate descent (100s) ===
    print(f"\n=== Phase 7: Coord Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 15:
            t_alloc = min(rem * 0.8, 80)
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "deep_cd")

    # === Phase 8: Final polishing loop (remaining - 30s) ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        attempt = 0
        while time.monotonic() < deadline - 15:
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            attempt += 1

            if attempt % 3 == 1:
                # Small perturbation + LP-FW
                noise = rng.standard_normal(n_best) * 0.002
                f_start = project_simplex(best_seq + noise)
                t_alloc = min(rem * 0.6, 50)
                a_opt, _ = lp_fw_optimize(f_start, time_limit=t_alloc)
                try_seq(a_opt, f"final_lpfw_{attempt}")
            elif attempt % 3 == 2:
                # L-BFGS
                t_alloc = min(rem * 0.5, 40)
                a_opt, _ = lbfqg_optimize(best_seq, time_limit=t_alloc)
                try_seq(a_opt, f"final_lbfqg_{attempt}")
            else:
                # SG polish
                t_alloc = min(rem * 0.4, 30)
                a_opt, _ = sg_polish(best_seq, time_limit=t_alloc,
                                     lr=0.001, mom=0.95,
                                     top_k=max(20, n_best // 8),
                                     temp=3e-5)
                try_seq(a_opt, f"final_sg_{attempt}")

    # Final CD pass
    if best_seq is not None and time.monotonic() < deadline - 5:
        rem = min(deadline - time.monotonic() - 3, 25)
        if rem > 5:
            a_opt, _ = coordinate_descent(best_seq, time_limit=rem)
            try_seq(a_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
