def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Compute 2*n*max(conv(p,p)) for probability vector p (sum=1)."""
    n = len(p)

            k_minus_j = k - np.arange(n)
            valid = (k_minus_j >= 0) & (k_minus_j < n)
            sg[valid] += 2.0 * w[i] * p[k_minus_j[valid]]

        # Multiplicative update
        p = p * np.exp(-lr * sg)
        p = np.maximum(p, 1e-15)
        p = p / p.sum()

        # Check improvement
        conv = fast_convolve(p)
        obj = conv.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Learning rate and temperature scheduling
        lr *= 0.9995
        temperature *= 1.01

        # Asymmetry check: if sequence becomes too symmetric, boost asymmetry
        if it % 50 == 0:
            sym_diff = np.abs(p - p[::-1]).sum()
            if sym_diff < 0.01:
                # Add asymmetry
                rng = np.random.default_rng(it * 7)
                drift = rng.standard_normal(n) * 0.001
                drift = drift - drift.mean()  # zero mean to preserve sum
                p = p + drift
                p = np.maximum(p, 1e-15)
                p = p / p.sum()

        # Restart if stuck
        if no_improve > 200:
            rng = np.random.default_rng(it * 13 + 3)
            noise = rng.standard_normal(n) * 0.003
            p = np.maximum(best_p + noise, 1e-15)
            p = p / p.sum()
            lr = 0.002
            no_improve = 0

        if lr < 1e-7:
            break

    return best_p, best_obj


def mass_shifting(p0, time_limit, verbose=False):
    """
    Mass-shifting heuristic: iteratively moves mass from peak-contributing
    positions to positions that reduce the overall maximum.
    """
    p = np.maximum(p0.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(len(p0)) / len(p0)
    else:
        p = p / s
    n = len(p)

    conv = fast_convolve(p)
    best_obj = conv.max()
    best_p = p.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(p)
        peak_k = int(np.argmax(conv))
        peak_val = conv.max()
        improved = False

        # Find positions contributing to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        # p[j] * p[peak_k - j] contributes to the peak

        # Find the position with the largest contribution
        idx = np.arange(js, je + 1)
        contributions = p[idx] * p[peak_k - idx]
        j_worst = idx[np.argmax(contributions)]

        # Try moving mass from j_worst to various other positions
        old_val = p[j_worst]
        if old_val < 1e-10:
            break

        # Candidate positions to move mass to: those NOT contributing to the peak
        non_contributing = np.ones(n, dtype=bool)
        non_contributing[peak_k - js:peak_k - js + (je - js + 1)] = False
        # Actually, position j contributes to peak at k if j is in [max(0,k-n+1), min(n-1,k)]
        # AND k-j is in [0, n-1]. So j contributes if max(0,k-n+1) ≤ j ≤ min(n-1,k)
        non_contributing[js:je+1] = False

        if non_contributing.sum() == 0:
            break

        # Choose target: position with smallest current value among non-contributing
        target_candidates = np.where(non_contributing)[0]
        # Prefer targets that are far from peak_k (to minimize new convolution values)
        # Score: -p[target] + distance to peak (prefer far and small)
        dist = np.abs(target_candidates - peak_k)
        dist = dist / (n + 1)  # normalize
        score = -p[target_candidates] + 0.1 * dist
        target = target_candidates[np.argmax(score)]

        # Try different amounts of mass transfer
        best_trial = None
        best_trial_obj = peak_val

        for frac in [0.1, 0.2, 0.3, 0.5, 0.7]:
            transfer = old_val * frac
            trial = p.copy()
            trial[j_worst] = old_val - transfer
            trial[target] = p[target] + transfer
            trial = np.maximum(trial, 0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            trial = trial / ts
            obj = fast_convolve(trial).max()
            if obj < best_trial_obj - 1e-12:
                best_trial_obj = obj
                best_trial = trial

        if best_trial is not None:
            p = best_trial
            if best_trial_obj < best_obj - 1e-12:
                best_obj = best_trial_obj
                best_p = p.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 10:
                # Try a different peak or a different strategy
                # Move mass from the second-worst peak contributor
                peak_indices = np.argsort(conv)[-3:]
                moved = False
                for pk in peak_indices:
                    pk = int(pk)
                    js2 = max(0, pk - n + 1)
                    je2 = min(n - 1, pk)
                    if je2 <= js2:
                        continue
                    idx2 = np.arange(js2, je2 + 1)
                    contribs2 = p[idx2] * p[pk - idx2]
                    j2 = idx2[np.argmax(contribs2)]
                    old2 = p[j2]
                    if old2 < 1e-10:
                        continue

                    non_contr2 = np.ones(n, dtype=bool)
                    non_contr2[js2:je2+1] = False
                    if non_contr2.sum() == 0:
                        continue
                    tc = np.where(non_contr2)[0]
                    # Pick the position with smallest value
                    target2 = tc[np.argmin(p[tc])]

                    for frac in [0.2, 0.4, 0.6]:
                        transfer = old2 * frac
                        trial = p.copy()
                        trial[j2] = old2 - transfer
                        trial[target2] = p[target2] + transfer
                        trial = np.maximum(trial, 0)
                        ts = trial.sum()
                        if ts < 1e-15:
                            continue
                        trial = trial / ts
                        obj = fast_convolve(trial).max()
                        if obj < best_obj - 1e-12:
                            p = trial
                            best_obj = obj
                            best_p = p.copy()
                            improved = True
                            moved = True
                            break
                    if moved:
                        break

                if not improved:
                    # Perturb and try again
                    rng = np.random.default_rng(hash(str(no_improve)) % 10000 + it)
                    noise = rng.standard_normal(n) * 0.002
                    p = np.maximum(best_p + noise, 1e-15)
                    p = p / p.sum()
                    no_improve = 0
            else:
                break
        else:
            no_improve = 0

    return best_p, best_obj


def coordinate_descent_v2(p0, time_limit, verbose=False):
    """
    Coordinate descent that specifically targets peak reduction.
    For each coordinate, finds the optimal value that minimizes the max convolution.
    """
    p = np.maximum(p0.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(len(p0)) / len(p0)
    else:
        p = p / s
    n = len(p)

    best_p = p.copy()
    best_obj = fast_convolve(p).max()
    t0 = time.monotonic()
    epoch = 0

    while time.monotonic() - t0 < time_limit:
        epoch += 1
        improved = False

        # Process coordinates in order of importance (peak contributors first)
        conv = fast_convolve(p)
        peak_k = int(np.argmax(conv))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)

        # Order: peak contributors first, then others
        order = list(range(js, je + 1)) + [i for i in range(n) if i < js or i > je]

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = p[j]
            best_j_obj = best_obj
            best_j_val = old_val

            # Try a range of new values
            # The optimal value is where the peak from this coordinate balances with others
            search_range = np.linspace(0.0, min(old_val * 3, 0.5), 15)
            # Also try values near the current one
            if old_val > 0:
                search_range = np.unique(np.concatenate([
                    search_range,
                    old_val * np.array([0.3, 0.5, 0.7, 0.8, 1.2, 1.5, 2.0])
                ]))
                search_range = search_range[(search_range >= 0) & (search_range <= 0.5)]

            for new_val in search_range:
                if abs(new_val - old_val) < 1e-10:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                trial = p.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial = trial / ss

                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-12:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    p[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    p[mask] *= scale
                    p = np.maximum(p, 0)
                    p = p / p.sum()
                if best_j_obj < best_obj - 1e-12:
                    best_obj = best_j_obj
                    best_p = p.copy()
                    improved = True

        if not improved:
            # Try random restart
            rng = np.random.default_rng(epoch * 37 + 11)
            noise = rng.standard_normal(n) * 0.001
            p = np.maximum(best_p + noise, 1e-15)
            p = p / p.sum()

        if epoch > 100:
            break

    return best_p, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization exploiting asymmetry to minimize

    rng = np.random.default_rng(seed)

    best_p = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_p, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:

            best_p = a.copy()
