"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using a hybrid approach:
LP-FW with lazy constraints, peak-specific coordinate descent,
multiplicative weights (mirror descent), and diverse restarts.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_at_peak(f, k_star):
    """Compute subgradient of (f*f)[k_star] with respect to f."""
    n = len(f)
    sg = np.zeros(n)
    for j in range(n):
        val = 0.0
        if j + k_star < n:
            val += f[j + k_star]
        if 0 <= j - k_star:
            val += f[j - k_star]
        sg[j] = 2.0 * val
    return sg

def lp_direction_lazy(f, M, n, top_k=30):

    conv = fast_convolve(f)
    # Get top_k peak locations
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    c = -np.ones(n)
    A_ub_rows = []
    b_ub_vals = []
    
    for k in peaks:
        row = np.zeros(n)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            row[j] = f[k - j]
        A_ub_rows.append(row)
        b_ub_vals.append(M)
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'time_limit': 30, 'presolve': True})
    if res.success and res.x is not None:
        return res.x
    return None

            # Redistribute to coordinates with smallest sg
            others = np.delete(np.arange(n), j_best)
            others_sorted = others[np.argsort(sg[others])]
            # Give to the ones with smallest impact
            top_others = others_sorted[:max(1, n // 4)]
            weights = 1.0 / (sg[top_others] + 1e-10)
            weights = weights / weights.sum()

            f_new[j_best] = new_val
            f_new[top_others] += diff * weights
            f_new = np.maximum(f_new, 0)
            ss = f_new.sum()
            if ss < 1e-15:
                continue
            f_new = f_new / ss
            
            val = fast_convolve(f_new).max()
            if val < best_obj - 1e-16:
                best_obj = val
                best_f = f_new.copy()
                f = f_new
                improved = True
                break
        
        if not improved:
            # Try increasing a coordinate with small sg
            j_small = int(np.argmin(sg))

                new_val = old_val * mult
                if new_val > 0.5:
                    continue
                diff = new_val - old_val
                # Take from coordinates with large sg
                others = np.delete(np.arange(n), j_small)
                others_sorted = others[np.argsort(-sg[others])]
                top_others = others_sorted[:max(1, n // 4)]
                weights = sg[top_others].copy()
                total = weights.sum()
                if total < 1e-15:
                    break
                weights = weights / total
                # Need to ensure we don't go negative
                max_frac = np.minimum(f[top_others] / (weights * diff + 1e-15), 1.0)
                actual_frac = max_frac.min()
                if actual_frac < 0.01:
                    continue
                
                f_new = f.copy()
                f_new[j_small] = new_val
                f_new[top_others] -= actual_frac * diff * weights
                f_new = np.maximum(f_new, 0)
                ss = f_new.sum()
                if ss < 1e-15:
                    continue
                f_new = f_new / ss
                
                val = fast_convolve(f_new).max()

                    best_f = f_new.copy()
                    f = f_new

                best_f = f.copy()

def multiplicative_weights(f0, time_limit, eta0=0.05, decay=0.9995, verbose=False):

    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    conv = fast_convolve(f)
    best_obj = conv.max()
    eta = eta0
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Use top peaks for subgradient
        k_actual = min(10, 2 * n - 1)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        # Softmax weights
        scaled = (pv - pv.max()) / (eta * 0.1 + 1e-10)
        w = np.exp(scaled)
        w = w / w.sum()
        
        # Compute weighted subgradient
        sg = np.zeros(n)
        for i in range(k_actual):
            if w[i] < 1e-10:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                sg[j] += 2.0 * w[i] * f[k - j]
        
        # Multiplicative update
        f_new = f * np.exp(-eta * sg)
        f_new = np.maximum(f_new, 0)
        ss = f_new.sum()
        if ss < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / ss
        
        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()
        
        if M_new < best_obj - 1e-16:
            best_obj = M_new
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 1000 == 0:
                print(f"    MW iter {it}: obj={best_obj:.12f}")
        else:
            no_improve += 1
        
        f = f_new
        eta *= decay
        
        if no_improve > 200:
            rng = np.random.default_rng(it * 31 + 17)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            eta = max(eta0 * 0.3, 0.001)
            no_improve = 0
    
    return best_f, best_obj

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using hybrid approach:
LP-FW with lazy constraints, vectorized peak-shaving coordinate descent,
multiplicative weights (mirror descent), and diverse restarts.
Key innovations: vectorized subgradients, adaptive peak targeting,

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgrad_weighted(f, peaks, weights):
    """Vectorized computation of weighted subgradient at multiple peaks."""
    n = len(f)
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-12:
            continue
        k = int(peaks[i])
        w = 2.0 * weights[i]
        if k == 0:
            sg += w * f
        else:
            sg[:n - k] += w * f[k:]
            sg[k:] += w * f[:n - k]
    return sg


def lp_direction_lazy(f, M, n, top_k=30):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for top_k peaks, g >= 0."""
    conv = fast_convolve(f)
    total_len = len(conv)
    top_k = min(top_k, total_len)
    peaks = np.argpartition(conv, -top_k)[-top_k:]

    A_ub_rows = []
    b_ub_vals = []

    for k in peaks:
        row = np.zeros(n)
        k_int = int(k)
        js = max(0, k_int - n + 1)
        je = min(n - 1, k_int)
        idx = np.arange(js, je + 1)
        row[idx] = f[k_int - idx]
        A_ub_rows.append(row)
        b_ub_vals.append(M)

    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'time_limit': 25, 'presolve': True})
    if res.success and res.x is not None:
        return res.x
    return None

def peak_shave_descent(f0, time_limit, verbose=False):
    """Vectorized peak-shaving coordinate descent targeting the dominant peak."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        k_star = int(np.argmax(conv))

        # Compute subgradient at the peak
        if k_star == 0:
            sg = 2.0 * f
        else:
            sg = np.zeros(n)
            sg[:n - k_star] += 2.0 * f[k_star:]
            sg[k_star:] += 2.0 * f[:n - k_star]

        improved = False

        # Strategy 1: Decrease the coordinate with largest marginal impact
        # on the peak, redistribute to lowest-impact coordinates
        j_dec = int(np.argmax(sg))
        old_val = f[j_dec]

        if old_val > 1e-10:
            for frac in [0.3, 0.5, 0.7, 0.8, 0.9]:
                new_val = old_val * (1.0 - frac)
                diff = old_val - new_val

                # Redistribute to coordinates with smallest sg
                others = np.delete(np.arange(n), j_dec)
                others_sorted = others[np.argsort(sg[others])]
                n_redist = max(1, n // 5)
                redist_idx = others_sorted[:n_redist]

                w_redist = 1.0 / (sg[redist_idx] + 1e-8)
                w_sum = w_redist.sum()
                if w_sum < 1e-15:
                    continue
                w_redist = w_redist / w_sum

                f_new = f.copy()
                f_new[j_dec] = new_val
                f_new[redist_idx] += diff * w_redist
                f_new = np.maximum(f_new, 0.0)
                ss = f_new.sum()
                if ss < 1e-15:
                    continue
                f_new = f_new / ss

                val = fast_convolve(f_new).max()
                if val < best_obj - 1e-16:
                    best_obj = val
                    best_f = f_new.copy()
                    f = f_new
                    improved = True
                    break

        # Strategy 2: Spread out - decrease multiple high-sg coordinates
        if not improved:
            top_dec = np.argsort(-sg)[:max(1, n // 8)]
            bottom_inc = np.argsort(sg)[:max(1, n // 8)]
            # Ensure no overlap
            overlap = set(top_dec) & set(bottom_inc)
            if overlap:
                bottom_inc = np.array([x for x in bottom_inc if x not in overlap])
                if len(bottom_inc) == 0:
                    bottom_inc = np.array([0])

            for frac in [0.1, 0.2, 0.3, 0.5]:
                total_dec = 0.0
                f_new = f.copy()
                for j in top_dec:
                    dec_amount = f[j] * frac
                    f_new[j] -= dec_amount
                    total_dec += dec_amount

                f_new = np.maximum(f_new, 0.0)
                w_inc = 1.0 / (sg[bottom_inc] + 1e-8)
                w_inc = w_inc / w_inc.sum()
                f_new[bottom_inc] += total_dec * w_inc

                ss = f_new.sum()
                if ss < 1e-15:
                    continue
                f_new = f_new / ss
                val = fast_convolve(f_new).max()
                if val < best_obj - 1e-16:
                    best_obj = val
                    best_f = f_new.copy()
                    f = f_new
                    improved = True
                    break

        # Strategy 3: Small multiplicative perturbation
        if not improved and it % 5 == 0:
            rng = np.random.default_rng(it * 31627 + 3)
            perturb = 1.0 + rng.standard_normal(n) * 0.01
            f_new = f * perturb
            f_new = np.maximum(f_new, 0.0)
            ss = f_new.sum()
            if ss > 1e-15:
                f_new = f_new / ss
                val = fast_convolve(f_new).max()
                if val < best_obj - 1e-16:
                    best_obj = val
                    best_f = f_new.copy()
                    f = f_new
                else:
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.002)

    return best_f, best_obj


def multiplicative_weights_opt(f0, time_limit, eta0=0.03, decay=0.9995,
                               n_peaks=8, verbose=False):
    """Mirror descent with multiplicative (Exponentiated Gradient) updates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    conv = fast_convolve(f)
    best_obj = conv.max()
    eta = eta0
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Select top peaks with softmax weighting
        kp = min(n_peaks, 2 * n - 1)
        peaks = np.argpartition(conv, -kp)[-kp:]
        pv = conv[peaks]
        temp = max((pv.max() - pv.min()) * 0.3, 1e-10)
        scaled = (pv - pv.max()) / temp
        w = np.exp(scaled)
        w = w / w.sum()

        sg = compute_subgrad_weighted(f, peaks, w)

        # Multiplicative update
        f_new = f * np.exp(-eta * sg)
        f_new = np.maximum(f_new, 0.0)
        ss = f_new.sum()
        if ss < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / ss

        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()

        if M_new < best_obj - 1e-16:
            best_obj = M_new
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    MW iter {it}: obj={best_obj:.12f}")
        else:
            no_improve += 1

        f = f_new
        eta *= decay

        if no_improve > 300:
            rng = np.random.default_rng(it * 104729 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            eta = max(eta0 * 0.4, 0.002)
            no_improve = 0

    return best_f, best_obj


def simulate_annealing(f0, time_limit, T0=0.001, T_decay=0.999, verbose=False):
    """Simulated annealing on the simplex for local minima escape."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    T = T0
    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Propose a move: swap mass between two random coordinates
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = rng.uniform(-0.01, 0.01)
        f_new = f.copy()
        f_new[i] = max(0.0, f[i] + delta)
        f_new[j] = max(0.0, f[j] - delta)
        ss = f_new.sum()
        if ss < 1e-15:
            continue
        f_new = f_new / ss

        obj_new = fast_convolve(f_new).max()
        delta_obj = obj_new - best_obj

        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(T, 1e-10)):
            f = f_new
            if obj_new < best_obj - 1e-16:
                best_obj = obj_new
                best_f = f_new.copy()
        T *= T_decay

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid multi-phase optimizer combining LP-FW (lazy), peak-shaving CD,
    multiplicative weights, simulated annealing, and multi-length search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.6f}")

    n = len(best_seq)

    # === Phase 1: Main LP-FW with moderate top_k ===
    print(f"\n=== Phase 1: LP-FW (n={n}, top_k=30) ===")
    t_rem = deadline - time.monotonic()
    t1 = min(350, t_rem * 0.22)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t1, verbose=True, top_k=30)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")
        else:
            print(f"  LP-FW score: {sc:.6f}")

    # === Phase 2: Peak-shaving coordinate descent ===
    print(f"\n=== Phase 2: Peak Shaving CD (best={best_score:.6f}) ===")
    t_rem = deadline - time.monotonic()
    t2 = min(180, t_rem * 0.12)
    if t2 > 20:
        a_opt, _ = peak_shave_descent(best_seq, time_limit=t2)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")
        else:
            print(f"  Peak CD score: {sc:.6f}")

    # === Phase 3: Multiplicative weights fine-tuning ===
    print(f"\n=== Phase 3: Multiplicative Weights (best={best_score:.6f}) ===")
    t_rem = deadline - time.monotonic()
    t3 = min(120, t_rem * 0.1)
    if t3 > 15:
        a_opt, _ = multiplicative_weights_opt(best_seq, time_limit=t3,
                                              eta0=0.02, decay=0.9997, n_peaks=6)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 4: LP-FW with different top_k ===
    print(f"\n=== Phase 4: LP-FW varied top_k (best={best_score:.6f}) ===")
    for tk in [15, 50]:
        t_rem = deadline - time.monotonic()
        if t_rem < 40:
            break
        t4 = min(130, t_rem * 0.15)
        if t4 < 15:
            continue
        print(f"  top_k={tk}, t={t4:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t4, top_k=tk)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 5: Peak shaving again ===
    print(f"\n=== Phase 5: Peak Shaving round 2 (best={best_score:.6f}) ===")
    t_rem = deadline - time.monotonic()
    t5 = min(120, t_rem * 0.1)
    if t5 > 15:
        a_opt, _ = peak_shave_descent(best_seq, time_limit=t5)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 6: Simulated annealing for exploration ===
    print(f"\n=== Phase 6: Simulated Annealing (best={best_score:.6f}) ===")
    t_rem = deadline - time.monotonic()
    t6 = min(80, t_rem * 0.08)
    if t6 > 10:
        a_opt, _ = simulate_annealing(best_seq, time_limit=t6, T0=0.0005, T_decay=0.9995)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-length (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    for delta in [-64, 64, -128, 128, 256, -32, 32]:
        t_rem = deadline - time.monotonic() - 80
        if t_rem < 20:
            break
        nl = n_best + delta
        if nl < 100 or nl > 2048:
            continue
        f_interp = interpolate_sequence(best_seq, nl)
        t_alloc = min(80, t_rem * 0.2)
        if t_alloc < 10:
            continue
        print(f"  n={nl}, t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc, top_k=30)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            n = nl
            print(f"  NEW BEST: {sc:.6f} (n={nl})")

    # === Phase 8: Final deep LP-FW ===
    print(f"\n=== Phase 8: Final LP-FW (best={best_score:.6f}) ===")
    t_rem = deadline - time.monotonic()
    if t_rem > 30:
        t8 = min(t_rem * 0.4, 100)
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t8, top_k=40)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 9: Final peak shaving + MW ===
    print(f"\n=== Phase 9: Final polish (best={best_score:.6f}) ===")
    t_rem = deadline - time.monotonic()
    if t_rem > 20:
        t9a = min(t_rem * 0.4, 50)
        a_opt, _ = peak_shave_descent(best_seq, time_limit=t9a)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    t_rem = deadline - time.monotonic()
    if t_rem > 15:
        t9b = min(t_rem * 0.5, 40)
        a_opt, _ = multiplicative_weights_opt(best_seq, time_limit=t9b,
                                              eta0=0.008, decay=0.9999, n_peaks=4)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 10: Micro perturbations ===
    print(f"\n=== Phase 10: Micro perturbations (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    for trial in range(20):
        t_rem = deadline - time.monotonic()
        if t_rem < 3:
            break
        scale = 0.0001 + trial * 0.00008
        noise = rng.standard_normal(n_best) * scale
        f_pert = project_simplex(best_seq + noise)
        sc = float(evaluate_sequence(f_pert.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = f_pert.copy()
            print(f"  NEW BEST [pert{trial}]: {sc:.6f}")
            t_rem2 = deadline - time.monotonic()
            if t_rem2 > 5:
                a_opt, _ = multiplicative_weights_opt(
                    f_pert, time_limit=min(t_rem2 * 0.3, 8),
                    eta0=0.003, decay=0.9999, n_peaks=4)
                sc2 = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = a_opt.copy()
                    print(f"  NEW BEST [pert{trial}_mw]: {sc2:.6f}")

    # Final LP-FW squeeze with remaining time
    t_rem = deadline - time.monotonic()
    if t_rem > 10:
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(t_rem * 0.8, 30), top_k=35)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  FINAL LP-FW: {sc:.6f}")

    # Final verification
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
