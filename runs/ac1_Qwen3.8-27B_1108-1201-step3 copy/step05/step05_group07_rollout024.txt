log-softmax L-BFGS with temperature annealing, and coordinate descent polishing.
Key novel contribution: iterative LP-based refinement that finds the optimal
sequence fitting under the current peak envelope, guaranteed to decrease
the objective. Combined with gradient-based polishing and multi-length search.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def objective_value(a):
    """Compute the exact objective R = 2n*max(conv(a,a))/(sum a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s * s)


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    return np.log(np.maximum(a, 1e-300))


def soft_max_obj_grad(z, T, n):
    """
    Compute the soft-max smoothed objective and gradient w.r.t. z.
    Objective: (2n/T) * log(sum_k exp(T*b_k)) where b = conv(softmax(z), softmax(z))
    """
    a = softmax_from_z(z)
    b = fast_convolve(a)

    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp

    w = exp_Tb / exp_Tb.sum()

    # Gradient: d/dz of sum_k w_k * c[k] where c[k] = sum_j a[j]*a[k-j]
    # d c[k]/d a[i] = 2*a[k-i] if 0<=k-i<n (from both factors)
    # But more carefully: for the full convolution of a with itself,
    # c[k] = sum_{j valid} a[j]*a[k-j]
    # d c[k]/d a[i] = a[k-i] (from first factor, j=i) + a[k-i] (from second factor, k-j=i)
    # = 2*a[k-i] if 0 <= k-i < n, else 0
    # Wait, need to be more careful for the boundary.
    # c[k] = sum_{j: max(0,k-n+1)<=j<=min(n-1,k)} a[j]*a[k-j]
    # d/d a[i]: first term (j=i) valid if max(0,k-n+1)<=i<=min(n-1,k), contributes a[k-i]
    #           second term (k-j=i, i.e., j=k-i) valid if max(0,k-n+1)<=k-i<=min(n-1,k), contributes a[k-i]
    # Both conditions are equivalent to 0<=k-i<n and 0<=i<n (and 0<=k<n OR 0<=k-i<n which is same thing)
    # Actually both are valid iff 0<=i<n and 0<=k-i<n.
    # So d c[k]/d a[i] = 2*a[k-i] if 0<=i<n and 0<=k-i<n, else 0.
    
    # Total gradient of sum_k w_k c[k] w.r.t. a[i]:
    # = 2 * sum_{k: 0<=k-i<2n-1} w[k] * a[k-i] (with 0<=i<n)
    # = 2 * sum_{j: 0<=j<n, 0<=i+j<2n-1} w[i+j] * a[j]
    # = 2 * cross_correlation(w, a)[i] where cross[i] = sum_j w[i+j]*a[j]
    
    # Compute via FFT:
    # cross[i] = sum_{j=0}^{n-1} w[i+j] * a[j] for i = 0, ..., 2n-2
    # Let w_rev[k] = w[2n-2-k] (reverse w)
    # cross[i] = sum_j a[j] * w[i+j] = sum_j a[j] * w_rev[2n-2-i-j]
    # Linear convolution: (a * w_rev)[m] = sum_j a[j] * w_rev[m-j]
    # So cross[i] = (a * w_rev)[2n-2-i]
    
    w_rev = w[::-1]
    conv_len = n + (2*n-1) - 1  # = 3n-2
    fft_size2 = 1 << (conv_len - 1).bit_length()
    A_fft = rfft(a, fft_size2)
    W_fft = rfft(w_rev, fft_size2)
    conv_result = irfft(A_fft * W_fft, fft_size2)[:conv_len]
    
    # cross[i] = conv_result[2n-2-i] for i=0..n-1
    # cross[0] = conv_result[2n-2], cross[n-1] = conv_result[n-1]
    grad_a = 2.0 * conv_result[n-1:2*n-1][::-1]
    
    # Chain rule for softmax: grad_z[j] = a[j]*(grad_a[j] - sum_i grad_a[i]*a[i])
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)

    return obj, grad_z


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lbgfs_optimize(z0, T, n, time_limit, maxiter=3000):
    """L-BFGS in log-space with time limit enforcement."""
    t0 = time.monotonic()
    elapsed = [t0]

    def func(z):
        elapsed[0] = time.monotonic()
        if elapsed[0] - t0 > time_limit:
            return 1e15
        obj, _ = soft_max_obj_grad(z, T, n)
        return obj

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = soft_max_obj_grad(z, T, n)
        return g

    try:
        res = minimize(
            fun=func,
            x0=z0,
            jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 25, 'disp': False}
        )
        return res.x, res.fun
    except Exception:
        return z0, func(z0)


def lp_refine(f, time_limit, max_iter=200, t=0.3):
    """
    LP-based envelope refinement.
    Given f (normalized to sum 1), iteratively:
    1. Compute M = max(f*f)
    2. Solve LP: max sum(g) s.t. g>=0, f*g <= M
    3. Normalize g, interpolate f <- (1-t)*f + t*g
    """
    f = f.copy().astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f = f / s
    n = len(f)
    t0 = time.monotonic()

    best_f = f.copy()
    best_max = fast_convolve(f).max()
    
    # Pre-build the constraint matrix structure
    # A[k, i] = f_norm[k-i] if 0 <= k-i < n, else 0
    # But f changes at each iteration, so we rebuild each time.
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = fast_convolve(f).max()
        if M <= best_max * (1 - 1e-12):
            best_max = M
            best_f = f.copy()
        
        # Build constraint matrix A (2n-1 x n)
        A = np.zeros((2*n-1, n))
        for i in range(n):
            A[i:i+n, i] = f
        
        # LP: minimize -sum(g) subject to A*g <= M, g >= 0
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs',
                         options={'time_limit': max(1, time_limit - (time.monotonic()-t0)),
                                  'presolve': True})
        except Exception:
            break
        
        if not res.success or res.x is None:
            break
        
        g = np.maximum(res.x, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        
        g_norm = g / g_sum  # sum = 1
        
        # Try different interpolation parameters
        improved = False
        for t_try in [t, 0.5, 0.2, 0.1]:
            f_new = (1 - t_try) * f + t_try * g_norm
            obj_new = fast_convolve(f_new).max()
            if obj_new < best_max - 1e-14:
                f = f_new
                best_max = obj_new
                best_f = f.copy()
                improved = True
                break
        
        if not improved:
            # Try very small step
            f_try = f + 0.01 * (g_norm - f)
            obj_try = fast_convolve(f_try).max()
            if obj_try < best_max - 1e-14:
                f = f_try
                best_max = obj_try
                best_f = f.copy()
            else:
                break
    
    return best_f, best_max


def gradient_polish(a0, time_limit, temps=None):
    """Polish using soft-max L-BFGS with temperature annealing."""
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    if s < 1e-15:
        return a0
    a = a / s
    z = z_from_a(a)
    
    if temps is None:
        temps = [5, 10, 20, 50, 100, 200]
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    for T in temps:
        t_rem = time_limit - (time.monotonic() - time_limit + time_limit)  
        # Compute remaining time properly
        elapsed_so_far = 0  # We'll track externally
        pass
    
    # Simpler approach: allocate time per temperature
    n_temps = len(temps)
    time_per_T = time_limit / n_temps
    
    for i, T in enumerate(temps):
        remaining = (n_temps - i) * time_per_T
        z_new, _ = lbgfs_optimize(z, T, n, remaining * 0.95, maxiter=2000)
        a_new = softmax_from_z(z_new)
        obj_new = fast_convolve(a_new).max()
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            z = z_new
        else:
            z = z_new  # Still update z for next temperature
    
    return best_a


def coordinate_descent(a0, time_limit):
    """Cycle through coordinates with line search on multipliers."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    multipliers = np.array([0.5, 0.7, 0.8, 1.0, 1.2, 1.3, 1.5, 1.8])
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                denom = 1.0 - old_val
                if abs(denom) < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                # Ensure non-negative
                if a_trial.min() < -1e-10:
                    continue
                a_trial = np.maximum(a_trial, 0)
                a_trial_sum = a_trial.sum()
                if a_trial_sum < 1e-10:
                    continue
                a_trial = a_trial / a_trial_sum
                
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if abs(denom) > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a_sum = a.sum()
                    if a_sum > 1e-10:
                        a /= a_sum
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_obj


def peak_shave(a0, time_limit, n_peaks=3):
    """Reduce peaks by adjusting contributing elements."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        peak_idx = int(np.argmax(b))
        
        # Find contributing pairs
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        idx = np.arange(js, je + 1)
        contributions = a[idx] * a[peak_idx - idx]
        
        if contributions.max() < 1e-15:
            break
        
        max_i = idx[np.argmax(contributions)]
        max_j = peak_idx - max_i
        
        # Try reducing the larger of a[max_i], a[max_j]
        target = max_i if a[max_i] >= a[max_j] else max_j
        old_val = a[target]
        if old_val < 1e-10:
            break
        
        improved = False
        for factor in [0.95, 0.9, 0.85, 0.8]:
            new_val = old_val * factor
            diff = old_val - new_val
            if diff < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[target] = new_val
            mask = np.ones(n, dtype=bool)
            mask[target] = False
            others = a_trial[mask]
            if others.sum() < 1e-15:
                continue
            a_trial[mask] = others + diff * others / others.sum()
            a_trial = a_trial / a_trial.sum()
            
            val = fast_convolve(a_trial).max()
            if val < best_obj - 1e-14:
                a = a_trial
                best_obj = val
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            break
    
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        # One more LP pass
        if rem > 30 and n_best <= 3000:
            t_lp = min(rem * 0.3, 60.0)
            refined_final, _ = lp_refine(best_seq, time_limit=t_lp, max_iter=30, t=0.2)
            try_seq(refined_final, "lp_final")
            rem = deadline - time.monotonic() - 30
        
        # Final high-T gradient
        if rem > 20:
            z0 = z_from_a(best_seq)
            t_alloc = min(rem * 0.5, 40.0)
            z_fin, _ = lbgfs_optimize(z0, 500, n_best, t_alloc, maxiter=2000)
            a_fin = softmax_from_z(z_fin)
            try_seq(a_fin, "final_T500")
            
            rem = deadline - time.monotonic() - 10
            if rem > 10:
                # Perturbation + re-optimize
                noise = rng.standard_normal(n_best) * 0.003
                a_pert = project_simplex(best_seq + noise)
                z_pert = z_from_a(a_pert)
                t_alloc2 = min(rem * 0.7, 20.0)
                z_p, _ = lbgfs_optimize(z_pert, 200, n_best, t_alloc2, maxiter=1000)
                a_p = softmax_from_z(z_p)
                try_seq(a_p, "perturb_final")
        
        # Final coordinate descent
        rem = deadline - time.monotonic() - 5
        if rem > 10 and best_seq is not None:
            a_cd2, _ = coordinate_descent(best_seq, time_limit=rem * 0.8)
            try_seq(a_cd2, "cd_final")
    
    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
    
    # Ensure non-negative and finite
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    
    # Final score
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
