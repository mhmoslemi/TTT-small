import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns (g, dual_vars) where dual_vars are the Lagrange multipliers."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        g = res.x
        dual = res.ineqlin.marginals if res.ineqlin is not None and res.ineqlin.marginals is not None else np.zeros(2*n-1)
        return g, dual
    return None, None


def subgradient_from_weights(w, a, n):
    """Compute 2*(w cross-correlate a) truncated to length n, using FFT."""
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    return 2.0 * R


def golden_section_search(f_curr, f_target, t_eval, n_iter=25):
    """Golden section line search: minimize t_eval(alpha) for alpha in [0,1]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_lo = 0.0
    a_hi = 1.0
    c_pt = a_hi - gr * (a_hi - a_lo)
    d_pt = a_lo + gr * (a_hi - a_lo)
    c_val = t_eval(c_pt)
    d_val = t_eval(d_pt)
    for _ in range(n_iter):
        if c_val < d_val:
            a_hi = d_pt
            d_pt = c_pt
            d_val = c_val
            c_pt = a_hi - gr * (a_hi - a_lo)
            c_val = t_eval(c_pt)
        else:
            a_lo = c_pt
            c_pt = d_pt
            c_val = d_val
            d_pt = a_lo + gr * (a_hi - a_lo)
            d_val = t_eval(d_pt)
    best_alpha = (a_lo + a_hi) / 2.0
    best_val = t_eval(best_alpha)
    # Also check endpoints
    v0 = t_eval(0.0)
    v1 = t_eval(1.0)
    if v0 < best_val:
        best_alpha, best_val = 0.0, v0
    if v1 < best_val:
        best_alpha, best_val = 1.0, v1
    return best_alpha, best_val

def _smooth_obj_grad(t, beta):
    """Compute smooth log-sum-exp objective and gradient in t-space."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)

    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()

    obj = conv_max + np.log(exp_shifted.sum()) / beta

    grad_a = subgradient_from_weights(w, a, n)
    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)

    return obj, grad_t


def _smooth_obj_only(t, beta):
    return _smooth_obj_grad(t, beta)[0]


def lbfqg_optimize(f0, time_limit, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)

    best_f = f.copy()

    eps = 1e-15

    t0_time = time.monotonic()

    # Two-stage: first coarse (low beta) to find basin, then fine (high beta)

    for bi, beta in enumerate(betas):

        try:
            maxiter = max(100, int(sub_time * 8))
            res = minimize(
                _smooth_obj_grad, t, args=(beta,),
                jac=True, method='L-BFGS-B',
                options={'maxiter': maxiter, 'gtol': 1e-10, 'maxls': 40,
                         'maxfun': max(maxiter * 5, 500)}
            )
            t = res.x

    return best_f, best_obj


def peak_equalization(f0, time_limit, verbose=False):
    """Iteratively equalize the top K peaks by adjusting the sequence.
    Uses the KKT condition: at optimum, weighted subgradient is constant on support."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()

    K = max(5, n // 16)
    lr = 0.02
    decay = 0.9995
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()

        # Find top K peaks
        k_actual = min(K, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        # Weight peaks by how much they exceed the mean
        peak_vals = conv[peaks]
        mean_val = conv.mean()
        excess = np.maximum(peak_vals - mean_val, 0)
        w = excess / (excess.sum() + 1e-30)

        w_full = np.zeros(2 * n - 1)
        w_full[peaks] = w

        # Compute subgradient
        sg = subgradient_from_weights(w_full, f, n)

        # Projected gradient step with step size
        f_new = project_simplex(f - lr * sg)
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()

        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()

        if M_new < best_obj - 1e-14:
            best_obj = M_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            if verbose and no_improve == 0 and best_obj < 1.51:
                print(f"    PeakEq: obj={best_obj:.10f}")
        else:
            no_improve += 1
            f = best_f.copy()  # revert

        lr *= decay
        if lr < 0.0005:
            lr = 0.0005

        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            no_improve = 0
            lr = 0.01

    return best_f, best_obj


def pair_coordinate_descent(f0, time_limit, verbose=False):
    """Pair-coordinate descent: optimize pairs of coordinates simultaneously."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    sweep = 0
    while time.monotonic() - t0 < time_limit:
        improved = False
        sweep += 1

        # Random order for diversity
        order = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            j = order[idx]
            if f[j] < 1e-14:
                continue

            # Find partner: the index that has the largest interaction with j
            # (i.e., the one where changing both a[j] and a[partner] affects the peak most)
            # Simple heuristic: choose a random partner or the one with similar value
            partner_pool = [k for k in range(n) if k != j and f[k] > 1e-10]
            if not partner_pool:
                continue
            # Pick a few candidates and try
            candidates = rng.choice(partner_pool, size=min(3, len(partner_pool)), replace=False) if len(partner_pool) > 3 else np.array(partner_pool)

            old_j = f[j]
            best_pair_obj = best_obj
            best_vals = None

            for p in candidates:
                old_p = f[p]
                # Try a grid of (a_j, a_p) adjustments
                for dj in [-0.3, -0.15, 0.0, 0.15, 0.3]:
                    new_j = max(0, old_j * (1 + dj))
                    if new_j == old_j:
                        continue
                    for dp in [-0.3, -0.15, 0.0, 0.15, 0.3]:
                        new_p = max(0, old_p * (1 + dp))
                        # Renormalize rest
                        rest_sum = 1.0 - old_j - old_p + new_j + new_p
                        if rest_sum < 1e-10:
                            continue
                        scale = (1.0 - new_j - new_p) / (1.0 - old_j - old_p) if (1.0 - old_j - old_p) > 1e-10 else 1.0
                        if scale < 0 or scale > 2:
                            continue
                        f_trial = f.copy()
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        mask[p] = False
                        f_trial[mask] *= scale
                        f_trial[j] = new_j
                        f_trial[p] = new_p
                        f_trial = f_trial / f_trial.sum()
                        obj = fast_convolve(f_trial).max()
                        if obj < best_pair_obj - 1e-15:
                            best_pair_obj = obj
                            best_vals = (new_j, new_p, scale, p)

            if best_vals is not None:
                new_j, new_p, scale, p = best_vals
                f[j] = new_j
                f[p] = new_p
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                mask[p] = False
                f[mask] *= scale
                f = f / f.sum()
                if best_pair_obj < best_obj - 1e-14:
                    best_obj = best_pair_obj
                    best_f = f.copy()
                    improved = True

        if not improved and sweep > 2:
            break
        if sweep > 50:
            break

    return best_f, best_obj


def mwu_optimize(f0, time_limit, verbose=False):
    """Multiplicative Weight Update with adaptive beta and momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    beta_ls = 40.0
    eta = 0.04
    momentum = 0.75
    v = np.zeros(n)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        conv_max = conv.max()
        shifted = beta_ls * (conv - conv_max)
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()

        grad_a = subgradient_from_weights(w, f, n)
        v = momentum * v + (1.0 - momentum) * grad_a

        log_f = np.log(f + 1e-30) - eta * v
        f_new = softmax_fn(log_f)
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()

        obj_new = fast_convolve(f_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and best_obj < 1.51:
                print(f"    MWU: obj={best_obj:.10f}")
        else:
            no_improve += 1
            eta *= 0.98
            beta_ls *= 1.02
            if no_improve > 400:
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.008)
                v = np.zeros(n)
                eta = 0.03
                beta_ls = 30.0
                no_improve = 0

        eta = max(eta, 0.003)

    return best_f, best_obj


def coordinate_descent_1d(f0, time_limit):
    """1D coordinate descent with multi-multiplication trial."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in [0.4, 0.6, 0.75, 0.85, 0.9, 1.1, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0 or scale > 3:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining LP-FW, L-BFGS, peak equalization,
    pair-coordinate descent, MWU, and 1D coordinate descent with multi-length
    exploration and structured restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512

    # === Phase 1: LP-FW on parent (350s) ===
    print("\n=== Phase 1: LP-FW on parent ===")
    t1 = min(350, deadline - time.monotonic() - 900)
    if t1 > 30:
        a_opt, _ = lp_fw_optimize(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "lpfw_parent")

    # Try reversed
    t1r = min(80, deadline - time.monotonic() - 800)
    if t1r > 20:
        a_opt, _ = lp_fw_optimize(parent[::-1].copy(), time_limit=t1r)
        try_seq(a_opt, "lpfw_rev")

    # === Phase 2: L-BFGS (400s) ===
    print(f"\n=== Phase 2: L-BFGS (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    t2 = min(400, deadline - time.monotonic() - 700)
    if t2 > 40:
        a_opt, _ = lbfqg_optimize(base, time_limit=t2, verbose=True)
        try_seq(a_opt, "lbfqg_main")

        # Perturbed start
        t2b = min(100, deadline - time.monotonic() - 550)
        if t2b > 20:
            noise = rng.standard_normal(len(base)) * 0.015
            f_pert = project_simplex(base + noise)
            a_opt, _ = lbfqg_optimize(f_pert, time_limit=t2b)
            try_seq(a_opt, "lbfqg_pert")

    # === Phase 3: Peak Equalization (200s) ===
    print(f"\n=== Phase 3: Peak Equalization (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    t3 = min(200, deadline - time.monotonic() - 450)
    if t3 > 20:
        a_opt, _ = peak_equalization(base, time_limit=t3, verbose=True)
        try_seq(a_opt, "peak_eq")

    # === Phase 4: MWU (150s) ===
    print(f"\n=== Phase 4: MWU (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    t4 = min(150, deadline - time.monotonic() - 350)
    if t4 > 20:
        a_opt, _ = mwu_optimize(base, time_limit=t4, verbose=True)
        try_seq(a_opt, "mwu_main")

    # === Phase 5: Multi-length exploration (200s) ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    target_lengths = [640, 896, 1024, 1280]
    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 250
        if rem < 40:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.35, 70)
        print(f"  Trying n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc)
        try_seq(a_opt, f"len_{target_n}")

    # === Phase 6: Deep optimization on best (300s) ===
    print(f"\n=== Phase 6: Deep Optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # 6a: LP-FW deep
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(rem * 0.4, 120)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_lpfw")

        # 6b: L-BFGS deep
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_alloc = min(rem * 0.5, 120)
            a_opt, _ = lbfqg_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_lbfqg")

        # 6c: Peak equalization deep
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_alloc = min(rem * 0.5, 80)
            a_opt, _ = peak_equalization(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "deep_peak_eq")

        # 6d: Pair coordinate descent
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_alloc = min(rem * 0.5, 60)
            a_opt, _ = pair_coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "deep_pair_cd")

    # === Phase 7: Final polishing (remaining) ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Alternate methods with small perturbations
        for attempt in range(8):
            rem = deadline - time.monotonic() - 20
            if rem < 15:
                break

            noise = rng.standard_normal(n_best) * 0.002
            f_start = project_simplex(best_seq + noise)

            if attempt % 3 == 0:
                t_alloc = min(rem * 0.6, 50)
                a_opt, _ = lp_fw_optimize(f_start, time_limit=t_alloc)
                try_seq(a_opt, f"final_lpfw_{attempt}")
            elif attempt % 3 == 1:
                t_alloc = min(rem * 0.6, 50)
                a_opt, _ = lbfqg_optimize(f_start, time_limit=t_alloc)
                try_seq(a_opt, f"final_lbfqg_{attempt}")
            else:
                t_alloc = min(rem * 0.6, 50)
                a_opt, _ = peak_equalization(f_start, time_limit=t_alloc)
                try_seq(a_opt, f"final_peak_{attempt}")

        # 1D coordinate descent final
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_opt, _ = coordinate_descent_1d(best_seq, time_limit=min(rem, 40))
            try_seq(a_opt, "final_cd1d")

        # Pair CD final
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt, _ = pair_coordinate_descent(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "final_pair")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
