import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, n, top_k=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k (or top_k most important), g >= 0."""
    # Build constraint matrix
    # (f*g)[k] = sum_j f[j] * g[k-j] for valid j
    # For constraint k: sum over j where 0<=j<n and 0<=k-j<n of f[j]*g[k-j]
    # = sum over j from max(0,k-n+1) to min(n-1,k) of f[j]*g[k-j]
    # Let i = k-j, so j = k-i, and the constraint is:
    # sum over i from max(0,k-n+1) to min(n-1,k) of f[k-i]*g[i]
    
    # Full constraint matrix would be (2n-1) x n
    # For active set, we select the most important constraints
    
    if top_k is not None:
        # Compute current convolution to find active constraints
        # We want constraints where (f*g) is closest to M
        # Actually, we want all constraints since g is what we're optimizing
        # Let's just use all constraints for now
        conv_f = fast_convolve(f)
        # Most binding constraints are where conv_f is large
        # Actually all constraints matter, but let's use the top_k largest
        indices = np.argpartition(conv_f, -top_k)[-top_k:]
    else:
        indices = np.arange(2 * n - 1)
    
    num_constraints = len(indices)
    A_ub = np.zeros((num_constraints, n))
    for idx in range(num_constraints):
        k = int(indices[idx])
        # (f*g)[k] = sum_i f[k-i]*g[i] for valid i
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            A_ub[idx, i] = f[k - i]
    
    b_ub = np.full(num_constraints, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    options = {'time_limit': 30, 'presolve': True}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return res.x
    return None


def solve_lp_direction_fast(f, M, n, top_k=None):
    """Faster LP direction using structured constraints."""
    if top_k is not None and top_k < 2 * n - 1:
        conv_f = fast_convolve(f)
        indices = np.argpartition(conv_f, -top_k)[-top_k:]
    else:
        indices = np.arange(2 * n - 1)
    
    # Build sparse-ish constraint matrix more efficiently
    num_constraints = len(indices)
    A_ub = np.zeros((num_constraints, n))
    
    for ci, k in enumerate(indices):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            A_ub[ci, i] = f[k - i]
    
    b_ub = np.full(num_constraints, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    options = {'time_limit': 20, 'presolve': True}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return res.x
    return None


def fw_optimize(f0, time_limit, verbose=False, use_away=False, top_k_constraints=None):
    """Frank-Wolfe optimization with optional away-steps."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP for direction
        g = solve_lp_direction_fast(f, M * m_relax, n, top_k=top_k_constraints)
        if g is None:
            m_relax = min(m_relax * 1.1, 5.0)
            if m_relax > 5.0:
                m_relax = 1.0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 50:
                # Perturb
                rng = np.random.default_rng(it * 7919)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_relax = 1.0
                no_improve = 0
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        
        # Coarse
        alphas = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
        for alpha in alphas:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Fine
        if best_alpha > 0:
            if best_alpha >= 1.0:
                lo, hi = 0.7, 1.0
            elif best_alpha <= 0.05:
                lo, hi = 0.0, 0.1
            else:
                lo, hi = max(0.0, best_alpha - 0.2), min(1.0, best_alpha + 0.2)
            
            for _ in range(30):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-12:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            ss = f_new.sum()
            if ss > 1e-15:
                f_new = f_new / ss
            f = f_new
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0
            
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
    
    return best_f, best_obj

                if val < best_j_obj - 1e-15:

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(f, n, k_indices):
    """Build constraint matrix rows for given lag indices."""
    num_k = len(k_indices)
    A = np.zeros((num_k, n))
    for ci, k in enumerate(k_indices):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            A[ci, i] = f[k - i]
    return A


def solve_lp_direction(f, M, n, top_k=None, time_per_lp=15):
    """Solve LP: max 1^T g s.t. (f*g)[k] <= M, g >= 0, using active set."""
    if top_k is not None and top_k < 2 * n - 1:
        conv_f = fast_convolve(f)
        k_indices = np.argpartition(conv_f, -top_k)[-top_k:]
    else:
        k_indices = np.arange(2 * n - 1)
    
    A_ub = build_constraint_matrix(f, n, k_indices)
    b_ub = np.full(len(k_indices), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    opts = {'time_limit': time_per_lp, 'presolve': True}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=opts)
    if res.success and res.x is not None:
        return res.x
    return None


def fw_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP direction finding and adaptive line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0
    
    # Adaptive top_k: start with all, reduce if LP is too slow
    top_k = None
    lp_time_avg = 0.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        lp_t0 = time.monotonic()
        g = solve_lp_direction(f, M * m_relax, n, top_k=top_k, time_per_lp=10)
        lp_time_avg = 0.7 * lp_time_avg + 0.3 * (time.monotonic() - lp_t0)
        
        if g is None:
            m_relax = min(m_relax * 1.08, 4.0)
            if lp_time_avg > 8 and top_k is None:
                top_k = max(200, n)
            if lp_time_avg > 8 and top_k > 150:
                top_k = max(100, top_k // 2)
            if m_relax > 4.0:
                m_relax = 1.0
                continue
            continue
        
        lp_time_avg = min(lp_time_avg, 10.0)
        if lp_time_avg < 3 and top_k is not None:
            top_k = None
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 60:
                rng = np.random.default_rng(it * 7919 + 13)
                scale = 0.002 + 0.002 * min(no_improve // 60, 3)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_relax = 1.0
                no_improve = 0
            continue
        
        # Line search: coarse + fine bisection
        best_alpha = 0.0
        best_val = M
        
        alphas_coarse = [0.03, 0.07, 0.15, 0.25, 0.4, 0.6, 0.8, 1.0]
        for alpha in alphas_coarse:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            if best_alpha >= 1.0:
                lo, hi = 0.7, 1.0
            elif best_alpha <= 0.03:
                lo, hi = 0.0, 0.08
            else:
                lo, hi = max(0.0, best_alpha - 0.15), min(1.0, best_alpha + 0.15)
            
            for _ in range(25):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-13:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            f = f_new
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0
            
            if M < best_obj - 1e-16:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if verbose and it % 50 == 0:
            print(f"    FW iter {it}: M={M:.10f}, best={best_obj:.10f}")
    
    return best_f, best_obj


def softmax_adam_optimize(f0, time_limit, verbose=False):
    """Optimize using softmax parametrization and Adam on p-norm objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    theta = np.log(f + 1e-30)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    lr_init = 0.02
    t = 0
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        theta_max = theta.max()
        exp_theta = np.exp(theta - theta_max)
        f = exp_theta / exp_theta.sum()
        
        conv = fast_convolve(f)
        M = conv.max()
        
        if M < best_obj - 1e-15:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Use p-norm with adaptive p
        p = min(50, 5 + t * 0.05)
        scaled = conv / p
        scaled -= scaled.max()
        exp_scaled = np.exp(scaled * p)
        soft_max = (1.0 / p) * np.log(np.sum(exp_scaled) / len(conv)) + conv.max() / 1.0
        
        # Actually use a cleaner p-norm: (sum conv^p)^(1/p)
        # Gradient: d/d(f[j]) of (sum_k conv[k]^p)^(1/p)
        # = (1/p) * (sum_k conv[k]^p)^(1/p - 1) * p * sum_k conv[k]^(p-1) * d(conv[k])/d(f[j])
        # = (sum_k conv[k]^p)^(1/p - 1) * sum_k conv[k]^(p-1) * d(conv[k])/d(f[j])
        
        conv_p = np.maximum(conv, 1e-30) ** p
        conv_p_sum = conv_p.sum()
        norm_p = conv_p_sum ** (1.0 / p)
        # weights for each k
        w_k = (conv_p_sum ** (1.0/p - 1.0)) * conv ** (p - 1)
        
        # Gradient of conv[k] w.r.t. f:
        # d(conv[k])/d(f[j]) = 2*f[k-j] if 0<=k-j<n, else 0
        # grad_f[j] = sum_k w_k[k] * 2*f[k-j]
        # = 2 * cross_correlation(w_k, f)[j]
        
        pad_size = 1 << (3 * n - 2).bit_length()
        W = np.fft.rfft(w_k, pad_size)
        F = np.fft.rfft(f, pad_size)
        H = np.fft.irfft(W * np.conj(F), pad_size)
        grad_f = 2.0 * H[:n]
        
        # Convert to theta-space: grad_theta = f * (grad_f - dot(grad_f, f))
        dot_gf = np.dot(grad_f, f)
        grad_theta = f * (grad_f - dot_gf)
        
        # Adam
        if not hasattr(softmax_adam_optimize, '_m'):
            m = np.zeros(n)
            v = np.zeros(n)
        
        # Use local adam state
        if t == 1:
            m = np.zeros(n)
            v = np.zeros(n)
        
        m = beta1 * m + (1 - beta1) * grad_theta
        v = beta2 * v + (1 - beta2) * grad_theta ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        lr_t = lr_init * max(0.05, 1.0 / (1.0 + 0.01 * t))
        theta = theta - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        if no_improve > 300:
            f = best_f.copy()
            theta = np.log(f + 1e-30)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 1
            lr_init = max(0.002, lr_init * 0.5)
            no_improve = 0
            if lr_init < 0.001:
                break
        
        if verbose and t % 200 == 0:
            print(f"    Adam iter {t}: obj={best_obj:.10f}, M={M:.10f}")
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Coordinate descent with adaptive multipliers."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve_passes = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        cur_conv = fast_convolve(f)
        cur_max = cur_conv.max()
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            
            if old_val < 1e-6:
                candidates = [0.001, 0.003, 0.01]
            else:
                candidates = [old_val * m for m in [0.6, 0.8, 0.9, 1.1, 1.25, 1.5]]
            
            best_j_val = old_val
            best_j_obj = cur_max
            
            for new_val in candidates:
                if new_val < 0 or new_val > 0.95:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                improved = True
        
        if not improved:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                break
        else:
            no_improve_passes = 0
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Linear interpolation to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. LP-FW on parent with adaptive active set
    2. Multi-length exploration
    3. Softmax Adam fine optimization
    4. Coordinate descent polish
    5. Micro-perturbation final search
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(768) / 768
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}")
    
    n_parent = len(parent)
    
    # === Phase 1: LP-FW on parent ===
    print(f"\n=== Phase 1: LP-FW on parent n={n_parent} ===")
    rem = deadline - time.monotonic()
    t1 = min(300, rem * 0.2)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a_opt, obj = fw_optimize(parent, time_limit=t1, verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.8f}")
        else:
            print(f"  No improvement (fw_obj={obj:.8f})")
    
    # === Phase 2: Multi-length exploration ===
    print(f"\n=== Phase 2: Multi-length (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    candidate_lengths = []
    for delta in [-128, -64, -32, 32, 64, 128, 256, -256, 640, 896, 1024, 512]:
        nl = n_parent + delta if abs(delta) <= 256 else delta
        if 128 <= nl <= 1536 and nl != n_parent and nl not in candidate_lengths:
            candidate_lengths.append(nl)
    candidate_lengths.sort(key=lambda x: abs(x - n_best))
    
    length_results = {}
    for target_n in candidate_lengths[:6]:
        rem = deadline - time.monotonic() - 250
        if rem < 30:
            break
        f_interp = interpolate_sequence(best_seq, target_n)
        t_alloc = min(120, rem * 0.15)
        if t_alloc < 20:
            continue
        print(f"  n={target_n}, t={t_alloc:.0f}s")
        a_opt, obj = fw_optimize(f_interp, time_limit=t_alloc)
        sc = float(evaluate_sequence(a_opt.tolist()))
        length_results[target_n] = (a_opt, sc)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.8f} (n={target_n})")
        else:
            print(f"  n={target_n}: score={sc:.8f}")
    
    # === Phase 3: Softmax Adam on best ===
    print(f"\n=== Phase 3: Softmax Adam (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 200
    if rem > 30:
        t_sa = min(rem * 0.3, 250)
        print(f"  t={t_sa:.0f}s")
        a_opt, obj = softmax_adam_optimize(best_seq, time_limit=t_sa, verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.8f}")
        else:
            print(f"  No improvement (obj={obj:.8f})")
    
    # === Phase 4: Coordinate descent ===
    print(f"\n=== Phase 4: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 150
    if rem > 30:
        t_cd = min(rem * 0.4, 180)
        print(f"  t={t_cd:.0f}s")
        a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.8f}")
    
    # === Phase 5: LP-FW on best (round 2) ===
    print(f"\n=== Phase 5: LP-FW round 2 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 100
    if rem > 30:
        t_fw2 = min(rem * 0.4, 150)
        print(f"  t={t_fw2:.0f}s")
        a_opt, obj = fw_optimize(best_seq, time_limit=t_fw2)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.8f}")
    
    # === Phase 6: Another softmax Adam pass ===
    print(f"\n=== Phase 6: Softmax Adam round 2 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 60
    if rem > 20:
        t_sa2 = min(rem * 0.5, 80)
        a_opt, obj = softmax_adam_optimize(best_seq, time_limit=t_sa2)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.8f}")
    
    # === Phase 7: Final coordinate descent ===
    print(f"\n=== Phase 7: Final CD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 30
    if rem > 15:
        t_cd2 = min(rem * 0.7, 60)
        a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd2)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.8f}")
    
    # === Phase 8: Micro-perturbation ===
    print(f"\n=== Phase 8: Micro-perturbation (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    for trial in range(30):
        rem = deadline - time.monotonic()
        if rem < 3:
            break
        scale = 0.0001 + trial * 0.00005
        noise = rng.standard_normal(n_best) * scale
        f_pert = project_simplex(best_seq + noise)
        sc = float(evaluate_sequence(f_pert.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = f_pert.copy()
            print(f"  NEW BEST [pert{trial}]: {sc:.8f}")
    
    # === Phase 9: Final squeeze ===
    rem = deadline - time.monotonic()
    if rem > 10:
        t_fin = min(rem * 0.9, 40)
        a_opt, obj = fw_optimize(best_seq, time_limit=t_fin)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  FINAL: {sc:.8f}")
    
    # Final verification
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
