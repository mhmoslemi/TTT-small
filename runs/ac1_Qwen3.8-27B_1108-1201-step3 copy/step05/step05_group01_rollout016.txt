def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Numerically stable softmax."""
    v = v - v.max()

def solve_lp_full(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    # A_ub has shape (2n-1, n)
    # Row k: sum_j f[j] * g[j+k-n] (with appropriate indexing)
    # Actually, (f*g)[k] = sum_j f[j] * g[k-j] where valid j gives 0 <= j < n and 0 <= k-j < n
    # So (f*g)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[j] * g[k-j]
    # Let me re-derive: conv(f,g)[k] = sum_j f[j]*g[k-j] for j such that 0<=j<n and 0<=k-j<n
    # i.e., max(0, k-n+1) <= j <= min(n-1, k)
    
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # f[j] contributes to conv[k] for k = j, j+1, ..., j+n-1
        # conv[k] = sum_j f[j] * g[k-j], so g[k-j] means g[m] where m = k-j
        # So for fixed j, g[m] appears in conv[j+m] with coefficient f[j]
        # i.e., A_ub[j+m, m] += f[j] for m = 0, ..., n-1
        A_ub[j:j+n, :] = 0  # clear (should already be 0)
        for m in range(n):
            A_ub[j+m, m] = f[j]
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

    conv = fast_convolve(f)

            if 0 <= m < n:
                A_ub[idx, m] = f[j]
    
    b_ub = np.full(len(top_k), M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

        return res.x
    return None

def spectral_init(n, mode='gaussian'):
    """Generate structured initializations."""
    if mode == 'uniform':
        return np.ones(n) / n
    elif mode == 'gaussian':
        x = np.linspace(-3, 3, n)
        a = np.exp(-x**2 / 4)
        return a / a.sum()
    elif mode == 'triangle':
        x = np.linspace(0, 1, n)
        a = np.maximum(0, 1 - 2*np.abs(x - 0.5))
        return a / a.sum()
    elif mode == 'cosine':
        x = np.arange(n) / n
        a = np.cos(np.pi * x) ** 2
        a = np.maximum(a, 0)
        return a / a.sum()
    elif mode == 'two_peaks':
        x = np.arange(n) / n
        a = np.exp(-20*(x-0.25)**2) + np.exp(-20*(x-0.75)**2)
        return a / a.sum()
    elif mode == 'flat_edges':
        a = np.ones(n)
        a[:n//8] = 0.3
        a[-(n//8):] = 0.3
        return a / a.sum()
    else:
        return np.ones(n) / n


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def build_conv_matrix(n):
    """Build the banded convolution matrix structure (indices only)."""
    # For convolution: (f*g)[k] = sum_m f[k-m]*g[m]

    # We return the row structure for efficient filling
    rows = []
    cols = []
    for k in range(2 * n - 1):
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        for m in range(m_start, m_end + 1):
            rows.append(k)
            cols.append(m)
    return np.array(rows), np.array(cols)


def solve_lp_direction(f, M, n, top_k=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Uses top-K constraints if specified."""
    rows, cols = build_conv_matrix(n)

        conv_ff = fast_convolve(f)

        cols_sel = cols[row_mask]

        cols_sel = cols

def lse_lbfgs_optimize(f0, time_limit, verbose=False):
    """L-BFGS on log-sum-exp with annealed beta in softmax parameterization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    
    beta_holder = [10.0]
    
    def obj_grad(t):
        beta = beta_holder[0]
        a = softmax_fn(t)
        conv = fast_convolve(a)
        conv_max = conv.max()
        shifted = beta * (conv - conv_max)
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()
        obj = conv_max + np.log(exp_shifted.sum()) / beta
        
        N = 1 << (3 * n - 2).bit_length()
        A = np.fft.rfft(a, N)
        W = np.fft.rfft(w, N)
        R = np.fft.irfft(W * np.conj(A), N)[:n]
        grad_a = 2.0 * R
        dot = np.dot(a, grad_a)
        grad_t = a * (grad_a - dot)
        return obj, grad_t
    
    betas = [5, 10, 20, 50, 100, 200, 400, 800]
    for beta in betas:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 15:
            break
        beta_holder[0] = float(beta)
        t = np.log(f + 1e-15)
        try:
            res = minimize(obj_grad, t, jac=True, method='L-BFGS-B',
                          options={'maxiter': 150, 'gtol': 1e-10, 'maxls': 40})
            f = softmax_fn(res.x)
            f = np.maximum(f, 0)
            f = f / f.sum()
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_f = f.copy()
                if verbose:
                    print(f"    LSE-LBFGS beta={beta}: obj={best_obj:.10f}")
        except Exception:
            break
    
    return best_f, best_obj


def gradient_optimize(f0, time_limit, verbose=False):
    """Momentum-based gradient on log-sum-exp with adaptive parameters."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    beta = 20.0
    lr = 0.005
    momentum = 0.92
    v = np.zeros(n)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        conv_max = conv.max()
        shifted = beta * (conv - conv_max)
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()
        
        N = 1 << (3 * n - 2).bit_length()
        A = np.fft.rfft(f, N)
        W = np.fft.rfft(w, N)
        R = np.fft.irfft(W * np.conj(A), N)[:n]
        grad = 2.0 * R
        
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        obj_new = fast_convolve(f_new).max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            beta = min(beta * 1.03, 1000.0)
            if verbose and best_obj < 1.51:
                print(f"    Grad: obj={best_obj:.10f}, beta={beta:.1f}")
        else:
            no_improve += 1
            lr *= 0.97
            beta = max(beta * 0.97, 5.0)
        
        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            v = np.zeros(n)
            lr = 0.005
            beta = 20.0
            no_improve = 0
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def spectral_init(n, mode='gaussian'):
    """Generate structured initializations."""
    if mode == 'uniform':
        return np.ones(n) / n
    elif mode == 'gaussian':
        x = np.linspace(-3, 3, n)
        a = np.exp(-x**2 / 4)
        return a / a.sum()
    elif mode == 'triangle':
        x = np.linspace(0, 1, n)
        a = np.maximum(0, 1 - 2 * np.abs(x - 0.5))
        return a / a.sum()
    elif mode == 'cosine':
        x = np.arange(n) / n
        a = np.cos(np.pi * x) ** 2
        a = np.maximum(a, 0.001)
        return a / a.sum()
    elif mode == 'two_peaks':
        x = np.arange(n) / n
        a = np.exp(-30 * (x - 0.25) ** 2) + np.exp(-30 * (x - 0.75) ** 2)
        return a / a.sum()
    elif mode == 'broad_peaks':
        x = np.arange(n) / n
        a = np.exp(-10 * (x - 0.3) ** 2) + np.exp(-10 * (x - 0.7) ** 2)
        return a / a.sum()
    elif mode == 'wavy':
        x = np.arange(n) / n
        a = 1.0 + 0.3 * np.cos(4 * np.pi * x) + 0.1 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 'inverse_edges':
        x = np.arange(n) / n
        a = 1.0 / (1.0 + 5.0 * (2 * np.abs(x - 0.5) - 0.5) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def symmetry_enhance(a):
    """Average a sequence with its reversal and project to simplex."""
    avg = 0.5 * (a + a[::-1])
    avg = np.maximum(avg, 0)
    s = avg.sum()
    if s > 1e-15:
        avg /= s
    return avg


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization:
    1. Enhanced LP-FW with adaptive M and top-K constraints
    2. Multi-initialization exploration
    3. LSE-LBFGS polishing
    4. Multi-length exploration
    5. Symmetry exploitation
    6. Final deep optimization with alternating methods
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(768) / 768
    
    n_parent = len(parent)
    
    # === Phase 1: LP-FW on parent with aggressive M ===
    print("\n=== Phase 1: Aggressive LP-FW on parent ===")
    t1 = min(350, deadline - time.monotonic() - 1200)
    if t1 > 30:
        a_opt, obj = lp_fw_optimize(parent, time_limit=t1, verbose=True, aggressive=True)
        try_seq(a_opt, "lpfw_aggr")
        # Symmetry enhance
        a_sym = symmetry_enhance(a_opt)
        try_seq(a_sym, "lpfw_sym")
    
    # === Phase 2: Quick LP-FW on reversed parent ===
    t1r = min(80, deadline - time.monotonic() - 1000)
    if t1r > 20:
        a_rev = parent[::-1].copy()
        a_opt, _ = lp_fw_optimize(a_rev, time_limit=t1r, aggressive=True)
        try_seq(a_opt, "lpfw_reversed")
    
    # === Phase 3: Multi-initialization search ===
    print(f"\n=== Phase 3: Multi-init (best={best_score:.6f}) ===")
    
    init_modes = ['gaussian', 'cosine', 'two_peaks', 'wavy', 'inverse_edges', 'broad_peaks']
    
    for mode in init_modes:
        rem = deadline - time.monotonic() - 700
        if rem < 25:
            break
        a0 = spectral_init(n_parent, mode=mode)
        t_alloc = min(50, rem * 0.25)
        a_opt, _ = lp_fw_optimize(a0, time_limit=t_alloc, aggressive=True)
        try_seq(a_opt, f"init_{mode}")
    
    # Random perturbations of current best
    base = best_seq if best_seq is not None else parent
    for trial in range(4):
        rem = deadline - time.monotonic() - 500
        if rem < 20:
            break
        noise = rng.standard_normal(len(base)) * (0.03 + 0.02 * trial)
        a0 = project_simplex(base + noise)
        t_alloc = min(40, rem * 0.3)
        a_opt, _ = lp_fw_optimize(a0, time_limit=t_alloc, aggressive=True)
        try_seq(a_opt, f"perturb_{trial}")
    
    # === Phase 4: Deep LP-FW on best ===
    print(f"\n=== Phase 4: Deep LP-FW (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    t4 = min(400, deadline - time.monotonic() - 600)
    if t4 > 40:
        a_opt, _ = lp_fw_optimize(base, time_limit=t4, verbose=True, aggressive=True)
        try_seq(a_opt, "deep_lpfw")
        a_sym = symmetry_enhance(a_opt)
        try_seq(a_sym, "deep_lpfw_sym")
    
    # === Phase 5: LSE-LBFGS polishing ===
    print(f"\n=== Phase 5: LSE-LBFGS (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    t5 = min(200, deadline - time.monotonic() - 300)
    if t5 > 30:
        a_opt, _ = lse_lbfgs_optimize(base, time_limit=t5, verbose=True)
        try_seq(a_opt, "lse_lbfgs")
    
    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    n_best = len(base)
    
    target_lengths = [512, 640, 768, 896, 1024, 1280, 1536, 2048]
    target_lengths = [l for l in target_lengths if l != n_best]
    # Sort by proximity to current best length
    target_lengths.sort(key=lambda x: abs(x - n_best))
    
    for target_n in target_lengths[:4]:
        rem = deadline - time.monotonic() - 250
        if rem < 35:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(70, rem * 0.35)
        print(f"  Trying n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc, aggressive=True)
        try_seq(a_opt, f"len_{target_n}")
        
        # Quick LSE-LBFGS on the result

        if rem2 > 20 and a_opt is not None:
            a_opt2, _ = lse_lbfgs_optimize(a_opt, time_limit=min(30, rem2 * 0.5))
            try_seq(a_opt2, f"len_{target_n}_lbfqg")
    
    # === Phase 7: Final deep optimization ===
    print(f"\n=== Phase 7: Final Deep (best={best_score:.6f}) ===")
    if best_seq is not None:
        # Alternate LP-FW and gradient with perturbations
        for attempt in range(6):
            rem = deadline - time.monotonic() - 25
            if rem < 20:
                break
            
            if attempt % 3 == 0:
                # Direct LP-FW
                t_alloc = min(rem * 0.5, 100)
                a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True, aggressive=True)
                try_seq(a_opt, f"final_lpfw_{attempt}")
            elif attempt % 3 == 1:
                # Perturb + LP-FW
                noise = rng.standard_normal(len(best_seq)) * 0.004
                f_start = project_simplex(best_seq + noise)
                t_alloc = min(rem * 0.5, 80)
                a_opt, _ = lp_fw_optimize(f_start, time_limit=t_alloc)
                try_seq(a_opt, f"final_pert_{attempt}")
            else:
                # Gradient method
                t_alloc = min(rem * 0.5, 60)
                a_opt, _ = gradient_optimize(best_seq, time_limit=t_alloc)
                try_seq(a_opt, f"final_grad_{attempt}")
        
        # Final LSE-LBFGS polish
        rem = deadline - time.monotonic() - 10
        if rem > 20:
            a_opt, _ = lse_lbfgs_optimize(best_seq, time_limit=min(rem, 80), verbose=True)
            try_seq(a_opt, "final_lse")
        
        # Symmetry final
        a_sym = symmetry_enhance(best_seq)
        try_seq(a_sym, "final_sym")
        
        # One more round on symmetry-enhanced
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_opt, _ = lp_fw_optimize(a_sym, time_limit=min(rem, 30), aggressive=True)
            try_seq(a_opt, "final_sym_lpfw")
    
    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
