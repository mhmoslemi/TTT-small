At the current point a, compute conv = a*a. Find the lag k* where conv is maximized. The subgradient of conv[k*] with respect to a is: for each i, the gradient component is 2*sum_j a_j * delta_{i,j+k*} (with appropriate bounds). This gives a direction to move.

Actually, this is essentially what the LP-FW does, but in a continuous (non-LP) way. Let me formalize:

Given current a (normalized to sum=1), compute conv = a*a and find k* = argmax conv.
The subgradient of the objective with respect to a is:
grad_i = 2 * sum_{j: j+k* valid} a_{j} * [i = j] + 2 * sum_{j: j-k* valid} a_{j} * [i = j]

       = 2 * conv_local[i] (where conv_local is the cross-correlation of a with a shifted by k*)

Actually, let me be more precise. conv[k] = sum_j a_j a_{j+k}. The derivative with respect to a_i is:
d(conv[k])/da_i = sum_j (delta_{ij} a_{j+k} + a_j delta_{i,j+k}) = a_{i+k} + a_{i-k} (where out-of-bounds terms are 0).

So the subgradient at the maximizing lag k* is:
grad_i = 2 * (a_{i+k*} + a_{i-k*})

This is a simple computation! We can use this for a subgradient descent method.

But since we're on the simplex (a >= 0, sum a = 1), we need to project after each step. And since the subgradient is noisy (it only uses one lag), we might want to use a "smoothed" version that weights all lags.

This is exactly what the log-sum-exp smoothing does! The gradient of log-sum-exp(conv/T) is a weighted average of the per-lag gradients, with weights proportional to exp(conv[k]/T). As T -> 0, this concentrates on the maximizing lag (recovering the subgradient). As T is large, it's a smooth average.

"""
Multi-stage optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses diverse initializations, log-sum-exp smooth Adam optimization, 
LP Frank-Wolfe, subgradient methods, coordinate descent, and 
progressive length exploration with adaptive restarts.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad(f, T):
    """Compute smooth gradient of logsumexp(conv(f,f)/T) via FFT."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / T
    scaled -= scaled.max()
    e = np.exp(scaled)
    w = e / e.sum()
    
    N = 1 << (2 * (2 * n - 1) - 1).bit_length()
    A_fft = np.fft.rfft(f, N)
    W_fft = np.fft.rfft(w, N)
    # cross-correlation: corr[i] = sum_j f[j] * w[j+i] for i=0..n-1
    # This is the auto-correlation structure we need
    corr = np.fft.irfft(A_fft * np.conj(W_fft), N)[:n]
    grad = (2.0 / T) * corr
    return grad, conv, w


def smooth_optimize(f0, time_limit, T_start=0.5, T_end=0.01, verbose=False):
    """Optimize using log-sum-exp smooth approximation with Adam."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t_adam = 0
    
    alpha = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        # Exponential cooling with a floor
        progress = min(elapsed / time_limit, 1.0)
        T = T_start * (T_end / T_start) ** (progress ** 0.7)
        T = max(T, T_end)
        lr = alpha * (0.3 ** (progress ** 0.8))
        
        grad, conv, w = smooth_grad(f, T)
        
        t_adam += 1
        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad ** 2
        m_hat = m_state / (1 - beta1 ** t_adam)
        v_hat = v_state / (1 - beta2 ** t_adam)
        
        f = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f)
        
        cur_obj = fast_convolve(f).max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 2000:
            seed_val = int(elapsed * 1000) % (2**31)
            rng_p = np.random.default_rng(seed_val)
            scale = 0.002 + 0.004 * min(no_improve // 500, 5)
            f = project_simplex(best_f + rng_p.standard_normal(n) * scale)
            m_state = np.zeros(n)
            v_state = np.zeros(n)
            t_adam = 0
            no_improve = 0
    
    return best_f, best_obj


def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lpfw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        g = lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.2
            if m_exp > 10.0:
                m_exp = 1.0
                continue
            else:
                continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 40:
                # Perturb and restart
                rng_p = np.random.default_rng(it * 7919 + 13)
                scale = 0.003 * min(1 + no_improve // 40, 5)
                f = project_simplex(best_f + rng_p.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue
        
        best_alpha = 0.0
        best_val = M
        
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.2)
            hi = min(1.0, best_alpha + 0.2)
            for _ in range(15):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)
    
    return best_f, best_obj


def subgradient_optimize(f0, time_limit, verbose=False):
    """Subgradient descent with momentum on the max-conv objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    momentum = np.zeros(n)
    mu = 0.9  # momentum coefficient
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / time_limit, 1.0)
        
        conv = fast_convolve(f)
        # Use a smoothed subgradient: weight lags by exp(conv[k]/T)
        T = 0.05 * (1.0 - 0.5 * progress) + 0.01
        scaled = (conv - conv.max()) / T
        w = np.exp(scaled)
        w = w / w.sum()
        
        # Compute weighted gradient
        # grad_i = 2 * sum_k w[k] * (a_{i+k} + a_{i-k})
        # This is a cross-correlation-like operation
        N = 1 << (3 * n - 1).bit_length()
        f_fft = np.fft.rfft(f, N)
        w_fft = np.fft.rfft(w, N)
        # cross-correlation of f with w
        corr = np.fft.irfft(f_fft * np.conj(w_fft), N)[:n]
        # Also need the "anti-correlation"
        w_rev = w[::-1].copy()
        wr_fft = np.fft.rfft(w_rev, N)
        corr2 = np.fft.irfft(f_fft * np.conj(wr_fft), N)[:n]
        
        grad = (2.0 / T) * (corr + corr2)
        
        # Step size: diminishing
        lr = 0.01 / (1.0 + 0.01 * it)
        
        momentum = mu * momentum + grad
        f = f - lr * momentum
        f = project_simplex(f)
        
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 1000:
            rng_p = np.random.default_rng(int(elapsed * 100) % (2**31))
            scale = 0.003 * min(1 + no_improve // 200, 5)
            f = project_simplex(best_f + rng_p.standard_normal(n) * scale)
            momentum = np.zeros(n)
            no_improve = 0
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with exact line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    
    # Precompute the convolution structure
    # For coordinate j, changing f[j] to x while scaling others:
    # new_f[j] = x, new_f[k] = f[k] * (1-x) / (1-f[j]) for k != j
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue
            
            # Try several candidate values for f[j]
            candidates = [0.0, 0.001, old_val * 0.5, old_val * 0.8, old_val * 1.2, 
                         old_val * 1.5, 0.01, 0.05, 0.1]
            candidates = [c for c in candidates if 0 <= c < 1.0]
            
            best_c = old_val
            best_local = best_obj
            
            for c in candidates:
                scale = (1.0 - c) / denom
                f_trial = f.copy()
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] = f[mask] * scale
                f_trial[j] = c
                val = fast_convolve(f_trial).max()
                if val < best_local - 1e-14:
                    best_local = val
                    best_c = c
            
            if abs(best_c - old_val) > 1e-10:
                scale = (1.0 - best_c) / denom
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] = f[mask] * scale
                f[j] = best_c
                f = f / f.sum()
                if best_local < best_obj - 1e-13:
                    best_obj = best_local
                    best_f = f.copy()
                improved = True
        
        if not improved and passes > 2:
            break
    
    return best_f, best_obj


def make_geometric(n, r):
    """Geometric sequence normalized to sum=1."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_symmetric_geo(n, r):
    """Symmetric (rising-falling) geometric sequence."""
    half = (n + 1) // 2
    asc = r ** np.arange(half, dtype=np.float64)
    desc = r ** np.arange(half - 1, 0, -1, dtype=np.float64)
    a = np.concatenate([asc, desc])
    if len(a) < n:
        a = np.concatenate([a, np.ones(n - len(a)) * a[-1]])
    elif len(a) > n:
        a = a[:n]
    return a / a.sum()


def make_cosine(n, power=2.0):
    """Cosine window raised to a power."""
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** power + 0.005
    return a / a.sum()


def make_bell(n, width=0.3):
    """Gaussian bell-shaped sequence."""
    x = np.linspace(0, 1, n)
    center = 0.5
    a = np.exp(-((x - center) ** 2) / (2 * width ** 2)) + 0.001
    return a / a.sum()


def make_random_smooth(n, seed):
    """Random smooth sequence (interpolated random values)."""
    rng = np.random.default_rng(seed)
    n_ctrl = max(3, n // 20)
    ctrl = rng.random(n_ctrl) + 0.1
    x_ctrl = np.linspace(0, 1, n_ctrl)
    x = np.linspace(0, 1, n)
    a = np.interp(x, x_ctrl, ctrl)
    a = np.maximum(a, 0)
    ss = a.sum()
    if ss < 1e-15:
        return np.ones(n) / n
    return a / ss


def make_two_level(n, frac, seed):
    """Two-level sequence: fractions of entries are high vs low."""
    rng = np.random.default_rng(seed)
    a = np.full(n, 0.3)
    n_high = int(frac * n)
    high_idx = rng.choice(n, n_high, replace=False)
    a[high_idx] = 1.0
    return a / a.sum()


def interpolate_seq(f, new_n):
    """Linear interpolation to new length, normalized to sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
