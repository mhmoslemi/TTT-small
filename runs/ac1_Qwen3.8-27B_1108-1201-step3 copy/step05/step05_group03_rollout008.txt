Minimizes 2*n*max(auto_conv)/(sum)^2 using:
1. Diverse initialization (geometric ramps, cosine, parent perturbation)
2. Smooth subgradient descent with adaptive temperature
3. Precise coordinate descent with wide search brackets

import numpy as np
from scipy.optimize import minimize_scalar
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def geometric_sequence(n, r):
    a = r ** np.arange(n)
    return normalize(a)


def cosine_sequence(n):
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** 2
    return normalize(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    return normalize(np.interp(new_x, old_x, f))


def smooth_subgradient(a, time_limit, T0=0.01, T_min=0.001, decay=0.9999,
                       lr0=0.005, top_k=50, verbose=False):
    """Smooth subgradient with adaptive temperature."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    T = T0
    cur_lr = lr0
    t0 = time.monotonic()
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        b = fast_convolve(a)
        
        # Compute softmax weights on top-k lags
        k_actual = min(top_k, len(b))
        # Focus on the region near the max
        max_val = b.max()
        # Use soft weights
        scaled = b / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        # Compute gradient: da_j = sum_k w[k] * (a[k-j] + a[j-k])
        # Use FFT for efficiency
        # c1[j] = sum_k w[k] * a[k-j] = (w_reversed * a)[j + (len(w)-1)]... 
        # Actually, let's just do it directly for the top few lags
        grad = np.zeros(n)
        # Find lags with significant weight
        active = np.where(w > w.max() * 0.01)[0]
        for k in active:
            wk = w[k]
            if wk < 1e-10:
                continue
            # db[k]/da[j] = a[k-j] + a[j-k] (with 0 for out of range)
            # a[k-j] is valid when 0 <= k-j < n, i.e., j in [k-n+1, k]
            lo1 = max(0, k - n + 1)
            hi1 = min(n - 1, k)
            if lo1 <= hi1:
                idx = np.arange(lo1, hi1 + 1)
                grad[idx] += wk * a[k - idx]
            # a[j-k] is valid when 0 <= j-k < n, i.e., j in [k, k+n-1]
            lo2 = max(0, k)
            hi2 = min(n - 1, k + n - 1)
            if lo2 <= hi2:
                idx = np.arange(lo2, hi2 + 1)
                grad[idx] += wk * a[idx - k]
        
        # Mirror descent step
        a = np.maximum(a * np.exp(-cur_lr * grad), 1e-15)
        a = a / a.sum()
        
        sc = compute_score(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
        
        # Adapt T and lr
        T = max(T * decay, T_min)
        cur_lr *= 0.99999
        
        if verbose and step % 500 == 0:
            print(f"      SG step {step}: score={best_score:.10f}, T={T:.6f}, lr={cur_lr:.6f}")
    
    return best_a, best_score


def coord_descent_wide(a, time_limit, verbose=False):
    """Coordinate descent with wide search range and parabolic refinement."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    stagnant = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        order = np.random.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                    break
            
            old_val = a[j]
            
            # Define objective: fix a[j] = v, rescale others to maintain sum=1
            def obj_v(v):
                v = max(0.0, v)
                if v >= 1.0:
                    return float('inf')
                other_sum = 1.0 - old_val
                if other_sum < 1e-12:
                    return float('inf')
                new_a = a.copy()
                new_a[j] = v
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                new_a[mask] *= (1.0 - v) / other_sum
                conv = fast_convolve(new_a)
                return conv.max()
            
            # Wide search: try several brackets
            best_v = old_val
            best_obj = obj_v(old_val)
            
            # Try geometric spread of candidate values
            candidates = [old_val * f for f in [0.1, 0.3, 0.5, 0.7, 1.3, 1.7, 2.0, 3.0, 5.0]]
            candidates = [c for c in candidates if 0 < c < 1.0]
            # Also try absolute values
            candidates.extend([0.001, 0.005, 0.01, 0.05, 0.1, 0.2])
            candidates = [c for c in candidates if 0 <= c < 0.9999]
            
            for cv in candidates:
                ov = obj_v(cv)
                if ov < best_obj:
                    best_obj = ov
                    best_v = cv
            
            if best_v != old_val:
                # Refine with golden section in a neighborhood of best_v
                lo = max(0.0, best_v * 0.5)
                hi = min(0.999, best_v * 2.0)
                if hi - lo < 1e-8:
                    lo = max(0.0, best_v - 0.01)
                    hi = min(0.999, best_v + 0.01)
                
                try:
                    res = minimize_scalar(obj_v, bounds=(lo, hi), method='bounded',
                                          options={'xatol': 1e-10, 'maxiter': 40})
                    if res.success and res.fun < best_obj:
                        best_v = res.x
                        best_obj = res.fun
                except Exception:
                    pass
                
                # Apply
                other_sum = 1.0 - old_val
                if other_sum > 1e-12:
                    new_a = a.copy()
                    new_a[j] = best_v
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    new_a[mask] *= (1.0 - best_v) / other_sum
                    new_a = normalize(new_a)
                    sc = compute_score(new_a)
                    if sc < best_score - 1e-14:
                        a = new_a
                        best_score = sc
                        best_a = a.copy()
                        improved = True
        
        stagnant = 0 if improved else stagnant + 1
        if stagnant >= 3:
            break
    
    return best_a, best_score


def pair_exchange(a, time_limit, n_iter=5000, verbose=False):
    """Random pair exchange: transfer mass between two coordinates."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    rng = np.random.default_rng(99)
    t0 = time.monotonic()
    
    for _ in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        
        a_i, a_j = a[i], a[j]
        if a_i < 1e-13 and a_j < 1e-13:
            continue
        
        # Try transferring various fractions
        best_sc = best_score
        best_f = 0.0
        best_dir = 0  # 0: no change, 1: j->i, -1: i->j
        
        for f in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5]:
            # j -> i
            delta = a_j * f
            if delta > 0:
                new_a = a.copy()
                new_a[j] = a_j - delta
                new_a[i] = a_i + delta
                sc = compute_score(new_a)
                if sc < best_sc:
                    best_sc = sc
                    best_f = f
                    best_dir = 1
            
            # i -> j
            delta = a_i * f
            if delta > 0:
                new_a = a.copy()
                new_a[i] = a_i - delta
                new_a[j] = a_j + delta
                sc = compute_score(new_a)
                if sc < best_sc:
                    best_sc = sc
                    best_f = f
                    best_dir = -1
        
        if best_dir != 0:
            delta = (a_j if best_dir == 1 else a_i) * best_f
            if best_dir == 1:
                a[j] -= delta
                a[i] += delta
            else:
                a[i] -= delta
                a[j] += delta
            a = normalize(a)
            if best_sc < best_score:
                best_score = best_sc
                best_a = a.copy()
    
    return best_a, best_score
