3. Use soft-max: f(z) ≈ 2n * (1/β) * log(sum_k exp(β * (a*a)_k))
4. Optimize z using L-BFGS or Adam
5. After convergence, a = softmax(z) is our answer

This is a clean, differentiable optimization! The main computational cost is computing (a*a) and its gradient at each step, which can be done via FFT in O(n log n).

Let me estimate the speed:
- n = 512
- Each function evaluation: FFT + some arithmetic ≈ O(n log n) ≈ 512 * 9 ≈ 4600 ops
- L-BFGS typically needs ~100-1000 iterations
- Total: ~10^7 ops, should run in < 1 second

This should be MUCH faster than LP-FW! And it can find a better optimum because it's a proper gradient-based method.

The key question is: how smooth is the log-sum-exp approximation? With β = 50-100, it should be a good approximation to the max while remaining smooth enough for gradient methods.

with softmax parameterization, multi-length search, and homotopy in the
smoothness parameter beta.

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax(z):
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def compute_grad_a(w, a):
    """
    Compute grad_a[j] = 2 * sum_m a[m] * w[j+m]
    where w is the weight vector (length 2n-1) and a is length n.
    """
    n = len(a)
    w_flipped = w[::-1]
    conv = np.convolve(a, w_flipped)
    return 2.0 * conv[n-1:n-1+n]

def lse_objective(z, beta):
    """Compute the smooth objective and gradient w.r.t. z."""
    a = softmax(z)
    n = len(a)
    b = fast_convolve(a)
    
    max_b = b.max()
    scaled = beta * (b - max_b)
    log_sum = max_b + np.log(np.sum(np.exp(scaled))) / beta
    
    # Weights for the LSE
    w = np.exp(scaled)
    w /= w.sum()
    
    # Gradient w.r.t. a
    grad_a = compute_grad_a(w, a)
    
    # Gradient w.r.t. z (softmax chain rule)
    dot = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot)
    
    return log_sum, grad_z

            val, _ = lse_objective(z, beta)
            return val

            _, g = lse_objective(z, beta)
            return g
        
        res = minimize(fun, z, jac=jac, method='L-BFGS-B', 
                      options={'maxiter': n_steps_per_beta, 'maxfun': n_steps_per_beta * 3})
        
        if res.success or res.fun < fun(z):

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth log-sum-exp (LSE) 
optimization with softmax parameterization, multi-length search, homotopy on 
the smoothness parameter, and spectral domain polishing.

Key innovations over previous approach:
- Replaces slow LP-Frank-Wolfe with fast L-BFGS on a smooth surrogate
- Uses softmax parameterization to handle constraints elegantly
- Homotopy on LSE parameter for global convergence
- Multi-length exploration with diverse initializations
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_and_grad(z, beta):
    """
    Compute the log-sum-exp smooth approximation of max(b) and its gradient w.r.t. z.
    b = auto-convolution of softmax(z).
    """
    a = softmax(z)
    n = len(a)
    b = fast_convolve(a)
    
    # LSE computation (numerically stable)
    max_b = b.max()
    scaled = beta * (b - max_b)
    log_sum_exp = max_b + np.log(np.sum(np.exp(scaled))) / beta
    
    # Softmax weights over b
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(b)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= w_sum
    
    # Gradient of LSE w.r.t. a:
    # grad_a[m] = 2 * sum_{n'} a[n'] * w[m + n']
    # This is 2 * (w cross-correlated with a)[m]
    # Using FFT for efficiency
    total_len = len(w) + n - 1  # 3n - 2
    fft_sz = 1 << (total_len - 1).bit_length()
    
    W = np.fft.rfft(w, fft_sz)
    A = np.fft.rfft(a, fft_sz)
    # Cross-correlation (w ⋆ a)[m] = sum_n' a[n'] * w[m+n']
    # = IFFT(conj(FFT(a)) * FFT(w))[:n]
    cross = np.fft.irfft(np.conj(A) * W, fft_sz)[:n]
    grad_a = 2.0 * cross
    
    # Chain rule: z -> a (softmax) -> b -> LSE
    # a_j = exp(z_j) / sum(exp(z_i))
    # da_j / dz_i = a_j * (delta_{ij} - a_i)
    # dLSE/dz_i = sum_j (dLSE/da_j) * (da_j/dz_i)
    # = sum_j grad_a[j] * a_j * (delta_{ij} - a_i)
    # = a_i * (grad_a[i] - sum_j grad_a[j] * a_j)
    
    dot = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot)
    
    return log_sum_exp, grad_z


def run_lse_optimization(z0, beta_schedule, max_iter_per_beta=300, time_limit=120.0):
    """
    Run L-BFGS optimization with homotopy on beta.
    """
    t0 = time.monotonic()
    z = z0.copy()
    best_z = z0.copy()
    best_val = np.inf
    
    for beta in beta_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        
        def fun(z):
            val, _ = lse_and_grad(z, beta)
            return val
        
        def jac(z):
            _, g = lse_and_grad(z, beta)
            return g
        
        try:
            res = minimize(fun, z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter_per_beta,
                                   'maxfun': max_iter_per_beta * 5,
                                   'gtol': 1e-8})
            if res.fun < best_val:
                best_val = res.fun
                best_z = res.x.copy()
            z = res.x
        except Exception:
            # If L-BFGS fails, try a simpler gradient step
            val, grad = lse_and_grad(z, beta)
            # Gradient descent with backtracking
            lr = 1.0
            for _ in range(10):
                z_new = z - lr * grad
                val_new, _ = lse_and_grad(z_new, beta)
                if val_new < val:
                    z = z_new
                    break
                lr *= 0.5
            if val < best_val:
                best_val = val
                best_z = z.copy()
    
    return best_z


def subgradient_polish(a0, time_limit=60.0):
    """
    Fast subgradient-based polishing using multiplicative updates.
    Targets the top peaks with adaptive learning rate.
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    lr = 0.02
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        
        # Use soft weights over top peaks (temperature annealing)
        top_k = min(15, len(b))
        peak_indices = np.argsort(b)[-top_k:]
        pv = b[peak_indices]
        temp = 0.005 + 0.02 * max(0, 1 - it / 500)
        w = np.exp((pv - pv.max()) / temp)
        w /= w.sum()
        
        # Compute weighted subgradient
        grad_a = np.zeros(n)
        for i in range(len(peak_indices)):
            if w[i] < 1e-10:
                continue
            k = int(peak_indices[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad_a[j] += 2.0 * w[i] * a[k - j]
        
        # Multiplicative update (mirror descent)
        a = a * np.exp(-lr * grad_a)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive learning rate
        if no_improve > 30:
            lr *= 0.7
            no_improve = 0
        elif no_improve == 0:
            lr = min(lr * 1.03, 0.15)
        
        if lr < 1e-7:
            break
    
    return best_a


def spectral_polish(a0, time_limit=30.0):
    """
    Spectral domain polish: flatten the power spectrum to reduce auto-conv peaks.
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    lr = 0.1
    
    while time.monotonic() - t0 < time_limit:
        # Compute FFT
        fft_size = 1 << (2 * n - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        
        # Target: flatten the power spectrum
        # Replace power with geometric mean of current and flat
        target = np.ones_like(power) * power.mean()
        blend = 0.5  # Blend factor
        new_power = blend * power + (1 - blend) * target
        
        # Reconstruct: keep phase, adjust magnitude
        new_mag = np.sqrt(np.maximum(new_power, 1e-30))
        phase = A / np.maximum(np.abs(A), 1e-30)
        new_A = new_mag * phase
        
        # Inverse FFT
        a_new = np.fft.irfft(new_A, fft_size)[:n]
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.1, 0.5)
        else:
            lr *= 0.7
            if lr < 0.01:
                break
    
    return best_a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_diverse_inits(n, seed=42):
    """Create diverse initial sequences of length n."""
    rng = np.random.default_rng(seed)
    x = np.arange(n) / max(n - 1, 1)
    inits = []
    
    # 1. Uniform
    inits.append(np.ones(n) / n)
    
    # 2. Raised cosine (bell)
    a = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
    a = np.maximum(a, 1e-10)
    inits.append(a / a.sum())
    
    # 3. Inverse cosine (U-shape) - might help flatten edges
    a = 0.5 * (1 - np.cos(2 * np.pi * (x - 0.5)))
    a = np.maximum(a, 1e-10)
    inits.append(a / a.sum())
    
    # 4. Cosine-modulated
    for freq in [3, 5, 7]:
        a = 1.0 + 0.3 * np.cos(2 * np.pi * freq * x)
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())
    
    # 5. Random (softmax)
    z = rng.standard_normal(n)
    inits.append(softmax(z))
    
    # 6. Gaussian
    center = np.arange(n) - n / 2.0
    a = np.exp(-0.5 * (center / (n * 0.25)) ** 2)
    inits.append(a / a.sum())
    
    # 7. Bi-modal
    c1, c2 = n * 0.25, n * 0.75
    a = np.exp(-0.5 * ((np.arange(n) - c1) / (n * 0.1)) ** 2) + \
        np.exp(-0.5 * ((np.arange(n) - c2) / (n * 0.1)) ** 2)
    inits.append(a / a.sum())
    
    return inits
