    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    conv = fast_convolve(f)
    best_obj = conv.max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    temperature = 0.001
    lr = 0.05
    top_k = 30
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  MD iter {it} ({elapsed:.1f}s): obj={best_obj:.10f}, temp={temperature:.6f}")
        
        conv = fast_convolve(f)
        M = conv.max()
        
        # Find active peaks
        threshold = M - temperature
        active = np.where(conv >= threshold)[0]
        
        # Compute weights (softmax over active peaks)
        pv = conv[active]
        scaled = (pv - M) / temperature  # negative or zero
        w = np.exp(scaled)
        w /= w.sum()
        
        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(active)):
            k = int(active[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            indices = np.arange(j_start, j_end + 1)
            sg[indices] += 2.0 * w[i] * f[k - indices]
        
        # Multiplicative update (mirror descent with negative entropy)
        # f_new[j] ∝ f[j] * exp(-lr * sg[j] / max(sg))
        sg_max = sg.max()
        if sg_max < 1e-15:
            break
        f_new = f * np.exp(-lr * sg / sg_max)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()
        
        # Evaluate
        new_conv = fast_convolve(f_new)
        new_obj = new_conv.max()
        
        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        
        # Annealing
        temperature *= 1.002
        lr *= 0.9998

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast mirror descent (multiplicative
weight updates with softmax-weighted subgradients), LP-FW polishing, adaptive
temperature scheduling, multi-start diversification, and final coordinate
descent squeezing. Key improvements: FFT-accelerated subgradients, no LP
relaxation needed, aggressive iteration count via O(n) updates, and diverse

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, conv, top_k, temperature):
    """Compute weighted subgradient of max(conv) using softmax over top peaks."""
    n = len(f)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    
    pv = conv[peaks]
    scaled = (pv - pv.max()) / temperature
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start > j_end:
            continue
        indices = np.arange(j_start, j_end + 1)
        sg[indices] += 2.0 * w[i] * f[k - indices]
    
    return sg


def mirror_descent(f0, time_limit, verbose=False, top_k=40, t0_temp=0.0005):

    f = np.maximum(f0.astype(np.float64), 0.0)

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {'time_limit': 30, 'presolve': True}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return res.x
    return None

def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with multiplicative updates on individual elements."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        cur_conv = fast_convolve(f)
        cur_max = cur_conv.max()
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            best_j_val = old_val
            best_j_obj = cur_max
            
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.6]:
                new_val = old_val * mult
                if new_val < 1e-15 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved and passes > 3:
            break
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

            continue
        print(f"  n={target_n}: t={t_alloc:.0f}s")
        a_opt, _ = mirror_descent(f_interp, time_limit=t_alloc,
                                  top_k=50, t0_temp=0.0003)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST (n={target_n}): {sc:.6f}")
    
    # === Phase 7: Final LP-FW squeeze ===
    print(f"\n=== Phase 7: Final LP-FW Squeeze (best={best_score:.6f}) ===")
    t_remaining = deadline - time.monotonic() - 20
    if t_remaining > 15:
        t_sq = min(t_remaining * 0.6, 120)
        print(f"  Budget: {t_sq:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_sq)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")
    
    # === Phase 8: Final micro perturbations ===
    print(f"\n=== Phase 8: Micro Perturbations (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    for trial in range(15):
        t_remaining = deadline - time.monotonic()
        if t_remaining < 3:
            break
        scale = 0.0002 + trial * 0.0001
        noise = rng.standard_normal(n_best) * scale
        f_pert = project_simplex(best_seq + noise)
        sc = float(evaluate_sequence(f_pert.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = f_pert.copy()
            print(f"  NEW BEST [pert{trial}]: {sc:.6f}")
            # Quick MD polish
            t_remaining = deadline - time.monotonic()
            if t_remaining > 5:
                a_opt, _ = mirror_descent(f_pert, time_limit=min(t_remaining * 0.3, 10),
                                          top_k=40, t0_temp=0.0001)
                sc2 = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = a_opt.copy()
                    print(f"  NEW BEST [pert{trial}_md]: {sc2:.6f}")
    
    # Final mirror descent burst with remaining time
    t_remaining = deadline - time.monotonic()
    if t_remaining > 10:
        a_opt, _ = mirror_descent(best_seq, time_limit=min(t_remaining * 0.8, 60),
                                  top_k=80, t0_temp=0.00005)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  FINAL MD: {sc:.6f}")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
