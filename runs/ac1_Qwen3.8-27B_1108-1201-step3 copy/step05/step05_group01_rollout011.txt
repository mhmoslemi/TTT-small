def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_smooth_grad(a, beta):
    """Compute gradient of log-sum-exp smoothed max of auto-convolution."""
    n = len(a)
    conv = fast_convolve(a)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()

    fft_size = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    R = np.fft.irfft(W * np.conj(A), fft_size)[:n]
    grad = 2.0 * R
    return grad, w, conv

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using accelerated gradient descent
with log-sum-exp smoothing, LP-Frank-Wolfe hybrid, and multi-length search.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_smooth_grad(a, beta):
    n = len(a)
    conv = fast_convolve(a)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    
    fft_size = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    R = np.fft.irfft(W * np.conj(A), fft_size)[:n]
    grad = 2.0 * R
    return grad, w, conv

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_direction(f, M, n):
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.success:
        return res.x
    return None

def accelerated_sgd(f0, time_limit, betas=None, verbose=False):
    """Accelerated gradient descent with Nesterov momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    if betas is None:
        betas = [10, 20, 50, 100, 200, 400, 800, 1600, 3200]
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    
    for beta_idx, beta in enumerate(betas):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        phase_time = remaining * 0.85 / (len(betas) - beta_idx)
        phase_end = time.monotonic() + phase_time
        
        # Nesterov parameters
        t = 1.0
        mu = 0.9
        y = f.copy()
        v = np.zeros(n)
        lr = 0.5 / (beta + 2 * n)
        no_improve = 0
        
        it = 0
        while time.monotonic() < phase_end:
            # Nesterov lookahead
            y = f + (t - 1) / (t + 1) * (f - y) if t > 1 else f
            # Actually, standard Nesterov: y = f + (beta_t - 1)/(beta_{t+1}) * (f - y_prev)
            # Simpler: just use momentum
            grad, w, conv = compute_smooth_grad(y, beta)
            grad_tangent = grad - np.mean(grad)
            
            v = mu * v - lr * grad_tangent
            f_new = project_simplex(y + v)
            
            obj = fast_convolve(f_new).max()
            
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_f = f_new.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 200:
                    lr *= 0.9
                    no_improve = 0
            
            f = f_new
            t += 1
            it += 1
            
            if verbose and it % 10000 == 0:
                print(f"      AGD beta={beta}, it={it}, obj={best_obj:.10f}, lr={lr:.6f}")
    
    return best_f, best_obj
