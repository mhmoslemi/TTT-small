"""
Hybrid optimizer combining LP-based best response directions, exact coordinate
descent, 2D pair optimization, smoothed L-BFGS-B, and multi-scale search to
minimize 2*n*max(auto_conv)/sum^2. Key novel elements: LP dual-guided
subgradient, simultaneous 2D coordinate moves, and adaptive n exploration.

import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def geometric_sequence(n, r):
    a = r ** np.arange(n)
    return normalize(a)


def cosine_sequence(n):
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** 2
    return normalize(a)


def gaussian_sequence(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-x ** 2 / (2.0 * sigma ** 2))
    return normalize(a)


def symmetric_geometric(n, r):
    """Symmetric (palindromic) geometric-like sequence."""
    half = n // 2
    if n % 2 == 0:
        a_half = r ** np.arange(half)
        a = np.concatenate([a_half, a_half[::-1]])
    else:
        a_half = r ** np.arange(half + 1)
        a = np.concatenate([a_half, a_half[-2::-1]])
    return normalize(a)


def flat_target_sequence(n):
    """Sequence whose convolution is close to flat: based on central binomial coefficients."""
    if n % 2 == 0:
        k = np.arange(n // 2)
        coeffs = _binom_2k_k(k)
        a_half = coeffs
        a = np.concatenate([a_half, a_half[::-1]])
    else:
        k = np.arange(n // 2 + 1)
        coeffs = _binom_2k_k(k)
        a_half = coeffs
        a = np.concatenate([a_half, a_half[-2::-1]])
    return normalize(a)


def _binom_2k_k(k):
    """Compute C(2k,k)/4^k for array of k values."""
    k = np.asarray(k, dtype=np.float64)
    # Use log-gamma for numerical stability
    from scipy.special import gammaln
    log_val = gammaln(2 * k + 1) - 2 * gammaln(k + 1) - k * np.log(4.0)
    return np.exp(log_val)


def lp_best_response(f, M, time_per_lp=0.5):
    """Solve: max sum(b) s.t. b >= 0, (f*b)_k <= M for all k.
    Returns the optimal b vector."""
    n = len(f)
    # Build constraint matrix: (f*b)_k = sum_j f[k-j] * b[j]
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        js = np.arange(j_lo, j_hi + 1)
        A_ub[k, js] = f[k - js]

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        t_start = time.monotonic()
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'time_limit': time_per_lp})
        if result.success:
            return np.maximum(result.x, 0.0)
        else:
            return None
    except Exception:
        return None


def lp_dual_weights(f, M, time_per_lp=0.5):
    """Solve LP dual to get importance weights for subgradient."""
    n = len(f)
    # Dual: min M * sum(lambda) s.t. sum_k lambda_k * f[k-j] >= 1 for all j
    # => min M * sum(lambda) s.t. A^T * lambda >= 1, lambda >= 0
    # where A[k,j] = f[k-j] (same as primal constraint matrix)

    A_ub = np.zeros((n, 2 * n - 1))
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        js = np.arange(j_lo, j_hi + 1)
        A_ub[js, k] = f[k - js]

    c = M * np.ones(2 * n - 1)
    A_ub_dual = -A_ub  # We need A^T * lambda >= 1, i.e., -A^T * lambda <= -1
    b_ub_dual = -np.ones(n)
    bounds = [(0, None)] * (2 * n - 1)

    try:
        result = linprog(c, A_ub=A_ub_dual, b_ub=b_ub_dual, bounds=bounds,
                         method='highs', options={'time_limit': time_per_lp})
        if result.success:
            lam = np.maximum(result.x, 0.0)
            s = lam.sum()
            if s > 1e-15:
                return lam / s
    except Exception:
        pass
    return None


def subgradient_lp_guided(a, n_steps, time_limit, lr0=0.01, decay=0.9998,
                          use_lp=True, verbose=False):
    """Subgradient method with LP-dual-guided weights."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    cur_lr = lr0
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        M = conv.max()

        # Get weights: either LP dual or softmax over top peaks
        if use_lp and step % 10 == 0:
            lam = lp_dual_weights(a, M, time_per_lp=0.3)
            if lam is None:
                # Fallback to softmax
                k_actual = min(50, len(conv))
                peaks = np.argpartition(conv, -k_actual)[-k_actual:]
                pv = conv[peaks]
                temp = max(1e-6, pv.max() - pv.min()) * 0.5
                scaled = pv / temp
                scaled -= scaled.max()
                w = np.exp(scaled)
                w /= w.sum()
                lam = np.zeros(2 * n - 1)
                lam[peaks] = w
        else:
            k_actual = min(50, len(conv))
            peaks = np.argpartition(conv, -k_actual)[-k_actual:]
            pv = conv[peaks]
            temp = max(1e-6, pv.max() - pv.min()) * 0.5
            scaled = pv / temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            w /= w.sum()
            lam = np.zeros(2 * n - 1)
            lam[peaks] = w

        # Compute subgradient
        sg = np.zeros(n)
        active = np.where(lam > 1e-12)[0]
        for k in active:
            w_k = lam[k]
            if w_k < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w_k * a[k - idx]

        # Multiplicative update
        a = np.maximum(a * np.exp(-cur_lr * sg), 1e-15)
        a = a / a.sum()

        sc = compute_score(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()

        cur_lr *= decay

        if verbose and step % 500 == 0:
            print(f"      SG-LP step {step}: score={best_score:.10f}, lr={cur_lr:.6f}")

    return best_a, best_score


def coord_descent_1d(a, time_limit, verbose=False):
    """Exact 1D coordinate descent with renormalization."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    pass_count = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 3:
        pass_count += 1
        improved = False
        no_improve = 0
        order = np.random.permutation(n) if pass_count > 1 else np.arange(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            other_sum = 1.0 - old_val
            if other_sum < 1e-12:
                continue

            def obj_v(v):
                v = max(0.0, min(v, 1.0))
                new_a = a.copy()
                new_a[j] = v
                scale = (1.0 - v) / other_sum
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                new_a[mask] *= scale
                conv = fast_convolve(new_a)
                return conv.max()

            # Determine search bounds
            if old_val < 1e-10:
                lo, hi = 0.0, 0.1
            else:
                lo = max(0.0, old_val * 0.2)
                hi = min(0.999, old_val * 5.0 + 0.05)
                if hi - lo < 1e-6:
                    lo = max(0.0, old_val - 0.05)
                    hi = min(0.999, old_val + 0.05)

            try:
                res = minimize_scalar(obj_v, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-10, 'maxiter': 40})
                if res.success and res.fun < obj_v(old_val) - 1e-15:
                    v_new = res.x
                    new_a = a.copy()
                    new_a[j] = v_new
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    scale = (1.0 - v_new) / other_sum
                    new_a[mask] *= scale
                    new_a = np.maximum(new_a, 0.0)
                    s = new_a.sum()
                    if s > 1e-15:
                        new_a /= s
                    sc = compute_score(new_a)
                    if sc < best_score - 1e-14:
                        a = new_a
                        best_score = sc
                        best_a = a.copy()
                        improved = True
            except Exception:
                pass

        if not improved:
            no_improve += 1

        if verbose and pass_count % 3 == 0:
            print(f"      CD pass {pass_count}: score={best_score:.10f}")

        if pass_count > 300:
            break

    return best_a, best_score


def coord_descent_2d(a, time_limit, n_pairs=100, verbose=False):
    """2D coordinate descent: optimize pairs of coordinates simultaneously."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    rng = np.random.default_rng(42)
    t0 = time.monotonic()
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        improved = False

        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break

            i, j = rng.integers(0, n, 2)
            if i == j:
                continue

            old_i, old_j = a[i], a[j]
            other_sum = 1.0 - old_i - old_j
            if other_sum < 1e-12:
                continue

            def obj_2d(v_i, v_j):
                v_i = max(0.0, min(v_i, 1.0))
                v_j = max(0.0, min(v_j, 1.0))
                if v_i + v_j > 1.0 - 1e-12:
                    return float('inf')
                scale = (1.0 - v_i - v_j) / other_sum
                new_a = a.copy()
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                mask[j] = False
                new_a[mask] *= scale
                new_a[i] = v_i
                new_a[j] = v_j
                conv = fast_convolve(new_a)
                return conv.max()

            # Grid search in 2D (small grid around current point)
            best_v = (old_i, old_j)
            best_f = obj_2d(old_i, old_j)

            # Try a small grid
            di = max(0.01, old_i * 0.5)
            dj = max(0.01, old_j * 0.5)
            for fi in [-1, -0.5, 0, 0.5, 1]:
                for fj in [-1, -0.5, 0, 0.5, 1]:
                    vi = old_i + fi * di
                    vj = old_j + fj * dj
                    if vi < 0 or vj < 0 or vi + vj > 0.999:
                        continue
                    f = obj_2d(vi, vj)
                    if f < best_f - 1e-15:
                        best_f = f
                        best_v = (vi, vj)

            if best_v != (old_i, old_j):
                vi, vj = best_v
                scale = (1.0 - vi - vj) / other_sum
                new_a = a.copy()
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                mask[j] = False
                new_a[mask] *= scale
                new_a[i] = vi
                new_a[j] = vj
                new_a = np.maximum(new_a, 0.0)
                s = new_a.sum()
                if s > 1e-15:
                    new_a /= s
                sc = compute_score(new_a)
                if sc < best_score - 1e-14:
                    a = new_a
                    best_score = sc
                    best_a = a.copy()
                    improved = True

        if verbose and iteration % 10 == 0:
            print(f"      CD2D iter {iteration}: score={best_score:.10f}")

        if not improved and iteration > 3:
            break
        if iteration > 100:
            break

    return best_a, best_score


def pair_swap_local(a, time_limit, n_pers=200, verbose=False):
    """Random pair transfer local search."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    rng = np.random.default_rng(99999)
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        for _ in range(n_pers):
            if time.monotonic() - t0 > time_limit:
                break

            i, j = rng.integers(0, n, 2)
            if i == j:
                continue

            a_j = a[j]
            a_i = a[i]
            if a_j < 1e-12 and a_i < 1e-12:
                continue

            best_delta = 0.0
            best_sc = best_score

            fracs = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.1, 1.3]
            for f in fracs:
                delta = a_j * f
                if a_j - delta < -1e-12:
                    continue
                new_a = a.copy()
                new_a[j] = max(0.0, a_j - delta)
                new_a[i] = a_i + max(0.0, delta)
                # Also try reverse
                if a_i > 1e-12:
                    delta_r = a_i * f
                    new_a2 = a.copy()
                    new_a2[i] = max(0.0, a_i - delta_r)
                    new_a2[j] = a_j + max(0.0, delta_r)
                    conv2 = fast_convolve(new_a2)
                    sc2 = 2.0 * n * conv2.max()
                    if sc2 < best_sc - 1e-14:
                        best_sc = sc2
                        best_delta = -delta_r

                conv = fast_convolve(new_a)
                sc = 2.0 * n * conv.max()
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_delta = delta

            if best_delta != 0.0:
                if best_delta > 0:
                    a[j] -= best_delta
                    a[i] += best_delta
                else:
                    a[i] -= (-best_delta)
                    a[j] += (-best_delta)
                a = np.maximum(a, 0.0)
                a = normalize(a)
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

    return best_a, best_score


def optimize_pipeline(a, time_limit, verbose=False):
    """Multi-stage optimization pipeline."""
    n = len(a)
    a = normalize(a.copy())

            best_score = sc_sg
            best_a = a_sg.copy()
            a = a_sg.copy()

    # Stage 2: Coordinate descent 1D

    t2 = min(rem * 0.3, 180)
    if rem > t2 + 30:
        a_cd, sc_cd = coord_descent_1d(a, time_limit=t2)

            best_score = sc_cd
            best_a = a_cd.copy()
            a = a_cd.copy()

    # Stage 3: 2D coordinate descent

            best_a = a_final.copy()
        if verbose:
            print(f"        After final CD: {best_score:.10f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)
