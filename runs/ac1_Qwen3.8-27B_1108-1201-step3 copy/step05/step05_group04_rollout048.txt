Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-guided sequential approximation
with adaptive step sizes, multi-start, and subgradient polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_direction(f, M, top_k=None):
    """
    Solve the LP: max sum(g) s.t. (f*g)(k) <= M for all k, g >= 0.
    Returns the optimal g or None if infeasible.
    """
    n = len(f)
    num_conv = 2 * n - 1
    
    if top_k is not None and top_k < num_conv:
        # Only use top-k constraints
        conv = fast_convolve(f)
        top_idx = np.argsort(conv)[::-1][:top_k]
    else:
        top_idx = np.arange(num_conv)
    
    # Build constraint matrix A for (f*g)(k) <= M
    # A[k,j] = f(k-j) if 0 <= k-j <= n-1, else 0
    # But we only need rows for top_idx
    rows = []
    for k in top_idx:
        # (f*g)(k) = sum_j f(k-j) * g(j) for max(0,k-n+1) <= j <= min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            coeffs = np.zeros(n)
            coeffs[j_start:j_end+1] = f[k - j_end : k - j_start + 1][::-1]
            rows.append(coeffs)
    
    if not rows:
        return None
    
    A_ub = np.array(rows)
    b_ub = np.full(len(top_idx), M)
    
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status == 0:
        return res.x
    return None

def compute_subgradient(f, w):
    """Compute subgradient: sg[j] = sum_i w[i+j] * f[i] for j=0..n-1."""
    return np.correlate(w, f, 'valid')

def get_soft_max_weights(conv, temp, top_k=None):
    """Get soft-max weights over convolution values."""
    shifted = conv / temp
    shifted -= shifted.max()
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
        return w
    w /= ws
    if top_k is not None and top_k < len(conv):
        idx = np.argpartition(w, -top_k)[-top_k:]
        w_masked = np.zeros(len(conv))
        w_masked[idx] = w[idx]
        w_masked /= w_masked.sum()
        w = w_masked
    return w

def adam_optimize(f0, time_limit, lr=0.005, beta1=0.9, beta2=0.999,
                  eps=1e-8, temp_init=0.008, temp_decay=0.9995,

        sg = compute_subgradient(f, w)

        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f_new)

        temp = max(temp * temp_decay, 1e-7)
        cur_lr = lr * (0.9999 ** t)
        cur_lr = max(cur_lr, lr * 0.1)

            v = np.zeros(n)
            t = 0
            temp = max(temp_init * 0.7, 0.003)
            cur_lr = lr
            no_improve = 0

def lp_guided_descent(f0, time_limit, initial_top_k=100, max_top_k=500,
                      n_line_search=10, verbose=True):
    """
    LP-guided sequential approximation method.
    At each step, solve the LP to find a descent direction, then line search.
    """
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    conv = fast_convolve(f)
    best_obj = conv.max()
    
    t0 = time.monotonic()
    iteration = 0
    top_k = initial_top_k
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Solve LP to find direction
        g = solve_lp_direction(f, M, top_k=min(top_k, 2*n-1))
        
        if g is None:
            # LP failed, fall back to subgradient
            rng = np.random.default_rng(iteration * 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        
        # Rescale g to have same sum as f (= 1)
        g_scaled = g / g_sum
        
        # Line search
        best_t = 0
        best_val = best_obj
        
        for t_val in np.linspace(0.05, 0.95, n_line_search):
            f_trial = (1 - t_val) * f + t_val * g_scaled
            f_trial = np.maximum(f_trial, 0)
            f_trial = f_trial / f_trial.sum()
            trial_conv = fast_convolve(f_trial)
            trial_obj = trial_conv.max()
            if trial_obj < best_val:
                best_val = trial_obj
                best_t = t_val
        
        if best_t > 0:
            f_new = (1 - best_t) * f + best_t * g_scaled
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            f = f_new
            
            if best_val < best_obj - 1e-15:
                best_obj = best_val
                best_f = f.copy()
                if verbose and iteration % 10 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"    LP iter {iteration}: obj={best_obj:.10f}, t={best_t:.3f}, elapsed={elapsed:.1f}s")
        else:
            # No improvement, try different top_k or perturb
            top_k = min(top_k * 2, max_top_k)
            if top_k >= max_top_k:
                # Try perturbing
                rng = np.random.default_rng(iteration * 13)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                top_k = initial_top_k
                best_obj = fast_convolve(f).max()
                best_f = f.copy()
    
    return best_f, best_obj

def make_smooth_random_init(n, rng, n_components=5):
    """Create a smooth random initialization."""
    f = np.zeros(n)
    x = np.linspace(0, 1, n)
    for _ in range(n_components):
        c = rng.uniform(0.1, 0.9)
        s = rng.uniform(0.05, 0.25)
        a = rng.uniform(0.5, 2.0)
        f += a * np.exp(-0.5 * ((x - c) / s) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:

    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-guided sequential approximation,
Adam subgradient polishing, length exploration, and multi-start strategies.
Key innovation: actual LP solves (via HiGHS) for direction finding at each step,
which gives exact first-order optimal descent directions for the linearized problem.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, w):
    """Compute subgradient: sg[j] = sum_i w[i+j] * f[i] for j=0..n-1."""
    return np.correlate(w, f, 'valid')


def get_soft_max_weights(conv, temp, top_k=None):
    """Get soft-max weights over convolution values."""
    shifted = conv / temp
    shifted -= shifted.max()
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
        return w
    w /= ws
    if top_k is not None and top_k < len(conv):
        idx = np.argpartition(w, -top_k)[-top_k:]
        w_masked = np.zeros(len(conv))
        w_masked[idx] = w[idx]
        w_masked /= w_masked.sum()
        w = w_masked
    return w


def solve_lp_direction(f, M, top_k=None):
    """
    Solve LP: max sum(g) s.t. (f*g)(k) <= M for k in top_k, g >= 0.
    Returns optimal g vector or None if infeasible/unbounded.
    """
    n = len(f)
    num_conv = 2 * n - 1

    if top_k is not None and top_k < num_conv:
        conv = fast_convolve(f)
        top_idx = np.argsort(conv)[::-1][:top_k]
    else:
        top_idx = np.arange(num_conv)

    rows = []
    for k in top_idx:
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            coeffs = np.zeros(n)
            # (f*g)(k) = sum_{j} f(k-j) * g(j)
            # For j in [j_start, j_end], k-j is in [0, n-1]
            idx_f = k - np.arange(j_start, j_end + 1)
            coeffs[j_start:j_end + 1] = f[idx_f]
            rows.append(coeffs)

    if not rows:
        return None

    A_ub = np.array(rows)
    b_ub = np.full(len(top_idx), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def adam_optimize(f0, time_limit, lr=0.005, beta1=0.9, beta2=0.999,
                  eps=1e-8, temp_init=0.008, temp_decay=0.9995,
                  top_k=None, verbose=False):
    """Adam optimizer with simplex projection and soft-max subgradients."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    temp = temp_init

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)

        w = get_soft_max_weights(conv, temp, top_k)
        sg = compute_subgradient(f, w)

        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f_new)

        temp = max(temp * temp_decay, 1e-7)
        cur_lr = lr * (0.9999 ** t)
        cur_lr = max(cur_lr, lr * 0.1)

        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            rng = np.random.default_rng(t * 17 + 3)
            noise_scale = 0.003 + 0.002 * min(no_improve // 100, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            temp = max(temp_init * 0.7, 0.003)
            cur_lr = lr
            no_improve = 0

    return best_f, best_obj


def lp_guided_descent(f0, time_limit, initial_top_k=80, max_top_k=400,
                      n_line_search=12, verbose=True):
    """
    LP-guided sequential approximation: at each step solve LP for optimal
    descent direction, then line search for best step size.
    """
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    conv = fast_convolve(f)
    best_obj = conv.max()

    t0 = time.monotonic()
    iteration = 0
    top_k = initial_top_k
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        rem_time = time_limit - (time.monotonic() - t0)
        if rem_time < 2:
            break

        iteration += 1
        conv = fast_convolve(f)
        M = conv.max()

        g = solve_lp_direction(f, M, top_k=min(top_k, 2 * n - 1))

        if g is None:
            stall_count += 1
            if stall_count > 10:
                break
            continue

        g_sum = g.sum()
        if g_sum < 1e-10:
            break

        g_scaled = g / g_sum

        # Check if direction is meaningful (sum(g) significantly > 1)
        improvement_ratio = g_sum  # since sum(f)=1, ratio = g_sum/1

        # Line search with multiple candidates
        best_t = 0.0
        best_val = best_obj
        candidates = np.linspace(0.05, min(0.95, 0.5 * improvement_ratio), n_line_search)
        candidates = np.unique(np.concatenate([candidates, [0.1, 0.3, 0.5, 0.7]]))
        candidates = candidates[(candidates > 0) & (candidates < 1)]

        for t_val in candidates:
            f_trial = (1.0 - t_val) * f + t_val * g_scaled
            f_trial = np.maximum(f_trial, 0.0)
            ss = f_trial.sum()
            if ss < 1e-15:
                continue
            f_trial = f_trial / ss
            trial_conv = fast_convolve(f_trial)
            trial_obj = trial_conv.max()
            if trial_obj < best_val - 1e-16:
                best_val = trial_obj
                best_t = t_val

        if best_t > 0:
            f_new = (1.0 - best_t) * f + best_t * g_scaled
            f_new = np.maximum(f_new, 0.0)
            ss = f_new.sum()
            if ss > 1e-15:
                f = f_new / ss
            else:
                f = best_f.copy()

            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_f = f.copy()
                stall_count = 0
                top_k = max(30, top_k // 2)  # Reduce constraints as we converge
                if verbose and (iteration % 5 == 0 or best_val < best_obj):
                    elapsed = time.monotonic() - t0
                    print(f"    LP iter {iteration}: obj={best_obj:.10f}, t={best_t:.3f}%, g_sum={g_sum:.4f}, elapsed={elapsed:.1f}s")
        else:
            stall_count += 1
            top_k = min(top_k * 2, max_top_k)
            if stall_count > 15:
                # Perturb and restart
                rng = np.random.default_rng(iteration * 13 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                best_obj = fast_convolve(f).max()
                best_f = f.copy()
                top_k = initial_top_k
                stall_count = 0

    return best_f, best_obj


def subgradient_polish(f0, time_limit, temp=0.002, lr=0.002, top_k=None):
    """Quick subgradient polishing with decaying learning rate."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    t = 0
    cur_lr = lr

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)
        w = get_soft_max_weights(conv, temp, top_k)
        sg = compute_subgradient(f, w)

        f_new = f - cur_lr * sg
        f = project_simplex(f_new)
        cur_lr *= 0.9999

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()

    return best_f, best_obj


def make_smooth_random_init(n, rng, n_components=5):
    """Create a smooth random initialization as sum of Gaussians."""
    f = np.zeros(n)
    x = np.linspace(0, 1, n)
    for _ in range(n_components):
        c = rng.uniform(0.1, 0.9)
        s = rng.uniform(0.05, 0.25)
        a = rng.uniform(0.5, 2.0)
        f += a * np.exp(-0.5 * ((x - c) / s) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_symmetric_init(n, shape='bump'):

        f = np.sin(np.pi * x) ** 2

        x = np.linspace(0, 1, n)
        f = 1 - np.abs(2 * x - 1)
    else:
        f = np.ones(n)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_gaussian_init(n, center=0.5, sigma=0.15):
    """Create a Gaussian-like initialization."""
    x = np.linspace(0, 1, n)
    f = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def try_seq_update(state, a, label=""):
    """Update best state dict with a candidate sequence."""
    if a is None:
        return
    a = np.maximum(a, 0.0)
    if a.sum() < 0.01:
        return
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return
    if np.isfinite(score) and score < state['best_score']:
        state['best_score'] = score
        state['best_seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")


def sort_by_distance(arr, ref):
    """Return indices sorted by distance to ref."""
    return sorted(range(len(arr)), key=lambda i: abs(arr[i] - ref))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: LP-guided descent (exact direction finding),
    Adam subgradient, length exploration, and final polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    state = {'best_seq': None, 'best_score': float('inf')}

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if np.isfinite(p_score):
                state['best_score'] = p_score
                state['best_seq'] = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        print("No parent found, using uniform init n=768")

    n_parent = len(parent)

    # === Phase 1: LP-guided descent with diverse starts ===
    print(f"\n=== Phase 1: LP-guided descent (n={n_parent}) ===")
    t_phase1_end = time.monotonic() + 500

    initializations = {}
    initializations['parent'] = parent.copy()
    initializations['gauss_015'] = make_gaussian_init(n_parent, 0.5, 0.15)
    initializations['gauss_025'] = make_gaussian_init(n_parent, 0.5, 0.25)
    initializations['bump'] = make_symmetric_init(n_parent, 'bump')
    initializations['smooth_rand'] = make_smooth_random_init(n_parent, rng, 5)
    initializations['parent_pert'] = project_simplex(parent + rng.standard_normal(n_parent) * 0.01)

    for name, init in initializations.items():
        rem = t_phase1_end - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(rem * 0.4, 80)
        print(f"  LP init '{name}', t={t_alloc:.0f}s...")
        a_opt, obj = lp_guided_descent(init, time_limit=t_alloc,
                                        initial_top_k=60, max_top_k=300,
                                        n_line_search=10, verbose=False)
        try_seq_update(state, a_opt, f"p1_lp_{name}")

    # Adam from parent as well
    rem = t_phase1_end - time.monotonic()
    if rem > 40:
        a_opt, _ = adam_optimize(parent, time_limit=min(rem * 0.6, 100),
                                  lr=0.005, temp_init=0.006,
                                  temp_decay=0.9993, top_k=max(20, n_parent // 10))
        try_seq_update(state, a_opt, "p1_adam_parent")

    # === Phase 2: Extended LP refinement ===
    print(f"\n=== Phase 2: LP refinement (best={state['best_score']:.8f}) ===")
    t_phase2_end = time.monotonic() + 450
    base = state['best_seq']
    n_base = len(base)

    lp_configs = [
        (50, 15), (80, 12), (120, 10), (200, 8)
    ]

    for ci, (tk, nls) in enumerate(lp_configs):
        rem = t_phase2_end - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(rem * 0.55, 120)
        print(f"  LP config top_k={tk}, nls={nls}, t={t_alloc:.0f}s")
        a_opt, _ = lp_guided_descent(base, time_limit=t_alloc,
                                      initial_top_k=tk, max_top_k=500,
                                      n_line_search=nls, verbose=False)
        try_seq_update(state, a_opt, f"p2_lp_{ci}")
        base = state['best_seq']
        n_base = len(base)

        # Quick subgradient between LP runs
        rem = t_phase2_end - time.monotonic()
        if rem > 20:
            a_sub, _ = subgradient_polish(base, time_limit=min(rem * 0.3, 40),
                                           temp=0.002, lr=0.003)
            try_seq_update(state, a_sub, f"p2_sub_{ci}")
            base = state['best_seq']
            n_base = len(base)

    # === Phase 3: Alternating fine LP + subgradient ===
    print(f"\n=== Phase 3: Fine alternating (best={state['best_score']:.8f}) ===")
    t_phase3_end = time.monotonic() + 400
    base = state['best_seq']
    n_base = len(base)

    for alt_i in range(4):
        rem = t_phase3_end - time.monotonic()
        if rem < 25:
            break

        # Fine LP with small top_k (focus on most critical constraints)
        t_lp = min(rem * 0.4, 80)
        a_opt, _ = lp_guided_descent(base, time_limit=t_lp,
                                      initial_top_k=30, max_top_k=100,
                                      n_line_search=15, verbose=False)
        try_seq_update(state, a_opt, f"p3_fine_lp_{alt_i}")
        base = state['best_seq']
        n_base = len(base)

        rem = t_phase3_end - time.monotonic()
        if rem < 15:
            break

        # Subgradient with very low temperature
        t_sub = min(rem * 0.4, 50)
        a_sub, _ = subgradient_polish(base, time_limit=t_sub,
                                       temp=0.001, lr=0.001,
                                       top_k=max(10, n_base // 20))
        try_seq_update(state, a_sub, f"p3_fine_sub_{alt_i}")
        base = state['best_seq']
        n_base = len(base)

        # Small perturbation + quick LP
        rem = t_phase3_end - time.monotonic()
        if rem < 15:
            break
        noise = rng.standard_normal(n_base) * 0.003
        f_pert = project_simplex(base + noise)
        t_pert = min(rem * 0.4, 40)
        a_opt, _ = lp_guided_descent(f_pert, time_limit=t_pert,
                                      initial_top_k=40, max_top_k=150,
                                      n_line_search=10, verbose=False)
        try_seq_update(state, a_opt, f"p3_pert_{alt_i}")
        base = state['best_seq']
        n_base = len(base)

    # === Phase 4: Length exploration ===
    print(f"\n=== Phase 4: Length exploration (best={state['best_score']:.8f}) ===")
    t_phase4_end = time.monotonic() + 300
    base = state['best_seq']
    n_base = len(base)

    target_lengths = [1024, 640, 512, 896, 1280, 384]
    target_lengths = [t for t in target_lengths if t != n_base and 256 <= t <= 1536]
    # Sort by distance to current best
    idx_sorted = sort_by_distance(target_lengths, n_base)
    target_lengths = [target_lengths[i] for i in idx_sorted[:4]]

    for target_n in target_lengths:
        rem = t_phase4_end - time.monotonic()
        if rem < 50:
            break
        f_interp = interpolate_sequence(base, target_n)

        # Adam warm-up
        t_adam = min(rem * 0.3, 60)
        a_opt, _ = adam_optimize(f_interp, time_limit=t_adam,
                                  lr=0.004, temp_init=0.006,
                                  temp_decay=0.9994, top_k=max(15, target_n // 15))
        try_seq_update(state, a_opt, f"p4_adam_n{target_n}")

        rem = t_phase4_end - time.monotonic()
        if rem < 20:
            continue

        # LP refinement
        t_lp = min(rem * 0.5, 80)
        a_opt2, _ = lp_guided_descent(a_opt, time_limit=t_lp,
                                       initial_top_k=50, max_top_k=200,
                                       n_line_search=10, verbose=False)
        try_seq_update(state, a_opt2, f"p4_lp_n{target_n}")

    # === Phase 5: Final deep polish ===
    print(f"\n=== Phase 5: Final polish (best={state['best_score']:.8f}) ===")
    base = state['best_seq']
    n_base = len(base)

    rem = deadline - time.monotonic() - 15
    if rem > 30:
        # Long fine LP
        t_final = min(rem * 0.5, 100)
        print(f"  Final deep LP, t={t_final:.0f}s")
        a_opt, _ = lp_guided_descent(base, time_limit=t_final,
                                      initial_top_k=25, max_top_k=80,
                                      n_line_search=20, verbose=False)
        try_seq_update(state, a_opt, "p5_deep_lp")

        base = state['best_seq']
        n_base = len(base)

    rem = deadline - time.monotonic() - 10
    if rem > 20:
        # Subgradient final
        t_final_sub = min(rem * 0.4, 50)
        a_opt, _ = subgradient_polish(base, time_limit=t_final_sub,
                                       temp=0.0005, lr=0.0005,
                                       top_k=max(5, n_base // 30))
        try_seq_update(state, a_opt, "p5_final_sub")

        base = state['best_seq']
        n_base = len(base)

    # Tiny perturbations in remaining time
    rem = deadline - time.monotonic() - 5
    while rem > 15:
        noise = rng.standard_normal(n_base) * 0.001
        f_pert = project_simplex(base + noise)
        t_quick = min(rem * 0.4, 15)
        a_opt, _ = lp_guided_descent(f_pert, time_limit=t_quick,
                                      initial_top_k=30, max_top_k=80,
                                      n_line_search=8, verbose=False)
        try_seq_update(state, a_opt, "p5_micro")
        base = state['best_seq']
        n_base = len(base)
        rem = deadline - time.monotonic() - 3
        if not np.isfinite(base.sum()):
            break

    # Fallback
    if state['best_seq'] is None:
        state['best_seq'] = np.ones(512) / 512
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(state['best_seq'].tolist()))
        if np.isfinite(fs):
            state['best_score'] = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {state['best_score']:.10f} (n={len(state['best_seq'])})")
    return [float(x) for x in state['best_seq']]
