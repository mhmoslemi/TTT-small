import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def smooth_obj_grad(a, tau):
    """Smoothed objective and gradient on the simplex."""
    conv = fast_convolve(a)
    max_c = conv.max()
    shifted = np.clip((conv - max_c) / tau, -100, 100)
    E = np.exp(shifted)
    E_sum = E.sum()
    obj = max_c + tau * np.log(E_sum)
    w = E / E_sum
    
    n = len(a)
    grad_a = np.zeros(n)
    for j in range(n):
        k_lo = j
        k_hi = min(j + n, len(conv))
        grad_a[j] = 2.0 * np.dot(w[k_lo:k_hi], a[:k_hi - k_lo])
    
    return obj, grad_a

def adam_smooth_minimize(a_init, tau_init, time_limit, lr=0.01, verbose=False):
    """
    Minimize the smoothed objective using Adam optimizer on the simplex.
    Uses the log-softmax parameterization for the simplex constraint.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Initialize Adam moments
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    tau = tau_init
    
    t0 = time.monotonic()
    best_obj = float('inf')
    best_a = a.copy()
    
    while time.monotonic() - t0 < time_limit:
        obj, grad_a = smooth_obj_grad(a, tau)
        
        # Project gradient onto tangent space of simplex
        # The gradient on the simplex is: grad - (grad · a) * 1 (for the log-softmax parameterization)
        # Actually, for the simplex projection, we just project the gradient
        mean_grad = np.dot(grad_a, a)
        grad_proj = grad_a - mean_grad
        
        t += 1
        m = beta1 * m + (1 - beta1) * grad_proj
        v = beta2 * v + (1 - beta2) * grad_proj ** 2
        m_hat = m / (1 - beta1 ** t)

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines log-sum-exp smooth gradient optimization (Adam with log-softmax parameterization),
coarse-to-fine multi-scale approach, LP-Frank-Wolfe refinement, and coordinate descent polish.
Exploits FFT-based convolution and diverse initializations across multiple sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_obj_grad(a, tau):
    """
    Compute smoothed objective (log-sum-exp approximation of max) and gradient.
    a is assumed to be on the simplex (sum=1, a>=0).
    Returns (objective_value, gradient_wrt_a).
    """
    conv = fast_convolve(a)
    max_c = conv.max()
    shifted = (conv - max_c) / tau
    shifted = np.clip(shifted, -80, 80)
    E = np.exp(shifted)
    E_sum = E.sum()
    obj = max_c + tau * np.log(E_sum)
    w = E / E_sum

    n = len(a)
    # grad_a[j] = 2 * sum_k w[k] * a[k-j] for valid indices
    # Use FFT-based cross-correlation for efficiency when n is large
    if n > 256:
        L = 1 << (len(conv) + n - 1 - 1).bit_length()
        W = np.fft.rfft(w, L)
        A = np.fft.rfft(a, L)
        corr = np.fft.irfft(W * np.conj(A), L)
        grad_a = 2.0 * corr[:n]
    else:
        grad_a = np.zeros(n)
        for j in range(n):
            k_lo = j
            k_hi = min(j + n, len(conv))
            if k_lo < k_hi:
                grad_a[j] = 2.0 * np.dot(w[k_lo:k_hi], a[:k_hi - k_lo])

    return obj, grad_a


def adam_logsoftmax(a_init, tau_init, time_limit, lr=0.005, decay_tau=0.997,
                    verbose=False, label=""):
    """
    Minimize smoothed objective using Adam on log-softmax parameterization.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-12)
    a = a / a.sum()
    theta = np.log(a)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t = 0
    tau = tau_init
    lr_current = lr

    t0 = time.monotonic()
    best_obj = float('inf')
    best_a = a.copy()
    stuck_count = 0
    last_improve = 0

    while time.monotonic() - t0 < time_limit:
        a = np.exp(theta - theta.max())
        a = a / a.sum()

        obj, grad_a = smooth_obj_grad(a, tau)

        # Gradient w.r.t. theta: dL/dtheta_i = a_i * (g_i - sum_j g_j * a_j)
        mean_g = np.dot(grad_a, a)
        grad_theta = a * (grad_a - mean_g)

        t += 1
        m = beta1 * m + (1 - beta1) * grad_theta
        v = beta2 * v + (1 - beta2) * grad_theta ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        theta = theta - lr_current * m_hat / (np.sqrt(v_hat) + eps)

        if t % 200 == 0:
            theta = theta - theta.mean()

        a = np.exp(theta - theta.max())
        a = a / a.sum()

        conv = fast_convolve(a)
        actual = conv.max()
        if actual < best_obj - 1e-12:
            best_obj = actual
            best_a = a.copy()
            stuck_count = 0
            last_improve = t
        elif t - last_improve > 300:
            stuck_count += 1
            if stuck_count > 5:
                # Perturb and reset
                rng = np.random.default_rng(t * 13 + 7)
                theta_pert = np.log(np.maximum(best_a, 1e-12)) + rng.standard_normal(n) * 0.01
                theta = theta_pert - theta_pert.mean()
                m = np.zeros(n)
                v = np.zeros(n)
                lr_current = lr * 0.5
                stuck_count = 0
                last_improve = t

        tau = max(tau * decay_tau, 0.0003 * best_obj)
        lr_current *= 0.9999

        if verbose and t % 500 == 0:
            score = 2 * n * actual
            print(f"    {label} iter {t}: score={score:.6f}, tau={tau:.6f}")

    return best_a, best_obj


def solve_lp_active(f, M, n, top_k=40):
    """Solve LP with only top-k active constraints for speed."""
    conv = fast_convolve(f)
    # Get top k peak indices
    k = min(top_k, len(conv))
    peak_indices = np.argpartition(conv, -k)[-k:]
    peak_indices = sorted(peak_indices)

    # Build constraint matrix: (f * g)[peak_idx] <= M
    A_ub = np.zeros((len(peak_indices), n))
    b_ub = np.full(len(peak_indices), M)

    for i, pk in enumerate(peak_indices):
        pk = int(pk)
        js = max(0, pk - n + 1)
        je = min(n - 1, pk)
        idx = np.arange(js, je + 1)
        A_ub[i, idx] = f[pk - idx]

    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x
    return None


def lp_fw_refine(f0, time_limit, top_k=50, verbose=False):
    """LP-Frank-Wolfe with active constraint management."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp_active(f, M * m_exp, n, top_k=top_k)
        if g is None:
            m_exp *= 1.2
            if m_exp > 10.0:
                rng = np.random.default_rng(it * 17)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.1
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                m_exp = min(m_exp * 1.1, 5.0)
            if no_improve > 150:
                rng = np.random.default_rng(it * 31)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue
        else:
            no_improve = 0

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            m_exp = 1.0
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 40 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 4.0)

        if no_improve > 200:
            break

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit):
    """Coordinate descent: for each coordinate, try to reduce the max peak."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        conv = fast_convolve(f)
        peak_k = int(np.argmax(conv))

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old = f[j]
            if old < 1e-12:
                continue

            # Try reducing f[j] and redistributing
            best_j_val = old
            best_j_obj = best_obj

            for frac in [0.5, 0.7, 0.85, 1.2, 1.5]:
                new_val = old * frac
                if new_val < 0 or new_val > 0.9:
                    continue
                trial = f.copy()
                trial[j] = new_val
                diff = old - new_val
                if abs(diff) > 1e-14:
                    # Redistribute to indices that don't contribute to peak
                    mask = np.ones(n, dtype=bool)
                    # Indices contributing to peak: i where i + (peak_k - i) is valid
                    # i.e., i in [max(0, peak_k-n+1), min(n-1, peak_k)]
                    js = max(0, peak_k - n + 1)
                    je = min(n - 1, peak_k)
                    mask[js:je + 1] = False
                    mask[j] = False
                    if mask.sum() > 0:
                        w_redist = f[mask].copy()
                        w_redist[w_redist < 0] = 0
                        wsum = w_redist.sum()
                        if wsum > 1e-14:
                            trial[mask] += diff * w_redist / wsum
                        else:
                            trial[mask] += diff / mask.sum()
                    else:
                        trial += diff / n
                trial = np.maximum(trial, 0)
                trial = trial / trial.sum()
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old:
                trial = f.copy()
                trial[j] = best_j_val
                diff = old - best_j_val
                if abs(diff) > 1e-14:
                    js = max(0, peak_k - n + 1)
                    je = min(n - 1, peak_k)
                    mask = np.ones(n, dtype=bool)
                    mask[js:je + 1] = False
                    mask[j] = False
                    if mask.sum() > 0:
                        w_redist = f[mask].copy()
                        wsum = w_redist.sum()
                        if wsum > 1e-14:
                            trial[mask] += diff * w_redist / wsum
                        else:
                            trial[mask] += diff / mask.sum()
                    else:
                        trial += diff / n
                trial = np.maximum(trial, 0)
                trial = trial / trial.sum()
                f = trial
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and sweeps > 3:
            break

    return best_f, best_obj


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length, preserving non-negativity and sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, f)
    fi = np.maximum(fi, 0)
    s = fi.sum()
    if s > 1e-15:
        fi /= s
    return fi


def make_gaussian(n, sigma_frac=0.25):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):

            make_gaussian(t_n, 0.22),
            make_gaussian(t_n, 0.35),
            make_bell(t_n, 0.45),
            make_optimal_shape(t_n),
            make_asymmetric(t_n, 0.25),
            make_two_peak(t_n, 0.3),
        ]

        # Add random perturbations of the best
        for r in range(2):
            noise = rng.standard_normal(t_n) * 0.02
        other_inits.append(project_simplex(f_interp + rng.standard_normal(t_n) * 0.01))
        other_inits.append(project_simplex(f_interp + rng.standard_normal(t_n) * 0.03))

        # Evaluate all inits quickly
        best_init_local = f_interp
        best_init_score = 2 * t_n * fast_convolve(f_interp).max()

        for a_init in other_inits:
            a_init = np.maximum(a_init, 0)
            a_init = a_init / a_init.sum()
            obj = fast_convolve(a_init).max()
            sc_est = 2 * t_n * obj
            if np.isfinite(sc_est) and sc_est < best_init_score:
                best_init_score = sc_est
                best_init_local = a_init

        # Run smooth optimization on the best init
        t_alloc = min(rem * 0.6, 120)
        tau0 = 0.03 * fast_convolve(best_init_local).max()
        a_opt, obj = adam_logsoftmax(best_init_local, tau0, time_limit=t_alloc,
                                     lr=0.005, decay_tau=0.997, verbose=True,
                                     label=f"refine_n{t_n}")
        try_seq(a_opt, f"refine_n{t_n}")

        # Also try with different LR
        if time.monotonic() < deadline - 60:
            t_alloc2 = min(rem * 0.3, 60)
            a_opt2, obj2 = adam_logsoftmax(best_init_local, tau0, time_limit=t_alloc2,
                                           lr=0.002, decay_tau=0.998, verbose=False,
                                           label=f"refine2_n{t_n}")
            try_seq(a_opt2, f"refine2_n{t_n}")

    # === Phase 3: LP-FW refinement on best found so far ===
    print("\n=== Phase 3: LP-FW Refinement ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 60:
            n_best = len(best_seq)
            t_alloc = min(rem * 0.4, 300)
            print(f"  LP-FW on n={n_best} for {t_alloc:.0f}s")
            a_opt, obj = lp_fw_refine(best_seq, time_limit=t_alloc, top_k=60, verbose=True)
            try_seq(a_opt, "lpfw_best")

            # Try LP-FW with perturbation
            if time.monotonic() < deadline - 60:
                t_alloc2 = min(rem * 0.2, 120)
                f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.005)
                a_opt2, obj2 = lp_fw_refine(f_pert, time_limit=t_alloc2, top_k=50, verbose=False)
                try_seq(a_opt2, "lpfw_perturb")

    # === Phase 4: Coordinate descent polish ===
    print("\n=== Phase 4: Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 40:
            t_alloc = min(rem * 0.3, 180)
            print(f"  CD polish for {t_alloc:.0f}s on n={len(best_seq)}")
            a_opt, obj = coordinate_descent_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "cd_polish")

    # === Phase 5: Final smooth optimization with very small tau ===
    print("\n=== Phase 5: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            n_best = len(best_seq)
            t_alloc = min(rem * 0.7, 120)
            tau_tiny = 0.001 * fast_convolve(best_seq).max()
            print(f"  Final Adam (tau={tau_tiny:.8f}) for {t_alloc:.0f}s")
            a_opt, obj = adam_logsoftmax(best_seq, tau_tiny, time_limit=t_alloc,
                                         lr=0.002, decay_tau=0.999, verbose=True,
                                         label="final")
            try_seq(a_opt, "final_adam")

            # One more LP-FW squeeze
            if time.monotonic() < deadline - 20:
                t_alloc2 = min(rem * 0.2, 60)
                a_opt2, obj2 = lp_fw_refine(best_seq, time_limit=t_alloc2, top_k=80, verbose=False)
                try_seq(a_opt2, "final_lpfw")

    # === Phase 6: Try symmetric constraint ===
    print("\n=== Phase 6: Symmetric Optimization ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        # Optimizing symmetric sequences: a_j = a_{n-1-j}
        # Only optimize first half
        sym_n = 640
        half_n = sym_n // 2 + 1

        # Create symmetric init
        a_sym_full = interpolate_seq(best_seq, sym_n) if best_seq is not None else make_gaussian(sym_n, 0.3)
        a_sym = (a_sym_full + a_sym_full[::-1]) / 2.0
        a_sym = a_sym / a_sym.sum()

        # Optimize the first half, mirror to get full
        t_alloc = min(rem * 0.5, 120)

        def sym_adam(a_half_init, time_lim):
            """Adam on symmetric sequence (optimize first half)."""
            n_half = len(a_half_init)
            n_full = 2 * n_half - 1
            a_h = np.maximum(a_half_init, 1e-12)

            def expand(half):
                full = np.zeros(2 * len(half) - 1)
                full[:len(half)] = half
                full[len(half):] = half[:0:-1]
                return full

            f = expand(a_h)
            f = f / f.sum()

            t0s = time.monotonic()
            best_f = f.copy()
            best_o = fast_convolve(f).max()

            # Simple projected gradient on the half
            lr_s = 0.003
            step = 0
            tau_s = 0.02 * best_o

            while time.monotonic() - t0s < time_lim:
                step += 1
                f = expand(a_h)
                obj_sm, grad_full = smooth_obj_grad(f, tau_s)
                grad_h = grad_full[:n_half]
                a_h_new = a_h - lr_s * grad_h
                a_h_new = np.maximum(a_h_new, 1e-12)

                f_new = expand(a_h_new)
                f_new = f_new / f_new.sum()
                obj_new = fast_convolve(f_new).max()

                if obj_new < best_o:
                    best_o = obj_new
                    best_f = f_new.copy()
                    a_h = a_h_new.copy()
                else:
                    lr_s *= 0.95

                tau_s *= 0.998

            return best_f, best_o

        a_sym_opt, obj_sym = sym_adam(a_sym[:half_n], t_alloc)
        try_seq(a_sym_opt, "symmetric")

    # === Phase 7: Multi-length final check ===
    print("\n=== Phase 7: Multi-Length Check ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            # Check if a different length gives better results
            for check_n in [512, 768, 896, 1024, 1280]:
                if time.monotonic() > deadline - 20:
                    break
                if len(best_seq) == check_n:
                    continue
                f_c = interpolate_seq(best_seq, check_n)
                sc = try_seq(f_c, f"check_n{check_n}")
                if sc < best_score + 0.005:
                    # Worth optimizing at this length
                    t_alloc = min(rem * 0.3, 60)
                    tau0 = 0.01 * fast_convolve(f_c).max()
                    a_o, _ = adam_logsoftmax(f_c, tau0, time_limit=t_alloc, lr=0.003,
                                             decay_tau=0.998, verbose=False,
                                             label=f"multilen_{check_n}")
                    try_seq(a_o, f"opt_n{check_n}")

    # Final verification
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\n{'='*60}")
    print(f"FINAL: score={best_score:.6f}, n={len(best_seq)}")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
