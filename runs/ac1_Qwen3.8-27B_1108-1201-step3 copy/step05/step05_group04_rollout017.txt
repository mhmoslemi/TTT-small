def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def smooth_val_and_grad(a, tau):
    """
    Compute smooth approximation to max(convolve(a,a)) and its gradient w.r.t. a.
    Assumes a is already normalized (sum=1, a>=0).
    Returns (smooth_value, gradient_vector).
    """
    b = fast_convolve(a)
    shifted = b / tau
    m = shifted.max()
    w = np.exp(shifted - m)
    w_sum = w.sum()
    w = w / w_sum
    val = tau * (m + np.log(w_sum))
    grad = 2.0 * np.correlate(w, a, 'valid')
    return val, grad


def lbfgs_objective_and_grad(x, tau, n):
    """Objective and gradient for L-BFGS-B optimizer."""
    a = np.maximum(x, 1e-15)
    s = a.sum()
    if s < 1e-15:
        return 1e10, np.zeros(n)
    a = a / s
    b = fast_convolve(a)
    shifted = b / tau
    m = shifted.max()
    w = np.exp(shifted - m)
    w_sum = w.sum()
    w = w / w_sum
    val = tau * (m + np.log(w_sum))
    # Gradient w.r.t. a (normalized)
    grad_a = 2.0 * np.correlate(w, a, 'valid')
    # Account for normalization: the effective gradient w.r.t. x is grad_a/s - val/s * ones
    # Simplified: just use grad_a (approximation works well in practice)
    grad_a = grad_a - np.mean(grad_a)
    return val, grad_a


def run_lbfgs(a0, tau, time_limit, maxiter=3000):
    """Run L-BFGS-B on the smoothed objective with time limit."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    t0 = time.monotonic()

    def fun_grad(x):
        if time.monotonic() - t0 > time_limit:
            return 1e10, np.zeros(n)
        return lbfgs_objective_and_grad(x, tau, n)

    try:
        result = minimize(fun_grad, a, method='L-BFGS-B',
                          bounds=[(0, None)] * n,
                          options={'maxiter': maxiter, 'maxfun': 10000,
                                   'gtol': 1e-12, 'maxls': 30})
        a_opt = np.maximum(result.x, 0)
        s = a_opt.sum()
        if s > 1e-15:
            a_opt = a_opt / s
        return a_opt, result.fun, time.monotonic() - t0
    except Exception:
        return a, fast_convolve(a).max(), time.monotonic() - t0


def mirror_descent(a0, time_limit, lr_init=0.01, temp_init=0.01,
                   temp_decay=0.9995, verbose=False):
    """Mirror descent (multiplicative weights) with subgradient."""
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    temp = temp_init
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    t = 0

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(a)

        shifted = conv / temp
        shifted -= shifted.max()
        w = np.exp(shifted)
        w /= w.sum()
        sg = 2.0 * np.correlate(w, a, 'valid')

        update = -lr * sg
        update -= update.max()
        a_new = a * np.exp(update)
        fs = a_new.sum()
        if fs < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / fs
        a = a_new

        temp = max(temp * temp_decay, 1e-6)
        lr *= 0.9999
        lr = max(lr, lr_init * 0.05)

        conv = fast_convolve(a)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and t % 300 == 0:
                print(f"    MD iter {t}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        if no_improve > 400:
            rng = np.random.default_rng(t * 31 + 7)
            noise = 0.005 + 0.003 * min(no_improve // 200, 4)
            a = np.maximum(best_a + rng.standard_normal(n) * noise, 0)
            a /= a.sum()
            temp = max(temp_init * 0.5, 0.003)
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def coordinate_descent_fast(a0, time_limit):
    """Fast coordinate descent with incremental convolution updates."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    b = fast_convolve(a)
    best_obj = b.max()
    best_a = a.copy()
    t0 = time.monotonic()
    multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.6, 2.0]

    passes = 0
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-14:
                continue

            cur_peak = b.max()
            best_j_obj = cur_peak
            best_j_mult = 1.0

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 1e-15 or new_val > 0.9999:
                    continue

                delta = new_val - old_val
                # Incremental update: b_new = b + 2*delta*a_shifted
                # a_shifted[k] = a[k-j] for valid k
                b_trial = b.copy()
                b_trial[j:j + n] += 2.0 * delta * a
                # Also the "other direction" is already captured by the symmetric structure
                # Wait - let me reconsider. b_k = sum_i a_i * a_{k-i}
                # Changing a_j: delta_b_k = delta * (a[k-j] + a[k-j]) = 2*delta*a[k-j] when valid
                # a[k-j] is valid when 0 <= k-j < n, i.e., j <= k <= j+n-1
                # So b_trial[k] = b[k] + 2*delta*a[k-j] for k in [j, j+n-1]
                # This is what b_trial[j:j+n] += 2*delta*a does. ✓

                new_sum = s + delta
                trial_obj = b_trial.max() / (new_sum * new_sum)

                if trial_obj < best_j_obj - 1e-15:
                    best_j_obj = trial_obj
                    best_j_mult = mult

            if best_j_mult != 1.0:
                new_val = old_val * best_j_mult
                delta = new_val - old_val
                b[j:j + n] += 2.0 * delta * a
                a[j] = new_val
                s = s + delta
                # Renormalize to avoid drift
                a = a / s
                b = fast_convolve(a)
                s = 1.0

                if b.max() < best_obj - 1e-14:
                    best_obj = b.max()
                    best_a = a.copy()
                    improved = True

        if not improved and passes > 4:
            break

    best_a = best_a / best_a.sum()
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian_init(n, center=0.5, sigma=0.15):
    x = np.linspace(0, 1, n)
    f = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_symmetric_init(n, shape='bump'):
    if shape == 'bump':
        x = np.linspace(0, 1, n)
        f = np.sin(np.pi * x) ** 2
    elif shape == 'triangle':
        x = np.linspace(0, 1, n)
        f = 1 - np.abs(2 * x - 1)
    elif shape == 'plateau':
        x = np.linspace(0, 1, n)
        f = np.ones(n)
        mask = (x > 0.2) & (x < 0.8)
        f = np.where(mask, 1.0, 0.3)
    else:
        f = np.ones(n)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_smooth_random_init(n, rng, n_components=5):
    f = np.zeros(n)
    x = np.linspace(0, 1, n)
    for _ in range(n_components):
        c = rng.uniform(0.1, 0.9)
        s = rng.uniform(0.05, 0.25)
        a = rng.uniform(0.5, 2.0)
        f += a * np.exp(-0.5 * ((x - c) / s) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_spectral_init(n, rng):
    """
    Create a sequence with approximately flat power spectrum.
    Random phases in frequency domain, inverse DFT, project to nonnegative.
    """
    # Use FFT length that's a power of 2 >= n
    fft_n = 1 << (n - 1).bit_length()
    # Create random phases with conjugate symmetry for real output
    phases = rng.uniform(0, 2 * np.pi, fft_n)
    # Enforce conjugate symmetry: phases[fft_n-k] = -phases[k]
    for k in range(1, fft_n // 2 + 1):
        phases[fft_n - k] = -phases[k]
    phases[0] = 0.0
    if fft_n % 2 == 0:
        phases[fft_n // 2] = 0.0

    # Construct spectrum with flat magnitude
    spec = np.exp(1j * phases)
    # Inverse DFT
    a_full = np.fft.irfft(spec, fft_n)
    # Take first n elements
    a = a_full[:n]
    # Project to nonnegative
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        # Fallback: use absolute value
        a = np.abs(a_full[:n])
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    return a / s


def make_half_spectral_init(n, rng):
    """
    Hybrid: combine spectral init with a smooth base shape.
    """
    spect = make_spectral_init(n, rng)
    base = make_gaussian_init(n, 0.5, 0.3)
    f = 0.5 * spect + 0.5 * base
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def annealed_lbfgs(a0, time_limit, tau_start=None, tau_end=None, n_steps=5):
    """
    Run L-BFGS with decreasing temperature (annealing).
    Each step uses the result of the previous as initialization.
    """
    n = len(a0)
    a = a0.copy() / a0.sum()

    if tau_start is None:
        tau_start = 5.0 / n
    if tau_end is None:
        tau_end = 0.5 / n

    t0 = time.monotonic()
    taus = np.geomspace(tau_start, tau_end, n_steps)

    for i, tau in enumerate(taus):
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        t_alloc = rem * 0.7
        a, _, _ = run_lbfgs(a, tau, time_limit=t_alloc, maxiter=2000)

    return a


def annealed_mirror_descent(a0, time_limit, temp_start=None, temp_end=None, n_phases=4):
    """
    Mirror descent with decreasing temperature.
    """
    n = len(a0)
    a = a0.copy() / a0.sum()

    if temp_start is None:
        temp_start = 3.0 / n
    if temp_end is None:
        temp_end = 0.3 / n

    t0 = time.monotonic()
    temps = np.geomspace(temp_start, temp_end, n_phases)

    for i, temp in enumerate(temps):
        rem = time_limit - (time.monotonic() - t0)
        if rem < 10:

        t_alloc = rem * 0.6

        a, _ = mirror_descent(a, time_limit=t_alloc, lr_init=lr,
                              temp_init=temp, temp_decay=0.9999)

    return a

            print(f"  Final CD, t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent_fast(best_seq, time_limit=t_cd)
            try_seq(a_opt, "p6_cd")

            best_seq_for_next = a_opt if a_opt is not None else best_seq

        # Tiny L-BFGS refinement
        rem = deadline - time.monotonic() - 8
        if rem > 10:
            t_alloc = min(rem * 0.4, 30)
            print(f"  Final L-BFGS, t={t_alloc:.0f}s")
            a_opt, _, _ = run_lbfgs(best_seq_for_next if 'best_seq_for_next' in dir() else best_seq,
                                    tau=0.2 / n_best, time_limit=t_alloc, maxiter=3000)
            try_seq(a_opt, "p6_lbfgs")

        # Final CD after L-BFGS
        rem = deadline - time.monotonic() - 5
        if rem > 8:
            t_cd2 = min(rem * 0.7, 20)
            a_opt, _ = coordinate_descent_fast(best_seq, time_limit=t_cd2)
            try_seq(a_opt, "p6_final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
