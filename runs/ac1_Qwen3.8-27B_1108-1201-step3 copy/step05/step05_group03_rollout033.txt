import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_refinement(f, M, top_k_constraints=None, verbose=False):
    """
    Solve LP: maximize sum(g) subject to g >= 0, (f*g)_k <= M for top-K k.
    Returns the optimal g (unnormalized).

    cols = []
    vals = []

    #   f[j] * g[k-j]

    # We use the current convolution f*f as a proxy
    current_conv = fast_convolve(f)

    """
    LP-based iterative refinement.
    At each step: find g via LP, blend with current a.

    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    iter_count = 0
    while time.monotonic() - t0 < time_limit and iter_count < n_iters_max:
        iter_count += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        # Solve LP
        top_k = max(50, int(top_k_frac * (2 * n - 1)))
        g_raw = lp_refinement(a, M, top_k_constraints=top_k)
        g_sum = g_raw.sum()
        if g_sum < 1e-10:
            break
        g = g_raw / g_sum
        
        # Blend
        new_a = (1 - t_step) * a + t_step * g
        new_a = np.maximum(new_a, 0.0)
        new_a = new_a / new_a.sum()
        
        new_score = compute_score(new_a)
        if new_score < best_score - 1e-12:
            a = new_a
            best_score = new_score
            best_a = a.copy()
        else:
            # Try smaller step
            t_step = t_step * 0.5
            if t_step < 0.01:
                break
        
        if verbose and iter_count % 10 == 0:
            print(f"    LP iter {iter_count}: score={best_score:.10f}, M={M:.8f}")
    
    return best_a, best_score


def coord_descent(a, time_limit, verbose=False):
    """Coordinate descent with golden-section search per coordinate."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    pass_count = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        order = np.random.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            
            if old_val < 1e-12:
                candidates_v = [0.001, 0.005, 0.01, 0.02]
            else:
                lo = max(0.0, old_val - 0.02)
                hi = min(0.99, old_val + 0.02)
            
            def obj_v(v):
                v = max(0.0, min(1.0, v))
                new_a = a.copy()
                new_a[j] = v
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                other_sum = 1.0 - old_val
                if other_sum < 1e-12:
                    return float('inf')
                scale = (1.0 - v) / other_sum
                new_a[mask] *= scale
                conv = fast_convolve(new_a)
                return conv.max()
            
            try:
                res = minimize_scalar(obj_v, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-9, 'maxiter': 40})
                if res.success:
                    v_new = res.x
                    new_a = a.copy()
                    new_a[j] = v_new
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    other_sum = 1.0 - old_val
                    if other_sum > 1e-12:
                        scale = (1.0 - v_new) / other_sum
                        new_a[mask] *= scale
                    new_a = normalize(new_a)
                    sc = compute_score(new_a)
                    if sc < best_score - 1e-14:
                        a = new_a
                        best_score = sc
                        best_a = a.copy()
                        improved = True
            except Exception:
                pass
        
        if not improved and pass_count > 2:
            break
        if pass_count > 300:
            break
    
    return best_a, best_score

def geometric_seq(n, r):
    a = (1.0 - r) * r ** np.arange(n)
    return normalize(a)


def uniform_seq(n):
    return np.ones(n) / n


def ramp_seq(n, c):
    """Linear ramp with antisymmetric component."""
    x = np.linspace(0, 1, n)
    a = 1.0 + c * (2 * x - 1)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def cosine_seq(n):
    x = np.linspace(0, np.pi, n)
    return normalize(np.sin(x) ** 2)


def gaussian_seq(n, sigma_frac):
    x = np.arange(n) - n / 2.0
    sigma = sigma_frac * n
    return normalize(np.exp(-x ** 2 / (2 * sigma ** 2)))


def two_peak_seq(n, sep_frac, width):
    """Two Gaussian peaks."""
    x = np.arange(n)
    c1 = n * (0.5 - sep_frac)
    c2 = n * (0.5 + sep_frac)
    s = width * n
    a = np.exp(-(x - c1) ** 2 / (2 * s ** 2)) + np.exp(-(x - c2) ** 2 / (2 * s ** 2))
    return normalize(a)


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    return normalize(np.interp(new_x, old_x, f))

"""
Minimizes 2*n*max(auto_conv)/(sum)^2 using LP-based iterative refinement
(cutting-plane style), diverse parametric initializations including antisymmetric
ramp perturbations, coordinate descent with golden-section search, and pair-swap
local search. Key innovation: LP finds globally optimal update directions at each
step, escaping local minima of pure coordinate descent.
"""

import numpy as np
from scipy.optimize import minimize_scalar, linprog
from scipy.sparse import csr_matrix
from functools import partial
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def coord_obj(v, a, j, old_val):
    """Evaluate max-conv when coordinate j is set to v (with renormalization)."""
    n = len(a)
    v = max(0.0, min(1.0, v))
    new_a = a.copy()
    new_a[j] = v
    mask = np.ones(n, dtype=bool)
    mask[j] = False
    other_sum = 1.0 - old_val
    if other_sum < 1e-12:
        return float('inf')
    scale = (1.0 - v) / other_sum
    new_a[mask] *= scale
    conv = fast_convolve(new_a)
    return conv.max()


def lp_solve(f, M, top_k):
    """
    Solve LP: maximize sum(g) subject to g >= 0, (f*g)_k <= M for top-K k.
    Uses the current f*f convolution to identify which constraints matter.
    """
    n = len(f)
    current_conv = fast_convolve(f)
    k_indices = np.argsort(current_conv)[::-1][:top_k]
    
    rows = []
    cols = []
    vals = []
    b_ub = []
    
    for k in k_indices:
        ks = int(k)
        j_lo = max(0, ks - n + 1)
        j_hi = min(ks, n - 1)
        for j in range(j_lo, j_hi + 1):
            gi = ks - j
            if 0 <= gi < n:
                rows.append(len(b_ub))
                cols.append(gi)
                vals.append(f[j])
        b_ub.append(M)
    
    if len(b_ub) == 0:
        return np.ones(n) / n
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(len(b_ub), n))
    c = -np.ones(n)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=np.array(b_ub),
                      bounds=[(0, None)] * n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.ones(n) / n


def lp_iterate(a, time_limit, top_k_frac=0.5, t_init=0.4, verbose=False):
    """LP-based iterative refinement."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    t_step = t_init
    stall_count = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        
        top_k = max(50, int(top_k_frac * (2 * n - 1)))
        g_raw = lp_solve(a, M, top_k)
        g_sum = g_raw.sum()
        if g_sum < 1e-10:
            break
        g = g_raw / g_sum
        
        new_a = (1.0 - t_step) * a + t_step * g
        new_a = np.maximum(new_a, 0.0)
        ns = new_a.sum()
        if ns < 1e-15:
            break
        new_a = new_a / ns
        
        new_score = compute_score(new_a)
        if new_score < best_score - 1e-13:
            a = new_a
            best_score = new_score
            best_a = a.copy()
            stall_count = 0
            if t_step < t_init:
                t_step = min(t_init, t_step * 1.2)
        else:
            stall_count += 1
            t_step *= 0.7
            if t_step < 0.02:
                break
        
        if verbose and (best_a is not a):
            print(f"    LP: score={best_score:.10f}")
    
    return best_a, best_score


def coord_descent(a, time_limit, verbose=False):
    """Coordinate descent with golden-section search per coordinate."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    pass_count = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        order = np.random.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            
            if old_val < 1e-12:
                lo, hi = 0.0, 0.03
            else:
                lo = max(0.0, old_val - 0.03)
                hi = min(0.99, old_val + 0.03)
            
            if hi - lo < 1e-8:
                continue
            
            obj = partial(coord_obj, a=a, j=j, old_val=old_val)
            try:
                res = minimize_scalar(obj, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-9, 'maxiter': 35})
                if res.success:
                    v_new = res.x
                    new_a = a.copy()
                    new_a[j] = v_new
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    other_sum = 1.0 - old_val
                    if other_sum > 1e-12:
                        scale = (1.0 - v_new) / other_sum
                        new_a[mask] *= scale
                    new_a = normalize(new_a)
                    sc = compute_score(new_a)
                    if sc < best_score - 1e-14:
                        a = new_a
                        best_score = sc
                        best_a = a.copy()
                        improved = True
            except Exception:
                pass
        
        if not improved and pass_count > 2:
            break
        if pass_count > 400:
            break
    
    return best_a, best_score


def pair_swap(a, time_limit, n_pers=600, verbose=False):
    """Random pair transfer local search."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    rng = np.random.default_rng(777)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        for _ in range(n_pers):
            if time.monotonic() - t0 > time_limit:
                break
            i, j = rng.integers(0, n, 2)
            if i == j:
                continue
            a_j, a_i = a[j], a[i]
            if a_j < 1e-13 and a_i < 1e-13:
                continue
            
            best_delta = 0.0
            best_sc = best_score
            fracs = [0.03, 0.08, 0.15, 0.3, 0.5, 0.75, 0.9]
            
            for f in fracs:
                delta = a_j * f
                if a_j - delta < -1e-15:
                    continue
                new_a = a.copy()
                new_a[j] = a_j - delta
                new_a[i] = a_i + delta
                conv = fast_convolve(new_a)
                sc = 2.0 * n * conv.max()
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_delta = delta
            
            if best_delta != 0.0:
                a[j] -= best_delta
                a[i] += best_delta
                a = normalize(a)
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
    
    return best_a, best_score


def block_shift_search(a, time_limit, verbose=False):
    """Try shifting blocks of consecutive coordinates."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    rng = np.random.default_rng(999)
    t0 = time.monotonic()
    
    block_sizes = [5, 10, 20, 40, 80]
    
    while time.monotonic() - t0 < time_limit:
        for _ in range(50):
            if time.monotonic() - t0 > time_limit:
                break
            bs = block_sizes[rng.integers(0, len(block_sizes))]
            if bs >= n:
                continue
            start = rng.integers(0, n - bs)
            shift = rng.integers(1, max(2, n // 8))
            
            new_a = a.copy()
            if start + shift + bs <= n:
                block = new_a[start:start + bs].copy()
                new_a[start:start + bs] = new_a[start + shift:start + shift + bs]
                new_a[start + shift:start + shift + bs] = block
            else:
                continue
            
            new_a = normalize(new_a)
            sc = compute_score(new_a)
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = new_a.copy()
                a = new_a.copy()
    
    return best_a, best_score


def geometric_seq(n, r):
    a = (1.0 - r) * r ** np.arange(n)
    return normalize(a)


def uniform_seq(n):
    return np.ones(n) / n


def ramp_seq(n, c):
    x = np.linspace(0, 1, n)
    a = 1.0 + c * (2.0 * x - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def cosine_seq(n):
    x = np.linspace(0, np.pi, n)
    return normalize(np.sin(x) ** 2)


def gaussian_seq(n, sigma_frac):
    x = np.arange(n) - n / 2.0
    sigma = sigma_frac * n
    return normalize(np.exp(-x ** 2 / (2.0 * sigma ** 2)))


def two_peak_seq(n, sep_frac, width):
    x = np.arange(n)
    c1 = n * (0.5 - sep_frac)
    c2 = n * (0.5 + sep_frac)
    s = width * n
    a = np.exp(-(x - c1) ** 2 / (2.0 * s ** 2)) + np.exp(-(x - c2) ** 2 / (2.0 * s ** 2))
    return normalize(a)


def quadratic_ramp_seq(n, c):
    """Quadratic bump: 1 - c*(2x-1)^2, antisymmetric in log-space."""
    x = np.linspace(0, 1, n)
    a = 1.0 - c * (2.0 * x - 1.0) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def sine_modulated_seq(n, freq, amp):
    """Uniform modulated by a sine."""
    x = np.arange(n)
    a = 1.0 + amp * np.sin(2.0 * np.pi * freq * x / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    return normalize(np.interp(new_x, old_x, f))


def optimize_multi_stage(a, time_limit, verbose=False):
    """LP -> CD -> pair_swap -> block_shift -> CD"""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    # Stage 1: LP refinement
    t1 = min(time_limit * 0.30, 100)
    rem = time_limit - (time.monotonic() - t0)
    if rem > t1 + 20:
        a1, sc1 = lp_iterate(a, time_limit=t1, top_k_frac=0.5, t_init=0.4, verbose=verbose)
        if sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
            a = a1.copy()
        if verbose:
            print(f"      After LP: {best_score:.10f}")
    
    # Stage 2: Coordinate descent
    rem = time_limit - (time.monotonic() - t0)
    t2 = min(rem * 0.30, 120)
    if rem > t2 + 15:
        a2, sc2 = coord_descent(a, time_limit=t2, verbose=verbose)
        if sc2 < best_score:
            best_score = sc2
            best_a = a2.copy()
        if verbose:
            print(f"      After CD: {best_score:.10f}")
    
    # Stage 3: Pair swap
    rem = time_limit - (time.monotonic() - t0)
    t3 = min(rem * 0.20, 60)
    if rem > t3 + 10:
        a3, sc3 = pair_swap(best_a, time_limit=t3)
        if sc3 < best_score:
            best_score = sc3
            best_a = a3.copy()
        if verbose:
            print(f"      After PS: {best_score:.10f}")
    
    # Stage 4: Block shift
    rem = time_limit - (time.monotonic() - t0)
    t4 = min(rem * 0.15, 40)
    if rem > t4 + 10:
        a4, sc4 = block_shift_search(best_a, time_limit=t4)
        if sc4 < best_score:
            best_score = sc4
            best_a = a4.copy()
        if verbose:
            print(f"      After BS: {best_score:.10f}")
    
    # Stage 5: Final CD polish
    rem = time_limit - (time.monotonic() - t0)
    if rem > 10:
        a5, sc5 = coord_descent(best_a, time_limit=min(rem, 80))
        if sc5 < best_score:
            best_score = sc5
            best_a = a5.copy()
        if verbose:
            print(f"      After final CD: {best_score:.10f}")
    
    return best_a, best_score

            a_cd2, sc = coord_descent(best_a, time_limit=min(rem * 0.5, 40))
            try_seq(a_cd2, "cd2")
    
    # === Final verification ===
    if best_a is None:
        best_a = cosine_seq(768)
        best_score = compute_score(best_a)
    
    final_score = float(evaluate_sequence([float(x) for x in best_a]))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_a)})")
    return [float(x) for x in best_a]
