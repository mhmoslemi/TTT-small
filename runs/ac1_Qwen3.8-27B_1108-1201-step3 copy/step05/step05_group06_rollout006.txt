"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: top-k subgradient with backtracking, log-sum-exp Adam with cosine-annealed
temperature, peak-directed coordinate refinement, structured restarts, and

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def topk_subgradient(a, k=5):
    """Compute subgradient of max(conv(a,a)) averaged over top-k peaks."""
    n = len(a)
    c = fast_convolve(a)
    # Find top-k positions
    k_actual = min(k, len(c))
    top_idx = np.argpartition(c, -k_actual)[-k_actual:]
    top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
    
    # Gradient of c[k] w.r.t. a[j] = a[k-j] + a[j] if 0 <= k-j < n, else 0
    g = np.zeros(n)
    for k_pos in top_idx:
        # For each j, contribution is a[k_pos - j] if 0 <= k_pos - j < n
        # This is a shifted (and reversed) version of a
        lo = max(0, k_pos - n + 1)
        hi = min(k_pos, n - 1)
        # j ranges from lo to hi (these are valid j where k_pos - j is in [0, n-1])
        # a[k_pos - j] for j in [lo, hi] is a[hi-k_pos+n-1 ... ] ... 
        # Actually: for j in range(lo, hi+1), k_pos - j ranges from k_pos-hi to k_pos-lo
        # a[k_pos - j] = a[k_pos - lo], a[k_pos - lo - 1], ..., a[k_pos - hi]
        # So it's a reversed slice of a from index (k_pos-hi) to (k_pos-lo)
        a_part = a[k_pos - hi : k_pos - lo + 1][::-1]  # a[k_pos-j] for j=lo..hi
        a_seg = a[lo:hi+1]
        g[lo:hi+1] += a_part + a_seg
    
    # Weight by the peak values (normalize so top peak has weight 1)
    peak_vals = c[top_idx]
    max_peak = peak_vals.max()
    if max_peak > 0:
        weights = peak_vals / max_peak
        # Recompute with weights
        g = np.zeros(n)
        for idx, k_pos in enumerate(top_idx):
            w = weights[idx]
            lo = max(0, k_pos - n + 1)
            hi = min(k_pos, n - 1)
            a_part = a[k_pos - hi : k_pos - lo + 1][::-1]
            a_seg = a[lo:hi+1]
            g[lo:hi+1] += w * (a_part + a_seg)
    
    return g, c, c.max()


def smooth_grad_fft(a, T):
    """Compute gradient of logsumexp(conv(a,a)/T) via FFT efficiently."""
    n = len(a)
    c = fast_convolve(a)
    scaled = c / T
    scaled -= scaled.max()
    e = np.exp(scaled)
    w = e / e.sum()
    
    # grad_i = (2/T) * sum_j a[j] * w[i+j] for i=0..n-1
    # This is a cross-correlation
    N = 1 << (3 * n - 1).bit_length()
    A_fft = np.fft.rfft(a, N)
    W_fft = np.fft.rfft(w, N)
    corr = np.fft.irfft(A_fft * np.conj(W_fft), N)[:n]
    grad = (2.0 / T) * corr
    return grad, c


def adam_optimize(a0, time_limit, T_start=0.3, T_end=0.005, mode='cosine'):

    v_state = np.zeros(n)
    t_adam = 0
    
    alpha = 0.01
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        frac = min(elapsed / max(time_limit, 0.1), 1.0)
        
        # Cosine annealed temperature
        if mode == 'cosine':
            T = T_end + 0.5 * (T_start - T_end) * (1 + np.cos(np.pi * frac))
        else:
            T = T_start * (T_end / T_start) ** frac
        T = max(T, T_end)
        
        # Learning rate with warmup and decay
        warmup_frac = min(elapsed / max(time_limit * 0.1, 1.0), 1.0)
        lr = alpha * warmup_frac * (1.0 - 0.5 * frac)
        
        grad, c = smooth_grad_fft(a, T)
        
        t_adam += 1
        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad ** 2
        m_hat = m_state / (1 - beta1 ** t_adam)
        v_hat = v_state / (1 - beta2 ** t_adam)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)
        
        cur_obj = fast_convolve(a_new).max()
        
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            a = a_new  # Still move (momentum), just don't update best
        
        if no_improve > 2000:
            # Structured restart: try a few different perturbations
            rng_r = np.random.default_rng(it * 7919 + 13)

                    noise = rng_r.standard_normal(n) * 0.003
                elif attempt == 1:
                    # Shift perturbation
                    shift = rng_r.integers(1, max(2, n // 50))
                    noise = np.zeros(n)
                    noise[shift:] = rng_r.standard_normal(n - shift) * 0.005
                    noise[:shift] = rng_r.standard_normal(shift) * 0.005
                else:

                    noise = (1 - 2 * rng_r.random(n)) * 0.002
                f_pert = project_simplex(best_a + noise)
                obj_pert = fast_convolve(f_pert).max()
                if obj_pert < best_obj:
                    best_a = f_pert.copy()
                    best_obj = obj_pert
                    a = f_pert.copy()
                    break
            m_state = np.zeros(n)
            v_state = np.zeros(n)
            t_adam = 0
            no_improve = 0

def topk_gradient_step(a, time_limit, k=3, lr_init=0.02, decay=0.9995):
    """Direct subgradient method on top-k peaks with backtracking."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        g, c, cur_max = topk_subgradient(a, k=k)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        g_normed = g / g_norm
        
        # Backtracking line search
        best_alpha = 0.0
        best_val = cur_max
        alpha = lr
        for _ in range(15):
            a_trial = a - alpha * g_normed
            a_trial = project_simplex(a_trial)
            obj = fast_convolve(a_trial).max()
            if obj < best_val:
                best_val = obj
                best_alpha = alpha
            alpha *= 0.5
            if alpha < 1e-8:
                break
        
        if best_alpha > 1e-10:
            a = a - best_alpha * g_normed
            a = project_simplex(a)
            if best_val < best_obj - 1e-13:
                best_obj = best_val
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            lr *= 0.8
        
        # Decay learning rate
        lr = lr_init * (decay ** it)
        lr = max(lr, 1e-6)
        
        if no_improve > 500:
            # Perturb and reset
            rng_p = np.random.default_rng(it * 31 + 7)
            noise = rng_p.standard_normal(n) * 0.004
            a = project_simplex(best_a + noise)
            no_improve = 0
            lr = lr_init
    
    return best_a, best_obj


def peak_directed_cd(a0, time_limit):

def lp_direction_fast(a, M, n):
    """Solve LP for Frank-Wolfe direction using sparse structure."""
    c_vec = -np.ones(n)
    # A_ub: (a * g)[k] <= M for all k, g >= 0
    # (a*g)[k] = sum_j a[j] * g[k-j] (valid indices)
    # For efficiency, build the matrix
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j+n, j] = a
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def make_geometric(n, r):
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_symmetric_geo(n, r):
    half = (n + 1) // 2
    asc = r ** np.arange(half, dtype=np.float64)
    desc = r ** np.arange(half - 1, 0, -1, dtype=np.float64)
    a = np.concatenate([asc, desc])
    if len(a) < n:
        a = np.concatenate([a, np.ones(n - len(a)) * 0.5])
    a = a / a.sum()
    return a


def make_cosine(n, power=2.0, eps=0.005):
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** power + eps
    return a / a.sum()


def make_richardson(n):
    """Bump function (Richardson's kernel) - smooth and compact."""
    x = np.linspace(-1, 1, n)
    a = np.where(np.abs(x) < 1, (1 - x**2) ** 4, 0)
    a = np.maximum(a, 0)
    ss = a.sum()
    if ss < 1e-15:
        return np.ones(n) / n
    return a / ss


def make_battletechtng(n):
    """Sinusoidal with tapered ends."""
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) * (0.5 + 0.5 * np.cos(4*x)) + 0.01
    a = np.maximum(a, 0)
    return a / a.sum()


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    # === Phase 3: LP-Frank-Wolfe ===
    print(f"\n=== Phase 3: LP-Frank-Wolfe (best={results['best_score']:.6f}) ===")
    if results['best_seq'] is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_alloc = min(rem * 0.5, 180)
            print(f"  LP-FW (n={len(results['best_seq'])}, t={t_alloc:.0f}s)")
            a_opt, _ = lpfw_optimize(results['best_seq'], time_limit=t_alloc, verbose=True)
            _try(a_opt, "lpfw")
    
    # === Phase 4: Peak-directed CD ===
    print(f"\n=== Phase 4: Peak CD (best={results['best_score']:.6f}) ===")
    if results['best_seq'] is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 15:
            t_alloc = min(rem * 0.5, 80)
            print(f"  Peak CD (t={t_alloc:.0f}s)")
            a_opt, _ = peak_directed_cd(results['best_seq'], time_limit=t_alloc)
            _try(a_opt, "peak_cd")
    
    # === Phase 5: Fine-tuning with low temperature Adam ===
    print(f"\n=== Phase 5: Fine Adam (best={results['best_score']:.6f}) ===")
    if results['best_seq'] is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_alloc = min(rem * 0.6, 120)
            print(f"  Fine Adam (t={t_alloc:.0f}s)")
            a_opt, _ = adam_optimize(results['best_seq'], time_limit=t_alloc, T_start=0.05, T_end=0.002, mode='cosine')
            _try(a_opt, "fine_adam")
    
    # === Phase 6: Structured transforms ===
    print(f"\n=== Phase 6: Structured Transforms (best={results['best_score']:.6f}) ===")
    if results['best_seq'] is not None:
        transforms = []
        
        # Reverse
        transforms.append((results['best_seq'][::-1].copy(), "reversed"))
        
        # Blend with uniform
        for w in [0.9, 0.85, 0.8, 0.75]:
            n_b = len(results['best_seq'])
            blend = w * results['best_seq'] + (1-w) * (np.ones(n_b) / n_b)
            transforms.append((project_simplex(blend), f"blend_w{w}"))
        
        # Cyclic shifts
        n_b = len(results['best_seq'])
        for shift in [n_b//10, n_b//8, n_b//16]:
            shifted = np.roll(results['best_seq'], shift)
            transforms.append((shifted, f"shift_{shift}"))
        
        for tf, label in transforms:
            _try(tf, label)
            # Quick Adam on the transform
            rem = deadline - time.monotonic() - 60
            if rem < 15:
                break
            t_alloc = min(rem * 0.2, 30)
            a_opt, _ = adam_optimize(tf, time_limit=t_alloc, T_start=0.03, T_end=0.003)
            _try(a_opt, f"adam_{label}")
    
    # === Phase 7: Length exploration ===
    print(f"\n=== Phase 7: Length Exploration (best={results['best_score']:.6f}) ===")
    if results['best_seq'] is not None:
        n_best = len(results['best_seq'])
        target_lengths = []
        for tl in [n_best + 64, n_best + 128, n_best * 2]:
            if tl != n_best and tl <= 2048:
                target_lengths.append(tl)
        
        for tl in target_lengths[:3]:
            rem = deadline - time.monotonic() - 40
            if rem < 30:
                break
            f_interp = interpolate_seq(results['best_seq'], tl)
            _try(f_interp, f"interp_n{tl}")
            
            rem = deadline - time.monotonic() - 30
            if rem < 15:
                continue
            t_alloc = min(rem * 0.5, 60)
            a_opt, _ = adam_optimize(f_interp, time_limit=t_alloc, T_start=0.15, T_end=0.01)
            _try(a_opt, f"adam_n{tl}")
    
    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={results['best_score']:.6f}) ===")
    if results['best_seq'] is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            t_alloc = min(rem, 50)
            a_opt, _ = adam_optimize(results['best_seq'], time_limit=t_alloc,
                                     T_start=0.02, T_end=0.001, mode='cosine')
            _try(a_opt, "final_adam")
        
        rem = deadline - time.monotonic() - 10
        if rem > 8:
            t_alloc = min(rem, 30)
            a_opt, _ = topk_gradient_step(results['best_seq'], time_limit=t_alloc, k=3, lr_init=0.008)
            _try(a_opt, "final_topk")
    
    # Fallback
    if results['best_seq'] is None:
        results['best_seq'] = np.ones(768) / 768
        results['best_score'] = float(evaluate_sequence(results['best_seq'].tolist()))
    
    # Final verification
    try:
        fs = float(evaluate_sequence(results['best_seq'].tolist()))
        if np.isfinite(fs):
            results['best_score'] = fs
    except Exception:
        pass
    
    print(f"\nFINAL SCORE: {results['best_score']:.6f} (n={len(results['best_seq'])})")
    return [float(x) for x in results['best_seq']]
