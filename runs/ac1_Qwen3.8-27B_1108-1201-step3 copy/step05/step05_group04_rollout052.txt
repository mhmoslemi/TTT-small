```python
import numpy as np
import time
import cvxpy as cp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def exact_subgradient(f, conv):
    """Compute exact subgradient of max(conv(f,f)) at f.
    sg[i] = 2/|K*| * sum_{k in K*} f[k-i] (with out-of-range = 0)
    """
    n = len(f)
    max_val = conv.max()
    # Use a small tolerance to include near-maximizers
    tol = 1e-10
    K_star = np.where(conv >= max_val - tol)[0]
    
    sg = np.zeros(n)
    for k in K_star:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                sg[i] += f[j]
    sg *= 2.0 / len(K_star)
    return sg
```

```python
def lp_direction(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0"""
    n = len(f)
    g = cp.Variable(n)
    
    # Build constraint: conv(f,g)[k] <= M for all k
    # conv(f,g)[k] = sum_j f[j]*g[k-j] (with g out of range = 0)
    # = sum_{j: 0<=k-j<n} f[j]*g[k-j]
    # Let i = k-j, then j = k-i, constraint is: sum_{i: 0<=k-i<n} f[k-i]*g[i] <= M
    
    # The constraint matrix C: C[k,i] = f[k-i] if 0<=k-i<n, else 0
    # Size: (2n-1) x n
    
    constraints = []
    for k in range(2*n-1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[i] = f[j]
        constraints.append(row @ g <= M)
    
    constraints.append(g >= 0)
    
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS, verbose=False)
    
    if g.value is None:
        return np.ones(n) / n
    return np.maximum(g.value, 0)
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based direction finding (exactly solves the LP from the state-of-the-art)
2. Subgradient with Nesterov momentum and exact subgradients
3. Coordinate descent polishing with golden-section search
4. Multi-start with diverse initializations and length exploration
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build (2n-1) x n matrix C where C[k,i] = f[k-i] if 0<=k-i<n else 0."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = f
    return C


def lp_direction(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns normalized g."""
    n = len(f)
    C = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return np.ones(n) / n


def exact_subgradient(f, conv):
    """Compute exact subgradient of max(conv(f,f)) w.r.t. f.
    sg[i] = 2/|K*| * sum_{k in K*} f[k-i] (out-of-range f = 0)."""
    n = len(f)
    max_val = conv.max()
    K_star = np.where(conv >= max_val - 1e-12)[0]
    sg = np.zeros(n)
    for k in K_star:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        sg[lo:hi + 1] += f[k - lo:k - hi + 1][::-1]
    sg *= 2.0 / len(K_star)
    return sg


def soft_subgradient(f, conv, temp, top_k=None):
    """Soft subgradient using softmax weights on convolution peaks."""
    shifted = conv / temp
    shifted -= shifted.max()
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        idx = np.argmax(conv)
        w = np.zeros(len(conv))
        w[idx] = 1.0
    else:
        w /= ws
    if top_k is not None and top_k < len(conv):
        idx = np.argpartition(w, -top_k)[-top_k:]
        w_masked = np.zeros(len(conv))
        w_masked[idx] = w[idx]
        w_masked /= w_masked.sum()
        w = w_masked
    sg = np.zeros(len(f))
    for k in range(len(conv)):
        if w[k] > 1e-15:
            lo = max(0, k - len(f) + 1)
            hi = min(len(f) - 1, k)
            sg[lo:hi + 1] += w[k] * f[k - lo:k - hi + 1][::-1]
    sg *= 2.0
    return sg


def lp_phase(f0, time_limit, verbose=False):
    """LP-based direction finding iterations."""
    f = f0.copy()
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    step_init = 0.5

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        g = lp_direction(f, M)

        # Line search over step sizes
        best_t = step_init
        best_trial_obj = best_obj + 1
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            f_trial = (1.0 - t) * f + t * g
            f_trial = project_simplex(f_trial)
            obj_t = fast_convolve(f_trial).max()
            if obj_t < best_trial_obj:
                best_trial_obj = obj_t
                best_t = t

        f_new = (1.0 - best_t) * f + best_t * g
        f_new = project_simplex(f_new)
        f = f_new

        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    LP iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            if no_improve > 30:
                # Reduce step size or restart with noise
                step_init = max(0.1, step_init * 0.7)
                if no_improve > 80:
                    rng = np.random.default_rng(it * 7 + 1)
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                    no_improve = 0
                    step_init = 0.5

    return best_f, best_obj


def momentum_subgradient(f0, time_limit, L=2.0, verbose=False):
    """Subgradient method with Nesterov momentum and adaptive step size."""
    f = f0.copy()
    y = f0.copy()
    n = len(f)
    t = 0
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    cur_L = L

    while time.monotonic() - t0 < time_limit:
        t += 1
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0

        conv_y = fast_convolve(y)
        sg = exact_subgradient(y, conv_y)

        # Gradient step with projection
        f_new = project_simplex(y - sg / cur_L)

        # Nesterov extrapolation
        y_new = f_new + (t / (t_k + 1.0)) * (f_new - f)
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = np.ones(n) / n
        else:
            y_new /= s_y

        f = f_new
        y = y_new

        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
            if verbose and t % 200 == 0:
                print(f"    Mom iter {t}: obj={best_obj:.10f}, L={cur_L:.3f}")
        else:
            no_improve += 1
            if no_improve > 200:
                # Restart with noise
                rng = np.random.default_rng(t * 13 + 5)
                noise_scale = 0.002 + 0.003 * min(no_improve // 100, 4)
                f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
                y = f.copy()
                t = 0
                no_improve = 0
                cur_L = max(1.0, L * 0.5)

    return best_f, best_obj


def adam_subgradient(f0, time_limit, lr=0.003, temp_init=0.005,
                     temp_decay=0.9995, top_k=None, verbose=False):
    """Adam optimizer with soft subgradients and simplex projection."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    temp = temp_init
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)
        sg = soft_subgradient(f, conv, temp, top_k)

        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f_new)

        temp = max(temp * temp_decay, 1e-7)
        cur_lr = max(lr * (0.9999 ** t), lr * 0.1)

        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            rng = np.random.default_rng(t * 17 + 3)
            noise_scale = 0.002 + 0.002 * min(no_improve // 100, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            temp = max(temp_init * 0.7, 0.002)
            cur_lr = lr
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with local search per coordinate."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            cur_conv = fast_convolve(f)
            cur_peak = cur_conv.max()
            best_j_val = old_val
            best_j_obj = cur_peak

            # Try various multipliers
            multipliers = [0.2, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0, 3.0]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 2:
            break
        if verbose and passes % 5 == 0:
            print(f"    CD pass {passes}: obj={best_obj:.10f}")

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian_init(n, center=0.5, sigma=0.15):
    """Create a Gaussian-like initialization."""
    x = np.linspace(0, 1, n)
    f = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_symmetric_init(n, shape='bump'):
    """Create a symmetric initialization."""
    if shape == 'bump':
        x = np.linspace(0, 1, n)
        f = np.sin(np.pi * x) ** 2
    elif shape == 'triangle':
        x = np.linspace(0, 1, n)
        f = 1 - np.abs(2 * x - 1)
    else:
        f = np.ones(n)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_smooth_random_init(n, rng, n_components=5):
    """Create a smooth random initialization."""
    f = np.zeros(n)
    x = np.linspace(0, 1, n)
    for _ in range(n_components):
        c = rng.uniform(0.1, 0.9)
        s = rng.uniform(0.05, 0.25)
        a = rng.uniform(0.5, 2.0)
        f += a * np.exp(-0.5 * ((x - c) / s) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()

                try_seq(a_opt, "p7_cd")

            rem = deadline - time.monotonic() - 10
            for pi in range(3):
                rem = deadline - time.monotonic() - 8
                if rem < 10:
                    break
                noise = rng.standard_normal(n_best) * 0.001
                f_pert = project_simplex(best_seq + noise)
                a_opt, _ = momentum_subgradient(f_pert, time_limit=min(rem * 0.4, 25), L=2.0)
                try_seq(a_opt, f"p7_mom_{pi}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
