from scipy.optimize import linprog, minimize_scalar
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def lp_direction(f, M, top_k=50):
    """
    Find the LP direction: maximize sum(g) subject to g >= 0 and (f*g)_k <= M for top-k peaks.
    """
    n = len(f)
    conv = fast_convolve(f)
    # Get top-k peak indices
    k_idx = np.argpartition(conv, -top_k)[-top_k:]
    
    # Build constraint matrix: A_ub @ g <= M * ones
    # Row k: sum_i f[i] * g[k-i] <= M for valid i
    # g has n variables (g_0 to g_{n-1})
    # For constraint at position k: sum over i where 0<=i<n and 0<=k-i<n
    #   i.e., max(0,k-n+1) <= i <= min(n-1,k)
    #   g_{k-i} appears with coefficient f_i
    
    # A_ub[i, j] = coefficient of g_j in constraint i
    A_ub = np.zeros((len(k_idx), n))
    for idx, k in enumerate(k_idx):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i  # column index (g_j)
            A_ub[idx, j] = f[i]
    
    # Objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    b_ub = np.full(len(k_idx), M)
    
    # Solve LP
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        if res.success:
            g = res.x
            return g
    except Exception:
        pass
    return None

def lp_optimize(a, time_limit=300, top_k=50, verbose=False):

    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iter_count = 0

        M = conv.max()
        sum_a = a.sum()

        g = lp_direction(a, M, top_k=top_k)
        if g is None:

        # Rescale g to match normalization

        if g_sum < 1e-15:

        g = g / g_sum  # normalize to sum=1
        
        # Try different step sizes
        best_t = 0
        best_sc = best_score
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:

            new_a = normalize(new_a)
            sc = compute_score(new_a)
            if sc < best_sc:

                best_t = t

                    a = new_a
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if not improved:

    return best_a, best_score

"""
Minimizes 2*n*max(auto_conv)/(sum)^2 using LP-based direction finding,

import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_direction(f, M, top_k=50):
    """Find LP direction: max sum(g) s.t. g>=0, (f*g)_k <= M for top-k peaks."""
    n = len(f)
    conv = fast_convolve(f)
    k_idx = np.argpartition(conv, -top_k)[-top_k:]
    
    # Build constraint matrix
    # For constraint at position k: sum_i f[i] * g[k-i] <= M
    # g has n variables
    A_ub = np.zeros((len(k_idx), n))
    for idx, k in enumerate(k_idx):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            A_ub[idx, j] = f[i]
    
    c = -np.ones(n)
    b_ub = np.full(len(k_idx), M)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

    best_a = a.copy()

        g_sum = g.sum()
        if g_sum < 1e-15:

def subgradient_optimize(a, time_limit=300, top_k=30, verbose=False):
    """Averaged subgradient with adaptive step size."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    
    avg_a = np.zeros(n)
    avg_count = 0
    lr = 0.1
    lr_decay = 0.9995
    
    t0 = time.monotonic()
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        # Find top-k peaks
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        
        # Weight by how close to max
        temp = max(1e-6, M - pv.min()) * 0.3
        w = np.exp((pv - M) / temp)
        w /= w.sum()
        
        # Compute subgradient
        sg = np.zeros(n)
        for i in range(k_actual):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                sg[j] += 2.0 * w[i] * a[k - j]
        
        # Multiplicative update (mirror descent)
        a = np.maximum(a * np.exp(-lr * sg), 1e-15)
        a = a / a.sum()
        
        avg_a += a
        avg_count += 1
        
        sc = compute_score(a)
        sc_avg = compute_score(avg_a / avg_count)
        
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
        if sc_avg < best_score - 1e-14:
            best_score = sc_avg

def coord_descent(a, time_limit=300, verbose=False):
    """Coordinate descent with golden-section search."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    pass_count = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        order = np.random.permutation(n) if pass_count > 1 else np.arange(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            
            def obj_v(v):
                v = max(0.0, v)
                new_a = a.copy()
                new_a[j] = v
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                other_sum = 1.0 - old_val
                if other_sum < 1e-12:
                    return float('inf')
                scale = (1.0 - v) / other_sum
                new_a[mask] *= scale
                conv = fast_convolve(new_a)
                return conv.max()
            
            if old_val < 1e-10:
                lo, hi = 0.0, 0.01
            else:
                lo = max(0.0, old_val * 0.2)
                hi = min(0.99, old_val * 5.0)
            
            try:
                res = minimize_scalar(obj_v, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-8, 'maxiter': 20})
                if res.success:
                    v_new = res.x
                    new_a = a.copy()
                    new_a[j] = v_new
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    other_sum = 1.0 - old_val
                    if other_sum > 1e-12:
                        scale = (1.0 - v_new) / other_sum
                        new_a[mask] *= scale
                    new_a = normalize(new_a)
                    sc = compute_score(new_a)
                    if sc < best_score - 1e-14:
                        a = new_a
                        best_score = sc
                        best_a = a.copy()
                        improved = True
            except Exception:
                pass
        
        if pass_count > 5 and not improved:
            break
        if pass_count > 100:
            break
    
    return best_a, best_score


def pair_swap(a, time_limit=100, n_pers=200, verbose=False):
    """Random pair swap local search."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    rng = np.random.default_rng(42)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        for _ in range(n_pers):
            if time.monotonic() - t0 > time_limit:
                break
            i, j = rng.integers(0, n, 2)
            if i == j:
                continue
            a_j, a_i = a[j], a[i]
            if a_j < 1e-12 and a_i < 1e-12:
                continue
            
            best_delta = 0.0
            best_sc = best_score
            for f in [0.1, 0.2, 0.3, 0.5, 0.7]:
                delta = a_j * f
                new_a = a.copy()
                new_a[j] = a_j - delta
                new_a[i] = a_i + delta
                if new_a[j] < 0:
                    continue
                conv = fast_convolve(new_a)
                sc = 2.0 * n * conv.max() / (new_a.sum()**2)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_delta = delta
            
            if best_delta != 0.0:
                a[j] -= best_delta
                a[i] += best_delta
                a = normalize(a)
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
    
    return best_a, best_score


def geometric_sequence(n, r):
    a = (1.0 - r) * r ** np.arange(n)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    return normalize(np.interp(new_x, old_x, f))

Let me try a different angle: **what if we parameterize the sequence as a sum of shifted top-hat functions?**

a = sum_{m=0}^{M-1} c_m * rect_m

where rect_m is a top-hat (rectangle) of some width centered at position m. If we choose the positions and weights carefully, the auto-convolution might be flatter.

Actually, this is related to the "Wigner distribution" and "time-frequency" analysis. The auto-convolution is related to the "ambiguity function" of the signal.
