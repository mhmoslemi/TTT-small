Minimizes 2*n*max(auto_conv)/(sum)^2 using LP-based predictor-corrector method,
spectral flattening (Gerchberg-Saxton), and coordinate descent polishing.
Key innovation: Linear programming to find optimal descent directions that
respect the convolution peak constraint, combined with spectral domain
flattening for global exploration.
"""

import numpy as np
import time
from scipy.optimize import minimize_scalar

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_B_matrix(n, f):
    """Build the B matrix where B[k,i] = f[i] if 0 <= k-i < n else 0."""
    B = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        B[k, i_start:i_end+1] = f[i_start:i_end+1]
    return B


def lp_direction(f, M, use_lv=True):
    """Solve LP: max sum(g) s.t. g >= 0, B @ g <= M."""
    if not HAS_CVXPY:
        return None
    
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    
    B = build_B_matrix(n, f)
    constraints = [B @ g <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    try:
        prob.solve(solver=cp.CLARABEL)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                return None
    
    if g.value is not None:
        return np.maximum(g.value, 0)

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    rssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (rssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (rssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_direction(f, M, solver=None):
    """Solve LP: max sum(g) s.t. g >= 0, cross_conv(f,g) <= M elementwise."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    
    # Build constraint: for each k, sum_i f[i]*g[k-i] <= M
    # This is (B @ g) <= M where B[k,i] = f[i] if valid
    # Use efficient construction
    terms = []
    for k in range(2*n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        terms.append(cp.sum(f[i_lo:i_hi+1] * g[i_lo:i_hi+1]) <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(g)), terms)
    
    solvers_to_try = [solver] if solver else [cp.CLARABEL, cp.ECOS, cp.SCS]
    for sol in solvers_to_try:
        if sol is None:
            continue
        try:
            prob.solve(solver=sol)
            if g.value is not None:
                return np.maximum(g.value, 0)
        except Exception:
            continue
    
    return None

def coord_descent(a, time_limit, verbose=False):
    """Coordinate descent with line search."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    rng = np.random.default_rng(999)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            
            # Try modifying coordinate j
            # new_a[j] = v, others scaled to maintain sum=1
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            other_sum = 1.0 - old_val
            
            if other_sum < 1e-12:
                continue
            
            def obj_v(v):
                if v < 0 or v > 0.999:
                    return float('inf')
                new_a = a.copy()
                new_a[j] = v
                new_a[mask] *= (1.0 - v) / other_sum
                conv = fast_convolve(new_a)
                return conv.max()
            
            # Golden section / bounded minimize
            lo = max(0.0, old_val * 0.1)
            hi = min(0.5, old_val * 5.0 + 0.1)
            
            try:
                res = minimize_scalar(obj_v, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-10, 'maxiter': 25})
                if res.success:
                    v_new = res.x
                    new_a = a.copy()
                    new_a[j] = v_new
                    new_a[mask] *= (1.0 - v_new) / other_sum
                    new_a = normalize(new_a)
                    sc = compute_score(new_a)
                    if sc < best_score - 1e-13:
                        a = new_a
                        best_score = sc
                        best_a = a.copy()
                        improved = True
            except Exception:
                pass
        
        if not improved:
            break
    
    return best_a, best_score

    g = cp.Variable(n, nonneg=True)

    # B[k, i] = f[i] if max(0, k-n+1) <= i <= min(n-1, k) else 0

        i_hi = min(n-1, k)
        B[k, i_lo:i_hi+1] = f[i_lo:i_hi+1]
    
    constraints = [B @ g <= M, g >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    try:
        prob.solve(solver=cp.CLARABEL)
    except:
        try:
            prob.solve(solver=cp.ECOS)
        except:
            prob.solve(solver=cp.SCS)
    
    if g.value is not None:
        return np.maximum(g.value, 0)
    return None

"""
Minimizes 2*n*max(auto_conv)/(sum)^2 using LP-based predictor-corrector method
with HiGHS solver, spectral flattening (Gerchberg-Saxton), and coordinate descent.
Key innovation: Linear programming to find optimal descent directions respecting
the convolution peak constraint, combined with spectral domain operations for
global exploration and coordinate descent for local polishing.
"""

import numpy as np
import time
from scipy.optimize import minimize_scalar, linprog

try:
    import cvxpy as cp

except ImportError:
    HAS_CVXPY = False


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_B_matrix(n, f):
    """Build B where B[k,i] = f[i] if 0<=k-i<n else 0. Shape (2n-1, n)."""
    B = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        B[k, i_lo:i_hi + 1] = f[i_lo:i_hi + 1]
    return B


def lp_direction_scipy(f, M):
    """Solve LP via scipy: max sum(g) s.t. g>=0, B@g <= M."""
    n = len(f)
    B = build_B_matrix(n, f)
    c = -np.ones(n)
    res = linprog(c, A_ub=B, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        return np.maximum(res.x, 0)
    return None


def lp_direction_cvxpy(f, M):
    """Solve LP via cvxpy: max sum(g) s.t. g>=0, B@g <= M."""
    if not HAS_CVXPY:
        return None
    n = len(f)
    B = build_B_matrix(n, f)
    g = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [B @ g <= M])
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None:
                return np.maximum(g.value, 0)
        except Exception:
            continue
    return None


def lp_direction(f, M):
    """Try scipy first, fall back to cvxpy."""
    g = lp_direction_scipy(f, M)
    if g is not None:
        return g
    return lp_direction_cvxpy(f, M)

def spectral_flatten(a_init, time_limit, alpha=0.08, verbose=False):
    """Gradchberg-Saxton style spectral flattening."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        log_mag = np.log(mag + 1e-30)
        target = np.mean(log_mag)

        new_log = (1.0 - alpha) * log_mag + alpha * target
        new_mag = np.exp(new_log)

        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new / s

        conv = fast_convolve(a)
        sc = 2.0 * n * conv.max() / (s * s)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and it % 500 == 0:
            print(f"    Spectral iter {it}: score={best_score:.10f}")

    return best_a, best_score


def coord_descent(a_init, time_limit, verbose=False):
    """Coordinate descent with bounded minimization per coordinate."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    rng = np.random.default_rng(777)
    t0 = time.monotonic()
    pass_count = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            other_sum = 1.0 - old_val
            if other_sum < 1e-12:
                continue

            lo = max(0.0, old_val * 0.05)
            hi = min(0.5, old_val * 10.0 + 0.05)
            if hi - lo < 1e-8:
                lo = max(0.0, old_val - 0.1)
                hi = min(0.5, old_val + 0.1)

            def obj_j(v):
                if v < -1e-10 or v > 0.999:
                    return float('inf')
                v = max(0.0, v)
                new_a = a.copy()
                new_a[j] = v
                new_a[mask] *= (1.0 - v) / other_sum
                conv = fast_convolve(new_a)
                return conv.max()

            try:
                res = minimize_scalar(obj_j, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-10, 'maxiter': 20})
                if res.success:
                    v_new = res.x
                    new_a = a.copy()
                    new_a[j] = v_new
                    new_a[mask] *= (1.0 - v_new) / other_sum
                    new_a = normalize(new_a)
                    sc = compute_score(new_a)
                    if sc < best_score - 1e-13:
                        a = new_a
                        best_score = sc
                        best_a = a.copy()
                        improved = True
            except Exception:
                pass

        if not improved and pass_count > 1:
            break

        if verbose and pass_count % 3 == 0:
            print(f"    CD pass {pass_count}: score={best_score:.10f}")

    return best_a, best_score

        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        temp = max(1e-8, pv.max() - pv.min()) * 0.5
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        sg = np.zeros(n)
        for i in range(k_actual):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        a = np.maximum(a * np.exp(-cur_lr * sg), 1e-15)
        a = a / a.sum()

        avg_a += a
        avg_count += 1

        sc = compute_score(a)
        avg_candidate = avg_a / avg_count
        sc_avg = compute_score(avg_candidate)

        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
        if sc_avg < best_score - 1e-14:
            best_score = sc_avg
            best_a = avg_candidate.copy()

        cur_lr *= decay

    return best_a, best_score


def geometric_seq(n, r):
    """Geometric sequence normalized."""
    a = r ** np.arange(n)
    return normalize(a)


def exp_ramp_seq(n, beta):
    """Exponential decay sequence."""
    x = np.arange(n) / max(1, n - 1)
    a = np.exp(-beta * x)
    return normalize(a)


def cosine_seq(n):
    """Cosine-shaped sequence."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    return normalize(a)


def gaussian_seq(n, sigma_frac):
    """Gaussian-shaped sequence."""
    x = np.arange(n) - n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-x ** 2 / (2.0 * sigma ** 2))
    return normalize(a)


def random_smooth_seq(n, seed, smooth_k):
    """Random sequence smoothed with box filter."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    kernel = np.ones(smooth_k) / smooth_k
    a = np.convolve(a, kernel, mode='same')
    return normalize(a)


def evaluate_candidate(a, label, best_score_ref, best_a_ref):
    """Evaluate and track best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf'), best_score_ref, best_a_ref
    score = compute_score(a)
    if score < best_score_ref:
        best_score_ref = score
        best_a_ref = a.copy()
        print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
    return score, best_score_ref, best_a_ref


def optimize_pipeline(a_init, time_limit):
    """Run multi-stage optimization: LP -> spectral -> LP -> CD."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    # Stage 1: LP predictor-corrector
    rem = time_limit - (time.monotonic() - t0)
    t1 = min(rem * 0.5, rem - 60)
    if t1 > 30:
        a_lp, sc_lp = lp_predictor_corrector(a, time_limit=t1)
        if sc_lp < best_score:
            best_score = sc_lp
            best_a = a_lp.copy()
            a = a_lp.copy()

    # Stage 2: Spectral flattening
    rem = time_limit - (time.monotonic() - t0)
    t2 = min(rem * 0.25, 120)
    if t2 > 15:
        a_spec, sc_spec = spectral_flatten(a, time_limit=t2, alpha=0.1)
        if sc_spec < best_score:
            best_score = sc_spec
            best_a = a_spec.copy()
            a = a_spec.copy()

    # Stage 3: More LP
    rem = time_limit - (time.monotonic() - t0)
    t3 = min(rem * 0.3, rem - 30)
    if t3 > 15:
        a_lp2, sc_lp2 = lp_predictor_corrector(a, time_limit=t3)
        if sc_lp2 < best_score:
            best_score = sc_lp2
            best_a = a_lp2.copy()
            a = a_lp2.copy()

    # Stage 4: Coordinate descent
    rem = time_limit - (time.monotonic() - t0)
    if rem > 10:
        a_cd, sc_cd = coord_descent(a, time_limit=rem)
        if sc_cd < best_score:
            best_score = sc_cd
            best_a = a_cd.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining LP predictor-corrector, spectral flattening,
    subgradient, and coordinate descent with diverse structured initializations.
    Targets score < 1.503 by exploiting the linear structure of the
    cross-convolution constraint in the LP step.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_score = float('inf')
    best_a = None

    # === Generate diverse starting points ===
    print("=== Generating starting points ===")
    starts = []

    # Try parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            starts.append((parent, "parent"))
            p_score = compute_score(parent)
            print(f"  Parent: n={len(parent)}, score={p_score:.10f}")
    except Exception as e:
        print(f"  No parent available: {e}")

    # Geometric
    for n in [640, 768, 896, 1024]:
        for r in [0.996, 0.998, 0.999]:
            starts.append((geometric_seq(n, r), f"geom_{r}_{n}"))

    # Exponential ramp
    for n in [768, 1024]:
        for beta in [1.5, 3.0, 5.0, 8.0]:
            starts.append((exp_ramp_seq(n, beta), f"exp_{beta}_{n}"))

    # Cosine
    for n in [768, 1024]:
        starts.append((cosine_seq(n), f"cos_{n}"))

    # Gaussian
    for n in [768, 1024]:
        for sf in [0.25, 0.35, 0.5]:
            starts.append((gaussian_seq(n, sf), f"gauss_{sf}_{n}"))

    # Random smooth
    for n in [768, 1024]:
        for sk in [10, 30]:
            starts.append((random_smooth_seq(n, seed + n + sk, sk), f"rand_{sk}_{n}"))

    # Evaluate all
    start_scores = []
    for a, label in starts:
        sc = compute_score(a)
        start_scores.append((sc, a, label))
    start_scores.sort(key=lambda x: x[0])

    print(f"  Top 8 starts:")
    for i, (sc, a, label) in enumerate(start_scores[:8]):
        print(f"    {sc:.10f} [{label}] (n={len(a)})")
        _, best_score, best_a = evaluate_candidate(a, label, best_score, best_a)

    # === Phase 1: LP optimization on top diverse starts ===
    print(f"\n=== Phase 1: LP on diverse starts ===")
    selected = []
    seen = set()
    for sc, a, label in start_scores:
        # Extract type prefix
        parts = label.split('_')
        ptype = parts[0]
        if ptype not in seen and len(selected) < 6:
            selected.append((a, label, sc))
            seen.add(ptype)

    # Ensure parent if good
    if "parent" not in seen:
        for sc, a, label in start_scores:
            if label == "parent" and sc < best_score * 1.1:
                selected.append((a, label, sc))
                seen.add("parent")
                break

    for a_init, label, init_sc in selected:
        rem = deadline - time.monotonic() - 400
        if rem < 60:
            break
        t_alloc = min(rem * 0.6, 350)
        print(f"  LP-PC [{label}] (init={init_sc:.8f}), t={t_alloc:.0f}s")
        a_opt, sc_opt = lp_predictor_corrector(a_init, time_limit=t_alloc, verbose=True)
        _, best_score, best_a = evaluate_candidate(a_opt, label + "_lp", best_score, best_a)

    # === Phase 2: Full pipeline on best ===
    print(f"\n=== Phase 2: Full pipeline (best={best_score:.10f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 60:
            t_alloc = min(rem * 0.7, 400)
            print(f"  Pipeline on best, t={t_alloc:.0f}s")
            a_opt, sc_opt = optimize_pipeline(best_a, time_limit=t_alloc)
            _, best_score, best_a = evaluate_candidate(a_opt, "pipeline", best_score, best_a)

    # === Phase 3: Subgradient exploration from best ===
    print(f"\n=== Phase 3: Subgradient (best={best_score:.10f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(rem * 0.5, 200)
            a_sg, sc_sg = subgradient_descend(best_a, time_limit=t_alloc, top_k=50, lr0=0.003)
            _, best_score, best_a = evaluate_candidate(a_sg, "sg", best_score, best_a)
            # Then LP polish on the subgradient result
            rem = deadline - time.monotonic() - 150
            if rem > 20:
                a_lp, sc_lp = lp_predictor_corrector(a_sg, time_limit=min(rem, 150))
                _, best_score, best_a = evaluate_candidate(a_lp, "sg_lp", best_score, best_a)

    # === Phase 4: Spectral from best ===
    print(f"\n=== Phase 4: Spectral (best={best_score:.10f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_alloc = min(rem * 0.5, 150)
            a_spec, sc_spec = spectral_flatten(best_a, time_limit=t_alloc, alpha=0.15)
            _, best_score, best_a = evaluate_candidate(a_spec, "spec", best_score, best_a)
            rem = deadline - time.monotonic() - 100
            if rem > 15:
                a_lp, sc_lp = lp_predictor_corrector(a_spec, time_limit=min(rem, 100))
                _, best_score, best_a = evaluate_candidate(a_lp, "spec_lp", best_score, best_a)

    # === Phase 5: Dimension exploration ===
    print(f"\n=== Phase 5: Dimension exploration (best={best_score:.10f}) ===")
    if best_a is not None:
        best_n = len(best_a)
        for target_n in [640, 896, 1024, 1280]:
            if target_n == best_n:
                continue
            rem = deadline - time.monotonic() - 100
            if rem < 30:
                break
            t_alloc = min(rem * 0.4, 120)
            x_old = np.linspace(0, 1, best_n)
            x_new = np.linspace(0, 1, target_n)
            a_interp = normalize(np.interp(x_new, x_old, best_a))
            a_opt, sc = lp_predictor_corrector(a_interp, time_limit=t_alloc)
            _, best_score, best_a = evaluate_candidate(a_opt, f"dim_{target_n}", best_score, best_a)

    # === Phase 6: Final polish with LP + CD ===
    print(f"\n=== Phase 6: Final polish (best={best_score:.10f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            t_lp = min(rem * 0.6, 200)
            a_lp, sc_lp = lp_predictor_corrector(best_a, time_limit=t_lp)
            _, best_score, best_a = evaluate_candidate(a_lp, "final_lp", best_score, best_a)

        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_cd, sc_cd = coord_descent(best_a, time_limit=min(rem, 150))
            _, best_score, best_a = evaluate_candidate(a_cd, "final_cd", best_score, best_a)

    # === Phase 7: Tiny perturbations + re-optimize ===
    print(f"\n=== Phase 7: Perturbations (best={best_score:.10f}) ===")
    if best_a is not None:
        for trial in range(4):
            rem = deadline - time.monotonic() - 15
            if rem < 10:
                break
            n_best = len(best_a)
            noise = rng.standard_normal(n_best) * 0.003
            a_pert = project_simplex(best_a + noise)
            t_alloc = min(rem * 0.5, 60)
            a_opt, sc = lp_predictor_corrector(a_pert, time_limit=t_alloc)
            _, best_score, best_a = evaluate_candidate(a_opt, f"pert_{trial}", best_score, best_a)

    # === Final verification ===
    if best_a is None:
        best_a = cosine_seq(768)
        best_score = compute_score(best_a)

    # Scale back to original scale (the evaluate function handles any positive scale)
    # But we want to keep values reasonable
    final_scale = 1.0
    final_a = best_a * final_scale

    final_score = float(evaluate_sequence([float(x) for x in final_a]))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_a)})")
    return [float(x) for x in final_a]
