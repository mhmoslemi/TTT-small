"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP-based Frank-Wolfe, smoothed subgradient descent with adaptive
temperature, targeted peak reduction, coordinate descent, and multi-length
exploration. Key innovations: efficient vectorized subgradients, structured
restarts, and a peak-targeting local search that directly reduces elements
contributing to the highest convolution lags.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_subgradient(f, conv, top_k=30, temp=0.005):
    """
    Compute smoothed subgradient of max(conv(a,a)) w.r.t. a.
    Uses softmax weighting over the top-k lags.
    db_k/da[i] = 2*a[k-i] if 0<=k-i<n, else 0.
    """
    n = len(f)
    k_actual = min(top_k, len(conv))
    # Get top-k indices
    if k_actual == len(conv):
        peak_indices = np.arange(len(conv))
    else:
        peak_indices = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peak_indices]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peak_indices))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for idx in range(len(peak_indices)):
        if w[idx] < 1e-15:
            continue
        k = int(peak_indices[idx])
        wi = 2.0 * w[idx]
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        # sg[i] += wi * f[k-i] for i in [i_start, i_end]
        # f[k-i] for i=i_start..i_end = f reversed in that range
        segment = f[i_start:i_end + 1][::-1]
        sg[i_start:i_end + 1] += wi * segment

    return sg


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive M relaxation and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) * (1.0 / len(f0))
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Solve LP: max sum(g) s.t. (f*g)[k] <= M*m_relax, g >= 0
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M * m_relax)
        bounds = [(0, None)] * n

        options = {'time_limit': min(30, max(5, (time_limit - (time.monotonic() - t0)) * 0.1)),
                   'presolve': True}
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options=options)

        if not res.success or res.x is None:
            m_relax *= 1.1
            if m_relax > 10.0:
                m_relax = 1.0
                no_improve += 5
                if no_improve > 60:
                    break
            continue

        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 50:
                break
            m_relax = min(m_relax * 1.01, 3.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M + 1e-15

        alphas = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine bisection
        if best_alpha > 0:
            if best_alpha >= 1.0:
                lo, hi = 0.6, 1.0
            elif best_alpha <= 0.05:
                lo, hi = 0.0, 0.15
            else:
                lo = max(0.0, best_alpha - 0.2)
                hi = min(1.0, best_alpha + 0.2)

            for _ in range(20):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-13:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            st = f_new.sum()
            if st < 1e-15:
                break
            f_new = f_new / st
            conv = fast_convolve(f_new)
            M = conv.max()
            m_relax = 1.0
            f = f_new

            if M < best_obj - 1e-16:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.03, 5.0)

        if no_improve > 60:
            # Restart with noise
            rng = np.random.default_rng(it * 104729 + 7)
            scale = 0.001 + 0.003 * min(no_improve // 60, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0
            no_improve = 0

    return best_f, best_obj


def sg_descent(f0, time_limit, lr_init=0.005, mom=0.92, top_k=30,
               temp_init=0.01, temp_end=0.0005, verbose=False):
    """Smoothed subgradient descent with adaptive temperature and momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) * (1.0 / len(f0))
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    total_iters_estimated = max(1, int(time_limit / max(0.001, time_limit / 5000)))

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = elapsed / time_limit

        # Adaptive temperature
        temp = temp_init * (temp_end / temp_init) ** progress
        # Adaptive learning rate
        lr = lr_init * (1.0 - 0.5 * progress)

        # Compute subgradient
        sg = compute_smoothed_subgradient(f, conv, top_k=top_k, temp=temp)

        # Momentum update
        v = mom * v - lr * sg
        f = project_simplex(f + v)

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-16:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    SG iter {it}: obj={best_obj:.10f}, temp={temp:.6f}")
        else:
            no_improve += 1

        if no_improve > 300:
            # Restart
            rng = np.random.default_rng(it * 31 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            conv = fast_convolve(f)
            no_improve = 0

    return best_f, best_obj


def peak_reduction(f0, time_limit, verbose=False):
    """
    Targeted peak reduction: identify lags where conv is near max,
    and reduce elements that contribute most to those peaks.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) * (1.0 / len(f0))
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Find top lags
        top_k = min(10, len(conv))
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        # Only consider lags that are > 0.9 * M
        peak_indices = peak_indices[conv[peak_indices] > 0.9 * M]

        if len(peak_indices) == 0:
            break

        # For each peak lag, find the elements contributing most
        # and try to reduce them
        improved = False
        for k in peak_indices:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            # Contribution of element i to b_k is 2*f[k-i]
            # We want to reduce f[i] where f[k-i] is large
            # i.e., reduce f[i] where the partner f[k-i] is large

            contributions = f[k - np.arange(i_start, i_end + 1)] * f[i_start:i_end + 1]
            # Try reducing the top contributors
            top_contrib = np.argsort(contributions)[::-1][:5]

            for tc in top_contrib:
                i = i_start + tc
                if contributions[tc] < 1e-10:
                    continue
                # Try reducing f[i] by a small amount
                for frac in [0.7, 0.5, 0.3, 0.1]:
                    f_trial = f.copy()
                    old = f_trial[i]
                    f_trial[i] = old * frac
                    # Redistribute the mass to elements that don't contribute to this peak
                    remaining = old * (1 - frac)
                    # Find elements that DON'T contribute to any peak lag
                    safe_mask = np.ones(n, dtype=bool)
                    for k2 in peak_indices:
                        s2 = max(0, k2 - n + 1)
                        e2 = min(n - 1, k2)
                        safe_mask[s2:e2 + 1] = False
                    # Also mark the element itself
                    safe_mask[i] = False

                    safe_count = safe_mask.sum()
                    if safe_count < 1:
                        continue

                    # Distribute to safe elements proportional to current values
                    safe_weights = f[safe_mask]
                    sw_sum = safe_weights.sum()
                    if sw_sum < 1e-15:
                        f_trial[safe_mask] += remaining / safe_count
                    else:
                        f_trial[safe_mask] += remaining * safe_weights / sw_sum

                    f_trial = np.maximum(f_trial, 0)
                    st = f_trial.sum()
                    if st < 1e-15:
                        continue
                    f_trial = f_trial / st
                    new_obj = fast_convolve(f_trial).max()

                    if new_obj < M - 1e-14:
                        f = f_trial
                        M = new_obj
                        improved = True
                        break

        if improved:
            if M < best_obj - 1e-16:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 20 == 0:
                    print(f"    PeakRed iter {it}: obj={best_obj:.10f}")
            no_improve_count = 0
        else:
            # Try a random perturbation
            rng = np.random.default_rng(it * 997 + 3)
            f = project_simplex(f + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(f)

        if time.monotonic() - t0 > time_limit:
            break

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with multiplicative updates on simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) * (1.0 / len(f0))
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        cur_obj = best_obj

        # Random order for diversity
        rng = np.random.default_rng(passes * 7919)
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-12:
                continue

            old_val = f[j]
            multipliers = [0.5, 0.7, 0.85, 1.15, 1.3, 1.6]
            best_j_obj = cur_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial[0] += 1.0 - ss  # fix rounding
                f_trial = np.maximum(f_trial, 0)
                f_trial /= f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-12:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f /= f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 6:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    # === Phase 8: Micro perturbations ===
    print(f"\n=== Phase 8: Micro-perturb (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    for trial in range(15):
        rem = deadline - time.monotonic()
        if rem < 3:
            break
        scale = 0.0002 + trial * 0.00015
        noise = rng.standard_normal(n_best) * scale
        f_pert = project_simplex(best_seq + noise)
        sc = evaluate_sequence(f_pert.tolist())
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = f_pert.copy()
            print(f"  NEW BEST [micro{trial}]: {sc:.6f}")
            # Quick polish
            rem2 = deadline - time.monotonic()
            if rem2 > 8:
                a_opt, _ = sg_descent(f_pert, time_limit=min(rem2 * 0.4, 15),
                                      lr_init=0.0005, mom=0.96,
                                      top_k=20, temp_init=0.002, temp_end=0.0003)
                try_update(a_opt, f"micro{trial}_sg")

    # === Final LP-FW squeeze ===
    rem = deadline - time.monotonic()
    if rem > 15:
        print(f"\n=== Final LP-FW squeeze (t={rem:.0f}s) ===")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem * 0.85, 90))
        try_update(a_opt, "Final LP-FW")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
