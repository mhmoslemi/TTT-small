def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def convex_eval(f, j, x):
    """Evaluate max conv when f[j]=x and others scaled to maintain sum=1."""
    n = len(f)
    old_val = f[j]
    denom = 1.0 - old_val
    if denom < 1e-10:
        return 1e10
    if x < 0 or x >= 1.0:
        return 1e10
    scale = (1.0 - x) / denom
    f_trial = f.copy()
    f_trial[j] = x
    mask = np.ones(n, dtype=bool)
    mask[j] = False
    f_trial[mask] *= scale
    return fast_convolve(f_trial).max()


def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 60})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def smooth_grad(f, T):
    """Compute smooth gradient of logsumexp(conv(f,f)/T) via FFT cross-correlation."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / T
    scaled -= scaled.max()
    e = np.exp(scaled)
    w = e / e.sum()
    
    N = 1 << (3 * n - 1).bit_length()
    A_fft = np.fft.rfft(f, N)
    W_fft = np.fft.rfft(w, N)
    corr = np.fft.irfft(A_fft * np.conj(W_fft), N)[:n]
    grad = (2.0 / T) * corr
    return grad, conv, w

    beta1 = 0.9

def single_cd(f0, time_limit, verbose=False):
    """Single coordinate descent with golden section search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

    t0 = time.monotonic()
    passes = 0
    ORDER = 0.6180339887
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        cur_conv = fast_convolve(f)
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-10:
                continue
            
            lo = 0.0
            hi = 0.999
            c = hi - ORDER * (hi - lo)
            d = lo + ORDER * (hi - lo)
            
            fc = convex_eval(f, j, c)
            fd = convex_eval(f, j, d)
            
            for _ in range(12):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - ORDER * (hi - lo)
                    fc = convex_eval(f, j, c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + ORDER * (hi - lo)
                    fd = convex_eval(f, j, d)
            
            opt_val = (lo + hi) / 2.0
            opt_obj = convex_eval(f, j, opt_val)
            
            if opt_obj < best_obj - 1e-13:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - opt_val) / denom
                    f[j] = opt_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                best_obj = opt_obj
                best_f = f.copy()
                improved = True
        
        if not improved and passes > 2:
            break

def make_geometric(n, r):
    """Geometric sequence normalized to sum=1."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_modulated(n, r, freq, amp):
    """Geometric sequence modulated by cosine."""
    base = r ** np.arange(n, dtype=np.float64)
    mod = 1.0 + amp * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
    a = base * mod
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_multi_freq(n, freqs, amps, r=0.99):
    """Multi-frequency modulated geometric sequence."""
    base = r ** np.arange(n, dtype=np.float64)
    mod = np.ones(n)
    for freq, amp in zip(freqs, amps):
        mod += amp * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
    a = base * np.maximum(mod, 0.01)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_cosine(n, power=2.0):
    """Cosine window raised to a power."""
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** power + 0.005
    return a / a.sum()


def interpolate_seq(f, new_n):
    """Linear interpolation to new length, normalized to sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def subgradient_optimize(f0, time_limit, verbose=False, seed=42):
    """Subgradient method with step-size decay for the non-smooth max-conv objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    step_size = 0.5
    no_improve = 0
    
    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break
        elapsed = time.monotonic() - t0
        progress = elapsed / time_limit
        step_size = 0.5 * (0.9999 ** it) * (1.0 - 0.5 * progress)
        
        conv = fast_convolve(f)
        max_val = conv.max()
        active = np.where(conv >= max_val - 1e-10)[0]
        
        # Subgradient: average of gradients of active constraints
        # grad_k[i] = a[i] if i+k is in [0,2n-2], plus a[i-k] if valid
        # For conv_k = sum_i a[i]*a[k-i], grad[i] = a[k-i] + a[i-k] (if valid indices)
        grad = np.zeros(n)
        for k in active:
            for i in range(n):
                if 0 <= k - i < n:
                    grad[i] += f[k - i]
                if 0 <= i - k < n:
                    grad[i] += f[i - k]
        
        norm = np.linalg.norm(grad)
        if norm < 1e-15:
            break
        
        step = step_size / norm
        f = f - step * grad
        f = project_simplex(f)
        
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 5000:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            no_improve = 0
            step_size = max(step_size, 0.1)
    
    return best_f, best_obj

                                   verbose=True, seed=1)
        _try(a_opt, "smooth_p1")
    
    # Subgradient on best
    rem = deadline - time.monotonic() - 800
    if rem > 20:
        t_alloc = min(rem, 120)
        a_opt, _ = subgradient_optimize(best_seq, time_limit=t_alloc, seed=2)
        _try(a_opt, "subgrad_p1")
    
    # Smooth on interpolated longer sequence
    for target_n in [1024, 1536, 2048]:
        if target_n <= n_parent:
            continue
        rem = deadline - time.monotonic() - 700
        if rem < 30:
            break
        f_interp = interpolate_seq(best_seq, target_n)
        t_alloc = min(rem, 100)
        a_opt, _ = smooth_optimize(f_interp, time_limit=t_alloc, T_start=0.3, T_end=0.01, seed=10+target_n)
        _try(a_opt, f"smooth_interp_n{target_n}")
    
    # === Phase 2: LP-FW ===
    print(f"\n=== Phase 2: LP-FW (best={best_score:.6f}) ===")
    
    rem = deadline - time.monotonic() - 500
    if rem > 30:
        t_alloc = min(rem, 200)
        print(f"  LP-FW n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = lpfw_optimize(best_seq, time_limit=t_alloc, verbose=True)
        _try(a_opt, "lpfw_p2")
    
    # LP-FW with perturbation
    rem = deadline - time.monotonic() - 400
    if rem > 30:
        noise = rng.standard_normal(len(best_seq)) * 0.005
        f_pert = project_simplex(best_seq + noise)
        t_alloc = min(rem, 120)
        a_opt, _ = lpfw_optimize(f_pert, time_limit=t_alloc)
        _try(a_opt, "lpfw_pert")
    
    # LP-FW on longer sequence
    rem = deadline - time.monotonic() - 350
    if rem > 40:
        target_n = min(len(best_seq) * 2, 2048)
        f_interp = interpolate_seq(best_seq, target_n)
        t_alloc = min(rem, 150)
        a_opt, _ = lpfw_optimize(f_interp, time_limit=t_alloc)
        _try(a_opt, f"lpfw_n{target_n}")
    
    # === Phase 3: Coordinate descent ===
    print(f"\n=== Phase 3: Coordinate Descent (best={best_score:.6f}) ===")
    
    rem = deadline - time.monotonic() - 250
    if rem > 15:
        t_cd = min(rem * 0.5, 80)
        print(f"  Single CD t={t_cd:.0f}s")
        a_opt, _ = single_cd(best_seq, time_limit=t_cd)
        _try(a_opt, "cd_single")
    
    # === Phase 4: Longer sequences optimization ===
    print(f"\n=== Phase 4: Longer Sequences (best={best_score:.6f}) ===")
    
    for target_n in [2048, 3072]:
        rem = deadline - time.monotonic() - 150
        if rem < 40:
            break
        
        # Try multiple initializations at this length
        best_at_n = None
        best_at_n_score = float('inf')
        
        # Interpolated
        f_interp = interpolate_seq(best_seq, target_n)
        cur_score = 2.0 * target_n * fast_convolve(f_interp).max()
        if cur_score < best_at_n_score:
            best_at_n = f_interp
            best_at_n_score = cur_score
        
        # Geometric at this length
        for r in [0.997, 0.998, 0.999]:
            g = make_geometric(target_n, r)
            cur_score = 2.0 * target_n * fast_convolve(g).max()
            if cur_score < best_at_n_score:
                best_at_n = g
                best_at_n_score = cur_score
        
        # Modulated
        for freq in [5, 7, 10]:
            g = make_modulated(target_n, 0.997, freq, 0.4)
            cur_score = 2.0 * target_n * fast_convolve(g).max()
            if cur_score < best_at_n_score:
                best_at_n = g
                best_at_n_score = cur_score
        
        if best_at_n is not None:
            rem = deadline - time.monotonic() - 120
            t1 = min(rem * 0.5, 60)
            a_opt, _ = smooth_optimize(best_at_n, time_limit=t1, T_start=0.3, T_end=0.02,
                                       seed=target_n)
            _try(a_opt, f"smooth_long_n{target_n}")
            
            rem = deadline - time.monotonic() - 100
            if rem > 30:
                t2 = min(rem * 0.5, 50)
                a_opt, _ = lpfw_optimize(a_opt, time_limit=t2)
                _try(a_opt, f"lpfw_long_n{target_n}")
    
    # === Phase 5: Final polishing ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        
        rem = deadline - time.monotonic() - 50
        if rem > 10:
            a_opt, _ = smooth_optimize(best_seq, time_limit=min(rem, 60),
                                       T_start=0.03, T_end=0.002, verbose=True, seed=100)
            _try(a_opt, "smooth_fine")
        
        rem = deadline - time.monotonic() - 35
        if rem > 10:
            a_opt, _ = lpfw_optimize(best_seq, time_limit=min(rem, 40))
            _try(a_opt, "lpfw_final")
        
        # Multiple perturbations
        for pi in range(5):
            rem = deadline - time.monotonic() - 15
            if rem < 10:
                break
            noise = rng.standard_normal(n_best) * (0.001 + 0.001 * pi)
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = smooth_optimize(f_pert, time_limit=min(rem, 20),
                                       T_start=0.015, T_end=0.002, seed=200+pi)
            _try(a_opt, f"perturb_{pi}")
        
        # Subgradient final
        rem = deadline - time.monotonic() - 10
        if rem > 8:
            a_opt, _ = subgradient_optimize(best_seq, time_limit=min(rem, 15), seed=300)
            _try(a_opt, "subgrad_final")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
