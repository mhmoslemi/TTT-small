Uses linear programming to find optimal update directions (maximizing spread

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Cross-convolution of two equal-length sequences via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(f):
    """Build matrix A such that A @ g = convolve(f, g)."""
    n = len(f)
    size = 2 * n - 1
    A = np.zeros((size, n))
    for j in range(n):
        A[j:j + n, j] = f[j]
    return A


def lp_direction(f):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= max(f*f) for all k, g >= 0.
    Returns normalized g."""
    n = len(f)
    f = normalize(f)
    conv = fast_convolve(f)
    M = conv.max()

        return f.copy()

    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-8})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            g_sum = g.sum()

        for beta in [0.5, 1.0, 1.5, 2.0, 3.0]:
            starts.append((power_law_seq(n, beta), f"pow_b{beta}_n{n}"))
        for alpha in [1.0, 2.0, 3.0]:
            starts.append((cosine_exp_seq(n, alpha), f"cos_exp_a{alpha}_n{n}"))
        starts.append((two_sided_exp(n, 2.0), f"two_exp_n{n}"))
        starts.append((smooth_random_seq(n, rng), f"rand_smooth_n{n}"))

    # Quick evaluation
    start_scores = []
    for a, label in starts:
        sc = compute_score(a)
        start_scores.append((sc, a, label))
    start_scores.sort(key=lambda x: x[0])

    print(f"  Generated {len(starts)} starts. Top 5:")
    for i in range(min(5, len(start_scores))):
        sc, a, label = start_scores[i]
        print(f"    {i+1}. {label}: {sc:.10f}")
        record(a, label)

    # === Phase 1: LP optimization on diverse candidates ===
    print(f"\n=== Phase 1: LP optimization (best={best_score:.10f}) ===")

    # Select diverse candidates (different lengths and types)
    selected = []
    seen = set()
    for sc, a, label in start_scores:
        key = f"{len(a)}_{label.split('_')[0]}"
        if key not in seen:
            selected.append((a, label))
            seen.add(key)
        if len(selected) >= 5:
            break

    if parent is not None:
        parent_added = False
        for a, label in selected:
            if len(a) == len(parent) and label == "parent":
                parent_added = True
                break
        if not parent_added:
            selected.append((parent, "parent"))

    # Allocate time for LP on each
    n_sel = len(selected)
    rem_total = deadline - time.monotonic() - 700
    t_per = max(60, rem_total / max(1, n_sel))

    for a_init, label in selected:
        rem = deadline - time.monotonic() - 600
        if rem < 80:
            break
        t_alloc = min(t_per, rem * 0.6)
        if t_alloc < 30:
            continue
        print(f"  LP [{label}]: t={t_alloc:.0f}s, start_score={compute_score(a_init):.10f}")
        a_lp, sc_lp = lp_optimize(a_init, time_limit=t_alloc, verbose=True)
        record(a_lp, label + "_lp")

    # === Phase 2: Subgradient + CD on running best ===
    print(f"\n=== Phase 2: Subgradient polish (best={best_score:.10f}) ===")

    if best_a is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 60:
            t_sg = min(rem * 0.5, 250)
            print(f"  Subgradient: t={t_sg:.0f}s")
            a_sg, sc_sg = subgradient_nesterov(best_a, time_limit=t_sg, lr0=0.008,
                                               momentum=0.85, top_k=30, verbose=True)
            record(a_sg, "sg_phase2")

    # === Phase 3: Second LP round on best ===
    print(f"\n=== Phase 3: LP round 2 (best={best_score:.10f}) ===")

    if best_a is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 60:
            t_lp2 = min(rem * 0.7, 400)
            print(f"  LP round 2: t={t_lp2:.0f}s")
            a_lp2, sc_lp2 = lp_optimize(best_a, time_limit=t_lp2, verbose=True)
            record(a_lp2, "lp_round2")

    # === Phase 4: Coordinate descent + pair transfer ===
    print(f"\n=== Phase 4: Local polish (best={best_score:.10f}) ===")

    if best_a is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 40:
            t_cd = min(rem * 0.4, 150)
            print(f"  CD: t={t_cd:.0f}s")
            a_cd, sc_cd = coord_descent_precise(best_a, time_limit=t_cd, verbose=True)
            record(a_cd, "cd_phase4")

        rem = deadline - time.monotonic() - 60
        if rem > 30:
            t_ps = min(rem * 0.5, 100)
            print(f"  PairTransfer: t={t_ps:.0f}s")
            a_ps, sc_ps = pair_transfer(best_a, time_limit=t_ps, n_iters=3000, verbose=True)
            record(a_ps, "pair_phase4")

    # === Phase 5: Final LP + CD on absolute best ===
    print(f"\n=== Phase 5: Final optimization (best={best_score:.10f}) ===")

    if best_a is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 60:
            t_final_lp = min(rem * 0.5, 200)
            print(f"  Final LP: t={t_final_lp:.0f}s")
            a_fl, sc_fl = lp_optimize(best_a, time_limit=t_final_lp, verbose=False)
            record(a_fl, "final_lp")

        rem = deadline - time.monotonic() - 15
        if rem > 30:
            t_final_cd = min(rem * 0.8, 100)
            print(f"  Final CD: t={t_final_cd:.0f}s")
            a_fc, sc_fc = coord_descent_precise(best_a, time_limit=t_final_cd, verbose=False)
            record(a_fc, "final_cd")

    # === Phase 6: Try a few more lengths on the best shape ===
    print(f"\n=== Phase 6: Length exploration (best={best_score:.10f}) ===")

    if best_a is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 60:
            best_n = len(best_a)
            for target_n in [best_n - 64, best_n + 64, best_n - 128, best_n + 128]:
                if target_n < 256 or target_n > 2048 or target_n == best_n:
                    continue
                rem = deadline - time.monotonic() - 15
                if rem < 40:
                    break
                # Resample
                old_x = np.linspace(0, 1, best_n)
                new_x = np.linspace(0, 1, target_n)
                a_interp = np.interp(new_x, old_x, best_a)
                a_interp = normalize(a_interp)
                sc_int = compute_score(a_interp)
                # Quick LP
                t_quick = min(rem * 0.5, 60)
                a_opt, sc_opt = lp_optimize(a_interp, time_limit=t_quick, verbose=False)
                record(a_opt, f"dim_{target_n}")

    # === Fallback ===
    if best_a is None:
        best_a = geometric_seq(768, 2.0)
        best_score = compute_score(best_a)

    # === Final verification ===
    final_list = [float(x) for x in best_a]
    final_score = float(evaluate_sequence(final_list))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_a)})")
    return final_list
