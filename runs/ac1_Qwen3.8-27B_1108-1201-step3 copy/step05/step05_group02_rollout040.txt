def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast LP-Frank-Wolfe
with O(n) greedy dual solver, making each iteration 100-1000x faster
than general LP-based approaches. Enables millions of iterations for
superior convergence.

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A*A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def greedy_dual(f, n):
    """O(n) greedy solve of the dual LP."""
    y = np.zeros(2*n - 1)
    ws = 0.0
    for j in range(n):
        f_j = max(f[j], 1e-14)
        deficit = 1.0/f_j - ws
        if deficit > 0:
            y[j + n - 1] += deficit
            ws += deficit
        # Slide window: remove y[j], add y[j+n]
        if j + n < 2*n - 1:
            ws = ws - y[j] + y[j + n]
        else:
            ws = ws - y[j]
    return y

    """Compute FW direction using greedy dual + linear system."""
    y = greedy_dual(f, n)
    dual_obj = y.sum()

        return None  # At (near) optimum
    
    # Find free variables (dual constraint tight)
    T = []
    f_safe = np.maximum(f, 1e-14)
    for j in range(n):

        target = 1.0 / f_safe[j]
        if ws >= target * (1 - 1e-6) - 1e-10:
            T.append(j)
    T = np.array(T)

    # Need to be efficient here
    A = np.zeros((m + 1, d))
    b = np.zeros(m + 1)
    b[m] = M * dual_obj
    
    T_list = T.tolist()
    for idx in range(d):
        A[m, idx] = 1.0
    
    for i, k in enumerate(S):
        b[i] = M
        k_int = int(k)

        j_hi = min(n - 1, k_int)
        for idx, j in enumerate(T_list):
            if j_lo <= j <= j_hi:

    gs = g_T.sum()
    if gs < 1e-14:
        return None
    
    g = np.zeros(n)
    g[T] = g_T
    g /= g.sum()
    return g
