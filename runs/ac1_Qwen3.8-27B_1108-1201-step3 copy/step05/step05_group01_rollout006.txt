Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def smooth_obj_grad(t, beta):
    """Compute smooth log-sum-exp objective and gradient in t-space."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    obj = conv_max + np.log(exp_shifted.sum()) / beta
    
    # Gradient: grad_a[i] = 2 * sum_m w[m] * a[m+i] (cross-correlation)
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R
    
    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)
    
    return obj, grad_t


def cutting_plane_lp(f, M, n, max_iters=200):
    """Solve LP with cutting planes: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    # Start with all constraints, but we can be smarter
    # Actually for n up to ~1000, the full LP is fine with HiGHS
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

            t = res.x

def mirror_descent(f0, time_limit, verbose=False):
    """Mirror descent (exponential weights) with adaptive step size."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    eta = 0.1
    momentum = 0.8
    v = np.zeros(n)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        conv_max = conv.max()
        
        # Smooth weights for the subgradient
        beta = 50.0
        shifted = beta * (conv - conv_max)
        exp_s = np.exp(shifted)
        w = exp_s / exp_s.sum()
        
        # Gradient in a-space
        N = 1 << (3 * n - 2).bit_length()
        A = np.fft.rfft(f, N)
        W = np.fft.rfft(w, N)
        R = np.fft.irfft(W * np.conj(A), N)[:n]
        grad_a = 2.0 * R
        
        # Momentum
        v = momentum * v + (1.0 - momentum) * grad_a
        
        # Mirror descent update: f_new[i] ∝ f[i] * exp(-eta * v[i])
        log_f = np.log(f + 1e-30) - eta * v
        f_new = softmax_fn(log_f)
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()
        
        obj_new = fast_convolve(f_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 200 == 0:
                print(f"    MD: obj={best_obj:.10f}")
        else:
            no_improve += 1
            eta *= 0.98
            
        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            eta = 0.05
            no_improve = 0
        
        eta = max(eta, 0.001)
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Fine coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-14:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            
            # Try different multipliers
            for mult in [0.7, 0.85, 0.92, 0.97, 1.03, 1.08, 1.15, 1.3]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-16:
                    best_j_obj = obj
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic-like interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-strategy approach:
1. LP-based Frank-Wolfe with adaptive M expansion (primary local optimizer)
2. L-BFGS on log-sum-exp smoothed objective with aggressive multi-beta annealing

4. Randomized LP-FW for escaping local optima
5. Multi-length exploration with interpolation

7. Simulated annealing with structured perturbations

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def smooth_obj_grad(t, beta):
    """Compute smooth log-sum-exp objective and gradient in t-space."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    obj = conv_max + np.log(exp_shifted.sum()) / beta

    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R

    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)

    return obj, grad_t


def solve_lp(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x
    return None

def mirror_descent(f0, time_limit, verbose=False):
    """Mirror descent with adaptive step size and momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)

    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        conv_max = conv.max()

        beta = 40.0
        shifted = beta * (conv - conv_max)
        exp_s = np.exp(shifted)
        w = exp_s / exp_s.sum()

        N = 1 << (3 * n - 2).bit_length()
        A = np.fft.rfft(f, N)
        W = np.fft.rfft(w, N)
        R = np.fft.irfft(W * np.conj(A), N)[:n]
        grad_a = 2.0 * R

        v = momentum * v + (1.0 - momentum) * grad_a

        log_f = np.log(f + 1e-30) - eta * v
        f_new = softmax_fn(log_f)
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()

        obj_new = fast_convolve(f_new).max()

            f = f_new

            eta *= 0.985

            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            v = np.zeros(n)
            eta = 0.04
            no_improve = 0

        eta = max(eta, 0.001)

    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Fine coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    multipliers = [0.6, 0.75, 0.85, 0.92, 0.96, 1.04, 1.08, 1.15, 1.25, 1.4]

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-16:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def simulated_annealing(f0, time_limit, verbose=False):
    """Simulated annealing with structured block perturbations."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    temp = 0.02
    cool = 0.9995
    rng = np.random.default_rng(12345)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Block perturbation: pick a random contiguous block and rescale
        block_size = rng.integers(max(4, n // 32), min(n // 4, n))
        start = rng.integers(0, n - block_size)
        end = start + block_size

        f_trial = f.copy()
        # Gaussian perturbation on the block
        noise = rng.standard_normal(block_size) * temp
        f_trial[start:end] += noise
        f_trial = np.maximum(f_trial, 0)
        fs = f_trial.sum()
        if fs < 1e-15:
            f_trial = best_f.copy()
        else:
            f_trial = f_trial / fs

        obj_trial = fast_convolve(f_trial).max()

        # Metropolis accept
        delta = obj_trial - best_obj
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            f = f_trial
            if obj_trial < best_obj - 1e-14:
                best_obj = obj_trial
                best_f = f.copy()
                if verbose and it % 500 == 0:
                    print(f"    SA iter {it}: obj={best_obj:.10f}, temp={temp:.6f}")

        temp = max(temp * cool, 0.001)

        if temp <= 0.0011:
            # Restart
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            temp = 0.01

    return best_f, best_obj

                f_pert = project_simplex(base + noise)

                try_seq(a_opt, f"lbfqg_pert_{pert_scale}")

    # === Phase 3: Mirror Descent (150s) ===
    print(f"\n=== Phase 3: Mirror Descent (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    t3 = min(150, deadline - time.monotonic() - 400)
    if t3 > 20:
        a_opt, _ = mirror_descent(base, time_limit=t3, verbose=True)
        try_seq(a_opt, "mirror_descent")

    # === Phase 4: Multi-length exploration (200s) ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent

    target_lengths = sorted(set([
        max(128, n_parent // 2),
        max(128, n_parent * 3 // 4),
        n_parent,
        n_parent + 64,
        n_parent + 128,
        min(2048, n_parent * 5 // 4),
    ]))

    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 250
        if rem < 40:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.35, 70)
        print(f"  Trying n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lpfw_optimize(f_interp, time_limit=t_alloc)
        try_seq(a_opt, f"len_{target_n}")

    # === Phase 5: Deep optimization on best (350s) ===
    print(f"\n=== Phase 5: Deep Optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # 5a: LP-FW deep
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(rem * 0.4, 150)
            a_opt, _ = lpfw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_lpfw")

        # 5b: L-BFGS deep
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_alloc = min(rem * 0.5, 120)
            a_opt, _ = lbfqg_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_lbfqg")

        # 5c: Mirror descent
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t_alloc = min(rem * 0.4, 50)
            a_opt, _ = mirror_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "deep_md")

        # 5d: Simulated annealing
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_alloc = min(rem * 0.5, 40)
            a_opt, _ = simulated_annealing(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "deep_sa")

    # === Phase 6: Final polishing (remaining) ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Coordinate descent for fine polishing
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_alloc = min(rem * 0.4, 60)
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "polish_cd")

        # Multiple short LP-FW with tiny perturbations
        for attempt in range(6):
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break

            if attempt % 2 == 0:
                noise = rng.standard_normal(n_best) * 0.001
                f_start = project_simplex(best_seq + noise)
                t_alloc = min(rem * 0.7, 35)
                a_opt, _ = lpfw_optimize(f_start, time_limit=t_alloc)
                try_seq(a_opt, f"final_lpfw_{attempt}")
            else:
                f_start = best_seq.copy()
                t_alloc = min(rem * 0.7, 30)
                a_opt, _ = lbfqg_optimize(f_start, time_limit=t_alloc)
                try_seq(a_opt, f"final_lbfqg_{attempt}")

        # Final CD pass
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 15))
            try_seq(a_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
