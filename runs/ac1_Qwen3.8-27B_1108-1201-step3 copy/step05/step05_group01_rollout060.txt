- Minimize t

- And: sum_j d[j] = 0
- And: |d[j]| <= delta (to keep the step small)

This LP has n+1 variables and |S|+1 constraints. If |S| is small (say < 20), this is very fast.

    # So rcc[i] = conv(w_rev, a)[i + n_w - 1]
    # where w_rev = w[::-1]
    
    w_rev = w[::-1]
    N = 1 << (len(w_rev) + n - 1 - 1).bit_length()
    W_rev = np.fft.rfft(w_rev, N)
    A = np.fft.rfft(a, N)
    conv_result = np.fft.irfft(W_rev * A, N)
    # rcc[i] = conv_result[i + n_w - 1] for i = 0, ..., n-1
    offset = n_w - 1
    rcc = conv_result[offset:offset + n]
    return 2.0 * rcc

def hankel_matvec(w, v):
    """Compute H @ v where H[i,j] = 2*w[i+j] (HANKEL matrix).
    (H@v)[i] = 2 * sum_j w[i+j] * v[j] = 2 * cross_corr(w, v)[i]
    """
    return conv_gradient(v, w)  # Same computation!

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s

def fienup_init(n, iterations=300, seed=42):
    """GERCHBERG-SAXTON: find non-negative a with flat (zero-padded) power spectrum."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    
    # Initialize with random phases and flat magnitude
    phase = rng.uniform(0, 2 * np.pi, fft_size)
    A = np.exp(1j * phase) / np.sqrt(fft_size)
    
    best_a = np.ones(n) / n
    best_score = np.inf
    
    for it in range(iterations):
        # Time domain
        a_full = np.fft.ifft(A).real
        a = a_full[:n].copy()
        
        # Enforce non-negativity and normalize
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        
        if it % 50 == 0:
            score = fast_convolve(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        
        # Frequency domain: flatten magnitude
        a_padded = np.zeros(fft_size)
        a_padded[:n] = a
        A = np.fft.fft(a_padded)
        
        # Set magnitude to be as flat as possible
        # Keep DC to maintain sum
        dc = A[0]
        target_mag = np.ones(fft_size)
        # Scale so that total energy is preserved
        scale = np.abs(A).sum() / fft_size
        target_mag[:] = scale
        target_mag[0] = abs(dc)
        
        A = target_mag * np.exp(1j * np.angle(A))
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    else:
        a = best_a
    
    return a

def smooth_obj_and_grad_t(t, beta):
    """Compute smoothed objective and gradient w.r.t. t (softmax parameterization)."""
    n = len(t)
    a = softmax_fn(t)
    c = fast_convolve(a)
    c_max = c.max()
    shifted = beta * (c - c_max)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    obj = c_max + np.log(exp_s.sum()) / beta
    
    # grad_a[i] = 2 * sum_j w[i+j] * a[j]
    grad_a = conv_gradient(a, w)
    
    # grad_t = a * (grad_a - <a, grad_a>)
    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)
    
    return obj, grad_t

def lbfqg_optimize(a0, time_limit, verbose=False, seed=42):
    """L-BFGS optimization with adaptive beta schedule."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    
    n = len(a)
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    
    eps = 1e-15
    t = np.log(a + eps)
    
    t0_time = time.monotonic()
    
    # Beta schedule: start small, grow geometrically
    beta = 5.0
    grows = [5.0, 8.0, 13.0, 21.0, 34.0, 55.0, 89.0, 144.0, 233.0, 377.0, 610.0, 987.0, 1597.0]
    
    for beta in grows:
        remaining = time_limit - (time.monotonic() - t0_time)
        if remaining < 20:
            break
        
        # How many L-BFGS iterations? Scale with remaining time
        # Each iteration is O(n log n) for the gradient, plus L-BFGS overhead
        # Estimate: ~1ms per gradient for n=768, so ~1000 iters per second
        # L-BFGS does ~1 gradient eval per iteration (roughly)
        # Give it a fraction of remaining time
        
        sub_time = max(5, remaining * 0.3)  # Use 30% of remaining for this beta level
        max_iter = max(50, int(sub_time * 50))  # Rough estimate
        
        try:
            res = minimize(
                smooth_obj_and_grad_t, t, args=(beta,),
                jac=True, method='L-BFGS-B',
                options={'maxiter': max_iter, 'gtol': 1e-12, 'maxls': 50,
                         'maxfun': max_iter * 3}
            )
            t = res.x
            a = softmax_fn(t)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 1e-15:
                a = a / a_sum
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_a = a.copy()
                if verbose:
                    print(f"    L-BFGS beta={beta}: obj={best_obj:.10f}")
        except Exception as e:
            if verbose:
                print(f"    L-BFGS beta={beta} failed: {e}")
    
    return best_a, best_obj

def active_set_optimize(a0, time_limit, verbose=False):
    """Active set peak reduction using LP for direction finding."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        M = c.max()
        
        # Identify active set: lags where c[k] >= M * 0.999
        threshold = M * 0.999
        active = np.where(c >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        
        # Cap the active set size for LP efficiency
        if len(active) > 30:
            # Keep only the top 30
            top_idx = active[np.argsort(c[active])[-30:]]
            active = top_idx
        
        # Solve LP: minimize t
        # s.t. for each k in active: 2 * sum_j d[j] * a[k-j] <= t
        #      sum_j d[j] = 0
        #      |d[j]| <= 1 (for all j)
        
        # The constraint 2 * sum_j d[j] * a[k-j] <= t
        # For each k: sum_j (2*a[k-j]) * d[j] - t <= 0
        # (where a[k-j] = 0 if k-j < 0 or k-j >= n)
        
        # Variables: d[0], ..., d[n-1], t  (total n+1 variables)
        # Objective: minimize t  ->  c = [0, ..., 0, 1]
        
        m_active = len(active)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0  # Minimize t
        
        # Inequality constraints: A_ub @ x <= b_ub
        # For each active lag k: 2 * sum_j a[k-j] * d[j] - t <= 0
        A_ub = np.zeros((m_active + 1, n + 1))
        b_ub = np.zeros(m_active + 1)
        
        for idx, k in enumerate(active):
            for j in range(n):
                kj = k - j
                if 0 <= kj < n:
                    A_ub[idx, j] = 2.0 * a[kj]
            A_ub[idx, n] = -1.0  # -t term
        
        # Equality constraint: sum(d) = 0
        # Use A_eq
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        
        bounds = [(-1, 1)] * n + [(None, None)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                         A_eq=A_eq, b_eq=[0],
                         bounds=bounds, method='highs')
            if not res.success:
                no_improve += 1
                if no_improve > 50:
                    break
                continue
            
            d = res.x[:n]
            t_val = res.x[n]
            
            # The LP gives a direction d. Take a step a + alpha * d
            # and re-normalize.
            # Find alpha by line search.
            best_alpha = 0.0
            best_step_obj = M
            
            for alpha in [0.01, 0.03, 0.05, 0.1, 0.2, 0.3, 0.5]:
                a_trial = a + alpha * d
                a_trial = np.maximum(a_trial, 0)
                s_trial = a_trial.sum()
                if s_trial < 1e-10:
                    continue
                a_trial = a_trial / s_trial
                obj_trial = fast_convolve(a_trial).max()
                if obj_trial < best_step_obj:
                    best_step_obj = obj_trial
                    best_alpha = alpha
            
            if best_alpha > 0:
                a_new = a + best_alpha * d
                a_new = np.maximum(a_new, 0)
                a_new = a_new / a_new.sum()
                
                c_new = fast_convolve(a_new)
                obj_new = c_new.max()
                a = a_new
                
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a.copy()
                    no_improve = 0
                    if verbose:
                        print(f"    Active set: obj={best_obj:.10f}, |S|={len(active)}")
                else:
                    no_improve += 1
            else:
                no_improve += 1
        
            if no_improve > 100:
                # Perturb and restart
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
                no_improve = 0
        
        except Exception:
            no_improve += 1
            if no_improve > 50:
                break
    
    return best_a, best_obj

def md_optimize(a0, time_limit, verbose=False):
    """Mirror-descent / multiplicative weight optimization with HANKEL gradient."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    beta = 20.0
    eta = 0.1  # Learning rate
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        c_max = c.max()
        shifted = beta * (c - c_max)
        exp_s = np.exp(shifted)
        w = exp_s / exp_s.sum()
        
        # Gradient of smoothed max w.r.t. a
        grad_a = conv_gradient(a, w)
        
        # Mirror descent step: a_new[i] proportional to a[i] * exp(-eta * grad_a[i])
        log_a = np.log(a + 1e-30)
        a_new = np.exp(log_a - eta * grad_a)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = fast_convolve(a_new).max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
            # Slowly increase beta
            beta *= 1.02
            if verbose and best_obj < 1.51:
                print(f"    MD: obj={best_obj:.10f}, beta={beta:.1f}, eta={eta:.4f}")
        else:
            no_improve += 1
            eta *= 0.95
            if no_improve > 200:
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
                eta = 0.08
                beta = 20.0
                no_improve = 0
        
        eta = max(eta, 0.001)
        beta = min(beta, 2000.0)
    
    return best_a, best_obj
