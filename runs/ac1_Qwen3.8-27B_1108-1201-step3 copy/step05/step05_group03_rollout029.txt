OK this is getting too complex in my head. Let me write a cleaner implementation.

Key decisions:
1. Use softmax reparameterization: a = softmax(b), optimize over b.
2. Objective: soft-max of top K peaks of a*a.
3. Use L-BFGS-B for optimization.
4. Compute gradient analytically using FFT-based cross-correlation.

Let me write the gradient computation more carefully:

Given a (length n), conv = a * a (full, length 2n-1).
For a peak at position k with weight w_k:
d(conv[k])/d(a[j]) = 2 * a[k-j] if 0 <= k-j < n, else 0.

So grad_a[j] = 2 * sum_{k in peaks} w_k * a[k-j] * 1[0 <= k-j < n]

This is: grad_a = 2 * cross_correlation(w_vec, a) where w_vec has w_k at position k and 0 elsewhere.

cross_correlation at lag j: sum_k w_k * a[k-j]

Using the property that cross_correlation(w, a)[j] = convolution(w, flip(a))[j + len(a) - 1]:
- flip(a) = a[::-1] (length n)
- w has length 2n-1
- convolution has length (2n-1) + n - 1 = 3n-2
- cross_correlation[j] = conv_result[j + n - 1]

So:
```python
fft_size = 1 << (3*n - 2 - 1).bit_length()
W = np.fft.rfft(w_vec, fft_size)
Af = np.fft.rfft(a[::-1], fft_size)
cross_corr_full = np.fft.irfft(W * Af, fft_size)
grad_a = 2.0 * cross_corr_full[n-1 : n-1+n]
```

Then for the softmax parameterization:
db[m] = grad_a[m] * a[m] - a[m] * dot(grad_a, a)

```python
import numpy as np
from scipy.optimize import minimize
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def softmax_obj_grad(b, beta, K):
    """Compute objective and gradient for softmax parameterization.
    a = softmax(b), minimize soft-max of top K peaks."""
    a = np.exp(b - b.max())
    a = a / a.sum()
    conv = fast_convolve(a)
    
    # Get top K
    K_actual = min(K, len(conv))
    top_idx = np.argpartition(conv, -K_actual)[-K_actual:]
    top_vals = conv[top_idx]
    
    # Soft-max value
    max_val = top_vals.max()
    scaled = beta * (top_vals - max_val)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(np.sum(exp_scaled))
    obj = (log_sum + beta * max_val) / beta  # This is log(sum(exp(beta*x))/beta) = logsumexp(beta*x)/beta
    
    # Weights
    weights = exp_scaled / np.sum(exp_scaled)
    
    # Gradient of obj w.r.t. conv[top_idx]:
    # d/d(conv[k]) [logsumexp(beta * top_vals) / beta] = weights[i] for k = top_idx[i]
    
    # Gradient of conv w.r.t. a:
    # d(conv[k])/d(a[j]) = 2*a[k-j] if 0 <= k-j < n
    n = len(a)
    w_vec = np.zeros(2 * n - 1)
    w_vec[top_idx] = weights
    # (Note: factor of 2 will be included in grad_a)
    
    # Compute grad_a[j] = 2 * sum_k w_vec[k] * a[k-j] * 1[0<=k-j<n]
    # Using cross-correlation via FFT:
    fft_size = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w_vec, fft_size)
    Af = np.fft.rfft(a[::-1], fft_size)
    cross_corr = np.fft.irfft(W * Af, fft_size)
    grad_a = 2.0 * cross_corr[n-1 : n-1+n]
    
    # Gradient w.r.t. b (softmax chain rule):
    # da[j]/db[m] = a[j]*(delta_{jm} - a[m])
    # db[m] = sum_j grad_a[j] * da[j]/db[m] = grad_a[m]*a[m] - a[m]*sum_j grad_a[j]*a[j]
    da_db_sum = np.dot(grad_a, a)
    grad_b = grad_a * a - a * da_db_sum
    
    return obj, grad_b

def optimize_softmax(a_init, beta=50.0, K=50, maxiter=200, time_limit=300):
    """Optimize using softmax parameterization and L-BFGS-B."""
    n = len(a_init)
    a_init = normalize(a_init.copy())
    # Initialize b = log(a_init) (with small epsilon)
    b0 = np.log(np.maximum(a_init, 1e-15))
    
    t0 = time.monotonic()
    
    def obj_func(b):
        val, _ = softmax_obj_grad(b, beta, K)
        return val
    
    def grad_func(b):
        _, g = softmax_obj_grad(b, beta, K)
        return g
    
    best_a = a_init.copy()
    best_score = compute_score(a_init)
    
    # Use L-BFGS-B
    res = minimize(obj_func, b0, jac=grad_func, method='L-BFGS-B',
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8,
                            'maxfun': 500})
    
    if res.x is not None:
        a_opt = np.exp(res.x - res.x.max())
        a_opt = a_opt / a_opt.sum()
        sc = compute_score(a_opt)
        if sc < best_score:
            best_score = sc
            best_a = a_opt
    
    # Also try with different beta values
    for beta_try in [10, 20, 100, 200]:
        if time.monotonic() - t0 > time_limit:
            break
        b0 = np.log(np.maximum(best_a, 1e-15))
        
        def obj_func2(b):
            val, _ = softmax_obj_grad(b, beta_try, K)
            return val
        
        def grad_func2(b):
            _, g = softmax_obj_grad(b, beta_try, K)
            return g
        
        res = minimize(obj_func2, b0, jac=grad_func2, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8,
                                'maxfun': 500})
        if res.x is not None:
            a_opt = np.exp(res.x - res.x.max())
            a_opt = a_opt / a_opt.sum()
            sc = compute_score(a_opt)
            if sc < best_score:
                best_score = sc
                best_a = a_opt
    
    return best_a, best_score


def projected_gradient(a_init, time_limit=100, lr=0.01, decay=0.999, K=30):
    """Projected gradient method on the simplex."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        conv = fast_convolve(a)
        
        # Find top K peaks
        K_actual = min(K, len(conv))
        top_idx = np.argpartition(conv, -K_actual)[-K_actual:]
        top_vals = conv[top_idx]
        
        # Softmax weights
        temp = max(1e-10, (top_vals.max() - top_vals.min()) * 0.3)
        scaled = (top_vals - top_vals.max()) / temp
        w = np.exp(scaled)
        w /= w.sum()
        
        # Compute gradient of soft-max peak
        w_vec = np.zeros(2 * n - 1)
        w_vec[top_idx] = w
        fft_size = 1 << (3 * n - 2).bit_length()
        W = np.fft.rfft(w_vec, fft_size)
        Af = np.fft.rfft(a[::-1], fft_size)
        cross_corr = np.fft.irfft(W * Af, fft_size)
        grad_a = 2.0 * cross_corr[n-1:n-1+n]
        
        # Gradient step
        a_new = a - lr * grad_a
        a_new = project_simplex(a_new)
        
        # Check improvement
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce learning rate and try smaller step
            lr *= 0.5
            a_new = a - lr * grad_a
            a_new = project_simplex(a_new)
            sc = compute_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
            else:
                lr = max(lr * 1.5, 0.001)
        
        if step % 100 == 0:
            lr = max(lr * 0.95, 0.0005)
    
    return best_a, best_score


def simulated_annealing(a_init, time_limit=200, T0=0.01, T1=0.0001):
    """Simulated annealing with structured moves."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    T = T0
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        # Temperature schedule
        T = T0 * (T1 / T0) ** (step / max(1, int(time_limit * 10)))
        
        # Choose move type
        move_type = rng.integers(0, 4)
        
        a_try = a.copy()
        
        if move_type == 0:
            # Pair transfer
            i, j = rng.integers(0, n, 2)
            if i != j and a[j] > 1e-10:
                delta = a[j] * rng.uniform(0.01, 0.3)
                a_try[j] -= delta
                a_try[i] += delta
        
        elif move_type == 1:
            # Coordinate scaling
            i = rng.integers(0, n)
            a_try[i] *= rng.uniform(0.5, 2.0)
        
        elif move_type == 2:
            # Block transfer
            block_size = rng.integers(3, 10)
            start = rng.integers(0, n - block_size)
            delta = a[start:start+block_size].sum() * rng.uniform(0.05, 0.2)
            target = rng.integers(0, n - block_size)
            if target != start:
                a_try[start:start+block_size] -= delta / block_size
                a_try[target:target+block_size] += delta / block_size
        
        elif move_type == 3:
            # Gradient-guided move
            conv = fast_convolve(a)
            peak_idx = int(np.argmax(conv))
            # Move mass away from positions that contribute to the peak
            js = max(0, peak_idx - n + 1)
            je = min(n - 1, peak_idx)
            source = rng.integers(js, je + 1)
            target = rng.integers(0, n)
            if source != target:
                delta = a[source] * rng.uniform(0.01, 0.1)
                a_try[source] -= delta
                a_try[target] += delta
        
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-10:
            continue
        a_try = a_try / s
        
        sc_try = compute_score(a_try)
        sc_cur = compute_score(a)
        
        if sc_try < sc_cur:
            a = a_try
            if sc_try < best_score:
                best_score = sc_try
                best_a = a_try.copy()
        else:
            # Accept with probability exp(-(sc_try - sc_cur) / T)
            if rng.random() < np.exp(-(sc_try - sc_cur) / T):
                a = a_try
    
    return best_a, best_score


def cosine_power_sequence(n, gamma):
    """a[i] = (0.5 + 0.5*cos(2*pi*i/(n-1)))^gamma"""
    x = np.linspace(0, 2*np.pi, n)
    a = (0.5 + 0.5 * np.cos(x)) ** gamma
    a = np.maximum(a, 0)
    return normalize(a)


def lorentzian_sequence(n, gamma):
    """a[i] = 1 / (1 + gamma * ((i - (n-1)/2) / (n/2))^2)^gamma"""
    x = (np.arange(n) - (n-1)/2.0) / (n/2.0)
    a = 1.0 / (1.0 + gamma * x**2) ** gamma
    return normalize(a)


def power_law_sequence(n, alpha):
    """a[i] = (i+1)^(-alpha) * (n-i)^(-alpha)"""
    i = np.arange(n)
    a = (i + 1.0) ** (-alpha) * (n - i) ** (-alpha)
    return normalize(a)


def exponential_sequence(n, alpha):
    """a[i] = exp(-alpha * |i - (n-1)/2| / (n/2))"""
    x = np.abs(np.arange(n) - (n-1)/2.0) / (n/2.0)
    a = np.exp(-alpha * x)
    return normalize(a)


def geometric_sequence(n, r):
    """Geometric sequence."""
    a = r ** np.arange(n)
    return normalize(a)
```

```python
"""
Minimizes 2*n*max(auto_conv)/(sum)^2 using:
1. Softmax-reparameterized non-convex optimization (L-BFGS-B) with analytical
   gradients on a log-sum-exp smoothing of the peak autocorrelation.
2. Projected gradient descent on the simplex targeting active peaks.
3. Simulated annealing with gradient-guided moves.
4. Diverse parametric initializations (cosine-power, Lorentzian, power-law,
   exponential, geometric) across multiple sequence lengths.
5. Dimension interpolation and aggressive final polish.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_conv/sum^2 (same as evaluate_sequence but faster)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_normalize(b):
    """Convert log-parameters to probability vector via softmax."""
    b_shifted = b - b.max()
    a = np.exp(b_shifted)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(b)) / len(b)
    return a / s


def logsumexp_obj_grad(b, beta, K):
    """
    Compute objective (log-sum-exp of top K peaks / beta) and gradient
    for the softmax parameterization a = softmax(b).
    """
    n = len(b)
    a = softmax_normalize(b)
    conv = fast_convolve(a)

    K_actual = min(K, len(conv))
    top_idx = np.argpartition(conv, -K_actual)[-K_actual:]
    top_vals = conv[top_idx]

    max_val = top_vals.max()
    scaled = beta * (top_vals - max_val)
    exp_scaled = np.exp(scaled)
    sum_exp = np.sum(exp_scaled)
    log_sum_ex = np.log(sum_exp)
    obj = (log_sum_ex + beta * max_val) / beta

    weights = exp_scaled / sum_exp

    # Compute grad_a[j] = 2 * sum_k weights[k] * a[k-j] for valid k-j
    grad_a = np.zeros(n)
    for idx in range(K_actual):
        k = int(top_idx[idx])
        wt = 2.0 * weights[idx]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        # a[k-j] for j in [js, je]: a[k-js], a[k-js-1], ..., a[k-je]
        # = a[k-je : k-js+1][::-1]
        grad_a[js:je + 1] += wt * a[k - je:k - js + 1][::-1]

    # Softmax chain rule: db[m] = grad_a[m]*a[m] - a[m]*dot(grad_a, a)
    grad_a_dot_a = np.dot(grad_a, a)
    grad_b = grad_a * a - a * grad_a_dot_a

    return obj, grad_b


def optimize_softmax(a_init, beta_list, K, maxiter_per, time_limit):
    """
    Optimize using softmax parameterization and L-BFGS-B with
    increasing beta (annealing the smoothness parameter).
    """
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for beta in beta_list:
        if time.monotonic() - t0 > time_limit - 10:
            break
        b0 = np.log(np.maximum(a, 1e-15))

        def obj_func(b):
            val, _ = logsumexp_obj_grad(b, beta, K)
            return val

        def grad_func(b):
            _, g = logsumexp_obj_grad(b, beta, K)
            return g

        try:
            res = minimize(obj_func, b0, jac=grad_func, method='L-BFGS-B',
                           options={'maxiter': maxiter_per, 'ftol': 1e-15,
                                    'gtol': 1e-8, 'maxfun': maxiter_per * 3})
        except Exception:
            continue

        if res.x is not None:
            a_opt = softmax_normalize(res.x)
            sc = compute_score(a_opt)
            if sc < best_score:
                best_score = sc
                best_a = a_opt.copy()
                a = a_opt

    return best_a, best_score

def simulated_annealing(a_init, time_limit, T0=0.005, T1=0.00005):
    """Simulated annealing with gradient-guided structured moves."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    cur_score = best_score
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    # Estimate total iterations
    est_total = int(time_limit * 200)  # rough estimate
    est_total = max(est_total, 100)

    step = 0
    while time.monotonic() - t0 < time_limit:
        step += 1
        T = T0 * (T1 / T0) ** (step / est_total)

        a_try = a.copy()
        move_type = rng.integers(0, 5)

        if move_type == 0:
            # Pair transfer
            i, j = rng.integers(0, n, 2)
            if i != j and a[j] > 1e-10:
                delta = a[j] * rng.uniform(0.02, 0.25)
                a_try[j] -= delta
                a_try[i] += delta

        elif move_type == 1:
            # Coordinate rescale
            i = rng.integers(0, n)
            if a[i] > 1e-10:
                a_try[i] *= rng.uniform(0.5, 2.0)

        elif move_type == 2:
            # Block redistribution
            bs = int(rng.integers(3, 8))
            if n > 2 * bs:
                src = int(rng.integers(0, n - bs))
                dst = int(rng.integers(0, n - bs))
                if abs(src - dst) > bs:
                    frac = rng.uniform(0.05, 0.15)
                    block_sum = a[src:src + bs].sum()
                    transfer = block_sum * frac
                    a_try[src:src + bs] -= transfer / bs
                    a_try[dst:dst + bs] += transfer / bs

        elif move_type == 3:
            # Gradient-guided: move away from peak contributors
            conv = fast_convolve(a)
            peak_idx = int(np.argmax(conv))
            js = max(0, peak_idx - n + 1)
            je = min(n - 1, peak_idx)
            # Weighted random source from peak contributors
            contrib = a[js:je + 1].copy()
            total_c = contrib.sum()
            if total_c > 1e-15:
                contrib /= total_c
                src = int(rng.choice(je - js + 1, p=contrib)) + js
                dst = int(rng.integers(0, n))
                if src != dst:
                    delta = a[src] * rng.uniform(0.02, 0.15)
                    a_try[src] -= delta
                    a_try[dst] += delta

        else:
            # Multi-coordinate perturbation
            n_pert = int(rng.integers(3, 10))
            idxs = rng.integers(0, n, n_pert)
            for idx in idxs:
                a_try[idx] *= rng.uniform(0.7, 1.3)

        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-10:
            continue
        a_try = a_try / s

        sc_try = compute_score(a_try)

        if sc_try < cur_score:
            a = a_try
            cur_score = sc_try
            if sc_try < best_score:
                best_score = sc_try
                best_a = a_try.copy()
        else:
            diff = sc_try - cur_score
            if diff / T < 50:  # Avoid overflow
                if rng.random() < np.exp(-diff / T):
                    a = a_try
                    cur_score = sc_try

    return best_a, best_score


def cosine_power_sequence(n, gamma):
    """a[i] proportional to (0.5 + 0.5*cos(2*pi*i/(n-1)))^gamma."""
    x = np.linspace(0, 2 * np.pi, n)
    a = (0.5 + 0.5 * np.cos(x)) ** gamma
    a = np.maximum(a, 0)
    return normalize(a)


def lorentzian_sequence(n, gamma):
    """a[i] = 1/(1 + gamma*((i-(n-1)/2)/(n/2))^2)^gamma."""
    x = (np.arange(n) - (n - 1) / 2.0) / (n / 2.0)
    a = (1.0 / (1.0 + gamma * x ** 2)) ** gamma
    return normalize(a)


def power_law_sequence(n, alpha):
    """a[i] = (i+1)^(-alpha) * (n-i)^(-alpha)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** (-alpha) * (n - i) ** (-alpha)
    return normalize(a)


def exponential_sequence(n, alpha):
    """a[i] = exp(-alpha * |i-(n-1)/2|/(n/2))."""
    x = np.abs(np.arange(n) - (n - 1) / 2.0) / (n / 2.0)
    a = np.exp(-alpha * x)
    return normalize(a)


def geometric_sequence(n, r):
    """Geometric sequence normalized."""
    a = r ** np.arange(n, dtype=np.float64)
    return normalize(a)


def sym_geometric_sequence(n, r):
    """Symmetric geometric: a[i] = r^|i-(n-1)/2|."""
    x = np.abs(np.arange(n) - (n - 1) / 2.0)
    a = r ** x
    return normalize(a)


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)


def optimize_pipeline(a_init, time_limit):
    """
    Full optimization pipeline: softmax L-BFGS -> projected gradient ->
    simulated annealing -> final softmax polish.
    """
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    # Stage 1: Softmax L-BFGS with increasing beta
    t1 = min(time_limit * 0.35, 120)
    if time_limit - (time.monotonic() - t0) > t1 + 20:
        a_opt, sc = optimize_softmax(a, beta_list=[20, 50, 100, 200],
                                     K=30, maxiter_per=150, time_limit=t1)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            a = a_opt
        print(f"      After softmax LBFGS: {best_score:.10f}")

    # Stage 2: Projected gradient
    rem = time_limit - (time.monotonic() - t0)
    t2 = min(rem * 0.3, 80)
    if rem > t2 + 15:
        a_opt, sc = projected_gradient(a, time_limit=t2, lr_init=0.005, K=30)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            a = a_opt
        print(f"      After projected grad: {best_score:.10f}")

    # Stage 3: Simulated annealing
    rem = time_limit - (time.monotonic() - t0)
    t3 = min(rem * 0.25, 50)
    if rem > t3 + 15:
        a_opt, sc = simulated_annealing(a, time_limit=t3)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            a = a_opt
        print(f"      After SA: {best_score:.10f}")

    # Stage 4: Final softmax polish with high beta
    rem = time_limit - (time.monotonic() - t0)
    if rem > 30:
        a_opt, sc = optimize_softmax(a, beta_list=[200, 500],
                                     K=15, maxiter_per=300,
                                     time_limit=min(rem, 120))
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
        print(f"      After final polish: {best_score:.10f}")

    return best_a, best_score


def propose_candidate():
    """
    Multi-phase search for the best non-negative sequence minimizing
    2*n*max(auto_conv)/(sum)^2. Combines parametric initialization,
    L-BFGS-B on smoothed peak objective, projected gradient, simulated
    annealing, dimension exploration, and final aggressive polish.
    """
    deadline = time.monotonic() + 1760.0
    rng = np.random.default_rng(42)

    best_a_global = None
    best_score_global = float('inf')

    def try_seq(a, label):
        nonlocal best_a_global, best_score_global
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        score = compute_score(a)
        if score < best_score_global:
            best_score_global = score
            best_a_global = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # === Generate diverse starting points ===
    print("=== Generating starting points ===")
    starts = []

    lengths = [384, 512, 640, 768, 896, 1024, 1280]
    key_lengths = [512, 768, 1024]

    for n in key_lengths:
        for gamma in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
            starts.append((cosine_power_sequence(n, gamma), f"cos_pow_g{gamma}_n{n}"))
        for gamma in [0.5, 1.0, 2.0, 4.0, 8.0]:
            starts.append((lorentzian_sequence(n, gamma), f"lor_g{gamma}_n{n}"))
        for alpha in [0.3, 0.5, 0.8, 1.2, 2.0]:
            starts.append((power_law_sequence(n, alpha), f"pow_a{alpha}_n{n}"))
        for alpha in [0.5, 1.0, 2.0, 4.0]:
            starts.append((exponential_sequence(n, alpha), f"exp_a{alpha}_n{n}"))
        for r in [0.999, 0.9995, 0.9999]:
            starts.append((sym_geometric_sequence(n, r), f"symgeo_r{r}_n{n}"))

    # Also try some for other lengths
    for n in [384, 640, 896, 1280]:
        starts.append((cosine_power_sequence(n, 1.0), f"cos_pow_n{n}"))
        starts.append((lorentzian_sequence(n, 2.0), f"lor_n{n}"))
        starts.append((power_law_sequence(n, 0.8), f"pow_n{n}"))

    # Parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            starts.append((parent, "parent"))
            p_score = compute_score(parent)
            print(f"  Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    # Evaluate all starts quickly
    print(f"  Evaluating {len(starts)} starting points...")
    start_scores = []
    for a, label in starts:
        sc = compute_score(a)
        start_scores.append((sc, a, label))
    start_scores.sort(key=lambda x: x[0])

    for i, (sc, a, label) in enumerate(start_scores[:10]):
        print(f"    Start #{i + 1} [{label}]: {sc:.8f}")
        try_seq(a, label)

    # Select diverse top starts (different families)
    selected = []
    selected_families = set()
    for sc, a, label in start_scores:
        if len(selected) >= 10:
            break
        # Extract family identifier
        if label == "parent":
            fam = "parent"
        else:
            # Get part before _n
            parts = label.split('_n')
            base = parts[0] if len(parts) > 1 else label
            # Get the family (first token)
            fam = base.split('_')[0]
        if fam not in selected_families:
            selected.append((a, label, sc))
            selected_families.add(fam)

    # Always include parent if available
    if "parent" not in selected_families:
        for sc, a, label in start_scores:
            if label == "parent":
                selected.append((a, label, sc))
                selected_families.add("parent")
                break

    # Also include top 2 unconditionally (even if same family)
    for sc, a, label in start_scores[:2]:
        if label not in [s[1] for s in selected]:
            selected.append((a, label, sc))

    # === Phase 1: Optimize diverse starts ===
    print(f"\n=== Phase 1: Optimize {len(selected)} diverse starts ===")
    avail_time = deadline - time.monotonic() - 400
    t_per = max(60, min(avail_time / len(selected), 200))

    for a_init, label, init_sc in selected:
        rem = deadline - time.monotonic() - 300
        if rem < 60:
            break
        t_alloc = min(t_per, rem)
        print(f"  [{label}]: n={len(a_init)}, init={init_sc:.6f}, t={t_alloc:.0f}s")
        a_opt, sc = optimize_pipeline(a_init, time_limit=t_alloc)
        try_seq(a_opt, label + "_opt")

    # === Phase 2: Dimension exploration ===
    print(f"\n=== Phase 2: Dimension exploration (best={best_score_global:.8f}) ===")
    if best_a_global is not None:
        best_n = len(best_a_global)
        target_ns = [384, 512, 640, 768, 896, 1024, 1280, 1536]
        for target_n in target_ns:
            if target_n == best_n:
                continue
            rem = deadline - time.monotonic() - 200
            if rem < 50:
                break
            a_interp = interpolate_sequence(best_a_global, target_n)
            t_alloc = min(rem * 0.35, 120)
            print(f"  Interpolating to n={target_n}, t={t_alloc:.0f}s")
            a_opt, sc = optimize_pipeline(a_interp, time_limit=t_alloc)
            try_seq(a_opt, f"dim_{target_n}")

    # === Phase 3: Aggressive final optimization ===
    print(f"\n=== Phase 3: Aggressive final optimization (best={best_score_global:.8f}) ===")
    if best_a_global is not None:
        # Long softmax with very high beta
        rem = deadline - time.monotonic() - 100
        if rem > 40:
            t1 = min(rem * 0.4, 150)
            print(f"  Long softmax (beta=500, K=10): t={t1:.0f}s")
            a_opt, sc = optimize_softmax(best_a_global, beta_list=[100, 300, 500, 1000],
                                         K=10, maxiter_per=400, time_limit=t1)
            try_seq(a_opt, "agg_softmax")

        # Long projected gradient
        rem = deadline - time.monotonic() - 80
        if rem > 30:
            t2 = min(rem * 0.4, 100)
            print(f"  Long projected grad: t={t2:.0f}s")
            a_opt, sc = projected_gradient(best_a_global, time_limit=t2,
                                           lr_init=0.002, K=20)
            try_seq(a_opt, "agg_PG")

        # Long simulated annealing
        rem = deadline - time.monotonic() - 60
        if rem > 30:
            t3 = min(rem * 0.4, 80)
            print(f"  Long SA: t={t3:.0f}s")
            a_opt, sc = simulated_annealing(best_a_global, time_limit=t3,
                                            T0=0.002, T1=0.00001)
            try_seq(a_opt, "agg_SA")

        # Perturbation restarts
        rem = deadline - time.monotonic() - 40
        for trial in range(8):
            if rem < 25:
                break
            n_best = len(best_a_global)
            noise_scale = 0.003 * (1 + trial * 0.5)
            noise = rng.standard_normal(n_best) * noise_scale
            a_pert = project_simplex(best_a_global + noise)
            t_alloc = min(rem, 30)
            a_opt, sc = optimize_pipeline(a_pert, time_limit=t_alloc)
            try_seq(a_opt, f"perturb_{trial}")
            rem = deadline - time.monotonic() - 30

    # === Final verification ===
    if best_a_global is None:
        best_a_global = cosine_power_sequence(768, 1.0)
        best_score_global = compute_score(best_a_global)

    final_score = float(evaluate_sequence([float(x) for x in best_a_global]))
    if np.isfinite(final_score):
        best_score_global = final_score

    print(f"\nFINAL SCORE: {best_score_global:.8f} (n={len(best_a_global)})")
    return [float(x) for x in best_a_global]
```
