1. At point x_t, compute the subgradient g_t of f at x_t.
2. Solve: min_{y in P} <g_t, y> (the "linear oracle")
3. Update: x_{t+1} = (1 - gamma_t) x_t + gamma_t * y_t

import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_direction(f, M, n, top_k=None):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for top_k peaks, g >= 0.
    If top_k is None, use all 2n-1 constraints.
    """
    # Use only top_k constraints for speed
    if top_k is not None:
        # Find the top_k peak positions of f*f
        ff = fast_convolve(f)
        peak_indices = np.argpartition(ff, -top_k)[-top_k:]
    else:
        peak_indices = np.arange(2 * n - 1)
    
    num_constraints = len(peak_indices)
    c = -np.ones(n)
    A_ub = np.zeros((num_constraints, n))
    for idx, k in enumerate(peak_indices):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # (f*g)[k] = sum_{j=js}^{je} f[j] * g[k-j]

def compute_lp_row(f, k, n, nh):
    """Compute the row of the LP matrix for constraint (f*g)[k] <= M."""
    row = np.zeros(nh)
    for j in range(n):
        kj = k - j
        if 0 <= kj < n:
            free_i = min(kj, n - 1 - kj)
            row[free_i] += f[j]
    return row

    else:

    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():

            kj = k - j

                fi = free_index(kj, n)
