def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    # Use indices where conv is largest as a proxy
    
    # Actually, for the LP direction, the relevant constraints are where f*g would be largest
    # If g is close to f, then f*g is close to f*f = conv, so the tightest constraints
    # are where conv is largest

    active_idx = sorted_idx[:top_k]

    for i, k in enumerate(active_idx):

def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with multiplicative updates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            cur_conv = fast_convolve(f)
            best_j_val = old_val
            best_j_obj = cur_conv.max()
            
            for mult in [0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 2.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved and passes > 3:
            break
    
    return best_f, best_obj

def sg_polish(f0, time_limit, lr=0.003, mom=0.94, top_k=50, temp=0.00005):
    """Subgradient polishing with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-16:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.999
        cur_temp *= 1.005
        
        if no_improve > 300:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0
    
    return best_f, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_cutting_plane(f, M, n, max_iter_cp=15, lp_time_limit=20):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M, g >= 0 using cutting plane."""
    conv = fast_convolve(f)
    sorted_idx = np.argsort(conv)[::-1]
    active_set = [int(sorted_idx[0])]
    
    g = None
    for _ in range(max_iter_cp):
        k_list = active_set
        A_ub = np.zeros((len(k_list), n))
        for i, k in enumerate(k_list):
            j_start = max(0, k - (n - 1))
            j_end = min(n - 1, k)
            A_ub[i, j_start:j_end + 1] = f[k - np.arange(j_start, j_end + 1)]
        
        b_ub = np.full(len(k_list), M)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        options = {'time_limit': lp_time_limit, 'presolve': True}
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options=options)
        if not res.success or res.x is None:
            return None
        
        g = res.x
        fg = np.convolve(f, g)
        violations = np.where(fg > M * 1.0001)[0]
        if len(violations) == 0:
            return g
        
        worst = int(violations[np.argmax(fg[violations])])
        if worst not in active_set:
            active_set.append(worst)
    
    return g

def coordinate_descent_fast(f0, time_limit):
    """Faster coordinate descent using incremental convolution updates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_convolve(f)
        cur_max = conv.max()
        
        # Process indices in order of their contribution to the max
        order = np.arange(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-8:
                continue
            
            best_j_val = old_val
            best_j_obj = cur_max
            
            for mult in [0.6, 0.8, 0.9, 1.1, 1.3, 1.6]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.002, mom=0.94, top_k=40, temp=0.00004):
    """Subgradient polishing with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-16:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.999
        cur_temp *= 1.004
        
        if no_improve > 250:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.0015)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0002)
            no_improve = 0
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic-like interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    if old_n < 2:
        return np.full(new_n, f[0] / new_n * old_n)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.6f}")
    
    # === Phase 1: Main LP-FW on parent ===
    print(f"\n=== Phase 1: LP-FW parent n={len(parent)} ===")
    t_remaining = deadline - time.monotonic()
    t1 = min(280, t_remaining * 0.18)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a_opt, obj = lp_fw_optimize(parent, time_limit=t1, verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")
    
    # === Phase 2: Systematic length search ===
    print(f"\n=== Phase 2: Length search (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    
    lengths_to_try = set()
    # Nearby
    for delta in [-48, -32, -16, 16, 32, 48, 96, 192, -96, -192, 384, -384]:
        nl = n_best + delta
        if 64 <= nl <= 2500:
            lengths_to_try.add(nl)
    # Interesting fixed lengths
    for nl in [128, 192, 256, 320, 384, 448, 512, 576, 640, 704, 768,
               832, 896, 960, 1024, 1152, 1280, 1408, 1536, 1792, 2048]:
        if 64 <= nl <= 2500:
            lengths_to_try.add(nl)
    
    # Sort by proximity to current best
    lengths_to_try = sorted(lengths_to_try - {n_best}, key=lambda x: abs(x - n_best))
    
    for target_n in lengths_to_try:
        rem = deadline - time.monotonic() - 250
        if rem < 15:
            break
        f_interp = interpolate_sequence(best_seq, target_n)
        t_alloc = min(70, rem * 0.12)
        if t_alloc < 8:
            continue
        print(f"  n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f} (n={target_n})")
    
    # === Phase 3: Diverse initializations on best length ===
    print(f"\n=== Phase 3: Diverse inits (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    
    inits = []
    inits.append(("best", best_seq.copy()))
    
    # Small noise
    noise = rng.standard_normal(n_best) * 0.003
    inits.append(("noise_s", project_simplex(best_seq + noise)))
    
    # Medium noise
    noise = rng.standard_normal(n_best) * 0.008
    inits.append(("noise_m", project_simplex(best_seq + noise)))
    
    # Edge boost
    f_e = best_seq.copy().astype(np.float64)
    q = n_best // 4
    f_e[:q] *= 1.25
    f_e[-q:] *= 1.25
    f_e[q:3*q] *= 0.82
    inits.append(("edge", project_simplex(f_e)))
    
    # Center boost
    f_c = best_seq.copy().astype(np.float64)
    f_c[q:3*q] *= 1.25
    f_c[:q] *= 0.82
    f_c[-q:] *= 0.82
    inits.append(("center", project_simplex(f_c)))
    
    # Cosine window
    x = np.linspace(0, np.pi, n_best)
    f_cos = np.cos(x / 2) ** 2
    f_cos = np.maximum(f_cos, 0.001)
    inits.append(("cosine", project_simplex(f_cos)))
    
    # Gaussian
    centers = np.arange(n_best)
    f_gauss = np.exp(-0.5 * ((centers - n_best/2) / (n_best/4))**2)
    inits.append(("gauss", project_simplex(f_gauss)))
    
    # Two-bump
    f_2b = np.zeros(n_best)
    f_2b[:n_best//2] = np.exp(-0.5 * ((np.arange(n_best//2) - n_best//4) / (n_best//8))**2)
    f_2b[n_best//2:] = np.exp(-0.5 * ((np.arange(n_best//2) - n_best//4) / (n_best//8))**2)
    f_2b = np.maximum(f_2b, 0.001)
    inits.append(("two_bump", project_simplex(f_2b)))
    
    for idx, (label, f_init) in enumerate(inits):
        rem = deadline - time.monotonic() - 120
        if rem < 12:
            break
        # Skip "best" since we already optimized it
        if label == "best":
            continue
        t_alloc = min(100, rem * 0.15)
        if t_alloc < 8:
            continue
        print(f"  [{label}] t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f}")
    
    # === Phase 4: Deep LP-FW on current best ===
    print(f"\n=== Phase 4: Deep LP-FW (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 80
    if rem > 20:
        t_deep = min(rem * 0.45, 150)
        print(f"  t={t_deep:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_deep, verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")
    
    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: CD (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 50
    if rem > 15:
        t_cd = min(rem * 0.5, 80)
        print(f"  t={t_cd:.0f}s")
        a_opt, _ = coordinate_descent_fast(best_seq, time_limit=t_cd)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")
    
    # === Phase 6: SG polish ===
    print(f"\n=== Phase 6: SG Polish (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 30
    if rem > 10:
        t_sg = min(rem * 0.6, 50)
        a_opt, _ = sg_polish(best_seq, time_limit=t_sg,
                             lr=0.0015, mom=0.94,
                             top_k=max(15, n_best // 6),
                             temp=0.00002)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")
    
    # === Phase 7: Final micro-search ===
    print(f"\n=== Phase 7: Micro-search (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    for trial in range(20):
        rem = deadline - time.monotonic()
        if rem < 2:
            break
        scale = 0.0001 + trial * 0.00008
        noise = rng.standard_normal(n_best) * scale
        f_pert = project_simplex(best_seq + noise)
        sc = float(evaluate_sequence(f_pert.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = f_pert.copy()
            print(f"  NEW BEST [pert{trial}]: {sc:.6f}")
            # Quick SG polish
            rem2 = deadline - time.monotonic()
            if rem2 > 5:
                a_opt, _ = sg_polish(f_pert, time_limit=min(rem2 * 0.3, 8),
                                     lr=0.0005, mom=0.95,
                                     top_k=max(10, n_best // 5),
                                     temp=0.00001)
                sc2 = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = a_opt.copy()
                    print(f"  NEW BEST [pert{trial}_sg]: {sc2:.6f}")
    
    # === Final LP-FW squeeze ===
    rem = deadline - time.monotonic()
    if rem > 8:
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem * 0.9, 50))
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  FINAL: {sc:.6f}")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
