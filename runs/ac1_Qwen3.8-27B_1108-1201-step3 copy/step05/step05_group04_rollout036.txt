Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using exact subgradient descent
with active-set tracking, backtracking line search, asymmetric
initializations, length exploration, and coordinate descent polishing.

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def conv_subgradient_peak(a, peak_indices, top_k=None):
    """
    Compute subgradient as uniform mixture of gradients at peak indices.
    grad_k[j] = 2*a[k-j] for valid k-j.
    """
    n = len(a)
    sg = np.zeros(n)
    peaks = peak_indices
    if top_k is not None and len(peaks) > top_k:
        peaks = peaks[:top_k]
    for k in peaks:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        j = np.arange(lo, hi + 1)
        sg[j] += 2.0 * a[k - j]
    sg /= len(peaks)
    return sg

def exact_subgrad_descent(a0, time_limit, lr_init=0.01, verbose=False):
    """
    Exact subgradient descent with active-set tracking and backtracking.
    """
    a = a0.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    iter_count = 0
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        conv = fast_convolve(a)
        cur_max = conv.max()
        
        # Find peak indices (within 1% of max for numerical stability)
        threshold = cur_max * 0.999
        peak_mask = conv >= threshold
        peak_indices = np.where(peak_mask)[0]
        
        if len(peak_indices) == 0:
            peak_indices = np.array([np.argmax(conv)])
        
        # Compute subgradient
        sg = conv_subgradient_peak(a, peak_indices, top_k=20)
        
        # Backtracking line search
        best_trial_obj = cur_max
        best_trial_a = a.copy()
        step_lr = lr
        for _ in range(20):
            a_trial = project_simplex(a - step_lr * sg)
            conv_trial = fast_convolve(a_trial)
            trial_obj = conv_trial.max()
            if trial_obj < best_trial_obj:
                best_trial_obj = trial_obj
                best_trial_a = a_trial.copy()
            if trial_obj < cur_max * 0.999:
                break
            step_lr *= 0.8
        
        a = best_trial_a
        lr = max(step_lr * 1.1, 1e-6)  # increase step if it worked
        
        conv = fast_convolve(a)
        cur_obj = conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and iter_count % 50 == 0:
                print(f"    Subgrad iter {iter_count}: obj={best_obj:.10f}, n_peaks={len(peak_indices)}")
        else:
            no_improve += 1
        
        if no_improve > 100:

            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            lr = lr_init
            no_improve = 0
    
    return best_a, best_obj

    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        cur_max = conv.max()

        # Use tighter tolerance for peak detection
        sg = compute_peak_subgradient(a, conv, tolerance=0.99999, top_k=10)

        # Shorter line search for fine tuning
        trial_lr = lr
        accepted = False
        for _ in range(15):
            a_new = project_simplex(a - trial_lr * sg)
            conv_new = fast_convolve(a_new)
            obj_new = conv_new.max()
            if obj_new < cur_max - 1e-6 * cur_max:
                a = a_new
                lr = min(trial_lr * 1.05, lr_init * 3)
                accepted = True
                break
            trial_lr *= 0.6

        if not accepted:
            a = project_simplex(a - trial_lr * 0.5 * sg)
            lr = max(trial_lr * 0.8, 1e-10)

        conv = fast_convolve(a)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            rng = np.random.default_rng(it * 23 + 11)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.0005)
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def coordinate_descent(a0, time_limit, verbose=False):
    """Coordinate descent with multiple scale factors per coordinate."""
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    factors = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[j] < 1e-13:
                continue

            old_val = a[j]
            cur_conv = fast_convolve(a)
            cur_peak = cur_conv.max()
            best_val = old_val
            best_val_obj = cur_peak

            for f in factors:
                new_val = old_val * f
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss

                obj = fast_convolve(a_trial).max()
                if obj < best_val_obj - 1e-15:
                    best_val_obj = obj
                    best_val = new_val

            if best_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_val) / denom
                    a[j] = best_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    ss = a.sum()
                    if ss > 1e-15:
                        a /= ss

                obj = fast_convolve(a).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True

        if not improved and passes > 1:
            break

    return best_a, best_obj


def pair_swap_search(a0, time_limit, n_pairs=100, verbose=False):
    """Try random pair swaps to escape local minima."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for _ in range(n_pairs):
        if time.monotonic() - t0 > time_limit:
            break

        # Pick random pair
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if a[i] < 1e-6 or a[j] < 1e-6:
            continue

        # Try transferring mass
        for frac in [0.1, 0.2, 0.3, 0.5, 0.7]:
            da = frac * a[i]
            a_trial = a.copy()
            a_trial[i] = a[i] - da
            a_trial[j] = a[j] + da
            if a_trial[i] < 0:
                continue
            ss = a_trial.sum()
            a_trial /= ss
            obj = fast_convolve(a_trial).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_trial.copy()
                a = a_trial.copy()
                break

    return best_a, best_obj


def make_asymmetric_init(n, rng, style='linear'):
    """Create diverse asymmetric initializations."""
    x = np.linspace(0, 1, n)
    if style == 'linear':
        p = rng.uniform(0.5, 2.5)
        f = x ** p
    elif style == 'reciprocal':
        p = rng.uniform(0.5, 2.0)
        f = (1 - x + 1e-8) ** p
    elif style == 'quadratic':
        c = rng.uniform(0.15, 0.85)
        sigma = rng.uniform(0.08, 0.25)
        asym = rng.uniform(-0.8, 0.8)
        f = np.exp(-0.5 * ((x - c) / sigma) ** 2) * np.exp(asym * x)
    elif style == 'bump_power':
        p = rng.uniform(0.4, 3.0)
        f = np.sin(np.pi * x ** p) ** 2
    elif style == 'double_peak':
        c1, c2 = sorted(rng.uniform(0.1, 0.9, size=2))
        w1 = rng.uniform(0.5, 1.5)
        s1 = rng.uniform(0.05, 0.2)
        s2 = rng.uniform(0.05, 0.2)
        f = w1 * np.exp(-((x - c1) / s1) ** 2) + np.exp(-((x - c2) / s2) ** 2)
    elif style == 'ramp_decay':
        f = np.exp(-x * rng.uniform(0.5, 3.0))
    elif style == 'smooth_random':
        f = np.zeros(n)
        for _ in range(rng.integers(3, 8)):
            c = rng.uniform(0.05, 0.95)
            s = rng.uniform(0.05, 0.3)
            amp = rng.uniform(0.3, 1.5)
            f += amp * np.exp(-0.5 * ((x - c) / s) ** 2)
    else:
        f = np.ones(n)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: diverse initialization, deep subgradient optimization,
    coordinate descent polishing, length exploration, and pair-swap refinement.
    """
    rng = np.random.default_rng(seed)
    total_budget = max(1.0, float(budget_s))
    deadline = time.monotonic() + total_budget - 15.0

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            old_score = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)}, improvement={old_score-score:.2e})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            if np.isfinite(p_score):
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_asymmetric_init(512, rng, 'linear')
        print("No parent found, using asymmetric init n=512")
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    # === Phase 1: Diverse quick search (250s) ===
    print(f"\n=== Phase 1: Diverse quick search (budget 250s) ===")
    t1_end = time.monotonic() + 250

    n_parent = len(parent)
    lengths = sorted(set([256, 384, 512, 640, 768, 1024, 1280]))
    if n_parent not in lengths:
        lengths.append(n_parent)
    lengths = [l for l in lengths if 128 <= l <= 1536]

    styles = ['linear', 'reciprocal', 'quadratic', 'bump_power', 'double_peak', 'ramp_decay', 'smooth_random']

    # Sort lengths by proximity to parent length
    lengths.sort(key=lambda x: abs(x - n_parent))
    lengths = lengths[:6]  # Limit to 6 lengths

    for n_target in lengths:
        for style in styles:
            rem = t1_end - time.monotonic()
            if rem < 8:
                break
            t_alloc = min(rem * 0.35, 35)
            if t_alloc < 4:
                break

            if n_target == n_parent and style == 'linear':
                init = parent.copy()
            else:
                init = make_asymmetric_init(n_target, rng, style)

            a_opt, _ = subgrad_optimize(init, time_limit=t_alloc, lr_init=0.006)
            try_seq(a_opt, f"p1_n{n_target}_{style}")

    # === Phase 2: Deep optimization on best (700s) ===
    print(f"\n=== Phase 2: Deep optimization (best={best_score:.10f}, budget 700s) ===")
    t2_end = time.monotonic() + 700

    if best_seq is not None:
        n_best = len(best_seq)

        # Run 1: Direct deep optimization
        rem = t2_end - time.monotonic()
        if rem > 30:
            t_alloc = min(rem * 0.4, 250)
            print(f"  Deep run 1, t={t_alloc:.0f}s")
            a_opt, _ = subgrad_optimize(best_seq, time_limit=t_alloc, lr_init=0.004)
            try_seq(a_opt, "p2_deep1")

        # Run 2: Slightly perturbed start
        rem = t2_end - time.monotonic()
        if rem > 20:
            t_alloc = min(rem * 0.35, 180)
            init2 = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)
            print(f"  Deep run 2 (perturbed), t={t_alloc:.0f}s")
            a_opt, _ = subgrad_optimize(init2, time_limit=t_alloc, lr_init=0.005)
            try_seq(a_opt, "p2_deep2")

        # Run 3: Different perturbation
        rem = t2_end - time.monotonic()
        if rem > 20:
            t_alloc = min(rem * 0.35, 150)
            init3 = project_simplex(best_seq + rng.standard_normal(n_best) * 0.001)
            print(f"  Deep run 3 (small perturbation), t={t_alloc:.0f}s")
            a_opt, _ = subgrad_optimize(init3, time_limit=t_alloc, lr_init=0.003)
            try_seq(a_opt, "p2_deep3")

        # Run 4: Mix with a different structure
        rem = t2_end - time.monotonic()
        if rem > 20:
            t_alloc = min(rem * 0.5, 120)
            other_init = make_asymmetric_init(n_best, rng, 'quadratic')
            init4 = project_simplex(0.7 * best_seq + 0.3 * other_init)
            print(f"  Deep run 4 (mixed init), t={t_alloc:.0f}s")
            a_opt, _ = subgrad_optimize(init4, time_limit=t_alloc, lr_init=0.005)
            try_seq(a_opt, "p2_deep4")

    # === Phase 3: Coordinate descent (400s) ===
    print(f"\n=== Phase 3: Coordinate descent (best={best_score:.10f}, budget 400s) ===")
    t3_end = time.monotonic() + 400

    if best_seq is not None:
        rem = t3_end - time.monotonic()
        if rem > 30:
            t_cd = min(rem * 0.6, 200)
            print(f"  CD round 1, t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "p3_cd1")

            # Pair swap after CD
            rem = t3_end - time.monotonic()
            if rem > 15 and best_seq is not None:
                a_ps, _ = pair_swap_search(best_seq, time_limit=min(rem * 0.4, 60), n_pairs=200)
                try_seq(a_ps, "p3_pairswap")
                if a_ps is not None:
                    a_opt, _ = coordinate_descent(a_ps, time_limit=min(rem * 0.4, 60))
                    try_seq(a_opt, "p3_cd2")

            # Another CD round
            rem = t3_end - time.monotonic()
            if rem > 20 and best_seq is not None:
                t_cd2 = min(rem * 0.7, 100)
                a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd2)
                try_seq(a_opt, "p3_cd3")

    # === Phase 4: Length exploration (300s) ===
    print(f"\n=== Phase 4: Length exploration (best={best_score:.10f}, budget 300s) ===")
    t4_end = time.monotonic() + 300

    if best_seq is not None:
        n_best = len(best_seq)
        target_lengths = sorted(set([
            n_best - 128, n_best - 64, n_best + 64, n_best + 128,
            n_best - 32, n_best + 32,
            512, 640, 768, 896, 1024, 1152
        ]))
        target_lengths = [t for t in target_lengths if 128 <= t <= 2048 and t != n_best]
        target_lengths.sort(key=lambda x: abs(x - n_best))
        target_lengths = target_lengths[:5]

        for tl in target_lengths:
            rem = t4_end - time.monotonic()
            if rem < 20:
                break
            f_interp = interpolate_sequence(best_seq, tl)
            t_alloc = min(rem * 0.5, 80)
            print(f"  Length n={tl}, t={t_alloc:.0f}s")
            a_opt, _ = subgrad_optimize(f_interp, time_limit=t_alloc, lr_init=0.005)
            try_seq(a_opt, f"p4_n{tl}")

            # Quick coordinate descent
            rem = t4_end - time.monotonic()
            if rem > 10 and best_seq is not None:
                a_cd, _ = coordinate_descent(a_opt, time_limit=min(rem * 0.4, 40))
                try_seq(a_cd, f"p4_cd_n{tl}")

    # === Phase 5: Final fine polishing (remaining time) ===
    print(f"\n=== Phase 5: Final fine polishing (best={best_score:.10f}) ===")

    if best_seq is not None:
        n_best = len(best_seq)

        # Fine subgradient
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            t_fine = min(rem * 0.4, 80)
            print(f"  Fine subgrad, t={t_fine:.0f}s")
            a_opt, _ = subgrad_optimize_fine(best_seq, time_limit=t_fine, lr_init=0.0008)
            try_seq(a_opt, "p5_fine_subgrad")

        # Fine coordinate descent
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t_cd = min(rem * 0.4, 60)
            print(f"  Fine CD, t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "p5_fine_cd")

        # Tiny perturbations + quick optimize
        rem = deadline - time.monotonic() - 5
        for pi in range(4):
            rem = deadline - time.monotonic() - 5
            if rem < 8:
                break
            noise = rng.standard_normal(n_best) * 0.0008
            f_pert = project_simplex(best_seq + noise)
            t_alloc = min(rem * 0.4, 25)
            a_opt, _ = subgrad_optimize_fine(f_pert, time_limit=t_alloc, lr_init=0.0005)
            try_seq(a_opt, f"p5_pert_{pi}")

        # Final pair swap
        rem = deadline - time.monotonic() - 3
        if rem > 5:
            a_ps, _ = pair_swap_search(best_seq, time_limit=min(rem * 0.6, 20), n_pairs=300)
            try_seq(a_ps, "p5_final_pairswap")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    print(f"{'='*60}")
    return [float(x) for x in best_seq]
