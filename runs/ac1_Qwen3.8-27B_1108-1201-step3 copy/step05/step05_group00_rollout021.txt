(1) Smooth surrogate optimization via Adam with adaptive temperature,
(2) LP-Frank-Wolfe refinement with adaptive relaxation,
(3) Minimax multiplicative weights with inner projected gradient,

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(f, tau):

    Compute gradient of F(a) = log(sum_k exp((a*a)[k]/tau)).

        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        # Adaptive temperature decay
        tau *= 0.9997
        if tau < 1e-7:
            tau = 1e-7

        # Adaptive learning rate
        if no_improve > 100:
            lr *= 0.85
        if no_improve > 400:
            # Perturb from best and restart
            rng = np.random.default_rng(t * 13 + 7)
            noise_scale = 0.002 + 0.003 * min(no_improve / 1000.0, 3.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = lr_init * 0.7
            tau = 0.05 * fast_convolve(f).max()
            if verbose:
                obj_score = 2 * n * best_obj
                print(f"    Adam restart at iter {t}: obj={obj_score:.8f}")

        if verbose and t % 500 == 0:
            obj_score = 2 * n * best_obj
            print(f"    Adam iter {t}: obj={obj_score:.8f}, tau={tau:.2e}, lr={lr:.5f}")

    return best_f, 2 * n * best_obj


def lp_fw_optimize(f0, time_limit, verbose=False):
    """
    LP-based Frank-Wolfe with adaptive relaxation.
    At each step: solve LP to find vertex direction, do line search.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Solve LP: max sum(g) s.t. (f*g)[k] <= M*m_exp, g >= 0
        g = _solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.2
            if m_exp > 6.0:
                rng = np.random.default_rng(it * 17 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.1
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 20:
                m_exp = min(m_exp * 1.1, 5.0)
            if no_improve > 80:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue
        else:
            no_improve = 0

        # Line search: golden section
        best_alpha = 0.0
        best_val = M

        # Coarse search
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine search (bisection)
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.06)
            hi = min(1.0, best_alpha + 0.06)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = np.maximum(f + mid * (g - f), 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = np.maximum(f + best_alpha * (g - f), 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={2*n*best_obj:.8f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

        if no_improve > 100:
            break

    return best_f, 2 * n * best_obj


def _solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def minimax_multiplicative_weights(f0, time_limit, verbose=False):
    """
    Minimax approach: alternate between inner QP (projected gradient)
    and outer multiplicative weight update.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    # Initialize p (outer variable)
    p = np.ones(2 * n - 1) / (2 * n - 1)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    lr_p = 0.15
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Inner: solve min a^T Q(p) a via projected gradient (few iterations)
        a_inner = f.copy()
        inner_iters = min(30, max(10, 2000 // (it + 1)))
        inner_lr = 2.0 / (n * 1.1)

        for _ in range(inner_iters):
            # Gradient of a^T Q(p) a = 2 * (Q(p) * a)
            # (Q(p)*a)[j] = sum_l p[j+l] * a[l]
            p_rev = p[::-1]
            total_len = n + len(p) - 1
            fft_size = 1 << (total_len - 1).bit_length()
            A = np.fft.rfft(a_inner, fft_size)
            P = np.fft.rfft(p_rev, fft_size)
            conv_r = np.fft.irfft(A * P, fft_size)
            grad = conv_r[n - 1:2 * n - 1][::-1]  # This is (Q*p_a)[j]
            grad = 2.0 * grad

            a_inner = a_inner - inner_lr * grad
            a_inner = project_simplex(a_inner)

        f = a_inner
        conv = fast_convolve(f)
        M = conv.max()

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            if verbose and it % 50 == 0:
                print(f"    Minimax iter {it}: obj={2*n*best_obj:.8f}")

        # Outer: multiplicative weight update on p
        # Weight positions proportional to their convolution value
        scaled = conv / (0.02 * M + 1e-15)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        p = (1 - lr_p) * p + lr_p * w
        p = p / p.sum()

        # Adaptive lr_p
        if it % 100 == 0:
            lr_p = max(0.03, lr_p * 0.95)

    return best_f, 2 * n * best_obj


def peak_targeted_cd(f0, time_limit, verbose=False):
    """
    Peak-targeted coordinate descent: identifies peak-contributing pairs
    and redistributes mass to reduce the peak.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        # Get top peaks
        top_k = min(5, 2 * n - 1)
        peak_indices = np.argsort(conv)[-top_k:]

        for pk in peak_indices:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[pk - idx]
            j_max = idx[np.argmax(products)]
            j_other = pk - j_max

            if j_max == j_other:
                # Diagonal element: just reduce f[j_max]
                old_val = f[j_max]
                if old_val < 1e-10:
                    continue
                for factor in [0.5, 0.7, 0.85]:
                    trial = f.copy()
                    trial[j_max] *= factor
                    diff = old_val - trial[j_max]
                    # Redistribute to non-peak-contributing elements
                    contrib_mask = np.ones(n, dtype=bool)
                    # Elements contributing to top peaks
                    for pk2 in peak_indices[:3]:
                        js2 = max(0, int(pk2) - n + 1)
                        je2 = min(n - 1, int(pk2))
                        contrib_mask[js2:je2 + 1] = False
                    if contrib_mask.sum() > 0:
                        trial[contrib_mask] += diff * f[contrib_mask] / f[contrib_mask].sum()
                    trial = np.maximum(trial, 0)
                    trial /= trial.sum()
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break
            else:
                # Off-diagonal: reduce the product
                old_jm = f[j_max]
                old_jo = f[j_other]
                for scale_jm in [0.6, 0.8]:
                    new_jm = old_jm * scale_jm
                    diff = old_jm - new_jm
                    trial = f.copy()
                    trial[j_max] = new_jm
                    # Redistribute to j_other's "opposite" (non-contributing)
                    mask = np.ones(n, dtype=bool)
                    mask[j_max] = False
                    # Also avoid adding to elements that contribute to the same peak
                    mask[j_other] = False
                    if mask.sum() > 0:
                        trial[mask] += diff * f[mask] / f[mask].sum()
                    trial = np.maximum(trial, 0)
                    trial /= trial.sum()
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break

        if not improved:
            no_improve += 1
            if no_improve > 15:
                # Try a larger perturbation
                rng = np.random.default_rng(int(time.monotonic() * 1000))
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                no_improve = 0
        else:
            no_improve = 0

    return best_f, 2 * n * best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_asymmetric_gaussian(n, center_frac=0.4, sigma_frac=0.3):
    """Asymmetric Gaussian initialization."""
    x = np.arange(n) - n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.5):
    """Flat-topped bell initialization."""
    x = np.arange(n) - n / 2.0
    half = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Diverse initialization (20s)
    Phase 2: Adam smooth optimization (850s)
    Phase 3: LP-FW refinement (400s)
    Phase 4: Multi-length exploration (300s)
    Phase 5: Fine-tuning (180s + buffer)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
        return score

    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_asymmetric_gaussian(512, 0.35, 0.28)
        try_seq(parent, "default_init")

    # Try diverse initializations
    inits = []
    for n_try in [512, 768]:
        inits.append(make_asymmetric_gaussian(n_try, 0.3, 0.25))
        inits.append(make_asymmetric_gaussian(n_try, 0.5, 0.2))
        inits.append(make_bell(n_try, 0.5))
        inits.append(np.ones(n_try) / n_try)

    # Asymmetric: exponential decay
    for n_try in [512, 768]:
        a_exp = np.exp(-np.arange(n_try) / (n_try * 0.4))
        a_exp /= a_exp.sum()
        inits.append(a_exp)

    for i, a_init in enumerate(inits):
        if time.monotonic() > deadline - 1400:
            break
        sc = try_seq(a_init, f"init_{i}")

    # Pick best for main optimization
    work_seq = best_seq if best_seq is not None else parent
    work_n = len(work_seq)
    print(f"  Working with n={work_n}, score={best_score:.6f}")

    # === Phase 2: Adam smooth optimization (main workhorse) ===
    print(f"\n=== Phase 2: Adam Smooth Optimization (n={work_n}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem, 850)
    if t2 > 60:
        print(f"  Time allocated: {t2:.0f}s")
        # Run Adam on the best sequence
        a_opt, obj = adam_smooth_optimize(work_seq, time_limit=t2 * 0.6,
                                          lr_init=0.004, verbose=True)
        try_seq(a_opt, "adam_main")

        # Run again with different parameters from the result
        if time.monotonic() < deadline - 400:
            rem2 = deadline - time.monotonic() - 400
            t2b = min(rem2, 250)
            a_opt2, obj2 = adam_smooth_optimize(best_seq, time_limit=t2b,
                                                tau_init=0.03 * fast_convolve(best_seq).max(),
                                                lr_init=0.002, verbose=True)
            try_seq(a_opt2, "adam_fine")

    # === Phase 3: LP-FW refinement ===
    print(f"\n=== Phase 3: LP-FW Refinement ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 350
        if rem > 60:
            t3 = min(rem, 400)
            print(f"  Time allocated: {t3:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t3, verbose=True)
            try_seq(a_opt, "lpfw")

            # Try LP-FW from a perturbed start
            if time.monotonic() < deadline - 250:
                rem2 = deadline - time.monotonic() - 250
                t3b = min(rem2, 150)
                perturbed = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.005)
                a_opt2, obj2 = lp_fw_optimize(perturbed, time_limit=t3b, verbose=False)
                try_seq(a_opt2, "lpfw_pert")

    # === Phase 4: Minimax multiplicative weights ===
    print(f"\n=== Phase 4: Minimax MW ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 60:
            t4 = min(rem, 200)
            print(f"  Time allocated: {t4:.0f}s")
            a_opt, obj = minimax_multiplicative_weights(best_seq, time_limit=t4, verbose=True)
            try_seq(a_opt, "minimax")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length Exploration ===")
    if best_seq is not None:
        base = best_seq
        for target_n in [1024, 1280, 640]:
            rem = deadline - time.monotonic() - 100
            if rem < 50:
                break
            f_interp = interpolate_sequence(base, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")

            # Quick Adam on interpolated
            t5 = min(rem * 0.4, 100)
            if t5 > 20:
                a_opt, obj = adam_smooth_optimize(f_interp, time_limit=t5, lr_init=0.003)
                try_seq(a_opt, f"adam_n{target_n}")

    # === Phase 6: Peak-targeted coordinate descent ===
    print(f"\n=== Phase 6: Peak-Targeted CD ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 30:
            t6 = min(rem, 120)
            print(f"  Time allocated: {t6:.0f}s")
            a_opt, obj = peak_targeted_cd(best_seq, time_limit=t6, verbose=True)
            try_seq(a_opt, "peak_cd")

    # === Phase 7: Final Adam polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t7 = min(rem, 60)
            a_opt, obj = adam_smooth_optimize(
                best_seq, time_limit=t7,
                tau_init=0.02 * fast_convolve(best_seq).max(),
                lr_init=0.001, verbose=False)
            try_seq(a_opt, "final_polish")

    # === Phase 8: Final LP-FW squeeze ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t8 = min(rem, 30)
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t8, verbose=False)
            try_seq(a_opt, "final_lpfw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
