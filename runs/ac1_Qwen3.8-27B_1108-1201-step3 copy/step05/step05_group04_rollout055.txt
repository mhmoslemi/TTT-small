"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding,
corrected subgradient Adam, mirror descent, and multi-start exploration.
Key improvements: correct subgradient (factor of 2), LP-based direction search,

import numpy as np
import time
import scipy.optimize as opt


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def correct_subgradient(f, w_full):
    """
    Compute the correct subgradient of max(conv(f,f)) with respect to f.
    w_full: soft-max weights over the full convolution (length 2n-1).
    Returns: (n,) array.
    """
    n = len(f)
    # The correct subgradient is 2 * correlate(w_full, f, 'valid')
    # because conv(k) involves pairs (f[i], f[i+k]) and each f[j] appears
    # in both the "forward" and "backward" roles.
    return 2.0 * np.correlate(w_full, f, 'valid')


def smooth_max(conv, temp):
    """Log-sum-exp approximation of max."""
    shifted = conv / temp
    shifted -= shifted.max()
    return temp * np.log(np.sum(np.exp(shifted)))


def get_smooth_max_grad(conv, temp):
    """Gradient of log-sum-exp w.r.t. conv elements (i.e., softmax weights)."""
    shifted = conv / temp
    shifted -= shifted.max()
    w = np.exp(shifted)
    w /= w.sum()
    return w


def solve_lp_direction(f, M_override=None):
    """
    Solve LP to find direction g that minimizes max_k conv(f,g)[k].
    LP: minimize t
        s.t. conv(f,g)[k] <= t for all k
             sum(g) = 1, g >= 0
    Returns: (g, t_opt) or (None, None) on failure.
    """
    n = len(f)
    # Build constraint matrix
    # Variables: g_0..g_{n-1}, t (total n+1)
    # Constraints: for each lag, sum of coefficients * g - t <= 0
    
    n_cons = 2 * n - 1
    A = np.zeros((n_cons, n + 1))
    b = np.zeros(n_cons)
    
    # Positive lags k = 0..n-1
    for k in range(n):
        row = k
        A[row, n] = -1.0  # -t
        for i in range(n - k):
            A[row, i + k] += f[i]
    
    # Negative lags k = 1..n-1 (lag -k)
    for k in range(1, n):
        row = n - 1 + k
        A[row, n] = -1.0
        for i in range(k, n):
            A[row, i - k] += f[i]
    
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g_i >= 0, t >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    res = opt.linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
    
    if not res.success:
        return None, None
    
    g = res.x[:n]
    t_opt = res.x[n]
    
    # Ensure g is valid
    g = np.maximum(g, 0)
    s = g.sum()
    if s < 1e-15:
        return None, None
    g /= s
    
    return g, t_opt


def adam_optimize(f0, time_limit, lr=0.005, beta1=0.9, beta2=0.999,
                  eps=1e-8, temp_init=0.005, temp_decay=0.9995,
                  top_k=None, use_correct_sg=True):
    """Adam optimizer with simplex projection and correct subgradients."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    m = np.zeros(n)
    v = np.zeros(n)
    t_step = 0
    temp = temp_init

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr

    while time.monotonic() - t0 < time_limit:
        t_step += 1
        conv = fast_convolve(f)

        if use_correct_sg:
            w = get_smooth_max_grad(conv, temp)
            if top_k is not None and top_k < len(conv):
                idx = np.argpartition(w, -top_k)[-top_k:]
                w_masked = np.zeros(len(conv))
                w_masked[idx] = w[idx]
                w_masked /= w_masked.sum()
                w = w_masked
            sg = correct_subgradient(f, w)
        else:
            w = get_smooth_max_grad(conv, temp)
            sg = np.correlate(w, f, 'valid')

        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** t_step)
        v_hat = v / (1 - beta2 ** t_step)

        f_new = f - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f_new)

        temp = max(temp * temp_decay, 1e-7)
        cur_lr = lr * (0.99995 ** t_step)
        cur_lr = max(cur_lr, lr * 0.1)

        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            rng = np.random.default_rng(t_step * 31 + 7)
            noise_scale = 0.005 + 0.003 * min(no_improve // 100, 4)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            t_step = 0
            temp = max(temp_init * 0.8, 0.003)
            cur_lr = lr
            no_improve = 0

    return best_f, best_obj


def mirror_descent_optimize(f0, time_limit, lr_init=0.01, lr_decay=0.999,
                            temp_init=0.005, temp_decay=0.9995,
                            top_k=None):
    """Mirror descent with correct subgradients."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    temp = temp_init
    cur_lr = lr_init

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    t_step = 0

    while time.monotonic() - t0 < time_limit:
        t_step += 1
        conv = fast_convolve(f)
        w = get_smooth_max_grad(conv, temp)
        if top_k is not None and top_k < len(conv):
            idx = np.argpartition(w, -top_k)[-top_k:]
            w_masked = np.zeros(len(conv))
            w_masked[idx] = w[idx]
            w_masked /= w_masked.sum()
            w = w_masked
        sg = correct_subgradient(f, w)

        log_update = -cur_lr * sg
        log_update -= log_update.max()
        f_new = f * np.exp(log_update)
        fs = f_new.sum()
        if fs < 1e-15:
            f_new = np.ones(n) / n
        else:
            f_new = f_new / fs
        f = f_new

        temp = max(temp * temp_decay, 1e-7)
        cur_lr *= lr_decay
        cur_lr = max(cur_lr, lr_init * 0.05)

        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 150:
            rng = np.random.default_rng(t_step * 13 + 5)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            temp = max(temp_init * 0.7, 0.003)
            cur_lr = lr_init
            no_improve = 0

    return best_f, best_obj


def lp_based_optimize(f0, time_limit, max_iters=5000):
    """LP-based direction finding optimization."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_lp_improve = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve LP for direction
        g, t_opt = solve_lp_direction(f)
        if g is None:
            break

        # Try different step sizes
        best_alpha = 0
        best_trial_obj = best_obj

        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            f_trial = (1 - alpha) * f + alpha * g
            f_trial = project_simplex(f_trial)
            conv_trial = fast_convolve(f_trial)
            obj_trial = conv_trial.max()
            if obj_trial < best_trial_obj - 1e-15:
                best_trial_obj = obj_trial
                best_alpha = alpha

        if best_alpha > 0:
            f = (1 - best_alpha) * f + best_alpha * g
            f = project_simplex(f)
            conv = fast_convolve(f)
            cur_obj = conv.max()
            if cur_obj < best_obj - 1e-15:
                best_obj = cur_obj
                best_f = f.copy()
                no_lp_improve = 0
            else:
                no_lp_improve += 1
        else:
            no_lp_improve += 1

        if no_lp_improve > 50:
            # Perturb and restart
            rng = np.random.default_rng(it * 17 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
            no_lp_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Coordinate descent with local search."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            multipliers = [0.3, 0.5, 0.7, 0.8, 1.1, 1.3, 1.5, 2.0, 0.1]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 1:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian_init(n, center=0.5, sigma=0.15):
    """Create a Gaussian-like initialization."""
    x = np.linspace(0, 1, n)
    f = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_symmetric_init(n, shape='bump'):
    """Create a symmetric initialization."""
    if shape == 'bump':
        x = np.linspace(0, 1, n)
        f = np.sin(np.pi * x) ** 2
    elif shape == 'triangle':
        x = np.linspace(0, 1, n)
        f = 1 - np.abs(2 * x - 1)
    elif shape == 'raised_cosine':
        x = np.linspace(0, 1, n)
        f = 0.5 * (1 + np.cos(2 * np.pi * x))
    else:
        f = np.ones(n)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_smooth_random_init(n, rng, n_components=5):
    """Create a smooth random initialization."""
    f = np.zeros(n)
    x = np.linspace(0, 1, n)
    for _ in range(n_components):
        c = rng.uniform(0.1, 0.9)
        s = rng.uniform(0.05, 0.25)
        a = rng.uniform(0.5, 2.0)
        f += a * np.exp(-0.5 * ((x - c) / s) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()

    base = best_seq if best_seq is not None else parent

    # === Phase 2: Corrected Adam optimization ===
    print(f"\n=== Phase 2: Corrected Adam (best={best_score:.8f}) ===")
    t_phase2_end = time.monotonic() + 400
    n_base = len(base)

    configs = [
        {'lr': 0.008, 'temp_init': 0.004, 'temp_decay': 0.9993, 'top_k': max(20, n_base // 10)},
        {'lr': 0.005, 'temp_init': 0.006, 'temp_decay': 0.9995, 'top_k': max(30, n_base // 8)},
        {'lr': 0.010, 'temp_init': 0.003, 'temp_decay': 0.9992, 'top_k': max(15, n_base // 15)},
        {'lr': 0.006, 'temp_init': 0.008, 'temp_decay': 0.9996, 'top_k': None},
        {'lr': 0.004, 'temp_init': 0.005, 'temp_decay': 0.9994, 'top_k': max(25, n_base // 12)},
    ]

    for ci, cfg in enumerate(configs):
        rem = t_phase2_end - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(rem * 0.6, 100)

        print(f"  Adam cfg {ci} (lr={cfg['lr']}), t={t_alloc:.0f}s")
        a_opt, _ = adam_optimize(base, time_limit=t_alloc, **cfg)
        try_seq(a_opt, f"p2_adam_{ci}")

        # Mirror descent variant
        rem = t_phase2_end - time.monotonic()
        if rem < 20:
            break
        t_md = min(rem * 0.4, 50)
        a_opt, _ = mirror_descent_optimize(base, time_limit=t_md,
                                           lr_init=cfg['lr'] * 1.5,
                                           temp_init=cfg['temp_init'],
                                           temp_decay=cfg['temp_decay'],
                                           top_k=cfg['top_k'])
        try_seq(a_opt, f"p2_mirror_{ci}")

    base = best_seq if best_seq is not None else parent

    # === Phase 3: Length exploration ===
    print(f"\n=== Phase 3: Length exploration (best={best_score:.8f}) ===")
    t_phase3_end = time.monotonic() + 350

    n_best = len(base)
    target_lengths = [256, 384, 512, 640, 768, 896, 1024, 1280, 1536]
    target_lengths = [t for t in target_lengths if t != n_best and 128 <= t <= 2048]
    target_lengths.sort(key=lambda x: abs(x - n_best))
    target_lengths = target_lengths[:5]

    for target_n in target_lengths:
        rem = t_phase3_end - time.monotonic()
        if rem < 50:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.5, 80)

        print(f"  Length n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = adam_optimize(f_interp, time_limit=t_alloc,
                                 lr=0.006, temp_init=0.005,
                                 temp_decay=0.9994, top_k=max(20, target_n // 10))
        try_seq(a_opt, f"p3_n{target_n}")

        rem = t_phase3_end - time.monotonic()
        if rem > 20:
            a_opt, _ = lp_based_optimize(f_interp, time_limit=min(rem, 40))
            try_seq(a_opt, f"p3_lp_n{target_n}")

    base = best_seq if best_seq is not None else parent

    # === Phase 4: Deep optimization from best ===
    print(f"\n=== Phase 4: Deep optimization (best={best_score:.8f}) ===")
    t_phase4_end = time.monotonic() + 300
    n_best = len(base)

    # Multiple Adam runs from perturbed best
    for si in range(4):
        rem = t_phase4_end - time.monotonic()
        if rem < 25:
            break
        t_alloc = min(rem * 0.5, 70)
        noise = rng.standard_normal(n_best) * (0.003 + 0.002 * si)
        f_pert = project_simplex(base + noise)
        print(f"  Deep Adam pert {si}, t={t_alloc:.0f}s")
        a_opt, _ = adam_optimize(f_pert, time_limit=t_alloc,
                                 lr=0.003, temp_init=0.003,
                                 temp_decay=0.9997, top_k=max(15, n_best // 16))
        try_seq(a_opt, f"p4_pert_{si}")

    # LP from best
    rem = t_phase4_end - time.monotonic()
    if rem > 30:
        print(f"  LP from best, t={min(rem, 100):.0f}s")
        a_opt, _ = lp_based_optimize(base, time_limit=min(rem, 100))
        try_seq(a_opt, "p4_lp_best")

    base = best_seq if best_seq is not None else parent

    # === Phase 5: Coordinate descent polish ===
    print(f"\n=== Phase 5: Coordinate descent (best={best_score:.8f}) ===")
    t_phase5_end = time.monotonic() + 150
    n_best = len(base)

    rem = t_phase5_end - time.monotonic()
    if rem > 20:
        t_cd = min(rem * 0.7, 100)
        print(f"  CD, t={t_cd:.0f}s")
        a_opt, _ = coordinate_descent(base, time_limit=t_cd)
        try_seq(a_opt, "p5_cd")

        rem = t_phase5_end - time.monotonic()
        if rem > 15 and best_seq is not None:
            perturbed = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.002)
            a_opt, _ = coordinate_descent(perturbed, time_limit=min(rem, 30))
            try_seq(a_opt, "p5_cd_pert")

    base = best_seq if best_seq is not None else parent

    # === Phase 6: Final fine-tuning ===
    print(f"\n=== Phase 6: Final fine-tuning (best={best_score:.8f}) ===")
    n_best = len(base)

    rem = deadline - time.monotonic() - 10
    if rem > 15:
        t_alloc = min(rem * 0.5, 50)
        print(f"  Fine Adam, t={t_alloc:.0f}s")
        a_opt, _ = adam_optimize(base, time_limit=t_alloc,
                                 lr=0.002, temp_init=0.002,
                                 temp_decay=0.9998, top_k=max(10, n_best // 25))
        try_seq(a_opt, "p6_fine")

        # Very fine LP
        rem = deadline - time.monotonic() - 10
        if rem > 20:
            print(f"  Fine LP, t={min(rem, 60):.0f}s")
            a_opt, _ = lp_based_optimize(base, time_limit=min(rem, 60))
            try_seq(a_opt, "p6_lp_fine")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
