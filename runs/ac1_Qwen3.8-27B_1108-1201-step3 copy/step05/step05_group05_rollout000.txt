LP-based iterative refinement for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses the exact LP direction (maximize sum(g) s.t. conv(f,g)<=M, g>=0) instead of
subgradient approximation, combined with coordinate descent polish and
multi-length exploration.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def build_lp_pattern(n):
    """Precompute sparsity pattern for the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp_direction(f, M, rows, cols):
    """Solve LP: maximize sum(g) subject to conv(f,g) <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    
    # Fill in the data values

    for idx in range(len(rows)):
        k = rows[idx]
        j = cols[idx]
        i = k - j
        data[idx] = f[i]
    
    A_ub = coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        return None

                f = project_simplex(best_f + rng.standard_normal(n) * noise_scale, target_sum=1.0)
                t = 0.05
                no_improve = 0
            elif no_improve > 80:
                # Try larger step with blend toward uniform
                f_blend = 0.9 * f + 0.1 * (np.ones(n) / n)
                f_blend = f_blend / f_blend.sum()
                f = f_blend
                t = 0.03
        
        iter_count += 1
        if verbose and iter_count % 50 == 0:
            print(f"    LP iter {iter_count}: obj={best_obj:.10f}, t={t:.4f}, no_imp={no_improve} ({elapsed:.0f}s)")
    
    return best_f, best_obj


def coordinate_descent(a0, time_limit):
    """Coordinate descent with local search per coordinate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    pass_count = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            multipliers = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.3, 1.8]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved and pass_count > 3:
            break

    return best_a, best_obj


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        mag = 1.0
        A[k] = mag * np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def tapered_init(n, rng):
    """Tapered initialization with small boundaries."""
    x = np.linspace(0, 1, n)
    # Sine taper
    a = np.sin(np.pi * x)
    # Add some randomness
    a = a * (1.0 + 0.1 * rng.standard_normal(n))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def polyphase_init(n, rng):
    """Initialize using polyphase-like structure."""
    a = np.ones(n) / n
    # Create a pattern with reduced boundary effects
    boundary_width = max(3, n // 32)
    for i in range(boundary_width):
        taper = (i + 1) / (boundary_width + 1)
        a[i] *= taper * taper
        a[n - 1 - i] *= taper * taper
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolated_init(parent, n, rng):
    """Interpolate parent to new length with random perturbation."""
    old_n = len(parent)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, n)
    f_interp = np.interp(new_x, old_x, parent)
    noise = rng.standard_normal(n) * 0.005
    f_interp = f_interp + noise
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return f_interp / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase LP-based optimization with diverse initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # === Phase 1: Generate diverse initializations ===
    print("\n=== Phase 1: Initializations ===")
    
    target_lengths = []
    if parent is not None:
        target_lengths.append(len(parent))
    for n_try in [384, 512, 640, 768, 896, 1024, 1280]:
        if n_try not in target_lengths:
            target_lengths.append(n_try)
    
    # Remove very large lengths if parent is small
    target_lengths = [n for n in target_lengths if n <= 1536]
    target_lengths.sort()
    
    inits = {}
    for n in target_lengths[:6]:
        best_init = None
        best_init_obj = float('inf')
        
        methods = [
            ("flat_spec", flat_spectrum_init(n, rng)),
            ("tapered", tapered_init(n, rng)),
            ("polyphase", polyphase_init(n, rng)),
        ]
        
        if parent is not None:
            if n == len(parent):
                methods.append(("parent", parent.copy()))
            else:
                methods.append(("interp", interpolated_init(parent, n, rng)))
        
        # Also try a "concave" shape
        x = np.linspace(0, 1, n)
        methods.append(("concave", (1 - (2*x - 1)**4) / (1 - (2*x - 1)**4).sum()))
        
        for name, init in methods:
            try:
                init = np.maximum(init, 0)
                ss = init.sum()
                if ss < 1e-10:
                    continue
                init = init / ss
                obj = fast_convolve(init).max()
                if obj < best_init_obj:
                    best_init_obj = obj
                    best_init = init.copy()
            except Exception:
                pass
        
        if best_init is not None:
            inits[n] = (best_init, best_init_obj)
            print(f"  n={n}: best init obj={best_init_obj:.8f}")

    # === Phase 2: LP-based optimization ===
    print(f"\n=== Phase 2: LP Optimization (best={best_score:.6f}) ===")
    
    sorted_inits = sorted(inits.items(), key=lambda x: x[1][1])
    
    # Allocate time proportional to quality
    total_lp_time = min(1100, max(200, (deadline - time.monotonic()) * 0.6))
    n_optimize = min(4, len(sorted_inits))
    
    for rank, (n_len, (init, init_obj)) in enumerate(sorted_inits[:n_optimize]):
        rem = deadline - time.monotonic() - 400
        if rem < 30:
            break
        
        # Allocate more time to better initializations
        time_fracs = [0.45, 0.25, 0.18, 0.12]
        time_alloc = min(total_lp_time * time_fracs[rank], rem)
        
        print(f"  LP optimize n={n_len} (init_obj={init_obj:.8f}), t={time_alloc:.0f}s")
        
        a_opt, obj = lp_optimize(init, time_limit=time_alloc, verbose=(rank == 0))
        try_seq(a_opt, f"lp_n{n_len}_r{rank}")
        
        # Quick coordinate descent after LP
        rem = deadline - time.monotonic() - 350
        if rem > 20 and best_score < 1.51:
            t_cd = min(rem * 0.3, 80)
            a_cd, obj_cd = coordinate_descent(a_opt, time_limit=t_cd)
            try_seq(a_cd, f"cd_after_lp_n{n_len}")

    # === Phase 3: Refined LP from best ===
    print(f"\n=== Phase 3: Refined LP (best={best_score:.6f}) ===")
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        for attempt in range(3):
            rem = deadline - time.monotonic() - 200
            if rem < 30:
                break
            t_alloc = min(rem / (4 - attempt), 150)
            
            if attempt == 0:
                f_init = best_seq.copy()
            elif attempt == 1:
                noise = rng.standard_normal(n_best) * 0.003
                f_init = project_simplex(best_seq + noise, target_sum=1.0)
            else:
                # Blend with reversed
                f_init = 0.7 * best_seq + 0.3 * best_seq[::-1]
                f_init = project_simplex(f_init, target_sum=1.0)
            
            a_opt, obj = lp_optimize(f_init, time_limit=t_alloc)
            try_seq(a_opt, f"lp_refine_{attempt}")
            
            # CD polish
            rem = deadline - time.monotonic() - 180
            if rem > 15:
                a_cd, obj_cd = coordinate_descent(a_opt, time_limit=min(rem, 50))
                try_seq(a_cd, f"cd_refine_{attempt}")

    # === Phase 4: Try longer sequences ===
    print(f"\n=== Phase 4: Longer Sequences (best={best_score:.6f}) ===")
    
    if best_seq is not None:
        n_best = len(best_seq)
        for target_n in [int(n_best * 1.25), int(n_best * 1.5), int(n_best * 2)]:
            if target_n <= n_best or target_n > 2048:
                continue
            rem = deadline - time.monotonic() - 100
            if rem < 60:
                continue
            
            f_interp = interpolated_init(best_seq, target_n, rng)
            t_alloc = min(rem * 0.4, 100)
            print(f"  Longer n={target_n}, t={t_alloc:.0f}s")
            
            a_opt, obj = lp_optimize(f_interp, time_limit=t_alloc)
            try_seq(a_opt, f"long_n{target_n}")
            
            rem = deadline - time.monotonic() - 80
            if rem > 20:
                a_cd, obj_cd = coordinate_descent(a_opt, time_limit=min(rem, 40))
                try_seq(a_cd, f"cd_long_n{target_n}")

    # === Phase 5: Final coordinate descent polish ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.6f}) ===")
    
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t_final = min(rem * 0.7, 120)
            print(f"  Final CD, t={t_final:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_final)
            try_seq(a_opt, "final_cd")
            
            # One more LP round if time allows
            rem = deadline - time.monotonic() - 5
            if rem > 30:
                a_opt, obj = lp_optimize(best_seq, time_limit=min(rem, 60))
                try_seq(a_opt, "final_lp")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
