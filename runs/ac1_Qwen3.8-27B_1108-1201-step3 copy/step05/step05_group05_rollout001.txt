This is related to our problem because if we could find a sequence whose auto-convolution is perfectly flat (all equal except at the peak), we'd achieve the theoretical minimum.

For non-negative sequences, perfect flatness is impossible, but we can get close. The question is how to systematically search for good sequences.

**Another idea: Start from known good structures**

Some well-known sequences with good autocorrelation properties:
- Barker sequences (length 7, 11, 13) - but these are ±1 sequences
- m-sequences
- Zadoff-Chu sequences

For non-negative sequences, we can try:
- Raised cosine shapes
- Windowed sinc functions
- Solutions to the "minimum peak power" problem

**Practical approach: Focus on improving the subgradient method**

I think the most likely path to improvement is:
1. Better gradient computation (use the exact formula, not numerical)
2. Smarter line search (backtracking)
3. Better momentum scheduling
4. More diverse restarts
5. Longer local refinement phase

Let me also think about the **exact subgradient vs. smoothed gradient** tradeoff:
- Smoothed gradient (softmax): Gives a direction that considers all peaks, which can help escape from "sticking" to one peak
- Exact subgradient: Gives the steepest descent direction for the current max, which can be more aggressive

I'll use a hybrid: compute the smoothed gradient but with a relatively sharp temperature (so it focuses on the top few peaks).

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def compute_smooth_gradient(a, conv, tau, top_k=50):
    """Compute smooth gradient of log-sum-exp approximation to max(conv)."""
    n = len(a)
    # Find top-k peaks
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    
    # Compute softmax weights
    peak_vals = conv[peaks]
    max_val = peak_vals.max()
    exp_vals = np.exp(tau * (peak_vals - max_val))
    weights = exp_vals / exp_vals.sum()
    
    if not np.isfinite(weights).all() or weights.sum() < 1e-30:
        # Fallback: uniform weight on the single max
        grad = np.zeros(n)
        k = np.argmax(conv)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        grad[js:je+1] = 2.0 * a[k-je:k-js+1][::-1]
        return grad
    
    # Compute gradient as weighted sum of individual gradients
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        grad[js:je+1] += weights[i] * 2.0 * a[k-je:k-js+1][::-1]
    
    return grad

        if no_improve > 200:

                    obj_new = obj_ext
                else:

        a = a_new

            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:

            a = project_simplex(best_a + rng_r.standard_normal(n) * noise_scale, target_sum=1.0)

            no_improve = 0
            tau = tau_init
            momentum = 0.8

def coordinate_refinement(a0, time_limit, verbose=False):
    """Coordinate descent with binary search on each coordinate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    pass_count = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        
        # Randomize coordinate order for diversity
        np.random.shuffle = None  # Don't want to mess with global state
        order = np.random.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-14:
                continue
            
            # Try multiple candidate values
            old_remaining = 1.0 - old_val
            if old_remaining < 1e-10:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            best_j_a = a.copy()
            
            # Geometric search around current value
            candidates = [old_val * m for m in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]]
            # Also try moving mass to/from this coordinate
            candidates.extend([max(0, old_val + 0.01 * old_remaining) for _ in [1]])
            candidates.extend([max(0, old_val - 0.01 * old_remaining) for _ in [1]])
            
            for new_val in candidates:
                if new_val < 0 or new_val > 0.9999:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= (1.0 - new_val) / old_remaining
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
                    best_j_a = a_trial.copy()
            
            if best_j_val != old_val:
                a = best_j_a
                if best_obj > best_j_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved and pass_count > 3:
            break
    
    return best_a, best_obj

def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        mag = 1.0
        A[k] = mag * np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def random_init(n, rng):
    """Random non-negative initialization."""
    a = rng.exponential(1.0, n)
    return a / a.sum()

def cosine_init(n):
    """Raised cosine window."""
    x = np.arange(n) / (n - 1)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    return a / a.sum()

def gaussian_init(n):
    """Gaussian window."""
    x = np.linspace(-3, 3, n)
    a = np.exp(-x**2 / 2)
    return a / a.sum()

def tukey_init(n, r=0.5):
    """Tukey (tapered cosine) window."""
    x = np.arange(n)
    a = np.ones(n)
    m = int(r * n / 2)
    if m > 0:
        for i in range(m):
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / n - 1 + r) / r))
            a[n-1-i] = a[i]
    return a / a.sum()

def haard_init(n, rng):
    """Haar-like random initialization with smooth transitions."""
    # Start with a random binary pattern and smooth it
    bits = (rng.random(n) > 0.5).astype(float)
    # Smooth with a Gaussian kernel
    sigma = max(1, n // 32)
    kernel = np.exp(-np.arange(-sigma, sigma+1)**2 / (2*sigma**2))
    kernel /= kernel.sum()
    a = np.convolve(bits, kernel, mode='same')
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
