def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def cross_corr(w, a):
    """Cross-correlation for gradient computation."""
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:n]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgrad_direction(a, k, top_k=10, temperature=0.01):

    scaled -= scaled.max()
    w = np.exp(scaled)

import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr(w, a):
    """Cross-correlation: result[i] = sum_k w[k] * a[k-i] for valid indices."""
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:n]


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def soft_max_obj_grad(z, T, n):
    """
    Compute the soft-max smoothed objective and its gradient w.r.t. z.
    """
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp

    w = exp_Tb / exp_Tb.sum()
    grad_a = 2.0 * n * cross_corr(w, a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)

    return obj, grad_z


def lbgfs_optimize(z0, T, n, time_limit, maxiter=3000):
    """L-BFGS in log-space with time limit enforcement."""
    t0 = time.monotonic()
    stop_flag = [False]

    def func(z):
        if time.monotonic() - t0 > time_limit:
            stop_flag[0] = True
            return 1e15
        obj, _ = soft_max_obj_grad(z, T, n)
        return obj

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            stop_flag[0] = True
            return np.zeros_like(z)
        _, g = soft_max_obj_grad(z, T, n)
        return g

    try:
        res = minimize(
            fun=func,
            x0=z0,
            jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 30, 'disp': False}
        )

    except Exception:

def heavy_ball_subgrad(a0, time_limit=60.0, top_k=30, mom=0.92, lr_init=0.003,
                       temp_init=0.01, temp_growth=1.003, lr_decay=0.9995,
                       max_no_improve=300):
    """Heavy ball method with top-k subgradient and adaptive parameters."""
    a = a0.copy()
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute gradient
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-13:
                continue
            k_pos = int(peaks[idx])
            js = max(0, k_pos - n + 1)
            je = min(n - 1, k_pos)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] += 2.0 * w[idx] * a[k_pos - idx_arr]

        # Normalize gradient
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn

        # Heavy ball update
        v = mom * v - lr * grad
        a_new = project_simplex(a + v)

        b_new = fast_convolve(a_new)
        obj = b_new.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                # Perturbed restart
                noise = np.random.default_rng(int(time.monotonic() * 1e9) % (2**31)).standard_normal(n)
                a = project_simplex(best_a + noise * 0.002)
                v = np.zeros(n)
                lr = max(lr * 0.5, 0.0003)
                no_improve = 0

        lr *= lr_decay
        temp *= temp_growth

    return best_a, best_obj


def active_set_line_search(a0, time_limit=30.0, alpha_init=0.01, shrink=0.7):
    """
    Active set method: focus on the top peak, compute exact subgradient,
    perform backtracking line search.
    """
    a = a0.copy()
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    alpha = alpha_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()
        # Find peaks very close to max (active set)
        active_tol = max_b * 0.001 + 1e-12
        active = np.where(b >= max_b - active_tol)[0]

        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Compute subgradient (average over active set)
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] += a[k - idx_arr]
        grad *= 2.0 / len(active)

        # Project gradient onto tangent space of simplex
        mean_grad = grad.mean()
        grad_tangent = grad - mean_grad

        # Backtracking line search
        improved = False
        for _ in range(20):
            a_trial = project_simplex(a - alpha * grad_tangent)
            obj = fast_convolve(a_trial).max()
            if obj < max_b - 1e-14:
                a = a_trial
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a.copy()
                no_improve = 0
                improved = True
                break
            alpha *= shrink

        if not improved:
            no_improve += 1
            alpha *= 0.5
            if no_improve > 100:
                break

    return best_a, best_obj


def pair_redistribute(a0, n_peaks=10, n_iters=500, time_limit=30.0):
    """Targeted pair redistribution to reduce top peaks."""
    a = a0.copy()
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_inner = b.copy()
        b_inner[0] = -1
        b_inner[-1] = -1
        k_p = min(n_peaks, n - 1)
        peaks = np.argpartition(b_inner, -k_p)[-k_p:]
        peaks = sorted(peaks.tolist(), key=lambda x: -b[x])

        improved = False
        for k in peaks:
            if time.monotonic() - t0 > time_limit:
                break
            if b[k] < best_obj * 0.92:
                break

            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            contributions = a[idx_arr] * a[k - idx_arr]

            if contributions.max() < 1e-15:
                continue

            max_i = idx_arr[np.argmax(contributions)]
            max_j = k - max_i

            for target in [max_i, max_j]:
                old_val = a[target]
                if old_val < 1e-12:
                    continue

                for frac in [0.85, 0.75, 0.6]:
                    new_val = old_val * frac
                    diff = old_val - new_val
                    if diff < 1e-15:
                        continue

                    # Redistribute to "safe" positions (not contributing to top peaks)
                    safe = np.ones(n, dtype=bool)
                    for pk in peaks[:5]:
                        s_js = max(0, pk - n + 1)
                        s_je = min(n - 1, pk)
                        safe[s_js:s_je + 1] = False
                    safe[target] = False

                    safe_idx = np.where(safe)[0]
                    if len(safe_idx) == 0:
                        safe = np.ones(n, dtype=bool)
                        safe[target] = False
                        safe_idx = np.where(safe)[0]

                    safe_vals = a[safe_idx]
                    if safe_vals.sum() < 1e-12:
                        continue

                    a_trial = a.copy()
                    a_trial[target] = new_val
                    a_trial[safe_idx] = safe_vals + diff * safe_vals / safe_vals.sum()
                    a_trial = a_trial / a_trial.sum()

                    new_obj = fast_convolve(a_trial).max()
                    if new_obj < best_obj - 1e-14:
                        a = a_trial
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break

        if not improved:
            break

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining subgradient, L-BFGS, active-set,
    and pair redistribution methods with multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512

    # === Phase 1: Heavy-ball subgradient on parent (30% time) ===
    print("\n=== Phase 1: Heavy-ball subgradient ===")
    n_parent = len(parent)
    t_budget = min(300.0, (deadline - time.monotonic()) * 0.25)
    if t_budget > 30:
        a_hb, obj_hb = heavy_ball_subgrad(parent, time_limit=t_budget, top_k=30,
                                          mom=0.93, lr_init=0.004, temp_init=0.008,
                                          temp_growth=1.002, lr_decay=0.9997)
        try_seq(a_hb, "heavy_ball_p1")

        # Second pass with different parameters
        rem = deadline - time.monotonic()
        if rem > 100:
            t_budget2 = min(rem * 0.15, 120.0)
            base = best_seq if best_seq is not None else parent
            a_hb2, _ = heavy_ball_subgrad(base, time_limit=t_budget2, top_k=50,
                                          mom=0.95, lr_init=0.002, temp_init=0.003,
                                          temp_growth=1.001, lr_decay=0.9998)
            try_seq(a_hb2, "heavy_ball_p1b")

    # === Phase 2: L-BFGS temperature annealing (30% time) ===
    print("\n=== Phase 2: L-BFGS temperature annealing ===")
    base = best_seq if best_seq is not None else parent
    n_base = len(base)
    z_curr = z_from_a(base)

    # Aggressive temperature schedule: start low, go high
    temps = [3, 8, 20, 50, 150, 400]
    for i, T in enumerate(temps):
        rem = deadline - time.monotonic()
        remaining_temps = len(temps) - i
        if rem < remaining_temps * 15:
            break
        t_alloc = min(rem / remaining_temps * 0.8, 120.0)
        if t_alloc < 10:
            break
        print(f"  L-BFGS n={n_base}, T={T}, t={t_alloc:.0f}s")
        z_curr = lbgfs_optimize(z_curr, T, n_base, t_alloc, maxiter=2500)
        a_curr = softmax_from_z(z_curr)
        try_seq(a_curr, f"lbgfs_T{T}")

    # === Phase 3: Active-set line search (10% time) ===
    print(f"\n=== Phase 3: Active-set line search (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 40:
            t_as = min(rem * 0.1, 60.0)
            a_as, _ = active_set_line_search(best_seq, time_limit=t_as,
                                             alpha_init=0.005, shrink=0.6)
            try_seq(a_as, "active_set")

    # === Phase 4: Pair redistribution (10% time) ===
    print(f"\n=== Phase 4: Pair redistribution (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t_pr = min(rem * 0.1, 45.0)
            a_pr, _ = pair_redistribute(best_seq, n_peaks=15, n_iters=800,
                                        time_limit=t_pr)
            try_seq(a_pr, "pair_redist")

    # === Phase 5: Multi-length exploration (15% time) ===
    print(f"\n=== Phase 5: Multi-length exploration (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent

    # Try nearby lengths (slightly smaller and larger)
    candidate_lengths = []
    n_best = len(base)
    for factor in [0.75, 0.85, 1.15, 1.3]:
        cl = int(n_best * factor)
        cl = max(128, min(4096, cl))
        if cl != n_best and cl not in candidate_lengths:
            candidate_lengths.append(cl)

    # Also try some standard lengths
    for std_n in [640, 768, 1024, 1280, 1536]:
        if std_n != n_best and std_n not in candidate_lengths:
            candidate_lengths.append(std_n)

    # Sort by proximity to current best length
    candidate_lengths.sort(key=lambda x: abs(x - n_best))
    candidate_lengths = candidate_lengths[:4]  # Try at most 4

    for target_n in candidate_lengths:
        rem = deadline - time.monotonic()
        if rem < 80:
            break

        f_interp = interpolate_sequence(base, target_n)

        # Quick subgradient
        t_sg = min(rem * 0.2, 40.0)
        if t_sg > 10:
            a_sg, _ = heavy_ball_subgrad(f_interp, time_limit=t_sg, top_k=20,
                                         mom=0.92, lr_init=0.003, temp_init=0.01)
            try_seq(a_sg, f"sg_n{target_n}")

        # L-BFGS refinement
        rem = deadline - time.monotonic()
        if rem > 30:
            z_new = z_from_a(best_seq if (best_seq is not None and len(best_seq) == target_n) else a_sg)
            t_lbgfs = min(rem * 0.3, 60.0)
            if t_lbgfs > 10:
                z_new = lbgfs_optimize(z_new, 30, target_n, t_lbgfs, maxiter=2000)
                a_new = softmax_from_z(z_new)
                try_seq(a_new, f"lbgfs_n{target_n}_T30")

                rem = deadline - time.monotonic()
                if rem > 20:
                    t_final = min(rem * 0.2, 30.0)
                    if t_final > 8:
                        z_new = lbgfs_optimize(z_new, 150, target_n, t_final, maxiter=2000)
                        a_new = softmax_from_z(z_new)
                        try_seq(a_new, f"lbgfs_n{target_n}_T150")

    # === Phase 6: Final deep refinement (remaining time) ===
    print(f"\n=== Phase 6: Final deep refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_final = len(best_seq)

        # Deep subgradient with very small learning rate
        rem = deadline - time.monotonic()
        if rem > 30:
            t_deep = min(rem * 0.25, 90.0)
            a_deep, _ = heavy_ball_subgrad(best_seq, time_limit=t_deep, top_k=40,
                                           mom=0.96, lr_init=0.001, temp_init=0.002,
                                           temp_growth=1.001, lr_decay=0.9999,
                                           max_no_improve=500)
            try_seq(a_deep, "deep_subgrad")

        # Final L-BFGS at very high T
        rem = deadline - time.monotonic()
        if rem > 20:
            z0 = z_from_a(best_seq)
            t_alloc = min(rem * 0.3, 60.0)
            if t_alloc > 8:
                print(f"  Final L-BFGS T=500, t={t_alloc:.0f}s")
                z_fin = lbgfs_optimize(z0, 500, n_final, t_alloc, maxiter=3000)
                a_fin = softmax_from_z(z_fin)
                try_seq(a_fin, "final_T500")

        # Final pair redistribution
        rem = deadline - time.monotonic()
        if rem > 15:
            t_pr = min(rem * 0.4, 30.0)
            a_final, _ = pair_redistribute(best_seq, n_peaks=20, n_iters=2000,
                                           time_limit=t_pr)
            try_seq(a_final, "final_pair")

        # Perturbation + re-optimize
        rem = deadline - time.monotonic()
        if rem > 15:
            for attempt in range(3):
                rem = deadline - time.monotonic()
                if rem < 8:
                    break
                noise = rng.standard_normal(n_final) * 0.003
                a_pert = project_simplex(best_seq + noise)
                t_alloc = min(rem * 0.4, 15.0)
                if t_alloc < 5:
                    break
                a_p, _ = heavy_ball_subgrad(a_pert, time_limit=t_alloc, top_k=30,
                                            mom=0.93, lr_init=0.002, temp_init=0.005)
                try_seq(a_p, f"perturb_{attempt}")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
