def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using iterated LP-based direction finding,
soft-max subgradient polishing with momentum, coordinate descent, and multi-length exploration.
The key innovation is using the LP relaxation to find optimal descent directions that

import numpy as np
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_soft_subgradient(f, T, top_k=80):
    """Compute soft-max weighted subgradient of max(f*f)."""
    n = len(f)
    b = fast_convolve(f)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    F = rfft(f, fft_size)
    grad = 2.0 * irfft(np.conj(W) * F, fft_size)[:n]
    return grad, b.max()


def solve_lp_direction(f, top_k=100):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for top peaks, g >= 0."""
    n = len(f)
    b = fast_convolve(f)
    M = b.max()
    
    peak_idx = np.argpartition(b, -top_k)[-top_k:]
    peak_idx = sorted(set(peak_idx.tolist()))
    
    A_rows = []
    for k in peak_idx:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        row = np.zeros(n)
        g_idx = np.arange(js, je + 1)
        f_idx = k - g_idx
        row[g_idx] = f[f_idx]
        A_rows.append(row)
    
    A = np.array(A_rows)
    rhs = np.full(len(peak_idx), M)
    
    g = cp.Variable(n, nonneg=True)
    obj = cp.Maximize(cp.sum(g))
    constraints = [A @ g <= rhs]
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=5000, eps=1e-6)
        if g.value is not None:
            gv = np.maximum(g.value, 0.0)
            if gv.sum() > 1e-10:
                return gv
    except Exception:
        pass
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=5000, abstol=1e-7)
        if g.value is not None:
            gv = np.maximum(g.value, 0.0)
            if gv.sum() > 1e-10:

        pass

def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def soft_max_obj_and_grad(z, T, n):
    """Compute soft-max smoothed objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    
    w = exp_Tb / exp_Tb.sum()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * n * irfft(np.conj(W) * A, fft_size)[:n]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    
    return obj, grad_z

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        rem = deadline - time.monotonic() - 300
        if rem > 20:
            t_alloc = min(rem * 0.5, 120.0)
            print(f"  Time budget: {t_alloc:.0f}s")
            f_cd, obj_cd = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(f_cd, "cd_phase3")
    
    # === Phase 4: L-BFGS in log-space with temperature annealing ===
    print(f"\n=== Phase 4: L-BFGS log-space ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 250
        if rem > 30:
            z0 = z_from_a(best_seq)
            z_curr = z0
            temps = [20, 50, 100, 300]
            per_T = min(rem * 0.15, 60.0)
            
            for T in temps:
                rem = deadline - time.monotonic() - 200
                if rem < 15:
                    break
                t_alloc = min(per_T, rem * 0.5)
                print(f"  L-BFGS n={n_best}, T={T}, t={t_alloc:.0f}s")
                z_curr, _ = lbfgs_logspace(z_curr, T, n_best, time_limit=t_alloc, maxiter=2000)
                a_curr = softmax_from_z(z_curr)
                try_seq(a_curr, f"lbfgs_T{T}")
    
    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length exploration ===")
    base = best_seq if best_seq is not None else parent
    
    for target_n in [1024, 1536, 2048]:
        rem = deadline - time.monotonic() - 150
        if rem < 30:
            break
        
        f_interp = interpolate_sequence(base, target_n)
        try_seq(f_interp, f"interp_n{target_n}")
        
        # LP refine at new length
        rem = deadline - time.monotonic() - 120
        if rem > 30:
            t_alloc = min(rem * 0.4, 80.0)
            f_lp_new, _ = lp_refine(f_interp, time_limit=t_alloc, top_k=80)
            try_seq(f_lp_new, f"lp_n{target_n}")
        
        # SG polish at new length
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_alloc = min(rem * 0.3, 50.0)
            best_at_n = best_seq if best_seq is not None and len(best_seq) == target_n else (f_lp_new if 'f_lp_new' in dir() else f_interp)
            f_sg_new, _ = subgradient_polish(best_at_n, time_limit=t_alloc,
                                             T_init=10.0, lr_init=0.01)
            try_seq(f_sg_new, f"sg_n{target_n}")
    
    # === Phase 6: Final intensive polish on best ===
    print(f"\n=== Phase 6: Final polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        
        # LP + SG + CD cycle
        for cycle in range(3):
            rem = deadline - time.monotonic() - 30
            if rem < 20:
                break
            
            # LP
            t_lp = min(rem * 0.2, 40.0)
            if t_lp > 10:
                print(f"  Cycle {cycle}: LP, t={t_lp:.0f}s")
                f_lp_c, _ = lp_refine(best_seq, time_limit=t_lp, top_k=120)
                try_seq(f_lp_c, f"final_lp_{cycle}")
            
            # SG
            rem = deadline - time.monotonic() - 25
            if rem < 10:
                break
            t_sg = min(rem * 0.25, 50.0)
            if t_sg > 10:
                print(f"  Cycle {cycle}: SG, t={t_sg:.0f}s")
                f_sg_c, _ = subgradient_polish(best_seq, time_limit=t_sg,
                                              T_init=50.0, lr_init=0.005, mom=0.95)
                try_seq(f_sg_c, f"final_sg_{cycle}")
            
            # CD
            rem = deadline - time.monotonic() - 20
            if rem < 10:
                break
            t_cd = min(rem * 0.3, 40.0)
            if t_cd > 8:
                print(f"  Cycle {cycle}: CD, t={t_cd:.0f}s")
                f_cd_c, _ = coordinate_descent(best_seq, time_limit=t_cd)
                try_seq(f_cd_c, f"final_cd_{cycle}")
    
    # === Phase 7: Last-ditch high-T L-BFGS ===
    print(f"\n=== Phase 7: Final L-BFGS ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            n_best = len(best_seq)
            z0 = z_from_a(best_seq)
            t_alloc = min(rem * 0.7, 30.0)
            z_final, _ = lbfgs_logspace(z0, 500, n_best, time_limit=t_alloc, maxiter=2000)
            a_final = softmax_from_z(z_final)
            try_seq(a_final, "final_T500")
    
    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
