Key innovations: FISTA with adaptive smoothing, BB step-size subgradient,

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v_shift = v - v.max()
    e = np.exp(v_shift)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def get_smooth_grad(a, beta):
    """
    Compute the smooth approximation of max(conv(a,a)) and its gradient.
    Uses log-sum-exp with temperature 1/beta.
    Returns (smooth_obj, grad) where grad is w.r.t. a.
    """
    conv = fast_convolve(a)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    
    obj = conv_max + np.log(exp_shifted.sum()) / beta
    
    # grad_a[i] = 2 * sum_m w[m] * a[m+i] (cross-correlation of w and a)
    n = len(a)
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R
    
    return obj, grad_a

    if s < 1e-15:

    else:
        f = f / s
    n = len(f)
    
    if betas is None:
        betas = [20.0, 50.0, 100.0, 200.0, 400.0, 800.0, 1600.0]
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    
    for beta in betas:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        
        sub_time = remaining / (len(betas) - betas.index(beta))
        
        # Lipschitz constant estimate: L = 4 * max(a) * beta roughly
        # More carefully: the gradient involves cross-correlation, so
        # L <= 2 * beta * ||a||_inf * 2 (rough bound)

        # Let's use a practical bound: L = 4 * beta / n (for normalized a)
        L = 4.0 * beta / n * 2  # conservative
        
        # FISTA parameters
        t_k = 1.0
        x = f.copy()
        y = f.copy()
        step = 0.0
        it = 0

import time


def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    v_shift = v - v.max()
    e = np.exp(v_shift)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def compute_subgradient_topk(a, top_k):
    """Compute subgradient using top-k peaks of the convolution."""
    n = len(a)
    conv = fast_convolve(a)
    k_actual = min(top_k, len(conv))
    
    # Get top-k peak indices
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    
    # Weight by softmax for smoothness
    if pv.max() > 0:
        scaled = pv / max(pv.max() * 0.01, 1e-10)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
    else:
        w = np.ones(len(peaks)) / len(peaks)
    
    # Compute subgradient: sg[j] = 2 * sum_i w[i] * a[peaks[i] - j]
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * a[k - idx]
    
    return sg, conv.max()

    best_f = f.copy()

    v = np.zeros(n)  # momentum
    mom = 0.8
    prev_f = f.copy()
    prev_grad = None
    
    # Initial step size
    sg, _ = compute_subgradient_topk(f, top_k)
    sg_norm = np.linalg.norm(sg)
    if sg_norm < 1e-15:
        step = 0.01
    else:
        step = 0.01 / sg_norm
    
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:

            gy = sg - prev_grad
            ys_norm_sq = np.dot(ys, ys)

            best_f = f_new.copy()
            no_improve = 0
            prev_f = f.copy()
            f = f_new

def fista_optimize(f0, time_limit, verbose=False):
    """
    FISTA (accelerated proximal gradient) on log-sum-exp smooth approximation.
    Adaptive beta scheduling.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    
    # Beta schedule: start smooth, end sharp
    total_betas = [30, 60, 120, 240, 480, 960]
    
    for beta in total_betas:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        sub_time = remaining / (len(total_betas) - total_betas.index(beta))
        
        # Estimate Lipschitz constant
        # The gradient is 2 * cross_corr(w, a), and ||grad|| <= 2 * ||w||_1 * ||a||_inf * 2
        # For the smooth max: L = beta * 2 * ||a||_inf * 2 + 2 * ||cross_corr_op||
        # Practical: L ~ 4 * beta (since a is normalized, ||a||_inf <= 1)
        L = 4.0 * beta * 1.5
        step = 1.0 / L
        
        # FISTA
        x = f.copy()
        y = f.copy()
        t_k = 1.0
        
        it = 0
        while time.monotonic() - t0 < t0 + sub_time:
            it += 1
            
            conv_y = fast_convolve(y)
            conv_max = conv_y.max()
            shifted = beta * (conv_y - conv_max)
            exp_shifted = np.exp(shifted)
            w = exp_shifted / exp_shifted.sum()
            
            # Gradient: 2 * cross_corr(w, y)
            N = 1 << (3 * n - 2).bit_length()
            Y = np.fft.rfft(y, N)
            W = np.fft.rfft(w, N)
            R = np.fft.irfft(W * np.conj(Y), N)[:n]
            grad = 2.0 * R
            
            # Gradient step + projection
            z = y - step * grad
            x_new = project_simplex(z)
            
            # FISTA momentum
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            y = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
            y = np.maximum(y, 0.0)
            y_sum = y.sum()
            if y_sum > 1e-15:
                y = y / y_sum
            t_k = t_new
            
            # Check
            obj_new = fast_convolve(x_new).max()
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = x_new.copy()
                if verbose and it % 500 == 0:
                    print(f"    FISTA beta={beta}, iter={it}: obj={best_obj:.10f}")
            
            x = x_new
        
        f = best_f.copy()
    
    return best_f, best_obj


def multiplicative_update(f0, time_limit, verbose=False):
    """
    Multiplicative weight update: a[j] *= exp(-eta * pressure[j])
    where pressure[j] = sum_k w[k] * a[k-j] (contribution to peaks)
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    eta = 0.1
    beta_ls = 50.0
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        conv_max = conv.max()
        
        shifted = beta_ls * (conv - conv_max)
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()
        
        # Pressure on each element
        N = 1 << (3 * n - 2).bit_length()
        A = np.fft.rfft(f, N)
        W = np.fft.rfft(w, N)
        R = np.fft.irfft(W * np.conj(A), N)[:n]
        pressure = R  # pressure[j] = sum_k w[k] * f[k-j]
        
        # Multiplicative update
        log_f = np.log(f + 1e-30)
        f_new = softmax_fn(log_f - eta * pressure)
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()
        
        obj_new = fast_convolve(f_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    MU iter {it}: obj={best_obj:.10f}, eta={eta:.4f}")
        else:
            no_improve += 1
            eta *= 0.97
            beta_ls *= 1.02
        
        if no_improve > 150:

            f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
            eta = 0.05
            no_improve = 0
        
        eta = max(eta, 0.001)
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with local line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    # Pre-compute which coordinates are "active" (non-negligible)
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-10:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            
            # Try various multipliers
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 0.3]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Linear interpolation to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def diverse_initializations(parent, rng, n_target):
    """Generate diverse initial sequences."""
    candidates = []
    
    # 1. Parent interpolated
    if parent is not None:
        if len(parent) == n_target:
            candidates.append(parent.copy())
        else:
            candidates.append(interpolate_sequence(parent, n_target))
    
    # 2. Uniform
    candidates.append(np.ones(n_target) / n_target)
    
    # 3. Random
    r = rng.dirichlet(np.ones(n_target))
    candidates.append(r)
    
    # 4. Perturbed parent
    if parent is not None:
        p = interpolate_sequence(parent, n_target)
        candidates.append(project_simplex(p + rng.standard_normal(n_target) * 0.05))
    
    # 5. Triangular
    x = np.linspace(-1, 1, n_target)
    tri = np.maximum(0, 1 - np.abs(x))
    tri = tri / tri.sum()
    candidates.append(tri)
    
    # 6. Cosine window
    x = np.linspace(0, np.pi, n_target)
    cos_w = np.maximum(0, np.cos(x))
    cos_s = cos_w.sum()
    if cos_s > 0:
        candidates.append(cos_w / cos_s)
    
    # 7. Two-lobed
    center = n_target // 2
    x1 = np.linspace(-1, 0, n_target // 2)
    x2 = np.linspace(0.2, 1, n_target - n_target // 2)
    lobes = np.concatenate([np.maximum(0, 1 - np.abs(x1)), np.maximum(0, 1 - np.abs(x2))])
    ls = lobes.sum()
    if ls > 0:
        candidates.append(lobes / ls)
    
    return candidates
