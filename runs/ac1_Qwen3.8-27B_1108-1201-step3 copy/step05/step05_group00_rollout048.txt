Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using subgradient descent with momentum,
accelerated Frank-Wolfe with away steps, and coordinate descent refinement.
Exploits FFT-based convolution for efficiency and uses the previous best

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgrad(conv, a, k, n):
    """Compute subgradient of (a*a)_k at point a."""
    sg = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        sg[j] = 2.0 * a[k - j]
    return sg

    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        # Find top peaks for subgradient
        n_peaks = min(5, 2*n-1)
        peak_idx = np.argpartition(conv, -n_peaks)[-n_peaks:]
        peak_vals = conv[peak_idx]
        # Weights: softmax over peak values
        max_pv = peak_vals.max()
        weights = np.exp((peak_vals - max_pv) / (max_pv * 0.1 + 1e-10))
        weights /= weights.sum()
        
        # Compute combined subgradient
        sg = np.zeros(n)
        for i, k in enumerate(peak_idx):
            if weights[i] < 1e-10:
                continue
            js = max(0, int(k) - n + 1)
            je = min(n - 1, int(k))
            sg[js:je+1] += weights[i] * 2.0 * a[int(k)-je:int(k)-js+1][::-1]
        
        # Momentum update
        v = momentum * v - lr * sg
        a_new = a + v
        a_new = project_simplex(a_new)
        
        # Check improvement
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            # Revert and reduce lr
            lr *= 0.95
            v *= momentum
            no_improve += 1
        
        a = a_new
        
        if verbose and it % 100 == 0:
            print(f"    SG iter {it}: obj={best_obj:.10f}, lr={lr:.6f}")
        
        if no_improve > 200:
            break
        if lr < 1e-8:
            break
    
    return best_a, best_obj


def fw_away_steps(a0, time_limit, verbose=False):
    """Frank-Wolfe with away steps on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        k_star = int(np.argmax(conv))
        
        # Compute subgradient at peak
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        sg = np.zeros(n)
        sg[js:je+1] = 2.0 * a[k_star-je:k_star-js+1][::-1]
        
        # FW direction: vertex with minimum subgradient
        # sg[j] = 2*a[k_star - j] for valid j, 0 otherwise
        # We want to minimize sg[j] over j
        # For j where sg[j] = 0 (i.e., k_star - j < 0 or >= n), those are best
        valid_js = np.arange(js, je+1)
        invalid_js = np.setdiff1d(np.arange(n), valid_js)
        
        if len(invalid_js) > 0:
            j_star = invalid_js[0]  # Any invalid position gives sg=0
        else:
            j_star = valid_js[np.argmin(sg[valid_js])]
        
        v = np.zeros(n)
        v[j_star] = 1.0
        
        # Away step: find coordinate to move away from
        # The "worst" coordinate in the support is the one with largest a_j * sg_j
        support = np.where(a > 1e-10)[0]
        if len(support) > 0:
            away_scores = a[support] * sg[support]
            j_away = support[np.argmax(away_scores)]
        else:
            j_away = -1
        
        # Try standard FW step and away step
        best_alpha = 0.0
        best_val = M
        
        # Standard FW: a + alpha*(v - a)
        for alpha in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5]:
            a_trial = (1 - alpha) * a + alpha * v
            a_trial = project_simplex(a_trial)
            val = fast_convolve(a_trial).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
                best_mode = 'fw'
        
        # Away FW: a + alpha*(v - a) - alpha * gamma * e_{j_away}
        # where gamma = a_{j_away} / (1 - a_{j_away})
        if j_away >= 0 and a[j_away] > 1e-10:
            gamma = a[j_away] / max(1 - a[j_away], 1e-10)
            for alpha in [0.01, 0.05, 0.1, 0.2, 0.3]:
                a_trial = a + alpha * (v - a)
                a_trial[j_away] -= alpha * gamma
                a_trial = np.maximum(a_trial, 0)
                a_trial = project_simplex(a_trial)
                val = fast_convolve(a_trial).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha
                    best_mode = 'away'
        
        if best_alpha > 0:
            if best_mode == 'fw':
                a_new = (1 - best_alpha) * a + best_alpha * v
            else:
                a_new = a + best_alpha * (v - a)
                a_new[j_away] -= best_alpha * gamma
            a_new = np.maximum(a_new, 0)
            a_new = project_simplex(a_new)
            a = a_new
            
            if best_val < best_obj - 1e-12:
                best_obj = best_val
                best_a = a.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"    FW-A iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
        
        if no_improve > 100:
            # Perturb and restart
            rng = np.random.default_rng(it)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            no_improve = 0
        
        if no_improve > 500:
            break
    
    return best_a, best_obj


def accelerated_fw(a0, time_limit, verbose=False):
    """Accelerated Frank-Wolfe (SAW) with away steps."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        k_star = int(np.argmax(conv))
        
        # Subgradient
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        sg = np.zeros(n)
        sg[js:je+1] = 2.0 * a[k_star-je:k_star-js+1][::-1]
        
        # FW vertex
        valid_js = np.arange(js, je+1)
        invalid_js = np.setdiff1d(np.arange(n), valid_js)
        if len(invalid_js) > 0:
            j_star = int(invalid_js[np.argmin(sg[invalid_js])])
        else:
            j_star = int(valid_js[np.argmin(sg[valid_js])])
        
        v = np.zeros(n)
        v[j_star] = 1.0
        
        # Away step
        support = np.where(a > 1e-10)[0]
        j_away = -1
        gamma = 0
        if len(support) > 0:
            away_scores = a[support] * sg[support]
            j_away = int(support[np.argmax(away_scores)])
            gamma = a[j_away] / max(1 - a[j_away], 1e-10)
        
        # Line search with both FW and away steps
        best_alpha = 0.0
        best_val = M
        best_mode = 'none'
        
        for alpha in np.arange(0.02, 1.0, 0.03):
            # FW step
            a_trial = (1 - alpha) * a + alpha * v
            a_trial = project_simplex(a_trial)
            val = fast_convolve(a_trial).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
                best_mode = 'fw'
            
            # Away step
            if j_away >= 0:
                a_trial2 = a + alpha * (v - a)
                a_trial2[j_away] -= alpha * gamma
                a_trial2 = np.maximum(a_trial2, 0)
                a_trial2 = project_simplex(a_trial2)
                val2 = fast_convolve(a_trial2).max()
                if val2 < best_val:
                    best_val = val2
                    best_alpha = alpha
                    best_mode = 'away'
        
        if best_alpha > 0:
            if best_mode == 'fw':
                a_new = (1 - best_alpha) * a + best_alpha * v
            else:
                a_new = a + best_alpha * (v - a)
                a_new[j_away] -= best_alpha * gamma
            a_new = np.maximum(a_new, 0)
            a_new = project_simplex(a_new)
            a = a_new
            
            if best_val < best_obj - 1e-12:
                best_obj = best_val
                best_a = a.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"    AccFW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
        
        if no_improve > 50:
            rng = np.random.default_rng(it * 7 + 3)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            no_improve = 0
        
        if no_improve > 300:
            break
    
    return best_a, best_obj


def multiplicative_descent_v2(a0, time_limit, verbose=False):
    """Improved multiplicative descent with adaptive temperature and peak targeting."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    lr = 0.005
    temp = 0.01
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        # Find top peaks
        n_peaks = min(10, 2*n-1)
        peak_idx = np.argpartition(conv, -n_peaks)[-n_peaks:]
        peak_vals = conv[peak_idx]
        
        # Exponential weights
        scaled = peak_vals / max(temp, 1e-10)
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peak_idx))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= ws
        
        # Compute subgradient (weighted)
        sg = np.zeros(n)
        for i, k in enumerate(peak_idx):
            if w[i] < 1e-12:
                continue
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            sg[js:je+1] += w[i] * 2.0 * a[k-je:k-js+1][::-1]
        
        # Multiplicative update
        a = a * np.exp(-lr * sg)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        conv_new = fast_convolve(a)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    MD2 iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
        
        # Adaptive parameters
        lr *= 0.9995
        temp *= 1.01
        
        if no_improve > 200:
            rng = np.random.default_rng(it * 13 + 5)
            a = np.maximum(best_a + rng.standard_normal(n) * 0.003, 1e-15)
            a = a / a.sum()
            lr = 0.003
            temp = 0.01
            no_improve = 0
        
        if lr < 1e-7:
            break
    
    return best_a, best_obj


def coordinate_descent_v2(a0, time_limit, verbose=False):
    """Coordinate descent: for each coordinate, find optimal value via ternary search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-12:
                # Try small values
                candidates = [0.001, 0.005, 0.01]
            else:
                candidates = [old_val * m for m in [0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5]]
            
            best_j_obj = fast_convolve(a).max()
            best_j_val = old_val
            
            for new_val in candidates:
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_initial(n, shape='gaussian', sigma=0.25):
    """Create various initial sequences."""
    if shape == 'gaussian':
        x = np.arange(n) - n / 2.0
        a = np.exp(-0.5 * (x / (n * sigma)) ** 2)
    elif shape == 'bell':
        x = np.arange(n) - n / 2.0
        hw = n * 0.3
        t = np.clip(np.abs(x) / hw, 0, 1.0)
        a = 1.0 - t ** 6
    elif shape == 'uniform':
        a = np.ones(n)
    elif shape == 'arc':
        x = np.arange(1, n) / (n + 1)
        a = 1.0 / np.sqrt(x * (1 - x) + 1e-10)
    elif shape == 'cosine':
        x = np.arange(n) / n
        a = np.cos(np.pi * x) ** 2
        a = np.maximum(a, 0)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):

    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass
    
    if parent is None:
        parent = make_initial(512, 'gaussian')
    
    # === Phase 1: Try diverse initializations ===
    print("=== Phase 1: Diverse Initializations ===")
    inits = []
    for n_try in [512, 768, 1024]:
        for shape in ['gaussian', 'bell', 'cosine', 'arc']:
            a_init = make_initial(n_try, shape)
            inits.append(a_init)
            sc = try_seq(a_init, f"init_n{n_try}_{shape}")
    
    if parent is not None:
        inits.append(parent)
    
    # Find best init by actual objective
    best_init = None
    best_init_obj = float('inf')
    for a_init in inits:
        s = a_init.sum()
        if s < 1e-10:
            continue
        a_norm = a_init / s
        obj = 2 * len(a_norm) * fast_convolve(a_norm).max()
        if obj < best_init_obj:
            best_init_obj = obj

    print(f"  Best init: obj={best_init_obj:.6f} (n={len(best_init)})")
    
    # === Phase 2: Subgradient with momentum ===
    print(f"\n=== Phase 2: Subgradient + Momentum (n={len(best_init)}) ===")
    rem = deadline - time.monotonic()

    # === Phase 3: Accelerated FW with away steps ===
    print(f"\n=== Phase 3: Accelerated FW + Away Steps ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 60:
            t_alloc = min(rem * 0.2, 200)
            print(f"  Allocating {t_alloc:.0f}s (n={len(best_seq)})")

    # === Phase 4: Multiplicative descent ===
    print(f"\n=== Phase 4: Multiplicative Descent V2 ===")

        rem = deadline - time.monotonic()

    print(f"\n=== Phase 8: Subgradient Polish ===")
    if best_seq is not None:

        if rem > 20:
