    g = np.zeros(n)
    # For i in [0, n-1], k_star - i is in [k_star - (n-1), k_star]
    # We need 0 <= k_star - i < n, i.e., k_star - (n-1) <= i <= k_star
    # So i ranges from max(0, k_star - n + 1) to min(n-1, k_star)
    i_min = max(0, k_star - n + 1)
    i_max = min(n - 1, k_star)
    g[i_min:i_max+1] = 2 * a[k_star - np.arange(i_min, i_max + 1)]
    return g

def project_simplex(v):
    """Project onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def linprog_gla(a, b, top_k=5):
    """Solve the linearized GLA problem."""
    n = len(a)
    # Get top K peak positions
    idx = np.argpartition(b, -top_k)[-top_k:]
    idx = idx[np.argsort(b[idx])[::-1]]  # sort by value descending
    
    # Build LP:
    # Variables: a'[0],...,a'[n-1], t (index n)
    # Minimize t
    # Subject to:
    #   sum(a') = 1
    #   a' >= 0
    #   g_j^T a' - t <= g_j^T a - b[k_j]  for j in top_k
    
    c = np.zeros(n + 1)
    c[n] = 1  # minimize t
    
    # Equality: sum(a') = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1
    b_eq = [1.0]
    
    # Inequalities: g_j^T a' - t <= g_j^T a - b[k_j]
    A_ub = np.zeros((top_k, n + 1))
    b_ub = np.zeros(top_k)
    for idx_j, k_j in enumerate(idx):
        g_j = subgradient_at_peak(a, k_j)
        A_ub[idx_j, :n] = g_j
        A_ub[idx_j, n] = -1
        b_ub[idx_j] = g_j @ a - b[k_j]
    
    # Bounds: a' >= 0, t free (or t >= 0)
    bounds = [(0, None)] * n + [(-np.inf, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n]
    return None

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using exact subgradient (folding) mirror descent,
linearized GLA with LP, efficient pairwise coordinate descent, and length exploration.

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def subsgradient_peak(a, k_star):
    """Compute db[k_star]/da[i] = 2*a[k_star-i] if 0<=k_star-i<n, else 0."""
    n = len(a)
    g = np.zeros(n)
    i_min = max(0, k_star - n + 1)
    i_max = min(n - 1, k_star)
    if i_min <= i_max:
        idx = np.arange(i_min, i_max + 1)
        g[idx] = 2.0 * a[k_star - idx]
    return g


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def gla_linprog_step(a, b, top_k=8):
    """One linearized GLA step: solve LP to reduce top-K peaks."""
    n = len(a)
    idx = np.argpartition(b, -top_k)[-top_k:]
    idx = idx[np.argsort(b[idx])[::-1]]

    # LP: minimize t
    # s.t. sum(a') = 1, a' >= 0
    #      g_j^T a' - t <= g_j^T a - b[k_j]  for j in top_k
    c = np.zeros(n + 1)
    c[n] = 1.0

    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    b_ub = np.zeros(top_k)

        pass
    return None

                    diff[lo2:hi2+1] -= a[0:hi2-lo2+1]

                # Try several transfer fractions
                for frac in [0.05, 0.1, 0.2, 0.4]:
                    delta = a[i] * frac
                    # Change in conv: 2*delta*diff + delta^2 * (second order, small)
                    new_conv = conv + 2.0 * delta * diff
                    # Second order correction (only at k=2j, k=i+j, k=2i)
                    if 2 * j < 2 * n - 1:
                        new_conv[2 * j] += delta * delta
                    if i + j < 2 * n - 1:
                        new_conv[i + j] -= 2.0 * delta * delta
                    if 2 * i < 2 * n - 1:
                        new_conv[2 * i] += delta * delta

                    obj = new_conv.max()
                    if obj < best_obj_local - 1e-14:
                        best_obj_local = obj
                        best_delta = delta
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                conv = fast_convolve(a)  # Recompute exactly
                if conv.max() < best_obj - 1e-14:
                    best_obj = conv.max()
                    best_a = a.copy()
                    improved = True

        if not improved and passes > 3:
            break
        if verbose and passes % 10 == 0:
            print(f"    CD pass {passes}: obj={best_obj:.10f}")

    return best_a, best_obj


def interp_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    1. Mirror descent with exact folding subgradient
    2. Linearized GLA with LP
    3. Efficient coordinate descent
    4. Length exploration
    5. Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return None
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return None
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return None
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if np.isfinite(p_score):
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        print("No parent found, using uniform init n=768")

    n_parent = len(parent)
    base = best_seq.copy() if best_seq is not None else parent

    # === Phase 1: Mirror descent with exact subgradient ===
    print(f"\n=== Phase 1: Mirror descent (n={n_parent}, best={best_score:.8f}) ===")
    p1_end = time.monotonic() + 280

    # Run from parent
    t_alloc = min(p1_end - time.monotonic(), 120)
    if t_alloc > 10:
        print(f"  MD from parent, t={t_alloc:.0f}s")
        a_opt, _ = mirror_descent_exact(base, time_limit=t_alloc)
        try_seq(a_opt, "p1_md_parent")

    # Run from perturbed parent
    rem = p1_end - time.monotonic()
    if rem > 20:
        pert = normalize(base + rng.standard_normal(n_parent) * 0.02)
        t_alloc = min(rem, 100)
        a_opt, _ = mirror_descent_exact(pert, time_limit=t_alloc, lr_init=0.03)
        try_seq(a_opt, "p1_md_pert")

    # === Phase 2: GLA with LP ===
    print(f"\n=== Phase 2: GLA-LP (n={n_parent}, best={best_score:.8f}) ===")
    p2_end = time.monotonic() + 300
    if best_seq is not None:
        base = best_seq.copy()
        n_base = len(base)

        # GLA with different top_k values
        for tk in [5, 10, 15]:
            rem = p2_end - time.monotonic()
            if rem < 15:
                break
            t_alloc = min(rem * 0.5, 90)
            print(f"  GLA top_k={tk}, t={t_alloc:.0f}s")
            a_opt, _ = gla_loop(base, time_limit=t_alloc, top_k=tk)
            try_seq(a_opt, f"p2_gla_k{tk}")

        # GLA from perturbed
        rem = p2_end - time.monotonic()
        if rem > 30 and best_seq is not None:
            n_best = len(best_seq)
            pert = normalize(best_seq + rng.standard_normal(n_best) * 0.01)
            t_alloc = min(rem * 0.7, 80)
            a_opt, _ = gla_loop(pert, time_limit=t_alloc, top_k=8)
            try_seq(a_opt, "p2_gla_pert")

    # === Phase 3: Coordinate descent ===
    print(f"\n=== Phase 3: Coordinate descent (best={best_score:.8f}) ===")
    p3_end = time.monotonic() + 280
    if best_seq is not None:
        rem = p3_end - time.monotonic()
        if rem > 20:
            t_cd = min(rem * 0.8, 200)
            print(f"  CD, t={t_cd:.0f}s")
            a_opt, _ = coord_descent_efficient(best_seq, time_limit=t_cd)
            try_seq(a_opt, "p3_cd")

    # === Phase 4: Length exploration ===
    print(f"\n=== Phase 4: Length exploration (best={best_score:.8f}) ===")
    p4_end = time.monotonic() + 300
    if best_seq is not None:
        n_best = len(best_seq)
        target_lengths = [640, 768, 896, 1024, 1280]
        target_lengths = [t for t in target_lengths if t != n_best and 256 <= t <= 2048]
        target_lengths.sort(key=lambda x: abs(x - n_best))
        target_lengths = target_lengths[:3]

        for tn in target_lengths:
            rem = p4_end - time.monotonic()
            if rem < 30:
                break
            f_interp = interp_to_length(best_seq, tn)
            t_alloc = min(rem * 0.4, 80)
            print(f"  n={tn}, MD t={t_alloc:.0f}s")
            a_opt, _ = mirror_descent_exact(f_interp, time_limit=t_alloc, lr_init=0.04)
            try_seq(a_opt, f"p4_n{tn}_md")

            rem = p4_end - time.monotonic()
            if rem > 20:
                t_gla = min(rem * 0.5, 50)
                a_opt, _ = gla_loop(normalize(a_opt), time_limit=t_gla, top_k=8)
                try_seq(a_opt, f"p4_n{tn}_gla")

    # === Phase 5: Deep optimization on best ===
    print(f"\n=== Phase 5: Deep optimization (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Fine mirror descent
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_alloc = min(rem * 0.4, 120)
            print(f"  Fine MD, t={t_alloc:.0f}s")
            a_opt, _ = mirror_descent_exact(best_seq, time_limit=t_alloc, lr_init=0.01)
            try_seq(a_opt, "p5_fine_md")

        # GLA with more constraints
        rem = deadline - time.monotonic() - 20
        if rem > 20 and best_seq is not None:
            t_gla = min(rem * 0.4, 100)
            print(f"  GLA k=20, t={t_gla:.0f}s")
            a_opt, _ = gla_loop(best_seq, time_limit=t_gla, top_k=20)
            try_seq(a_opt, "p5_gla_k20")

        # CD polish
        rem = deadline - time.monotonic() - 15
        if rem > 15 and best_seq is not None:
            t_cd = min(rem * 0.6, 80)
            a_opt, _ = coord_descent_efficient(best_seq, time_limit=t_cd)
            try_seq(a_opt, "p5_cd")

        # Final tiny perturbation + optimize
        rem = deadline - time.monotonic() - 10
        for pi in range(3):
            rem = deadline - time.monotonic() - 8
            if rem < 8 or best_seq is None:
                break
            noise = rng.standard_normal(len(best_seq)) * 0.003
            f_pert = normalize(best_seq + noise)
            t_alloc = min(rem * 0.5, 30)
            a_opt, _ = mirror_descent_exact(f_pert, time_limit=t_alloc, lr_init=0.005)
            try_seq(a_opt, f"p5_pert_{pi}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
