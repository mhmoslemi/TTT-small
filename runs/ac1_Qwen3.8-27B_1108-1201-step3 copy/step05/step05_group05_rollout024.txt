def compute_h_weighted_fft(a, w):
    """
    Compute h[i] = sum_j a[j] * (w[i+j] + w[i-j]) efficiently via FFT.
    a: length n
    w: length 2n-1 (the softmax weights over convolution lags)
    Returns: h of length n
    """
    n = len(a)
    # T2 = conv(a, w)[0:n]
    # T1 = conv(a_rev, w)[(n-1):(n-1+n)]
    
    fft_size = 1 << (3*n - 1).bit_length()  # enough for linear convolutions
    
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    
    # conv(a, w) = ifft(A * W)
    conv_aw = np.fft.irfft(A * W, fft_size)
    T2 = conv_aw[:n]
    
    # conv(a_rev, w)
    a_rev = a[::-1]
    Ar = np.fft.rfft(a_rev, fft_size)
    conv_arw = np.fft.irfft(Ar * W, fft_size)
    T1 = conv_arw[(n-1):(2*n-1)]
    
    return T1 + T2

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated full log-sum-exp smoothed gradient (O(n log n) per step)
2. Adaptive line search with projection onto probability simplex
3. Temperature annealing schedule
4. Squared parameterization for barrier-free exploration
5. Peak-shaving local search targeting argmax lag
6. Multi-length, multi-restart exploration

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_grad_fft(a, w):
    """
    Compute h[j] = sum_k w[k] * (a[k-j] + a[j-k]) via FFT.
    a: length n, w: length 2n-1
    Returns: h of length n (gradient of weighted conv sum w.r.t. a)
    """
    n = len(a)
    fft_size = 1 << (3 * n - 1).bit_length()

    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)

    # T2[j] = conv(a, w)[j] for j=0..n-1
    conv_aw = np.fft.irfft(A * W, fft_size)
    T2 = conv_aw[:n]

    # T1[j] = conv(a_rev, w)[n-1+j] for j=0..n-1
    a_rev = a[::-1]
    Ar = np.fft.rfft(a_rev, fft_size)
    conv_arw = np.fft.irfft(Ar * W, fft_size)
    T1 = conv_arw[(n - 1):(2 * n - 1)]

    return T1 + T2

def optimize_sqr_param(x0, time_limit, T_init=20.0):
    """
    Optimize using squared parameterization: a = x^2 / ||x||^2.
    No projection needed - non-negativity is automatic.
    """
    n = len(x0)
    x = x0.astype(np.float64).copy()
    # Ensure x is not all zeros
    if np.linalg.norm(x) < 1e-10:
        x = np.random.default_rng(42).standard_normal(n)

    best_a = None
    best_obj = float('inf')

    T = T_init
    t0 = time.monotonic()
    patience = 0
    iter_count = 0

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    lr = 0.05

    while time.monotonic() - t0 < time_limit:
        iter_count += 1

        # Compute a from x
        x2 = x * x
        S = x2.sum()
        if S < 1e-30:
            break
        a = x2 / S

        # Compute gradient in x-space
        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        exp_s = np.exp(scaled)
        w = exp_s / exp_s.sum()

        h = compute_grad_fft(a, w)  # gradient of smoothed obj w.r.t. a (up to const)

        # Chain rule: dF/dx[i] = 2*x[i]/S * (h[i] - a[i] * sum_j a[j]*h[j])
        R = np.dot(a, h)
        grad_x = (2.0 * x / S) * (h - a * R)

        # Normalize gradient
        gn = np.linalg.norm(grad_x)
        if gn < 1e-15:
            break
        grad_x = grad_x / gn

        # Adam step
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * grad_x * grad_x
        m_hat = m / (1 - beta1 ** iter_count)
        v_hat = v / (1 - beta2 ** iter_count)

        x_trial = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Evaluate
        x2_t = x_trial * x_trial
        S_t = x2_t.sum()
        if S_t < 1e-30:
            x = x + 0.01 * np.random.default_rng(iter_count).standard_normal(n)
            patience += 10
            continue
        a_trial = x2_t / S_t
        obj_trial = fast_convolve(a_trial).max()

        if obj_trial < best_obj - 1e-14:
            best_obj = obj_trial
            best_a = a_trial.copy()
            patience = 0
        else:
            patience += 1

        x = x_trial

        # Schedule
        if patience > 60:
            T *= 0.8
            lr *= 0.9
            if T < 0.5:
                T = 2.0
                lr = 0.03
                # Perturb

                x = x + rng_r.standard_normal(n) * 0.05
                patience = 0
        elif patience == 0:

    if best_a is None:

        best_obj = fast_convolve(best_a).max()

def peak_shaving(a0, time_limit, verbose=False):
    """
    Focused local search: repeatedly identify the peak lag and try to reduce it
    by transferring mass from contributing pairs to non-contributing positions.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        peak_lag = int(np.argmax(conv))
        peak_val = conv[peak_lag]

        # Find contributing pairs
        # c[peak_lag] = sum_i a[i] * a[peak_lag - i] (valid indices)
        js = max(0, peak_lag - (n - 1))
        je = min(n - 1, peak_lag)
        if js > je:
            js, je = 0, n - 1
        idx = np.arange(js, je + 1)
        products = a[idx] * a[peak_lag - idx]
        total_contrib = products.sum()

        if total_contrib < 1e-15:
            break

        # Find the pair with largest product
        max_pair_local = int(np.argmax(products))
        i1 = int(idx[max_pair_local])
        i2 = peak_lag - i1

        if i1 < 0 or i1 >= n or i2 < 0 or i2 >= n:
            stall += 100
            continue

        # Try to reduce a[i1]*a[i2] by transferring mass
        # Option 1: reduce a[i1], give to random non-contributing position
        # Option 2: reduce both a[i1] and a[i2]
        # Option 3: redistribute among multiple positions

        improved = False
        for frac in [0.1, 0.2, 0.3, 0.5]:
            if improved:
                break

            # Reduce a[i1] and a[i2], give to random positions
            amt1 = frac * a[i1]
            amt2 = frac * a[i2]

            # Find target positions (not contributing to peak)
            contrib_set = set(idx.tolist())
            non_contrib = [j for j in range(n) if j not in contrib_set and a[j] < 0.5 * a.max()]
            if not non_contrib:
                non_contrib = [j for j in range(n) if j != i1 and j != i2]
            if not non_contrib:
                continue

            t1 = int(rng.integers(0, len(non_contrib)))
            t2 = int(rng.integers(0, len(non_contrib)))
            dest1 = non_contrib[t1]
            dest2 = non_contrib[t2]

            a_trial = a.copy()
            a_trial[i1] -= amt1
            a_trial[i2] -= amt2
            a_trial[dest1] += amt1
            a_trial[dest2] += amt2
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial /= ss

            obj_trial = fast_convolve(a_trial).max()
            if obj_trial < best_obj - 1e-14:
                best_obj = obj_trial
                best_a = a_trial.copy()
                a = a_trial.copy()
                improved = True
                stall = 0

        if not improved:
            stall += 1
            if stall > 100:
                # Perturb
                noise = rng.standard_normal(n) * 0.003
                a = project_simplex(best_a + noise, 1.0)
                stall = 0

    return best_a, best_obj


def fast_coordinate_descent(a0, time_limit, verbose=False):

        pass_count += 1
        improved = False

                a_trial /= a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                remaining = 1.0 - old_val
                if remaining > 1e-10:
                    scale = (1.0 - best_j_val) / remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved and pass_count > 1:
            break

    return best_a, best_obj


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        A[k] = np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = complex(np.cos(phases[half]), 0)
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def random_init(n, rng):
    """Random non-negative initialization."""
    a = rng.exponential(1.0, n)
    return a / a.sum()


def triangular_init(n):
    """Triangular distribution."""
    x = np.arange(n) / max(n - 1, 1)
    a = 1.0 - np.abs(2.0 * x - 1.0)
    return a / a.sum()


def raised_cosine_init(n):
    """Raised cosine window."""
    x = np.arange(n) / max(n - 1, 1)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    return a / a.sum()


def gaussian_init(n):
    """Gaussian bell."""
    x = np.linspace(-2.5, 2.5, n)
    a = np.exp(-x**2)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining:
    - FFT-accelerated smoothed gradient with adaptive line search
    - Squared parameterization for barrier-free exploration
    - Peak-shaving local search
    - Fast coordinate descent
    - Multi-length and multi-restart exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # === Phase 1: Diverse initializations ===
    print("\n=== Phase 1: Initializations ===")
    target_lengths = []
    if parent is not None:
        target_lengths.append(len(parent))
    for n_try in [384, 512, 640, 768, 896, 1024, 1280]:
        if n_try not in target_lengths:
            target_lengths.append(n_try)
    target_lengths.sort(key=lambda x: abs(x - 768))

    inits = {}
    for n in target_lengths[:6]:
        candidates = [
            ("flat", flat_spectrum_init(n, rng)),
            ("random", random_init(n, rng)),
            ("tri", triangular_init(n)),
            ("cos", raised_cosine_init(n)),
            ("gauss", gaussian_init(n)),
        ]
        if parent is not None:
            candidates.append(("parent_interp", interpolate_sequence(parent, n)))
            # Perturbed parent
            noise = rng.standard_normal(n) * 0.02
            candidates.append(("parent_pert", project_simplex(
                interpolate_sequence(parent, n) + noise, 1.0)))

        best_init = None
        best_init_obj = float('inf')
        for name, init in candidates:
            init = np.maximum(init, 0)
            ss = init.sum()
            if ss < 1e-10:
                continue
            init = init / ss
            obj = fast_convolve(init).max()
            if obj < best_init_obj:
                best_init_obj = obj
                best_init = init
                best_name = name

        if best_init is not None:
            inits[n] = (best_init, best_init_obj)
            print(f"  n={n} ({best_name}): obj={best_init_obj:.6f}")

    # === Phase 2: Smoothed gradient optimization ===
    print(f"\n=== Phase 2: Smoothed Gradient (best={best_score:.6f}) ===")
    sorted_inits = sorted(inits.items(), key=lambda x: x[1][1])
    elapsed = time.monotonic()
    remaining = deadline - elapsed

    # Allocate time to top 3 initializations
    n_opt = min(3, len(sorted_inits))
    time_per_opt = min(remaining * 0.4 / n_opt, 200)

    for rank, (n_len, (init, init_obj)) in enumerate(sorted_inits[:n_opt]):
        rem = deadline - time.monotonic() - 200
        if rem < 30:
            break
        t_alloc = min(time_per_opt, rem)
        print(f"  Smooth opt n={n_len} (obj={init_obj:.6f}), t={t_alloc:.0f}s")

        # Method 1: Direct smoothed gradient
        a_opt, obj = optimize_smooth(init, time_limit=t_alloc * 0.6,
                                     T_init=25.0, verbose=(rank == 0))
        try_seq(a_opt, f"smooth_n{n_len}_r{rank}")

        # Method 2: Squared parameterization
        rem = deadline - time.monotonic() - 180
        if rem > 20:
            x_init = np.sqrt(np.maximum(a_opt, 0)) * np.sqrt(n_len)
            x_init = x_init + rng.standard_normal(n_len) * 0.1
            x
            a_opt2, obj2 = optimize_sqr_param(x_init, time_limit=min(rem, t_alloc * 0.4))
            try_seq(a_opt2, f"sqr_n{n_len}_r{rank}")

    # === Phase 3: Peak shaving on best ===
    print(f"\n=== Phase 3: Peak Shaving (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_ps = min(rem * 0.4, 120)
            print(f"  Peak shaving t={t_ps:.0f}s")
            a_opt, obj = peak_shaving(best_seq, time_limit=t_ps)
            try_seq(a_opt, "peak_shave")

    # === Phase 4: Coordinate descent ===
    print(f"\n=== Phase 4: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_cd = min(rem * 0.5, 100)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, obj = fast_coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")

    # === Phase 5: Refined iterations ===
    print(f"\n=== Phase 5: Refined (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(5):
            rem = deadline - time.monotonic() - 60
            if rem < 20:
                break
            t_alloc = min(rem / (5 - p), 80)

            if p == 0:
                f_init = best_seq.copy()
                T_init = 5.0
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.005
                f_init = project_simplex(best_seq + noise, 1.0)
                T_init = 10.0
            elif p == 2:
                # Mirror mix
                f_init = 0.5 * best_seq + 0.5 * best_seq[::-1]
                f_init = project_simplex(f_init, 1.0)
                T_init = 15.0
            elif p == 3:
                # Blend with uniform
                f_init = 0.85 * best_seq + 0.15 * np.ones(n_best) / n_best
                f_init = project_simplex(f_init, 1.0)
                T_init = 5.0
            else:
                # Squared param from best
                x_init = np.sqrt(np.maximum(best_seq, 0)) * np.sqrt(n_best)
                x_init += rng.standard_normal(n_best) * 0.05
                a_opt, obj = optimize_sqr_param(x_init, time_limit=t_alloc)
                try_seq(a_opt, f"refine_sqr_{p}")
                continue

            a_opt, obj = optimize_smooth(f_init, time_limit=t_alloc, T_init=T_init)
            try_seq(a_opt, f"refine_{p}")

            # Quick coordinate descent after refinement
            rem = deadline - time.monotonic() - 50
            if rem > 15:
                a_opt, obj = fast_coordinate_descent(best_seq, time_limit=min(rem, 40))
                try_seq(a_opt, f"cd_refine_{p}")

    # === Phase 6: Longer sequences ===
    print(f"\n=== Phase 6: Longer Sequences (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for target_n in [n_best + 128, n_best + 256]:
            if target_n > 2048:
                continue
            rem = deadline - time.monotonic() - 50
            if rem < 30:
                continue

            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.5, 80)
            print(f"  Longer n={target_n}, t={t_alloc:.0f}s")
            a_opt, obj = optimize_smooth(f_interp, time_limit=t_alloc, T_init=15.0)
            try_seq(a_opt, f"long_n{target_n}")

            rem = deadline - time.monotonic() - 30
            if rem > 15:
                a_opt, obj = fast_coordinate_descent(a_opt, time_limit=min(rem, 30))
                try_seq(a_opt, f"cd_long_n{target_n}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_opt, obj = optimize_smooth(best_seq, time_limit=min(rem, 40), T_init=2.0)
            try_seq(a_opt, "final_smooth")

        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt, obj = fast_coordinate_descent(best_seq, time_limit=min(rem, 25))
            try_seq(a_opt, "final_cd")

        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt, obj = peak_shaving(best_seq, time_limit=min(rem, 15))
            try_seq(a_opt, "final_shave")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
