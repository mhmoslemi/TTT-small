def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def solve_lp_direction_selected(f, M, n, active_indices=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for active k, g >= 0.
    Uses only active constraints for speed."""
    if active_indices is None:
        active_indices = np.arange(2 * n - 1)
    
    m = len(active_indices)
    if m == 0:
        return np.ones(n) / n
    
    c = -np.ones(n)
    A_ub = np.zeros((m, n))
    for idx, k in enumerate(active_indices):
        # (f*g)[k] = sum_j f[j] * g[k-j] for valid j
        # g[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            # A_ub[idx, j] = f[j] for j in [js, je], but we need g[k-j]
            # So the coefficient of g[i] (where i = k-j, so j = k-i) is f[k-i]
            # i ranges from k-je to k-js
            i_start = k - je
            i_end = k - js
            A_ub[idx, i_start:i_end+1] = f[je:js-1:-1]  # f[je], f[je-1], ..., f[js]
    
    b_ub = np.full(m, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x
    return None


def get_active_indices(conv, fraction=0.3):
    """Get indices of convolution values that are in the top fraction of the max."""
    conv_max = conv.max()
    threshold = conv_max * (1 - fraction)
    active = np.where(conv >= threshold)[0]
    # Always include the global max
    max_idx = np.argmax(conv)
    if max_idx not in active:
        active = np.append(active, max_idx)
    return active


def lp_fw_fast(f0, time_limit, verbose=False):
    """LP-FW with constraint selection for speed."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    # Constraint selection parameters
    frac = 0.4  # fraction of constraints to use
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 50 == 0 and verbose:
            print(f"    LP-FW iter {it}: M={M:.10f}, no_improve={no_improve}")
        
        # Get active constraints
        active = get_active_indices(conv, fraction=frac)
        
        g = solve_lp_direction_selected(f, M * m_exp, n, active)
        if g is None:
            m_exp *= 1.2
            if m_exp > 5.0:
                # Try with all constraints
                g = solve_lp_direction_selected(f, M * 1.1, n, np.arange(2*n-1))
                m_exp = 1.0
            if g is None:
                continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 40:
                # Perturb and restart
                rng = np.random.default_rng(it * 13 + 7)
                scale = 0.002 + 0.001 * min(no_improve // 40, 10)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            # Bisection refinement
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

    return best_f, best_obj

def smooth_optimize_v2(f0, time_limit, verbose=False):
    """Smooth optimization using log-sum-exp with adaptive temperature."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    
    # Adaptive temperature schedule
    beta = 50.0
    base_lr = 0.05
    lr = base_lr
    momentum = np.zeros(n)
    mom = 0.85
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 200 == 0 and verbose:
            print(f"    SmoothOpt iter {it}: obj={best_obj:.10f}, beta={beta:.1f}, lr={lr:.5f}")
        
        conv = fast_convolve(f)
        conv_max = conv.max()
        
        # Log-sum-exp approximation
        shifted = beta * (conv - conv_max)
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()
        
        obj_smooth = conv_max + np.log(exp_shifted.sum()) / beta
        
        # Gradient: d(obj_smooth)/da[m] = 2 * sum_k w[k] * a[k-m] (if 0<=k-m<n)
        # Use FFT for efficiency
        N = 1 << (4 * n - 3).bit_length()
        A = np.fft.rfft(f, N)
        w_ext = np.zeros(N)
        w_ext[:len(conv)] = w
        W = np.fft.rfft(w_ext)
        # The cross-correlation (a ⋆ w)[m] = sum_i a[i]*w[i+m]
        # DFT: DFT(a ⋆ w) = conj(DFT(a)) * DFT(w) ... no
        # Actually: (a ⋆ w)[m] = sum_i a[i] * w[i+m]
        # This equals (flip(a) * w)[m] where * is linear convolution
        # flip(a)[j] = a[n-1-j]
        # (flip(a) * w)[k] = sum_j flip(a)[j] * w[k-j] = sum_j a[n-1-j] * w[k-j]
        # Let i = n-1-j: = sum_i a[i] * w[k-n+1+i]
        # So (flip(a) * w)[k] = (a ⋆ w)[k - (n-1)] = (a ⋆ w)[k - n + 1]
        # Therefore (a ⋆ w)[m] = (flip(a) * w)[m + n - 1]
        
        A_flip = np.fft.rfft(f[::-1], N)
        R = np.fft.irfft(A_flip * W, N)
        grad = 2.0 * R[n-1 : 2*n-2]  # (a ⋆ w)[0] to (a ⋆ w)[n-2]... 
        # Hmm, let me verify the length. R has length N. 
        # (flip(a) * w) has length n + (2n-1) - 1 = 3n-2
        # (a ⋆ w)[m] = R[m + n - 1] for m = 0, ..., n-1
        # So we need R[n-1 : 2n-1] which is n elements.
        grad = 2.0 * R[n-1 : 2*n-1]
        
        if len(grad) != n:
            # Fallback to direct computation
            grad = np.zeros(n)
            k_top = np.where(w > 1e-8)[0]
            for k in k_top:
                wi = w[k]
                ms = max(0, int(k) - n + 1)
                me = min(n - 1, int(k))
                if ms <= me:
                    # grad[ms:me+1] += 2*wi * f[k-me : k-ms][::-1]
                    vals = f[ms:me+1]
                    # f[k-m] for m in ms..me is f[k-ms], f[k-ms-1], ..., f[k-me]
                    # = f[k-me : k-ms+1][::-1]
                    grad[ms:me+1] += 2.0 * wi * f[k-me:k-ms+1][::-1]
        
        # SGD step with momentum
        momentum = mom * momentum - lr * grad
        f_new = f + momentum
        f_new = np.maximum(f_new, 0.0)
        fs = f_new.sum()
        if fs < 1e-15:
            f_new = best_f.copy()
        else:
            f_new = f_new / fs
        
        obj_new = fast_convolve(f_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            f = f_new  # still take the step
        
        # Adaptive learning rate and temperature
        if no_improve > 100:
            lr *= 0.95
            beta *= 1.05
            no_improve = 0
        if no_improve > 500:
            # Restart
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            momentum = np.zeros(n)
            lr = base_lr * 0.5
            no_improve = 0
        
        lr = max(lr, 0.001)
        beta = min(beta, 2000.0)

    return best_f, best_obj


def targeted_peak_reduce(f0, time_limit, verbose=False):
    """Targeted peak reduction: identify peaks and reduce contributing elements."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    eta = 0.1
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 100 == 0 and verbose:
            print(f"    PeakReduce iter {it}: obj={best_obj:.10f}, eta={eta:.4f}")
        
        conv = fast_convolve(f)
        conv_max = conv.max()
        
        # Find top peaks (by value, not index)
        num_peaks = min(10, len(conv))
        peak_idx = np.argpartition(conv, -num_peaks)[-num_peaks:]
        peak_vals = conv[peak_idx]
        
        # Weight peaks by how much they exceed the "median" level
        med_level = np.median(conv)
        excess = np.maximum(peak_vals - med_level, 0)
        w_peaks = excess / (excess.sum() + 1e-15)
        
        # Compute the "pressure" at each position
        # pressure[j] = sum over peaks k of w[k] * (contribution of a[j] to conv[k])
        # The contribution of a[j] to conv[k] is: a[k-j] if 0<=k-j<n, plus a[j-k] if 0<=j-k<n
        # But for reduction purposes, we want to know: if we decrease a[j] by delta,
        # how much does conv[k] decrease? It decreases by delta * a[k-j] (if valid) 
        # + delta * a[j-k] (if valid and j != k-j)
        # For simplicity, let's just use: pressure[j] = sum_k w[k] * (a[k-j] if 0<=k-j<n else 0)
        # (ignoring the symmetric term for now, it's approximately the same)
        
        pressure = np.zeros(n)
        for i in range(len(peak_idx)):
            k = int(peak_idx[i])
            wi = w_peaks[i]
            if wi < 1e-10:
                continue
            # a[k-j] for j in [max(0,k-n+1), min(n-1,k)]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                # a[k-j] for j in js..je is a[k-je], a[k-je+1], ..., a[k-js]
                pressure[js:je+1] += wi * f[k-je:k-js+1][::-1]
        
        # Multiplicative update: reduce positions with high pressure
        # a[j] <- a[j] * (1 - eta * pressure[j] / pressure.max())
        p_max = pressure.max()
        if p_max > 1e-15:
            reduction = 1.0 - eta * pressure / p_max
            reduction = np.maximum(reduction, 0.1)  # don't reduce by more than 90%
            f_new = f * reduction
            f_new = np.maximum(f_new, 0.0)
            fs = f_new.sum()
            if fs > 1e-15:
                f_new = f_new / fs
                obj_new = fast_convolve(f_new).max()
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_f = f_new.copy()
                    f = f_new
                    no_improve = 0
                else:
                    no_improve += 1
                    eta *= 0.9
            else:
                no_improve += 1
        
        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            eta = 0.05
            no_improve = 0
        
        eta = max(eta, 0.01)

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def coordinate_descent_v2(f0, time_limit, num_mults=5):
    """Coordinate descent with more multipliers."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    mults = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in mults:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj
