Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an LP-based iterative method
combined with soft-max L-BFGS polishing. The LP approach finds the optimal
direction at each step by solving a linear program that maximizes the sum
of the candidate sequence subject to the convolution being bounded by the
current peak. This is fundamentally different from gradient-based methods
and provides a convex optimization step at each iteration.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_fixed(f, g):
    """Linear convolution of fixed f with variable g via FFT."""
    n = len(f)
    m = len(g)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]

def build_toeplitz_matrix(f):
    """Build the Toeplitz matrix A where A[k,j] = f[k-j] if valid."""
    n = len(f)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[idx]
    return A

def solve_lp_direction(f, M):
    """
    Solve the LP: max sum(g) s.t. f*g[k] <= M for all k, g >= 0.
    Returns the optimal g.
    """
    n = len(f)
    A = build_toeplitz_matrix(f)
    c = -np.ones(n)  # minimize -sum(g)
    b = M * np.ones(2*n-1)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    else:
        return f / f.sum()  # fallback

def lp_descent(f, n_iters=100, step_size=0.1, time_limit=60.0):
    """
    LP-based descent method.
    """
    t0 = time.monotonic()
    f = f.copy().astype(np.float64)
    s = f.sum()
    f = f / s  # normalize to sum = 1
    
    best_f = f.copy()
    best_M = fast_convolve(f).max()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(f)
        M = b.max()
        
        # Solve LP
        g = solve_lp_direction(f, M)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum  # normalize to sum = 1
        
        # Try different step sizes
        best_t = step_size
        best_trial_M = M
        
        for t_candidate in [step_size * 0.5, step_size, step_size * 1.5, step_size * 2.0]:
            f_trial = (1 - t_candidate) * f + t_candidate * g
            f_trial = np.maximum(f_trial, 0)
            fs = f_trial.sum()
            if fs < 1e-10:
                continue
            f_trial = f_trial / fs
            trial_M = fast_convolve(f_trial).max()
            if trial_M < best_trial_M:
                best_trial_M = trial_M
                best_t = t_candidate
        
        f = (1 - best_t) * f + best_t * g
        f = np.maximum(f, 0)
        fs = f.sum()
        if fs < 1e-10:
            break
        f = f / fs
        
        M = fast_convolve(f).max()
        if M < best_M - 1e-12:
            best_M = M
            best_f = f.copy()
    
    return best_f, best_M

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an LP-based iterative descent
method combined with soft-max L-BFGS polishing. The key innovation is using
a linear program at each iteration to find the optimal direction: we maximize
sum(g) subject to f*g <= M (current peak) and g >= 0. This provides a
convex optimization step that can navigate the non-convex landscape more
effectively than pure gradient methods.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_two(f, g):
    """Linear convolution of f and g via FFT."""
    n = len(f)
    m = len(g)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def build_conv_matrix(f):

        # (f*g)[k] += f[k-j] * g[j]

def solve_lp_direction(f, M, method='highs'):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    """
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b = M * np.ones(2 * n - 1)
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, 
                      method=method, options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return f / f.sum()


def lp_descent(f_init, time_limit=60.0, initial_step=0.1, verbose=False):
    """
    LP-based descent: iteratively solve LP and take steps.
    """
    t0 = time.monotonic()
    f = f_init.copy().astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        return f, fast_convolve(f).max()
    f = f / s
    
    best_f = f.copy()
    best_M = fast_convolve(f).max()
    step = initial_step
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP
        g = solve_lp_direction(f, best_M)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Line search over step sizes
        best_trial_M = best_M
        best_t = 0
        
        step_candidates = [step * 0.25, step * 0.5, step, step * 1.5, step * 2.0]
        for t_cand in step_candidates:
            if t_cand <= 0 or t_cand >= 1:
                continue
            f_trial = (1 - t_cand) * f + t_cand * g
            f_trial = np.maximum(f_trial, 0)
            fs = f_trial.sum()
            if fs < 1e-10:
                continue
            f_trial = f_trial / fs
            trial_M = fast_convolve(f_trial).max()
            if trial_M < best_trial_M - 1e-14:
                best_trial_M = trial_M
                best_t = t_cand
        
        if best_t == 0:
            # No improvement found, try smaller steps or break
            step *= 0.5
            if step < 1e-6:
                break
            continue
        
        f = (1 - best_t) * f + best_t * g
        f = np.maximum(f, 0)
        fs = f.sum()
        f = f / fs
        
        M_new = fast_convolve(f).max()
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_f = f.copy()
            if verbose and it % 100 == 0:
                print(f"    LP iter {it}: M={M_new:.10f}, t={best_t:.4f}")
        else:
            # M didn't decrease, revert
            f = best_f.copy()
            step *= 0.7
            if step < 1e-6:
                break
    
    return best_f, best_M


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)

def lbgfs_optimize(z0, T, n, time_limit, maxiter=3000):
    """L-BFGS in log-space with time limit enforcement."""
    t0 = time.monotonic()
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        obj, _ = soft_max_obj_grad(z, T, n)
        return obj
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = soft_max_obj_grad(z, T, n)
        return g
    
    try:
        res = minimize(
            fun=func,
            x0=z0,
            jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 25, 'disp': False}
        )
        return res.x, res.fun
    except Exception:
        return z0, func(z0)


def project_simplex(v):
    """Euclidean projection onto {x : sum(x)=1, x>=0}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def peak_shave(a, n_peaks=5, n_steps=100, time_limit=30.0):
    """Peak shaving: iteratively reduce the highest peaks."""
    t0 = time.monotonic()
    a = a.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        # Find top peaks (excluding edges)
        b_inner = b.copy()
        b_inner[0] = -1
        b_inner[-1] = -1
        peak_indices = np.argpartition(b_inner, -n_peaks)[-n_peaks:]
        peak_indices = [k for k in peak_indices if b[k] > best_obj * 0.95]
        
        if not peak_indices:
            break
        
        improved = False
        for k in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contributions = a[idx] * a[k - idx]
            
            max_local = np.argmax(contributions)
            i = idx[max_local]
            j = k - i
            
            for target in [i, j]:
                old_val = a[target]
                if old_val < 1e-10:
                    continue
                factor = 0.95
                new_val = old_val * factor
                diff = old_val - new_val
                if diff < 1e-15:
                    continue
                
                mask = np.ones(n, dtype=bool)
                mask[target] = False
                others = a[mask]
                if others.sum() < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[target] = new_val
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                
                val = fast_convolve(a_trial).max()
                if val < best_obj - 1e-12:
                    a = a_trial
                    best_obj = val
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            break
    
    return best_a, best_obj


def subgradient_polish(a0, time_limit=60.0, lr=0.005, mom=0.9, top_k=30):
    """Subgradient method with softmax-weighted active peaks."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = 0.001
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]
        
        v = mom * v - cur_lr * sg
        a = project_simplex(a + v)
        co = fast_convolve(a).max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.999
        cur_temp *= 1.01
        
        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 1000000)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0
    
    return best_a, best_obj

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an LP-based iterative descent
method. At each step, we solve a linear program to find the optimal direction
g that maximizes sum(g) subject to the cross-convolution f*g being bounded
by the current peak M. This provides a convex subproblem at each iteration.
Combined with soft-max L-BFGS polishing and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_two(f, g):
    n = len(f)
    m = len(g)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def build_conv_matrix(f):
    """Build matrix A where (A @ g)[k] = sum_i f[i]*g[k-i] (valid i)."""
    n = len(f)
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    idx = k - j
    valid = (idx >= 0) & (idx < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = f[idx[valid]]
    return A


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b = M * np.ones(2 * n - 1)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return f / f.sum()
