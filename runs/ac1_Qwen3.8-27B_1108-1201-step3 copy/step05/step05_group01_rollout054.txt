import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if not len(rho):
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s

def compute_subgradient(a, w):
    """
    Compute the weighted subgradient of max(conv(a,a)).
    w is a weight vector over the convolution positions (length 2n-1).
    Returns gradient vector of length n.
    """
    n = len(a)
    g = np.zeros(n)
    # For each position k with weight w[k]:
    # grad_a[i] += w[k] * (a[k-i] + a[i-k]) with out-of-range = 0
    # We only iterate over non-negligible weights
    thresh = 1e-12
    for k in range(len(w)):
        wk = w[k]
        if wk < thresh:
            continue
        # a[k-i] is valid for i in [max(0, k-n+1), min(n-1, k)]
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            g[i] += wk * a[k - i]
        # a[i-k] is valid for i in [max(0, k), min(n-1, k+n-1)]
        i_start2 = max(0, k)
        i_end2 = min(n - 1, k + n - 1)
        for i in range(i_start2, i_end2 + 1):
            g[i] += wk * a[i - k]
    return g

def compute_subgradient_fft(a, w):

    Second term: convolution(w, a) at position i (approximately)

    # sum_k w[k] * a[k-i] = sum_m w_rev[m] * a[m + len(w) - 1 - i]

    # Then sum_k w[k] * a[k-i] = sum_k w[k] * a_flip[n-1-k+i]

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    # Build the constraint matrix (banded/Toeplitz structure)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.15
            if m_exp > 10.0:
                m_exp = 1.0

        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 80:
                # Try larger perturbation
                rng = np.random.default_rng(it * 31 + 7)
                scale = 0.005 + 0.005 * min(no_improve // 80, 5)
                noise = rng.standard_normal(n) * scale
                f = project_simplex(best_f + noise)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
                continue
            m_exp = min(m_exp * 1.08, 5.0)
            continue

        # Backtracking line search
        best_alpha = 0.0
        best_val = M
        alpha = 1.0
        c1 = 0.01  # Armijo condition
        slope = np.dot(f - g, g - f)  # negative
        
        for _ in range(30):
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                break
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
            if val <= M + c1 * alpha * slope:
                break
            alpha *= 0.5
        
        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

    return best_f, best_obj

def soft_sg_optimize(f0, time_limit, verbose=False):
    """Soft subgradient descent with adaptive temperature and step size."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    eta = 0.01
    temperature = 0.001
    top_k = min(20, 2 * n - 1)
    momentum = 0.85
    v = np.zeros(n)
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        conv_max = conv.max()

        # Find top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        # Softmax weights over peaks
        pv = conv[peaks]
        scaled = (pv - pv.max()) / temperature
        w_peaks = np.exp(scaled)
        ws = w_peaks.sum()
        if ws < 1e-300:
            w_peaks = np.zeros(len(peaks))
            w_peaks[np.argmax(pv)] = 1.0
        else:
            w_peaks /= ws

        # Build full weight vector
        w = np.zeros(len(conv))
        w[peaks] = w_peaks

        # Compute subgradient
        g = compute_subgradient_fft(f, w)

        # Momentum update
        v = momentum * v + (1 - momentum) * g

        # Projected gradient step
        f_new = project_simplex(f - eta * v)

        obj_new = fast_convolve(f_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 100 == 0:
                print(f"    SoftSG iter {it}: obj={best_obj:.10f}, T={temperature:.6f}")
        else:
            f = f_new  # Still update to keep momentum going
            no_improve += 1
            eta *= 0.995
            temperature *= 1.01

        # periodic restart
        if no_improve > 500:
            rng = np.random.default_rng(it * 17 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            eta = 0.005
            temperature = 0.0005
            no_improve = 0

        eta = max(eta, 0.0005)

    return best_f, best_obj

def pair_swap_sa(f0, time_limit, verbose=False):
    """Pair-swap local search with simulated annealing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    total_time = time_limit
    T_max = 0.001
    T_min = 0.00001
    swap_fraction = 0.01  # fraction of mass to swap

    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Annealing schedule
        progress = (time.monotonic() - t0) / total_time
        T = T_max * (T_min / T_max) ** progress

        # Pick two random positions
        i, j = np.random.default_rng(it).choice(n, size=2, replace=False)

        if f[i] < 1e-15 or f[j] < 1e-15:
            continue

        # Try swapping
        delta = swap_fraction * min(f[i], f[j])
        
        # Move mass from i to j
        f_trial = f.copy()
        f_trial[i] -= delta
        f_trial[j] += delta
        f_trial = np.maximum(f_trial, 0)
        f_trial = f_trial / f_trial.sum()

        obj_trial = fast_convolve(f_trial).max()
        delta_obj = obj_trial - best_obj

        if delta_obj < 0:
            f = f_trial
            if obj_trial < best_obj:
                best_obj = obj_trial
                best_f = f.copy()
            no_improve = 0
        else:
            # Metropolis acceptance
            if np.random.default_rng(it + 1000).random() < np.exp(-delta_obj / max(T, 1e-10)):
                f = f_trial
            no_improve += 1

        if no_improve > 2000:
            # Restart
            rng = np.random.default_rng(it * 7 + 1)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
            no_improve = 0

        if verbose and it % 1000 == 0:
            print(f"    PairSwapSA iter {it}: best={best_obj:.10f}, T={T:.6f}")

    return best_f, best_obj
