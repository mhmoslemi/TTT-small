import time

def fast_autoconv(a):
    """Compute auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(A * A, fft_size)
    return conv[:size]

def compute_score(a):
    """Compute the score."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_autoconv(a)
    return 2.0 * n * conv.max() / (s * s)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

"""
Minimizes 2*n*max(auto_conv)/(sum)^2 using fast subgradient method with
adaptive learning rate, periodic coordinate refinement, multi-start
exploration, and dimension search. Uses FFT-based convolution and
efficient gradient computation via precomputed reverse array.
"""

import numpy as np
import time


def fast_autoconv(a):
    """Compute full auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score_fast(a):
    """Compute score = 2*n*max_conv/sum^2 (fast, no clipping)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_autoconv(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a), dtype=np.float64) / len(a)
    return a / s


def compute_smoothed_subgradient(a, conv, top_m=40, temperature=0.5):
    """
    Compute smoothed subgradient of max(conv).
    g[j] = 2 * sum_k w[k] * a[k-j] for valid indices.
    """
    n = len(a)
    k_actual = min(top_m, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]

    # Softmax weights with temperature
    pv_max = pv.max()
    pv_min = pv.min()
    range_pv = max(1e-12, pv_max - pv_min)
    temp = range_pv * temperature
    scaled = (pv - pv_max) / temp
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_actual) / k_actual
    else:
        w /= w_sum

    # Compute gradient
    g = np.zeros(n, dtype=np.float64)
    for i in range(k_actual):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start > j_end:
            continue
        # a[k-j] for j in [j_start, j_end] = a[k-j_start : k-j_end-1 : -1]
        g[j_start:j_end + 1] += 2.0 * w[i] * a[k - j_start:k - j_end - 1:-1]

    return g


def subgradient_optimize(a, time_limit, lr0=0.05, decay=0.9999, top_m=40,
                         cooldown_steps=5000, verbose=False):
    """
    Mirror descent (multiplicative weight) with smoothed subgradient.
    Periodic restarts for exploration.
    """
    n = len(a)
    a = normalize(a.copy())
    conv = fast_autoconv(a)
    best_score = compute_score_fast(a)
    best_a = a.copy()

    t0 = time.monotonic()
    step = 0
    cur_lr = lr0

    while time.monotonic() - t0 < time_limit:
        batch = 64 if step < 1000 else 128
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            step += 1

            # Compute subgradient
            g = compute_smoothed_subgradient(a, conv, top_m=top_m, temperature=0.5)

            # Multiplicative update
            log_update = cur_lr * g
            # Clip for numerical stability
            log_update = np.clip(log_update, -5.0, 5.0)
            a = a * np.exp(-log_update)
            a = normalize(a)

            # Update conv
            conv = fast_autoconv(a)
            sc = compute_score_fast(a)

            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a.copy()

            cur_lr *= decay

        # Periodic restart: add noise around best
        if step % cooldown_steps == 0:
            noise_scale = 0.002
            a = best_a.copy()
            a += np.random.standard_normal(n) * noise_scale
            a = normalize(a)
            conv = fast_autoconv(a)
            cur_lr = lr0 * 0.3

        if verbose and step % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SG step {step}: score={best_score:.10f}, lr={cur_lr:.6f}, t={elapsed:.0f}s")

    return best_a, best_score


def coord_refine(a, time_limit, n_coords_per_pass=20, verbose=False):

    Coordinate-wise refinement: pick coordinates with highest gradient
    contribution and do local 1D optimization using a few evaluations.

    a = normalize(a.copy())
    conv = fast_autoconv(a)
    best_score = compute_score_fast(a)
    best_a = a.copy()

    t0 = time.monotonic()
    pass_count = 0

        pass_count += 1
        improved = False

        # Compute gradient to identify important coordinates
        g = compute_smoothed_subgradient(a, conv, top_m=30, temperature=0.3)
        g_abs = np.abs(g)

        # Select coordinates to optimize (in random order among top candidates)
        if pass_count == 1:
            order = np.arange(n)
        else:
            # Bias toward high-gradient coordinates
            probs = g_abs + 1e-10
            probs /= probs.sum()
            order = np.random.choice(n, size=n, replace=False, p=probs)

        for j in order[:n_coords_per_pass * pass_count % n + n_coords_per_pass]:
            if time.monotonic() - t0 > time_limit:

            j = int(j)

            if old_val < 1e-14:
                continue

            # Try several candidate values
            candidates = [
                old_val * 0.7, old_val * 0.85, old_val * 1.15, old_val * 1.3,

            ]
            # Clamp
            candidates = [max(0.0, min(0.99, c)) for c in candidates]
            candidates = list(set(candidates))

            best_v = old_val
            best_sc = best_score

            other_sum = 1.0 - old_val
            if other_sum < 1e-12:
                continue

            for v in candidates:
                if abs(v - old_val) < 1e-10:
                    continue
                new_a = a.copy()
                new_a[j] = v
                new_a[j:j+1] = v
                # Rescale others to maintain sum=1
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                new_a[mask] *= (other_sum) / (other_sum)  # already sum=1 minus old_val
                # Actually: we want new_a to sum to 1
                # new_a[j] = v, others should sum to 1-v
                # Currently others sum to 1-old_val (since a sums to 1)
                # Scale: others *= (1-v)/(1-old_val)
                new_a = a.copy()
                new_a[j] = v
                scale = (1.0 - v) / other_sum if other_sum > 1e-12 else 0.0
                new_a[mask] *= scale
                new_a = np.maximum(new_a, 0.0)
                sc = compute_score_fast(new_a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_v = v

            if best_v != old_val:
                a[j] = best_v
                scale = (1.0 - best_v) / other_sum
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = np.maximum(a, 0.0)
                a = normalize(a)
                conv = fast_autoconv(a)
                sc = compute_score_fast(a)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a.copy()
                    improved = True

        if verbose and pass_count % 5 == 0:

        if not improved and pass_count > 3:

def pair_transfer(a, time_limit, n_tries=5000, verbose=False):

    a = normalize(a.copy())

    best_a = a.copy()
    rng = np.random.default_rng(777)

            if time.monotonic() - t0 > time_limit:
                break

            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

            a_j = a[j]
            if a_j < 1e-12:
                continue

            # Try transferring a fraction of a_j to a_i
            frac = rng.uniform(0.05, 0.4)
            delta = a_j * frac

            new_a = a.copy()
            new_a[j] = a_j - delta
            new_a[i] = a[i] + delta
            sc = compute_score_fast(new_a)

            if sc < best_score - 1e-12:
                a = new_a
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


def geometric_init(n, r):
    """Geometric (exponentially decaying) sequence."""
    a = r ** np.arange(n, dtype=np.float64)
    return normalize(a)


def cosine_init(n):
    """Raised cosine sequence."""
    x = np.linspace(0, np.pi, n)
    a = 0.5 * (1 + np.cos(x))
    return normalize(a)


def exponential_init(n, alpha):
    """Exponential decay with adjustable rate."""
    a = np.exp(-alpha * np.arange(n, dtype=np.float64))
    return normalize(a)


def power_law_init(n, p):
    """Power law decay: a[i] = (i+1)^(-p)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    return normalize(a)


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1.0, old_n)
    new_x = np.linspace(0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a)
    return normalize(a_interp)


def proposed_multi_phase_optimize(a, time_limit, verbose=False):
    """
    Multi-phase: fast SG -> coord refine -> pair transfer -> coord refine.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score_fast(a)
    t0 = time.monotonic()

    # Phase A: Subgradient (exploratory + refinement)
    t_a = min(time_limit * 0.4, 300)
    rem = time_limit - (time.monotonic() - t0)
    if rem > t_a + 30:
        a_sg, sc_sg = subgradient_optimize(a, time_limit=t_a, lr0=0.03,
                                           decay=0.99995, top_m=50,
                                           cooldown_steps=3000)
        if sc_sg < best_score:
            best_score = sc_sg
            best_a = a_sg.copy()
        if verbose:
            print(f"        After SG: {best_score:.10f}")

    # Phase B: Coordinate refinement
    rem = time_limit - (time.monotonic() - t0)
    t_b = min(rem * 0.3, 200)
    if rem > t_b + 20:
        a_cd, sc_cd = coord_refine(best_a, time_limit=t_b,
                                   n_coords_per_pass=30)
        if sc_cd < best_score:
            best_score = sc_cd
            best_a = a_cd.copy()
        if verbose:
            print(f"        After CD: {best_score:.10f}")

    # Phase C: Pair transfer
    rem = time_limit - (time.monotonic() - t0)
    t_c = min(rem * 0.25, 150)
    if rem > t_c + 15:
        a_pt, sc_pt = pair_transfer(best_a, time_limit=t_c, n_tries=8000)
        if sc_pt < best_score:
            best_score = sc_pt
            best_a = a_pt.copy()
        if verbose:
            print(f"        After PT: {best_score:.10f}")

    # Phase D: Final coordinate polish
    rem = time_limit - (time.monotonic() - t0)
    if rem > 15:
        a_final, sc_final = coord_refine(best_a, time_limit=rem - 5,
                                         n_coords_per_pass=40)
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_final.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-start optimization combining subgradient method, coordinate
    refinement, pair transfer, and dimension exploration.
    Starts from parent sequence and diverse structured initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_a = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_a, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        score = compute_score_fast(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # === Generate starting points ===
    print("=== Generating starting points ===")
    starts = []

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            starts.append((parent, "parent"))
            p_score = compute_score_fast(parent)
            print(f"  Parent: n={len(parent)}, score={p_score:.10f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"  Could not load parent: {e}")

    # Geometric sequences
    for r in [0.997, 0.998, 0.999, 0.9995]:
        for n in [512, 768, 1024]:
            starts.append((geometric_init(n, r), f"geom_r{r}_n{n}"))

    # Exponential
    for alpha in [0.003, 0.005, 0.008, 0.01]:
        for n in [512, 768, 1024]:
            starts.append((exponential_init(n, alpha), f"exp_a{alpha}_n{n}"))

    # Power law
    for p in [0.5, 0.8, 1.0, 1.2]:
        for n in [512, 768]:
            starts.append((power_law_init(n, p), f"pow_p{p}_n{n}"))

    # Cosine
    for n in [512, 768, 1024]:
        starts.append((cosine_init(n), f"cos_n{n}"))

    # Random smooth
    for n in [512, 768, 1024]:
        a = rng.random(n)
        kernel_size = 21
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
        starts.append((normalize(a), f"rand_smooth_n{n}"))

    # Evaluate all starts quickly
    print(f"  Evaluating {len(starts)} starting points...")
    start_scores = []
    for a, label in starts:
        sc = compute_score_fast(a)
        start_scores.append((sc, a, label))
        try_seq(a, label)
    start_scores.sort(key=lambda x: x[0])

    print("  Top 5 starts:")
    for i, (sc, a, label) in enumerate(start_scores[:5]):
        print(f"    #{i+1} [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: Optimize top diverse starting points ===
    print(f"\n=== Phase 1: Optimize diverse starts (budget_s={budget_s}) ===")

    # Select diverse starts (best from each type)
    selected = []
    selected_types = set()
    for sc, a, label in start_scores:
        if len(selected) >= 5:
            break
        # Extract type prefix
        parts = label.split('_')
        type_key = parts[0]
        if type_key not in selected_types:
            selected.append((a, label, sc))
            selected_types.add(type_key)

    # Ensure parent is in selected
    parent_in = any(l == "parent" for _, l, _ in selected)
    if not parent_in and parent is not None:
        while len(selected) >= 5:
            selected.pop()
        selected.append((parent, "parent", compute_score_fast(parent)))

    # Allocate time
    n_sel = len(selected)
    t_per_start = min((deadline - time.monotonic() - 300) / max(1, n_sel), 280)

    for idx, (a_init, label, sc_init) in enumerate(selected):
        rem = deadline - time.monotonic() - 250
        if rem < 30:
            break
        t_alloc = min(t_per_start, rem)
        print(f"  [{idx+1}/{n_sel}] Optimizing [{label}]: t={t_alloc:.0f}s")
        a_opt, sc_opt = proposed_multi_phase_optimize(a_init, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, label + "_opt")

    # === Phase 2: Dimension exploration ===
    print(f"\n=== Phase 2: Dimension exploration (best={best_score:.10f}) ===")
    if best_a is not None:
        best_n = len(best_a)
        target_ns = [448, 512, 640, 704, 768, 832, 896, 1024, 1152, 1280]
        target_ns = [tn for tn in target_ns if tn != best_n]

        for target_n in target_ns:
            rem = deadline - time.monotonic() - 150
            if rem < 40:
                break
            t_alloc = min(rem * 0.3, 120)
            a_interp = interpolate_to_length(best_a, target_n)
            a_opt, sc = proposed_multi_phase_optimize(a_interp, time_limit=t_alloc)
            if sc < best_score:
                try_seq(a_opt, f"dim_{target_n}")

    # === Phase 3: Intensive local search on best ===
    print(f"\n=== Phase 3: Intensive local search (best={best_score:.10f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 30:
            # Long subgradient run
            t_sg = min(rem * 0.4, 300)
            print(f"  Long SG: t={t_sg:.0f}s")
            a_sg, sc = subgradient_optimize(best_a, time_limit=t_sg, lr0=0.02,
                                            decay=0.99997, top_m=60,
                                            cooldown_steps=5000)
            try_seq(a_sg, "long_sg")

            # Coordinate refinement
            rem = deadline - time.monotonic() - 50
            if rem > 20:
                t_cd = min(rem * 0.4, 200)
                print(f"  CD refine: t={t_cd:.0f}s")
                a_cd, sc = coord_refine(best_a, time_limit=t_cd, n_coords_per_pass=50)
                try_seq(a_cd, "cd_refine")

            # Pair transfer
            rem = deadline - time.monotonic() - 40
            if rem > 15:
                t_pt = min(rem * 0.4, 150)
                print(f"  Pair transfer: t={t_pt:.0f}s")
                a_pt, sc = pair_transfer(best_a, time_limit=t_pt, n_tries=10000)
                try_seq(a_pt, "pair_transfer")

            # Multiple perturbation restarts
            rem = deadline - time.monotonic() - 30
            if rem > 20:
                n_best = len(best_a)
                t_per_pert = min(rem / 4, 60)
                for trial in range(4):
                    rem = deadline - time.monotonic() - 25
                    if rem < 15:
                        break
                    for noise_scale in [0.001, 0.003, 0.005]:
                        rem = deadline - time.monotonic() - 25
                        if rem < 15:
                            break
                        noise = rng.standard_normal(n_best) * noise_scale
                        a_pert = normalize(best_a + noise)
                        a_opt, sc = proposed_multi_phase_optimize(
                            a_pert, time_limit=min(rem, 40))
                        try_seq(a_opt, f"perturb_{trial}_{noise_scale}")

    # === Phase 4: Final ultra-precise polish ===
    print(f"\n=== Phase 4: Final polish (best={best_score:.10f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            # Very targeted coordinate search
            t_final = rem
            print(f"  Final CD: t={t_final:.0f}s")
            a_final, sc = coord_refine(best_a, time_limit=t_final, n_coords_per_pass=60)
            try_seq(a_final, "final_polish")

    # === Final verification ===
    if best_a is None:
        best_a = geometric_init(768, 0.999)
        best_score = compute_score_fast(best_a)

    # Verify with actual evaluate_sequence
    final_list = [float(x) for x in np.clip(best_a, 0.0, 1000.0)]
    final_score = float(evaluate_sequence(final_list))
    if np.isfinite(final_score):
        if final_score < best_score:
            best_score = final_score
            print(f"  Verification improved: {best_score:.10f}")
        else:
            print(f"  Verification: {best_score:.10f} (eval: {final_score:.10f})")

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_a)})")
    return [float(x) for x in np.clip(best_a, 0.0, 1000.0)]
