    """Simple version of peak shaving for length n=768."""
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    n = len(f)
    conv_val = np.convolve(f, f, mode='full')
    top_k = 20
    # Get top positions
    indices = np.argsort(conv_val)[-top_k:]
    # For each index, find the two coefficients in the product (at lag k, j and k-j)
    best_improvement = 0
    best_factor = 1.0

    # We'll try to zero out single factors and see if the score improves
    for k in indices:
        js = max(0, k - n + 1)
        je = min(n-1, k)
        # Calculate the product
        j_list = np.arange(js, je+1)
        products = f[j_list] * f[k - j_list]  # This handles k-j for j from js to je
        j_max = j_list[np.argmax(products)]
        j_val = f[j_max]
        if j_val < 1e-12:
            continue

        # Try reductions
        try_factors = [0.3, 0.5, 0.7, 0.85, 1.0, 0.9, 0.95, 0.75]
        for factor in try_factors:
            new_val = j_val * factor
            if new_val < 1e-12:
                continue

            candidate = f.copy()
            candidate[j_max] = new_val
            # Normalize
            s = candidate.sum()
            if s < 1e-15:
                new_val = (new_val / j_val) * s  # Not really, but we hope to project if losing sum entirely
            candidate = candidate / s
            conv = fast_convolve(candidate)
            new_score = 2 * n * conv.max()  # (without the sum, so we need to use the evaluated function)
            # Instead, we evaluate_sequence
            eval_score = evaluate_sequence(candidate.tolist())
            if eval_score < best_improvement / n:  # Best improvement in terms of this factor? Let's hope.
                best_improvement = eval_score
                best_factor = factor
                break

    if best_improvement > 0:
        return candidate.tolist(), best_improvement
    else:
        return f.tolist(), best_score
