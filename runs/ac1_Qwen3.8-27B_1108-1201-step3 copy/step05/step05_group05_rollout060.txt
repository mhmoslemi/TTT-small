Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP best-response iteration
with multiple initializations, sequence lengths, and local refinement.
"""

import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, csc_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):

    rows = []
    cols = []
    data = []

        start_k = j
        end_k = j + n
        for k in range(start_k, min(end_k, 2*n-1)):
            rows.append(k)
            cols.append(j)
            data.append(a[k - j])
    C = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return C


def lp_best_response(a, M, solver='highs'):
    """
    Solve LP: max sum(g) s.t. conv(a,g) <= M (elementwise), g >= 0.
    Returns the optimal g or None if infeasible.
    """
    n = len(a)
    C = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=[(0, None)]*n, method=solver)
    if res.success:
        return res.x
    return None


def lp_iteration(a, max_iters=200, tol=1e-10, verbose=False):
    """
    Iterate the LP best-response until convergence.
    a is normalized to sum 1.
    Returns the best sequence found.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    for it in range(max_iters):
        conv = fast_convolve(a)
        M = conv.max()
        
        g = lp_best_response(a, M)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        g_conv = fast_convolve(g)
        g_peak = g_conv.max()
        
        if g_peak < best_obj - tol:
            best_obj = g_peak
            best_a = g.copy()
            a = g.copy()
            if verbose:
                print(f"    LP iter {it}: peak={best_obj:.10f} (full step)")
        else:
            # Try interpolation
            improved = False
            for t in [0.5, 0.3, 0.1, 0.05, 0.01]:
                a_trial = (1-t)*a + t*g
                a_trial = np.maximum(a_trial, 0)
                ts = a_trial.sum()
                if ts < 1e-15:
                    continue
                a_trial = a_trial / ts
                trial_conv = fast_convolve(a_trial)
                trial_peak = trial_conv.max()
                if trial_peak < best_obj - tol:
                    best_obj = trial_peak
                    best_a = a_trial.copy()

                    improved = True
                    if verbose:
                        print(f"    LP iter {it}: peak={best_obj:.10f} (t={t})")
                    break
            
            if not improved:
                # Check convergence
                if abs(M - best_obj) < 1e-12:
                    if verbose:
                        print(f"    LP iter {it}: converged, peak={best_obj:.10f}")
                    break
    
    return best_a, best_obj


def hann_window(n):
    """Hann window normalized to sum 1."""
    x = np.arange(n)
    a = 0.5 * (1 - np.cos(2 * np.pi * x / n))
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def raised_cosine(n, beta=1.0):
    """Raised cosine window."""
    x = np.arange(n)
    t = (x - (n-1)/2) / (n/2)
    a = np.where(np.abs(t) <= beta, 0.5*(1+np.cos(np.pi*t/beta)), 0.0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        mag = 1.0
        A[k] = mag * np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def random_init(n, rng):
    """Random non-negative initialization."""
    a = rng.exponential(1.0, n)
    return a / a.sum()


def gaussian_init(n):
    """Gaussian initialization."""
    x = np.linspace(-3, 3, n)
    a = np.exp(-x**2 / 2)
    return a / a.sum()


def polynomial_taper(n, p=2):
    """Polynomial taper: (1 - (2x/n - 1)^p) type window."""
    x = np.linspace(0, 1, n)
    t = 2*x - 1
    a = np.maximum(1 - np.abs(t)**p, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def coord_descent_refine(a0, time_limit=30):
    """Simple coordinate descent for local refinement."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            best_j_val = old_val
            best_j_obj = best_obj
            
            for mult in [0.7, 0.85, 1.15, 1.3]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP best-response iteration
with diverse initializations (Hann, raised cosine, polynomial tapers, flat-spectrum),
multi-length exploration, perturbation-based restarts, and coordinate descent refinement.
The core algorithm iteratively solves a linear program to find the best-response
sequence that keeps the cross-convolution bounded, which converges to a local
minimum of the auto-convolution peak.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build sparse matrix C where C @ g = full convolve(a, g)."""
    n = len(a)
    rows = []
    cols = []
    data = []
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        ks = np.arange(k_start, k_end)
        rows.extend(ks.tolist())
        cols.extend([j] * len(ks))
        data.extend(a[ks - j].tolist())
    C = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return C


def lp_best_response(a, M):
    """
    Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0.
    Returns optimal g or None.
    """
    n = len(a)
    C = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-10,
                           'primal_feasibility_tolerance': 1e-10,
                           'dual_edge_tolerance': 1e-10, 'primal_edge_tolerance': 1e-10})
    if res.success:
        return res.x
    return None


def lp_iteration(a0, time_limit, verbose=False):
    """
    Run LP best-response iterations until time limit.
    a0 should be normalized to sum 1.
    Returns (best_a, best_peak).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_a = a.copy()
    best_peak = conv.max()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        if M < best_peak - 1e-14:
            best_peak = M
            best_a = a.copy()

        if no_improve > 30:
            break

        g = lp_best_response(a, M)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        g_peak = fast_convolve(g).max()

        if g_peak < best_peak - 1e-12:
            a = g.copy()
            best_peak = g_peak
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 10 == 0:
                elapsed = time.monotonic() - t0
                print(f"      LP it={it}: peak={best_peak:.10f} ({elapsed:.0f}s)")
        else:
            improved = False
            for t_val in [0.7, 0.5, 0.3, 0.1]:
                a_trial = (1 - t_val) * a + t_val * g
                a_trial = np.maximum(a_trial, 0)
                ts = a_trial.sum()
                if ts < 1e-15:
                    continue
                a_trial = a_trial / ts
                tp = fast_convolve(a_trial).max()
                if tp < best_peak - 1e-12:
                    a = a_trial.copy()
                    best_peak = tp
                    best_a = a.copy()
                    no_improve = 0
                    improved = True
                    if verbose and it % 10 == 0:
                        elapsed = time.monotonic() - t0
                        print(f"      LP it={it} (t={t_val}): peak={best_peak:.10f} ({elapsed:.0f}s}")
                    break
            if not improved:
                no_improve += 1

        if no_improve > 10 and no_improve % 10 == 0:
            rng_p = np.random.default_rng(hash((it, n)) & 0xFFFFFFFF)
            noise = rng_p.standard_normal(n) * 0.01
            a = project_simplex(best_a + noise, target_sum=1.0)
            no_improve = 0

    return best_a, best_peak

def hann_window(n):
    """Hann window normalized to sum 1."""
    x = np.arange(n)
    a = 0.5 * (1 - np.cos(2 * np.pi * x / n))
    a[0] = 0.0
    a[-1] = 0.0
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def raised_cosine(n, beta=1.0):
    """Raised cosine window."""
    x = np.arange(n, dtype=np.float64)
    t = (x - (n - 1) / 2.0) / (n / 2.0)
    a = np.where(np.abs(t) <= beta, 0.5 * (1 + np.cos(np.pi * t / beta)), 0.0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def polynomial_taper(n, p=2.0):
    """Polynomial taper window."""
    x = np.linspace(0, 1, n)
    t = 2 * x - 1
    a = np.maximum(1 - np.abs(t) ** p, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def flat_spectrum_init(n, rng):
    """Non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        mag = 1.0
        A[k] = mag * np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def gaussian_init(n, sigma=0.3):
    """Gaussian window."""
    x = np.linspace(-1, 1, n)
    a = np.exp(-x ** 2 / (2 * sigma ** 2))
    return a / a.sum()


def double_gaussian_init(n, rng):
    """Two Gaussian peaks."""
    c1 = rng.uniform(0.2, 0.4)
    c2 = rng.uniform(0.6, 0.8)
    w = 0.08
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
    return a / a.sum()


def random_init(n, rng):
    """Random exponential initialization."""
    a = rng.exponential(1.0, n)
    return a / a.sum()


def prolate_init(n, rng):
    """Approximate prolate spheroidal wave function via iterative multiplication."""
    a = np.ones(n) / n
    bandwidth = n * 0.25
    for _ in range(50):
        # Time concentration
        A = np.fft.rfft
        a_fft = np.fft.rfft(a, n)
        mask = np.ones(len(a_fft), dtype=bool)
        center = n // 2
        fb = int(np.round(bandwidth))
        mask[:n - center - fb] = False
        mask[center + fb:] = False
        a_fft[~mask] = 0
        a = np.fft.irfft(a_fft, n)
        a = np.maximum(a, 0)
        a = a / a.sum()
    return a


def coord_descent(a0, time_limit, verbose=False):
    """Coordinate descent with multiplicative updates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    pass_count = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_val = old_val
            best_j_obj = best_obj

            multipliers = [0.6, 0.8, 0.9, 1.1, 1.25, 1.5]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved and pass_count > 1:
            break

    return best_a, best_obj


def pair_local_search(a0, time_limit, seed=0):
    """Randomized pair transfer local search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    support = np.where(a > 1e-10)[0]
    if len(support) == 0:
        support = np.arange(n)

    while time.monotonic() - t0 < time_limit:
        for _ in range(100):
            if time.monotonic() - t0 > time_limit:
                break
            idx = rng.integers(0, len(support), size=2)
            i, j = int(support[idx[0]]), int(support[idx[1]])
            if i == j:
                continue
            if a[i] < 1e-12 or a[j] < 1e-12:
                continue

            for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                for direction in [1, -1]:
                    transfer = direction * frac * min(a[i], a[j])
                    new_i = a[i] + transfer
                    new_j = a[j] - transfer
                    if new_i < -1e-15 or new_j < -1e-15:
                        continue
                    new_i = max(0.0, new_i)
                    new_j = max(0.0, new_j)
                    a_trial = a.copy()
                    a_trial[i] = new_i
                    a_trial[j] = new_j
                    val = fast_convolve(a_trial).max()
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        break

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
