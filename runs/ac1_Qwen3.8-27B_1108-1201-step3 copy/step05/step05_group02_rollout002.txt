def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def cross_convolve(f, g):
    """Compute (f*g)_k = sum_j f_j * g_{k-j} for all k."""
    return np.convolve(f, g)

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_cutting_plane(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M, g >= 0 using cutting plane."""
    # Start with center constraint
    active = [n - 1]
    
    for it in range(30):
        # Build constraint matrix for active constraints
        A_ub = np.zeros((len(active), n))
        for i, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                A_ub[i, js:je+1] = f[k - js:k - js + (je - js + 1)][::-1]
        b_ub = np.full(len(active), M)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': 5, 'presolve': True})
        if not res.success or res.x is None:
            return None
        
        g = res.x
        
        # Check feasibility against all constraints
        fg = np.convolve(f, g)
        max_viol = fg.max() - M
        if max_viol < 1e-9 * max(1.0, M):
            return g
        
        # Add most violated constraint and neighbors
        worst = int(np.argmax(fg - M))
        new_c = [worst]
        for d in [-2, -1, 1, 2]:
            nk = worst + d
            if 0 <= nk < 2 * n - 1 and nk not in active:
                new_c.append(nk)
        
        for nc in new_c:
            if nc not in active:
                active.append(nc)
    
    # If we used up iterations, check if the last solution is feasible
    g = res.x
    fg = np.convolve(f, g)
    if fg.max() <= M * (1 + 1e-8):
        return g
    return None

def fw_line_search(f, g, n, conv_f):
    """Efficient line search for FW: minimize max_k (f + alpha*(g-f)) * (f + alpha*(g-f))_k."""
    d = g - f
    # Precompute (d*d) and (f*d)
    dd = fast_convolve(d)
    fd = np.convolve(f, d)
    
    best_alpha = 0.0
    best_val = conv_f.max()
    
    # Discrete search
    alphas = np.linspace(0, 1, 51)
    for alpha in alphas:
        # (f + alpha*d) * (f + alpha*d)_k = conv_f_k + 2*alpha*fd_k + alpha^2*dd_k
        vals = conv_f + 2 * alpha * fd + alpha * alpha * dd
        val = vals.max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    # Golden section refinement
    if best_alpha > 0 and best_alpha < 1:
        gr = (np.sqrt(5) - 1) / 2
        lo = max(0, best_alpha - 0.05)
        hi = min(1, best_alpha + 0.05)
        a1 = hi - gr * (hi - lo)
        a2 = lo + gr * (hi - lo)
        f1 = (conv_f + 2*a1*fd + a1*a1*dd).max()
        f2 = (conv_f + 2*a2*fd + a2*a2*dd).max()
        for _ in range(40):
            if f1 < f2:
                hi = a2
                a2 = a1
                f2 = f1
                a1 = hi - gr * (hi - lo)
                f1 = (conv_f + 2*a1*fd + a1*a1*dd).max()
            else:
                lo = a1
                a1 = a2
                f1 = f2
                a2 = lo + gr * (hi - lo)
                f2 = (conv_f + 2*a2*fd + a2*a2*dd).max()
        best_alpha = (lo + hi) / 2
        best_val = (conv_f + 2*best_alpha*fd + best_alpha*best_alpha*dd).max()
    
    return best_alpha, best_val

def sg_descent(f0, time_limit, lr_init=0.005, decay=0.999):
    """Subgradient descent for fast coarse optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    lr = lr_init
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Find active constraints (top k)
        k = max(10, n // 8)
        top_k = np.argpartition(conv, -k)[-k:]
        
        # Subgradient: gradient of max approximated by weighted sum of active gradients
        # grad_k of (a*a)_k w.r.t. a_j = a_{k-j} + a_{k-j} = 2*a_{k-j} (for interior)
        # More precisely: d(a*a)_k/da_j = a_{k-j} if 0 <= k-j < n (from the a_j term)
        #                              + a_{k-j} if 0 <= k-j < n (from the a_{k-j} term, when j != k-j)
        # Actually, (a*a)_k = sum_i a_i * a_{k-i}, so d/da_j = a_{k-j} + a_{k-j} = 2*a_{k-j}
        # (counting the j and k-j terms; if j == k-j, it's just a_j * 2... 
        #  actually by the product rule, d/da_j (a_j * a_{k-j}) = a_{k-j} + a_j * d(a_{k-j})/da_j
        #  = a_{k-j} + a_j * delta(j, k-j) ... this is getting confusing.
        # Let me just use the formula: grad_j = sum_{k in active} w_k * 2 * a_{k-j} (clipped)
        
        sg = np.zeros(n)
        for k_val in top_k:
            k_int = int(k_val)
            w = 1.0 / len(top_k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            for j in range(js, je + 1):
                # d/da_j of (a*a)_{k_int} = a_{k_int - j} (from the a_j factor)
                # + a_{k_int - j} (from the a_{k_int-j} factor, if j != k_int-j)
                # Actually, (a*a)_k = sum_i a_i * a_{k-i}
                # d/da_j = a_{k-j} + sum_i a_i * d(a_{k-i})/da_j
                #        = a_{k-j} + a_{k-j} (if j is in range for the second term)
                #        = 2 * a_{k-j} (if j != k-j)
                #        = a_j (if j == k-j, i.e., k = 2j)
                # For simplicity, use 2 * a_{k-j}
                idx = k_int - j
                if 0 <= idx < n:
                    sg[j] += 2.0 * w * f[idx]
        
        # Update
        f = project_simplex(f - lr * sg)
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= decay
        
        if no_improve > 200:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            lr = lr_init * 0.5
            no_improve = 0
    
    return best_f, best_obj

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def coordinate_descent(f0, time_limit):
    """Coordinate descent with multiplicative updates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        cur_conv = fast_convolve(f)
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            best_j_val = old_val
            best_j_obj = cur_conv.max()
            
            for mult in [0.5, 0.7, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                ft = f.copy()
                ft[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] *= scale
                ft /= ft.sum()
                val = fast_convolve(ft).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f /= f.sum()
                    if best_j_obj < best_obj - 1e-13:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True
        
        if not improved:
            break
    
    return best_f, best_obj

def make_gauss_init(n, sigma_frac=0.3):
    """Gaussian-shaped initialization."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    f = np.exp(-(x - center)**2 / (2 * sigma**2))
    f = f / f.sum()
    return f

def make_tri_init(n):
    """Triangular initialization."""
    x = np.arange(n)
    f = np.minimum(x, n - 1 - x) + 1
    f = f / f.sum()
    return f

def make_uniform_init(n):
    return np.ones(n) / n

def make_bi_lobe_init(n):
    """Two-lobe initialization."""
    x = np.arange(n)
    c1 = n * 0.25
    c2 = n * 0.75
    sigma = n * 0.1
    f = np.exp(-(x - c1)**2 / (2*sigma**2)) + np.exp(-(x - c2)**2 / (2*sigma**2))
    f = f / f.sum()
    return f
