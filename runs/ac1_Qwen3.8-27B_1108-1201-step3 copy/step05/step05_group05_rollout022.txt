def compute_subgrad(a, tau, top_k=50):
    """Compute smooth subgradient of max(conv(a,a)) using softmax weighting."""
    n = len(a)
    conv = fast_convolve(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]

    pv = conv[peaks]
    scaled = tau * (pv - pv.max())
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        coeff = 2.0 * w[i]
        # Handle the case where i = k-i (middle term counted once)
        if k % 2 == 0 and k // 2 in idx:
            mid = k // 2
            grad[mid] += w[i] * a[mid]  # single count
            mask = idx != mid
            grad[idx[mask]] += coeff * a[k - idx[mask]]
        else:
            grad[idx] += coeff * a[k - idx]

    return grad, conv

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break

        # Adaptive scheduling
        if no_improve > 500:
            momentum = min(0.96, momentum + 0.005)
            lr = max(0.002, lr * 0.95)
            tau = min(200.0, tau * 1.02)
        elif no_improve > 200:
            tau = min(300.0, tau * 1.01)
            momentum = min(0.93, momentum + 0.002)
            lr = max(0.005, lr * 0.98)
        elif no_improve > 50:
            tau = max(20.0, tau * 0.99)
            lr = min(0.06, lr * 1.01)
        else:
            tau = min(150.0, tau * 1.003)
            lr = min(0.05, lr * 1.002)

        grad, conv = compute_subgrad(a, tau, top_k=min(80, n))
        grad_centered = grad - grad.mean()

        v = momentum * v - lr * grad_centered
        a_new = project_simplex(a + v, target_sum=1.0)
        obj_new = fast_convolve(a_new).max()

        # Line search: try to extend
        if obj_new < best_obj:
            for mult in [2.0, 3.0]:
                a_ext = project_simplex(a + mult * v, target_sum=1.0)
                obj_ext = fast_convolve(a_ext).max()
                if obj_ext < obj_new:
                    a_new = a_ext
                    obj_new = obj_ext
                else:
                    break

        a = a_new

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG iter {iter_count}: obj={best_obj:.10f}, lr={lr:.4f}, tau={tau:.1f} ({elapsed:.0f}s)")

        # Restart if stuck
        if no_improve > 800:
            noise_scale = 0.003 + 0.003 * min(no_improve // 800, 5)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale, target_sum=1.0)
            v = np.zeros(n)
            no_improve = 0
            lr = 0.025
            tau = 40.0
            momentum = 0.8

    return best_a, best_obj


def peak_coordinate_descent(a0, time_limit, verbose=False):
    """Coordinate descent that specifically targets peak-contributing coordinates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    pass_count = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        if time.monotonic() - t0 > time_limit - 5:
            break

        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))
        improved = False

        # Focus on coordinates that contribute to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        peak_coords = list(range(js, je + 1))

        # Also include a few random coordinates
        other_coords = [i for i in range(n) if i not in peak_coords]
        if other_coords:
            n_samples = min(20, len(other_coords))
            peak_coords.extend(rng.choice(other_coords, n_samples, replace=False).tolist())

        for j in peak_coords:
            if time.monotonic() - t0 > time_limit - 2:
                break

            old_val = a[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.3]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved and pass_count > 3:
            break

    return best_a, best_obj


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        mag = 1.0
        A[k] = mag * np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def gaussian_modulated_init(n, rng):
    """Gaussian envelope with cosine modulation."""
    center = (n - 1) / 2.0
    x = np.arange(n)
    sigma = n * rng.uniform(0.15, 0.35)
    env = np.exp(-(x - center)**2 / (2 * sigma**2))

    n_mod = rng.integers(1, 6)
    mod = np.ones(n)
    for _ in range(n_mod):
        freq = rng.uniform(1, n // 4)
        amp = rng.uniform(0.05, 0.2)
        phase = rng.uniform(0, 2 * np.pi)
        mod += amp * np.cos(2 * np.pi * freq * x / n + phase)

    a = env * np.maximum(mod, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: LP-based direction finding, aggressive subgradient,
    peak-specific coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        parent_score = float(evaluate_sequence(parent.tolist()))
        best_seq = parent.copy()
        best_score = parent_score
        print(f"No parent, using uniform n=768, score={parent_score:.6f}")

    n_main = len(parent)
    print(f"Main length: n={n_main}")

    # === Phase 1: Quick LP-based optimization on parent ===
    print(f"\n=== Phase 1: LP-based Optimization (best={best_score:.6f}) ===")
    t_phase = min(250, max(30, (deadline - time.monotonic()) * 0.18))
    print(f"  LP t={t_phase:.0f}s, n={n_main}")
    a_lp, obj_lp = lp_based_optimize(parent, time_limit=t_phase, n_constraints=250, verbose=True)
    try_seq(a_lp, "lp_phase1")

    # === Phase 2: Aggressive subgradient ===
    print(f"\n=== Phase 2: Subgradient (best={best_score:.6f}) ===")
    if best_seq is not None:
        t_phase = min(350, max(30, (deadline - time.monotonic()) * 0.22))
        print(f"  SG t={t_phase:.0f}s, n={len(best_seq)}")
        a_sg, obj_sg = optimize_subgrad(best_seq, time_limit=t_phase, verbose=True)
        try_seq(a_sg, "sg_phase2")

    # === Phase 3: LP-based refinement ===
    print(f"\n=== Phase 3: LP Refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        t_phase = min(250, max(20, (deadline - time.monotonic()) * 0.16))
        print(f"  LP refine t={t_phase:.0f}s")
        a_lp2, obj_lp2 = lp_based_optimize(best_seq, time_limit=t_phase, n_constraints=200, verbose=True)
        try_seq(a_lp2, "lp_phase3")

    # === Phase 4: Peak coordinate descent ===
    print(f"\n=== Phase 4: Peak CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        t_phase = min(200, max(20, (deadline - time.monotonic()) * 0.13))
        print(f"  CD t={t_phase:.0f}s")
        a_cd, obj_cd = peak_coordinate_descent(best_seq, time_limit=t_phase)
        try_seq(a_cd, "cd_phase4")

    # === Phase 5: Multi-start subgradient with perturbation ===
    print(f"\n=== Phase 5: Multi-Start SG (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(6):
            rem = deadline - time.monotonic() - 60
            if rem < 30:
                break
            t_alloc = min(rem / max(1, (6 - p)), 80)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.005
                f_init = project_simplex(best_seq + noise, target_sum=1.0)
            elif p == 2:
                f_init = 0.7 * best_seq + 0.3 * best_seq[::-1]
                f_init = project_simplex(f_init, target_sum=1.0)
            elif p == 3:
                f_init = 0.85 * best_seq + 0.15 * (np.ones(n_best) / n_best)
                f_init = project_simplex(f_init, target_sum=1.0)
            elif p == 4:
                f_init = gaussian_modulated_init(n_best, rng)
            else:
                f_init = flat_spectrum_init(n_best, rng)

            a_opt, obj = optimize_subgrad(f_init, time_limit=t_alloc)
            try_seq(a_opt, f"sg_perturb_{p}")

    # === Phase 6: Longer sequences ===
    print(f"\n=== Phase 6: Longer Sequences (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for mult in [1.25, 1.5, 2.0]:
            target_n = int(n_best * mult)
            if target_n <= n_best or target_n > 2560:
                continue
            rem = deadline - time.monotonic() - 40
            if rem < 60:
                break

            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.4, 120)
            print(f"  Long n={target_n}, t={t_alloc:.0f}s")

            a_opt, obj = optimize_subgrad(f_interp, time_limit=t_alloc)
            try_seq(a_opt, f"sg_long_n{target_n}")

            # Quick LP on the longer sequence
            rem = deadline - time.monotonic() - 30
            if rem > 30:
                t_lp = min(rem * 0.5, 80)
                a_opt2, obj2 = lp_based_optimize(a_opt, time_limit=t_lp, n_constraints=150)
                try_seq(a_opt2, f"lp_long_n{target_n}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        # Quick LP
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_lp = min(rem * 0.4, 60)
            a_opt, obj = lp_based_optimize(best_seq, time_limit=t_lp, n_constraints=200)
            try_seq(a_opt, "final_lp")

        # Quick subgradient
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_sg = min(rem * 0.4, 50)
            a_opt, obj = optimize_subgrad(best_seq, time_limit=t_sg)
            try_seq(a_opt, "final_sg")

        # Quick CD
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_opt, obj = peak_coordinate_descent(best_seq, time_limit=min(rem, 40))
            try_seq(a_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
