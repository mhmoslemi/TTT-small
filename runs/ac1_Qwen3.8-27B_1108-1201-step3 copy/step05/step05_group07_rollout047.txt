**Approach 6: Exploit the connection to spectral flatness**

The DFT of `a` has power spectrum `|A(f)|^2`. The convolution `conv(a,a)` is related to the aperiodic autocorrelation, not the periodic one. But for "almost" periodic sequences (where the boundaries don't matter much), the aperiodic and periodic autocorrelations are similar.

For periodic sequences, the aperiodic autocorrelation at lag k is approximately `|A(k/n)|^2 / n` (by Parseval). So minimizing the max of the aperiodic autocorrelation is like minimizing the max of the power spectrum, i.e., making the spectrum as flat as possible.

A flat power spectrum means the sequence is "white" - like noise. But we're constrained to non-negative values, which makes this harder.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an LP-based iterative
water-filling approach combined with soft-max L-BFGS polishing.
The LP approach finds the best direction to move within the current
"envelope" (max convolution bound), providing a fundamentally different
optimization path from pure gradient methods.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    """Compute the score locally (same as evaluate_sequence but faster)."""
    a = np.maximum(a, 0.0)
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S ** 2))

def cross_corr(w, a):
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:n]

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()

def z_from_a(a):
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    """
    Build the constraint matrix for the LP.
    A[k, j] = a[k-j] if 0 <= k-j < n, else 0.
    Shape: (2n-1) x n
    """
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A

def build_conv_matrix_fast(a):
    """
    Build the constraint matrix using scipy's tolu or manual approach.
    Actually for n up to ~2000, the dense approach is fine.
    Let's use a more efficient construction.
    """
    n = len(a)
    # The matrix is a "convolution" matrix. Row k has a[k-j] at column j
    # for valid j. This is equivalent to a block structure.
    # For efficiency, use scipy.sparse
    from scipy.sparse import csr_matrix
    # Row k: nonzero at columns [max(0,k-n+1), min(k,n-1)] with values a[k-j]
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        js = range(j_start, j_end + 1)
        rows.extend([k] * (j_end - j_start + 1))
        cols.extend(list(js))
        vals.extend(list(a[k - j] for j in js))
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_iteration(a, time_limit=10.0):
    """
    One LP iteration: given current a, find the best g that satisfies
    the current envelope M = max(conv(a,a)).
    Returns g (same length as a, non-negative, same sum as a).
    """
    n = len(a)
    S = a.sum()
    M = fast_convolve(a).max()
    
    if M <= 0:
        return a.copy()
    
    # Build constraint matrix
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # Constraints: A @ g <= M * ones(2n-1)
    # Maximize: ones(n) @ g
    # Subject to: g >= 0
    
    # For efficiency, build the matrix using a structured approach
    # The matrix is a "convolution matrix" - it's a banded Toeplitz-like matrix
    
    # Use scipy lstsq or direct construction
    # For n up to ~2000, dense is fine
    if n <= 3000:
        A = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            A[k, j_start:j_end+1] = a[k - j_end:k - j_start + 1][::-1]
    else:
        # Use sparse
        from scipy.sparse import csr_matrix
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(a[k - js].tolist())
        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # LP: min -1^T g s.t. A g <= M, g >= 0
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs')
        if res.success:
            g = res.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                g = g * (S / g_sum)
                return g
    except Exception:
        pass
    
    return a.copy()

            if score < best_step_score - 1e-12:

def soft_max_obj_grad(z, T, n):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    grad_a = 2.0 * n * cross_corr(w, a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z

def lbgfs_optimize(z0, T, n, time_limit, maxiter=3000):
    t0 = time.monotonic()
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        obj, _ = soft_max_obj_grad(z, T, n)
        return obj
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = soft_max_obj_grad(z, T, n)
        return g
    
    try:
        res = minimize(
            fun=func,
            x0=z0,
            jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 25, 'disp': False}
        )
        return res.x, res.fun
    except Exception:
        return z0, func(z0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def peak_shave(a, n_peaks=5, n_steps=200, time_limit=30.0):

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a two-phase approach:
(1) LP-based best-response iteration (minimax LP) for fast convergence,
(2) Soft-max L-BFGS + subgradient polishing for final refinement.
The key innovation over the previous solution is the LP-based direction
finding, which provides more globally informed updates than gradient methods.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time
from scipy.sparse import csr_matrix


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    a = np.maximum(a, 0.0)
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S ** 2))


def cross_corr_fft(w, a):
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:n]
