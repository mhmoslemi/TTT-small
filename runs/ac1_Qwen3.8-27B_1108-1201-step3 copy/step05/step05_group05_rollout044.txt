import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def flat_spectrum_init(n, rng):

        k = peak_idx

        je = min(n - 1, k)

        contribs = a[idx] * a[k - idx]
        total_contrib = contribs.sum()
        
        if total_contrib < 1e-15:
            break
        
        # Find the position with the largest contribution
        # and try to reduce it by redistributing mass
        max_contrib_pos = idx[np.argmax(contribs)]
        
        # Try moving some mass from this position to a "less harmful" position

        # and ideally contributes to a low-convolution lag
        
        for _ in range(20):
            src = max_contrib_pos
            if a[src] < 1e-10:
                break
            
            # Choose a destination that minimizes impact on the peak
            # randomly from positions that don't pair with src to form the peak
            possible_dst = np.array([m for m in range(n) if m != src and (m + (k - m) != k or m == k - m)])
            if len(possible_dst) == 0:
                possible_dst = np.setdiff1d(np.arange(n), [src])
            
            dst = possible_dst[rng.integers(0, len(possible_dst))]
            
            for frac in [0.01, 0.005, 0.002]:
                amount = frac * a[src]
                a_trial = a.copy()
                a_trial[src] -= amount
                a_trial[dst] += amount
                new_obj = fast_convolve(a_trial).max()
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a_trial.copy()

def coordinate_descent_v2(a0, time_limit):

            old_remaining = 1.0 - a[j]

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def spectral_init(n, rng, n_peaks=5):
    """Initialize using spectral shaping."""
    # Create a flat-ish spectrum with some random structure
    A = np.ones(n, dtype=complex)
    A[0] = np.sqrt(n)
    # Add some random phase
    phases = rng.standard_normal(n) * 0.5
    A *= np.exp(1j * phases)
    # Ensure Hermitian symmetry
    for k in range(1, (n // 2) + 1):
        A[n - k] = np.conj(A[k])
    
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s
