        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = x.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    FISTA-LSE iter {it}: obj={best_obj:.10f}, eps={eps:.6f}")
        else:
            no_improve += 1

        # Adaptively decrease eps and step
        if it % 30 == 0:
            eps = max(eps * 0.88, eps_min)
            step = max(step * 0.92, 0.01)

        # Stuck detection: perturb and reset
        if no_improve > 200:
            rng = np.random.default_rng(it * 7 + 3)
            noise = rng.standard_normal(n) * 0.005
            y = project_simplex(best_f + noise)
            f_prev = y.copy()
            t_fista = 1.0
            step = 0.3
            eps = max(eps * 1.5, eps_min)
            no_improve = 0

        if step < 0.01 and eps <= eps_min * 1.1:
            break

    return best_f, best_obj


def subgradient_mirror_descent(f0, time_limit, verbose=False):
    """Subgradient method with multiplicative weight updates (mirror descent)."""
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    t0 = time.monotonic()
    lr = 0.01
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(f)
        c_max = c.max()
        peaks = np.where(c >= c_max - 1e-10)[0]

        # Use top few peaks with softmax weighting
        if len(peaks) > 15:
            top = np.argpartition(c, -15)[-15:]
        else:
            top = peaks

        pv = c[top]
        temp = 0.01
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        # Compute subgradient
        sg = np.zeros(n)
        for idx in range(len(top)):
            if w[idx] < 1e-12:
                continue
            k = int(top[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                ki = k - j
                coeff = 2.0
                if ki == j:
                    coeff = 1.0
                sg[j] += w[idx] * coeff * f[ki]

        # Multiplicative update
        f = f * np.exp(-lr * sg)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    SG-MD iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        lr *= 0.999
        if no_improve > 500:
            rng = np.random.default_rng(it * 13 + 5)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-15)
            f = f / f.sum()
            lr = 0.005
            no_improve = 0

        if lr < 1e-8:
            break

    return best_f, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.15
            if m_exp > 8.0:
                rng = np.random.default_rng(it * 17 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.1
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 30:
                m_exp = min(m_exp * 1.08, 5.0)
            if no_improve > 100:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue
        else:
            no_improve = 0

        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            m_exp = 1.0
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 5.0)

        if no_improve > 100:
            break

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with pairwise swap optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(0, n, max(1, n // 500)):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue
            best_j_val = old_val
            best_j_obj = best_obj
            for mult in [0.5, 0.7, 0.85, 1.15, 1.4]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.3):
    """Create a Gaussian-like initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    """Create a bell-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_asymmetric(n, decay=0.3):
    """Create an asymmetric (exponential decay) initial sequence."""
    x = np.arange(n).astype(np.float64)
    a = np.exp(-x / (n * decay))
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_wavy(n, freq=3.0, amp=0.3):
    """Create a wavy initial sequence."""
    x = np.arange(n).astype(np.float64)
    base = np.ones(n)
    wave = 1.0 + amp * np.sin(2.0 * np.pi * freq * x / n)
    a = base * wave
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

    def try_seq(a, label=""):

        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})", flush=True)
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}", flush=True)
            if p_score < best_score:
                best_score = p_score

    except Exception:

    if parent is None:
        parent = make_gaussian(768, 0.3)

    # === Phase 1: Quick evaluation of diverse initializations ===
    print("=== Phase 1: Diverse Initializations ===", flush=True)
    init_candidates = []

        init_candidates.append(make_gaussian(n_try, 0.25))

    for a_init in init_candidates:

    if parent is not None:
        init_candidates.append(parent)
        try_seq(parent, "parent")

    # Find best initialization for further optimization
    best_init = None
    best_init_obj = float('inf')
    for a_init in init_candidates:
        s = a_init.sum()
        if s < 1e-10:
            continue
        a_norm = a_init / s
        conv = fast_convolve(a_norm)
        obj = 2 * len(a_norm) * conv.max()
        if obj < best_init_obj:

            best_init = a_norm.copy()

    if best_init is None:
        best_init = parent
    print(f"  Best init: obj={best_init_obj:.6f} (n={len(best_init)})", flush=True)

    # === Phase 3: Subgradient Mirror Descent ===

            try_seq(a_opt, "sg_md")

    if best_seq is not None:

        rem = deadline - time.monotonic() - 300
        if rem > 60:

        for vname, es, em in variants:

            if rem < 30:

    # === Phase 7: LP-FW with perturbation ===
    if best_seq is not None:
        print(f"\n=== Phase 7: LP-FW variants ===", flush=True)
        n_best = len(best_seq)
        for p in range(3):

    # Final verification

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})", flush=True)
    return [float(x) for x in best_seq]
