def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])

    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0. Returns g/sum(g) or None."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            g = res.x
            gs = g.sum()
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None

def make_geometric(n, r):
    """Geometric sequence normalized to sum=1."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_symmetric_geo(n, r):
    """Symmetric geometric (rising then falling)."""
    half = (n + 1) // 2
    asc = r ** np.arange(half, dtype=np.float64)
    desc = r ** np.arange(half - 1, 0, -1, dtype=np.float64)
    a = np.concatenate([asc, desc])
    if len(a) < n:
        a = np.concatenate([a, np.ones(n - len(a))])
    a = a / a.sum()
    return a


def make_cosine(n, power=2.0):
    """Cosine window raised to power."""
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** power + 0.005

def make_power_law(n, alpha):

    i = np.arange(n, dtype=np.float64)
    a = (1.0 + i / n) ** (-alpha)
    a = a / a.sum()

def make_bump(n, center_frac, width_frac):

    i = np.arange(n, dtype=np.float64)
    center = center_frac * (n - 1)
    width = width_frac * (n - 1) / 2.0
    a = np.exp(-(i - center) ** 2 / (2.0 * width ** 2))
    a = a / a.sum()
    return a


def interpolate_seq(f, new_n):
    """Linear interpolation to new length, normalized to sum=1."""
    old_n = len(f)
    if old_n == new_n:

    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:

    return f_interp

        _try(a_opt, "nest_mixed")
    
    # === Phase 2: LP Frank-Wolfe ===
    print(f"\n=== Phase 2: LP-FW (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 450
    if rem > 30:
        t_alloc = min(rem, 200)
        print(f"  LP-FW n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = lpfw_optimize(best_seq, time_limit=t_alloc, verbose=True)
        _try(a_opt, "lpfw_p2")
    
    # LP-FW with perturbation
    rem = deadline - time.monotonic() - 350
    if rem > 25:
        noise = rng.standard_normal(len(best_seq)) * 0.005
        f_pert = project_simplex(best_seq + noise)
        t_alloc = min(rem, 120)
        a_opt, _ = lpfw_optimize(f_pert, time_limit=t_alloc)
        _try(a_opt, "lpfw_pert")
    
    # === Phase 3: More Nesterov from best ===
    print(f"\n=== Phase 3: More Nesterov (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 300
    if rem > 30:
        t_alloc = min(rem, 250)
        print(f"  Nesterov round 2, t={t_alloc:.0f}s")
        a_opt, _ = nesterov_smooth(best_seq, time_limit=t_alloc, T_start=0.2, T_end=0.003, verbose=True, base_lr=0.008)
        _try(a_opt, "nest_p3")
    
    # === Phase 4: Subgradient polishing ===
    print(f"\n=== Phase 4: Subgradient Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 20:
        t_alloc = min(rem, 150)
        print(f"  Subgrad t={t_alloc:.0f}s")
        a_opt, _ = subgradient_polish(best_seq, time_limit=t_alloc, verbose=True)
        _try(a_opt, "subgrad_p4")
    
    # === Phase 5: Try longer sequences ===
    print(f"\n=== Phase 5: Longer Sequences (best={best_score:.6f}) ===")
    for target_n in [1024, 1280, 1536]:
        rem = deadline - time.monotonic() - 100
        if rem < 40:
            break
        f_interp = interpolate_seq(best_seq, target_n)
        
        t1 = min(rem * 0.4, 80)
        a_opt, _ = nesterov_smooth(f_interp, time_limit=t1, T_start=0.3, T_end=0.01, base_lr=0.008)
        _try(a_opt, f"nest_n{target_n}")
        
        rem = deadline - time.monotonic() - 80
        if rem > 25:
            t2 = min(rem * 0.5, 60)
            a_opt, _ = lpfw_optimize(f_interp, time_limit=t2)
            _try(a_opt, f"lpfw_n{target_n}")
    
    # === Phase 6: Final fine polishing ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            a_opt, _ = nesterov_smooth(best_seq, time_limit=min(rem, 80),
                                       T_start=0.03, T_end=0.001, verbose=True, base_lr=0.004)
            _try(a_opt, "nest_fine")
        
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            a_opt, _ = subgradient_polish(best_seq, time_limit=min(rem, 50))
            _try(a_opt, "subgrad_fine")
        
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_opt, _ = lpfw_optimize(best_seq, time_limit=min(rem, 30))
            _try(a_opt, "lpfw_fine")
        
        # Multiple small perturbations
        for idx in range(5):
            rem = deadline - time.monotonic() - 15
            if rem < 12:
                break
            noise = rng.standard_normal(n_best) * 0.0015
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = nesterov_smooth(f_pert, time_limit=min(rem, 20),
                                       T_start=0.02, T_end=0.001, base_lr=0.003)
            _try(a_opt, f"perturb_{idx}")
        
        # Reversed
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            f_rev = best_seq[::-1].copy()
            a_opt, _ = nesterov_smooth(f_rev, time_limit=min(rem, 15),
                                       T_start=0.015, T_end=0.001, base_lr=0.003)
            _try(a_opt, "reversed")
        
        # Slight blend with uniform
        rem = deadline - time.monotonic() - 8
        if rem > 8:
            for blend in [0.95, 0.90, 0.85]:
                f_blend = blend * best_seq + (1 - blend) * (np.ones(n_best) / n_best)
                f_blend = project_simplex(f_blend)
                a_opt, _ = nesterov_smooth(f_blend, time_limit=min(rem / 3, 8),
                                           T_start=0.01, T_end=0.001, base_lr=0.002)
                _try(a_opt, f"blend_{blend}")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
