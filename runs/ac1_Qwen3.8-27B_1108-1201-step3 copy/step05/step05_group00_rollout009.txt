"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth gradient descent
with log-sum-exp peak approximation, multiplicative mirror descent updates,
annealed temperature scheduling, multi-length exploration, and coordinate
descent polish. Key innovation: FFT-based cross-correlation gradient that
uses all peaks (not just top-k) with adaptive softmax weighting.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_gradient(a, p, n):
    """
    Compute sg[j] = sum_k p[k] * a[k-j] via FFT cross-correlation.
    p has length 2n-1, a has length n.
    sg[j] = sum_m a[m] * p[j+m] for m in [0, n-1]
    """
    N = 1
    while N < 3 * n:
        N *= 2
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    # sg[j] = sum_k a[k] * p[j+k] = ifft(fft(p) * conj(fft(a)))[j]
    sg = np.fft.irfft(P * np.conj(A), N)[:n]
    return sg

    best_obj = conv.max()
    best_a = a.copy()

    no_improve = 0
    stagnation_threshold = 50

        # Adaptive tau and lr

        lr = lr_start * (lr_end / lr_start) ** progress

        scaled = tau * conv
        scaled -= scaled.max()
        p = np.exp(scaled)

def coordinate_descent_polish(a0, time_limit, verbose=False):
    """
    Coordinate descent that optimizes each coordinate while maintaining sum=1.
    Uses quadratic interpolation for the 1D subproblem.
    """
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    t0 = time.monotonic()
    improved_any = False

    # Precompute: for each coordinate j, the "other" contribution to each conv[k]
    # conv[k] = f[j]*f[k-j] + (other contributions)
    # When we change f[j] to x, conv[k] = x*f[k-j] + rest[k]
    # where rest[k] = conv[k] - f[j]*f[k-j]

    while time.monotonic() - t0 < time_limit:
        improved_any = False
        conv = fast_convolve(f)

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            # The contribution of f[j] to conv[k] is f[j] * f[k-j]
            # So rest[k] = conv[k] - f[j] * f[k-j] (for valid k-j)
            # When f[j] = x: new_conv[k] = x * f[k-j] + rest[k]
            # We want to minimize max(new_conv)

            # Find the peaks that this coordinate contributes to
            # k-j must be in [0, n-1], so k in [j, j+n-1]

            k_end = min(2 * n - 2, j + n - 1)

            if k_end < k_start:
                continue

            # = conv[k] + (x - x_old) * f[k-j]
            # We want to minimize max over k of: conv[k] + (x - x_old) * f[k-j]

            # This is a 1D minimax problem!
            # Let d = x - x_old. For each k: value_k = conv[k] + d * f[k-j]
            # We want min_x max_k value_k
            # The optimal d is where the max is minimized.
            #
            # For a given d, max_k (conv[k] + d * f[k-j]) is piecewise linear in d.
            # The minimum of a convex piecewise linear function is where two lines cross.
            #

            # Actually, if we change f[j] from old_val to x, we need to redistribute
            # (old_val - x) among other coordinates. For simplicity, let's just
            # scale the other coordinates: f'[j] = x, f'[k≠j] = f[k] * (1-x)/(1-old_val)
            # But this changes all conv values... Let's use a simpler approach.

            best_local_obj = conv.max()

                if x < 0 or x > 0.8:

                    continue

                f_trial = f.copy()
                f_trial[j] = x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:

def make_gaussian(n, sigma_frac=0.25):

def make_bell(n, width_frac=0.5):
    """Bell-shaped with flat top."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

                                             lr_start=0.5, lr_end=0.1, verbose=verbose)
        if verbose:
            print(f"    After SMD1: obj={obj:.10f}")

    # Stage 2: Fine SMD
    rem = time_limit - (time.monotonic() - t_start)
    t2 = min(rem * 0.3, 200)
    if t2 > 10:
        if verbose:
            print(f"    Stage 2 (SMD fine): {t2:.0f}s")
        current, obj = smooth_mirror_descent(current, t2, tau_start=10.0, tau_end=200.0,
                                             lr_start=0.3, lr_end=0.05, verbose=verbose)
        if verbose:
            print(f"    After SMD2: obj={obj:.10f}")

    # Stage 3: Alternative SMD with different schedule
    rem = time_limit - (time.monotonic() - t_start)
    t3 = min(rem * 0.25, 150)
    if t3 > 10:
        if verbose:
            print(f"    Stage 3 (SMD alt): {t3:.0f}s")
        current, obj = smooth_mirror_descent(current, t3, tau_start=20.0, tau_end=500.0,
                                             lr_start=0.2, lr_end=0.02, verbose=verbose)
        if verbose:
            print(f"    After SMD3: obj={obj:.10f}")

    # Stage 4: Coordinate descent
    rem = time_limit - (time.monotonic() - t_start)
    t4 = min(rem * 0.5, 100)
    if t4 > 5:
        if verbose:
            print(f"    Stage 4 (CD): {t4:.0f}s")
        current, obj = coordinate_descent_polish(current, t4, verbose=verbose)
        if verbose:
            print(f"    After CD: obj={obj:.10f}")

    return current


def perturbed_restart(best_a, n, rng, scale=0.003):
    """Create a perturbed version of the best sequence."""
    noise = rng.standard_normal(n) * scale
    a_new = best_a + noise
    a_new = np.maximum(a_new, 1e-15)
    a_new = a_new / a_new.sum()
    return a_new


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage optimization combining smooth mirror descent with annealed
    temperature, multi-length exploration, diverse initializations, and
    coordinate descent polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Quick scan of initializations ===
    print("=== Phase 1: Initialization Scan ===")
    init_pool = []

    if parent is not None:
        init_pool.append(("parent", parent))

    for n_try in [384, 512, 768, 1024]:
        init_pool.append((f"gauss_{n_try}", make_gaussian(n_try, 0.25)))
        init_pool.append((f"bell_{n_try}", make_bell(n_try, 0.55)))
        init_pool.append((f"cos_{n_try}", make_cosine_window(n_try)))
        init_pool.append((f"interf_{n_try}", make_interference(n_try)))

    # Score all inits (using the normalized objective)
    scored_inits = []
    for label, a_init in init_pool:
        a_norm = a_init / a_init.sum()
        conv = fast_convolve(a_norm)
        obj = 2 * len(a_norm) * conv.max()
        scored_inits.append((obj, label, a_norm))
    scored_inits.sort(key=lambda x: x[0])

    print(f"  Top 5 inits:")
    for obj, label, _ in scored_inits[:5]:
        print(f"    {label}: obj={obj:.6f}")

    # === Phase 2: Optimize top candidates ===
    print("\n=== Phase 2: Multi-stage Optimization ===")
    # Allocate time to top candidates
    top_k = min(4, len(scored_inits))
    candidates_to_optimize = scored_inits[:top_k]

    # Give more time to better candidates
    weights = []
    for i, (obj, label, a_init) in enumerate(candidates_to_optimize):
        w = 1.0 / (1.0 + 0.5 * i)
        weights.append(w)
    total_w = sum(weights)
    weights = [w / total_w for w in weights]

    # Also ensure we try the parent if it's good
    if parent is not None and best_score < 1.55:
        if not any("parent" in label for _, label, _ in candidates_to_optimize):
            candidates_to_optimize.append((best_score / 2, "parent_extra", parent))
            weights.append(0.5)
            total_w = sum(weights)
            weights = [w / total_w for w in weights]

    for idx, (init_obj, label, a_init) in enumerate(candidates_to_optimize):
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_alloc = rem * weights[idx] * 0.8
        t_alloc = max(15, min(t_alloc, 400))
        n_init = len(a_init)

        print(f"\n  Optimizing [{label}] (n={n_init}, init_obj={init_obj:.6f}, t={t_alloc:.0f}s)")
        try:
            result = multi_stage_optimize(a_init, time_limit=t_alloc, verbose=True)
            try_seq(result, label)
        except Exception as e:
            print(f"    Error: {e}")

    # === Phase 3: Interpolation to different lengths ===
    print("\n=== Phase 3: Length Exploration ===")
    if best_seq is not None:
        best_n = len(best_seq)
        for target_n in [best_n + 128, best_n + 256, best_n - 128, 1024, 1536]:
            if target_n < 128 or target_n > 3000:
                continue
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = try_seq(f_interp, f"interp_{target_n}")

            t_alloc = min(rem * 0.15, 100)
            if t_alloc > 10:
                print(f"  Optimizing interp n={target_n}: t={t_alloc:.0f}s")
                result = multi_stage_optimize(f_interp, time_limit=t_alloc)
                try_seq(result, f"opt_interp_{target_n}")

    # === Phase 4: Multiple restarts from best ===
    print("\n=== Phase 4: Restart & Refine ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for restart_idx in range(8):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            t_alloc = min(rem / (8 - restart_idx), 80)

            # Different perturbation strategies
            if restart_idx == 0:
                f_init = best_seq.copy()
                tag = "direct"
            elif restart_idx == 1:
                f_init = perturbed_restart(best_seq, n_best, rng, 0.002)
                tag = "noise1"
            elif restart_idx == 2:
                f_init = perturbed_restart(best_seq, n_best, rng, 0.005)
                tag = "noise2"
            elif restart_idx == 3:
                # Blend with uniform
                f_init = 0.9 * best_seq + 0.1 * (np.ones(n_best) / n_best)
                f_init = f_init / f_init.sum()
                tag = "blend_uniform"
            elif restart_idx == 4:
                # Reverse symmetry
                f_init = 0.5 * best_seq + 0.5 * best_seq[::-1]
                f_init = f_init / f_init.sum()
                tag = "sym"
            elif restart_idx == 5:
                # Slight shift
                f_init = np.zeros(n_best)
                shift = rng.integers(1, max(2, n_best // 20))
                f_init[shift:] = best_seq[:n_best - shift]
                f_init[:shift] = best_seq[n_best - shift:]
                f_init = f_init / f_init.sum()
                tag = f"shift{shift}"
            elif restart_idx == 6:
                # Blend with Gaussian
                gauss = make_gaussian(n_best, 0.3)
                f_init = 0.85 * best_seq + 0.15 * gauss
                f_init = f_init / f_init.sum()
                tag = "blend_gauss"
            else:
                # Larger perturbation
                f_init = perturbed_restart(best_seq, n_best, rng, 0.01)
                tag = "noise3"

            print(f"  Restart [{tag}]: t={t_alloc:.0f}s")
            result = multi_stage_optimize(f_init, time_limit=t_alloc)
            try_seq(result, f"restart_{tag}")

    # === Phase 5: Final intensive SMD on best ===
    print("\n=== Phase 5: Final Intensive Optimization ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t_alloc = min(rem * 0.6, 200)
            print(f"  Final SMD: t={t_alloc:.0f}s")
            # Use very high tau for precise max targeting
            n_best = len(best_seq)
            a_smd, obj = smooth_mirror_descent(best_seq, t_alloc,
                                                tau_start=50.0, tau_end=1000.0,
                                                lr_start=0.15, lr_end=0.01)
            try_seq(a_smd, "final_smd")

            # Coordinate descent on result
            rem = deadline - time.monotonic()
            if rem > 15:
                t_cd = min(rem * 0.5, 60)
                print(f"  Final CD: t={t_cd:.0f}s")
                a_cd, obj = coordinate_descent_polish(best_seq, t_cd)
                try_seq(a_cd, "final_cd")

    # === Phase 6: Try a completely different approach - frequency shaping ===
    print("\n=== Phase 6: Frequency Domain Approach ===")
    rem = deadline - time.monotonic()
    if rem > 60 and best_seq is not None:
        n_best = len(best_seq)
        t_alloc = min(rem * 0.4, 120)

        # Shape the Fourier transform to be flat (minimizes autocorrelation peaks)
        # Target: |A(f)|^2 = constant (radar code / Barker-like)
        # We optimize in the "amplitude" domain
        f_current = best_seq.copy()
        for fd_iter in range(int(t_alloc / 0.05) if t_alloc < 5 else 20):
            if time.monotonic() > deadline - 5:
                break
            A = np.fft.rfft(f_current)
            # Target: flat amplitude
            target_amp = np.ones(len(A)) * np.mean(np.abs(A))
            # Current amp
            cur_amp = np.abs(A)
            phase = np.angle(A)
            # Move towards flat amplitude
            alpha = 0.1
            new_amp = cur_amp + alpha * (target_amp - cur_amp)
            new_A = new_amp * np.exp(1j * phase)
            new_f = np.fft.irfft(new_A, n_best)
            new_f = np.maximum(new_f, 0)
            new_f = new_f / new_f.sum()
            obj = fast_convolve(new_f).max()
            if obj < fast_convolve(f_current).max():
                f_current = new_f
            if time.monotonic() > deadline - 5:
                break

        try_seq(f_current, "freq_shape")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
