def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    s = e.sum()

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast coordinate descent with
exact 1D optimization, combined with LP-FW for escaping local minima and
multi-start strategies with U-shaped initializations.
"""

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

    rho = np.nonzero(cond)[0][-1]

    return np.maximum(v - theta, 0.0)

    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s

def make_u_shape(n, p=2.0, c=0.0):
    """Create a U-shaped sequence of length n."""
    x = np.linspace(-1, 1, n)
    a = 1.0 + p * x**2 + c
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_bimodal(n, width=0.1):
    """Create a bimodal sequence with peaks at both ends."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - width) / (width/3))**2) + np.exp(-((x - 1 + width) / (width/3))**2)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def eval_1d(beta, alpha, T, C):
    """Evaluate 1D objective for coordinate descent."""
    one_minus_alpha = 1.0 - alpha
    if abs(one_minus_alpha) < 1e-15:
        # Degenerate case: alpha ≈ 1
        if beta > 0.999:
            return beta * beta
        return 1.0
    s = (1.0 - beta) / one_minus_alpha
    s2 = s * s
    val0 = beta * beta + s2 * T[0]
    if len(C) > 1:
        vals = s2 * T[1:] + beta * s * C[1:]
        val_rest = np.max(vals)
    else:
        val_rest = 0.0
    return max(val0, val_rest)

def coord_descend(f0, time_limit, verbose=False):
    """Fast coordinate descent with exact 1D optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    b = fast_convolve(f)
    best_obj = b.max()
    
    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        pass_improved = False
        prev_obj = best_obj
        
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            alpha = f[i]
            if alpha < 1e-14:
                continue
            if alpha > 1.0 - 1e-14:
                continue  # Can't reduce this
            
            # Compute C (length 2n-1)
            C = np.zeros(2 * n - 1)
            # C[k] += f[i+k] for k where 0 <= i+k < n
            # k ranges from 0 to n-1-i
            C[0:n-i] += f[i:n]
            # C[k] += f[i-k] for k where 0 <= i-k < n
            # k ranges from max(0, i-n+1) to min(i, 2n-2)
            lo = max(0, i - n + 1)
            hi = min(i, 2 * n - 2)
            if hi >= lo:
                # f[i-k] for k=lo,...,hi
                # = f[i-lo], f[i-lo-1], ..., f[i-hi]
                C[lo:hi+1] += f[i-hi:i-lo+1][::-1]
            # Subtract f[i] for k=0 (double counting)
            C[0] -= f[i]
            
            # Compute T
            T = np.empty(2 * n - 1)
            T[0] = b[0] - alpha * alpha
            T[1:] = b[1:] - alpha * C[1:]
            
            # Find optimal beta
            # Grid search
            grid = np.linspace(0.0, 0.9999, 30)
            # Add alpha to grid
            grid = np.insert(grid, 15, alpha)
            grid = np.sort(np.unique(grid))
            
            vals = np.empty(len(grid))
            for gi, bet in enumerate(grid):
                vals[gi] = eval_1d(bet, alpha, T, C)
            
            idx = np.argmin(vals)
            best_beta = grid[idx]
            best_val = vals[idx]
            
            # Golden section refinement in neighborhood
            lo_b = grid[max(0, idx - 1)]
            hi_b = grid[min(len(grid) - 1, idx + 1)]
            
            for _ in range(40):
                m1 = lo_b + 0.381966 * (hi_b - lo_b)
                m2 = lo_b + 0.618034 * (hi_b - lo_b)
                v1 = eval_1d(m1, alpha, T, C)
                v2 = eval_1d(m2, alpha, T, C)
                if v1 <= v2:
                    hi_b = m2
                else:
                    lo_b = m1
                if hi_b - lo_b < 1e-13:
                    break
            
            mid = (lo_b + hi_b) / 2.0
            v_mid = eval_1d(mid, alpha, T, C)
            if v_mid < best_val:
                best_beta = mid
                best_val = v_mid
            
            # Also check exact endpoints
            v0 = eval_1d(0.0, alpha, T, C)
            if v0 < best_val:
                best_beta = 0.0
                best_val = v0
            
            # Check near 1
            v1 = eval_1d(1.0 - 1e-10, alpha, T, C)
            if v1 < best_val:
                best_beta = 1.0 - 1e-10
                best_val = v1
            
            if best_val < best_obj - 1e-13:
                f[i] = best_beta
                fs = f.sum()
                if fs > 1e-15:
                    f = f / fs
                b = fast_convolve(f)
                cur_obj = b.max()
                if cur_obj < best_obj - 1e-14:
                    best_obj = cur_obj
                    best_f = f.copy()
                    pass_improved = True

            elapsed = time.monotonic() - t0
            print(f"    CD iter {it}: obj={best_obj:.10f} (t={elapsed:.1f}s)")

                if b.max() < best_obj - 1e-14:

                    best_f = f.copy()
    
    return best_f, best_obj

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    b = fast_convolve(f)
    M = b.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.15
            if m_exp > 10.0:
                m_exp = 1.0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 50:
                break
            m_exp = min(m_exp * 1.08, 5.0)
            continue
        
        best_alpha = 0.0
        best_val = M
        
        for alpha in [0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            b = fast_convolve(f_new)
            M = b.max()
            m_exp = 1.0
            f = f_new
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)
        
        if no_improve > 80:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.003 + 0.003 * min(no_improve // 80, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            b = fast_convolve(f)
            M = b.max()
            m_exp = 1.0
            no_improve = 0
    
    return best_f, best_obj

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
