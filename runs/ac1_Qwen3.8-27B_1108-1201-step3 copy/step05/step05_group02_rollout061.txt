    """Compute subgradient of max_{k in k_peaks} (f*f)[k] w.r.t. f."""
    n = len(f)
    sg = np.zeros(n)

        # d/d f[j] (f*f)[k] = 2*f[k-j] for valid j

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using mirror descent with exponential
weights, reduced-constraint LP-FW, coordinate descent, and multi-length search.
Key innovations: fast subgradient via FFT, adaptive peak tracking, diverse
restarts with structured initializations, and efficient time allocation.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """FFT-based linear auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, peaks):
    """Compute subgradient of sum_{k in peaks} (f*f)[k] w.r.t. f."""
    n = len(f)
    sg = np.zeros(n)
    for k in peaks:
        kk = int(k)
        js = max(0, kk - n + 1)
        je = min(n - 1, kk)
        if js <= je:
            idx = kk - np.arange(js, je + 1)
            sg[js:je + 1] += 2.0 * f[idx]
    return sg


def mirror_descent(f0, time_limit, top_k=80, eta_init=0.005, verbose=False):
    """Mirror descent with exponential weights for minimizing max auto-convolution."""
    f = np.maximum(f0.astype(np.float64).copy(), 1e-15)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    eta = eta_init
    no_improve = 0
    it = 0
    smooth_temp = 0.001

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        # Soft peaks: weight by exp((conv[k]-max)/temp) for smoother gradient
        pv = conv[peaks]
        scaled = (pv - pv.max()) / smooth_temp
        w = np.exp(np.clip(scaled, -20, 0))
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= w_sum

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-10:
                continue
            kk = int(peaks[i])
            js = max(0, kk - n + 1)
            je = min(n - 1, kk)
            if js <= je:
                idx = kk - np.arange(js, je + 1)
                sg[js:je + 1] += 2.0 * w[i] * f[idx]

        # Mirror descent update: f_j *= exp(-eta * sg_j)
        f = f * np.exp(-eta * sg)
        f = np.maximum(f, 1e-300)
        s = f.sum()
        if s < 1e-15:
            f = np.ones(n) / n
        else:
            f = f / s

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive step size
        if no_improve > 100:
            eta *= 0.85
            smooth_temp *= 1.05
        elif no_improve == 50:
            eta *= 0.95

        # Restart strategy
        if no_improve > 500:
            rng = np.random.default_rng(it * 13 + 7)
            scale = 0.003 + 0.003 * min((no_improve - 500) // 200, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            f = np.maximum(f, 1e-15)
            f = f / f.sum()
            conv = fast_convolve(f)
            eta = max(eta_init * 0.5, eta * 0.5)
            smooth_temp = 0.001
            no_improve = 0

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter {it} ({elapsed:.0f}s): obj={best_obj:.10f}, eta={eta:.6f}")

    return best_f, best_obj


def lp_fw_reduced(f0, time_limit, top_k_constraints=60, verbose=False):
    """LP-FW with only top-k convolution peaks as constraints (much faster LP)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Pick top-k constraints
        k_con = min(top_k_constraints, len(conv))
        peak_idx = np.argpartition(conv, -k_con)[-k_con:]
        peak_idx_sorted = sorted(peak_idx.tolist(), key=lambda x: -conv[x])

        # Build LP: max sum(g) s.t. (f*g)[k] <= M*m_relax for k in peak_idx, g >= 0
        c = -np.ones(n)
        A_rows = []
        b_vals = []
        for k in peak_idx_sorted:
            kk = int(k)
            js = max(0, kk - n + 1)
            je = min(n - 1, kk)
            row = np.zeros(n)
            # (f*g)[k] = sum_i f[i]*g[k-i], so coefficient of g[j] is f[k-j]
            for j in range(js, je + 1):
                row[j] = f[kk - j]
            if row.sum() > 0:
                A_rows.append(row)
                b_vals.append(M * m_relax)

        if len(A_rows) == 0:
            break

        A_ub = np.array(A_rows)
        b_ub = np.array(b_vals)
        bounds = [(0, None)] * n
        options = {'time_limit': 30, 'presolve': True}
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options=options)

        if not res.success or res.x is None:
            m_relax *= 1.1
            if m_relax > 5.0:
                m_relax = 1.0
            continue

        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 30:
                break
            m_relax = min(m_relax * 1.02, 4.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-12:
            f = f + best_alpha * (g - f)
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"    LP-FW iter {it} ({elapsed:.0f}s): obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.03, 4.0)

        if no_improve > 60:
            rng = np.random.default_rng(it * 31 + 11)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with multiplicative updates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-15:
                continue

            best_j_val = old_val
            best_j_obj = best_obj
            cur_conv_max = fast_convolve(f).max()

            for mult in [0.5, 0.7, 1.4, 2.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 3:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Linear interpolation to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

                a_opt, _ = mirror_descent(f_pert, time_limit=min(rem2 * 0.3, 8),
                                          top_k=40, eta_init=0.001)
                if update_best(a_opt):
                    print(f"  NEW BEST [pert{trial}_md]: {best_score:.8f}")

    # === Phase 9: Final MD pass ===
    rem = remaining_time()
    if rem > 10:
        a_opt, _ = mirror_descent(best_seq, time_limit=min(rem * 0.85, 40),
                                  top_k=40, eta_init=0.001)
        if update_best(a_opt):
            print(f"  FINAL MD: {best_score:.8f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
