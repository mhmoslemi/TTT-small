def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def objective_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fw_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe optimization for min max_k (a*a)_k s.t. a>=0, sum(a)=1."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Find the peak
        k_peak = np.argmax(conv)
        
        # Compute subgradient: 2 * a[k_peak - j] for valid j
        sg = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        for j in range(js, je + 1):
            sg[j] = 2.0 * f[k_peak - j]
        
        # FW step: find vertex that minimizes <sg, e_i> = sg[i]
        # The minimum is at i* = argmin(sg)
        i_star = np.argmin(sg)
        
        # But we can do better: solve the LP for the best direction in the simplex
        # min sg^T g s.t. g >= 0, sum(g) = 1
        # The solution is g = e_{i_star} (a vertex)
        # For the LP version: min sg^T g s.t. (f*g)_k <= M for all k, g >= 0
        # This gives a better direction
        
        # Let's use the LP version for better convergence
        # But first, try the simple FW step
        g_vertex = np.zeros(n)
        g_vertex[i_star] = 1.0
        
        # Actually, let's use the LP direction
        # But that's expensive. Let's use a "generalized FW" with a few candidate directions
        # and pick the best one via line search
        
        # Simple FW direction
        directions = [g_vertex]
        
        # Also try the "negative subgradient" direction projected to simplex
        neg_sg = -sg
        neg_sg_proj = project_simplex(neg_sg)
        if np.max(np.abs(neg_sg_proj - f)) > 1e-10:
            directions.append(neg_sg_proj)
        
        # Line search over all directions
        best_alpha = 0.0
        best_val = M
        best_dir = None
        
        for g in directions:
            for alpha in [0.1, 0.3, 0.5, 0.7, 1.0]:
                ft = (1 - alpha) * f + alpha * g
                ft = np.maximum(ft, 0)
                st = ft.sum()
                if st < 1e-15:
                    continue
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha
                    best_dir = g
        
        if best_dir is not None and best_alpha > 0:
            f_new = (1 - best_alpha) * f + best_alpha * best_dir
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            val = fast_convolve(f_new).max()
            if val < best_obj - 1e-14:
                best_obj = val
                best_f = f_new.copy()
                f = f_new
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 50:
            # Perturb
            rng = np.random.default_rng(it * 7 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001 * np.sqrt(1.0 / it))
            no_improve = 0
        
        if verbose and it % 100 == 0:
            print(f"    FW iter {it}: obj={best_obj:.10f}")
    
    return best_f, best_obj

def solve_convex_cvxpy(f0, n, time_limit, verbose=True):
    """Solve the convex problem directly using CVXPY."""
    # Use symmetric formulation
    if n % 2 != 0:
        n += 1
    m = n // 2
    
    x = cp.Variable(m, nonneg=True)
    
    # Build the symmetric sequence
    # a = [x_0, x_1, ..., x_{m-1}, x_{m-1}, ..., x_1, x_0]
    # a[j] for j = 0..m-1 is x[j]
    # a[j] for j = m..n-1 is x[n-1-j]
    
    # For the convolution, we need (a*a)_k for k = 0..n-1 (by symmetry)
    # (a*a)_k = sum_{j: valid} a[j] * a[k-j]
    
    t = cp.Variable()
    constraints = [cp.sum(x) * 2 - x[m-1] == 1]  # sum(a) = 2*sum(x) - x[m-1] = 1
    
    # Wait, for n even: a = [x_0, ..., x_{m-1}, x_{m-1}, ..., x_0]
    # sum(a) = 2 * (x_0 + ... + x_{m-1}) = 2 * sum(x)
    # Hmm wait: a has n = 2m elements
    # a[0] = x[0], a[1] = x[1], ..., a[m-1] = x[m-1]
    # a[m] = x[m-1], a[m+1] = x[m-2], ..., a[2m-1] = x[0]
    # sum(a) = (x[0]+...+x[m-1]) + (x[m-1]+x[m-2]+...+x[0]) = 2*sum(x)
    
    constraints = [2 * cp.sum(x) == 1]
    
    # Convolution constraints (by symmetry, k = 0..n-1 suffices)
    # Actually, (a*a)_k = (a*a)_{2n-2-k}, so we need k = 0..n-1 (n constraints)
    
    for k in range(n):
        term = cp.Constant(0)
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            # a[j] * a[k-j]
            # a[j] = x[j] if j < m, else x[n-1-j]
            # a[k-j] = x[k-j] if k-j < m, else x[n-1-(k-j)]
            
            ij = j if j < m else n - 1 - j
            ik = (k - j) if (k - j) < m else n - 1 - (k - j)
            
            term += x[ij] * x[ik]
        
        constraints.append(term <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    if verbose:
        print(f"  CVXPY: n={n}, m={m}, constraints={len(constraints)}")

        except Exception as e2:
            print(f"  CVXPY ECOS failed: {e2}")
            return f0, None
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"  CVXPY solved in {elapsed:.1f}s")

        return f0, None

    a = np.zeros(n)

        return f0, None

    obj = objective_normalized(a)
    return a, obj

def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
