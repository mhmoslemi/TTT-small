```python
import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft, rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def lp_direction(f, M):
    """
    Solve the LP: maximize sum(b_j) subject to b >= 0 and (f*b)_k <= M for all k.
    Returns the optimal b (unnormalized).
    """
    n = len(f)
    # Build constraint matrix A_ub: each row k corresponds to constraint (f*b)_k <= M
    # (f*b)_k = sum_{i: 0<=i<n, 0<=k-i<n} f_i * b_{k-i}
    # Let j = k-i, so i = k-j. Constraint: sum_{j: 0<=j<n, 0<=k-j<n} f_{k-j} * b_j <= M
    
    # For efficiency, let's build the sparse matrix
    # A_ub[k, j] = f_{k-j} if 0 <= k-j < n, else 0
    # b_j variable index = j
    
    # This is a (2n-1) x n matrix
    # Let's build it efficiently
    
    # For each j (column), the non-zero entries are at rows k where 0 <= k-j < n
    # i.e., j <= k <= j+n-1
    # A_ub[k, j] = f[k-j]
    
    # Use scipy sparse for efficiency
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    
    for j in range(n):
        # rows k where 0 <= k-j < n => j <= k <= j+n-1
        k_start = j
        k_end = j + n - 1
        # k ranges from 0 to 2n-2
        k_start = max(0, k_start)
        k_end = min(2*n-2, k_end)
        for k in range(k_start, k_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Objective: maximize sum(b) => minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Bounds: b_j >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    else:
        # Fallback: return f (no improvement)
        return f.copy()

def lp_optimize(f0, time_limit, verbose=False):
    """
    LP-based optimization: iteratively solve LP to find direction, take step.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_f = f.copy()
    best_M = M
    
    t0 = time.monotonic()
    iter_count = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        
        # Solve LP
        b = lp_direction(f, M)
        b = np.maximum(b, 0)
        s_b = b.sum()
        if s_b < 1e-10:
            break
        b = b / s_b  # normalize to sum 1
        
        # Check if b itself has lower peak
        conv_b = fast_convolve(b)
        M_b = conv_b.max()
        
        # Take a convex combination
        if M_b < M:
            # b is better, take full step
            f_new = b
            M_new = M_b
        else:
            # Try intermediate steps
            best_t = 0
            best_M_try = M
            for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                f_try = (1 - t) * f + t * b
                f_try = np.maximum(f_try, 0)
                s_try = f_try.sum()
                if s_try < 1e-10:
                    continue
                f_try = f_try / s_try
                M_try = fast_convolve(f_try).max()
                if M_try < best_M_try:
                    best_M_try = M_try
                    best_t = t
                    f_new = f_try
            
            if best_t > 0:
                M_new = best_M_try
            else:
                f_new = f
                M_new = M
        
        f = f_new
        M = M_new
        
        if M < best_M - 1e-14:
            best_M = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iter_count}: M={best_M:.10f}, no_imp={no_improve} ({elapsed:.0f}s)")
        
        # If stuck, add perturbation
        if no_improve > 10:
            rng = np.random.default_rng(iter_count)
            noise = rng.standard_normal(n) * 0.01
            f = project_simplex(best_f + noise, target_sum=1.0)
            M = fast_convolve(f).max()
            no_improve = 0

                # Perturbation was too aggressive
                f = best_f.copy()
                M = best_M
    
    return best_f, best_M
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based directional search,
spectral-shaped asymmetric initialization, refined subgradient optimization,
and multi-length exploration. Key innovation: linear programming provides
principled descent directions that exploit the full constraint coupling.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """Build sparse constraint matrix for LP: rows k, cols j, entry f[k-j]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_start = j
        k_end = j + n - 1
        k_start = max(0, k_start)
        k_end = min(2 * n - 2, k_end)
        if k_end >= k_start:
            kk = np.arange(k_start, k_end + 1)
            idx = kk - j
            rows.extend(kk.tolist())
            cols.extend([j] * len(kk))
            vals.extend(f[idx].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_direction(f, M, A_matrix=None):
    """
    Solve LP: maximize sum(b) s.t. A*b <= M, b >= 0.
    Returns optimal b or None on failure.
    """
    n = len(f)
    if A_matrix is None:
        A_matrix = build_lp_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_matrix, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': 30})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def lp_optimize(f0, time_limit, verbose=False):
    """LP-based iterative optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_f = f.copy()
    best_M = M
    A_matrix = build_lp_matrix(f)

    t0 = time.monotonic()
    iter_count = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 10:
            break

        b = lp_direction(f, M, A_matrix)
        if b is None:

        s_b = b.sum()
        if s_b < 1e-10:

        b_norm = b / s_b

        conv_b = fast_convolve(b_norm)
        M_b = conv_b.max()

        if M_b < M:
            # Guaranteed improvement: find best t
            best_t = 0.0
            best_M_try = M
            for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
                f_try = (1 - t) * f + t * b_norm
                f_try = np.maximum(f_try, 0)
                ss = f_try.sum()
                if ss < 1e-10:

                f_try = f_try / ss
                M_try = fast_convolve(f_try).max()
                if M_try < best_M_try:

                    best_t = t
            if best_t > 0:
                f = (1 - best_t) * f + best_t * b_norm
                f = np.maximum(f, 0)
                f = f / f.sum()
                M = fast_convolve(f).max()
        else:
            # Try small steps
            best_t = 0.0
            best_M_try = M
            for t in [0.05, 0.1, 0.15, 0.2]:
                f_try = (1 - t) * f + t * b_norm
                f_try = np.maximum(f_try, 0)
                ss = f_try.sum()
                if ss < 1e-10:

                f_try = f_try / ss
                M_try = fast_convolve(f_try).max()
                if M_try < best_M_try - 1e-14:

                    best_t = t
            if best_t > 0:

                f = np.maximum(f, 0)
                f = f / f.sum()
                M = best_M_try

            best_f = f.copy()
            no_improve = 0
        else:

            A_matrix = build_lp_matrix(f)
            no_improve = 0

def compute_grad(a, tau, top_k=200):
    """Compute smooth subgradient of max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]

    pv = conv[peaks]
    scaled = tau * (pv - pv.max())
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[i] * a[k - idx]

    return grad, conv


def subgradient_optimize(a0, time_limit, lr_init=0.05, tau_init=60.0, momentum_init=0.85, verbose=False):
    """Subgradient optimization with adaptive parameters and Nesterov-like weighting."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    lr = lr_init
    tau = tau_init
    momentum = momentum_init
    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0
    t_global = 0.0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        t_global += 1

        # Annealing schedule
        frac = min(1.0, t_global / max(time_limit * 0.7, 1))
        lr_t = lr_init * (1.0 - 0.7 * frac) + 0.003 * frac
        tau_t = tau_init * (1.0 + 2.0 * frac)
        mom_t = momentum_init + 0.08 * frac

        if no_improve > 300:
            lr_t *= 0.8
            tau_t *= 1.2
        elif no_improve > 150:
            tau_t *= 1.05

        grad, conv = compute_grad(a, tau_t, top_k=min(200, n))
        grad_centered = grad - grad.mean()

        v = mom_t * v - lr_t * grad_centered
        a_new = project_simplex(a + v, target_sum=1.0)
        obj_new = fast_convolve(a_new).max()

        # Backtracking line search
        if obj_new > best_obj and lr_t > 0.01:
            v_half = v * 0.5
            a_half = project_simplex(a + v_half, target_sum=1.0)
            obj_half = fast_convolve(a_half).max()
            if obj_half < obj_new:
                a_new = a_half
                obj_new = obj_half
                v = v_half

        a = a_new

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG iter {iter_count}: obj={best_obj:.10f}, lr={lr_t:.5f}, tau={tau_t:.1f} ({elapsed:.0f}s)")

        if no_improve > 500:
            rng_r = np.random.default_rng(iter_count * 37 + 11)
            noise_scale = 0.003 + 0.003 * min(no_improve // 500, 5)
            a = project_simplex(best_a + rng_r.standard_normal(n) * noise_scale, target_sum=1.0)
            v = np.zeros(n)
            no_improve = 0

    return best_a, best_obj


def coordinate_descent(a0, time_limit, verbose=False):
    """Coordinate descent with multi-multiplier trial."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    pass_count = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        pass_count += 1
        improved = False
        rng = np.random.default_rng(pass_count * 7 + 3)

        # Random coordinate order
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj
            multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved and pass_count > 3:
            break

    return best_a, best_obj


def asymmetric_init(n, alpha, shape='ramp', rng=None):
    """Generate asymmetric non-negative sequence."""
    x = np.linspace(0, 1, n)
    if shape == 'ramp':
        base = 1.0 + alpha * (2 * x - 1)
    elif shape == 'quadratic':
        base = 1.0 + alpha * (2 * x - 1)**2
    elif shape == 'gaussian_left':
        base = np.exp(-((x - 0.35) ** 2) / (2 * 0.25 ** 2)) + 0.3 * np.exp(-((x - 0.75) ** 2) / (2 * 0.15 ** 2))
    elif shape == 'gaussian_right':
        base = np.exp(-((x - 0.75) ** 2) / (2 * 0.25 ** 2)) + 0.3 * np.exp(-((x - 0.35) ** 2) / (2 * 0.15 ** 2))
    elif shape == 'half':
        base = np.ones(n)
        base[:n // 2] = 1.0 + alpha
        base[n // 2:] = 1.0 - alpha * 0.5
    else:
        base = np.ones(n)
    base = np.maximum(base, 0.01)
    return base / base.sum()


def spectral_init(n, rng):
    """Flat-spectrum initialization with random phases."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        mag = 1.0
        A[k] = mag * np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining LP-based directional search,
    subgradient optimization, spectral initialization, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_default = 768
        parent = asymmetric_init(n_default, 0.3, 'ramp', rng)
        print(f"No parent. Using default init n={n_default}")
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    n_parent = len(parent)

    # === Phase 1: LP optimization on parent ===
    print(f"\n=== Phase 1: LP on parent n={n_parent} (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lp = min(rem * 0.25, 400)
    if t_lp > 30:
        print(f"  Time allocated: {t_lp:.0f}s")
        a_opt, obj = lp_optimize(parent, time_limit=t_lp, verbose=True)
        try_seq(a_opt, "lp_parent")

    # === Phase 2: Subgradient refinement ===
    print(f"\n=== Phase 2: Subgradient refinement (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 600
        if rem > 30:
            t_sg = min(rem * 0.5, 250)
            print(f"  Time allocated: {t_sg:.0f}s")
            a_opt, obj = subgradient_optimize(best_seq, time_limit=t_sg,
                                              lr_init=0.04, tau_init=50.0,
                                              momentum_init=0.84, verbose=True)
            try_seq(a_opt, "sg_refine")

    # === Phase 3: Try asymmetric inits at different lengths ===
    print(f"\n=== Phase 3: Asymmetric inits + optimization (best={best_score:.8f}) ===")
    target_lengths = []
    if parent is not None:
        target_lengths.append(n_parent)
    for n_try in [640, 768, 896, 1024, 1280]:
        if n_try not in target_lengths and 384 <= n_try <= 2048:
            target_lengths.append(n_try)
    target_lengths.sort(key=lambda x: abs(x - n_parent))

    for n_try in target_lengths[:4]:
        if n_try == n_parent:
            continue
        rem = deadline - time.monotonic() - 400
        if rem < 60:
            break
        t_alloc = min(rem * 0.4, 150)
        print(f"  n={n_try}, t={t_alloc:.0f}s")

        # Multiple initializations
        inits = []
        for alpha in [0.1, 0.2, 0.3, 0.4]:
            for shape in ['ramp', 'gaussian_left', 'gaussian_right']:
                a_init = asymmetric_init(n_try, alpha, shape, rng)
                inits.append(a_init)
        a_spec = spectral_init(n_try, rng)
        inits.append(a_spec)
        if parent is not None:
            a_interp = interpolate_sequence(parent, n_try)
            # Make asymmetric
            a_asym = a_interp.copy()
            x = np.linspace(0, 1, n_try)
            a_asym *= (1.0 + 0.2 * (2 * x - 1))
            a_asym = np.maximum(a_asym, 0)
            a_asym /= a_asym.sum()
            inits.append(a_asym)
            inits.append(a_interp)

        # Pick best init
        best_init = None
        best_init_obj = float('inf')
        for a_init in inits:
            a_init = np.maximum(a_init, 0)
            ss = a_init.sum()
            if ss < 1e-10:
                continue
            a_init = a_init / ss
            obj = fast_convolve(a_init).max()
            if obj < best_init_obj:
                best_init_obj = obj
                best_init = a_init

        if best_init is not None:
            t_sg = min(t_alloc * 0.6, 100)
            a_opt, obj = subgradient_optimize(best_init, time_limit=t_sg,
                                              lr_init=0.05, tau_init=40.0,
                                              momentum_init=0.82)
            try_seq(a_opt, f"asym_n{n_try}")

            rem2 = deadline - time.monotonic() - 300
            if rem2 > 20 and fast_convolve(a_opt).max() < best_seq is not None and True:
                t_cd = min(rem2 * 0.3, 40)
                a_opt2, obj2 = coordinate_descent(a_opt, time_limit=t_cd)
                try_seq(a_opt2, f"cd_n{n_try}")

    # === Phase 4: LP on best found ===
    print(f"\n=== Phase 4: LP on best (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_lp = min(rem * 0.4, 300)
            print(f"  n={len(best_seq)}, t={t_lp:.0f}s")
            a_opt, obj = lp_optimize(best_seq, time_limit=t_lp, verbose=True)
            try_seq(a_opt, "lp_best")

    # === Phase 5: Subgradient with different parameters ===
    print(f"\n=== Phase 5: Subgradient variants (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        variants = [
            (0.06, 30.0, 0.80, "fast"),
            (0.03, 100.0, 0.90, "slow"),
            (0.04, 60.0, 0.85, "mid"),
        ]
        for i, (lr, tau, mom, name) in enumerate(variants):
            rem = deadline - time.monotonic() - 150
            if rem < 30:
                break
            t_alloc = min(rem * 0.35, 120)
            # Perturb for diversity
            noise = rng.standard_normal(n_best) * 0.005
            f_init = project_simplex(best_seq + noise, target_sum=1.0)
            a_opt, obj = subgradient_optimize(f_init, time_limit=t_alloc,
                                              lr_init=lr, tau_init=tau,
                                              momentum_init=mom)
            try_seq(a_opt, f"sg_{name}_{i}")

    # === Phase 6: Coordinate descent ===
    print(f"\n=== Phase 6: Coordinate Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_cd = min(rem * 0.5, 200)
            print(f"  t={t_cd:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd_final")

    # === Phase 7: LP on best again ===
    print(f"\n=== Phase 7: Final LP (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_lp = min(rem * 0.6, 150)
            a_opt, obj = lp_optimize(best_seq, time_limit=t_lp)
            try_seq(a_opt, "lp_final")

    # === Phase 8: Final polishing ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_cd = min(rem * 0.5, 50)
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "polish_cd")

        rem = deadline - time.monotonic() - 20
        for _ in range(3):
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            n_best = len(best_seq)
            noise = rng.standard_normal(n_best) * 0.003
            f_pert = project_simplex(best_seq + noise, target_sum=1.0)
            a_opt, obj = subgradient_optimize(f_pert, time_limit=min(rem, 20),
                                              lr_init=0.02, tau_init=80.0,
                                              momentum_init=0.88)
            try_seq(a_opt, "perturb")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
