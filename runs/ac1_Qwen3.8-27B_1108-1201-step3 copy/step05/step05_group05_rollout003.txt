def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def compute_soft_grad(a, conv, tau, top_k=200):
    """Compute gradient of soft approximation of max(conv)."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    # Use argpartition for efficiency
    peaks = np.argpartition(-conv, k_actual - 1)[:k_actual]
    pv = conv[peaks]
    # Soft weights
    max_pv = pv.max()
    scaled = (pv - max_pv) / tau
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(peaks))
        w[0] = 1.0
    else:
        w /= w_sum
    
    grad = np.zeros(n)

        if w[i] < 1e-15:

        k = int(peaks[i])

        je = min(n - 1, k)
        idx = np.arange(js, je + 1)

    s = a.sum()

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

Let me implement option 4 (mirror descent) as a new optimization method:

**Mirror Descent on the Simplex**:
- The mirror map is `Φ(a) = sum_i a_i * log(a_i)` (negative entropy).
- The Bregman divergence is `D_Φ(a, b) = sum_i (a_i log(a_i/b_i) - a_i + b_i)`.
- The update is: `a_new = argmin_{a in simplex} <g, a> + (1/lr) * D_Φ(a, a_old)`.
- This gives: `a_new[i] = a_old[i] * exp(-lr * g[i]) / sum_j a_old[j] * exp(-lr * g[j])`.

This is essentially the **multiplicative weights** or **exponential tilting** update. It's well-suited for problems where the variables are probabilities (non-negative and sum to 1).

Let me implement this:

```python
def mirror_descent_step(a, grad, lr):
    """Mirror descent step with entropy regularization."""
    # a_new[i] = a[i] * exp(-lr * grad[i]) / sum_j a[j] * exp(-lr * grad[j])
    log_weights = -lr * grad
    # For numerical stability, subtract the max
    log_weights -= log_weights.max()
    weights = np.exp(log_weights) * a
    ss = weights.sum()
    if ss < 1e-300:
        return a.copy()
    return weights / ss
```
