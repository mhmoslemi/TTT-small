       n = len(f)

       # Let's define w_ext of length 2n-1 (which is w itself)
       # v is the "convolution" of a (reversed) with w, shifted
       
       # Actually, the simplest way: 
       # v = np.array([np.dot(a, w[j:j+n]) for j in range(n)])
       # But this is O(n^2). For n=768, that's 590K operations per gradient eval.
       # For L-BFGS with ~1000 iterations, that's 590M operations. Might be slow in Python.
       
       # Better: use FFT
       # v[j] = <a, w[j:j+n]> = <a, shift(w, j)>
       # = ifft(conj(fft(a)) * fft(w))[j] ... let me verify with a small example
       
       # For n=3: a=(a0,a1,a2), w=(w0,w1,w2,w3,w4)
       # v[0] = a0*w0 + a1*w1 + a2*w2
       # v[1] = a0*w1 + a1*w2 + a2*w3
       # v[2] = a0*w2 + a1*w3 + a2*w4
       
       # This is the cross-correlation of a and w (with w the "long" signal)
       # In FFT terms: ifft(conj(FFT(a)) * FFT(w)) gives the cross-correlation
       # But we need to be careful about the exact definition
       
       # scipy.signal.correlate(a, w, mode='valid') would give exactly v
       # But we want FFT-based
       
       # The cross-correlation R[j] = sum_m a[m] * w[j+m] for j=0..n-1
       # (assuming w has length >= 2n-1)
       # 
       # FFT approach:
       # Let A = FFT(a, size=fft_size), W = FFT(w, size=fft_size)
       # Then IFFT(conj(A) * W)[j] = sum_m a[m] * w[(j+m) mod fft_size]
       # For j < n and fft_size >= 2n-1, this gives exactly v[j]
       
       fft_size = 1 << (2*n - 1 - 1).bit_length()
       A = rfft(a, fft_size)
       W = rfft(w, fft_size)
       v = irfft(np.conj(A) * W, fft_size)[:n]
       
       grad_a = 4.0 * n * v
       
       return obj, grad_a
   
   def projected_gradient_descent(a0, T, time_limit, lr=0.001, decay=0.999):
       """Projected gradient descent on the simplex."""
       n = len(a0)
       a = a0.copy().astype(np.float64)
       a /= a.sum()
       best_a = a.copy()
       best_obj = fast_convolve(a).max()
       t0 = time.monotonic()
       cur_lr = lr
       v = np.zeros(n)  # momentum
       mom = 0.9
       no_improve = 0
       
       while time.monotonic() - t0 < time_limit:
           obj, grad = smooth_max_obj_grad(a, T)
           
           # FISTA-style momentum
           v = mom * v - cur_lr * grad
           a_new = project_simplex(a + v)
           a = a_new
           
           cur_lr *= decay
           
           co = fast_convolve(a).max()
           if co < best_obj - 1e-15:
               best_obj = co
               best_a = a.copy()
               no_improve = 0
           else:
               no_improve += 1
           
           if no_improve > 500:
               # Restart
               a = project_simplex(best_a + 0.001 * np.random.default_rng(int(time.time()*1e6)).standard_normal(n))
               v = np.zeros(n)
               cur_lr = max(lr * 0.5, 1e-5)
               no_improve = 0
       
       return best_a, best_obj
   
   def lbfgs_optimize(a0, T, time_limit, maxiter=3000):
       """L-BFGS in log-space with soft-max objective."""
       n = len(a0)
       z0 = z_from_a(a0)
       t0 = time.monotonic()
       
       def func(z):
           if time.monotonic() - t0 > time_limit:
               return 1e15
           a = softmax_from_z(z)
           return smooth_max_obj_grad(a, T)[0]
       
       def jac(z):
           if time.monotonic() - t0 > time_limit:
               return np.zeros_like(z)
           a = softmax_from_z(z)
           obj, grad_a = smooth_max_obj_grad(a, T)
           S = np.dot(grad_a, a)
           grad_z = a * (grad_a - a * S)
           return grad_z
       
       try:
           res = minimize(
               fun=func,
               x0=z0,
               jac=jac,
               method='L-BFGS-B',
               options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                        'maxfun': maxiter * 25, 'disp': False}
           )
           a_out = softmax_from_z(res.x)
           return a_out
       except Exception:
           return a0
   
   def coordinate_descent(a0, time_limit=60.0):
       """Coordinate descent with best-multiplier search."""
       a = a0.copy().astype(np.float64)

       while time.monotonic() - t0 < time_limit:

               if time.monotonic() - t0 > time_limit:
                   break

               if old_val < 1e-12:

               best_j_val = old_val

                       continue
