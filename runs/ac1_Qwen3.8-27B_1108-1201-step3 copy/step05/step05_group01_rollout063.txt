    n = len(a)

    A = np.fft.rfft(a, fft_size)

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:

    return e / s

Enhancements over previous SOTA:

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def precompute_indices(n):
    rows = []
    cols = []
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        rows.extend(range(k_start, k_end))
        cols.extend([j] * (k_end - k_start))
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)

            for _ in range(20):

            conv = fast_convolve(f_new)

def _smooth_obj_grad(t, beta):
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    obj = conv_max + np.log(exp_shifted.sum()) / beta
    
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R
    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)
    
    return obj, grad_t


def lbfqg_optimize(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    
    n = len(f)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    eps = 1e-15
    t = np.log(f + eps)
    
    betas = [3.0, 6.0, 12.0, 25.0, 50.0, 100.0, 200.0, 400.0]
    t0_time = time.monotonic()
    
    for bi, beta in enumerate(betas):
        remaining = time_limit - (time.monotonic() - t0_time)
        if remaining < 20:
            break
        sub_time = remaining / (len(betas) - bi)
        
        try:
            if sub_time > 100:
                maxiter = 500
                maxfun = 3000
            elif sub_time > 50:
                maxiter = 300
                maxfun = 2000
            else:
                maxiter = 100
                maxfun = 1000
            
            res = minimize(
                _smooth_obj_grad, t, args=(beta,),
                jac=True, method='L-BFGS-B',
                options={'maxiter': maxiter, 'gtol': 1e-10, 'maxls': 40,
                         'maxfun': maxfun}
            )
            t = np.clip(res.x, -50, 50)
            a = softmax_fn(t)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 1e-15:
                a = a / a_sum
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = a.copy()
                if verbose:
                    print(f"    L-BFGS beta={beta}: obj={best_obj:.10f}")
        except Exception as e:
            if verbose:
                print(f"    L-BFGS beta={beta} failed: {e}")
    
    return best_f, best_obj


def swap_sa_optimize(f0, time_limit, verbose=False, seed=123):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    obj = conv.max()
    best_obj = obj
    best_f = f.copy()
    
    rng = np.random.default_rng(seed)
    T = 0.002
    T_min = 1e-5
    cooling = 0.99997
    no_improve = 0
    max_no_improve = 5000
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Pick random pair
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        
        # Pick delta
        max_delta = min(f[i], f[j], 0.01) if rng.random() < 0.5 else min(f[i], 0.01)
        if max_delta < 1e-10:
            continue
        delta = rng.random() * max_delta
        
        # Try moving mass from i to j
        f[i] -= delta
        f[j] += delta
        
        new_conv = fast_convolve(f)
        new_obj = new_conv.max()
        delta_obj = new_obj - obj
        
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / T):
            obj = new_obj
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
        else:
            # Revert
            f[i] += delta
            f[j] -= delta
            no_improve += 1
        
        T = max(T * cooling, T_min)
        
        if no_improve > max_no_improve:
            # Restart with perturbation
            scale = 0.005 + 0.003 * (no_improve // max_no_improve)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            obj = conv.max()
            no_improve = 0
            T = 0.003
    
    return best_f, best_obj


def sg_polish(f0, time_limit, verbose=False, seed=999):
    """Subgradient polishing with adaptive top-k peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    lr = 0.003
    mom = 0.93
    no_improve = 0
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        conv_max = conv.max()
        
        # Use top-k peaks with softmax weights
        k_actual = min(max(10, n // 10), len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        temp = max(0.001, (pv.max() - pv.min()) * 0.5)
        scaled = (pv - pv.max()) / temp
        w = np.exp(scaled)
        w /= w.sum()
        
        # Compute subgradient
        sg = np.zeros(n)
        for wi, pi in zip(w, peaks):
            if wi < 1e-12:
                continue
            k = int(pi)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * wi * f[k - idx]
        
        v = mom * v - lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.999
        
        if no_improve > 300:
            scale = 0.002
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            lr = 0.003
            no_improve = 0
    
    return best_f, best_obj


def generate_structured_starts(n, rng, base=None):
    """Generate diverse structured initializations."""
    starts = []
    
    if base is not None:
        # Perturbed versions of base
        for sigma in [0.005, 0.01, 0.02, 0.05]:
            starts.append(project_simplex(base + rng.standard_normal(n) * sigma))
    
    # Sine-based
    for freq in [1, 2, 3, 5, 8]:
        for phase in [0, np.pi/4, np.pi/2]:
            x = np.arange(n) / n
            a = 1.0 + 0.3 * np.sin(2 * np.pi * freq * x + phase)
            a = np.maximum(a, 0.01)
            starts.append(a / a.sum())
    
    # Cosine window
    x = np.arange(n) / n
    starts.append(np.cos(np.pi * x / 2)**2 / np.cos(np.pi * x / 2)**2.sum())
    
    # Triangle
    tri = 1.0 - np.abs(2 * x - 1)
    starts.append(tri / tri.sum())
    
    # Double bump
    for center1, center2 in [(0.25, 0.75), (0.2, 0.8), (0.3, 0.7)]:
        a = np.exp(-50 * (x - center1)**2) + np.exp(-50 * (x - center2)**2)
        starts.append(a / a.sum())
    
    # Gaussian center
    for width in [0.1, 0.2, 0.3, 0.5]:
        a = np.exp(-((x - 0.5)**2) / (2 * width**2))
        starts.append(a / a.sum())
    
    # Random Dirichlet
    for alpha in [0.5, 1.0, 2.0, 5.0]:
        a = rng.dirichlet(np.full(n, alpha))
        starts.append(a)
    
    return starts

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    if size <= 1:
        return np.array([a[0] * a[0]])
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:

    return e / s


def precompute_conv_indices(n):

def _smooth_obj_grad(t, beta):
    """Compute smooth log-sum-exp objective and gradient in t-space."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    obj = conv_max + np.log(exp_shifted.sum()) / beta

    fft_size = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    R = np.fft.irfft(W * np.conj(A), fft_size)[:n]
    grad_a = 2.0 * R
    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)

    return obj, grad_t

def coordinate_descent(f0, time_limit):
    """Coordinate descent with multi-ratio local search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 0.3]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    print("\n=== Phase 1: LP-FW on parent ===")
    t1 = min(350.0, deadline - time.monotonic() - 900)
    if t1 > 30:
        a_opt, obj_opt = lp_fw_optimize(parent, time_limit=t1, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  LP-FW parent: {score:.6f} (n={len(a_opt)})")

    # === Phase 2: L-BFGS on best (250s) ===
    print(f"\n=== Phase 2: L-BFGS (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    t2 = min(250.0, deadline - time.monotonic() - 700)
    if t2 > 20:
        a_opt, obj_opt = lbfqg_optimize(base, time_limit=t2, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  L-BFGS: {score:.6f} (n={len(a_opt)})")

    # === Phase 3: Simulated Annealing (400s) ===
    print(f"\n=== Phase 3: Simulated Annealing (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    t3 = min(400.0, deadline - time.monotonic() - 500)
    if t3 > 30:
        a_opt, obj_opt = swap_sa_optimize(base, time_limit=t3, verbose=True, seed=seed)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  SA: {score:.6f} (n={len(a_opt)})")

    # === Phase 4: LP-FW on SA result (250s) ===
    print(f"\n=== Phase 4: LP-FW round 2 (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    t4 = min(250.0, deadline - time.monotonic() - 400)
    if t4 > 20:
        a_opt, obj_opt = lp_fw_optimize(base, time_limit=t4, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score

            print(f"  LP-FW r2: {score:.6f} (n={len(a_opt)})")

    # === Phase 5: L-BFGS deep (150s) ===
    print(f"\n=== Phase 5: L-BFGS deep (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    t5 = min(150.0, deadline - time.monotonic() - 300)
    if t5 > 15:
        a_opt, obj_opt = lbfqg_optimize(base, time_limit=t5, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score

            print(f"  L-BFGS deep: {score:.6f} (n={len(a_opt)})")

    # === Phase 6: SA round 2 (150s) ===
    print(f"\n=== Phase 6: SA round 2 (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    t6 = min(150.0, deadline - time.monotonic() - 200)
    if t6 > 20:
        a_opt, obj_opt = swap_sa_optimize(base, time_limit=t6, verbose=True, seed=seed + 1000)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  SA r2: {score:.6f} (n={len(a_opt)})")

    # === Phase 7: Subgradient polish (100s) ===
    print(f"\n=== Phase 7: SG Polish (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    t7 = min(100.0, deadline - time.monotonic() - 100)
    if t7 > 15:
        a_opt, obj_opt = sg_polish(base, time_limit=t7, verbose=True, seed=seed + 2000)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  SG: {score:.6f} (n={len(a_opt)})")

    # === Phase 8: Multi-length exploration (80s) ===
    print(f"\n=== Phase 8: Multi-length (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    target_lengths = [512, 896, 1024]
    for tn in target_lengths:
        rem = deadline - time.monotonic() - 60
        if rem < 30:
            break
        f_interp = interpolate_sequence(base, tn)
        t_alloc = min(rem * 0.4, 40)
        if t_alloc < 10:
            break
        a_opt, obj_opt = lp_fw_optimize(f_interp, time_limit=t_alloc)
        score = float(evaluate_sequence(a_opt.tolist()))

    print(f"\n=== Phase 9: Final Refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # 9a: LP-FW final
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_alloc = min(rem * 0.5, 80)

        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_alloc = min(rem * 0.6, 50)
            a_opt, _ = swap_sa_optimize(best_seq, time_limit=t_alloc, seed=seed + 5000)

            if np.isfinite(score) and score < best_score:

                best_seq = a_opt.copy()
                print(f"  Final SA: {score:.6f}")

        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t_alloc = min(rem, 30)
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"  Final CD: {score:.6f}")

        # 9d: Very fine SG
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt, _ = sg_polish(best_seq, time_limit=min(rem, 15), seed=seed + 9000)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"  Final SG: {score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
