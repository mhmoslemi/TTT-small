    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_objective(f, T):
    """Log-sum-exp approximation of max(conv(f,f))."""
    conv = fast_convolve(f)
    Tconv = T * conv
    Tconv -= Tconv.max()  # numerical stability
    return conv.max() + np.log(np.sum(np.exp(Tconv))) / T

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast gradient methods with

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad(f, w, n):
    """Compute gradient of sum_k w[k]*conv[k] wrt f.
    w: softmax weights over convolution lags (length 2n-1)
    Returns grad of shape (n,)."""
    # grad[j] = 2 * sum_k w[k] * f[k-j] for valid k-j
    # = 2 * correlate(w, f, 'valid')[j]
    corr = np.correlate(w, f, mode='valid')
    return 2.0 * corr[:n]


def smooth_grad_and_val(f, T, top_k=None):
    """Compute (value, gradient) of log-sum-exp smoothed max(conv(f,f))."""
    n = len(f)
    conv = fast_convolve(f)
    conv_len = len(conv)
    
    if top_k is not None and top_k < conv_len:
        idx = np.argpartition(conv, -top_k)[-top_k:]
    else:
        idx = np.arange(conv_len)
    
    pv = conv[idx]
    Tpv = T * pv
    Tpv -= Tpv.max()
    softmax = np.exp(Tpv)
    Z = softmax.sum()
    if Z < 1e-300:

        w[np.argmax(pv)] = 1.0
    else:
        w = softmax / Z
    
    val = pv.max() + np.log(Z) / T
    
    # Build full weight vector for gradient computation
    w_full = np.zeros(conv_len)
    w_full[idx] = w
    
    grad = compute_grad(f, w_full, n)
    return val, grad

            T = max(T_init, T * 0.5)
            lr = max(lr * 2, 0.005)
            no_improve = 0
            conv = fast_convolve(f)
            exact_max = conv.max()
            if exact_max < best_obj:
                best_obj = exact_max
                best_f = f.copy()
        
        if verbose and t_step % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    adam iter {t_step} ({elapsed:.0f}s): obj={best_obj:.10f}, T={T:.0f}, lr={lr:.6f}")
    
    return best_f, best_obj


def subgradient_polish(f0, time_limit, top_k=30, lr=0.005, mom=0.9, verbose=False):
    """Subgradient method with momentum for exact max(conv) minimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        
        pv = conv[peaks]
        # Uniform weights over top peaks (or temperature-based)
        w = np.ones(len(peaks)) / len(peaks)
        
        w_full = np.zeros(len(conv))
        w_full[peaks] = w
        
        grad = compute_grad(f, w_full, n)
        
        v = mom * v - cur_lr * grad
        f = project_simplex(f + v)
        
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.9997
        
        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 1000))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.001)
            no_improve = 0
            conv = fast_convolve(f)
            if conv.max() < best_obj:
                best_obj = conv.max()
                best_f = f.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    sg: obj={best_obj:.10f}, lr={cur_lr:.6f}")
    
    return best_f, best_obj


def coordinate_descent_opt(f0, time_limit, verbose=False):
    """Coordinate descent with multiple trial multipliers per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    multipliers = [0.3, 0.5, 0.7, 0.8, 1.2, 1.4, 1.7, 2.5]
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        cur_conv = fast_convolve(f)
        cur_max = cur_conv.max()
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-15:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue
            
            best_j_obj = cur_max
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                f[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = f / f.sum()
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved and passes > 3:
            break
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_init(n, seed=0):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    inits = []
    
    # Uniform
    inits.append(("uniform", np.ones(n) / n))
    
    # Triangle (peaked center)
    x = np.linspace(0, 1, n)
    tri = 1 - np.abs(2 * x - 1)
    inits.append(("triangle", tri / tri.sum()))
    
    # Inverse triangle (peaked at edges)
    inv_tri = np.abs(2 * x - 1)
    inv_tri = inv_tri + 0.1  # avoid zeros
    inits.append(("inv_triangle", inv_tri / inv_tri.sum()))
    
    # Cosine window
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits.append(("cosine", cos_w / cos_w.sum()))
    
    # Hamming window
    ham = 0.54 - 0.46 * np.cos(2 * np.pi * x)
    inits.append(("hamming", ham / ham.sum()))
    
    # Blackman window
    black = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    black = np.maximum(black, 0.01)
    inits.append(("blackman", black / black.sum()))
    
    # Alternating (high-low pattern)
    alt = np.ones(n)
    alt[::2] = 2.0
    alt[1::2] = 0.5
    inits.append(("alternating", alt / alt.sum()))
    
    # Block pattern (high first half, low second half)
    block = np.ones(n)
    block[:n//2] = 1.5
    block[n//2:] = 0.5 + 0.5 * n/(2*n)  # ensure sum works out
    inits.append(("block", block / block.sum()))
    
    # Random
    rnd = rng.random(n) + 0.1
    inits.append(("random", rnd / rnd.sum()))
    
    # Bimodal
    bim = np.exp(-10 * (x - 0.25)**2) + np.exp(-10 * (x - 0.75)**2) + 0.05
    inits.append(("bimodal", bim / bim.sum()))
    
    # Low-frequency sinusoid modulated
    sine = 0.5 + 0.4 * np.sin(2 * np.pi * x * 3)
    sine = np.maximum(sine, 0.05)
    inits.append(("sine3", sine / sine.sum()))
    
    return inits
