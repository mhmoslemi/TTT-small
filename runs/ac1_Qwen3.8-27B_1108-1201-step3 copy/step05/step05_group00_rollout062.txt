"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using asymmetric initializations,
LP-Frank-Wolfe with warm starting, multiplicative weight mirror descent,
and Nesterov-accelerated subgradient methods across multiple sequence lengths.

Key insight: asymmetric sequences (geometric/exponential tilts) break the

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_geometric(n, r):
    """Geometric (exponentially decaying) sequence."""
    a = np.array([r ** i for i in range(n)])
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_exp_tilt(n, beta):
    """Exponential tilt: a_i proportional to exp(beta * i/(n-1))."""
    x = np.arange(n) / max(n - 1, 1)
    a = np.exp(beta * x)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_linear_ramp(n, alpha):
    """Linear ramp: a_i proportional to 1 + alpha * i/(n-1)."""
    x = np.arange(n) / max(n - 1, 1)
    a = 1.0 + alpha * x
    a = np.maximum(a, 0.01)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    """Bell-shaped initial sequence with flat top."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def solve_lp_direction(f, M, n, warm_start=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

                m_exp = 1.0

        else:

        # Line search

        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:

def mirror_descent(f0, time_limit, verbose=False, lr_init=0.004, decay=0.9996):
    """Multiplicative weight mirror descent with adaptive temperature."""
    f = np.maximum(f0.astype(np.float64), 1e-12)

        conv = fast_convolve(f)
        M = conv.max()

        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()

def nesterov_subgrad(f0, time_limit, verbose=False, top_k=40, T_init=0.005):
    """Nesterov-accelerated projected subgradient method."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    y = f.copy()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    T = T_init
    momentum = 0.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(y)
        M = conv.max()

        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            j_idx = np.arange(js, je + 1)
            a_idx = k - j_idx
            sg[js:je + 1] += 2.0 * w[i] * y[a_idx]

        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break

        # Polyak-style step size
        M_lower = 1.0 / n
        gamma = max(M - M_lower, 1e-4)
        step = 2.0 * gamma / (sg_norm * sg_norm + 1e-12)
        step = min(step, 0.1)

        # Heavy ball momentum
        momentum = 0.9 * momentum + 0.1 * step
        y_new = y - momentum * sg
        f_new = project_simplex(y_new)

        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    NS iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        f = f_new
        y = f.copy()
        T *= 1.002

        if no_improve > 150:
            rng = np.random.default_rng(it * 23 + 17)
            f = project_simplex(best_f + rng.standard_standard_normal if False else rng.standard_normal(n) * 0.002)
            y = f.copy()
            momentum = 0.0
            no_improve = 0
            T = T_init

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def peak_shaving(f0, time_limit, verbose=False):

            je = min(n - 1, k)
            if je < js:

            idx = np.arange(js, je + 1)
            products = f[idx] * f[k - idx]

                    weights[j_max] = 0

                    if wsum > 1e-15:
                        trial += diff * weights / wsum

                obj = fast_convolve(trial).max()

                    best_trial = trial

            if best_trial is not None:
                f = best_trial

                    if nw > 1e-15:

    print(f"\n=== Phase 5: Multi-Length Exploration ===")
    base = best_seq if best_seq is not None else best_init
    for target_n in [1024, 896, 768, 640, 512, 384]:
        if len(base) == target_n:
            continue
        rem = deadline - time.monotonic() - 200
        if rem < 30:
            break
        f_interp = interpolate_sequence(base, target_n)
        sc = try_seq(f_interp, f"interp_n{target_n}")

        t_alloc = min(rem * 0.3, 100)
        print(f"  LP-FW n={target_n}: t={t_alloc:.0f}s")
        a_opt, obj = lp_fw_optimize(f_interp, time_limit=t_alloc)
        try_seq(a_opt, f"lpfw_n{target_n}")

    # === Phase 6: Nesterov subgradient on best ===
    print(f"\n=== Phase 6: Nesterov Subgradient ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(rem, 150)
            print(f"  NS: t={t_alloc:.0f}s")
            a_opt, obj = nesterov_subgrad(best_seq, time_limit=t_alloc, verbose=True,
                                          top_k=50, T_init=0.003)
            try_seq(a_opt, "nesterov")

    # === Phase 7: Peak shaving ===
    print(f"\n=== Phase 7: Peak Shaving ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_alloc = min(rem, 100)
            print(f"  Peak shaving: t={t_alloc:.0f}s")
            a_opt, obj = peak_shaving(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_shave")

    # === Phase 8: LP-FW with diverse perturbations ===
    print(f"\n=== Phase 8: LP-FW Perturbations ===")
    if best_seq is not None:
        n_best = len(best_seq)
        variants = []
        # Small noise
        noise = rng.standard_normal(n_best) * 0.002
        variants.append(project_simplex(best_seq + noise))
        # Larger noise
        noise = rng.standard_normal(n_best) * 0.005
        variants.append(project_simplex(best_seq + noise))
        # Blend with optimal asymmetric init
        if best_init.shape[0] == n_best:
            variants.append((0.7 * best_seq + 0.3 * best_init) / 1.0)
        # Reverse
        variants.append(best_seq[::-1] / best_seq[::-1].sum())
        # Blend with uniform
        variants.append((0.85 * best_seq + 0.15 * np.ones(n_best) / n_best) / 1.0)

        for vi, v in enumerate(variants):
            rem = deadline - time.monotonic() - 100
            if rem < 20:
                break
            t_alloc = min(rem / (len(variants) - vi), 60)
            print(f"  LP-FW variant {vi}: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(v, time_limit=t_alloc)
            try_seq(a_opt, f"perturb_{vi}")

    # === Phase 9: Final mirror descent polish ===
    print(f"\n=== Phase 9: Final MD Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 20:
            t_alloc = min(rem, 60)
            print(f"  MD polish: t={t_alloc:.0f}s")
            a_opt, obj = mirror_descent(best_seq, time_limit=t_alloc, verbose=True,
                                        lr_init=0.001, decay=0.9998)
            try_seq(a_opt, "md_polish")

    # === Phase 10: Final LP-FW squeeze ===
    print(f"\n=== Phase 10: Final LP-FW Squeeze ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            t_alloc = min(rem, 30)
            print(f"  Final LP-FW: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_lpfw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
