- Compute the "gradient" of C at each of these positions.
- Take a weighted step in the direction that reduces all K peaks.

The gradient of C[k] with respect to a[j] is:
∂C[k]/∂a[j] = a[k-j] (if 0 <= k-j < n) + a[j-k] (if 0 <= j-k < n, i.e., k <= j < n and j-k >= 0)

Wait, let me be more careful:
C[k] = sum_i a[i] * a[k-i]
∂C[k]/∂a[j] = a[k-j] (from the term i=j) + a[j-k] (from the term k-i=j, i.e., i=k-j)

But wait, if j = k-j (i.e., k=2j), these are the same term, giving 2*a[j]. Otherwise, they're different.

∂C[k]/∂a[j] = a[k-j] + a[k-j] = 2*a[k-j] if j ≠ k-j (i.e., k ≠ 2j)
∂C[k]/∂a[j] = 2*a[j] if k = 2j

Wait no. C[k] = sum_i a[i]*a[k-i]. The terms involving a[j] are:
- i = j: gives a[j]*a[k-j] (valid if 0 <= k-j < n)
- k-i = j, i.e., i = k-j: gives a[k-j]*a[j] (valid if 0 <= k-j < n)

These are the same term when j = k-j, i.e., k = 2j. In that case, the term is a[j]^2, and the derivative is 2*a[j].
When j ≠ k-j, there are two separate terms, each giving a[j]*a[k-j], so the total derivative is 2*a[k-j].

So: ∂C[k]/∂a[j] = 2*a[k-j] if 0 <= k-j < n (and k is in the valid range for the convolution)
Otherwise: ∂C[k]/∂a[j] = 0.

Wait, I need to double-check. C[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i]*a[k-i].

The derivative with respect to a[j]:
- If 0 <= j < n and 0 <= k-j < n: a[j] appears as a[i] when i=j (giving a[j]*a[k-j]) and as a[k-i] when k-i=j, i.e., i=k-j (giving a[k-j]*a[j]).
  - If j ≠ k-j: both terms exist, giving 2*a[j]*a[k-j], derivative = 2*a[k-j].
  - If j = k-j (k=2j): only one term a[j]^2, derivative = 2*a[j].
- Otherwise: a[j] doesn't appear in C[k], derivative = 0.

So: ∂C[k]/∂a[j] = 2*a[k-j] if 0 <= k-j < n, else 0.

This is simple! The gradient of C[k] with respect to a is a shifted (and zero-padded) version of a, multiplied by 2.

For the subgradient of max(C), we take a convex combination of the gradients at the argmax positions (and possibly the top K positions with softmax weights).

```python
def subgradient_step(a, lr, K=10):
    n = len(a)
    C = fast_convolve(a)
    
    # Find top K peaks
    k_actual = min(K, len(C))
    peaks = np.argpartition(C, -k_actual)[-k_actual:]
    pv = C[peaks]
    
    # Softmax weights
    temp = max(1e-6, (pv.max() - pv.min()) * 0.5)
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # Compute weighted subgradient
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        # ∂C[k]/∂a[j] = 2*a[k-j] for 0 <= k-j < n
        # So sg[j] += w[i] * 2*a[k-j]
        # j ranges over [0, n-1], and k-j ranges over [k-(n-1), k]
        # Valid j: max(0, k-n+1) <= j <= min(n-1, k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]
    
    # Update: multiplicative (mirror descent) or additive (with projection)
    # Multiplicative: a_new = a * exp(-lr * sg), then normalize
    a_new = a * np.exp(-lr * sg)
    a_new = np.maximum(a_new, 1e-15)
    a_new /= a_new.sum()
    
    return a_new, C
```

"""
Minimizes 2*n*max(auto_conv)/(sum)^2 using exact 1D coordinate descent
with FFT-based convolution, multi-peak subgradient, diverse initializations,
and adaptive restarts.
"""

import numpy as np
from scipy.optimize import minimize_scalar
import time

def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.empty(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def coord_optimize(a, j, C=None):
    """Optimize coordinate j. Returns (new_a, new_score, new_C)."""
    n = len(a)
    v_old = a[j]
    S_total = a.sum()
    S0 = S_total - v_old
    
    if C is None:
        C = fast_convolve(a)
    
    if S0 < 1e-15:
        score = 2.0 * n * C.max() / (S_total * S_total)
        return (a.copy(), score, C)
    
    k_start = max(0, j)
    k_end = min(2 * n - 2, j + n - 1)
    
    if k_start > k_end:
        score = 2.0 * n * C.max() / (S_total * S_total)
        return (a.copy(), score, C)
    
    ks = np.arange(k_start, k_end + 1)
    kjs = ks - j  # = k - j, in [0, n-1]
    
    # C_base[k] = C[k] - contribution of a[j]
    C_base = C.copy()
    C_base[ks] -= 2.0 * v_old * a[kjs]
    
    two_j = 2 * j
    if k_start <= two_j <= k_end:
        # We subtracted 2*v_old*a[j] = 2*v_old^2, but should subtract v_old^2
        C_base[two_j] += v_old * v_old
    
    # Unaffected max
    mask_aff = np.zeros(2 * n - 1, dtype=bool)
    mask_aff[k_start:k_end + 1] = True
    if (~mask_aff).any():
        M_unaff = C[~mask_aff].max()
    else:
        M_unaff = 0.0
    
    # Precompute for f(v)
    intercepts = C_base[ks].copy()
    slopes = 2.0 * a[kjs].copy()
    is_2j = (ks == two_j)
    c_base_2j = C_base[two_j] if (k_start <= two_j <= k_end) else 0.0
    
    def f(v):
        if v < 0:
            return float('inf')
        c_new = intercepts + slopes * v
        if is_2j.any():
            c_new[is_2j] = c_base_2j + v * v
        max_c = max(M_unaff, c_new.max())
        S_new = S0 + v
        if S_new < 1e-15:
            return float('inf')
        return 2.0 * n * max_c / (S_new * S_new)
    
    v_lo = 0.0
    v_hi = max(v_old * 3.0, S0 / n * 2.0, 0.01)
    v_hi = min(v_hi, S0 * 5.0)
    v_hi = max(v_hi, v_old + 0.001)
    
    res = minimize_scalar(f, bounds=(v_lo, v_hi), method='bounded',
                          options={'xatol': 1e-12, 'maxiter': 50})
    
    if not np.isfinite(res.fun) or res.x < 1e-15:
        score = 2.0 * n * C.max() / (S_total * S_total)
        return (a.copy(), score, C)
    
    new_a = a.copy()
    new_a[j] = res.x
    new_C = fast_convolve(new_a)
    new_score = 2.0 * n * new_C.max() / (new_a.sum() ** 2)
    
    return (new_a, new_score, new_C)

def coord_descent_pass(a, C, order=None):
    """One pass of coordinate descent. Returns (new_a, new_C, improved)."""
    n = len(a)
    if order is None:
        order = np.arange(n)
    
    improved = False
    for j in order:
        new_a, new_score, new_C = coord_optimize(a, j, C)
        if new_score < 2.0 * n * C.max() / (a.sum() ** 2) - 1e-12:
            a = new_a
            C = new_C
            improved = True
    
    return a, C, improved

def subgradient_step(a, lr, K=20):
    """Multi-peak subgradient step with multiplicative update."""
    n = len(a)
    C = fast_convolve(a)
    
    k_actual = min(K, len(C))
    peaks = np.argpartition(C, -k_actual)[-k_actual:]
    pv = C[peaks]
    
    # Softmax weights
    diff = pv.max() - pv.min()
    temp = max(1e-6, diff * 0.3)
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]
    
    a_new = a * np.exp(-lr * sg)
    a_new = np.maximum(a_new, 1e-15)
    a_new /= a_new.sum()
    
    return a_new

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)

def binomial_decay_sequence(n):
    """Sequence based on central binomial coefficients (flat C inspired)."""
    k = np.arange(n, dtype=np.float64)
    # a[k] ∝ binom(2k,k) / 4^k ≈ 1/sqrt(pi*k) for large k
    # Use log computation to avoid overflow
    # log(binom(2k,k)/4^k) = log(binom(2k,k)) - k*log(4)
    # For large k: ≈ -0.5*log(pi*k)
    # For exact: use scipy.special or direct computation for small k
    # For simplicity, use the asymptotic: 1/sqrt(pi*(k+1))
    a = 1.0 / np.sqrt(np.pi * (k + 1))
    return normalize(a)

def gaussian_sequence(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-x ** 2 / (2 * sigma ** 2))
    return normalize(a)
