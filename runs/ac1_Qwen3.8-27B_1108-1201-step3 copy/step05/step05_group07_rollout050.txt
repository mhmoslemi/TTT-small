"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding
combined with log-softmax L-BFGS, multiplicative weight updates, and 
peak shaving. The LP finds the maximum-L1 direction compatible with the 
current peak level, which is iteratively mixed with the current sequence 
to reduce the peak. Multi-length exploration and multiple optimization 
phases ensure robustness against local minima.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def cross_corr(w, a):
    """Cross-correlation for gradient computation."""
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:n]


def soft_max_obj_grad(z, T, n):
    """Compute the soft-max smoothed objective and its gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    
    w = exp_Tb / exp_Tb.sum()
    grad_a = 2.0 * n * cross_corr(w, a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    
    return obj, grad_z


def lbgfs_optimize(z0, T, n, time_limit, maxiter=3000):
    """L-BFGS in log-space with time limit."""
    t0 = time.monotonic()
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        obj, _ = soft_max_obj_grad(z, T, n)
        return obj
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = soft_max_obj_grad(z, T, n)
        return g
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 25, 'disp': False}
        )
        return res.x, res.fun
    except Exception:
        return z0, func(z0)


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(a, M):
    """
    Solve the LP: max sum(g) s.t. g >= 0, (a*g)[k] <= M for all k.
    Returns the optimal g.
    """
    n = len(a)
    # Build constraint matrix A: (2n-1) x n
    # A[k, i] = a[k-i] if 0 <= k-i <= n-1, else 0
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for i in range(max(0, k - n + 1), min(n - 1, k) + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    
    A_mat = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    res = linprog(c, A_ub=A_mat, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    
    if res.success:
        return res.x
    return None


def lp_iteration(a, max_iters=200, time_limit=60.0, tol=1e-10):
    """
    LP-based iteration to reduce the peak of the autocorrelation.
    """
    t0 = time.monotonic()
    a = a.copy().astype(np.float64)
    s = a.sum()
    a /= s
    n = len(a)
    
    b = fast_convolve(a)
    M = b.max()
    best_a = a.copy()
    best_M = M
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP
        g = solve_lp_direction(a, M)
        if g is None:
            break
        
        s_g = g.sum()
        if s_g < 1e-10:
            break
        
        # Rescale g to sum = 1
        g /= s_g
        
        # Line search for best mixing parameter t
        best_t = 0
        best_M_new = M
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            h = (1 - t) * a + t * g
            h = np.maximum(h, 0)
            hs = h.sum()
            if hs < 1e-10:
                continue
            h /= hs
            b_h = fast_convolve(h)
            M_new = b_h.max()
            if M_new < best_M_new:
                best_M_new = M_new
                best_t = t
        
        if best_t == 0:
            break
        
        a = (1 - best_t) * a + best_t * g
        a = np.maximum(a, 0)
        a /= a.sum()
        
        b = fast_convolve(a)
        M = b.max()
        
        if M < best_M - tol:
            best_M = M
            best_a = a.copy()
        
        if abs(M - best_M) < tol:
            break
    
    return best_a, best_M


def multiplicative_weight_update(a, time_limit=60.0, eta=0.01, n_iters=5000):
    """
    Multiplicative weight update to reduce peak autocorrelation.
    """
    t0 = time.monotonic()
    a = a.copy().astype(np.float64)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        M = b.max()
        
        # Compute the gradient-like quantity: for each i, how much does a[i] contribute to peaks?
        # f_i = sum_k b_k * a[max(0,k-n+1)..min(n-1,k)] at position i
        # This is essentially a cross-correlation of b with a
        f = cross_corr(b, a)
        
        # Multiplicative update
        a_new = a * np.exp(-eta * f)
        a_new = np.maximum(a_new, 1e-300)
        a_new /= a_new.sum()
        
        b_new = fast_convolve(a_new)
        M_new = b_new.max()
        
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
        
        a = a_new
        # Decay eta slowly
        eta *= 0.999
    
    return best_a, best_M


def peak_shave(a, n_peaks=5, n_steps=200, time_limit=30.0):
    """Peak shaving: iteratively reduce the highest peaks."""
    t0 = time.monotonic()
    a = a.copy().astype(np.float64)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        b_inner = b.copy()
        b_inner[0] = -1
        b_inner[-1] = -1
        peak_indices = np.argpartition(b_inner, -n_peaks)[-n_peaks:]
        peak_indices = [k for k in peak_indices if b[k] > best_obj * 0.95]
        
        if not peak_indices:
            break
        
        improved = False
        for k in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contributions = a[idx] * a[k - idx]
            
            max_local = np.argmax(contributions)
            i = idx[max_local]
            j = k - i
            
            for target in [i, j]:
                old_val = a[target]
                if old_val < 1e-10:
                    continue
                for factor in [0.8, 0.9, 0.95]:
                    new_val = old_val * factor
                    diff = old_val - new_val
                    if diff < 1e-15:
                        continue
                    
                    mask = np.ones(n, dtype=bool)
                    mask[target] = False
                    others = a[mask]
                    if others.sum() < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[target] = new_val
                    a_trial[mask] = others + diff * others / others.sum()
                    a_trial /= a_trial.sum()
                    
                    val = fast_convolve(a_trial).max()
                    if val < best_obj - 1e-12:
                        a = a_trial
                        best_obj = val
                        best_a = a.copy()
                        improved = True
                        break
        
        if not improved:
            break
    
    return best_a, best_obj


def coordinate_descent(a0, time_limit=60.0):
    """Coordinate descent cycling through multipliers."""
    a = a0.copy().astype(np.float64)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    multipliers = [0.5, 0.7, 1.0, 1.3, 1.5, 1.8, 2.5]
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial /= a_trial.sum()
                
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a /= a.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
    
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding,
log-softmax L-BFGS with temperature annealing, multiplicative weight updates,
coordinate descent, and peak shaving. The LP finds maximum-L1 directions
compatible with the current peak level, iteratively mixed to reduce peaks.
Multi-length exploration and multiple optimization phases prevent local minima.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def cross_corr(w, a):
    """Cross-correlation: result[i] = sum_k w[k] * a[k-i] for valid indices."""
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:n]


def soft_max_obj_grad(z, T, n):
    """Compute soft-max smoothed objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    grad_a = 2.0 * n * cross_corr(w, a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z


def lbgfs_optimize(z0, T, n, time_limit, maxiter=3000):
    """L-BFGS in log-space with time limit."""
    t0 = time.monotonic()

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        obj, _ = soft_max_obj_grad(z, T, n)
        return obj

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = soft_max_obj_grad(z, T, n)
        return g

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 25, 'disp': False}
        )
        return res.x, res.fun
    except Exception:
        return z0, func(z0)


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(a, M):
    """Solve LP: max sum(g) s.t. g >= 0, (a*g)[k] <= M for all k."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    A_mat = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    res = linprog(c, A_ub=A_mat, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def lp_iteration(a, max_iters=200, time_limit=60.0, tol=1e-10):
    """LP-based iteration to reduce peak autocorrelation."""
    t0 = time.monotonic()
    a = a.copy().astype(np.float64)
    s = a.sum()
    a /= s
    n = len(a)
    b = fast_convolve(a)
    M = b.max()
    best_a = a.copy()
    best_M = M
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        g = solve_lp_direction(a, M)
        if g is None:
            break
        s_g = g.sum()
        if s_g < 1e-10:
            break
        g /= s_g

        best_t = 0
        best_M_new = M
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            h = (1 - t) * a + t * g
            h = np.maximum(h, 0)
            hs = h.sum()
            if hs < 1e-10:
                continue
            h /= hs
            b_h = fast_convolve(h)
            M_new = b_h.max()
            if M_new < best_M_new:
                best_M_new = M_new
                best_t = t

        if best_t == 0:
            no_improve += 1
            if no_improve > 10:
                break
            # Try a smaller perturbation
            g_pert = 0.5 * a + 0.5 * g
            g_pert = np.maximum(g_pert, 0)
            g_pert /= g_pert.sum()
            a = 0.9 * a + 0.1 * g_pert
            a = np.maximum(a, 0)
            a /= a.sum()
            continue

        a = (1 - best_t) * a + best_t * g
        a = np.maximum(a, 0)
        a /= a.sum()
        b = fast_convolve(a)
        M = b.max()

        if M < best_M - tol:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                break

    return best_a, best_M


def multiplicative_weight_update(a, time_limit=60.0, eta=0.02, n_iters=10000):
    """Multiplicative weight update to reduce peak autocorrelation."""
    t0 = time.monotonic()
    a = a.copy().astype(np.float64)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    no_improve = 0

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        f = cross_corr(b, a)
        a_new = a * np.exp(-eta * f)
        a_new = np.maximum(a_new, 1e-300)
        a_new /= a_new.sum()
        b_new = fast_convolve(a_new)
        M_new = b_new.max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        a = a_new
        eta *= 0.9995
        if no_improve > 500:
            break

    return best_a, best_M


def peak_shave(a, n_peaks=5, n_steps=200, time_limit=30.0):
    """Peak shaving: iteratively reduce the highest peaks."""
    t0 = time.monotonic()
    a = a.copy().astype(np.float64)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        b_inner = b.copy()
        b_inner[0] = -1
        b_inner[-1] = -1
        peak_indices = np.argpartition(b_inner, -n_peaks)[-n_peaks:]
        peak_indices = [k for k in peak_indices if b[k] > best_obj * 0.95]
        if not peak_indices:
            break
        improved = False
        for k in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contributions = a[idx] * a[k - idx]
            max_local = np.argmax(contributions)
            i = idx[max_local]
            j = k - i
            for target in [i, j]:
                old_val = a[target]
                if old_val < 1e-10:
                    continue
                for factor in [0.75, 0.85, 0.92]:
                    new_val = old_val * factor
                    diff = old_val - new_val
                    if diff < 1e-15:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[target] = False
                    others = a[mask]
                    if others.sum() < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[target] = new_val
                    a_trial[mask] = others + diff * others / others.sum()
                    a_trial /= a_trial.sum()
                    val = fast_convolve(a_trial).max()
                    if val < best_obj - 1e-12:
                        a = a_trial
                        best_obj = val
                        best_a = a.copy()
                        improved = True
                        break
        if not improved:
            break

    return best_a, best_obj


def coordinate_descent(a0, time_limit=60.0):
    """Coordinate descent cycling through multipliers."""
    a = a0.copy().astype(np.float64)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    multipliers = [0.4, 0.6, 0.8, 1.25, 1.5, 2.0, 3.0]

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            best_j_obj = best_obj
            best_j_val = old_val
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial /= a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        if not improved:
            break

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def try_update(a, best_seq, best_score, label):

    score = float(evaluate_sequence(a.tolist()))

    else:

    return best_seq, best_score


def subgradient_polish(a0, time_limit=60.0, lr=0.003, mom=0.92, top_k=30):
    """Subgradient method with softmax-weighted active peaks."""
    a = a0.copy().astype(np.float64)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = 0.001
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]
        v = mom * v - cur_lr * sg
        a = project_simplex(a + v)
        b = fast_convolve(a)
        co = b.max()
        if co < best_M - 1e-14:
            best_M = co
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        cur_lr *= 0.999
        cur_temp *= 1.01
        if no_improve > 200:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0

    return best_a, best_M
