"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding
(inspired by SOTA), adaptive step size, multiple initializations,
and coordinate descent polish.
"""

import numpy as np
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def conv_toeplitz(f):
    """Construct (2n-1) x n Toeplitz matrix C such that C @ g = conv(f, g)."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        g_start = k - i_end
        g_end = k - i_start
        C[k, g_start:g_end+1] = f[i_start:i_end+1][::-1]
    return C


def solve_lp_direction(f, M, tol=1e-9):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns normalized g."""
    n = len(f)
    C = conv_toeplitz(f)
    g = cp.Variable(n, nonneg=True)
    constraints = [C @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.GLOP, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.CBC, verbose=False)
        except Exception:
            return None
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    s = g_val.sum()
    if s < 1e-15:
        return None
    return g_val / s


def lp_iteration(f, M, max_steps=50):
    """
    One LP-based iteration: find direction g, then take a step.
    Returns updated f and new M.
    """
    n = len(f)
    f = f / f.sum()
    M = max(M, fast_convolve(f).max())
    
    g = solve_lp_direction(f, M)
    if g is None:
        return f, M
    
    # Ensure g is normalized to sum 1
    g = g / g.sum()
    
    # Binary search for step size
    best_t = 0.0
    best_obj = fast_convolve(f).max()
    
    lo, hi = 0.0, 1.0
    for _ in range(max_steps):
        mid = (lo + hi) / 2
        f_trial = (1 - mid) * f + mid * g
        f_trial = np.maximum(f_trial, 0)
        s = f_trial.sum()
        if s < 1e-15:
            hi = mid
            continue
        f_trial = f_trial / s
        obj = fast_convolve(f_trial).max()
        if obj <= M * (1 + 1e-12):  # Allow tiny numerical tolerance
            best_t = mid
            best_obj = obj
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-10:
            break
    
    if best_t > 0:
        f_new = (1 - best_t) * f + best_t * g
        f_new = np.maximum(f_new, 0)
        f_new = f_new / f_new.sum()
        new_M = fast_convolve(f_new).max()
        return f_new, new_M
    else:
        return f, M


def lp_optimize(f0, time_limit, verbose=False):
    """LP-based optimization with restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    M = fast_convolve(f).max()
    best_f = f.copy()
    best_M = M
    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        f_new, M_new = lp_iteration(f, M)
        
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        M = max(M_new, 1e-15)
        
        if verbose and iter_count % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iter_count}: M={best_M:.10f}, no_imp={no_improve} ({elapsed:.1f}s)")
        
        # Restart if stuck
        if no_improve > 30:
            rng_r = np.random.default_rng(iter_count * 13 + 5)
            noise = rng_r.standard_normal(n) * 0.003
            f = best_f + noise
            f = np.maximum(f, 0)
            f = f / f.sum()
            M = fast_convolve(f).max()
            no_improve = 0
    
    return best_f, best_M


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        mag = 1.0
        A[k] = mag * np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    ss = a.sum()
    if ss < 1e-10:
        return np.ones(n) / n
    return a / ss


def random_init(n, rng):
    a = rng.exponential(1.0, n)
    return a / a.sum()


def cosine_init(n):
    """Cosine-tapered initialization."""
    x = np.arange(n) / max(n - 1, 1)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    return a / a.sum()


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def coordinate_descent(a0, time_limit):
    """Coordinate descent for local polishing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_obj


def subgradient_step(f, tau, top_k=100):
    """Compute subgradient direction for max(conv(f,f))."""
    n = len(f)
    conv = fast_convolve(f)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    
    pv = conv[peaks]
    scaled = tau * (pv - pv.max())

def project_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)

            no_improve += 1

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding
(interior-point LP to minimize max(f*g) over simplex), adaptive step size,
multiple diverse initializations, subgradient backup, and coordinate descent polish.
Key innovation: direct LP formulation min t s.t. conv(f,g)<=t, sum(g)=1, g>=0.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def conv_toeplitz(f):
    """Construct (2n-1) x n matrix C such that C @ g = linear_conv(f, g)."""
    n = len(f)
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    idx = k - j
    mask = (idx >= 0) & (idx < n)
    C = np.where(mask, f[np.clip(idx, 0, n - 1)], 0.0)
    return C


def solve_lp_direction(f):
    """Solve LP: min t s.t. conv(f,g) <= t, sum(g)=1, g>=0. Returns (g, t)."""
    n = len(f)
    C = conv_toeplitz(f)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.hstack([C, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        return None, None
    g_val = np.maximum(res.x[:n], 0.0)
    s = g_val.sum()
    if s < 1e-15:
        return None, None
    g_val /= s
    t_val = float(res.x[n])
    return g_val, t_val


def lp_optimize(f0, time_limit, verbose=False):
    """LP-based descent: iteratively find direction minimizing max(f*g) on simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    M = fast_convolve(f).max()
    best_f = f.copy()
    best_M = M
    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0
    step_scale = 1.0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1

        g, t_lp = solve_lp_direction(f)
        if g is None:
            break

        # Binary search for optimal step size
        best_alpha = 0.0
        best_obj = fast_convolve(f).max()
        lo, hi = 0.0, 1.0

        for _ in range(25):
            mid = (lo + hi) / 2.0
            f_trial = (1.0 - mid) * f + mid * g
            f_trial = np.maximum(f_trial, 0.0)
            ss = f_trial.sum()
            if ss < 1e-15:
                hi = mid
                continue
            f_trial /= ss
            obj = fast_convolve(f_trial).max()
            if obj <= best_M * (1.0 + 1e-12):
                best_alpha = mid
                best_obj = obj
                lo = mid
            else:
                hi = mid
            if hi - lo < 1e-9:
                break

        if best_alpha > 1e-10:
            f_new = (1.0 - best_alpha) * f + best_alpha * g
            f_new = np.maximum(f_new, 0.0)
            f_new /= f_new.sum()
            new_M = fast_convolve(f_new).max()
            f = f_new

            if new_M < best_M - 1e-14:
                best_M = new_M
                best_f = f.copy()
                no_improve = 0
                step_scale = min(1.0, step_scale * 1.02)
            else:

                step_scale = max(0.3, step_scale * 0.95)

            f = best_f + noise

            f /= f.sum()
            best_M = fast_convolve(f).max()
            no_improve = 0
            step_scale = 1.0

def subgradient_step(f, tau, top_k=100):
    """Compute smooth subgradient of max(conv(f,f))."""
    n = len(f)
    conv = fast_convolve(f)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = tau * (pv - pv.max())
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[i] * f[k - idx]
    return grad


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def sg_optimize(f0, time_limit, tau=100.0, lr=0.05, momentum=0.85):
    """Subgradient optimization with momentum and adaptive params."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_M = fast_convolve(f).max()
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        grad = subgradient_step(f, tau, top_k=min(150, n))
        grad_c = grad - grad.mean()
        v = momentum * v - lr * grad_c
        f_new = project_simplex(f + v, 1.0)
        obj = fast_convolve(f_new).max()

        if obj < best_M - 1e-14:
            best_M = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        if no_improve > 200:
            lr = max(0.005, lr * 0.98)
            tau = min(500.0, tau * 1.01)
        if no_improve > 500:
            rng_r = np.random.default_rng(int(time.monotonic() * 10000))
            noise = rng_r.standard_normal(n) * 0.005
            f = project_simplex(best_f + noise, 1.0)
            v = np.zeros(n)
            no_improve = 0
            lr = 0.04
            tau = 80.0

    return best_f, best_M


def coordinate_descent(a0, time_limit):
    """Coordinate descent for local polishing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            best_j_val = old_val
            best_j_obj = best_obj
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        if not improved:
            break
    return best_a, best_obj


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for kk in range(1, half + 1):
        A[kk] = np.exp(1j * phases[kk])
        if kk < n - kk:
            A[n - kk] = np.conj(A[kk])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    ss = a.sum()
    if ss < 1e-10:
        return np.ones(n) / n
    return a / ss


def random_init(n, rng):
    a = rng.exponential(1.0, n)
    return a / a.sum()


def cosine_init(n):
    x = np.arange(n) / max(n - 1, 1)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    return a / a.sum()


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

                f_interp = interpolate_sequence(best_seq, target_n)
                t_alloc = min(rem * 0.4, 120)
                print(f"  Longer LP n={target_n}, t={t_alloc:.0f}s")
                a_opt, obj = lp_optimize(f_interp, time_limit=t_alloc)
                try_seq_wrapper(a_opt, f"long_n{target_n}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  FALLBACK: score={best_score:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
