"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using parametric initialization,
improved multiplicative weight optimization, and LP-Frank-Wolfe polishing.

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def make_power_law(n, beta):
    """a[j] ∝ (1 - j/n)^beta"""
    j = np.arange(n)
    a = (1.0 - j / n) ** beta
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_exponential(n, alpha):
    """a[j] ∝ exp(-alpha * j/n)"""
    j = np.arange(n)
    a = np.exp(-alpha * j / n)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_raised_cosine(n):
    """a[j] ∝ (1 + cos(pi * j/n)) / 2"""
    j = np.arange(n)
    a = (1.0 + np.cos(np.pi * j / n)) / 2.0
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_gaussian(n, sigma_frac):
    """Gaussian centered at n/2."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_one_sided_gaussian(n, sigma_frac):
    """Gaussian centered at 0 (one-sided)."""
    j = np.arange(n)
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (j / sigma) ** 2)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def compute_subgradient(f, k):
    """Compute gradient of (a*a)_k with respect to a."""
    n = len(f)
    grad = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        grad[j] += f[k - j]
        if k - j != j:
            grad[k - j] += f[j]
        else:
            grad[j] += f[j]  # diagonal case: 2*f[j]
    return grad

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Parametric family exploration (power-law, exponential, cosine, etc.)
2. Improved multiplicative weight mirror descent with adaptive step size
3. LP-Frank-Wolfe for final polishing
4. Multi-length exploration

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2 for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def make_power_law(n, beta):
    """a[j] proportional to (1 - j/n)^beta, normalized."""
    j = np.arange(n, dtype=np.float64)
    base = np.maximum(1.0 - j / n, 0.0)
    a = base ** beta
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_exponential(n, alpha):
    """a[j] proportional to exp(-alpha * j/n), normalized."""
    j = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * j / n)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_raised_cosine(n, center_frac=0.0):
    """a[j] proportional to (1+cos(pi*(j/n - center_frac)))/2, one-sided."""
    j = np.arange(n, dtype=np.float64)
    a = (1.0 + np.cos(np.pi * j / n)) / 2.0
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_gaussian(n, sigma_frac, center_frac=0.5):
    """Gaussian centered at center_frac*n."""
    x = np.arange(n, dtype=np.float64) - center_frac * n
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_bell(n, width_frac=0.6, center_frac=0.5):
    """Smooth bell shape."""
    x = np.arange(n, dtype=np.float64) - center_frac * n
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except:
        pass
    return None

def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.2
            if m_exp > 10.0:
                rng = np.random.default_rng(it * 17 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.1
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                m_exp = min(m_exp * 1.1, 5.0)
            if no_improve > 150:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue
        else:
            if np.max(np.abs(g - f)) > 1e-6:
                no_improve = 0

        # Line search
        best_alpha = 0.0
        best_val = M

        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            # Fine-tune
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 5.0)

        if no_improve > 150:
            break

    return best_f, best_obj

def solve_lp_direction_sparse(f, M, n):
    """Solve LP with sparse constraint matrix."""
    c = -np.ones(n)
    
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # A[j:j+n, j] = f
        row_start = j
        row_end = j + n
        rows.extend(range(row_start, row_end))
        cols.extend([j] * n)
        vals.extend(f)
    
    A_ub = sp.csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Parametric power-law initialization (a[j] ~ (1-j/n)^beta) with optimized beta
2. Fast multiplicative weight mirror descent with adaptive step size
3. Sparse LP-Frank-Wolfe for final polishing
4. Multi-length exploration and diverse restarts
"""

import numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2 for any non-negative a."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    n = len(a_norm)
    conv = fast_convolve(a_norm)
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_power_law(n, beta):
    """a[j] proportional to (1 - j/n)^beta, normalized to sum=1."""
    j = np.arange(n, dtype=np.float64)
    base = np.maximum(1.0 - j / float(n), 0.0)
    a = base ** beta
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_exponential(n, alpha):
    """a[j] proportional to exp(-alpha * j/n), normalized."""
    j = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * j / float(n))
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_raised_cosine_one_sided(n):
    """a[j] proportional to (1+cos(pi*j/n))/2 (one-sided raised cosine)."""
    j = np.arange(n, dtype=np.float64)
    a = (1.0 + np.cos(np.pi * j / float(n))) / 2.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_gaussian(n, sigma_frac, center_frac=0.5):
    """Gaussian centered at center_frac*n."""
    x = np.arange(n, dtype=np.float64) - center_frac * n
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_beta_dist(n, alpha, beta):
    """Beta distribution shape on [0,1]: a[j] ~ (j/n)^(alpha-1) * (1-j/n)^(beta-1)."""
    j = np.arange(n, dtype=np.float64)
    u = j / float(n)
    u = np.clip(u, 1e-10, 1.0 - 1e-10)
    a = u ** (alpha - 1) * (1.0 - u) ** (beta - 1)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def solve_lp_direction_sparse(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0. Using sparse matrix."""
    c = -np.ones(n)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        rows.extend(range(j, j + n))
        cols.extend([j] * n)
        vals.extend(f)
    A_ub = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = solve_lp_direction_sparse(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.15
            if m_exp > 10.0:
                rng = np.random.default_rng(it * 17 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.1
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 60:
                m_exp = min(m_exp * 1.1, 5.0)
            if no_improve > 200:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue
        elif np.max(np.abs(g - f)) > 1e-6:
            no_improve = 0

        # Line search: coarse
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 5.0)

        if no_improve > 200:
            break

    return best_f, best_obj


def multiplicative_descent(f0, time_limit, verbose=False, lr_init=0.01, n_peaks=5):
    """
    Improved multiplicative weight mirror descent.
    Uses the gradient identity: grad_k[m] = 2*a[k-m] for valid m.
    """
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    temperature = 0.05
    it = 0
    f_rev = f[::-1]

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Find top peaks
        k_actual = min(n_peaks, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        # Softmax weights over peaks
        pv = conv[peaks]
        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws

        # Compute subgradient using the identity:
        # grad[m] += 2 * w_i * f[k_i - m] for m in [max(0,k_i-n+1), min(n-1,k_i)]
        sg = np.zeros(n)
        f_rev = f[::-1]
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je >= js:
                # sg[js:je+1] += 2*w*f_rev[n-1-k+js : n-1-k+je+1]
                offset = n - 1 - k
                sg[js:je + 1] += 2.0 * w[i] * f_rev[offset + js:offset + je + 1]

        # Normalized multiplicative update
        sg_norm = sg / max(f.sum(), 1e-15)
        f = f * np.exp(-lr * sg_norm)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        new_obj = fast_convolve(f).max()
        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    MD iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        # Adaptive learning rate
        if no_improve > 100:
            lr *= 0.85
        if no_improve > 400:
            # Restart with perturbation
            rng = np.random.default_rng(it * 43 + 11)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.001, 1e-15)
            f = f / f.sum()
            no_improve = 0
            lr = lr_init * 0.5

        # Anneal temperature (sharper focus on dominant peak)
        temperature = max(temperature * 0.998, 0.001)

        if lr < 1e-9:
            break

    return best_f, best_obj


def subgradient_descent(f0, time_limit, verbose=False):
    """
    Projected subgradient with Nesterov-style averaging.
    Uses the gradient identity for efficient computation.
    """
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    avg_f = f.copy()
    avg_count = 1
    lr = 1.0 / np.sqrt(1.0)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Find the argmax peak (single peak for sharpness)
        k_peak = int(np.argmax(conv))

        # Subgradient: 2 * a[k_peak - m] for valid m
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        sg = np.zeros(n)
        f_rev = f[::-1]
        offset = n - 1 - k_peak
        sg[js:je + 1] = 2.0 * f_rev[offset + js:offset + je + 1]

        # Also add a small contribution from the second peak for stability
        conv_copy = conv.copy()
        conv_copy[k_peak] = -np.inf
        if len(conv_copy) > 0 and conv_copy.max() > 0.5 * M:
            k2 = int(np.argmax(conv_copy))
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            offset2 = n - 1 - k2
            sg[js2:je2 + 1] += 0.5 * f_rev[offset2 + js2:offset2 + je2 + 1]

        # Nesterov step size
        lr = 1.0 / np.sqrt(1.0 + 0.1 * it)

        # Update
        f_new = f - lr * sg
        f_new = project_simplex(f_new)

        # Nesterov averaging
        avg_f = avg_f + (f_new - avg_f) / avg_count
        avg_count += 1

        f = f_new

        # Check both current and averaged
        obj_avg = fast_convolve(avg_f).max()
        obj_cur = fast_convolve(f).max()

        if obj_avg < best_obj:
            best_obj = obj_avg
            best_f = avg_f.copy()
        if obj_cur < best_obj:
            best_obj = obj_cur
            best_f = f.copy()

        if verbose and it % 200 == 0:
            print(f"    SG iter {it}: best={best_obj:.10f}")

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent trying to reduce the maximum convolution coefficient."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve_rounds = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_convolve(f)
        peak_k = int(np.argmax(conv))

        # Find elements contributing most to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contrib = f[js:je + 1] * f[peak_k - js:peak_k - js + (je - js + 1)][::-1]

        # Try modifying the top contributors
        top_idx = js + np.argsort(contrib)[-5:]

        for j in top_idx:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-10:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for frac in [0.5, 0.7, 0.85, 1.2, 1.5]:
                new_val = old_val * frac
                if new_val < 0 or new_val > 0.9:
                    continue
                trial = f.copy()
                trial[j] = new_val
                # Redistribute
                diff = old_val - new_val
                if abs(diff) > 1e-12:
                    # Add to elements with smallest contribution to peak
                    mask = np.ones(n, dtype=bool)
                    # Reduce from non-contributing or low-contributing elements
                    trial = np.maximum(trial, 0)
                    tsum = trial.sum()
                    if tsum < 1e-15:
                        continue
                    # Scale others to maintain sum=1
                    mask2 = np.ones(n, dtype=bool)
                    mask2[j] = False
                    scale = (1.0 - new_val) / max(trial[mask2].sum(), 1e-15)
                    trial[mask2] *= scale
                    trial = np.maximum(trial, 0)
                    trial = trial / trial.sum()

                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                f[j] = best_j_val
                mask2 = np.ones(n, dtype=bool)
                mask2[j] = False
                osum = f[mask2].sum()
                if osum > 1e-15:
                    f[mask2] *= (1.0 - best_j_val) / osum
                f = np.maximum(f, 0)
                f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            no_improve_rounds += 1
            if no_improve_rounds > 5:
                break
        else:
            no_improve_rounds = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
