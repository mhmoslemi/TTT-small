    """Compute B^T d where B is the matrix such that conv(b, a) = B @ b.
    d has length len(a) + len(b) - 1 (the length of the convolution output).
    Returns vector of length len(b) = len(d) - len(a) + 1.
    """
    n_a = len(a)
    n_d = len(d)
    n_b = n_d - n_a + 1
    rev_a = a[::-1]
    # conv(d, rev_a) has length n_d + n_a - 1

    conv = np.fft.irfft(D * RA, fft_size)[:n_d + n_a - 1]
    return conv[n_a - 1 : n_a - 1 + n_b]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Frequency-domain phase optimization (TR-like)
2. QP-based refinement exploiting Toeplitz/convolution structure
3. Smoothed max subgradient with Nesterov acceleration
4. Coordinate descent and multiplicative weight updates
5. Multi-length exploration

import numpy as np
import time

def fast_convolve(a, b=None):
    if b is None:
        b = a
    n = len(a) + len(b) - 1
    fft_size = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:n]

def convolve(a, b):
    return fast_convolve(a, b)

def conv_adjoint(d, a):
    """Adjoint of 'convolve with a' operator.
    If y = convolve(b, a), then the gradient of y^T y w.r.t. b is 2*conv_adjoint(y, a).
    More precisely: if f(b) = convolve(b, a), then f^T(d) = conv_adjoint(d, a).
    """
    n_a = len(a)
    n_d = len(d)
    n_b = n_d - n_a + 1
    if n_b <= 0:
        return np.zeros(0)
    rev_a = a[::-1]
    total_len = n_d + n_a - 1
    fft_size = 1 << (total_len - 1).bit_length()
    D = np.fft.rfft(d, fft_size)
    RA = np.fft.rfft(rev_a, fft_size)
    conv = np.fft.irfft(D * RA, fft_size)[:total_len]
    return conv[n_a - 1 : n_a - 1 + n_b]

def project_simplex(v, target_sum=1.0):
    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def qp_step(b, a, target, lr=0.01, n_iters=100):
    """One round of projected gradient on min ||conv(b,a) - target||^2 s.t. sum(b)=1, b>=0.
    Returns updated b.
    """
    n = len(b)
    b = b.copy()
    for _ in range(n_iters):
        conv_ba = fast_convolve(b, a)
        residual = conv_ba - target
        grad = 2.0 * conv_adjoint(residual, a)
        # Center gradient for simplex
        grad = grad - grad.mean()
        b_new = project_simplex(b - lr * grad, 1.0)
        # Line search
        obj_old = np.sum(residual**2)
        obj_new = np.sum((fast_convolve(b_new, a) - target)**2)
        if obj_new < obj_old:
            b = b_new
        else:
            # Halve step
            b_mid = project_simplex(b - 0.5 * lr * grad, 1.0)
            obj_mid = np.sum((fast_convolve(b_mid, a) - target)**2)
            if obj_mid < obj_old:
                b = b_mid
            else:
                break
    return b

def freq_phase_optimize(a, n_iters=50, lr=0.5):
    """Optimize phases in frequency domain to minimize peak of auto-convolution.
    Returns a non-negative time-domain sequence (with clipping).
    """
    n = len(a)
    L = 1 << (2 * n - 1 - 1).bit_length()
    A = np.fft.fft(a, L)
    # Keep magnitudes, optimize phases
    mags = np.abs(A)
    mags[0] = np.sum(a)  # DC component is fixed (equals sum)
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    
    for it in range(n_iters):
        # Compute current auto-convolution
        conv = fast_convolve(a)
        peak = conv.max()
        
        # TR-like update: A -> A - lr * A * IFFT(|A|^2) / |IFFT(|A|^2)|^2 ... 
        # Actually, the standard TR update is:
        # x_new = x - lr * x * (|x|^2 - |x_target|^2) / |x|^2 ... 
        # Let me use a simpler approach:
        # The peak-reduction step: find the peak location, adjust phases to reduce it.
        
        # Simpler: gradient in phase domain
        # f = max_k IFFT(|A|^2)[k] ≈ use softmax smoothing
        tau = peak / 20
        logits = (conv - peak) / tau
        w = np.exp(logits)
        w = w / w.sum()
        
        # Gradient of sum_k w_k * conv_k w.r.t. A
        # conv = IFFT(|A|^2)
        # d(conv_k)/dA_m = ... this gets complicated
        # Let's just do: A -> A * (1 - lr * IFFT(|A|^2 * something))
        
        # Alternative: use the analogy with OFDM peak reduction
        # x = IFFT(A), we want to reduce peaks in x
        # Update: A_new = A - lr * FFT(x * d) where d is a "direction"
        # For auto-convolution, the "signal" is IFFT(|A|^2), not IFFT(A)
        
        # Let me try a different approach: 
        # minimize (1/T) sum exp(T * conv_k) by adjusting phases
        # This is like minimizing the log-sum-exp of the auto-convolution
        
        # The gradient w.r.t. phase of A_m:
        # Let A_m = |A_m| * exp(i*phi_m)
        # d(conv_k)/dphi_m = ... 
        
        # This is getting too complex. Let me just do a simple simulated annealing
        # on the phases.
        
        # Random phase perturbation
        delta = np.random.standard_normal(L) * 0.1
        A_trial = A + 1j * delta * mags
        A_trial[0] = mags[0]  # keep DC
        a_trial = np.fft.ifft(A_trial).real[:n]
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-10:

        a_trial = a_trial / s

            best_a = a_trial.copy()
            best_obj = obj_trial

    n_out = len(a) + len(b) - 1
    fft_size = 1 << (n_out - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:n_out]


def conv_adjoint(d, a):
    """Adjoint of 'convolve with a': if y = conv(b, a), then adjoint maps d -> B^T d.
    (B^T d)_i = sum_k d_k * a_{k-i} for valid indices.
    """
    n_a = len(a)
    n_d = len(d)
    n_b = n_d - n_a + 1
    if n_b <= 0:
        return np.zeros(0)
    rev_a = a[::-1]
    total_len = n_d + n_a - 1
    fft_size = 1 << (total_len - 1).bit_length()
    D = np.fft.rfft(d, fft_size)
    RA = np.fft.rfft(rev_a, fft_size)
    conv = np.fft.irfft(D * RA, fft_size)[:total_len]
    return conv[n_a - 1: n_a - 1 + n_b]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def estimate_lipschitz(a, n_power_iters=20):
    """Estimate Lipschitz constant of grad of ||conv(b,a)||^2 via power iteration on B^T B."""
    n = len(a)
    # B^T B is the auto-correlation matrix of a (Toeplitz)
    # Apply to a random vector
    rng = np.random.default_rng(0)
    v = rng.standard_normal(n)
    v /= np.linalg.norm(v)
    lambda_est = 0.0
    for _ in range(n_power_iters):
        # (B^T B) v = B^T (B v) = B^T (conv(v, a))
        conv_v = fast_convolve(v, a)
        Bt_conv = conv_adjoint(conv_v, a)
        # Power iteration step
        v_new = Bt_conv / (np.linalg.norm(Bt_conv) + 1e-30)
        lambda_est = float(v @ Bt_conv)
        v = v_new
    return lambda_est


def qp_alternate_step(a_current, target, n_iters=200, verbose=False):
    """Solve min ||conv(b, a_current) - target||^2 s.t. sum(b)=1, b>=0 using FISTA.
    Returns optimal b.
    """
    n = len(a_current)
    L = estimate_lipschitz(a_current, 15)
    L = max(L * 1.1, 0.1)  # Safety margin
    # Step size for FISTA: 1/L
    step = 1.0 / L

    b = a_current.copy()
    b_prev = a_current.copy()
    t_val = 1.0

    best_obj = np.inf
    best_b = b.copy()

    for it in range(n_iters):
        y = b + ((t_val - 1) / (t_val + 1)) * (b - b_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = np.ones(n) / n
        else:
            y = y / sy

        # Gradient at y: 2 * B^T (conv(y, a) - target)
        conv_ya = fast_convolve(y, a_current)
        residual = conv_ya - target
        grad = 2.0 * conv_adjoint(residual, a_current)

        # Proximal step: project onto simplex
        b_new = project_simplex(y - step * grad, 1.0)

        # Evaluate
        conv_new = fast_convolve(b_new, a_current)
        obj = np.sum((conv_new - target) ** 2)
        if obj < best_obj:
            best_obj = obj
            best_b = b_new.copy()

        b_prev = b
        b = b_new
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val ** 2)) / 2.0
        t_val = t_new

        if verbose and it % 50 == 0:
            print(f"      QP iter {it}: obj={obj:.8f}")

    return best_b


def nesterov_smoothed_max(a0, time_limit, tau0=50.0, verbose=False):
    """Nesterov-accelerated proximal gradient on log-sum-exp smoothed max of auto-convolution."""
    n = len(a0)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    a_prev = a.copy()
    t_val = 1.0
    tau = tau0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Compute smoothed gradient
        conv = fast_convolve(a)
        peak = conv.max()
        logits = (conv - peak) / tau
        w = np.exp(logits)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws

        # Gradient: grad_i = 2 * sum_k w_k * a_{k-i} = 2 * (a * w_reversed)(appropriate)
        # grad = 2 * conv_adjoint_w where we need sum_k w_k * a_{k-i}
        # This is the same as conv_adjoint but with w instead of residual
        # Actually: sum_k w_k * a_{k-i} is the cross-correlation of w and a
        # = conv_adjoint(w, a) ... let me check
        # conv_adjoint(d, a)_i = sum_k d_k * a_{k-i}
        # Yes! grad_i = 2 * conv_adjoint(w, a)_i

        grad = 2.0 * conv_adjoint(w, a)
        # Center for simplex
        grad_c = grad - grad.mean()

        # Nesterov step
        y = a + ((t_val - 1) / (t_val + 1)) * (a - a_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = np.ones(n) / n
        else:
            y = y / sy

        # Adaptive step size
        lr = 1.0 / (tau * max(peak, 0.01) * 2.0)
        lr = min(lr, 0.5)

        a_new = project_simplex(y - lr * grad_c, 1.0)

        # Evaluate true objective
        obj_new = fast_convolve(a_new).max()
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a_prev = a
        a = a_new
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val ** 2)) / 2.0
        t_val = t_new

        # Adaptive parameters
        if no_improve > 200:
            tau *= 0.95
            t_val = max(t_val, 1.0)
            if no_improve > 500:
                # Restart with perturbation
                rng_r = np.random.default_rng(int(time.monotonic() * 1000))
                noise = rng_r.standard_normal(n) * 0.003
                a = project_simplex(best_a + noise, 1.0)
                a_prev = a.copy()
                t_val = 1.0
                no_improve = 0
                tau = tau0

        t_val = min(t_val, 500.0)

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NS iter: obj={best_obj:.10f}, tau={tau:.1f}, t={t_val:.0f} ({elapsed:.0f}s)")

    return best_a, best_obj


def multiplicative_weight_optimize(a0, time_limit, verbose=False):
    """Mirror descent with entropic regularization (multiplicative weight updates)."""
    n = len(a0)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    eta = 0.5
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        conv = fast_convolve(a)
        peak = conv.max()

        # Softmax weights for the gradient
        tau = max(peak / 30, 1e-6)
        logits = (conv - peak) / tau
        w = np.exp(logits)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws

        # Gradient direction
        grad = conv_adjoint(w, a)

        # Multiplicative update: a_i <- a_i * exp(-eta * grad_i)
        # Use log-space for numerical stability
        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - eta * grad
        a_new = np.exp(log_a_new - log_a_new.max())
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 0.0)

        obj_new = fast_convolve(a_new).max()
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Try smaller step
            for shrink in [0.5, 0.25]:
                log_a_s = log_a - eta * shrink * grad
                a_s = np.exp(log_a_s - log_a_s.max())
                a_s = a_s / a_s.sum()
                obj_s = fast_convolve(a_s).max()
                if obj_s < best_obj - 1e-13:
                    best_obj = obj_s
                    best_a = a_s.copy()
                    a_new = a_s
                    no_improve = 0
                    break

        a = a_new

        # Adapt eta
        if no_improve > 100:
            eta *= 0.9
        elif no_improve < 20:
            eta = min(eta * 1.01, 2.0)

        if no_improve > 400:
            rng_r = np.random.default_rng(iter_count * 17 + 3)
            noise = rng_r.standard_normal(n) * 0.002
            a = project_simplex(best_a + noise, 1.0)
            no_improve = 0
            eta = 0.5

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MW iter {iter_count}: obj={best_obj:.10f}, eta={eta:.3f} ({elapsed:.0f}s)")

    return best_a, best_obj


def coordinate_descent(a0, time_limit, verbose=False):
    """Coordinate descent with multiplicative scaling of other coordinates."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    pass_count = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            multipliers = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved and pass_count > 2:
            break

    return best_a, best_obj


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        A[k] = np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def full_qp_alternate(a0, time_limit, verbose=False):
    """Full QP-based alternating optimization.
    Alternately minimize ||conv(b, a) - flat_target||^2 and update a <- b.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    outer_count = 0

    while time.monotonic() - t0 < time_limit:
        outer_count += 1
        # Compute target: flat (ideal) or partially flat
        current_conv = fast_convolve(a)
        flat_target = np.ones(2 * n - 1) / (2 * n - 1)
        # Blend: use mostly flat target
        alpha = min(1.0, outer_count / 5.0)  # Gradually increase flatness
        target = alpha * flat_target + (1 - alpha) * current_conv

        # Solve QP
        inner_iters = max(50, min(200, int((time_limit - (time.monotonic() - t0)) / 0.05)))
        b = qp_alternate_step(a, target, n_iters=inner_iters)

        # Update a (with damping)
        damp = 0.7
        a_new = damp * a + (1 - damp) * b
        a_new = project_simplex(a_new, 1.0)

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
        else:
            # Try without damping
            a_try = b
            obj_try = fast_convolve(a_try).max()
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
            else:
                # Try full damping
                a_try2 = 0.5 * a + 0.5 * b
                a_try2 = project_simplex(a_try2, 1.0)
                obj_try2 = fast_convolve(a_try2).max()
                if obj_try2 < best_obj - 1e-14:
                    best_obj = obj_try2
                    best_a = a_try2.copy()
                    a = a_try2
                else:
                    a = a_new  # Accept anyway to move forward

        if verbose and outer_count % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"      QP-Alt outer {outer_count}: obj={best_obj:.10f} ({elapsed:.0f}s)")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining QP-based alternating, Nesterov smoothed max,
    multiplicative weights, and coordinate descent for minimizing the auto-convolution peak.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        ss = a.sum()
        if ss < 1e-10:
            return float('inf')
        a_norm = a / ss
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # === Phase 1: Initialization ===
    print("\n=== Phase 1: Initialization ===")
    t_phase1 = time.monotonic()
    target_lengths = []
    if parent is not None:
        target_lengths.append(len(parent))
    for n_try in [384, 512, 640, 768, 896, 1024, 1280]:
        if n_try not in target_lengths:
            target_lengths.append(n_try)
    target_lengths = sorted(set(target_lengths), key=lambda x: abs(x - (len(parent) if parent is not None else 640)))
    target_lengths = target_lengths[:5]

    inits = {}
    for n in target_lengths:
        best_init = None
        best_init_obj = float('inf')
        methods = [
            ("flat_spec", flat_spectrum_init(n, rng)),
        ]
        # Random exponential
        a_rand = rng.exponential(1.0, n)
        methods.append(("random", a_rand / a_rand.sum()))
        # Interpolated parent
        if parent is not None:
            methods.append(("interp_p", interpolate_sequence(parent, n)))
        # Reversed interpolated
        if parent is not None:
            methods.append(("interp_pr", interpolate_sequence(parent[::-1], n)))

        for name, init in methods:
            try:
                init = np.maximum(init, 0)
                ss = init.sum()
                if ss < 1e-10:
                    continue
                init = init / ss
                obj = fast_convolve(init).max()
                if obj < best_init_obj:
                    best_init_obj = obj
                    best_init = init
            except Exception:
                pass
        if best_init is not None:
            inits[n] = (best_init, best_init_obj)
            print(f"  n={n}: best init obj={best_init_obj:.6f}")

    print(f"  Phase 1 done in {time.monotonic()-t_phase1:.1f}s")

    # === Phase 2: QP-based alternating optimization ===
    print(f"\n=== Phase 2: QP Alternating (best={best_score:.6f}) ===")
    t_phase2 = time.monotonic()
    sorted_inits = sorted(inits.items(), key=lambda x: x[1][1])
    n_qp = min(3, len(sorted_inits))
    time_per_qp = min(120, max(30, (deadline - time.monotonic() - 600) / max(n_qp, 1)))

    for rank, (n_len, (init, init_obj)) in enumerate(sorted_inits[:n_qp]):
        rem = deadline - time.monotonic() - 500
        if rem < 20:
            break
        t_alloc = min(time_per_qp, rem)
        print(f"  QP-Alt n={n_len} (init_obj={init_obj:.6f}), t={t_alloc:.0f}s")
        a_opt, obj = full_qp_alternate(init, time_limit=t_alloc)
        try_seq(a_opt, f"qp_n{n_len}_r{rank}")

    print(f"  Phase 2 done in {time.monotonic()-t_phase2:.1f}s")

    # === Phase 3: Nesterov smoothed max ===
    print(f"\n=== Phase 3: Nesterov Smoothed Max (best={best_score:.6f}) ===")
    t_phase3 = time.monotonic()
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_alloc = min(rem * 0.5, 150)
            print(f"  NS t={t_alloc:.0f}s")
            a_opt, obj = nesterov_smoothed_max(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "nesterov")

        # Also try on best init
        if sorted_inits:
            rem = deadline - time.monotonic() - 300
            if rem > 20:
                best_n, (best_init, best_io) = sorted_inits[0]
                t_alloc = min(rem * 0.4, 80)
                a_opt, obj = nesterov_smoothed_max(best_init, time_limit=t_alloc)
                try_seq(a_opt, f"nesterov_init_n{best_n}")

    print(f"  Phase 3 done in {time.monotonic()-t_phase3:.1f}s")

    # === Phase 4: Multiplicative weight optimization ===
    print(f"\n=== Phase 4: Multiplicative Weights (best={best_score:.6f}) ===")
    t_phase4 = time.monotonic()
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_alloc = min(rem * 0.4, 120)
            print(f"  MW t={t_alloc:.0f}s")
            a_opt, obj = multiplicative_weight_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "mw")

        # Try on perturbed best
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            n_best = len(best_seq)
            noise = rng.standard_normal(n_best) * 0.005
            pert = project_simplex(best_seq + noise, 1.0)
            t_alloc = min(rem * 0.4, 80)
            a_opt, obj = multiplicative_weight_optimize(pert, time_limit=t_alloc)
            try_seq(a_opt, "mw_perturb")

    print(f"  Phase 4 done in {time.monotonic()-t_phase4:.1f}s")

    # === Phase 5: QP alternating on best found ===
    print(f"\n=== Phase 5: QP Alt on Best (best={best_score:.6f}) ===")
    t_phase5 = time.monotonic()
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(rem * 0.5, 150)
            print(f"  QP-Alt on best, t={t_alloc:.0f}s")
            a_opt, obj = full_qp_alternate(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "qp_best")

            # Also try with perturbation
            rem = deadline - time.monotonic() - 150
            if rem > 20:
                n_best = len(best_seq)
                for p_noise in [0.003, 0.01]:
                    noise = rng.standard_normal(n_best) * p_noise
                    pert = project_simplex(best_seq + noise, 1.0)
                    t_alloc = min(rem * 0.4, 60)
                    a_opt, obj = full_qp_alternate(pert, time_limit=t_alloc)
                    try_seq(a_opt, f"qp_pert_{p_noise}")
                    rem = deadline - time.monotonic() - 100
                    if rem < 20:
                        break

    print(f"  Phase 5 done in {time.monotonic()-t_phase5:.1f}s")

    # === Phase 6: Coordinate descent ===
    print(f"\n=== Phase 6: Coordinate Descent (best={best_score:.6f}) ===")
    t_phase6 = time.monotonic()
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t_alloc = min(rem * 0.5, 80)
            print(f"  CD t={t_alloc:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "cd")

    print(f"  Phase 6 done in {time.monotonic()-t_phase6:.1f}s")

    # === Phase 7: Longer sequences ===
    print(f"\n=== Phase 7: Longer Sequences (best={best_score:.6f}) ===")
    t_phase7 = time.monotonic()
    if best_seq is not None:
        n_best = len(best_seq)
        for mult in [1.33, 1.5, 2.0]:
            target_n = int(n_best * mult)
            if target_n <= n_best or target_n > 2560:
                continue
            rem = deadline - time.monotonic() - 60
            if rem < 30:
                continue
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.4, 100)
            print(f"  Longer n={target_n}, t={t_alloc:.0f}s")

            # QP alternate
            a_opt, obj = full_qp_alternate(f_interp, time_limit=t_alloc)
            try_seq(a_opt, f"long_qp_n{target_n}")

            # CD
            rem = deadline - time.monotonic() - 30
            if rem > 15:
                a_opt, obj = coordinate_descent(a_opt, time_limit=min(rem, 40))
                try_seq(a_opt, f"long_cd_n{target_n}")

    print(f"  Phase 7 done in {time.monotonic()-t_phase7:.1f}s")

    # === Phase 8: Final intensive search ===
    print(f"\n=== Phase 8: Final Search (best={best_score:.6f}) ===")
    t_phase8 = time.monotonic()
    if best_seq is not None:
        n_best = len(best_seq)
        while time.monotonic() < deadline - 20:
            rem = deadline - time.monotonic() - 15
            if rem < 15:
                break
            # Alternate between methods
            choice = int(time.monotonic()) % 3
            if choice == 0:
                noise = rng.standard_normal(n_best) * 0.002
                f_init = project_simplex(best_seq + noise, 1.0)
                a_opt, obj = nesterov_smoothed_max(f_init, time_limit=min(rem, 30))
                try_seq(a_opt, "final_ns")
            elif choice == 1:
                noise = rng.standard_normal(n_best) * 0.004
                f_init = project_simplex(best_seq + noise, 1.0)
                a_opt, obj = full_qp_alternate(f_init, time_limit=min(rem, 30))
                try_seq(a_opt, "final_qp")
            else:
                a_opt, obj = coordinate_descent(best_seq, time_limit=min(rem, 20))
                try_seq(a_opt, "final_cd")

    print(f"  Phase 8 done in {time.monotonic()-t_phase8:.1f}s")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
