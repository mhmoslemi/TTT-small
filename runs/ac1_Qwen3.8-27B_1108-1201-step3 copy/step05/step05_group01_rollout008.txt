def nesterov_subgradient(f0, time_limit):
    f = f0 / f0.sum()
    y = f.copy()  # ghost point
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    eta = 0.5 / np.sqrt(n)  # initial step size
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute subgradient at y
        conv_y = fast_convolve(y)
        M = conv_y.max()
        active = np.where(conv_y >= M - 1e-10)[0]
        
        # Uniform weights on active set
        lambda_ = np.zeros(len(conv_y))
        lambda_[active] = 1.0 / len(active)
        
        # Subgradient: s[i] = 2 * sum_k lambda_[k] * y[k-i] (with bounds)
        s = np.zeros(n)
        for k in active:
            w = lambda_[k] * 2.0
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            s[idx] += w * y[k - idx]
        
        # Proximal step
        f_new = project_simplex(y - eta * s)
        
        # Update ghost point (Nesterov)
        beta_t = (1 + np.sqrt(1 + 4 * (t-1)**2)) / 2  # or simpler: 2/(t+2)
        y = f_new + ((t-1) / (t+1)) * (f_new - f)
        # Actually the standard Nesterov update is:
        # y_{t+1} = f_{t+1} + (t/(t+2)) * (f_{t+1} - f_t)
        
        f = f_new
        obj = fast_convolve(f).max()
        if obj < best_obj:
            best_obj = obj
            best_f = f.copy()
        
        eta = 0.5 / np.sqrt(n * t)  # diminishing step size
    
    return best_f, best_obj

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def conv_max(a):
    """Compute just the max of auto-convolution (fast path)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(A * A, fft_size)[:size]
    return conv.max(), conv


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def compute_subgradient(f, conv, tol=1e-8):
    """Compute exact subgradient of max_k (f*f)[k] using active set.
    Returns (subgradient_vector, max_val, active_indices)."""
    M = conv.max()
    active = np.where(conv >= M - tol * max(M, 1.0))[0]
    n = len(f)
    s = np.zeros(n)
    for k in active:
        w = 1.0 / len(active)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        s[idx] += w * f[k - idx]
    s *= 2.0
    return s, M, active


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def smooth_lse_obj(t, beta):
    """Log-sum-exp smoothed objective in softmax parameterization."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    shifted -= shifted.max()
    log_sum = np.log(np.exp(shifted).sum())
    obj = conv_max + log_sum / beta

    # Gradient
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()

    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R

    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)

    return obj, grad_t

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    from scipy.interpolate import PchipInterpolator
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    try:
        interp = PchipInterpolator(old_x, f, fill_value="extrapolate")
        f_interp = interp(new_x)
    except Exception:
        f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def cosine_window(n, alpha=1.0):
    """Cosine window sequence."""
    x = np.linspace(0, np.pi, n)
    w = np.cos(alpha * x / 2.0) ** 2
    w = np.maximum(w, 0)
    s = w.sum()
    if s > 0:
        w /= s
    return w


def triangular_window(n):
    """Triangular window sequence."""
    x = np.arange(n)
    mid = (n - 1) / 2.0
    w = 1.0 - np.abs(x - mid) / (mid + 0.5)
    w = np.maximum(w, 0)
    s = w.sum()
    if s > 0:
        w /= s
    return w

        a_opt, _ = nesterov_subgrad(f_interp, time_limit=t_alloc * 0.5)
        try_seq(a_opt, f"nesterov_n{target_n}")

        # Pairwise on interpolated
        rem2 = deadline - time.monotonic()
        if rem2 > 30:
            a_opt, _ = pairwise_local_search(a_opt, time_limit=min(rem2 * 0.3, 30))
            try_seq(a_opt, f"pairwise_n{target_n}")

    # === Phase 6: Diverse initializations (100s) ===
    print(f"\n=== Phase 6: Diverse Starts (best={best_score:.6f}) ===")
    n_ref = len(best_seq) if best_seq is not None else n_parent

    initializations = [
        ("cosine", cosine_window(n_ref)),
        ("triangular", triangular_window(n_ref)),
        ("random_smooth", None),  # handled below
        (" randomized_flat", None),  # handled below
    ]

    for name, init in initializations:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        if init is not None:
            a_opt, _ = nesterov_subgrad(init, time_limit=30)
            try_seq(a_opt, f"div_{name}")
        elif name == "random_smooth":
            # Random but smooth: interpolate a random 64-point sequence
            pts = rng.random(64)
            pts = np.maximum(pts, 0.01)
            init_s = interpolate_sequence(pts, n_ref)
            a_opt, _ = nesterov_subgrad(init_s, time_limit=30)
            try_seq(a_opt, "div_random_smooth")
        elif name == " randomized_flat":
            # Nearly uniform with small random perturbation
            init_flat = np.ones(n_ref) / n_ref + rng.standard_normal(n_ref) * 0.01
            init_flat = np.maximum(init_flat, 0)
            init_flat = init_flat / init_flat.sum()
            a_opt, _ = nesterov_subgrad(init_flat, time_limit=30)
            try_seq(a_opt, "div_flat_perturbed")

    # === Phase 7: Final deep optimization (remaining time) ===
    print(f"\n=== Phase 7: Final Deep (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # 7a: LP-FW deep
        rem = deadline - time.monotonic() - 60
        if rem > 30:
            t_alloc = min(rem * 0.4, 150)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "final_lpfw")

        # 7b: Pairwise on best
        rem = deadline - time.monotonic() - 40
        if rem > 20:
            t_alloc = min(rem * 0.5, 100)
            a_opt, _ = pairwise_local_search(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_pairwise")

        # 7c: Coordinate descent
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_alloc = min(rem * 0.6, 80)
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_cd")

        # 7d: Nesterov adaptive on best
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t_alloc = min(rem * 0.7, 60)
            a_opt, _ = nesterov_adaptive(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_nesterov_ad")

        # 7e: L-BFGS final
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            t_alloc = min(rem * 0.8, 50)
            a_opt, _ = lbfgs_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_lbfgs")

        # 7f: One more pairwise pass
        rem = deadline - time.monotonic() - 3
        if rem > 5:
            a_opt, _ = pairwise_local_search(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "final_pairwise2")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
