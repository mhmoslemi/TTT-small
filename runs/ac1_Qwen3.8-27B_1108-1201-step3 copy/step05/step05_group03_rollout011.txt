def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

"""
Minimizes 2*n*max(auto_conv)/(sum)^2 using LP-based direction finding
(equivalent to the state-of-the-art LP approach), random restarts,
multi-length exploration, and a soft-constraint variant for escaping
local minima. The core LP finds the best non-negative direction g that
keeps the cross-correlation (f*g) bounded by the current peak M,
then blends to monotonically reduce the peak.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_cross_matrix(f):
    """Build A where A[k,i] = f[k-i] if 0<=k-i<n, else 0.
    So that (f*g)[k] = (A @ g)[k]."""
    n = len(f)
    k_range = np.arange(2 * n - 1)[:, None]
    i_range = np.arange(n)[None, :]
    idx = k_range - i_range
    mask = (idx >= 0) & (idx < n)
    A = np.where(mask, f[idx.clip(0, n - 1)], 0.0)
    return A


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns (g, success)."""
    n = len(f)
    c = -np.ones(n)
    A = build_cross_matrix(f)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
        if result.success:
            return result.x, True
        return None, False
    except Exception:
        return None, False


def find_optimal_blend(f, g, current_peak):
    """Find t in [0,1] minimizing max(((1-t)*f + t*g) * ((1-t)*f + t*g))."""
    best_t = 0.0
    best_peak = current_peak

    # Coarse grid
    for t in np.linspace(0.01, 0.99, 40):
        h = (1.0 - t) * f + t * g
        peak = fast_convolve(h).max()
        if peak < best_peak:
            best_peak = peak
            best_t = t

    if best_t == 0.0:
        return 0.0, current_peak

    # Golden section refinement
    span = 2.0 / 40.0
    lo = max(0.0, best_t - span)
    hi = min(1.0, best_t + span)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for _ in range(40):
        if hi - lo < 1e-7:
            break
        m1 = hi - gr * (hi - lo)
        m2 = lo + gr * (hi - lo)
        p1 = fast_convolve((1.0 - m1) * f + m1 * g).max()
        p2 = fast_convolve((1.0 - m2) * f + m2 * g).max()
        if p1 < p2:
            hi = m2
        else:
            lo = m1
        mid = (lo + hi) / 2.0
        pm = fast_convolve((1.0 - mid) * f + mid * g).max()
        if pm < best_peak:
            best_peak = pm
            best_t = mid

    return best_t, best_peak

def structured_init(n, kind='hann'):
    """Create structured initial sequences."""
    if kind == 'hann':
        x = np.arange(n)
        a = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n - 1)))
    elif kind == 'blackman':
        x = np.arange(n)
        a = 0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1)) \
            + 0.08 * np.cos(4.0 * np.pi * x / (n - 1))
    elif kind == 'gauss':
        x = np.arange(n) - n / 2.0
        sigma = 0.3 * n
        a = np.exp(-x ** 2 / (2.0 * sigma ** 2))
    elif kind == 'tri':
        x = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * x - 1.0)
    elif kind == 'uniform':
        a = np.ones(n)
    else:
        a = np.ones(n)
    return normalize(np.maximum(a, 0.0))

            f_try[j] = v_new
            # Renormalize
            f_try = normalize(f_try)
            sc = compute_score(f_try)
            if sc < best_sc:
                best_sc = sc
                best_v = v_new

        if best_v != old:
            f[j] = best_v
            f = normalize(f)
            if best_sc < best_score:
                best_score = best_sc
                best_f = f.copy()

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function. Uses LP-based direction finding (the key technique
    from the state-of-the-art paper) combined with random restarts, multi-length
    exploration, and coordinate polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_a = None
    best_score = float('inf')

    def update_best(a, label):
        nonlocal best_a, best_score
        a = normalize(a)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")
        return sc

    # === Get parent ===
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        parent = normalize(parent)
        parent_score = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={parent_score:.10f}")
        update_best(parent, "parent")
    except Exception:
        parent = structured_init(768, 'hann')
        parent_score = compute_score(parent)
        print(f"No parent, using Hann: n={len(parent)}, score={parent_score:.10f}")
        update_best(parent, "hann_init")

    # === Phase 1: LP optimization from parent (main workhorse) ===
    print("\n=== Phase 1: LP optimization from parent ===")
    t_avail = deadline - time.monotonic() - 300
    t_lp = max(120, min(t_avail * 0.45, 600))
    print(f"  Time allocated: {t_lp:.0f}s")
    a1, sc1 = lp_optimize(parent, time_limit=t_lp, verbose=True, rng=rng)
    update_best(a1, "LP_parent")

    # === Phase 2: Soft LP (relaxed constraints for bigger moves) ===
    print(f"\n=== Phase 2: Soft LP (best={best_score:.10f}) ===")
    if time.monotonic() < deadline - 200:
        t_avail = deadline - time.monotonic() - 200
        t_sl = max(60, min(t_avail * 0.25, 200))
        print(f"  Time allocated: {t_sl:.0f}s")
        a2, sc2 = soft_lp_optimize(best_a, time_limit=t_sl, verbose=True, rng=rng)
        update_best(a2, "SoftLP")

    # === Phase 3: Random restarts + LP ===
    print(f"\n=== Phase 3: Random restarts (best={best_score:.10f}) ===")
    if time.monotonic() < deadline - 150:
        n_restarts = 0
        while time.monotonic() < deadline - 100 and n_restarts < 8:
            n_restarts += 1
            scale = rng.choice([0.005, 0.01, 0.02, 0.03, 0.05])
            noise = rng.standard_normal(len(best_a)) * scale
            a_pert = normalize(best_a + noise)
            t_rem = min(deadline - time.monotonic() - 60, 120)
            if t_rem < 30:
                break
            a_r, sc_r = lp_optimize(a_pert, time_limit=t_rem, verbose=False, rng=rng)
            update_best(a_r, f"restart_{n_restarts}")

    # === Phase 4: Different lengths ===
    print(f"\n=== Phase 4: Length exploration (best={best_score:.10f}) ===")
    if time.monotonic() < deadline - 120:
        target_lengths = [640, 896, 1024, 512]
        for tn in target_lengths:
            if time.monotonic() > deadline - 60:
                break
            # Interpolate best sequence to new length
            x_old = np.linspace(0, 1, len(best_a))
            x_new = np.linspace(0, 1, tn)
            a_interp = np.interp(x_new, x_old, best_a)
            a_interp = normalize(a_interp)

            t_rem = min(deadline - time.monotonic() - 40, 150)
            if t_rem < 30:
                break
            a_i, sc_i = lp_optimize(a_interp, time_limit=t_rem, verbose=False, rng=rng)
            update_best(a_i, f"len_{tn}")

    # === Phase 5: Structured inits + LP ===
    print(f"\n=== Phase 5: Structured inits (best={best_score:.10f}) ===")
    if time.monotonic() < deadline - 80:
        for kind in ['blackman', 'gauss', 'tri']:
            if time.monotonic() > deadline - 40:
                break
            n_struct = len(best_a)
            a_s = structured_init(n_struct, kind)
            t_rem = min(deadline - time.monotonic() - 30, 100)
            if t_rem < 20:
                break
            a_opt, sc_opt = lp_optimize(a_s, time_limit=t_rem, verbose=False, rng=rng)
            update_best(a_opt, f"struct_{kind}")

    # === Phase 6: Final polynomial LP pass (most iterations) ===
    print(f"\n=== Phase 6: Final LP polish (best={best_score:.10f}) ===")
    if time.monotonic() < deadline - 40:
        t_final = min(deadline - time.monotonic() - 20, 300)
        a_fin, sc_fin = lp_optimize(best_a, time_limit=t_final, verbose=True, rng=rng)
        update_best(a_fin, "final_LP")

    # === Phase 7: Coordinate polish ===
    print(f"\n=== Phase 7: Coordinate polish (best={best_score:.10f}) ===")
    if time.monotonic() < deadline - 20:
        t_cp = min(deadline - time.monotonic() - 10, 60)
        a_cp, sc_cp = coordinate_polish(best_a, time_limit=t_cp)
        update_best(a_cp, "coord_polish")

    # === Final verification ===
    if best_a is None:
        best_a = structured_init(768, 'hann')

    final_list = [float(max(0.0, min(1000.0, x))) for x in best_a]
    final_score = float(evaluate_sequence(final_list))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_a)})")
    return final_list
