def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def compute_subgrad_weighted(f, conv, temperature):
    """Compute weighted subgradient using soft peak weighting."""

    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= ws
    # Cross-correlation: sg[j] = 2 * sum_k w[k] * f[k-j]
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(f, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    return 2.0 * R, w

def coord_descent_fast(f0, time_limit, top_coords=50):

        if top_coords < n:
            high_idx = np.argsort(sg)[-top_coords:]
            low_idx = np.argsort(sg)[:top_coords]
            active = np.unique(np.concatenate([high_idx, low_idx]))
        else:
            active = np.arange(n)

        improved = False
        for j in active:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-10:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in [0.6, 0.8, 0.9, 1.1, 1.2, 1.4]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def fourier_init(n, num_coeffs, rng):
    """Generate a Fourier-parameterized initial sequence."""
    # Use random Fourier coefficients

    x = np.arange(n) / n

    f = np.maximum(f, 0.0)

    if s > 1e-10:
        f /= s
    return f

    """Cosine-shaped initialization."""
    x = np.arange(n) / n
    f = 1.0 + a * np.cos(2 * np.pi * x + phi)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s > 1e-10:
        f /= s
    return f


def gaussian_init(n, center_frac, width_frac):
    """Gaussian bump initialization."""
    x = np.arange(n)
    center = center_frac * n
    width = width_frac * n
    f = np.exp(-0.5 * ((x - center) / width) ** 2)
    f += 0.1 * np.ones(n)  # Add baseline to keep all positive
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s > 1e-10:
        f /= s
    return f


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining Nesterov subgradient, LP-FW,
    coordinate descent, and Fourier initializations with structured restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512

    # === Phase 1: Multi-start exploration (250s) ===
    print("\n=== Phase 1: Multi-Start Exploration ===")
    n_parent = len(parent)
    starts = []
    labels = []

    # Start 1: Parent
    starts.append(parent.copy())
    labels.append("parent")

    # Start 2: Uniform
    starts.append(np.ones(n_parent) / n_parent)
    labels.append("uniform")

    # Start 3: Cosine
    starts.append(cosine_init(n_parent, 0.3, 0.5))
    labels.append("cosine")

    # Start 4: Gaussian
    starts.append(gaussian_init(n_parent, 0.5, 0.3))
    labels.append("gaussian")

    # Start 5: Fourier random
    starts.append(fourier_init(n_parent, 20, rng))
    labels.append("fourier1")

    # Start 6: Fourier random 2
    starts.append(fourier_init(n_parent, 40, rng))
    labels.append("fourier2")

    # Start 7: Parent reversed
    starts.append(parent[::-1].copy())
    labels.append("parent_rev")

    # Start 8: Parent + smoothed noise
    noise = rng.standard_normal(n_parent) * 0.05
    # Smooth the noise
    kernel = np.ones(5) / 5
    noise_smooth = np.convolve(noise, kernel, mode='same')
    f_smooth = parent + noise_smooth
    f_smooth = np.maximum(f_smooth, 0)
    f_smooth /= f_smooth.sum()
    starts.append(f_smooth)
    labels.append("parent_noise")

    t_per_start = min(30, (deadline - time.monotonic() - 1400) / len(starts))
    for i, (start_f, label) in enumerate(zip(starts, labels)):
        rem = deadline - time.monotonic() - 1300
        if rem < 10:
            break
        t_alloc = min(t_per_start, rem)
        if t_alloc < 5:
            break
        # Quick subgradient optimization
        a_opt, _ = nesterov_subgrad(start_f, time_limit=t_alloc,
                                     lr0=0.003, momentum=0.85,
                                     top_k=50, temp0=0.00005)
        try_seq(a_opt, f"ns_{label}")

    print(f"  After Phase 1: best={best_score:.6f}")

    # === Phase 2: Deep LP-FW optimization (900s) ===
    print(f"\n=== Phase 2: Deep LP-FW (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    t2 = min(900, deadline - time.monotonic() - 500)
    if t2 > 30:
        a_opt, _ = lp_fw_v2(base, time_limit=t2, verbose=True)
        try_seq(a_opt, "lpfw_deep")

    # === Phase 3: Nesterov subgradient deep (300s) ===
    print(f"\n=== Phase 3: Nesterov Deep (best={best_score:.6f}) ===")
    if best_seq is not None:
        t3 = min(300, deadline - time.monotonic() - 300)
        if t3 > 20:
            a_opt, _ = nesterov_subgrad(best_seq, time_limit=t3,
                                         lr0=0.002, momentum=0.92,
                                         top_k=80, temp0=0.00003,
                                         verbose=True)
            try_seq(a_opt, "ns_deep")

    # === Phase 4: Multi-length exploration (200s) ===
    print(f"\n=== Phase 4: Multi-Length (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        # Try nearby lengths
        lengths_to_try = []
        for dn in [-64, -32, 32, 64, 128]:
            nl = n_best + dn
            if 128 <= nl <= 2048:
                lengths_to_try.append(nl)
        # Also try some round numbers
        for nl in [512, 640, 768, 896, 1024, 1280]:
            if nl not in lengths_to_try and 128 <= nl <= 2048:
                lengths_to_try.append(nl)

        for target_n in lengths_to_try[:4]:
            rem = deadline - time.monotonic() - 150
            if rem < 20:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.6, 60)
            if t_alloc < 10:
                continue
            a_opt, _ = lp_fw_v2(f_interp, time_limit=t_alloc)
            try_seq(a_opt, f"len_{target_n}")

    # === Phase 5: Coordinate descent polishing (150s) ===
    print(f"\n=== Phase 5: Coord Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        t5 = min(150, deadline - time.monotonic() - 60)
        if t5 > 15:
            a_opt, _ = coord_descent_fast(best_seq, time_limit=t5, top_coords=60)
            try_seq(a_opt, "cd_polish")

    # === Phase 6: Final deep LP-FW + Nesterov alternation (remaining) ===
    print(f"\n=== Phase 6: Final Alternation (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        attempt = 0
        while time.monotonic() < deadline - 10:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            attempt += 1

            if attempt % 3 == 1:
                # LP-FW with perturbation
                noise = rng.standard_normal(n_best) * 0.003
                f_start = project_simplex(best_seq + noise)
                t_alloc = min(rem * 0.6, 80)
                a_opt, _ = lp_fw_v2(f_start, time_limit=t_alloc)
                try_seq(a_opt, f"final_lpfw_{attempt}")
            elif attempt % 3 == 2:
                # Nesterov subgradient
                t_alloc = min(rem * 0.6, 60)
                a_opt, _ = nesterov_subgrad(best_seq, time_limit=t_alloc,
                                             lr0=0.001, momentum=0.93,
                                             top_k=60, temp0=0.00002)
                try_seq(a_opt, f"final_ns_{attempt}")
            else:
                # Coordinate descent
                t_alloc = min(rem * 0.5, 40)
                a_opt, _ = coord_descent_fast(best_seq, time_limit=t_alloc, top_coords=40)
                try_seq(a_opt, f"final_cd_{attempt}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
