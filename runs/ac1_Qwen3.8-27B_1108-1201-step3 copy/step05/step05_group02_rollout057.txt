"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using active-set LP-Frank-Wolfe,
mirror descent, and multi-scale refinement.
Key innovations: active constraint selection for faster LP, Nesterov acceleration,
and a deflation-based local search.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_active(f, M, n, top_k=50):
    """Solve LP with only the top-k most binding constraints."""
    conv = fast_convolve(f)
    # Get the top-k positions
    k_actual = min(top_k, len(conv))
    active_idx = np.argpartition(conv, -k_actual)[-k_actual:]
    # Sort them by value (highest first)
    active_idx = active_idx[np.argsort(conv[active_idx])[::-1]]
    
    # Build LP: max sum(g) s.t. (f*g)[k] <= M for k in active_idx, g >= 0
    c = -np.ones(n)
    A_ub = np.zeros((len(active_idx), n))
    b_ub = np.full(len(active_idx), M)
    
    for i, k in enumerate(active_idx):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[i, j] = f[k - j]
    
    bounds = [(0, None)] * n
    options = {'time_limit': 30, 'presolve': True, 'dual_feasibility_tolerance': 1e-8}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return res.x
    return None


def solve_lp_full(f, M, n):
    """Solve LP with all constraints (slower but exact)."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {'time_limit': 60, 'presolve': True}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return res.x
    return None

        f = f / s
    n = len(f)

    M = conv.max()

    best_f = f.copy()

    it = 0

            if g is None:

        else:

        # Golden section

        fc = obj_at_alpha(c_)
        fd = obj_at_alpha(d_)

                c_ = d_

            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new

def mirror_descent(f0, time_limit, lr=0.1, decay=0.999, verbose=False):
    """Mirror descent with softmax parameterization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    # Initialize w such that softmax(w) ≈ f
    w = np.log(f + 1e-30)
    v = np.zeros(n)  # momentum
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        
        f = np.exp(w - w.max())
        f = f / f.sum()
        
        conv = fast_convolve(f)
        M = conv.max()
        
        # Subgradient: positions contributing to the peak
        peak_idx = np.argmax(conv)
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        sg = np.zeros(n)
        for j in range(js, je + 1):
            sg[j] = 2.0 * f[peak_idx - j]
        
        # Also use top-k peaks for better signal
        k_actual = min(20, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        pv_norm = pv / (pv.max() + 1e-30)
        
        sg_full = np.zeros(n)
        for k in peaks:
            weight = pv_norm[pv == conv[k]][0] if conv[k] in pv else 0
            js2 = max(0, k - n + 1)
            je2 = min(n - 1, k)
            for j in range(js2, je2 + 1):
                sg_full[j] += weight * f[k - j]
        
        sg = 0.5 * sg + 0.5 * sg_full
        sg = sg / (np.max(sg) + 1e-30)
        
        v = 0.9 * v - lr * sg
        w = w + v
        
        # Renormalize w to prevent overflow
        w = w - w.max()
        
        if M < best_obj - 1e-16:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= decay
        
        if no_improve > 200:
            rng = np.random.default_rng(it * 13)
            f_pert = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            w = np.log(f_pert + 1e-30)
            v = np.zeros(n)
            lr = 0.05
            no_improve = 0

    return best_f, best_obj


def deflation_search(f0, time_limit, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)

        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Find top peaks
        k_actual = min(5, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        
        # For each peak, compute which f positions contribute most
        # and reduce them
        reduction = np.zeros(n)
        for k in peaks:
            if conv[k] < M * 0.95:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                # Contribution of f[j] to conv[k] is f[j]*f[k-j]
                reduction[j] += f[j] * f[k - j] / (M + 1e-30)
        
        # Apply reduction: f_new[j] = f[j] * (1 - eta * reduction[j])

        f_new = f * (1.0 - eta * reduction)
        f_new = np.maximum(f_new, 0.0)
        ss = f_new.sum()
        if ss < 1e-15:
            break
        f_new = f_new / ss
        
        new_obj = fast_convolve(f_new).max()
        if new_obj < best_obj - 1e-16:
            best_obj = new_obj
            best_f = f_new.copy()
            f = f_new
        else:
            # Try smaller eta
            eta = 0.01
            f_new = f * (1.0 - eta * reduction)
            f_new = np.maximum(f_new, 0.0)
            ss = f_new.sum()
            if ss < 1e-15:

            f_new = f_new / ss

                f_pert = project_simplex(best_f + rng.standard_normal(n) * 0.001)

    return best_f, best_obj


def pairwise_swap(f0, time_limit, verbose=False):
    """Try pairwise swaps of mass between positions."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        # Pick two random positions
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        
        # Try moving mass from i to j (and vice versa)
        best_local = best_obj
        best_delta = 0.0
        
        for delta in np.linspace(-0.5, 0.5, 11):
            f_trial = f.copy()
            f_trial[i] += delta * f[i]
            f_trial[j] -= delta * f[j]
            if f_trial[i] < 0 or f_trial[j] < 0:
                continue
            ss = f_trial.sum()
            if ss < 1e-15:
                continue
            f_trial = f_trial / ss
            obj = fast_convolve(f_trial).max()
            if obj < best_local:
                best_local = obj
                best_delta = delta
        
        if best_delta != 0.0:
            f[i] += best_delta * f[i]
            f[j] -= best_delta * f[j]
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-16:

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def coordinate_descent_fast(f0, time_limit, verbose=False):
    """Fast coordinate descent with exact 1D optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        conv = fast_convolve(f)
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            # For coordinate j, the conv values that involve f[j] are:
            # conv[k] for k in [j, j+n-1] (where f[j] * f[k-j] appears)
            # If we change f[j] to f[j]+d, the change in conv[k] is:
            # d * (f[k-j] + f[k-j]) = 2*d*f[k-j] for k in [j, j+n-1]
            # (approximately, since we're keeping sum=1 by scaling others)
            
            # Simpler: try a few values
            old_val = f[j]
            if old_val < 1e-15:
                continue
            
            best_j_obj = conv.max()
            best_j_val = old_val
            
            # Try multiplicative changes
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.9:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                # Renormalize (simple approach: scale all by sum)
                f_trial = f_trial / f_trial.sum()
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj:
                    best_j_obj = obj
                    best_j_val = new_val
            
            if best_j_val != old_val:
                f[j] = best_j_val
                f = f / f.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved and passes > 3:
            break

    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using active-set LP-Frank-Wolfe,
deflation search, mirror descent, and multi-phase local optimization.
Key innovations: active constraint selection for faster LP solving,

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_at_alpha(f, g, alpha):
    """Compute objective at f + alpha*(g-f) projected to simplex."""
    ft = f + alpha * (g - f)
    ft = np.maximum(ft, 0.0)
    st = ft.sum()
    if st < 1e-15:
        return float('inf')
    ft = ft / st
    return fast_convolve(ft).max()


def solve_lp_active(f, M, n, top_k=40):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for top-k k, g >= 0."""
    conv = fast_convolve(f)
    k_actual = min(top_k, len(conv))
    active_idx = np.argpartition(conv, -k_actual)[-k_actual:]
    active_idx = active_idx[np.argsort(conv[active_idx])[::-1]]

    c = -np.ones(n)
    A_ub = np.zeros((len(active_idx), n))
    b_ub = np.full(len(active_idx), M)

    for i, k in enumerate(active_idx):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[i, j] = f[k - j]

    bounds = [(0, None)] * n
    options = {'time_limit': 20, 'presolve': True}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return res.x
    return None

def golden_line_search(f, g, max_iters=35):
    """Golden section search for optimal alpha in [0,1]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)

    fd = obj_at_alpha(f, g, d_pt)

            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = obj_at_alpha(f, g, c_pt)

            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = obj_at_alpha(f, g, d_pt)

        pv = conv[peaks]
        pv_scaled = pv / (pv.max() + 1e-30)
        pv_weights = pv_scaled / pv_scaled.sum()

        sg = np.zeros(n)
        for idx_k in range(len(peaks)):
            k = int(peaks[idx_k])
            w_k = pv_weights[idx_k]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                sg[j] += w_k * f[k - j]

        sg_max = sg.max()
        if sg_max > 1e-30:
            sg = sg / sg_max

        v = 0.92 * v - lr * sg
        w = w + v
        w = w - w.max()

        if M < best_obj - 1e-16:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr = max(lr * decay, 0.001)

        if no_improve > 300:
            rng = np.random.default_rng(it * 13 + 1)
            f_pert = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            w = np.log(f_pert + 1e-30)
            v = np.zeros(n)
            lr = 0.03
            no_improve = 0

    return best_f, best_obj


def coordinate_descent_fast(f0, time_limit, verbose=False):
    """Coordinate descent with multiplicative updates and renormalization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-15:
                continue

            best_j_obj = fast_convolve(f).max()
            best_j_val = old_val

            for mult in [0.6, 0.75, 0.9, 1.1, 1.25, 1.4]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                f_trial = f_trial / f_trial.sum()
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                f[j] = best_j_val
                f = f / f.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 3:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
        else:
            parent = None
    except Exception as e:
        print(f"Could not load parent: {e}")
        parent = None

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.10f}")

    print(f"\n=== Phase 1: Active LP-FW (n={len(parent)}) ===")

        print(f"  Budget: {t1:.0f}s")
        a_opt, _ = lp_fw_optimize(parent, time_limit=t1, verbose=True, top_k=35, use_active=True)

    for trial in range(30):

        f_pert = project_simplex(best_seq + noise)
        sc = float(evaluate_sequence(f_pert.tolist()))

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
