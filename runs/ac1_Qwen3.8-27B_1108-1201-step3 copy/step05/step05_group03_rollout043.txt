def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()

    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def coord_descent_polish(a, time_limit, verbose=False):
    """
    Precise coordinate descent with golden-section search.
    Optimizes one coordinate at a time while maintaining sum=1.
    """
    n = len(a)
    f = normalize(a.copy())
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    pass_count = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        
        if pass_count > 1:
            order = np.random.permutation(n)
        else:
            order = np.arange(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            
            # Search range for coordinate j
            lo = 0.0
            hi = 0.999
            
            def obj_v(v):
                if v < -1e-12 or v > 1.0:
                    return float('inf')
                v = max(0.0, v)
                if abs(1.0 - v) < 1e-12:
                    return float('inf')
                new_f = f.copy()
                new_f[j] = v
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                scale = (1.0 - v) / (1.0 - old_val) if abs(1.0 - old_val) > 1e-12 else 1.0
                new_f[mask] *= scale
                new_f = np.maximum(new_f, 0.0)
                s = new_f.sum()
                if s < 1e-12:
                    return float('inf')
                new_f /= s
                conv = fast_convolve(new_f)
                return conv.max()
            
            try:
                res = minimize_scalar(obj_v, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-10, 'maxiter': 40})
                if res.success and res.fun < obj_v(old_val) - 1e-14:
                    v_new = max(0.0, res.x)
                    new_f = f.copy()
                    new_f[j] = v_new
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    scale = (1.0 - v_new) / (1.0 - old_val) if abs(1.0 - old_val) > 1e-12 else 1.0
                    new_f[mask] *= scale
                    new_f = np.maximum(new_f, 0.0)
                    new_f = normalize(new_f)
                    sc = compute_score(new_f)
                    if sc < best_score - 1e-14:
                        f = new_f
                        best_score = sc
                        best_f = f.copy()
                        improved = True
            except Exception:
                pass
            
            if pass_count >= 5 and j % 100 == 0:
                if time.monotonic() - t0 > time_limit:
                    break
        
        if verbose and pass_count % 3 == 0:
            print(f"    CD pass {pass_count}: score={best_score:.10f}")
        
        if not improved and pass_count > 2:

    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and trial < n_trials:
        trial += 1
        
        # Choose positions to exchange mass between
        # Bias towards positions that contribute to peak convolution values
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        a_i = f[i]
        a_j = f[j]
        if a_i < 1e-10 and a_j < 1e-10:
            continue
        
        # Try several transfer amounts
        best_delta = 0.0
        best_local_score = best_score

            # Transfer from j to i
            delta = a_j * frac
            new_f = f.copy()
            new_f[j] = a_j - delta
            new_f[i] = a_i + delta
            if new_f[j] < -1e-12:

            new_f = np.maximum(new_f, 0.0)
            s = new_f.sum()
            if s < 1e-12:

            if sc < best_local_score - 1e-13:
                best_local_score = sc
                best_delta = delta
        
        if best_delta != 0.0:
            f[j] -= best_delta
            f[i] += best_delta
            f = np.maximum(f, 0.0)
            f = normalize(f)
            sc = compute_score(f)
            if sc < best_score:
                best_score = sc
                best_f = f.copy()

def raised_cosine(n):
    """Raised cosine window, normalized."""
    x = np.linspace(0, np.pi, n)
    a = 0.5 * (1 + np.cos(x))
    return normalize(a)


def gaussian_seq(n, sigma_frac=0.35):
    """Gaussian-shaped sequence."""
    x = np.arange(n) - n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-x ** 2 / (2 * sigma ** 2))
    return normalize(a)


def geometric_seq(n, r):
    """Geometric sequence."""
    a = (1.0 - r) * r ** np.arange(n)
    return normalize(a)


def twosided_geometric(n, r, center=None):
    """Two-sided geometric (exponential decay from center)."""
    if center is None:
        center = n // 2
    indices = np.arange(n)
    a = r ** np.abs(indices - center)
    return normalize(a)


def triangular_seq(n):
    """Triangular (tent) sequence."""
    x = np.linspace(0, 1, n)
    a = 1.0 - np.abs(2.0 * x - 1.0)
    return normalize(a)


def hamming_window(n):

    a = 0.42 - 0.5 * np.cos(2 * np.pi * x / (n - 1)) + \
        0.08 * np.cos(4 * np.pi * x / (n - 1))
    a = np.maximum(a, 0.0)
    return normalize(a)


def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining LSE-smoothed mirror descent,
    coordinate descent, and pair exchange search.
    Explores multiple sequence lengths and structured initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_a = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_a, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score
    
    # === Phase 0: Try parent solution ===
    print("=== Phase 0: Parent solution ===")
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            p_score = try_seq(parent, "parent")
            print(f"  Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass
    
    # === Phase 1: Generate and evaluate diverse starting points ===
    print("\n=== Phase 1: Diverse initializations ===")
    starts = []
    
    # Various structured sequences at different lengths
    lengths = [512, 768, 1024, 1280, 1536, 2048]
    
    for n in lengths:
        starts.append((raised_cosine(n), f"rc_n{n}"))
        starts.append((gaussian_seq(n, 0.3), f"gauss30_n{n}"))
        starts.append((gaussian_seq(n, 0.4), f"gauss40_n{n}"))
        starts.append((triangular_seq(n), f"tri_n{n}"))
        starts.append((hamming_window(n), f"hamm_n{n}"))
        starts.append((blackman_window(n), f"black_n{n}"))
        starts.append((twosided_geometric(n, 0.999), f"twslice_n{n}"))
    
    # Geometric at various ratios
    for r in [0.998, 0.999, 0.9995]:
        for n in [768, 1024]:
            starts.append((geometric_seq(n, r), f"geom_r{r}_n{n}"))
    
    # Evaluate all starts
    print(f"  Evaluating {len(starts)} starting points...")
    start_scores = []
    for a, label in starts:
        sc = compute_score(a)
        start_scores.append((sc, a, label))
    start_scores.sort(key=lambda x: x[0])
    
    for i, (sc, a, label) in enumerate(start_scores[:8]):
        print(f"    Start#{i+1} [{label}]: {sc:.8f}")
        try_seq(a, label)
    
    # === Phase 2: Optimize top diverse starting points with mirror descent ===
    print(f"\n=== Phase 2: Mirror descent optimization ===")
    
    # Select diverse top starts
    selected = []
    selected_types = set()
    for sc, a, label in start_scores:
        if len(selected) >= 8:
            break
        # Extract type (before the last _nXXX)
        parts = label.rsplit('_n', 1)
        base = parts[0] if len(parts) > 1 else label
        if base not in selected_types:
            selected.append((a, label, sc))
            selected_types.add(base)
    
    # Include parent if good
    if best_a is not None and best_score < start_scores[0][0] * 1.1:
        if not any(label == "parent" for _, label, _ in selected):
            selected.append((best_a.copy(), "parent", best_score))
    
    n_selected = len(selected)
    # Time per optimization
    time_for_md = min((deadline - time.monotonic() - 300) / max(1, n_selected), 180)
    
    for idx, (a_init, label, init_sc) in enumerate(selected):
        rem = deadline - time.monotonic() - 200
        if rem < 30:
            break
        t_alloc = min(time_for_md, rem)
        
        print(f"  MD [{label}] t={t_alloc:.0f}s (init_score={init_sc:.6f})")
        
        # Mirror descent with LSE
        a_md, sc_md = mirror_descent_lse(
            a_init, time_limit=t_alloc * 0.6,
            tau0=1.0 / (len(a_init) * 0.5),
            tau_decay=0.9992,
            lr0=0.015,
            momentum=0.85,
            verbose=False
        )
        try_seq(a_md, label + "_md")
        
        # Polish with coordinate descent
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_cd = min(t_alloc * 0.3, rem)
            a_cd, sc_cd = coord_descent_polish(a_md, time_limit=t_cd)
            try_seq(a_cd, label + "_cd")
            
            # Pair exchange
            rem = deadline - time.monotonic() - 100
            if rem > 20:
                t_pe = min(t_alloc * 0.2, rem)
                a_pe, sc_pe = pair_exchange(a_cd, time_limit=t_pe, n_trials=3000)
                try_seq(a_pe, label + "_pe")
    
    # === Phase 3: Best sequence refinement with longer runs ===
    print(f"\n=== Phase 3: Deep refinement (best={best_score:.8f}) ===")
    
    if best_a is not None:
        # Extended mirror descent on best
        rem = deadline - time.monotonic() - 150
        if rem > 60:
            t_md2 = min(rem * 0.4, 300)
            print(f"  Extended MD: t={t_md2:.0f}s")
            a_ext, sc_ext = mirror_descent_lse(
                best_a, time_limit=t_md2,
                tau0=0.005,
                tau_decay=0.9998,
                lr0=0.005,
                momentum=0.9,
                verbose=True
            )
            try_seq(a_ext, "ext_md")
        
        # Extended coordinate descent
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_cd2 = min(rem * 0.4, 300)
            print(f"  Extended CD: t={t_cd2:.0f}s")
            a_cd2, sc_cd2 = coord_descent_polish(best_a, time_limit=t_cd2, verbose=True)
            try_seq(a_cd2, "ext_cd")
        
        # Pair exchange
        rem = deadline - time.monotonic() - 60
        if rem > 30:
            t_pe2 = min(rem * 0.4, 200)
            print(f"  Extended PE: t={t_pe2:.0f}s")
            a_pe2, sc_pe2 = pair_exchange(best_a, time_limit=t_pe2, n_trials=10000)
            try_seq(a_pe2, "ext_pe")
        
        # Final CD polish
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_final = min(rem, 120)
            print(f"  Final CD: t={t_final:.0f}s")
            a_fin, sc_fin = coord_descent_polish(best_a, time_limit=t_final)
            try_seq(a_fin, "final_cd")
    
    # === Phase 4: Dimension exploration ===
    print(f"\n=== Phase 4: Dimension exploration (best={best_score:.8f}) ===")
    
    if best_a is not None:
        best_n = len(best_a)
        target_lengths = [512, 640, 768, 896, 1024, 1280, 1536, 2048]
        
        for target_n in target_lengths:
            if target_n == best_n:
                continue
            rem = deadline - time.monotonic() - 60
            if rem < 40:
                break
            t_alloc = min(rem * 0.3, 100)
            
            a_interp = interpolate_to_length(best_a, target_n)
            a_opt, sc = mirror_descent_lse(
                a_interp, time_limit=t_alloc * 0.5,
                tau0=0.01, tau_decay=0.9995, lr0=0.01, momentum=0.85
            )
            if rem > t_alloc:
                a_opt, sc = coord_descent_polish(a_opt, time_limit=min(t_alloc * 0.5, rem - t_alloc))
            
            try_seq(a_opt, f"dim_{target_n}")
    
    # === Phase 5: Perturbation restarts ===
    print(f"\n=== Phase 5: Perturbation restarts (best={best_score:.8f}) ===")
    
    if best_a is not None:
        n_best = len(best_a)
        for trial in range(5):
            rem = deadline - time.monotonic() - 30
            if rem < 30:
                break
            ttrial = min(rem * 0.3, 80)
            
            noise_scale = [0.005, 0.01, 0.02, 0.01, 0.005][trial]
            noise = rng.standard_normal(n_best) * noise_scale
            a_pert = project_simplex(best_a + noise)
            
            a_opt, sc = mirror_descent_lse(
                a_pert, time_limit=ttrial * 0.5,
                tau0=0.01, tau_decay=0.999, lr0=0.01, momentum=0.85
            )
            rem = deadline - time.monotonic() - 20
            if rem > 15:
                a_opt, sc = coord_descent_polish(a_opt, time_limit=min(ttrial * 0.3, rem))
            
            try_seq(a_opt, f"perturb{trial}")
    
    # === Final verification ===
    if best_a is None:
        best_a = raised_cosine(768)
        best_score = compute_score(best_a)
    
    # Verify with actual evaluate_sequence
    final_seq = [float(x) for x in best_a]
    final_score = float(evaluate_sequence(final_seq))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_a)})")
    return final_seq
