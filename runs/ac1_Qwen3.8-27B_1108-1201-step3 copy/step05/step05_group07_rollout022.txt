import numpy as np

def fast_convolve(a):

    n = len(a)

def soft_max_obj_grad(z, T, n):
    """Compute the soft-max smoothed objective and its gradient w.r.t. z."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Soft-max L-BFGS with temperature annealing (smooth approximation phase)
2. Exact 1D golden-section coordinate descent on the true max-convolution objective
3. Simulated annealing for diversification and escaping local minima
4. Multi-length exploration with interpolation
Key innovation: exact coordinate descent that avoids FFT during 1D search,
giving O(n) per evaluation instead of O(n log n).
"""

import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def soft_max_obj_grad(z, T, n):
    """Compute the soft-max smoothed objective and its gradient w.r.t. z."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()

    b = fast_convolve(a)

    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp

    w = exp_Tb / exp_Tb.sum()

    # c_j = sum_k w_k * a_{k-j} via FFT cross-correlation
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A_fft = rfft(a, fft_size)
    c_full = irfft(np.conj(W) * A_fft, fft_size)
    c = c_full[:n]

    wb = np.dot(w, b)
    grad_z = 4.0 * n * a * (c - wb)

    return obj, grad_z


def lbgfs_optimize(z0, T, n, time_limit, maxiter=5000):
    """L-BFGS optimization in log-space with time limit."""
    t0 = time.monotonic()

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return soft_max_obj_grad(z, T, n)[0]

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return soft_max_obj_grad(z, T, n)[1]

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-12,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        return res.x
    except Exception:
        return z0


def exact_coordinate_descent(a, time_limit, n_sweeps=500, n_golden=50, seed=42):
    """
    Exact coordinate descent on the true max-convolution objective.
    For each coordinate i, finds optimal x via golden section search.
    Uses O(n) per evaluation by precomputing C, D, E vectors.
    """
    rng = np.random.default_rng(seed)
    a = a.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)

    b = fast_convolve(a)
    best_obj = b.max()
    best_a = a.copy()
    t0 = time.monotonic()

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    k_range = np.arange(2 * n - 1)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        order = rng.permutation(n)
        improved_count = 0

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            a_i_old = a[i]
            if a_i_old < 1e-18:
                # Try a few small values
                for x in [1e-4, 5e-4, 1e-3]:
                    denom = 1.0 - a_i_old
                    if denom < 1e-15:
                        continue
                    s_val = (1.0 - x) / denom
                    if s_val <= 0:
                        continue
                    a_trial = a.copy()
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    a_trial[mask] *= s_val
                    a_trial[i] = x
                    a_trial /= a_trial.sum()
                    b_trial = fast_convolve(a_trial)
                    if b_trial.max() < best_obj - 1e-14:
                        a = a_trial
                        b = b_trial
                        best_obj = b.max()
                        best_a = a.copy()
                        improved_count += 1
                        break
                continue

            # Precompute C_k, D_k, E_k for coordinate i
            ki = k_range - i
            valid_cross = (ki >= 0) & (ki < n) & (ki != i)

            D = np.zeros(2 * n - 1)
            D[valid_cross] = 2.0 * a[ki[valid_cross]]

            E = np.zeros(2 * n - 1)
            if 2 * i < 2 * n - 1:
                E[2 * i] = 1.0

            # contrib_k = terms in b_k that involve a[i]
            contrib = np.zeros(2 * n - 1)
            contrib[valid_cross] = 2.0 * a_i_old * a[ki[valid_cross]]
            if 2 * i < 2 * n - 1:
                contrib[2 * i] += a_i_old * a_i_old

            C = b - contrib
            C = np.maximum(C, 0.0)  # numerical safety

            denom = 1.0 - a_i_old
            if denom < 1e-15:
                continue

            # Golden section search for x in [lo, hi]
            lo = 0.0
            # Upper bound: don't let x exceed 1, and be reasonable
            hi = min(1.0, max(2.0 * a_i_old, 0.5))
            if a_i_old > 0.5:
                hi = 1.0

            x1 = hi - gr * (hi - lo)
            x2 = lo + gr * (hi - lo)

            def eval_at(x):
                s_val = (1.0 - x) / denom
                if s_val < 0:
                    return 1e10
                b_new = s_val * s_val * C + s_val * D * x + E * x * x
                return b_new.max()

            f1 = eval_at(x1)
            f2 = eval_at(x2)

            for _ in range(n_golden):
                if f1 < f2:
                    hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = hi - gr * (hi - lo)
                    f1 = eval_at(x1)
                else:
                    lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = lo + gr * (hi - lo)
                    f2 = eval_at(x2)

            x_opt = (lo + hi) / 2.0
            f_opt = eval_at(x_opt)

            if f_opt < best_obj - 1e-13:
                s_new = (1.0 - x_opt) / denom
                a_new = a.copy()
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                a_new[mask] *= s_new
                a_new[i] = x_opt
                a_new /= a_new.sum()
                a = a_new
                b = fast_convolve(a)
                best_obj = b.max()
                best_a = a.copy()
                improved_count += 1

        if improved_count == 0:
            break

    return best_a, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_sa = min(rem * 0.5, 120.0)
            print(f"  SA: t={t_sa:.0f}s, n={len(best_seq)}")
            a_sa, _ = simulated_annealing(best_seq, t_sa, seed=seed + 99)
            try_seq(a_sa, "sa")

            # CD after SA
            rem = deadline - time.monotonic() - 100
            if rem > 20:
                t_cd_post = min(rem * 0.5, 60.0)
                a_cd_post, _ = exact_coordinate_descent(a_sa, t_cd_post, n_sweeps=300,
                                                        n_golden=40, seed=seed + 200)
                try_seq(a_cd_post, "cd_post_sa")

    # === Phase 5: Try diverse initializations at best length ===
    print(f"\n=== Phase 5: Diverse initializations (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            inits = make_initializations(n_best, rng)
            per_init = rem / (len(inits) + 1)

            for init_idx, init in enumerate(inits):
                rem = deadline - time.monotonic() - 50
                if rem < 10:
                    break
                # Blend with best
                blend_a = 0.5 * best_seq + 0.5 * init
                blend_a = np.maximum(blend_a, 0)
                blend_a /= blend_a.sum()

                t_alloc = max(5.0, per_init * 0.7)
                a_blend_cd, _ = exact_coordinate_descent(blend_a, t_alloc,
                                                         n_sweeps=200, n_golden=30,
                                                         seed=seed + 1000 + init_idx)
                try_seq(a_blend_cd, f"blend_{init_idx}")

    # === Phase 6: Final deep coordinate descent ===
    print(f"\n=== Phase 6: Final deep CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_final = min(rem, 180.0)
            print(f"  Final CD: t={t_final:.0f}s, n={len(best_seq)}")
            a_final, _ = exact_coordinate_descent(best_seq, t_final, n_sweeps=2000,
                                                  n_golden=60, seed=seed + 500)
            try_seq(a_final, "final_cd")

            # One more L-BFGS polish
            rem = deadline - time.monotonic() - 10
            if rem > 10 and best_seq is not None:
                z_fin = np.log(np.maximum(best_seq, 1e-300))
                t_lf = min(rem, 30.0)
                z_fin = lbgfs_optimize(z_fin, 500.0, len(best_seq), t_lf, maxiter=2000)
                a_fin = np.exp(z_fin - z_fin.max())
                a_fin /= a_fin.sum()
                try_seq(a_fin, "final_lbgfs")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
