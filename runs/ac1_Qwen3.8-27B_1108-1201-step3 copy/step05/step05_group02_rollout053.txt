import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_Qf(p, f):
    """Compute Q*f where Q is Hankel with Q(i,j)=p(i+j), using direct dot products."""
    n = len(f)
    Qf = np.zeros(n)
    for i in range(n):
        Qf[i] = np.dot(p[i:i + n], f)
    return Qf


def compute_Qf_fft(p, f):
    """Compute Q*f via FFT where Q(i,j) = p(i+j)."""
    n = len(f)
    # Qf[i] = sum_j p(i+j) f(j) for j=0..n-1
    # This is the cross-correlation of p and f, restricted to lags 0..n-1
    # conv(p, f_rev)(i + n - 1) where f_rev = f[::-1]
    f_rev = f[::-1]
    conv = fast_convolve_padded(p, f_rev)
    # conv(m) = sum_j p(j) f_rev(m-j) = sum_j p(j) f(n-1-m+j)
    # We want Qf[i] = sum_j p(i+j) f(j)
    # Let k = i+j, so Qf[i] = sum_k p(k) f(k-i) for k=i..i+n-1
    # = cross_correlation(p, f)(i) in the "lower" sense
    # Using: Qf[i] = conv_full(p, f_flipped)(i + n - 1)
    # where f_flipped(j) = f(n-1-j)
    # conv_full(m) = sum_j p(j) f_flipped(m-j) = sum_j p(j) f(n-1-m+j)
    # At m = 2n-2-i: conv_full(2n-2-i) = sum_j p(j) f(n-1-2n+2+i+j) = sum_j p(j) f(i+1+j) -- not quite
    # Let me just use the direct method for clarity
    n = len(f)
    result = np.zeros(n)
    for i in range(n):
        result[i] = np.dot(p[i:i+n], f)
    return result


def fast_convolve_padded(a, b):
    """Full linear convolution of a and b via FFT."""
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def minimax_optimize(f0, time_limit, verbose=False):
    """
    Minimax alternating optimization:
    - Fix p (weights on convolution coefficients), solve QP for f
    - Update p based on new f
    The QP is: min f^T Q_p f + (mu/2)||f - f_prev||^2 s.t. f>=0, sum(f)=1
    Solved via projected gradient with backtracking.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    # Initialize p as uniform
    p = np.full(2 * n - 1, 1.0 / (2 * n - 1))

    t0 = time.monotonic()
    outer_it = 0
    no_improve = 0
    mu = 0.01  # proximal parameter

    while time.monotonic() - t0 < time_limit:
        outer_it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1:
            break

        # Update p based on current f (soft argmax)
        conv = fast_convolve(f)
        # Use temperature that decreases over iterations
        temp = max(0.005, 0.05 * (0.995 ** outer_it))
        p = np.exp(conv / temp)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.full(2 * n - 1, 1.0 / (2 * n - 1))
        else:
            p = p / p_sum

        # Solve QP: min f^T Q_p f + (mu/2)||f - f_old||^2 s.t. f>=0, sum=1
        f_old = f.copy()
        # Gradient of the QP at f: 2*Q_p*f + mu*(f - f_old)
        # We use projected gradient with backtracking line search
        lr = 1.0 / (2.0 * np.max(conv) + mu + 1e-10)
        inner_budget = max(20, int(rem * 2))  # max inner iterations

        for inner in range(inner_budget):
            Qf = np.array([np.dot(p[i:i + n], f) for i in range(n)])
            grad = 2.0 * Qf + mu * (f - f_old)
            f_new = project_simplex(f - lr * grad)
            if np.max(np.abs(f_new - f)) < 1e-12:
                f = f_new
                break
            f = f_new

        # Evaluate
        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            if verbose and outer_it % 50 == 0:
                print(f"    minimax iter {outer_it}: obj={best_obj:.12f}")
        else:
            no_improve += 1

        # Adaptive: reduce proximal weight over time
        mu *= 0.99

        # Restart if stuck
        if no_improve > 30:
            # Perturb f
            rng = np.random.default_rng(outer_it * 104729 + 7)
            noise = rng.standard_normal(n) * 0.003
            f = project_simplex(best_f + noise)
            no_improve = 0
            mu = 0.01

    return best_f, best_obj


def solve_lp_direction_topleaksed(f, M, n, top_k=None):
    """Solve LP: max sum(g) s.t. top-K (f*g)[k] <= M, g >= 0.
    Only constrain the top-k convolution coefficients to speed up."""
    if top_k is None:
        top_k = min(20, 2 * n - 1)
    # Compute f*g for all g to find which k are most restrictive
    # Actually, we need to constrain (f*g)(k) <= M for the k where f peaks
    # Use all constraints but limit to top-k by value of f convolution with ones
    ones = np.ones(n)
    fg_ones = fast_convolve(f * ones)  # = conv(f, ones) gives "width" profile
    # Actually, the binding constraints depend on g, so we use all 2n-1
    # But for speed, let's just use all
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {'time_limit': 30, 'presolve': True, 'dual_feasibility_tolerance': 1e-7}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    no_improve = 0
    it = 0
    m_relax = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break

        g = solve_lp_direction_topleaksed(f, M * m_relax, n)
        if g is None:
            m_relax = min(m_relax * 1.2, 10.0)
            if m_relax > 10.0:
                m_relax = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            m_relax = min(m_relax * 1.03, 6.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.02, 0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Bisection refinement
        if 0 < best_alpha < 1.0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(20):

                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-13:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_relax = 1.0
            f = f_new
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:

            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.05, 6.0)

        if no_improve > 60:

            scale = 0.002 + 0.003 * min(no_improve // 60, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)

            m_relax = 1.0
            no_improve = 0

    return best_f, best_obj


def bidirectional_cd(f0, time_limit, verbose=False):
    """Bidirectional coordinate descent: try moving each coordinate up and down."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    # Multipliers to try for each coordinate
    multipliers = [0.1, 0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 3.0]

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        cur_conv = fast_convolve(f)
        cur_max = cur_conv.max()

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj

            old_val = f[j]
            if old_val < 1e-15:
                continue

            denom = 1.0 - old_val
            if denom < 1e-10:
                continue

            best_j_obj = cur_max
            best_j_val = old_val
            best_j_conv = cur_conv.copy()

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                conv_trial = fast_convolve(f_trial)
                val = conv_trial.max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
                    best_j_conv = conv_trial

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                f[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = f / f.sum()
                cur_conv = best_j_conv
                cur_max = best_j_conv.max()
                improved = True
                if cur_max < best_obj - 1e-14:
                    best_obj = cur_max
                    best_f = f.copy()
                    if verbose and passes % 5 == 0:
                        print(f"    CD pass {passes}: obj={best_obj:.12f}")

        if not improved and passes > 3:
            break

    return best_f, best_obj


def nesterov_sg_polish(f0, time_limit, verbose=False):
    """Nesterov-accelerated subgradient method with adaptive temperature."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    y = f.copy()  # Nesterov's lookahead
    v = np.zeros(n)  # momentum
    best_f = f.copy()
    conv = fast_convolve(f)
    best_obj = conv.max()

    t0 = time.monotonic()
    lr = 0.01
    mom = 0.9
    temp = 0.01
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute subgradient at y
        conv_y = fast_convolve(y)
        # Use all coefficients above a threshold
        threshold = np.percentile(conv_y, 95)
        active = conv_y > threshold
        if not active.any():
            active[np.argmax(conv_y)] = True

        # Weights proportional to excess over threshold
        excess = np.maximum(conv_y - threshold, 0)
        w = excess / max(excess.sum(), 1e-30)

        # Subgradient: sg(j) = 2 * sum_k w(k) * f(k-j)
        sg = np.zeros(n)
        active_idx = np.where(active)[0]
        for k in active_idx:
            wk = w[k]
            if wk < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * wk * y[k - idx]

        # Nesterov step
        t_k = 2.0 / (it + 1.5)
        f_new = project_simplex(y - lr * sg)
        v = mom * v + (f_new - y)
        y = project_simplex(f_new + t_k * v)
        f = f_new

        # Evaluate
        conv = fast_convolve(f)
        obj = conv.max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive parameters
        lr *= 0.999
        temp *= 1.002

        if no_improve > 200:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            y = f.copy()
            v = np.zeros(n)
            lr = 0.005
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def gaussian_smooth(f, sigma):
    """Apply Gaussian smoothing to the sequence."""
    n = len(f)
    kernel_size = int(6 * sigma) | 1  # odd
    if kernel_size >= n:
        kernel_size = n | 1
    x = np.arange(kernel_size) - kernel_size // 2
    kernel = np.exp(-0.5 * (x / sigma) ** 2)
    kernel /= kernel.sum()
    f_smooth = np.convolve(f, kernel, mode='same')
    f_smooth = np.maximum(f_smooth, 0)
    ss = f_smooth.sum()
    if ss > 1e-15:
        f_smooth /= ss
    return f_smooth


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    methods = []

    # Uniform
    methods.append(np.ones(n) / n)

    # Triangular
    tri = np.abs(np.linspace(-1, 1, n))
    methods.append(tri / tri.sum())

    # Gaussian-like
    x = np.linspace(-3, 3, n)
    g = np.exp(-x ** 2)
    methods.append(g / g.sum())

    # Bimodal
    x = np.linspace(0, 1, n)
    b = np.exp(-((x - 0.25) / 0.1) ** 2) + np.exp(-((x - 0.75) / 0.1) ** 2)
    methods.append(b / b.sum())

    # Random smooth
    rough = rng.standard_normal(n)
    f = gaussian_smooth(np.abs(rough), sigma=max(3, n // 30))
    methods.append(f)

    # Periodic with noise
    period = max(3, n // 8)
    freq = 2 * np.pi / period
    base = 0.5 + 0.3 * np.sin(freq * np.arange(n))
    base = base + rng.standard_normal(n) * 0.05
    base = np.maximum(base, 0)
    methods.append(base / base.sum())

    return methods

    rem = total_deadline - time.monotonic()
    t5 = min(100, rem * 0.10)
    if t5 > 10:
        print(f"  Budget: {t5:.0f}s")
        a_opt, obj = nesterov_sg_polish(best_seq, time_limit=t5, verbose=True)
        update_best(a_opt, "nesterov_sg")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    candidate_lengths = []
    for delta in [-64, -32, 32, 64, 128, 192, 256, -128]:
        nl = n_best + delta
        if 128 <= nl <= 2560 and nl != n_best:
            candidate_lengths.append(nl)

    for target_n in candidate_lengths[:5]:
        rem = total_deadline - time.monotonic() - 60
        if rem < 20:
            break
        f_interp = interpolate_sequence(best_seq, target_n)
        t_alloc = min(80, rem * 0.15)
        if t_alloc < 10:
            continue
        print(f"  n={target_n}, t={t_alloc:.0f}s")
        # Quick minimax
        a_opt, _ = minimax_optimize(f_interp, time_limit=t_alloc * 0.5)
        update_best(a_opt, f"ml_minimax_{target_n}")
        # Quick LP-FW
        rem2 = total_deadline - time.monotonic() - 50
        if rem2 > 15:
            a_opt2, _ = lp_fw_optimize(a_opt, time_limit=min(40, rem2 * 0.4))
            update_best(a_opt2, f"ml_lpfw_{target_n}")

    # === Phase 7: Diverse initializations ===
    print(f"\n=== Phase 7: Diverse inits (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    inits = structured_init(n_best, rng)

    # Add perturbed versions of best
    inits.append(project_simplex(best_seq + rng.standard_normal(n_best) * 0.005))
    inits.append(project_simplex(0.7 * best_seq + 0.3 * np.ones(n_best) / n_best))
    inits.append(gaussian_smooth(best_seq, sigma=max(2, n_best // 20)))

    for idx, f_init in enumerate(inits):
        rem = total_deadline - time.monotonic() - 40
        if rem < 15:
            break
        t_alloc = min(60, rem * 0.12)
        if t_alloc < 8:
            continue
        print(f"  init {idx}, t={t_alloc:.0f}s")
        a_opt, _ = minimax_optimize(f_init, time_limit=t_alloc)
        update_best(a_opt, f"diverse_{idx}")

    # === Phase 8: Final combined polish ===
    print(f"\n=== Phase 8: Final polish (best={best_score:.8f}) ===")
    rem = total_deadline - time.monotonic()
    if rem > 30:
        # LP-FW
        t_a = min(rem * 0.35, 100)
        if t_a > 10:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_a)
            update_best(a_opt, "final_lpfw")

        # Minimax
        rem = total_deadline - time.monotonic()
        t_b = min(rem * 0.3, 60)
        if t_b > 8:
            a_opt, _ = minimax_optimize(best_seq, time_limit=t_b)
            update_best(a_opt, "final_minimax")

        # CD
        rem = total_deadline - time.monotonic()
        t_c = min(rem * 0.2, 40)
        if t_c > 5:
            a_opt, _ = bidirectional_cd(best_seq, time_limit=t_c)
            update_best(a_opt, "final_cd")

        # Micro perturbations
        rem = total_deadline - time.monotonic()
        for trial in range(12):
            if rem < 3:
                break
            scale = 0.0002 + trial * 0.00015
            noise = rng.standard_normal(len(best_seq)) * scale
            f_pert = project_simplex(best_seq + noise)
            update_best(f_pert, f"perturb_{trial}")
            # Quick SG polish on perturbation
            rem2 = total_deadline - time.monotonic()
            if rem2 > 5:
                a_opt, _ = nesterov_sg_polish(f_pert, time_limit=min(rem2 * 0.3, 8))
                update_best(a_opt, f"perturb_{trial}_sg")

    # === Final verification ===
    if best_seq is None:
        best_seq = np.ones(384) / 384

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
