def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def power_sum_optimize(f0, time_limit, p_start=100.0, p_end=5.0, verbose=False):
    """Optimize using power-sum smoothing with Adam."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_conv = fast_convolve(f)
    best_obj = best_conv.max()
    
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t_adam = 0
    
    alpha = 0.005
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / time_limit, 1.0)
        p = p_start * (p_end / p_start) ** progress
        p = max(p, p_end)
        lr = alpha * (1.0 + 2.0 * (1.0 - progress)) * (0.3 ** progress)
        
        grad, S_val, conv = power_sum_grad(f, p)
        
        t_adam += 1
        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad ** 2
        m_hat = m_state / (1 - beta1 ** t_adam)
        v_hat = v_state / (1 - beta2 ** t_adam)
        
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)
        
        conv_new = fast_convolve(f_new)
        cur_obj = conv_new.max()
        
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    PS iter {it}: obj={best_obj:.10f}, p={p:.1f}, lr={lr:.6f}")
        else:
            no_improve += 1
        
        f = f_new
        
        if no_improve > 500:

            v_state = np.zeros(n)
            t_adam = 0
            no_improve = 0
