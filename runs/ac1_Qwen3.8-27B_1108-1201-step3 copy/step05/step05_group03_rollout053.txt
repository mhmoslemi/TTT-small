def solve_lp(f, M):
    n = len(f)
    # Variables: g_0, ..., g_{n-1}
    # Objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    
    # Inequality constraints: (f*g)[k] <= M for k = 0, ..., 2n-2
    # A_ub @ g <= b_ub
    num_constraints = 2 * n - 1
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.full(num_constraints, M)

    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

LP-based iterative method for minimizing 2*n*max(auto_conv)/(sum)^2.
Uses the cutting-plane / active-set approach: at each iteration,
solve an LP to find the best direction, then perform a line search.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.optimize import minimize_scalar
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def solve_direction_lp(f, M):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k."""
    n = len(f)
    c = -np.ones(n)  # minimize -sum(g)
    
    # Build constraint matrix
    num_k = 2 * n - 1
    A_ub = np.zeros((num_k, n))
    b_ub = np.full(num_k, M)
    
    for k in range(num_k):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            A_ub[k, i] = f[k - i]
    
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success:

    return None

LP-based iterative optimization for minimizing 2*n*max(auto_conv)/(sum)^2.
Combines cutting-plane LP direction finding, line search, multi-length
exploration, and coordinate descent polishing.

import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize_nonneg(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_lp_matrix(f, M):
    """Build LP constraint matrix for the direction-finding problem."""
    n = len(f)
    num_k = 2 * n - 1
    A_ub = np.zeros((num_k, n))
    for k in range(num_k):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = f[k - j]
    b_ub = np.full(num_k, M)
    return A_ub, b_ub


def solve_direction_lp(f, M, time_limit=20.0):
    n = len(f)
    A_ub, b_ub = build_lp_matrix(f, M)
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'time_limit': time_limit})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def line_search_obj(t, f, g):
    h = (1.0 - t) * f + t * g
    return fast_convolve(h).max()


def do_line_search(f, g):
    res = minimize_scalar(line_search_obj, args=(f, g),
                          bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-7, 'maxiter': 50})
    return res.x


def geometric_seq(n, r):
    a = r ** np.arange(n, dtype=np.float64)

def cosine_seq(n):

    sigma = sigma_frac * n
    return normalize_nonneg(np.exp(-x**2 / (2.0 * sigma**2)))


def triangular_seq(n):
    x = np.linspace(0.0, 1.0, n)
    return normalize_nonneg(1.0 - np.abs(2.0 * x - 1.0))


def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    return normalize_nonneg(np.interp(new_x, old_x, a))
