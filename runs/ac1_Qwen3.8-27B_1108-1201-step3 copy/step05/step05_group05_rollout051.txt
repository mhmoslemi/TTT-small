import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax(x):
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()

def compute_objective_and_grad(a, tau, alpha=0.0, top_k=None):
    """
    Compute the smooth objective and its gradient wrt a.
    Objective: (1/tau)*log(sum(exp(tau*log(c_k)))) + alpha * var(c)

    log_c = np.log(c_safe)

    scaled = tau * log_c
    scaled -= scaled.max()
