"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding
(based on the SOTA approach), combined with Nesterov-accelerated subgradient,
coordinate descent, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build the (2n-1) x n matrix A where A@g = f*g (linear convolution)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            i = k - j
            if 0 <= i < n:
                rows.append(k)
                cols.append(j)
                vals.append(f[i])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_direction(f, M):
    """
    Solve LP: maximize sum(g) subject to g >= 0, (f*g)[k] <= M for all k.
    Returns g (unnormalized) and the optimal sum.
    """
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    else:
        return None


def lp_optimize(f0, time_limit, verbose=False):
    """
    LP-based optimization following the SOTA approach.
    Iteratively: solve LP for best direction, update f, repeat.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    iter_count = 0
    no_improve = 0
    t_blend = 0.15
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break
        
        M = fast_convolve(f).max()
        
        # Solve LP
        g = solve_lp_direction(f, M)
        if g is None:
            # Fallback: use uniform
            g = np.ones(n) / n
        else:
            g = np.maximum(g, 0)
            g_sum = g.sum()
            if g_sum < 1e-10:
                g = np.ones(n) / n
            else:
                g = g / g_sum
        
        # Try several blend factors
        best_t = t_blend
        best_blend_obj = best_obj
        
        for t_try in [t_blend * 0.5, t_blend, t_blend * 1.5, t_blend * 2.0]:
            f_try = (1 - t_try) * f + t_try * g
            f_try = np.maximum(f_try, 0)
            s_try = f_try.sum()
            if s_try < 1e-15:
                continue
            f_try = f_try / s_try
            obj_try = fast_convolve(f_try).max()
            if obj_try < best_blend_obj:
                best_blend_obj = obj_try
                best_t = t_try
        
        f = (1 - best_t) * f + best_t * g
        f = np.maximum(f, 0)
        s = f.sum()
        if s < 1e-15:
            f = best_f.copy()
            break
        f = f / s
        
        obj = fast_convolve(f).max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            # If stuck, perturb

                rng = np.random.default_rng(iter_count * 1009 + 7)
                noise = rng.standard_normal(n) * 0.003
                f = project_simplex(best_f + noise, target_sum=1.0)
                no_improve = 0
                t_blend = 0.1
            elif no_improve > 10:
                t_blend = max(0.03, t_blend * 0.7)
        
        # Adapt blend factor

            t_blend = min(0.3, t_blend * 1.05)
        
        if verbose and iter_count % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iter_count}: obj={best_obj:.10f}, t={best_t:.3f}, no_imp={no_improve} ({elapsed:.0f}s)")
    
    return best_f, best_obj


def subgradient_optimize(f0, time_limit, verbose=False):
    """Nesterov-accelerated subgradient method."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    # Nesterov momentum
    y = f.copy()
    v = np.zeros(n)
    beta = 0.9
    tau = 50.0
    lr0 = 0.05
    t0 = time.monotonic()
    no_improve = 0
    k = 0
    
    while time.monotonic() - t0 < time_limit:
        k += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1:
            break
        
        # Compute subgradient at y
        conv_y = fast_convolve(y)
        M_y = conv_y.max()
        # Top peaks
        top_k = min(200, 2 * n - 1)
        peaks = np.argpartition(conv_y, -top_k)[-top_k:]
        pv = conv_y[peaks]
        scaled = tau * (pv - pv.max())
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-15:
                continue
            kk = int(peaks[i])
            js = max(0, kk - n + 1)
            je = min(n - 1, kk)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w[i] * y[kk - idx]
        
        grad_centered = grad - grad.mean()
        
        # Nesterov step
        lr = lr0 / np.sqrt(k + 1)
        v = beta * v - lr * grad_centered
        x_new = project_simplex(y + v, target_sum=1.0)
        
        t_nest = (1 + beta) / (k + 1 + beta)  # Nesterov parameter
        y = x_new - t_nest * (x_new - y)
        y = project_simplex(y, target_sum=1.0)
        
        obj = fast_convolve(x_new).max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = x_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 300:
            rng = np.random.default_rng(k * 7 + 13)
            noise = rng.standard_normal(n) * 0.004
            y = project_simplex(best_f + noise, target_sum=1.0)
            x_new = y.copy()
            v = np.zeros(n)
            no_improve = 0
        
        if verbose and k % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG iter {k}: obj={best_obj:.10f}, lr={lr:.5f} ({elapsed:.0f}s)")
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    pass_count = 0
    
    # Pre-compute which indices are non-zero
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            
            multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 1.8]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f /= f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved and pass_count > 1:
            break
    
    return best_f, best_obj

def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        mag = 1.0
        A[k] = mag * np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    y = f.copy()

    lr0 = 0.06
    t0 = time.monotonic()
    no_improve = 0

        scaled = tau * (pv - pv.max())
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-15:

            kk = int(peaks[i])
            js = max(0, kk - n + 1)
            je = min(n - 1, kk)

        v = beta * v - lr * grad_centered
        x_new = project_simplex(y + v, target_sum=1.0)

        y = x_new - t_nest * (x_new - y)
        y = project_simplex(y, target_sum=1.0)

        if obj < best_obj - 1e-12:

def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    pass_count = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue
            best_j_val = old_val
            best_j_obj = best_obj
            multipliers = [0.4, 0.6, 0.75, 0.88, 1.12, 1.3, 1.5, 1.8]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f /= f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        if not improved and pass_count > 1:
            break
    return best_f, best_obj

            break
        improved = False
        order = rng.permutation(num_blocks)
        for bi in order:
            if time.monotonic() - t0 > time_limit:
                break
            start = int(bi * n / num_blocks)
            end = int((bi + 1) * n / num_blocks)
            if end <= start:
                end = start + 1
            block_idx = np.arange(start, end)
            block_sum = f[block_idx].sum()
            if block_sum < 1e-12:
                continue
            best_block = f[block_idx].copy()
            best_obj_b = best_obj
            for _ in range(15):
                pert = rng.standard_normal(len(block_idx)) * 0.15 * block_sum
                new_block = f[block_idx] + pert
                new_block = np.maximum(new_block, 0)
                nb_sum = new_block.sum()
                if nb_sum < 1e-12:
                    continue
                f_trial = f.copy()
                f_trial[block_idx] = new_block
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-12:
                    continue
                f_trial /= ss
                obj_trial = fast_convolve(f_trial).max()
                if obj_trial < best_obj_b:
                    best_obj_b = obj_trial
                    best_block = new_block.copy()
            if best_obj_b < best_obj - 1e-14:
                f[block_idx] = best_block
                f = np.maximum(f, 0)
                f /= f.sum()
                best_obj = best_obj_b
                best_f = f.copy()
                improved = True
        if not improved:
            break
    return best_f, best_obj


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        mag = 1.0
        A[k] = mag * np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: LP-based direction finding with adaptive bounds,
    golden-section line search, subgradient, block CD, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # === Phase 1: LP on parent ===
    print(f"\n=== Phase 1: LP on parent (best={best_score:.6f}) ===")
    if parent is not None:
        n_p = len(parent)
        rem = deadline - time.monotonic() - 500
        t_lp = min(rem * 0.4, 350)
        print(f"  LP n={n_p}, t={t_lp:.0f}s")
        a_opt, obj = lp_optimize(parent, time_limit=max(t_lp, 30), verbose=True)
        try_seq(a_opt, "lp_parent")

    # === Phase 2: LP on flat spectrum init ===
    print(f"\n=== Phase 2: LP on flat spec (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_b = len(best_seq)
        rem = deadline - time.monotonic() - 400
        if rem > 60:
            for init_id in range(3):
                rem = deadline - time.monotonic() - 350
                if rem < 30:
                    break
                fs_init = flat_spectrum_init(n_b, rng)
                t_alloc = min(rem * 0.25, 100)
                print(f"  LP flat_spec_{init_id} n={n_b}, t={t_alloc:.0f}s")
                a_opt, obj = lp_optimize(fs_init, time_limit=max(t_alloc, 20), verbose=False)
                try_seq(a_opt, f"lp_flat_{init_id}")

    # === Phase 3: Subgradient refinement ===
    print(f"\n=== Phase 3: Subgradient (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        t_sg = min(rem * 0.3, 150)
        if t_sg > 15:
            print(f"  SG n={len(best_seq)}, t={t_sg:.0f}s")
            a_opt, obj = subgradient_optimize(best_seq, time_limit=t_sg, verbose=False)
            try_seq(a_opt, "sg_best")

    # === Phase 4: Block coordinate descent ===
    print(f"\n=== Phase 4: Block CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        t_bc = min(rem * 0.3, 120)
        if t_bc > 15:
            bs = max(4, len(best_seq) // 64)
            print(f"  BCD n={len(best_seq)}, bs={bs}, t={t_bc:.0f}s")
            a_opt, obj = block_coordinate_optimize(best_seq, time_limit=t_bc, block_size=bs)
            try_seq(a_opt, "bcd")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 120
        t_cd = min(rem * 0.4, 100)
        if t_cd > 10:
            print(f"  CD t={t_cd:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")

    # === Phase 6: LP refinement with perturbations ===
    print(f"\n=== Phase 6: LP Refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_b = len(best_seq)
        for p in range(4):
            rem = deadline - time.monotonic() - 60
            if rem < 25:
                break
            t_alloc = min(rem / (4 - p + 0.5), 80)
            if t_alloc < 10:
                break
            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_b) * 0.003
                f_init = project_simplex(best_seq + noise, target_sum=1.0)
            elif p == 2:
                f_init = 0.7 * best_seq + 0.3 * best_seq[::-1]
                f_init = project_simplex(f_init, target_sum=1.0)
            else:
                noise = rng.standard_normal(n_b) * 0.008
                f_init = project_simplex(best_seq + noise, target_sum=1.0)
            print(f"  LP refine_{p} t={t_alloc:.0f}s")
            a_opt, obj = lp_optimize(f_init, time_limit=t_alloc, verbose=(p == 0))
            try_seq(a_opt, f"lp_refine_{p}")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-Length (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_b = len(best_seq)
        other_lengths = [512, 640, 896, 1024, 1280, 1536]
        other_lengths = [l for l in other_lengths if abs(l - n_b) > 50]
        other_lengths.sort(key=lambda x: abs(x - n_b))
        for n_try in other_lengths[:3]:
            rem = deadline - time.monotonic() - 50
            if rem < 35:
                break
            f_interp = interpolate_sequence(best_seq, n_try)
            t_alloc = min(rem * 0.3, 70)
            print(f"  LP n={n_try}, t={t_alloc:.0f}s")
            a_opt, obj = lp_optimize(f_interp, time_limit=max(t_alloc, 15), verbose=False)
            try_seq(a_opt, f"lp_n{n_try}")
            rem = deadline - time.monotonic() - 35
            if rem > 15 and best_score < 1.504:
                a_opt, obj = coordinate_descent(a_opt, time_limit=min(rem * 0.5, 25))
                try_seq(a_opt, f"cd_n{n_try}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_lp_f = min(rem * 0.4, 40)
            a_opt, obj = lp_optimize(best_seq, time_limit=t_lp_f, verbose=False)
            try_seq(a_opt, "final_lp")
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt, obj = coordinate_descent(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
