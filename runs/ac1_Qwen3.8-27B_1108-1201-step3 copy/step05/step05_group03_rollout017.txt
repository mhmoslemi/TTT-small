"""
Minimizes 2*n*max(auto_conv)/sum^2 using Adam optimization with softmax
parameterization, log-sum-exp objective with beta annealing, coordinate
descent polish, and LP-based refinement (Nemirovski-Yudin style).
Multi-start with diverse initializations and multi-length exploration.
"""

import numpy as np
from scipy.optimize import minimize_scalar
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_fn(theta):
    """Stable softmax."""
    theta = theta - theta.max()
    e = np.exp(theta)
    return e / e.sum()


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def adam_optimize(theta_init, n, time_limit, beta_start=5.0, beta_end=5000.0, lr=0.05):
    """Adam optimization with log-sum-exp objective and beta annealing."""
    t0 = time.monotonic()
    theta = theta_init.copy()
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    step = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    # Precompute FFT sizes
    conv_fft_size = 1 << (2 * n - 2).bit_length()  # >= 2n-1
    corr_fft_size = 1 << (3 * n - 3).bit_length()  # >= 3n-2
    
    best_score = float('inf')
    best_theta = theta.copy()
    best_f = None
    
    # Total estimated steps for beta schedule
    est_steps = 100000
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        
        # Beta schedule: exponential from beta_start to beta_end
        progress = min(step / est_steps, 1.0)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Compute f = softmax(theta)
        f = softmax_fn(theta)
        
        # Compute auto-convolution
        F_conv = np.fft.rfft(f, conv_fft_size)
        conv = np.fft.irfft(F_conv * F_conv, conv_fft_size)[:2*n-1]
        
        # Compute p = softmax(beta * conv)
        scaled = beta * conv
        scaled -= scaled.max()
        exp_scaled = np.exp(scaled)
        p = exp_scaled / exp_scaled.sum()
        
        # Compute gradient via cross-correlation
        # gr[i] = sum_m p[m+i] * f[m] for m=0..n-1
        P_fft = np.fft.rfft(p, corr_fft_size)
        F_fft = np.fft.rfft(f, corr_fft_size)
        cross_corr = np.fft.irfft(P_fft * np.conj(F_fft), corr_fft_size)[:n]
        
        # grad_theta[j] = f[j] * (cross_corr[j] - dot(cross_corr, f))
        dot_cf = np.dot(cross_corr, f)
        grad = f * (cross_corr - dot_cf)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        theta -= lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Periodically re-center theta (remove mean to avoid drift)
        if step % 1000 == 0:
            theta -= theta.mean()
        
        # Evaluate score every 100 steps
        if step % 100 == 0:
            f = softmax_fn(theta)
            score = 2.0 * n * conv.max() / 1.0  # sum=1 by construction
            # Actually recompute since f changed
            score = compute_score(f)
            if score < best_score:
                best_score = score
                best_theta = theta.copy()
                best_f = f.copy()
                
        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam step {step}: score={best_score:.8f}, beta={beta:.1f}, t={elapsed:.1f}s")
    
    if best_f is None:
        best_f = softmax_fn(best_theta)
        best_score = compute_score(best_f)
    
    return best_f, best_score
