def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr(w, a):
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:n]


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def soft_max_obj(z, T, n):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T) * log_sum_exp


def soft_max_grad(z, T, n):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def lbgfs_optimize(z0, T, n, time_limit, t0, maxiter=3000):
    elapsed = [0.0]

    def func(z):
        elapsed[0] = time.monotonic()
        if elapsed[0] - t0 > time_limit:
            return 1e15
        obj, _ = soft_max_obj_grad_combined(z, T, n)
        return obj

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = soft_max_obj_grad_combined(z, T, n)
        return g

    try:
        res = minimize(
            fun=func,
            x0=z0,
            jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 25, 'disp': False}
        )
        return res.x, res.fun
    except Exception:
        return z0, func(z0)

def subgradient_polish(a0, t0, time_limit, lr_init=0.005, momentum=0.9, top_k=30):
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    no_improve = 0
    cur_lr = lr_init
    cur_temp = 0.001
    restart_count = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        v = momentum * v - cur_lr * sg
        a = project_simplex(a + v)
        b = fast_convolve(a)
        co = b.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9995
        cur_temp *= 1.005

        if no_improve > 200:
            if restart_count < 10:
                rng = np.random.default_rng(int(time.monotonic() * 1e6) % 100000 + restart_count)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                v = np.zeros(n)
                cur_lr = max(lr_init * 0.5, 0.0005)
                no_improve = 0
                restart_count += 1
            else:
                break

    return best_a, best_obj

def peak_equalize(a0, t0, time_limit, n_iterations=500):
    """Iteratively equalize convolution peaks by reducing contributions."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    for _ in range(n_iterations):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Find the worst peak (excluding k=0 which is always large)
        b_search = b.copy()
        b_search[0] = -1
        b_search[-1] = -1
        k_peak = np.argmax(b_search)

        if b[k_peak] <= best_obj * 1.001:
            break

        # Find contributing pairs
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contributions = a[idx] * a[k_peak - idx]

        # Reduce the largest contributor
        max_local = np.argmax(contributions)
        i = idx[max_local]
        j = k_peak - i

        # Try reducing the larger of a[i], a[j]
        target = i if a[i] >= a[j] else j
        old_val = a[target]
        if old_val < 1e-12:
            continue

        # Reduce by a small amount
        for factor in [0.98, 0.95, 0.9, 0.85]:
            new_val = old_val * factor
            diff = old_val - new_val
            if diff < 1e-15:
                continue

            mask = np.ones(n, dtype=bool)
            mask[target] = False
            others = a[mask]
            others_sum = others.sum()
            if others_sum < 1e-15:
                continue

            a_trial = a.copy()
            a_trial[target] = new_val
            a_trial[mask] = others + diff * others / others_sum
            ss = a_trial.sum()
            if ss > 1e-15:
                a_trial /= ss

            val = fast_convolve(a_trial).max()
            if val < best_obj - 1e-14:
                a = a_trial
                best_obj = val
                best_a = a.copy()
                break

    return best_a, best_obj


def generate_structured_init(n, seed, parent=None):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)

    if parent is not None:
        # Interpolate parent to length n
        base = interpolate_sequence(parent, n)
    else:
        base = np.ones(n) / n

    # Choose a random strategy
    strategy = rng.integers(6)

    if strategy == 0:
        # Uniform + small noise
        a = base + rng.standard_normal(n) * 0.01
    elif strategy == 1:
        # Palindromic perturbation
        a = base.copy()
        for i in range(n // 2):
            delta = rng.standard_normal() * 0.02
            a[i] += delta
            a[n - 1 - i] += delta
    elif strategy == 2:
        # Tapered (raised cosine window * base)
        t = np.linspace(0, 1, n)
        window = 0.5 * (1 + np.cos(2 * np.pi * t))
        center_boost = 1.0 + 0.3 * np.exp(-4 * (t - 0.5) ** 2)
        a = base * window * center_boost
        noise = rng.standard_normal(n) * 0.01
        a = a + noise
    elif strategy == 3:
        # Two-peak structure
        a = base.copy()
        peak1 = rng.uniform(0.2, 0.4)
        peak2 = 1.0 - peak1
        width = max(2, n // 10)
        a = base + 0.1 * np.exp(-((np.arange(n) - peak1 * n) ** 2) / (2 * width ** 2))
        a = a + 0.1 * np.exp(-((np.arange(n) - peak2 * n) ** 2) / (2 * width ** 2))
    elif strategy == 4:
        # Frequency domain: modify power spectrum
        A = rfft(base, n)
        phase_mod = rng.standard_normal(len(A)) * 0.1
        magnitude_mod = 1.0 + rng.standard_normal(len(A)) * 0.05
        A_mod = A * np.exp(1j * phase_mod) * magnitude_mod
        a = irfft(A_mod, n)
    else:
        # Smooth random (low-pass filtered noise)
        noise = rng.standard_normal(n)
        # Simple low-pass: running average
        kernel_size = max(1, n // 50)
        kernel = np.ones(kernel_size) / kernel_size
        noise = np.convolve(noise, kernel, mode='same')
        a = base + 0.05 * noise / (np.std(noise) + 1e-15)

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

            rem2 = phase3_deadline - time.monotonic()
            if rem2 > 15:
                t0_2 = time.monotonic()
                t_alloc2 = min(rem2 * 0.5, 80.0)
                temps_ml2 = [100, 300, 600]
                a_opt2, _, _ = run_lbgfs_phase(a_opt, t0_2, t_alloc2, temps_ml2, maxiter_base=2000)
                try_seq(a_opt2, f"ml_n{target_n}_refine")

    print(f"  Phase 3 best: {best_score:.6f}")

    # === Phase 4: Subgradient + coordinate descent (20%) ===
    print(f"\n=== Phase 4: SG + CD refinement (budget ~{0.2*budget_s:.0f}s) ===")
    phase4_deadline = t_start + budget_s * 0.90

    if best_seq is not None:
        n_best = len(best_seq)

        # Subgradient
        t0 = time.monotonic()
        t_sg = min(phase4_deadline - t0, budget_s * 0.10)
        if t_sg > 10:
            a_sg, _ = subgradient_polish(best_seq, t0, t_sg,
                                         lr_init=0.003, momentum=0.92,
                                         top_k=max(20, n_best // 8))
            try_seq(a_sg, f"sg_n{n_best}")

        # Coordinate descent
        if best_seq is not None:
            t0 = time.monotonic()
            t_cd = min(phase4_deadline - t0, budget_s * 0.08)
            if t_cd > 10:
                a_cd, _ = coordinate_descent(best_seq, t0, t_cd)
                try_seq(a_cd, f"cd_n{n_best}")

        # Another subgradient pass on the best
        if best_seq is not None:
            t0 = time.monotonic()
            rem = phase4_deadline - t0
            if rem > 10:
                a_sg2, _ = subgradient_polish(best_seq, t0, rem,
                                              lr_init=0.001, momentum=0.94,
                                              top_k=max(30, n_best // 5))
                try_seq(a_sg2, f"sg2_n{n_best}")

    print(f"  Phase 4 best: {best_score:.6f}")

    # === Phase 5: Final polish (10%) ===
    print(f"\n=== Phase 5: Final polish (budget ~{0.1*budget_s:.0f}s) ===")

    if best_seq is not None:
        n_best = len(best_seq)

        # Peak equalization
        t0 = time.monotonic()
        t_pe = min(deadline - t0, budget_s * 0.05)
        if t_pe > 10:
            a_pe, _ = peak_equalize(best_seq, t0, t_pe, n_iterations=1000)
            try_seq(a_pe, f"peak_eq_n{n_best}")

        # Final L-BFGS with very high temperature
        if best_seq is not None:
            t0 = time.monotonic()
            rem = deadline - t0
            if rem > 10:
                t_final = min(rem * 0.5, 60.0)
                temps_final = [300, 800, 1500]
                a_final, _, _ = run_lbgfs_phase(best_seq, t0, t_final, temps_final, maxiter_base=3000)
                try_seq(a_final, f"final_T1500_n{n_best}")

        # Small perturbation restarts
        if best_seq is not None:
            t0 = time.monotonic()
            rem = deadline - t0
            for attempt in range(5):
                if time.monotonic() > deadline - 5:
                    break
                noise_scale = 0.001 + attempt * 0.001
                noise = rng.standard_normal(n_best) * noise_scale
                a_pert = project_simplex(best_seq + noise)
                t0_p = time.monotonic()
                t_alloc = min(deadline - t0_p - 3, 20.0)
                if t_alloc < 5:
                    break
                z_p = z_from_a(a_pert)
                z_p, _ = lbgfs_optimize(z_p, 500, n_best, t_alloc, t0_p, maxiter=1500)
                a_p = softmax_from_z(z_p)
                try_seq(a_p, f"perturb_{attempt}_s{noise_scale}")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        # This shouldn't happen but just in case
        best_seq = parent[:512] if len(parent) > 512 else parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)}, time={total_time:.1f}s)")
    return [float(x) for x in best_seq]
