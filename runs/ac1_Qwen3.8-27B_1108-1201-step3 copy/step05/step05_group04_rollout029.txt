            top_pos = np.argpartition(c, -k_eff)[-k_eff:]
        else:
            top_pos = np.arange(len(c))

        # Softmax weights over top positions
        max_c = c[top_pos].max()
        w = np.exp((c[top_pos] - max_c) / temp)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(top_pos))
            w[np.argmax(c[top_pos])] = 1.0
        else:
            w /= w_sum

        # Compute weighted gradient
        grad = np.zeros(n)
        for i, k in enumerate(top_pos):
            if w[i] > 1e-12:
                grad += w[i] * gradient_at_k(a, k)

        # Normalized gradient step
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            grad = grad / g_norm

        a_new = a - lr * grad
        a = project_simplex(a_new)

        temp *= temp_inc
        lr *= lr_dec
        lr = max(lr, lr_init * 0.01)

    return best_a, best_obj


def lp_refine(f, M, n, time_limit=30):
    """
    LP-based refinement: maximize sum(b) subject to b>=0, sum(b)=1,
    and (f*b)[k] <= M for all k.
    Returns the LP solution or None.
    """
    t0 = time.monotonic()
    
    # Build constraint matrix A: rows correspond to convolution lags
    # (f * b)[k] = sum_j f[j] * b[k-j] <= M
    # For b on simplex (sum b = 1, b >= 0)
    # Variables: b_0, ..., b_{n-1}
    
    # We need: sum_j f[j] * b[k-j] <= M for each k
    # This gives us a linear constraint in b
    
    # A_ub @ b <= b_ub where b_ub = M
    # (f*b)[k] = sum_{j: 0<=j<n, 0<=k-j<n} f[j] * b[k-j]
    # Let b_i be the variable at position i.
    # (f*b)[k] = sum_i f[k-i] * b_i (where f[m]=0 if m out of [0,n-1])
    
    # So A_ub[k, i] = f[k-i] if 0 <= k-i < n, else 0
    # A_ub is a matrix of size (2n-1) x n
    
    # To keep LP tractable, only include constraints where the bound is tight
    # (i.e., where (f*b_current) is close to M)
    
    # First, compute current convolution
    conv = fast_convolve(f)
    
    # Use only the top constraints (where conv is largest)
    n_constraints = min(2 * n - 1, max(20, n))
    top_lags = np.argpartition(conv, -n_constraints)[-n_constraints:]
    
    A_rows = []
    b_vals = []
    for k in top_lags:
        row = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            row[lo:hi + 1] = f[k - hi:k - lo + 1][::-1]
        A_rows.append(row)
        b_vals.append(M)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    
    # Objective: maximize sum(b) => minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Equality constraint: sum(b) = 1
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': max(1, time_limit - (time.monotonic() - t0))})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_iteration(f, time_limit=60, n_lp_constraints=None):
    """
    Full LP iteration: compute M = max conv(f,f), solve LP for g,
    then blend f and g.
    """
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max() * 1.001  # slight relaxation for feasibility
    
    g = lp_refine(f, M, n, time_limit=max(5, time_limit * 0.7))
    if g is None:
        return f
    
    g = np.maximum(g, 0)
    gs = g.sum()
    if gs < 1e-10:
        return f
    g = g / gs
    
    # Blend
    best_f = f.copy()
    best_obj = M
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
        blend = (1 - t) * f + t * g
        blend = project_simplex(blend)
        obj = fast_convolve(blend).max()
        if obj < best_obj:
            best_obj = obj
            best_f = blend.copy()
    
    return best_f


def coordinate_descent(f0, time_limit=60, verbose=False):
    """Coordinate descent: try scaling each coordinate and renormalizing."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-15:
                continue
            
            cur_obj = best_obj
            best_new_val = old_val
            
            multipliers = [0.5, 0.7, 0.85, 1.2, 1.5, 2.0]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                f_trial = project_simplex(f_trial)
                obj = fast_convolve(f_trial).max()
                if obj < cur_obj - 1e-14:
                    cur_obj = obj
                    best_new_val = new_val
            
            if best_new_val != old_val:
                f[j] = best_new_val
                f = project_simplex(f)
                if best_obj > cur_obj - 1e-14:
                    best_obj = cur_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved and passes > 1:
            break
    
    return best_f, best_obj


def make_gaussian_init(n, center=0.5, sigma=0.15):
    x = np.linspace(0, 1, n)
    f = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization:
    1. Evaluate parent and uniform baselines
    2. Subgradient descent from multiple starts
    3. LP-based refinement iterations
    4. Coordinate descent polishing
    5. Length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # === Phase 0: Baselines ===
    print("=== Phase 0: Baselines ===")
    
    # Parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            try_seq(parent, "parent")
            print(f"  Parent: n={len(parent)}, score={best_score:.10f}")
    except Exception:
        pass

    # Uniform baselines at various lengths
    for n in [128, 256, 384, 512, 640, 768, 896, 1024, 1280, 1536, 2048]:
        try_seq(np.ones(n), f"uniform_{n}")

    print(f"  Best after baselines: {best_score:.10f}")

    # Determine base for optimization
    if parent is not None:
        base = parent / parent.sum()
        n_base = len(parent)
    else:
        base = np.ones(768) / 768
        n_base = 768

    # === Phase 1: Subgradient descent from multiple starts ===
    print(f"\n=== Phase 1: Subgradient descent (n={n_base}) ===")
    t_phase1_end = time.monotonic() + 240

    starts = {
        'base': base.copy(),
        'uniform': np.ones(n_base) / n_base,
        'gauss': make_gaussian_init(n_base, 0.5, 0.15),
        'gauss_wide': make_gaussian_init(n_base, 0.5, 0.3),
        'pert': project_simplex(base + rng.standard_normal(n_base) * 0.01),
        'pert2': project_simplex(base + rng.standard_normal(n_base) * 0.05),
    }

    sg_configs = [
        {'top_k': 50, 'temp_init': 0.005, 'lr_init': 0.005},
        {'top_k': 100, 'temp_init': 0.01, 'lr_init': 0.008},
        {'top_k': 200, 'temp_init': 0.02, 'lr_init': 0.01},
        {'top_k': 500, 'temp_init': 0.05, 'lr_init': 0.012},
    ]

    phase1_best = None
    phase1_best_obj = float('inf')

    for sname, sinit in starts.items():
        rem = t_phase1_end - time.monotonic()
        if rem < 15:
            break
        t_alloc = min(rem * 0.3, 40)
        
        print(f"  SG from '{sname}', t={t_alloc:.0f}s")
        for ci, cfg in enumerate(sg_configs[:2]):
            rem = t_phase1_end - time.monotonic()
            if rem < 10:
                break
            a_opt, obj = subgradient_optimize(sinit, time_limit=min(rem * 0.5, 30), **cfg)
            try_seq(a_opt * parent.sum() if parent is not None else a_opt, f"sg_{sname}_{ci}")
            if obj < phase1_best_obj:
                phase1_best_obj = obj
                phase1_best = a_opt.copy()

    if phase1_best is not None:
        try_seq(phase1_best, "p1_best")

    # === Phase 2: LP-based refinement iterations ===
    print(f"\n=== Phase 2: LP refinement (best={best_score:.10f}) ===")
    t_phase2_end = time.monotonic() + 300

    if best_seq is not None:
        f_cur = best_seq.copy()
        f_cur = f_cur / f_cur.sum()
        n_cur = len(f_cur)
        
        for iteration in range(8):
            rem = t_phase2_end - time.monotonic()
            if rem < 30:
                break
            t_alloc = min(rem * 0.7, 60)
            print(f"  LP iter {iteration}, t={t_alloc:.0f}s")
            
            f_new = lp_iteration(f_cur, time_limit=t_alloc)
            f_new = f_new / f_new.sum()
            
            obj = fast_convolve(f_new).max()
            print(f"    LP iter {iteration} obj: {obj:.10f}")
            
            if obj < fast_convolve(f_cur).max():
                f_cur = f_new
                try_seq(f_cur * (best_seq.sum()), f"lp_iter{iteration}")
            else:
                # Try smaller steps
                half = 0.5 * f_cur + 0.5 * f_new
                half = project_simplex(half)
                obj_half = fast_convolve(half).max()
                if obj_half < fast_convolve(f_cur).max():
                    f_cur = half
                    try_seq(f_cur * (best_seq.sum()), f"lp_half_{iteration}")
                else:
                    break
    else:
        print("  No best found, skipping")

    # === Phase 3: Extended subgradient from current best ===
    print(f"\n=== Phase 3: Extended SG (best={best_score:.10f}) ===")
    t_phase3_end = time.monotonic() + 300

    if best_seq is not None:
        f_best = best_seq / best_seq.sum()
        n_best = len(best_seq)
        
        # Multiple SG runs with different hyperparams
        for cfg_i, (tk, tmp, lr) in enumerate([
            (50, 0.003, 0.004),
            (100, 0.008, 0.006),
            (200, 0.015, 0.008),
            (500, 0.03, 0.01),
            (1000, 0.05, 0.012),
        ]):
            rem = t_phase3_end - time.monotonic()
            if rem < 20:
                break
            t_alloc = min(rem * 0.3, 60)
            a_opt, obj = subgradient_optimize(f_best, time_limit=t_alloc,
                                              top_k=tk, temp_init=tmp, lr_init=lr)
            try_seq(a_opt, f"p3_sg_{cfg_i}")
            
            # Also try from perturbed version
            rem = t_phase3_end - time.monotonic()
            if rem > 15 and cfg_i < 3:
                perturb = project_simplex(f_best + rng.standard_normal(n_best) * 0.005)
                a_opt2, obj2 = subgradient_optimize(perturb, time_limit=min(rem * 0.4, 30),
                                                    top_k=tk, temp_init=tmp, lr_init=lr)
                try_seq(a_opt2, f"p3_sg_pert_{cfg_i}")

    # === Phase 4: Coordinate descent polishing ===
    print(f"\n=== Phase 4: CD polishing (best={best_score:.10f}) ===")
    t_phase4_end = time.monotonic() + 200

    if best_seq is not None:
        f_best = best_seq / best_seq.sum()
        n_best = len(best_seq)
        
        # CD on best
        rem = t_phase4_end - time.monotonic()
        if rem > 30:
            t_cd = min(rem * 0.5, 120)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, obj = coordinate_descent(f_best, time_limit=t_cd)
            try_seq(a_opt, "p4_cd")
            
            # CD from perturbed
            rem = t_phase4_end - time.monotonic()
            if rem > 20:
                pert = project_simplex(f_best + rng.standard_normal(n_best) * 0.003)
                a_opt, obj = coordinate_descent(pert, time_limit=min(rem * 0.5, 60))
                try_seq(a_opt, "p4_cd_pert")

    # === Phase 5: Length exploration ===
    print(f"\n=== Phase 5: Length exploration (best={best_score:.10f}) ===")
    t_phase5_end = time.monotonic() + 200

    if best_seq is not None:
        n_best = len(best_seq)
        target_lengths = [n for n in [256, 512, 768, 1024, 1280, 1536] if n != n_best]
        target_lengths.sort(key=lambda x: abs(x - n_best))
        target_lengths = target_lengths[:3]
        
        for tn in target_lengths:
            rem = t_phase5_end - time.monotonic()
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq / best_seq.sum(), tn)
            t_alloc = min(rem * 0.4, 60)
            
            a_opt, obj = subgradient_optimize(f_interp, time_limit=t_alloc,
                                              top_k=100, temp_init=0.01, lr_init=0.008)
            try_seq(a_opt, f"p5_n{tn}")

    # === Phase 6: Final fine-tuning ===
    print(f"\n=== Phase 6: Final (best={best_score:.10f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        f_best = best_seq / best_seq.sum()
        
        # Very fine SG
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt, _ = subgradient_optimize(f_best, time_limit=min(rem * 0.6, 40),
                                            top_k=50, temp_init=0.002, lr_init=0.002)
            try_seq(a_opt, "p6_fine")
        
        # Tiny perturbations
        for pi in range(5):
            rem = deadline - time.monotonic() - 5
            if rem < 5:
                break
            noise = rng.standard_normal(n_best) * 0.001
            fp = project_simplex(f_best + noise)
            a_opt, _ = subgradient_optimize(fp, time_limit=min(rem * 0.4, 15),
                                            top_k=50, temp_init=0.003, lr_init=0.003)
            try_seq(a_opt, f"p6_pert{pi}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
