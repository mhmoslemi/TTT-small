"""

Uses multi-phase strategy: global exploration, gradient refinement, and final polish.

import numpy as np

try:

except Exception:
    from numpy.fft import fft, ifft
    def next_fast_len(n):
        n = int(max(1, n))
        m = 1
        while m < n:
            m <<= 1
        return m

try:
    from scipy.optimize import minimize

def fft_len(n):
    return int(next_fast_len(max(1, 2 * int(n) - 1)))


def conv_sq(p):
    n = int(p.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = fft_len(n)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1].astype(np.float64, copy=False)


def internal_score(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.inf
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 1e-12:
        return np.inf
    c = conv_sq(p)
    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cmax) or cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (s * s))


def safe_normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or p.size == 0 or s <= 1e-12:
        n = max(1, int(p.size))
        return np.full(n, 1.0 / n, dtype=np.float64)
    return p / s


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def gradient_smoothed_max(p, T_factor):
    """
    Compute gradient of smoothed max of convolution w.r.t. p.
    Uses log-sum-exp with temperature T = T_factor * cmax.
    """
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0), np.empty(0), 1.0, 0.0

    L = int(next_fast_len(max(1, 3 * n - 2)))
    Fp = fft(p, L)
    c = np.real(ifft(Fp * Fp))[:2 * n - 1]

    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cmax) or cmax <= 0:
        return np.zeros(n), c, 1.0, 0.0

    T = max(T_factor * cmax, 1e-15)
    diff = (c - cmax) / T
    e = np.exp(diff)
    raw = float(e.sum())
    if not np.isfinite(raw) or raw <= 0:
        e = np.zeros_like(c)
        e[int(np.argmax(c))] = 1.0
        raw = 1.0

    w = e / raw
    # Gradient: d c_k / d p_j involves convolution of w with p
    # c_k = sum_i p_i * p_{k-i}
    # d c_k / d p_j = p_{k-j} + p_{j-k} (for valid indices)
    # grad_j = sum_k w_k * (p_{k-j} + p_{j-k})
    #        = sum_k w_k * p_{k-j} + sum_k w_k * p_{j-k}
    #        = (w * p)[j] + (w * p)[... ] 
    # Actually: sum_k w_k * p[k-j] is the cross-correlation of w and p at offset j
    # Let's use FFT: h = ifft(fft(w) * fft(p)) gives h[m] = sum_k w[k]*p[m-k] (circular)
    # We need the linear version.
    
    # Simpler approach: grad_j = sum_k w_k * (p[k-j] + p[j-k]) for valid k
    # = sum_{k: 0<=k-j<n} w_k * p[k-j] + sum_{k: 0<=j-k<n} w_k * p[j-k]
    # = sum_{i: 0<=i<n} w_{i+j} * p[i] + sum_{i: 0<=i<n} w_{j-i} * p[i]
    
    # Use FFT for efficiency
    # First term: sum_i w_{i+j} * p[i] for i+j < 2n-1
    # This is like a cross-correlation
    
    # Let's just use the FFT-based approach from the original code
    Fw = fft(w[::-1], L)
    h = np.real(ifft(Fp * Fw))
    
    if n == 1:
        G = h[:1]
    else:
        G = h[2 * n - 2 : n - 2 : -1]
    
    G = np.asarray(G, dtype=np.float64)
    if G.size != n:
        tmp = np.zeros(n)
        m = min(n, G.size)
        tmp[:m] = G[:m]
        G = tmp
    
    G = np.maximum(G, 0.0)
    g = 2.0 * G
    surr = float(cmax + T * np.log(raw))
    return g, c, T, surr

            inits.append(safe_normalize(np.maximum(p, 0.01)))
    
    # Cosine windows
    for a in [0.5, 1, 2, 3]:
        x = np.linspace(0, 1, n)
        p = np.abs(np.cos(a * np.pi * x))
        inits.append(safe_normalize(np.maximum(p, 0.001)))
    
    # Random smooth
    for _ in range(8):
        x = np.linspace(0, 1, n)

    # Two-peaked
    for mu1, mu2, sig in [(0.25, 0.75, 0.05), (0.2, 0.8, 0.1), (0.3, 0.7, 0.08)]:

        inits.append(safe_normalize(p))
    
    # W-shaped
    x = np.linspace(0, 1, n)
    p = 1 + 0.5*np.cos(4*np.pi*x)
    inits.append(safe_normalize(np.maximum(p, 0.01)))
    
    # M-shaped
    x = np.linspace(0, 1, n)

    # Triangular
    x = np.linspace(0, 1, n)

    inits.append(safe_normalize(p))
    
    # Inverted triangular
    p = 2*np.abs(x - 0.5)
    inits.append(safe_normalize(p + 0.1))

def optimize_gradient(p0, rng, time_limit, label=""):

    Gradient-based optimization with adaptive temperature and restarts.
    """
    p = project_simplex(np.asarray(p0, dtype=np.float64))

        return p, np.inf

    best_score = internal_score(best_p)

    prev_p = p.copy()

    iter_no = 0
    since_improve = 0
    T_factor = 0.1
    momentum = 0.0
    vel = np.zeros(n)

    while time.monotonic() < end:
        iter_no += 1
        
        # Adaptive temperature: decrease over time, increase when stuck
        T_factor = max(0.005, 0.1 * (0.999 ** iter_no))
        if since_improve > 200:

            base_lr = max(1.0/n, 0.01*pmax) / gabs
            lrs = [base_lr * f for f in [0.3, 0.1, 0.03, 0.01, 0.003]]
        else:
            lrs = [0.1, 0.03, 0.01]

        accepted = False
        
        for lr in lrs:
            if momentum > 0:
                vel = momentum * vel + (1 - momentum) * (-lr * g)
                v = p + vel
            else:
                v = p - lr * g
            
            pt = project_simplex(v)
            sc = internal_score(pt)
            
            if np.isfinite(sc) and sc < best_score - 1e-10:
                best_score = sc
                best_p = pt.copy()
                since_improve = 0
                current_score = sc
                prev_p = p
                p = pt
                accepted = True
                break
            elif np.isfinite(sc) and sc < current_score - 1e-10:
                # Accept if surrogate also improves
                sc2_c, sc2_cm, sc2_score, sc2_surr = internal_score(pt), 0, sc, 0
                if surr is not None and sc2_surr < surr + 0.5 * max(1.0, abs(surr)):
                    current_score = sc
                    prev_p = p
                    p = pt
                    accepted = True
                    since_improve += 1
                    break

        if not accepted:
            since_improve += 1
            T_factor = max(0.003, T_factor * 0.7)
            
            # Random perturbation
            if since_improve > 100:
                sigma = 0.05 / np.sqrt(n)
                if since_improve > 500:
                    sigma = 0.2 / np.sqrt(n)
                pt = project_simplex(p + rng.normal(0, sigma, n))
                sc = internal_score(pt)
                if np.isfinite(sc) and sc < best_score - 1e-10:
                    best_score = sc
                    best_p = pt.copy()

def _fft_len(n):

def _conv_sq(p):
    n = int(p.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = _fft_len(n)
    F = fft(p, L)

def _score(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)

        return np.inf
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 1e-12:

def _proj(v):
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-300:

    return out / s


def _grad_surrogate(p, Tfac):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0), np.empty(0), 1.0, 0.0
    L = int(next_fast_len(max(1, 3 * n - 2)))

    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cmax) or cmax <= 0:
        return np.zeros(n), c, 1.0, 0.0

    try:
        pa = np.asarray(height_sequence_1, dtype=np.float64)
        pa = np.where(np.isfinite(pa), pa, 0.0)
        pa = np.maximum(pa, 0.0)
        if pa.ndim == 1 and pa.size >= 32 and float(pa.sum()) > 0:
            parent = pa
    except Exception:
        parent = None

    best_p = None
    best_sc = np.inf

    def try_rec(p, sc):
        nonlocal best_p, best_sc
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64)
        if p.size == 0 or not np.all(np.isfinite(p)):
            return
        p = _normalize(p)
        if not np.isfinite(sc):
            sc = _score(p)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = float(sc)
            best_p = p.copy()
            print(f"  NEW BEST: {sc:.10f} (n={p.size})", flush=True)

    # Phase 1: Score initializations at several lengths
    print("Phase 1: Global exploration", flush=True)
    lengths = []
    if parent is not None:
        p_len = int(parent.size)
        lengths.append(p_len)
        for dl in [-200, -100, 0, 100, 200, 400, 800]:
            l = p_len + dl
            if 64 <= l <= 5000:
                lengths.append(l)

    base_lengths = [128, 192, 256, 384, 512, 700, 1024, 1200, 1536, 2048]
    for l in base_lengths:
        if l not in lengths:
            lengths.append(l)
    lengths = sorted(set(lengths))
    lengths = [l for l in lengths if 64 <= l <= 6000]

    candidates = []
    for n in lengths:
        if time.monotonic() > deadline - 600:
            break
        inits = _make_inits(n, rng, parent if n == len(parent) else None)
        for p in inits:
            sc = _score(p)
            if np.isfinite(sc):
                candidates.append((sc, p, n))
                try_rec(p, sc)
        candidates.sort(key=lambda x: x[0])

    print(f"Phase 1 done: {len(candidates)} candidates, best={best_sc:.10f}", flush=True)

    # Phase 2: Gradient optimization on top candidates
    print("Phase 2: Gradient optimization", flush=True)
    top = []
    seen_lengths = set()
    for sc, p, n in candidates:
        if n in seen_lengths:
            continue
        seen_lengths.add(n)
        top.append((sc, p, n))
        if len(top) >= 8:
            break

    top.sort(key=lambda x: x[0])

    rem = deadline - time.monotonic()
    if rem > 200 and top:
        n_best = len(top)
        time_per = max(30.0, min(350.0, (rem - 200) / n_best))
        for i, (sc, p, n) in enumerate(top):
            if time.monotonic() > deadline - 200:
                break
            limit = min(time_per, deadline - time.monotonic() - 200)
            if limit < 15:
                break
            use_lp = n <= 900
            p_best, sc_best = _optimize(p, rng, limit, f"opt{i}n{n}:", use_lp)
            try_rec(p_best, sc_best)
            # Re-optimize the current best
            if np.isfinite(sc_best) and sc_best < best_sc + 0.01:
                rem2 = deadline - time.monotonic() - 200
                if rem2 > 60:
                    p_b2, sc_b2 = _optimize(p_best, rng, min(rem2 * 0.4, 200.0), f"ref{i}:", use_lp)
                    try_rec(p_b2, sc_b2)

    # Phase 3: Coordinate descent polish
    print("Phase 3: Coordinate polish", flush=True)
    if best_p is not None and time.monotonic() < deadline - 30:
        limit = min(deadline - time.monotonic() - 20, 600.0)
        p_pol, sc_pol = _coord_descent(best_p, rng, limit, "coord:")
        try_rec(p_pol, sc_pol)

    # Phase 4: Final micro-optimization
    if best_p is not None and time.monotonic() < deadline - 10:
        limit = max(5.0, deadline - time.monotonic() - 5)
        p_fin, sc_fin = _optimize(best_p, rng, limit, "final:", False)
        try_rec(p_fin, sc_fin)

    if best_p is None:
        best_p = np.full(256, 1.0 / 256)
        best_sc = _score(best_p)

    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    best_p = np.minimum(best_p, 1000.0)
    s = float(best_p.sum())
    if s < 0.01:
        best_p = best_p * (1.0 / s) if s > 0 else np.full(best_p.size, 0.01)

    print(f"FINAL: score={best_sc:.10f}, n={best_p.size}", flush=True)
    return [float(x) for x in best_p]
