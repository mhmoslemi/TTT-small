LP best-response iteration for minimizing max autocorrelation of non-negative step functions.
Uses diverse initializations, adaptive LP tightening, and line search.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.fft import fft, ifft, next_fast_len

def conv_square(p):
    n = len(p)
    L = next_fast_len(2 * n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1]

def compute_score(p):
    n = len(p)
    s = float(np.sum(p))
    if s <= 1e-12:
        return np.inf
    c = conv_square(p)
    return 2.0 * n * float(np.max(c)) / (s * s)

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    k = int(np.sum(cond) - 1)
    if k < 0:
        return np.full(n, 1.0 / n)
    theta = (cssum[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s <= 1e-12:
        return np.full(n, 1.0 / n)
    return out / s

def build_conv_matrix(p):
    """Build sparse matrix A where (A @ g)[k] = (p*g)[k]."""
    n = len(p)
    rows = np.repeat(np.arange(n), n)  # for each i, j ranges 0..n-1
    # Actually let me think about this more carefully
    # (p*g)[k] = sum_i p[k-i] * g[i] for k-i in [0, n-1]
    # So A[k, i] = p[k-i] if 0 <= k-i <= n-1
    # For fixed i, k ranges from i to i+n-1
    rows_list = []
    cols_list = []
    data_list = []
    for i in range(n):
        ks = np.arange(i, i + n)
        js = np.arange(n)  # p[0], p[1], ..., p[n-1]
        rows_list.append(ks)
        cols_list.append(np.full(n, i))
        data_list.append(p)
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    data = np.concatenate(data_list)
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response(p, M):
    n = len(p)
    A = build_conv_matrix(p)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs', options={'disp': False, 'presolve': True})
    if res.status == 0:

        g = np.maximum(g, 0)
        s = g.sum()
        if s > 1e-12:

    return None

        # Symmetric exponential (V-shape)

"""
LP best-response iteration for minimizing max self-convolution of non-negative step functions.
Combines diverse shape initialization, adaptive LP tightening, over-relaxation, and multi-scale search.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.fft import fft, ifft, next_fast_len


def _conv_sq(p):
    """Full self-convolution of 1D vector p."""
    n = len(p)
    if n == 0:
        return np.empty(0)
    L = next_fast_len(2 * n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1].astype(np.float64, copy=False)


def _score(p):
    """Compute 2*n*max(p*p)/(sum(p))^2."""
    n = len(p)
    s = float(np.sum(p))
    if s <= 1e-12:
        return np.inf
    c = _conv_sq(p)
    cm = float(np.max(c))
    return 2.0 * n * cm / (s * s)


def _proj_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    k = int(np.sum(cond) - 1)
    if k < 0:
        return np.full(n, 1.0 / n)
    theta = (cssum[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s <= 1e-12:
        return np.full(n, 1.0 / n)
    return out / s


def _norm(p):
    """Normalize to sum 1, non-negative."""
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s <= 1e-12:
        return np.full(len(p), 1.0 / max(1, len(p)))
    return p / s


def _precompute_struct(n):
    """Pre-compute (row, col) structure for convolution matrix A[g] = p*p."""
    # A[k, i] = p[k-i] for 0<=k-i<=n-1
    # For column i, valid k range is [i, i+n-1], and p index is k-i = 0..n-1
    ii = np.tile(np.arange(n), n)  # column index (repeated n times)
    jj = np.tile(np.arange(n), 1)  # p index within each column
    kk = ii + jj  # row index = i + j
    return ii, jj, kk


def _make_A(ii, jj, kk, p, shape):
    """Build sparse convolution matrix."""
    data = p[jj]
    return csr_matrix((data, (kk, ii)), shape=shape)


def _lp_solve(p, M, ii, jj, kk, shape):
    """Solve: max sum(g) s.t. A@g <= M, g >= 0. Returns normalized g or None."""
    A = _make_A(ii, jj, kk, p, shape)
    c_obj = -np.ones(len(p))
    b_ub = np.full(shape[0], M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None),
                      method='highs', options={'disp': False, 'presolve': True,
                                               'time_limit': 30})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(g.sum())
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return None

                _try_record(p, sc)
        scored.sort(key=lambda x: x[0])

        # Keep top 8 for LP
        for sc, p in scored[:8]:
            all_jobs.append((sc, p, n, ii, jj, kk, shape, "n=%d" % n))

        print("  n=%d: best init score=%.6f, %d inits" % (n, scored[0][0], len(scored)), flush=True)

    all_jobs.sort(key=lambda x: x[0])

    # Phase 2: LP best-response on top jobs
    print("Phase 2: LP best-response optimization...")
    remaining = deadline - time.monotonic()
    n_jobs = min(len(all_jobs), 6)
    # Weight time allocation by promise
    if n_jobs > 0 and remaining > 60:
        time_per = max(20.0, remaining * 0.6 / max(1, n_jobs))
        for j in range(n_jobs):
            if time.monotonic() > deadline - 180:
                break
            sc, p, n, ii, jj, kk, shape, label = all_jobs[j]
            t_budget = min(time_per, deadline - time.monotonic() - 180)
            if t_budget < 10:
                break
            print("  Job %d (%s, score=%.6f): running LP for %.0fs..." %
                  (j, label, sc, t_budget), flush=True)
            t0 = time.monotonic()
            p_best, sc_best = _lp_cycle(p, ii, jj, kk, shape, t_budget, rng, verbose=False)
            _try_record(p_best, sc_best)
            print("    Done: score=%.8f (time=%.1fs)" % (sc_best, time.monotonic() - t0), flush=True)

    # Phase 3: Intensive polish on the best
    print("Phase 3: Intensive polishing...")
    if best_overall_p is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 30:
            p_best = _norm(best_overall_p)
            n = len(p_best)
            ii, jj, kk = _precompute_struct(n)
            shape = (2 * n - 1, n)
            p_pol, sc_pol = _polish(p_best, ii, jj, kk, shape, remaining, rng)
            _try_record(p_pol, sc_pol)

    # Phase 4: Try a few more LP cycles from random perturbations of best
    if best_overall_p is not None and time.monotonic() < deadline - 30:
        p_best = _norm(best_overall_p)
        n = len(p_best)
        ii, jj, kk = _precompute_struct(n)
        shape = (2 * n - 1, n)
        remaining = deadline - time.monotonic() - 10
        n_trials = min(5, max(1, int(remaining // 30)))
        for tr in range(n_trials):
            if time.monotonic() > deadline - 15:
                break
            p_trial = _proj_simplex(p_best + rng.normal(0, 0.5 / n, n))
            t_budget = min(25.0, deadline - time.monotonic() - 15)
            if t_budget < 5:
                break
            p_tr, sc_tr = _lp_cycle(p_trial, ii, jj, kk, shape, t_budget, rng)
            _try_record(p_tr, sc_tr)

    # Final: ensure valid output
    if best_overall_p is None:
        # Fallback: uniform of length 700
        best_overall_p = np.full(700, 1.0 / 700)
        best_overall_score = _score(best_overall_p)

    best_overall_p = _norm(best_overall_p)
    best_overall_p = np.minimum(best_overall_p, 1000.0)
    final_score = best_overall_score
    print("FINAL: score=%.10f, length=%d, elapsed=%.1fs" %
          (final_score, len(best_overall_p), time.monotonic() - start_time), flush=True)

    return [float(x) for x in best_overall_p]
