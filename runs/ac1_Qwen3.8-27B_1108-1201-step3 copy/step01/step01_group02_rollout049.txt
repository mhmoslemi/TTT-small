Improved LP-based optimization for minimizing peak auto-convolution of non-negative sequences.
Uses "minimize t" LP formulation with adaptive step sizes, Fourier-domain polishing,
and multi-start strategy.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.fft import fft, ifft
import time


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def build_conv_structure(n):
    """Precompute row/col indices for the convolution matrix conv(a, .)."""
    n_conv = 2 * n - 1
    rows = []
    cols = []
    for k in range(n_conv):
        m_start = max(0, k - (n - 1))
        m_end = min(n - 1, k)
        m_range = range(m_start, m_end + 1)
        j_range = range(k - m_end, k - m_start + 1)
        rows.extend([k] * len(m_range))
        cols.extend(m_range)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)

    """Given precomputed structure, extract data values for conv(a, .) matrix."""
    # A[k, m] = a[k-m] for valid k-m
    k_minus_m = row_idx - col_idx
    return a[k_minus_m]

    c[-1] = 1.0
    b_ub = np.zeros(n_conv)
    
    A_eq = csr_matrix((np.ones(n), (np.zeros(1, dtype=np.int32), np.arange(n))), shape=(1, n + 1))
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(0, res.x[:n])
            t_val = res.x[-1]
            return g, t_val
    except Exception:
        pass
    return None, None


def lp_refine(a, time_limit_s=300.0, verbose=True, seed=42):

    stagnant = 0

        if elapsed > time_limit_s:
            break
        
        remaining = time_limit_s - elapsed
        if remaining < 10.0:

            continue
        
        # Line search for step size

def fourier_polish(a, n_iters=10, verbose=True):
    """Polish using Fourier-domain flattening."""
    a = normalize(a.copy())
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    # Use a padded DFT for better frequency resolution
    N = 1  # power of 2 >= 4n
    while N < 4 * n:
        N *= 2
    
    for it in range(n_iters):
        A = fft(a, N)
        power = np.abs(A) ** 2
        # Target: make power more uniform
        # Use a soft flattening: sort and interpolate
        sorted_power = np.sort(power)
        # Create target: geometric mean interpolation
        target_level = np.geometric_mean(sorted_power[N // 8 : 7 * N // 8])
        # Blend current power with target
        blend = 0.3
        new_power = (1 - blend) * power + blend * target_level
        # Scale to maintain sum(a) = 1 (i.e., A[0] = 1)
        new_power[0] = power[0]  # Keep DC component
        # Reconstruct: need |A|^2 = new_power, keep phase
        phase = np.angle(A)
        new_A = np.sqrt(new_power) * np.exp(1j * phase)
        new_a = ifft(new_A).real[:n]
        new_a = normalize(new_a)
        s = score_sequence(new_a)
        if s < best_score:
            best_score = s
            best_a = new_a.copy()
            a = new_a.copy()
    
    return best_a, best_score


def local_search(a, time_limit_s=60.0, rng=None, verbose=True):
    """Random local search with window perturbations."""
    if rng is None:
        rng = np.random.default_rng(0)
    start_time = time.monotonic()
    a = normalize(a.copy())
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    accept_count = 0
    
    while time.monotonic() - start_time < time_limit_s:
        # Random window perturbation
        width = max(2, int(rng.integers(n // 200, n // 10)))
        start_pos = int(rng.integers(0, max(1, n - width)))
        candidate = a.copy()
        candidate[start_pos:start_pos + width] *= np.exp(rng.normal(0, 0.15, width))
        # Sometimes symmetrize
        if rng.random() < 0.3:
            candidate = 0.5 * (candidate + candidate[::-1])
        candidate = normalize(candidate)
        s = score_sequence(candidate)
        if s < best_score:
            best_score = s
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
    
    if verbose:
        print(f"    LS: {accept_count} accepts, best={best_score:.6f}")
    return best_a, best_score

Let me reconsider the problem from a variational perspective. We want to minimize:
`J(a) = 2n * max_k (a*a)(k) / (sum a)^2`

subject to `a >= 0`.

By scale invariance, set `sum(a) = 1`:
`J(a) = 2n * max_k (a*a)(k)`

This is equivalent to: find the smallest `M` such that there exists `a >= 0`, `sum(a)=1` with `(a*a)(k) <= M` for all `k`.

This is a feasibility problem: does there exist `a >= 0`, `sum(a)=1` with `a^T Q_k a <= M` for all `k`?

Where `Q_k` is the matrix with `(Q_k)_{j,i} = 1` if `j-i = k` or `i-j = k`... wait, let me be more careful.

`(a*a)(k) = sum_j a_j a_{j-k} = sum_{j,i} a_j B_k[j,i] a_i`

where `B_k[j,i] = 1` if `j - i = k` (i.e., `i = j-k`), else 0.

Wait, that's not symmetric. Let me write it as:
`(a*a)(k) = sum_j a_j a_{j-k}`

For `k >= 0`: this is `sum_{j=k}^{n-1} a_j a_{j-k}`.
For `k < 0`: let `k' = -k > 0`, then `(a*a)(-k') = sum_j a_j a_{j+k'} = sum_{j=0}^{n-1-k'} a_j a_{j+k'} = (a*a)(k')`.

So the auto-convolution is symmetric: `(a*a)(k) = (a*a)(-k)`. We only need to consider `k = 0, 1, ..., n-1`.

For `k >= 0`:
`(a*a)(k) = sum_{j=k}^{n-1} a_j a_{j-k}`

This is a quadratic form: `a^T Q_k a` where `Q_k` has `Q_k[j, j-k] = Q_k[j-k, j] = 1/2` for `j >= k`.

Wait, let me be more careful. `a^T Q_k a = sum_{j,i} a_j Q_k[j,i] a_i`.
We want this to equal `sum_{j=k}^{n-1} a_j a_{j-k}`.
So we need `Q_k[j, j-k] + Q_k[j-k, j] = 1` for `j >= k`.
Setting `Q_k[j, j-k] = Q_k[j-k, j] = 1/2` works.

So `Q_k` is a symmetric matrix with `1/2` on the k-th sub/super-diagonal.

Now, `(a*a)(k) <= M` is the constraint `a^T Q_k a <= M`.

The problem is:
Minimize `M`
Subject to: `a^T Q_k a <= M` for `k = 0, ..., n-1`
             `a >= 0`
             `sum(a) = 1`

This is a QCQP (quadratically constrained quadratic program) with a linear objective. It's NP-hard in general, but for this specific structure (bidiagonal Q_k matrices), there might be exploitable structure.

One approach: **SOCP relaxation**. Each constraint `a^T Q_k a <= M` can be relaxed to an SOCP constraint using the Schur complement:
`[M, a^T; a, I] >= 0` (SDP constraint)
But this is an SDP, which is expensive for large n.

Another approach: **sequential quadratic programming (SQP)** or **active set methods**. At each iteration, identify the active constraints (the lags where `(a*a)(k) = M`), and solve a QP subproblem.

But I think the LP approach (which is a special case of the bilinear complementary problem) is probably the most practical for our time budget.

At each iteration:

3. If `t < M`, update `a <- (1-α)a + αg` with appropriate α.
4. If `t >= M`, try a different approach (subgradient, perturbation, etc.).

The key insight is that step 2 gives the "best possible" direction (in the bilinear sense) for reducing the peak. If no such direction exists (`t >= M`), we're at a local minimum of the bilinear relaxation.

"""
Multi-phase optimization for minimizing peak auto-convolution of non-negative sequences.
Combines LP 'minimize t' formulation, subgradient descent, Fourier-domain polishing,
and local search with multi-start strategy.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.fft import fft, ifft
import time


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def build_conv_indices(n):
    """Precompute (row, col) indices for the convolution matrix."""
    n_conv = 2 * n - 1
    rows_list = []
    cols_list = []
    for k in range(n_conv):
        m_lo = max(0, k - (n - 1))
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            rows_list.extend([k] * (m_hi - m_lo + 1))
            cols_list.extend(range(m_lo, m_hi + 1))
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp(a, row_idx, col_idx, timeout=30.0):
    """Solve: minimize t s.t. conv(a,g) <= t, sum(g)=1, g>=0, t>=0.
    Returns (g, t_opt) or (None, None)."""
    n = len(a)
    n_conv = 2 * n - 1
    data = a[row_idx - col_idx]
    A = csr_matrix((data, (row_idx, col_idx)), shape=(n_conv, n))
    
    # Build full A_ub = [A, -I_(n_conv)]
    A_ub_rows = np.concatenate([row_idx, np.arange(n_conv)])
    A_ub_cols = np.concatenate([col_idx, (np.arange(n_conv) + n).astype(np.int32)])
    A_ub_data = np.concatenate([data, -np.ones(n_conv)])
    A_ub = csr_matrix((A_ub_data, (A_ub_rows, A_ub_cols)), shape=(n_conv, n + 1))
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    b_ub = np.zeros(n_conv)
    
    A_eq = csr_matrix((np.ones(n), (np.zeros(1, dtype=np.int32), np.arange(n))), shape=(1, n + 1))
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(0.0, res.x[:n])
            return g, float(res.x[-1])
    except Exception:
        pass
    return None, None

def subgradient_refine(a, n_iters=500, verbose=False):
    """Fast subgradient-based local refinement."""
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    no_improve = 0
    
    for it in range(n_iters):
        conv = np.convolve(a, a)
        k_star = int(np.argmax(conv))
        
        # Subgradient of (a*a)(k_star) w.r.t. a
        sg = np.zeros(n)
        if k_star < n:
            sg[:n - k_star] += a[k_star:]
            sg[k_star:] += a[:n - k_star]
        
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        sg /= sg_norm
        
        # Backtracking line search
        step = 0.3
        improved = False
        for _ in range(8):
            a_new = normalize(a - step * sg)
            s_new = score_sequence(a_new)
            if s_new < best_score - 1e-12:
                best_score = s_new
                best_a = a_new.copy()
                a = a_new.copy()
                improved = True
                no_improve = 0
                break
            step *= 0.5
        
        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
    
    if verbose:
        print(f"    Subgrad: best={best_score:.6f}")
    return best_a, best_score


def fourier_polish(a, n_iters=30, verbose=False):
    """Fourier-domain flattening polish."""
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    N = 1
    while N < 4 * n:
        N *= 2
    
    no_improve = 0

        A = fft(a, N)
        power = np.abs(A) ** 2
        ac_power = power[1:].copy()
        geo_mean = np.exp(np.mean(np.log(ac_power + 1e-30)))
        
        improved = False

            phase = np.angle(A)
            new_A = mag * np.exp(1j * phase)
            new_a = np.real(ifft(new_A))[:n]
            new_a = normalize(new_a)
            s = score_sequence(new_a)
            if s < best_score - 1e-12:

                best_a = new_a.copy()
                a = new_a.copy()
                improved = True
                break

            no_improve += 1
            if no_improve > 5:

            accepts += 1
