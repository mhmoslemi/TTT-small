def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _fft_convolve(a):
    """Full auto-convolution using FFT."""
    n = len(a)
    m = 1
    while m < 2 * n - 1:
        m *= 2
    fa = np.fft.rfft(a, m)
    result = np.fft.irfft(fa * fa, m)
    return result[:2 * n - 1]

    """Compute 2*n*max(convolve(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = _fft_convolve(a)

def _grad_smooth(a, tau, M):
    """Compute smoothed gradient of max(convolve(a,a)). Returns (grad, b_max)."""
    n = len(a)
    b = _fft_convolve(a)
    b_max = np.max(b)
    
    if M < len(b):
        k_idx = np.argpartition(b, -M)[-M:]
    else:
        k_idx = np.arange(len(b))
    
    b_sel = b[k_idx]
    wts = np.exp(tau * (b_sel - b_sel.max()))
    wts /= np.sum(wts)
    
    grad = np.zeros(n)
    for ki in range(len(k_idx)):
        if wts[ki] < 1e-12:
            continue
        k = int(k_idx[ki])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            grad[idx] += wts[ki] * 2.0 * a[k - idx]
    
    return grad, b_max

    vals = log_a + log_scores
    vals -= np.max(vals)
    a_new = np.exp(vals)
    a_new /= np.sum(a_new)
    return a_new
