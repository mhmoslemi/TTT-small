```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth L-BFGS optimization
with annealed temperature on a softmax parameterization.
Uses FFT-based gradient computation for the log-sum-exp smoothed surrogate.
Multi-start with diverse initializations and final LP-based polish.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time

def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def softmax(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def compute_objective_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def smooth_obj_grad(z, t):
    """
    Smooth surrogate F_t(softmax(z)) and gradient w.r.t. z.
    F_t(a) = (2n/t) * logsumexp(t * conv(a,a))
    """
    a = softmax(z)
    n = len(a)
    conv = fast_convolve(a)
    
    scaled = t * conv
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    
    log_sum_exp = scaled.max() + np.log(exp_scaled.sum())
    F = (2.0 * n / t) * log_sum_exp
    
    # grad_a[j] = 4*n * sum_i a[i] * p[i+j]
    a_rev = a[::-1]
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    A_rev_fft = np.fft.rfft(a_rev, fft_size)
    P_fft = np.fft.rfft(p, fft_size)
    conv_full = np.fft.irfft(A_rev_fft * P_fft, fft_size)[:3 * n - 2]
    
    cross = conv_full[n - 1: n - 1 + n]
    grad_a = 4.0 * n * cross
    
    dot = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot)
    
    return F, grad_z

def anneal_optimize(z0, t_levels, maxiter_per_level=50, time_limit=60.0):
    """
    Run L-BFGS on the smooth surrogate with an annealing schedule.
    z0: initial point in z-space (log-sum-exp parameterization)
    t_levels: list of temperatures (increasing)
    Returns: best z found, corresponding a, and objective.
    """
    t0 = time.monotonic()
    z = z0.copy()
    best_z = z.copy()
    best_a = softmax(z)
    best_obj = compute_objective_normalized(best_a)
    
    for t in t_levels:
        if time.monotonic() - t0 > time_limit:
            break
        result = minimize(
            smooth_obj_grad, z, args=(t,),
            method='L-BFGS-B', jac=True,
            options={'maxiter': maxiter_per_level, 'ftol': 1e-10, 'gtol': 1e-8}
        )
        z = result.x
        a = softmax(z)
        obj = compute_objective_normalized(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            best_z = z.copy()
    
    return best_z, best_a, best_obj

def lp_direction(a, M):
    """
    Solve LP: max sum(g) s.t. g >= 0, (a*g)_k <= M for all k.
    Returns optimal g (unnormalized).
    """
    n = len(a)
    c = -np.ones(n)
    
    # Build constraint matrix: A_ub[k, j] = a[k-j] if 0<=k-j<n else 0
    # A_ub[k, j] = a[k-j] when j <= k <= j+n-1
    # For column j: non-zero in rows j through j+n-1, values a[0], a[1], ..., a[n-1]
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = a
    
    b_ub = np.full(2 * n - 1, M)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def lp_polish(a, time_limit=60.0, max_iter=200):
    """LP-based polishing iteration."""
    a = np.maximum(a, 0).astype(np.float64)
    n = len(a)
    a_sum = a.sum()
    if a_sum < 1e-12:
        return a
    a = a / a_sum
    
    best_a = a.copy()
    best_obj = compute_objective_normalized(a)
    t0 = time.monotonic()
    t_blend = 0.5
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        
        g = lp_direction(a, M)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        
        g = g / g_sum
        
        a_new = (1 - t_blend) * a + t_blend * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-12:
            break
        a_new = a_new / s
        
        obj = compute_objective_normalized(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            t_blend = min(0.5, t_blend * 1.1)
        else:
            t_blend *= 0.7
            if t_blend < 0.01:
                break
    
    return best_a

def generate_initializations(n, rng):
    """Generate diverse initializations for length n."""
    x = np.arange(n, dtype=np.float64)
    inits = {}
    
    # Uniform
    inits['uniform'] = np.ones(n)
    
    # Gaussian center
    for sigma_frac in [0.1, 0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - n / 2) / (sigma_frac * n)) ** 2)
        inits[f'gauss_{sigma_frac}'] = g
    
    # Ramp
    inits['ramp'] = np.arange(1, n + 1, dtype=np.float64)
    
    # Bimodal (edges)
    bi = np.zeros(n)
    bi[:max(1, n // 4)] = 1.0
    bi[-max(1, n // 4):] = 1.0
    inits['bimodal'] = bi
    
    # Cosine
    inits['cosine'] = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    
    # Random
    inits['random1'] = np.abs(rng.standard_normal(n)) + 0.1
    inits['random2'] = rng.dirichlet(np.ones(np.maximum(1, n // 10)) * 10)[:n]  # hmm this won't work if n//10 > n
    
    # Better random
    inits['random2'] = np.abs(rng.standard_normal(n)) + 0.01
    
    # Sparse random
    sparse = np.zeros(n)
    n_sparse = max(1, n // 4)
    idx = rng.choice(n, size=n_sparse, replace=False)
    sparse[idx] = rng.uniform(0.5, 1.0, size=n_sparse)
    inits['sparse'] = sparse
    
    # Chirp-like
    inits['chirp'] = 0.5 + 0.5 * np.sin(2 * np.pi * x / n * 3)
    
    return inits
```

```python
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_objective_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def smooth_obj_grad(z, t):
    """
    Smooth surrogate F_t(softmax(z)) and gradient w.r.t. z.
    F_t(a) = (2n/t) * logsumexp(t * conv(a,a))
    Gradient verified: dF/da_j = 4n * sum_i a_i * p_{i+j}
    """
    a = softmax_fn(z)
    n = len(a)
    conv = fast_convolve(a)

    scaled = t * conv
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()

    log_sum_exp = scaled.max() + np.log(exp_scaled.sum())
    F = (2.0 * n / t) * log_sum_exp

    # grad_a[j] = 4*n * sum_{i=0}^{n-1} a[i] * p[i+j]
    # Compute via FFT: cross[j] = (a_rev * p)[n-1+j]
    a_rev = a[::-1]
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    A_rev_fft = np.fft.rfft(a_rev, fft_size)
    P_fft = np.fft.rfft(p, fft_size)
    conv_full = np.fft.irfft(A_rev_fft * P_fft, fft_size)[:3 * n - 2]

    cross = conv_full[n - 1: n - 1 + n]
    grad_a = 4.0 * n * cross

    # Chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot)

    return F, grad_z


def anneal_optimize(z0, t_levels, maxiter_per_level=50, time_limit=60.0, method='L-BFGS-B'):
    """
    Run L-BFGS on the smooth surrogate with an annealing schedule.
    Returns: best z, corresponding a, and exact objective.
    """
    t0 = time.monotonic()
    z = z0.copy()
    best_z = z.copy()
    best_a = softmax_fn(z)
    best_obj = compute_objective_normalized(best_a)

    for t in t_levels:
        if time.monotonic() - t0 > time_limit:
            break
        try:
            result = minimize(
                smooth_obj_grad, z, args=(t,),
                method=method, jac=True,
                options={'maxiter': maxiter_per_level, 'ftol': 1e-12, 'gtol': 1e-10}
            )
            z = result.x
        except Exception:
            break
        a = softmax_fn(z)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a /= s
        obj = compute_objective_normalized(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            best_z = z.copy()

    return best_z, best_a, best_obj


def lp_direction(a, M):
    """
    Solve LP: max sum(g) s.t. g >= 0, (a*g)_k <= M for all k=0,...,2n-2.
    Returns optimal g (unnormalized) or None.
    """
    n = len(a)
    c = -np.ones(n)

    # A_ub[k, j] = a[k-j] if 0<=k-j<n else 0
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = a

    b_ub = np.full(2 * n - 1, M)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_polish(a, time_limit=60.0, max_iter=300):
    """LP-based polishing iteration from the SOA description."""
    a = np.maximum(a, 0).astype(np.float64)
    n = len(a)
    a_sum = a.sum()
    if a_sum < 1e-12:
        return a
    a = a / a_sum

    best_a = a.copy()
    best_obj = compute_objective_normalized(a)
    t0 = time.monotonic()
    t_blend = 0.4

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        M = conv.max()

        g = lp_direction(a, M)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-12:
            break

        g = g / g_sum

        a_new = (1 - t_blend) * a + t_blend * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-12:
            break
        a_new = a_new / s

        obj = compute_objective_normalized(a_new)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
        else:
            t_blend *= 0.75
            if t_blend < 0.005:
                break

    return best_a


def generate_initializations(n, rng):
    """Generate diverse initializations for length n."""
    x = np.arange(n, dtype=np.float64)
    inits = {}

    inits['uniform'] = np.ones(n)

    for sf in [0.08, 0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - n / 2) / (sf * n)) ** 2)
        inits[f'gauss_{sf}'] = g

    inits['ramp'] = np.arange(1, n + 1, dtype=np.float64)

    bi = np.zeros(n)
    bi[:max(1, n // 5)] = 1.0
    bi[-max(1, n // 5):] = 1.0
    inits['bimodal'] = bi

    inits['cosine'] = 0.5 * (1 + np.cos(2 * np.pi * x / n))

    inits['random1'] = np.abs(rng.standard_normal(n)) + 0.05
    inits['random2'] = np.abs(rng.standard_normal(n)) + 0.01

    sparse = np.zeros(n)
    n_sparse = max(2, n // 3)
    idx = rng.choice(n, size=n_sparse, replace=False)
    sparse[idx] = rng.uniform(0.5, 1.0, size=n_sparse)
    inits['sparse'] = sparse

    inits['chirp'] = 0.3 + 0.7 * np.abs(np.sin(2 * np.pi * x * 2.0 / n))

    # Bimodal sharp (edge-peaked, motivated by conv_0 = a_0^2 constraint)
    edge = np.zeros(n)
    decay = np.exp(-np.arange(n) / (n * 0.1))
    edge[:n // 2] = decay[:n // 2]
    edge[n // 2:] = decay[:n // 2][::-1]
    inits['edge_sharp'] = edge

    return inits


def perturbed_restart(a, rng, sigma=0.1):
    """Create a perturbed version of a for escaping local minima."""
    noise = np.exp(rng.standard_normal(len(a)) * sigma)
    a_pert = a * noise
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def run_optimization(seed=42, budget_s=1800):
    """Main multi-phase optimization routine."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Access parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={parent.size}, sum={parent.sum():.6f}")
    except Exception:
        pass

    # ===================== PHASE 1: Quick Scan =====================
    print("Phase 1: Quick scan...")
    lengths = [128, 192, 256, 384, 512, 1024]

    scan_results = []
    phase1_deadline = time.monotonic() + min(400.0, (deadline - time.monotonic()) * 0.25)

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break

        inits = generate_initializations(n, rng)

        if parent is not None:
            if parent.size >= n:
                pp = parent[:n].copy()
                pp = np.maximum(pp, 0)
                if pp.sum() > 1e-10:
                    inits['parent'] = pp / pp.sum()
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
                pp = np.maximum(pp, 0)
                if pp.sum() > 1e-10:
                    inits['parent_interp'] = pp / pp.sum()

        n_configs = len(inits)
        time_per = max(3.0, (phase1_deadline - time.monotonic()) / max(1, n_configs * len(lengths) - 3))

        for name, a0 in inits.items():
            if time.monotonic() > phase1_deadline:
                break

            a0 = np.maximum(a0, 0).astype(np.float64)
            s = a0.sum()
            if s < 1e-10:
                continue
            a0 = a0 / s
            z0 = np.log(a0 + 1e-30)

            t_levels = [5, 15, 40, 100]
            t_alloc = min(time_per, phase1_deadline - time.monotonic())
            if t_alloc < 2:
                break

            _, a_opt, _ = anneal_optimize(z0, t_levels, maxiter_per_level=40, time_limit=t_alloc)

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Scan: n={n}, {name}, score={score:.6f}")

            scan_results.append((score, n, name, a_opt))

    # ===================== PHASE 2: Deep Optimization =====================
    print(f"Phase 2: Deep optimization. Best so far: {best_score:.6f}")
    scan_results.sort(key=lambda x: x[0] if np.isfinite(x[0]) else 1e10)

    top_configs = []
    seen_keys = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if np.isfinite(score) and key not in seen_keys:
            top_configs.append((n, name, a_opt))
            seen_keys.add(key)
        if len(top_configs) >= 12:
            break

    # Ensure diversity in n
    for score, n, name, a_opt in scan_results:
        if np.isfinite(score) and n not in set(tc[0] for tc in top_configs):
            top_configs.append((n, name, a_opt))
        if len(top_configs) >= 16:
            break

    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best', best_sequence.copy()))

    phase2_deadline = time.monotonic() + min(800.0, (deadline - time.monotonic()) * 0.45)
    time_per_deep = max(20.0, (phase2_deadline - time.monotonic()) / max(1, len(top_configs)))

    for i, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > phase2_deadline - 30:
            break

        a0 = np.maximum(a0, 0).astype(np.float64)
        s = a0.sum()
        if s < 1e-10:
            continue
        a0 = a0 / s
        z0 = np.log(a0 + 1e-30)

        t_alloc = min(time_per_deep, phase2_deadline - time.monotonic() - 30)
        if t_alloc < 10:
            break

        # Extended annealing
        t_levels = [3, 8, 20, 50, 120, 300, 700, 1500]
        _, a_opt, _ = anneal_optimize(z0, t_levels, maxiter_per_level=100, time_limit=t_alloc * 0.75)

        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Deep: n={n}, {name}, score={score:.6f}")

        # Also try a perturbed restart from the result
        if time.monotonic() < phase2_deadline - 30:
            a_pert = perturbed_restart(a_opt, rng, sigma=0.15)
            z_pert = np.log(a_pert + 1e-30)
            t_levels2 = [20, 60, 200, 600]
            t_alloc2 = min(30.0, phase2_deadline - time.monotonic() - 30)
            if t_alloc2 > 5:
                _, a_opt2, _ = anneal_optimize(z_pert, t_levels2, maxiter_per_level=60, time_limit=t_alloc2)
                score2 = float(evaluate_sequence(a_opt2.tolist()))
                if np.isfinite(score2) and score2 < best_score:
                    best_score = score2
                    best_sequence = a_opt2.copy()
                    print(f"  Deep-pert: n={n}, {name}, score={score2:.6f}")

    # ===================== PHASE 3: Polish & Escape =====================
    if best_sequence is not None:
        print(f"Phase 3: Polish & escape. Current best: {best_score:.6f}")
        n = len(best_sequence)

        # 3a: LP polish
        if time.monotonic() < deadline - 60:
            t_lp = min(120.0, (deadline - time.monotonic() - 60) * 0.3)
            a_lp = lp_polish(best_sequence, time_limit=t_lp)
            score = float(evaluate_sequence(a_lp.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_lp.copy()
                print(f"  LP polish: score={score:.6f}")

        # 3b: Very high temperature L-BFGS (near-exact max)
        if time.monotonic() < deadline - 30:
            a0 = np.maximum(best_sequence, 0).astype(np.float64)
            s = a0.sum()
            if s > 1e-10:
                a0 = a0 / s
                z0 = np.log(a0 + 1e-30)
                t_levels = [500, 1000, 2000, 5000]
                t_alloc = min(120.0, deadline - time.monotonic() - 30)
                _, a_opt, _ = anneal_optimize(z0, t_levels, maxiter_per_level=200, time_limit=t_alloc)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  High-t L-BFGS: score={score:.6f}")

        # 3c: Multiple random perturbations + re-optimize
        if time.monotonic() < deadline - 20:
            remaining = deadline - 20 - time.monotonic()
            n_restarts = max(2, int(remaining / 20))
            for r in range(n_restarts):
                if time.monotonic() > deadline - 20:
                    break
                sigma = 0.05 + 0.1 * (r % 4)
                a_pert = perturbed_restart(best_sequence, rng, sigma=sigma)
                z_pert = np.log(a_pert + 1e-30)
                t_levels = [30, 100, 400, 1200]
                t_single = min(40.0, (deadline - time.monotonic() - 20) / max(1, n_restarts - r))
                _, a_opt, _ = anneal_optimize(z_pert, t_levels, maxiter_per_level=80, time_limit=t_single)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Restart {r}: sigma={sigma:.2f}, score={score:.6f}")

        # 3d: Final LP polish on the absolute best
        if time.monotonic() < deadline - 10:
            t_lp2 = min(60.0, deadline - time.monotonic() - 10)
            a_final = lp_polish(best_sequence, time_limit=t_lp2)
            score = float(evaluate_sequence(a_final.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_final.copy()
                print(f"  Final LP: score={score:.6f}")

    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_sequence)})")
    return best_sequence


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Uses smooth L-BFGS with annealed log-sum-exp temperature
    on a softmax parameterization, combined with LP-based polishing and
    random restarts to escape local minima.
    """
    result = run_optimization(seed=seed, budget_s=budget_s)
    return [float(x) for x in result]
```
