    # Let me just build it directly
    # For efficiency, note that A[k,i] depends only on |k-i| in a specific way.
    # Actually A[k,i] = a[k-i] (if valid) + a[i-k] (if valid)
    
    # Build using the cross-convolution structure
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx_parts = []
    data_parts = []
    
    for i in range(n):
        # A[k, i] = a[k-i] + a[i-k] for k = 0, ..., 2n-2
        # a[k-i] is nonzero when 0 <= k-i < n, i.e., i <= k <= i+n-1
        # a[i-k] is nonzero when 0 <= i-k < n, i.e., i-n+1 <= k <= i
        
        # For a[k-i]: k ranges from i to i+n-1, values are a[0], a[1], ..., a[n-1]
        ks1 = np.arange(i, i + n, dtype=np.int32)
        ks1 = ks1[(ks1 >= 0) & (ks1 < n_rows)]
        vals1 = a[np.clip(ks1 - i, 0, n-1)]
        row_idx_parts.append(ks1)
        data_parts.append(vals1)
        
        # For a[i-k]: k ranges from max(0, i-n+1) to i, values are a[n-1-(i-k)]... 
        # a[i-k] where i-k ranges from 0 to n-1, so k ranges from i to i-(n-1) = i-n+1
        # k from max(0, i-n+1) to i
        ks2_start = max(0, i - n + 1)
        ks2 = np.arange(ks2_start, i + 1, dtype=np.int32)
        # a[i-k] for k in ks2: i-k ranges from i-ks2_start to i-i = 0
        # So a[i-k] = a[arange(i-ks2_start, -1, -1)]
        idx2 = i - ks2
        vals2 = a[idx2]
        row_idx_parts.append(ks2)
        data_parts.append(vals2)
    
    row_idx = np.concatenate(row_idx_parts)
    data = np.concatenate(data_parts)
    
    # Handle duplicate entries (where both a[k-i] and a[i-k] contribute to the same (k,i))
    A = csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))
    A.sum_duplicates()
    
    return A, aa


def solve_lp_cross(a, M, timeout=30.0):
    """Original LP: maximize sum(g) s.t. g>=0, a*g <= M."""
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a, mode='cross')
    c = -np.ones(n)
    b_ub = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception as e:
        print(f"    LP error: {e}")
    return None


def solve_lp_linearized(a, M, timeout=30.0):
    """Improved LP: maximize sum(g) s.t. g>=0, linearized(g*g) <= M at point a."""
    n = len(a)
    A, aa = build_linearized_conv_matrix(a)
    c = -np.ones(n)
    # Constraint: A @ g <= M - (a*a)[k] + sum_i A[k,i]*a[i]
    # = M - aa[k] + (A @ a)[k]
    offset = A @ a
    b_ub = M - aa + offset  # = M - aa + A@a
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception as e:
        print(f"    LP error: {e}")
    return None


def lp_refine_improved(a, time_limit_s=600.0, max_iters=500, verbose=True, use_linearized=True):
    """LP-based iterative refinement with improved constraints."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    stagnant = 0
    it = 0
    
    if verbose:
        print(f"    LP start: score={best_score:.6f}, n={n}")
    
    for it in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        
        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max() / (a.sum() ** 2)
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 20 == 0 or it < 5):
                print(f"    LP iter {it:4d}: score={score:.6f}")
        else:
            stagnant += 1
        
        if stagnant > 60:
            if verbose:
                print(f"    LP stagnant at iter {it}, best={best_score:.6f}")
            break
        
        M = conv.max()
        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break
        timeout = max(2.0, min(30.0, remaining * 0.2))
        
        if use_linearized:
            g = solve_lp_linearized(a, M, timeout=timeout)
        else:
            g = solve_lp_cross(a, M, timeout=timeout)
        
        if g is None:
            break
        
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            if verbose:
                print(f"    LP no improvement at iter {it}")
            break
        
        g = g / sum_g
        
        # Adaptive step size
        alpha = min(0.5, max(0.03, 2.0 / (1.0 + 0.2 * it + stagnant)))
        
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = normalize_nonneg(a_new)
        
        # Enforce symmetry (optional)
        # a_new = 0.5 * (a_new + a_new[::-1])
        # a_new = normalize_nonneg(a_new)
        
        new_score = score_sequence(a_new)
        if new_score <= score + 1e-12:
            a = a_new
        else:
            # Try smaller step
            for frac in [0.5, 0.25, 0.1]:
                a_try = (1.0 - alpha*frac) * a + alpha*frac * g
                a_try = normalize_nonneg(a_try)
                if score_sequence(a_try) < score:
                    a = a_try
                    break
            else:
                a = (1.0 - 0.01) * a + 0.01 * g
                a = normalize_nonneg(a)
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"    LP done: best={best_score:.6f} after {it+1} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def freq_domain_refine(a, time_limit_s=120.0, verbose=True):
    """Frequency-domain iterative refinement (Gerchberg-Saxton style)."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    # Use FFT size that's a power of 2 for efficiency
    N = 1
    while N < 2 * n - 1:
        N *= 2
    
    best_a = a.copy()
    best_score = score_sequence(a)
    
    if verbose:
        print(f"    FD start: score={best_score:.6f}, n={n}")
    
    # Target: flat convolution
    # DFT of flat sequence of length 2n-1 is concentrated at frequency 0
    m = np.arange(N)
    B_target = np.zeros(N, dtype=np.complex128)
    B_target[0] = 1.0  # Flat target in time = delta in freq
    # Actually, flat in time means B[m] = 0 for m != 0, B[0] = (2n-1)
    # Let's use a different target: slightly peaked at 0 to match the convolution structure
    # B_target[m] = exp(-m^2 / sigma_m^2) for a smooth target
    sigma_m = max(2.0, N / (4 * n))
    B_target = np.exp(-(m.astype(np.float64)**2) / (2 * sigma_m**2)).astype(np.complex128)
    
    it = 0
    stagnant = 0
    while time.monotonic() - start_time < time_limit_s:
        it += 1
        A = np.fft.fft(a, N)
        B = A * A  # B[m] = A[m]^2
        
        # Update: A_new = sqrt(B_target) * phase(A^2)
        # phase(A^2) = A^2 / |A^2| = (A/|A|)^2
        mask = np.abs(B) > 1e-30
        A_new = np.zeros(N, dtype=np.complex128)
        A_new[mask] = np.sqrt(B_target[mask]) * (B[mask] / np.abs(B[mask]))
        # For A[m] = 0, keep A_new[m] = 0
        A_new[~mask] = 0
        
        a_new = np.real(np.fft.ifft(A_new))[:n]
        a_new = np.maximum(0.0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Blend with current
        alpha = min(0.3, 1.0 / (1.0 + it * 0.01))
        a_candidate = (1.0 - alpha) * a + alpha * a_new
        a_candidate = normalize_nonneg(a_candidate)
        
        score = score_sequence(a_candidate)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_candidate.copy()
            a = a_candidate.copy()
            stagnant = 0
            if verbose and it % 50 == 0:
                print(f"    FD iter {it:4d}: score={score:.6f}")
        else:
            stagnant += 1
            a = a_candidate.copy()  # Still update to continue the iteration
        
        if stagnant > 200:
            if verbose:
                print(f"    FD stagnant at iter {it}, best={best_score:.6f}")
            break
    
    if verbose:
        print(f"    FD done: best={best_score:.6f} after {it} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def local_search_refine(a, time_limit_s=120.0, rng=None, verbose=True):
    """Local search with structured mutations."""
    if rng is None:
        rng = np.random.default_rng(42)
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    if verbose:
        print(f"    LS start: score={best_score:.6f}")
    
    accept_count = 0
    while time.monotonic() - start_time < time_limit_s:
        # Random choice of mutation type
        r = rng.random()
        candidate = a.copy()
        
        if r < 0.3:
            # Block scaling
            width = max(2, int(rng.integers(n // 200, n // 5)))
            start = int(rng.integers(0, max(1, n - width)))
            stop = min(n, start + width)
            candidate[start:stop] *= np.exp(rng.normal(0.0, 0.15, size=stop - start))
        elif r < 0.5:
            # Sinusoidal perturbation
            freq = rng.integers(1, n // 4 + 1)
            phase = rng.uniform(0, 2 * np.pi)
            amplitude = rng.uniform(0.01, 0.1)
            t = np.arange(n)
            perturbation = 1.0 + amplitude * np.sin(2 * np.pi * freq * t / n + phase)
            candidate *= perturbation
        elif r < 0.7:
            # Edge adjustment
            edge_width = max(2, n // 20)
            edge_factor = np.exp(rng.normal(0.0, 0.2))
            candidate[:edge_width] *= edge_factor
            candidate[-edge_width:] *= edge_factor
        elif r < 0.85:
            # Symmetric perturbation
            half = n // 2
            pert = np.exp(rng.normal(0.0, 0.1, size=half))
            candidate[:half] *= pert
            candidate[n-half:] *= pert[::-1]
        else:
            # Gaussian smoothing
            kernel_size = max(3, n // 50)
            kernel = np.exp(-0.5 * (np.arange(-kernel_size, kernel_size+1) / (kernel_size/3.0))**2)
            kernel /= kernel.sum()
            candidate = np.convolve(candidate, kernel, mode='same')
        
        # Enforce symmetry with some probability
        if rng.random() < 0.5:
            candidate = 0.5 * (candidate + candidate[::-1])
        
        candidate = normalize_nonneg(candidate)
        score = score_sequence(candidate)
        
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 30 == 0:
                print(f"    LS accept {accept_count}: score={best_score:.6f}")
    
    if verbose:
        print(f"    LS done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score
