    n = len(a)
    c = np.convolve(a, a)
    max_c = c.max()
    peak_indices = np.where(c >= max_c - 1e-10)[0]
    
    # Compute subgradient
    sens = np.zeros(n)
    for j in peak_indices:
        for k in range(n):
            i = j - k
            if 0 <= i < n:
                if j == 2*k:
                    sens[k] += a[i]
                else:
                    sens[k] += 2 * a[i]
    
    # Normalize sensitivity
    active = a > 1e-10
    if not active.any():
        return a
    avg_sens = sens[active].mean()
    
    # Multiplicative update
    a_new = a.copy()
    a_new[active] *= (avg_sens / (sens[active] + 1e-15)) ** eta
    a_new = np.maximum(0, a_new)
    a_new /= a_new.sum()
    return a_new

Hybrid optimization for minimizing peak auto-convolution of non-negative sequences.
Uses smoothed gradient descent (log-sum-exp), multiplicative updates, and LP refinement.
"""

import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def compute_smoothed_gradient(a, beta):
    """Compute gradient of log-sum-exp smoothed max of auto-convolution."""
    n = len(a)
    c = np.convolve(a, a)
    # Log-sum-exp for stability
    max_c = c.max()
    exp_c = np.exp(beta * (c - max_c))
    w = exp_c / exp_c.sum()
    
    # grad[k] = sum_j w[j] * dc[j]/da[k]
    # dc[j]/da[k] = 2*a[j-k] (if 0<=j-k<n and j!=2k) or a[k] (if j=2k)
    grad = np.zeros(n)
    for k in range(n):
        # Valid j range: j from k to min(k+n-1, 2n-2), and j-k in [0, n-1]
        j_start = k
        j_end = min(2 * n - 1, k + n)
        if j_start >= j_end:
            continue
        i_end = j_end - j_start  # = min(n, 2n-1-k)
        # Main term: 2 * sum w[j] * a[j-k] for j in [j_start, j_end)
        grad[k] = 2.0 * np.dot(w[j_start:j_end], a[:i_end])
        # Correction: subtract a[k]*w[2k] (since factor should be 1, not 2, when j=2k)
        if 2 * k < 2 * n - 1 and 2 * k >= j_start:
            i_val = 2 * k - k  # = k
            if i_val < i_end:
                grad[k] -= w[2 * k] * a[i_val]
    return grad, c.max()


def smoothed_grad_descent(a, time_limit=60.0, beta_start=50.0, beta_end=500.0, 
                           lr_start=0.5, lr_end=0.01, verbose=True):
    """Smoothed gradient descent with multiplicative updates."""
    start_time = time.monotonic()
    a = normalize_a(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    n_total_steps = 500
    stagnant = 0
    stagnant_threshold = 50
    
    for step in range(n_total_steps):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit:
            break
        
        progress = step / n_total_steps
        beta = beta_start * (beta_end / beta_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        
        grad, max_c = compute_smoothed_gradient(a, beta)
        
        # Multiplicative update
        # a_new[k] = a[k] * exp(-lr * (grad[k] - mean_grad_auc))
        active = a > 1e-10
        if not active.any():
            break
        
        # Use log-space for stability
        log_a = np.zeros(n)
        log_a[active] = np.log(a[active])
        
        # Center the gradient (to maintain sum)
        mean_grad = grad[active].mean()
        centered_grad = grad - mean_grad
        
        log_a[active] -= lr * centered_grad[active]
        
        # Clip to prevent overflow
        log_a = np.clip(log_a, -30, 30)
        a_new = np.exp(log_a)
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s
        
        score = score_sequence(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stagnant = 0
            if verbose and (step % 50 == 0 or step < 5):
                print(f"    SGD step {step}: score={score:.6f}, beta={beta:.1f}, lr={lr:.4f}")
        else:
            stagnant += 1
            # Try smaller step
            lr2 = lr * 0.3
            log_a[active] -= lr2 * centered_grad[active]
            log_a = np.clip(log_a, -30, 30)
            a_new = np.exp(log_a)
            a_new = np.maximum(0, a_new)
            s = a_new.sum()
            if s > 1e-15:
                a_new /= s
                score2 = score_sequence(a_new)
                if score2 < best_score:
                    best_score = score2
                    best_a = a_new.copy()
                    a = a_new.copy()
                    stagnant = 0
                else:
                    a = a_new.copy()  # still move a bit
            else:
                stagnant += 10
        
        a = a_new.copy()
        
        if stagnant > stagnant_threshold:
            if verbose:
                print(f"    SGD stagnant at step {step}, best={best_score:.6f}")
            break
    
    elapsed = time.monotonic() - start_time
    if verbose:
        print(f"    SGD done: best={best_score:.6f}, {elapsed:.1f}s, {step+1} steps")
    return best_a, best_score


def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_step(a, M, timeout=30.0):
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=60.0, verbose=True):
    start_time = time.monotonic()
    a = normalize_a(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    stagnant = 0
    
    for it in range(1000):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        conv = np.convolve(a, a)
        M = conv.max()
        score = 2.0 * n * M / (a.sum() ** 2)
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        if stagnant > 30:
            break
        
        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break
        timeout = max(2.0, min(30.0, remaining * 0.3))
        g = solve_lp_step(a, M, timeout=timeout)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            break
        g = g / sum_g
        
        # Adaptive step size
        alpha = 0.3
        for _ in range(5):
            a_new = normalize_a((1.0 - alpha) * a + alpha * g)
            new_score = score_sequence(a_new)
            if new_score <= score + 1e-12:
                a = a_new
                break
            alpha *= 0.5
        else:
            a = normalize_a((1.0 - alpha) * a + alpha * g)
    
    elapsed = time.monotonic() - start_time
    if verbose:
        print(f"    LP done: best={best_score:.6f}, {elapsed:.1f}s")
    return best_a, best_score


def local_search(a, time_limit=60.0, rng=None, verbose=False):
    if rng is None:
        rng = np.random.default_rng(42)
    start_time = time.monotonic()
    a = normalize_a(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    while time.monotonic() - start_time < time_limit:
        # Random block mutation
        width = int(rng.integers(max(2, n // 100), max(3, n // 5)))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        candidate = a.copy()
        candidate[start:stop] *= np.exp(rng.normal(0.0, 0.15, size=stop - start))
        candidate = normalize_a(candidate)
        score = score_sequence(candidate)
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
    
    if verbose:
        print(f"    LS done: best={best_score:.6f}")
    return best_a, best_score


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def random_smooth_init(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 10), max(2, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s

"""
Hybrid optimization for minimizing peak auto-convolution of non-negative sequences.
Uses FFT-accelerated smoothed gradient descent (log-sum-exp), LP refinement, and local search.
The key innovation is an O(n log n) gradient computation via FFT for the smoothed max-convolution objective.
"""

import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def score_sequence(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize_a(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def compute_grad_fft(a, beta):
    """
    Compute gradient of log-sum-exp smoothed max of auto-convolution using FFT.
    Returns (gradient, max_conv, weights).
    """
    n = len(a)
    c = np.convolve(a, a)  # length 2n-1
    # Softmax weights for stability
    max_c = c.max()
    bc = beta * (c - max_c)
    # Clip to prevent overflow
    bc = np.clip(bc, -100, 100)
    exp_bc = np.exp(bc)
    w = exp_bc / exp_bc.sum()
    
    # G[k] = sum_j w[j] * a[j-k] for valid j-k in [0, n-1]
    # This equals np.convolve(w, a[::-1], 'full')[n-1 : n-1+n]
    G = np.convolve(w, a[::-1], 'full')
    G_relevant = G[n - 1: n - 1 + n]
    
    # grad[k] = 2*G[k] - w[2k]*a[k]
    grad = 2.0 * G_relevant.copy()
    # Correction term: w[2k] * a[k] for 2k < 2n-1
    idx = 2 * np.arange(n)
    valid = idx < (2 * n - 1)
    grad[valid] -= w[idx[valid]] * a[valid]
    
    return grad, c.max(), w


def smoothed_gradient_descent(a, time_limit=60.0, beta_start=30.0, beta_end=300.0,
                               lr_start=1.0, lr_end=0.05, max_iters=3000, verbose=True):
    """
    Smoothed gradient descent with multiplicative (mirror) updates on the simplex.
    Minimizes log-sum-exp(beta * conv(a,a)) which approximates max(conv(a,a)).
    """
    start_time = time.monotonic()
    a = normalize_a(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    stagnant = 0
    stagnant_threshold = 200
    n_steps = 0
    
    for step in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit:
            break
        
        n_steps = step
        progress = min(1.0, step / max(1, max_iters * 0.7))
        beta = beta_start * (beta_end / beta_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        
        grad, max_c, w = compute_grad_fft(a, beta)
        
        # Multiplicative update in log-space
        active = a > 1e-12
        if not active.any():
            break
        
        # Center gradient for simplex stability
        mean_grad = grad[active].mean()
        centered = grad - mean_grad
        
        # Scale learning rate by gradient magnitude
        grad_range = grad[active].max() - grad[active].min()
        if grad_range < 1e-15:
            grad_range = 1.0
        effective_lr = lr / grad_range
        
        # Update in log space
        log_a = np.full(n, -30.0)
        log_a[active] = np.log(a[active])
        log_a[active] -= effective_lr * centered[active]
        log_a = np.clip(log_a, -40.0, 40.0)
        
        a_new = np.exp(log_a)
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            a = a  # no update
            stagnant += 1
            continue
        a_new /= s
        
        score = score_sequence(a_new)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            a = a_new.copy()
            stagnant = 0
            if verbose and (step % 200 == 0 or step < 5):
                print(f"    SGD [{elapsed:6.1f}s] step {step:5d}: score={score:.6f}, beta={beta:.1f}")
        else:
            # Half step
            log_a2 = np.full(n, -30.0)
            log_a2[active] = np.log(a[active])
            log_a2[active] -= effective_lr * 0.3 * centered[active]
            log_a2 = np.clip(log_a2, -40.0, 40.0)
            a_half = np.exp(log_a2)
            a_half = np.maximum(0, a_half)
            sh = a_half.sum()
            if sh > 1e-15:
                a_half /= sh
                score_h = score_sequence(a_half)
                if score_h < best_score:
                    best_score = score_h
                    best_a = a_half.copy()
                a = a_half.copy()
            stagnant += 1
        
        if stagnant > stagnant_threshold:
            if verbose:
                print(f"    SGD stagnant at step {step}, best={best_score:.6f}")
            break
    
    elapsed = time.monotonic() - start_time
    if verbose:
        print(f"    SGD done: best={best_score:.6f}, {elapsed:.1f}s, {n_steps+1} steps")
    return best_a, best_score


def build_conv_matrix(a):
    """Build sparse matrix for convolution constraints in LP."""
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_step(a, M, timeout=30.0):
    """Solve the LP: max sum(g) s.t. a*g <= M for all lags, g >= 0."""
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            x = np.maximum(0.0, res.x)
            if x.sum() > 0:
                return x
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=60.0, verbose=True):
    """LP-based iterative refinement."""
    start_time = time.monotonic()
    a = normalize_a(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    stagnant = 0
    
    it = 0
    for it in range(500):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        conv = np.convolve(a, a)
        M = conv.max()
        score = 2.0 * n * M / (a.sum() ** 2)
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 10 == 0 or it < 3):
                print(f"    LP [{elapsed:6.1f}s] iter {it:4d}: score={score:.6f}")
        else:
            stagnant += 1
        
        if stagnant > 30:
            break
        
        remaining = time_limit_s - elapsed
        if remaining < 8.0:
            break
        timeout = max(3.0, min(45.0, remaining * 0.4))
        g = solve_lp_step(a, M, timeout=timeout)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-8:
            break
        g = g / sum_g
        
        # Adaptive interpolation
        alpha = 0.4
        improved = False
        for trial in range(6):
            a_new = normalize_a((1.0 - alpha) * a + alpha * g)
            new_score = score_sequence(a_new)
            if new_score <= score + 1e-12:
                a = a_new
                improved = True
                break
            alpha *= 0.5
        if not improved:
            a = normalize_a((1.0 - alpha) * a + alpha * g)
    
    elapsed = time.monotonic() - start_time
    if verbose:
        print(f"    LP done: best={best_score:.6f}, {elapsed:.1f}s, {it+1} iters")
    return best_a, best_score


def local_search(a, time_limit=30.0, rng=None, verbose=False):
    """Random block-mutation local search."""
    if rng is None:
        rng = np.random.default_rng(123)
    start_time = time.monotonic()
    a = normalize_a(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    count = 0
    
    while time.monotonic() - start_time < time_limit:
        # Multiple mutation strategies
        strategy = rng.integers(0, 3)
        candidate = a.copy()
        
        if strategy == 0:
            # Block scaling
            width = int(rng.integers(max(2, n // 100), max(3, n // 4)))
            start = int(rng.integers(0, max(1, n - width)))
            stop = min(n, start + width)
            candidate[start:stop] *= np.exp(rng.normal(0.0, 0.2, size=stop - start))
        elif strategy == 1:
            # Smooth perturbation
            pert = rng.normal(0.0, 0.05, size=n)
            # Smooth it
            kernel = np.ones(5) / 5.0
            pert = np.convolve(pert, kernel, mode='same')
            candidate *= np.exp(pert)
        else:
            # Symmetric perturbation
            width = int(rng.integers(max(2, n // 200), max(3, n // 20)))
            start = int(rng.integers(0, max(1, n - width)))
            stop = min(n, start + width)
            pert = rng.normal(0.0, 0.1, size=stop - start)
            candidate[start:stop] *= np.exp(pert)
            candidate[stop - (stop - start):: -1] = candidate[start:stop]  # mirror
        
        candidate = normalize_a(candidate)
        score = score_sequence(candidate)
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            count += 1
    
    if verbose:
        print(f"    LS done: best={best_score:.6f}, {count} accepts")
    return best_a, best_score


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_blackman(n):
    if n == 1:
        return np.ones(1)
    t = np.arange(n, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def initial_super_gaussian(n, p=4.0):
    """Super-Gaussian: exp(-|x|^p), wider than Gaussian for p>2."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * 0.35
    x = (t - center) / width
    return np.exp(-np.abs(x) ** p)


def random_smooth_init(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 10), max(2, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Hybrid search: FFT-accelerated smoothed gradient descent + LP + local search.
    Tries multiple sequence lengths and initialization strategies.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + max(30.0, budget_s - 25.0)
    
    best_score = float('inf')
    best_seq = None
    
    # Try warm start from height_sequence_1
    warm_start = None
    try:
        hs = globals().get('height_sequence_1', None)
        if hs is not None:
            warm_arr = np.array(hs, dtype=np.float64)
            warm_arr = np.maximum(0, warm_arr)
            if warm_arr.sum() > 0.01:
                warm_start = warm_arr
                ws_score = score_sequence(warm_start)
                print(f"Warm start: n={len(warm_start)}, score={ws_score:.6f}")
    except Exception:
        pass
    
    # Build config list: (n, init_type, time_fraction)
    configs = []
    
    if warm_start is not None:
        configs.append((len(warm_start), 'warm', 0.22))
    
    # Focus on lengths that tend to work well
    configs.extend([
        (500, 'raised_cosine', 0.10),
        (500, 'super_gaussian', 0.08),
        (750, 'raised_cosine', 0.08),
        (1000, 'raised_cosine', 0.10),
        (1000, 'super_gaussian', 0.08),
        (1500, 'raised_cosine', 0.10),
        (2000, 'raised_cosine', 0.10),
        (2000, 'super_gaussian', 0.08),
        (500, 'gaussian', 0.06),
        (1000, 'gaussian', 0.06),
        (500, 'blackman', 0.05),
        (1000, 'blackman', 0.05),
        (500, 'random', 0.04),
        (1000, 'random', 0.04),
    ])
    
    for n, init_type, time_frac in configs:
        remaining = deadline - time.monotonic()
        if remaining < 40.0:
            break
        time_limit = min(time_frac * budget_s, remaining * 0.85)
        if time_limit < 20.0:
            continue
        
        print(f"\n=== n={n}, init={init_type}, budget={time_limit:.0f}s ===")
        
        # Generate initial sequence
        if init_type == 'warm' and warm_start is not None:
            a = normalize_a(warm_start.copy())
        elif init_type == 'raised_cosine':
            a = normalize_a(initial_raised_cosine(n))
        elif init_type == 'gaussian':
            a = normalize_a(initial_gaussian(n, sigma_frac=0.3))
        elif init_type == 'blackman':
            a = normalize_a(initial_blackman(n))
        elif init_type == 'super_gaussian':
            a = normalize_a(initial_super_gaussian(n, p=4.0))
        elif init_type == 'random':
            a = normalize_a(random_smooth_init(n, rng, n_bumps=int(rng.integers(2, 5))))
        else:
            a = normalize_a(initial_raised_cosine(n))
        
        init_score = score_sequence(a)
        print(f"    Init: score={init_score:.6f}")
        
        # Phase 1: Smoothed gradient descent (fast, many iterations)
        t1 = time_limit * 0.45
        a1, s1 = smoothed_gradient_descent(a, time_limit=t1,
                                            beta_start=30.0, beta_end=300.0,
                                            lr_start=1.0, lr_end=0.05,
                                            max_iters=5000, verbose=True)
        if s1 < best_score:
            best_score = s1
            best_seq = a1.copy()
            print(f"    After SGD: {s1:.6f} *** BEST ***")
        
        # Phase 2: LP refinement (slower, fewer iterations)
        remaining2 = deadline - time.monotonic()
        if remaining2 > 30:
            t2 = min(time_limit * 0.35, remaining2 * 0.7)
            a2, s2 = lp_refine(a1, time_limit_s=t2, verbose=True)
            if s2 < best_score:
                best_score = s2
                best_seq = a2.copy()
                print(f"    After LP: {s2:.6f} *** BEST ***")
            else:
                a2 = a1
        
        # Phase 3: Second SGD pass (higher beta for fine-tuning)
        remaining3 = deadline - time.monotonic()
        if remaining3 > 20:
            t3 = min(time_limit * 0.15, remaining3 * 0.7)
            a3, s3 = smoothed_gradient_descent(a2, time_limit=t3,
                                                beta_start=100.0, beta_end=500.0,
                                                lr_start=0.5, lr_end=0.02,
                                                max_iters=3000, verbose=False)
            if s3 < best_score:
                best_score = s3
                best_seq = a3.copy()
                print(f"    After SGD2: {s3:.6f} *** BEST ***")
            else:
                a3 = a2
        
        # Phase 4: Local search polish
        remaining4 = deadline - time.monotonic()
        if remaining4 > 10:
            t4 = min(time_limit * 0.05, remaining4 * 0.6)
            a4, s4 = local_search(best_seq, time_limit=t4, rng=rng, verbose=False)
            if s4 < best_score:
                best_score = s4
                best_seq = a4.copy()
                print(f"    After LS: {s4:.6f} *** BEST ***")
    
    # === Final polishing phase ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 90:
            print(f"\n=== FINAL POLISH (remaining={remaining:.0f}s) ===")
            
            # LP on best
            t_lp = remaining * 0.4
            a_pol, s_pol = lp_refine(best_seq, time_limit_s=t_lp, verbose=True)
            if s_pol < best_score:
                best_score = s_pol
                best_seq = a_pol.copy()
                print(f"  Final LP: {s_pol:.6f}")
            
            # High-beta SGD
            remaining2 = deadline - time.monotonic()
            if remaining2 > 40:
                a_pol2, s_pol2 = smoothed_gradient_descent(best_seq, time_limit=remaining2 * 0.5,
                                                            beta_start=200.0, beta_end=800.0,
                                                            lr_start=0.3, lr_end=0.01,
                                                            max_iters=5000, verbose=True)
                if s_pol2 < best_score:
                    best_score = s_pol2
                    best_seq = a_pol2.copy()
                    print(f"  Final SGD: {s_pol2:.6f}")
            
            # Final LP
            remaining3 = deadline - time.monotonic()
            if remaining3 > 30:
                a_pol3, s_pol3 = lp_refine(best_seq, time_limit_s=remaining3 * 0.6, verbose=True)
                if s_pol3 < best_score:
                    best_score = s_pol3
                    best_seq = a_pol3.copy()
                    print(f"  Final LP2: {s_pol3:.6f}")
            
            # Local search
            remaining4 = deadline - time.monotonic()
            if remaining4 > 15:
                a_pol4, s_pol4 = local_search(best_seq, time_limit=remaining4 * 0.7,
                                               rng=rng, verbose=True)
                if s_pol4 < best_score:
                    best_score = s_pol4
                    best_seq = a_pol4.copy()
                    print(f"  Final LS: {s_pol4:.6f}")
    
    if best_seq is None:
        print("No valid result found, using fallback")
        best_seq = initial_raised_cosine(500)
        best_score = score_sequence(best_seq)
    
    best_seq = np.array(best_seq, dtype=np.float64)
    best_seq = np.maximum(0.0, best_seq)
    # Clamp to [0, 1000] as per evaluator
    best_seq = np.minimum(1000.0, best_seq)
    
    print(f"\n{'='*60}")
    print(f"FINAL: score={best_score:.6f}, n={len(best_seq)}")
    print(f"{'='*60}")
    
    # Verify with actual evaluator
    try:
        verify = evaluate_sequence([float(x) for x in best_seq])
        print(f"Verified via evaluate_sequence: {verify:.6f}")
    except Exception as e:
        print(f"Verification error: {e}")
    
    print(f"Total time: {time.monotonic() - total_start:.1f}s")
    
    return [float(x) for x in best_seq]
