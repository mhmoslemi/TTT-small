from scipy.optimize import linprog

def solve_lp(f, M):
    n = len(f)
    # c = -1 (we want to maximize sum(g), so minimize -sum(g))
    c = -np.ones(n)
    
    # Constraint matrix: A_ub @ g <= b_ub
    # For each lag k (0 to 2n-2), sum_j f(j)*g(k-j) <= M
    # A[k, m] = f(k-m) if 0 <= k-m < n, else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for m in range(n):
            j = k - m
            if 0 <= j < n:
                A[k, m] = f[j]
    
    b = M * np.ones(2*n-1)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    return res.x

"""
LP-based iterative optimization for minimizing the peak of the self-convolution
of non-negative step functions. Uses the linear programming approach to find
the "flattest" non-negative sequence under a cross-convolution constraint,
combined with blending, adaptive step sizes, multi-start, and coordinate descent
post-processing.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]


def score_fast(a):
    """Compute the score (same as evaluate_sequence but faster)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def build_constraint_matrix(f):
    """Build the sparse constraint matrix for the LP.
    A[k, m] = f[k-m] if 0 <= k-m < n, else 0.
    Shape: (2n-1, n)
    """
    n = len(f)
    rows = []
    cols = []
    data = []
    for m in range(n):
        # For column m, A[k, m] = f[k-m] for k-m in [0, n-1], i.e., k in [m, m+n-1]
        k_start = m
        k_end = m + n - 1
        j_start = 0
        for k in range(k_start, k_end + 1):
            rows.append(k)
            cols.append(m)
            data.append(f[k - m])
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return A


def solve_lp(f, M, method='highs'):
    """Solve the LP: maximize sum(g) subject to f*g <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)  # minimize -sum(g)
    A = build_constraint_matrix(f)
    b = M * np.ones(2*n - 1)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method=method)
    if res.success:
        return res.x
    return None


def lp_iteration(f, t, max_lp_attempts=3):
    """
    One LP-based iteration.
    Returns (f_new, improved) where improved is True if peak decreased.
    """
    n = len(f)
    conv = fft_convolve(f, f)
    M = np.max(conv)
    
    g = solve_lp(f, M)
    if g is None:
        return f, False
    
    # Normalize g to have same sum as f
    sum_g = np.sum(g)
    if sum_g < 1e-10:
        return f, False
    sum_f = np.sum(f)
    g = g * (sum_f / sum_g)
    
    # Blend
    f_new = (1 - t) * f + t * g
    
    # Check if peak decreased
    conv_new = fft_convolve(f_new, f_new)
    M_new = np.max(conv_new)
    
    if M_new < M:
        return f_new, True
    else:
        # Try smaller step
        for t_attempt in [t * 0.5, t * 0.25, t * 0.1]:
            f_try = (1 - t_attempt) * f + t_attempt * g
            conv_try = fft_convolve(f_try, f_try)
            if np.max(conv_try) < M:
                return f_try, True
        return f, False


def optimize_lp(f_init, time_limit, initial_t=0.3, rng=None):
    """
    Run LP-based optimization until time limit.
    """
    f = f_init.astype(np.float64).copy()
    n = len(f)
    t0 = time.monotonic()
    
    # Normalize
    s = np.sum(f)
    if s < 1e-10:
        return f
    f = f * (np.sqrt(2.0 * n) / s)
    
    current_score = score_fast(f)
    best_f = f.copy()
    best_score = current_score
    t = initial_t
    stagnation = 0
    
    while time.monotonic() - t0 < time_limit:
        f_new, improved = lp_iteration(f, t)
        
        if improved:
            f = f_new
            new_score = score_fast(f)
            if new_score < best_score:
                best_score = new_score
                best_f = f.copy()
            stagnation = 0
        else:
            stagnation += 1
            # Reduce step size and restart
            t = max(0.01, t * 0.8)
            if stagnation > 20:
                # Try a random perturbation to escape
                if rng is not None:
                    f = f * np.exp(rng.normal(0, 0.05, n))
                    f = np.clip(f, 0, None)
                    s = np.sum(f)
                    if s > 0:
                        f = f * (np.sqrt(2.0 * n) / s)
                    stagnation = 0
                    t = initial_t
                else:
                    break
        
        if time.monotonic() - t0 > time_limit - 5:
            break
    
    return best_f


def coordinate_descent(a, time_limit, rng):
    """
    Coordinate descent: for each coordinate, find the optimal value
    that minimizes the peak of the convolution.
    """
    n = len(a)
    a = a.copy()
    current_score = score_fast(a)
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit - 1:
                break
            
            a_old = a[i]
            # Try a few candidate values
            candidates = [
                a_old * 0.5,
                a_old * 0.75,
                a_old * 1.25,
                a_old * 1.5,
                a_old + 0.1 * a.mean(),
                max(0, a_old - 0.1 * a.mean()),
            ]
            
            best_val = a_old
            best_local_score = current_score
            
            for val in candidates:
                val = max(0.0, val)
                if val == a_old:
                    continue
                a[i] = val
                s = score_fast(a)
                if s < best_local_score:
                    best_local_score = s
                    best_val = val
            
            a[i] = best_val
            if best_val != a_old:
                current_score = best_local_score
                improved = True
                if current_score < best_score:
                    best_score = current_score
                    best_a = a.copy()
        
        if not improved:
            # Try random perturbations
            idx = rng.integers(0, n)
            a[idx] *= np.exp(rng.normal(0, 0.1))
            a[idx] = max(0, a[idx])
    
    return best_a, best_score


def make_initial_sequence(n, init_type, rng):
    """Generate an initial sequence."""
    if init_type == 'random':
        a = 1.0 + 0.3 * rng.standard_normal(n)
    elif init_type == 'chirp':
        t = np.arange(n, dtype=np.float64)

        a = np.exp(-0.5 * ((t - center) / width) ** 2)
    else:
        a = np.ones(n)
    
    return np.clip(a, 0.01, None)

"""
LP-based iterative optimization for minimizing the normalized peak of self-convolution
of non-negative sequences. Uses linear programming to find the flattest non-negative
sequence under a cross-convolution constraint, combined with adaptive blending,
multi-start over lengths and initializations, and coordinate descent polishing.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute the evaluation score (matches evaluate_sequence)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def build_spA(f):
    """Build sparse constraint matrix for LP: A[k,m] = f[k-m] if 0<=k-m<n."""
    n = len(f)
    row_idx = (np.arange(n)[:, None] + np.arange(n)[None, :]).ravel()
    col_idx = np.tile(np.arange(n), n)
    data = np.tile(f, n)
    return csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. f*g <= M, g >= 0."""
    n = len(f)
    c_obj = -np.ones(n)
    A = build_spA(f)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def lp_iteration(f, M, t_init=0.1):
    """
    One LP-based iteration. Returns (f_new, M_new, success).
    """
    n = len(f)
    S = np.sum(f)
    g = solve_lp(f, M)
    if g is None or np.sum(g) < 1e-12:
        return f, M, False
    g = g * (S / np.sum(g))

    # Find max safe step via bisection on t
    lo, hi = 0.0, min(1.0, t_init * 3)

    def peak_at(t):
        f_try = (1 - t) * f + t * g
        return np.max(fft_convolve(f_try, f_try))

    if peak_at(hi) <= M:
        t_best = hi
    else:
        for _ in range(10):
            mid = (lo + hi) / 2
            if peak_at(mid) <= M:
                lo = mid
            else:
                hi = mid
        t_best = lo

    if t_best < 1e-5:
        # Try a tiny step
        t_best = 1e-4
        f_try = (1 - t_best) * f + t_best * g
        M_try = np.max(fft_convolve(f_try, f_try))
        if M_try >= M:
            return f, M, False
        f_new = f_try
        M_new = M_try
    else:
        f_new = (1 - t_best) * f + t_best * g
        M_new = np.max(fft_convolve(f_new, f_new))
        if M_new >= M:
            # Fallback: very small step
            t_best = 1e-5
            f_new = (1 - t_best) * f + t_best * g
            M_new = np.max(fft_convolve(f_new, f_new))
            if M_new >= M:
                return f, M, False

    f_new = np.maximum(f_new, 0)
    return f_new, M_new, True


def optimize_lp(f_init, time_limit, rng=None, verbose=True):
    """Run LP-based iterations until time limit."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    S = np.sqrt(2.0 * n)
    s = np.sum(f)
    if s < 1e-10:
        return f
    f = f * (S / s)

    conv = fft_convolve(f, f)
    M = np.max(conv)
    best_f = f.copy()
    best_M = M
    best_score = score_fast(f)
    t0 = time.monotonic()
    it = 0
    stagnation = 0
    t_init = 0.3

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit - 2:
            break

        f_new, M_new, success = lp_iteration(f, M, t_init)

        if success and M_new < M:
            f = f_new
            M = M_new
            it += 1
            stagnation = 0
            sc = 2.0 * n * M / (S * S)
            if sc < best_score:
                best_score = sc
                best_f = f.copy()
            if verbose and it % 50 == 0:
                print(f"      iter={it}, M={M:.6f}, score={sc:.6f}, t={elapsed:.1f}s")
        else:
            stagnation += 1
            t_init = max(0.02, t_init * 0.85)
            if stagnation > 30:

                    pert = np.exp(rng.normal(0, 0.1, n))
                    f = f * pert
                    f = np.maximum(f, 0)
                    f = f * (S / np.sum(f))
                    M = np.max(fft_convolve(f, f))
                    stagnation = 0
                    t_init = 0.3
                else:
                    break

    return best_f


def coordinate_descent_polish(a, time_limit, rng):
    """Polish with coordinate-wise multiplicative perturbations."""
    n = len(a)
    a = a.copy()
    cur_score = score_fast(a)
    best_a = a.copy()
    best_score = cur_score
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 1:
            break

        improved = False
        # Random subset of coordinates
        n_try = min(n, max(8, n // 4))
        indices = rng.choice(n, size=n_try, replace=False)

        for i in indices:
            if time.monotonic() - t0 > time_limit - 1:
                break
            old = a[i]
            if old < 1e-10:
                candidates = [0.01 * np.mean(a), 0.005 * np.mean(a)]
            else:
                candidates = [old * 0.5, old * 0.7, old * 1.3, old * 1.6,
                              max(0.0, old - 0.05 * np.mean(a)),
                              old + 0.05 * np.mean(a)]
            best_val = old
            best_s = cur_score
            for val in candidates:
                val = max(0.0, min(1000.0, val))
                a[i] = val
                s = score_fast(a)
                if s < best_s:
                    best_s = s
                    best_val = val
            a[i] = best_val
            if best_val != old:
                cur_score = best_s
                improved = True
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1

        if not improved:
            no_improve += 5
            if no_improve > 100:
                break

    return best_a, best_score


def make_initial(n, itype, rng):
    """Generate initial sequence of length n."""
    t = np.arange(n, dtype=np.float64)
    if itype == 'asymmetric':
        center = n * (0.3 + 0.4 * rng.random())
        width = n * (0.15 + 0.2 * rng.random())
        a = np.exp(-0.5 * ((t - center) / width) ** 2)
        a += 0.05 * np.ones(n)
    elif itype == 'bump':
        center = n / 2.0
        width = n * (0.15 + 0.25 * rng.random())
        a = np.exp(-0.5 * ((t - center) / width) ** 2)
        a += 0.05 * np.ones(n)
    elif itype == 'multi_bump':
        a = 0.1 * np.ones(n)
        for _ in range(rng.integers(2, 5)):
            c = rng.uniform(0, n)
            w = rng.uniform(n * 0.1, n * 0.3)
            a += np.exp(-0.5 * ((t - c) / w) ** 2)
    elif itype == 'chirp':
        alpha = 0.3 + 0.3 * rng.random()
        a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    elif itype == 'ramp':
        frac = rng.uniform(0.3, 0.7)
        a = np.where(t < n * frac, 1.5, 0.5)
    elif itype == 'random':
        a = 1.0 + 0.5 * rng.standard_normal(n)
    else:
        a = np.ones(n)
    return np.clip(a, 0.01, None)
