"""
Mirror-descent based optimization to minimize peak auto-convolution of non-negative sequences.
Uses softmax-regularized objective with FFT-accelerated gradient computation,
combined with local search and LP-based polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_linear_ramp(n):
    t = np.arange(1, n + 1, dtype=np.float64)
    return t


def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def initial_half_cosine(n):
    """Sequence that's high at edges, low in middle - like a 'U' shape"""
    t = np.arange(n, dtype=np.float64)
    # cos from -1 to 1 over the range, shifted to be non-negative
    # a[i] = 0.5 * (1 - cos(2*pi*(i+0.5)/n)) - this is 0 at center, 1 at edges
    # Actually let's use: a[i] = 0.5*(1-cos(pi*(2*(i+0.5)/n - 1)))
    # Simpler: a[i] = sin^2(pi*(i+0.5)/n) for i=0..n-1
    # This is 0 at i=-0.5 and i=n-0.5, peaked at i=n/2-0.5
    # That's the opposite of what we want.
    # Let's use: a[i] = cos^2(pi*(i+0.5)/n) - peaked at edges, 0 at center
    x = np.pi * (t + 0.5) / n
    return np.cos(x) ** 2


def hankel_matvec(p, a):
    """Compute H @ a where H[i,j] = p[i+j]. Uses FFT-based convolution."""
    n = len(a)
    p_flipped = p[::-1]
    # Need convolution of a (length n) and p_flipped (length 2n-1)
    # Result length: n + 2n - 1 - 1 = 3n - 2
    N = 1
    while N < 3 * n - 1:
        N *= 2
    fa = np.fft.rfft(a, N)
    fp = np.fft.rfft(p_flipped, N)
    conv = np.fft.irfft(fa * fp, N)
    # v[i] = conv[2n-2-i] for i=0,...,n-1
    v = conv[n - 1:2 * n - 1][::-1]
    return v


def mirror_descent_refine(a, time_limit_s=300.0, verbose=True):
    """
    Mirror descent with softmax-regularized objective.
    Minimizes log-sum-exp(gamma * (a*a)) which approximates max(a*a) for large gamma.
    """
    start = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_max = np.convolve(a, a).max()
    
    if verbose:
        best_score = 2.0 * n * best_max  # normalized score (sum=1)
        print(f"  MD start: max_conv={best_max:.6f}, score={best_score:.6f}, n={n}")
    
    gamma = 2.0  # initial temperature
    gamma_rate = 0.05  # increase rate
    eta = 0.5  # initial step size
    rho = 50.0  # step size decay rate
    stagnant = 0
    
    it = 0
    while time.monotonic() - start < time_limit_s:
        elapsed = time.monotonic() - start
        if elapsed > time_limit_s:
            break
        
        # Compute convolution
        conv = np.convolve(a, a)
        M = conv.max()
        
        if M < best_max - 1e-12:
            best_max = M
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 20 == 0 or it < 5):
                print(f"  MD iter {it:4d}: max={M:.6f}, score={2*n*M:.6f}, gamma={gamma:.1f}")
        else:
            stagnant += 1
        
        if stagnant > 100:
            if verbose:
                print(f"  MD stagnant at iter {it}, best={best_max:.6f}")
            break
        
        # Update gamma (increase to sharpen)
        gamma = min(50.0, gamma * (1.0 + gamma_rate))
        
        # Compute softmax weights over lags
        # p_k = exp(gamma * (conv_k - M)) / sum exp(gamma * (conv_j - M))
        shifted = conv - M  # all <= 0, max = 0
        log_p = gamma * shifted
        # For numerical stability, subtract max (which is 0 here)
        p = np.exp(log_p)
        p_sum = p.sum()
        if p_sum < 1e-300:
            # All underflow, just use argmax
            p = np.zeros_like(conv)
            p[np.argmax(conv)] = 1.0
        else:
            p = p / p_sum
        
        # Compute gradient: grad = 2 * H @ a where H[i,j] = p[i+j]
        grad = 2.0 * hankel_matvec(p, a)
        
        # Step size
        eta_t = eta / (1.0 + it / rho)
        
        # Multiplicative update
        a = a * np.exp(-eta_t * grad)
        
        # Floor to prevent underflow
        a = np.maximum(a, 1e-30)
        
        # Normalize
        a = normalize_nonneg(a)
        
        it += 1
    
    # Final score
    final_max = np.convolve(a, a).max()
    if final_max < best_max:
        best_max = final_max
        best_a = a.copy()
    
    if verbose:
        print(f"  MD done: best_max={best_max:.6f}, score={2*n*best_max:.6f}, iters={it}, time={time.monotonic()-start:.1f}s")
    
    return best_a, 2.0 * n * best_max


def local_search_refine(a, time_limit_s=60.0, rng=None, verbose=True):
    """Random local search to polish the solution."""
    if rng is None:
        rng = np.random.default_rng(0)
    start = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    if verbose:
        print(f"  LS start: score={best_score:.6f}")
    
    accepts = 0
    while time.monotonic() - start < time_limit_s:
        # Random perturbation
        width = int(rng.integers(max(2, n // 50), max(3, n // 5)))
        start_idx = int(rng.integers(0, max(1, n - width)))
        end_idx = min(n, start_idx + width)
        
        candidate = a.copy()
        noise = rng.normal(0.0, 0.1, size=end_idx - start_idx)
        candidate[start_idx:end_idx] *= np.exp(noise)
        
        # Occasionally mirror
        if rng.random() < 0.2:
            candidate = 0.5 * (candidate + candidate[::-1])
        
        candidate = normalize_nonneg(candidate)
        score = score_sequence(candidate)
        
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accepts += 1
            if verbose and accepts % 50 == 0:
                print(f"  LS accept {accepts}: score={best_score:.6f}")
    
    if verbose:
        print(f"  LS done: best={best_score:.6f}, accepts={accepts}, time={time.monotonic()-start:.1f}s")
    
    return best_a, best_score
