import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time

def fast_score(f):
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))

def compute_convolution(f):
    return np.convolve(f, f)

def subgradient_of_max_conv(f, k_star):
    """Compute subgradient of (f*f)[k_star] w.r.t. f."""
    n = len(f)
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            if 2 * i == k_star:
                grad[i] = f[j]  # = f[i]
            else:
                grad[i] = 2.0 * f[j]
    return grad

def mirror_descent_step(f, grad, eta):
    """Multiplicative update on the simplex."""
    f_new = f * np.exp(-eta * grad)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-15:
        return f
    return f_new / s

def log_sum_exp_approx(f, beta):
    """Smooth approximation of max(f*f)."""
    conv = np.convolve(f, f)
    # scale to avoid overflow
    max_c = conv.max()
    if max_c < 1e-15:
        return 0.0
    exp_vals = np.exp(beta * (conv - max_c))
    return max_c + np.log(exp_vals.sum()) / beta

def make_initial(n, rng, method='perturbed_uniform'):
    if method == 'perturbed_uniform':
        a = np.ones(n) + rng.normal(0.0, 0.3, n)
    elif method == 'cosine':
        x = np.linspace(0.0, np.pi, n)
        a = 0.5 * (1.0 + np.cos(x))
    elif method == 'symmetric_cosine':
        x = np.linspace(0.0, np.pi, n)
        a = np.sin(x) ** 2
    elif method == 'random':
        a = rng.random(n)
    elif method == 'two_peaks':
        x = np.arange(n, dtype=np.float64)
        s = max(1, n // 16)
        a = (np.exp(-0.5 * ((x - n // 4) / s) ** 2) +
             np.exp(-0.5 * ((x - 3 * n // 4) / s) ** 2))
    elif method == 'triangular':
        x = np.arange(n, dtype=np.float64)
        mid = n / 2.0
        a = 1.0 - np.abs(x - mid) / mid
    elif method == 'gaussian':
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-0.5 * ((x - n/2) / (n/6)) ** 2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.001)
    a = a / a.sum()
    return a

    """Mirror descent with adaptive parameters."""
    f = f_init.copy()
    n = len(f)
    best_f = f.copy()
    best_sc = fast_score(f)
    t0 = time.monotonic()
    total_time = budget_s
    it = 0
    while time.monotonic() - t0 < total_time:
        progress = (time.monotonic() - t0) / total_time
        beta = beta_start + (beta_end - beta_start) * progress
        eta = eta_start + (eta_end - eta_start) * progress
        
        for _ in range(100):  # inner batch
            if time.monotonic() - t0 >= total_time:
                break
            conv = np.convolve(f, f)
            k_star = int(np.argmax(conv))
            grad = subgradient_of_max_conv(f, k_star)
            # Normalize gradient for stability
            g_norm = np.sum(np.abs(grad))
            if g_norm < 1e-15:

            grad = grad / g_norm
            f = mirror_descent_step(f, grad, eta)
            it += 1
        
        sc = fast_score(f)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_f = f.copy()
    
    return best_f, best_sc

"""
Optimized multi-method search for low-autocorrelation non-negative sequences.
Combines LP refinement, mirror descent, and subgradient methods with
multi-start diversification and adaptive length selection.
"""
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time

# Pre-computed LP matrix structures (cached by n)
_lp_cache = {}

def get_lp_matrix(n):
    """Get pre-computed sparse LP matrix structure for length n."""
    if n in _lp_cache:
        return _lp_cache[n]
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    _lp_cache[n] = (rows, cols)
    return (rows, cols)

def fast_score(f):
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))

def solve_lp_fast(f, M, time_limit=2.0):
    """Fast LP solve using pre-computed matrix structure."""
    n = len(f)
    rows, cols = get_lp_matrix(n)
    # Data values: A[k,j] = f[k-j]
    data = f[rows - cols]  # since rows[k] - cols[k] gives the index into f
    A = sp.coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True,

    i_range = np.arange(i_min, i_max + 1)

    return grad

Key improvements over previous approach:

# Cache for LP matrix structures
_lp_struct_cache = {}

def get_lp_structure(n):
    """Pre-compute sparse matrix (row, col) indices for the LP constraint matrix."""
    if n in _lp_struct_cache:
        return _lp_struct_cache[n]
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_hi >= j_lo:
            rows_list.extend([k] * (j_hi - j_lo + 1))
            cols_list.extend(range(j_lo, j_hi + 1))
    rows = np.array(rows_list, dtype=np.int32)
    cols = np.array(cols_list, dtype=np.int32)
    _lp_struct_cache[n] = (rows, cols)
    return (rows, cols)

def fast_score(f):
    """Compute 2*n*max(conv)/sum^2 quickly."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))

def solve_lp(f, M, time_limit=3.0):
    """Solve max sum(g) s.t. A g <= M, g >= 0 where A[k,j]=f[k-j]."""
    n = len(f)
    rows, cols = get_lp_structure(n)
    data = f[rows - cols]
    A = sp.coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-12:
                return g / g.sum()
    except Exception:
        pass
    return None

def compute_subgrad(f, k_star):
    """Vectorized subgradient of (f*f)[k_star] w.r.t. f."""
    n = len(f)
    i_lo = max(0, k_star - n + 1)
    i_hi = min(n - 1, k_star)
    grad = np.zeros(n)
    if i_hi >= i_lo:
        i_range = np.arange(i_lo, i_hi + 1)
        j_range = k_star - i_range
        grad[i_range] = 2.0 * f[j_range]
        # Fix diagonal term
        if k_star % 2 == 0:
            i_d = k_star // 2
            if i_lo <= i_d <= i_hi:
                grad[i_d] = f[i_d]
    return grad

def mirror_descent_batch(f, n_iters, eta):
    """Run n_iters mirror descent steps on the simplex."""
    conv = np.convolve(f, f)
    for _ in range(n_iters):
        k_star = int(np.argmax(conv))
        grad = compute_subgrad(f, k_star)
        g_max = grad.max()
        if g_max < 1e-15:
            break
        log_w = np.log(np.maximum(f, 1e-30)) - eta * grad / g_max
        log_w -= log_w.max()  # numerical stability
        w = np.exp(log_w)
        s = w.sum()
        if s < 1e-15:
            break
        f = w / s
        conv = np.convolve(f, f)
    return f

        parent = parent / parent.sum()

    best_sc = np.inf
    
    def try_update(f):
        nonlocal best, best_sc
        s = fast_score(f)
        if np.isfinite(s) and s < best_sc:
            best_sc = s
            best = f.copy()
            print("  improved: %.6f (n=%d)" % (s, len(f)), flush=True)
    
    print("Starting multi-phase search. Budget: %.0fs" % budget_s, flush=True)
    
    # === Phase 1: Multi-start LP exploration ===
    # Try different (n, method) combinations
    n_values = [256, 384, 512, 640, 768, 1024]
    methods = list(range(10))
    
    # If we have a parent, include its length
    if parent is not None:
        pn = int(parent.size)
        if pn not in n_values and 100 <= pn <= 2048:
            n_values.append(pn)
    
    # Phase 1 budget: 40% of total
    p1_budget = 0.4 * (budget_s - 10)
    n_combos = len(n_values) * len(methods)
    per_combo = p1_budget / n_combos
    per_combo = max(5.0, min(per_combo, 30.0))
    
    print("Phase 1: Multi-start exploration (%d combos, ~%.0fs each)" % 
          (n_combos, per_combo), flush=True)
    
    phase1_results = []
    combo_idx = 0
    for n in n_values:
        for m in methods:
            if time.monotonic() > deadline - p1_budget:
                break
            combo_idx += 1
            start = make_init(n, rng, m)
            t_local = min(per_combo, deadline - time.monotonic())
            if t_local < 3:
                continue
            c_best, c_sc = run_lp_loop(start, t_local, t_init=0.35, t_decay=0.997,
                                       md_batch=30, md_eta=0.4)
            phase1_results.append((c_sc, c_best))
            try_update(c_best)
    
    # Sort phase 1 results
    phase1_results.sort(key=lambda x: x[0])
    print("Phase 1 best: %.6f" % (phase1_results[0][0] if phase1_results else np.inf), flush=True)
    
    # === Phase 2: Focused refinement on top candidates ===
    # Take top 3 from phase 1 (and parent if available)
    candidates = []
    if parent is not None:
        ps = fast_score(parent)
        if np.isfinite(ps):
            candidates.append((ps, parent.copy()))
    for sc, f in phase1_results[:5]:
        if np.isfinite(sc):
            candidates.append((sc, f.copy()))
    
    # Deduplicate by score (keep best per approximate score)
    candidates.sort(key=lambda x: x[0])
    unique_cands = []
    for sc, f in candidates:
        added = False
        for _, uf in unique_cands:
            if len(f) == len(uf) and np.linalg.norm(f - uf) < 0.01:
                added = True
                break
        if not added:
            unique_cands.append((sc, f))
    
    # Phase 2 budget: 40% of total
    p2_budget = 0.4 * (budget_s - 10)
    n_cands = min(4, len(unique_cands))
    per_cand = p2_budget / max(1, n_cands)
    
    print("Phase 2: Focused refinement (%d candidates, ~%.0fs each)" % 
          (n_cands, per_cand), flush=True)
    
    for i in range(n_cands):
        if time.monotonic() > deadline - p2_budget * 0.5:
            break
        sc_c, f_c = unique_cands[i]
        t_local = min(per_cand, deadline - time.monotonic())
        if t_local < 5:
            continue
        print("  Refining candidate %d (score=%.6f, n=%d) for %.0fs" % 
              (i, sc_c, len(f_c), t_local), flush=True)
        c_best, c_sc = run_lp_loop(f_c, t_local, t_init=0.3, t_decay=0.998,
                                    md_batch=80, md_eta=0.25)
        try_update(c_best)
    
    # === Phase 3: Local polish ===
    p3_budget = max(30.0, (deadline - time.monotonic()) - 5)
    if best is not None and p3_budget > 10:
        print("Phase 3: Local polish on best (score=%.6f, n=%d) for %.0fs" % 
              (best_sc, len(best), p3_budget), flush=True)
        c_best, c_sc = refine_local(best, p3_budget)
        try_update(c_best)
    
    # === Final verification ===
    if best is None:
        # Fallback: uniform sequence
        best = np.ones(512) / 512
        best_sc = 2.0
    
    # Verify with authoritative evaluator
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        best = np.ones(512) / 512
        final_score = float(evaluate_sequence(best.tolist()))
    
    print("FINAL: score=%.6f, n=%d" % (final_score, len(best)), flush=True)
    return [float(max(0.0, min(1000.0, x))) for x in best]
