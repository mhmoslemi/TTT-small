    weights = softmax(conv(a,a)/tau) -- the attention weights on each lag.

    # correlate(weights, a)[i] = sum_j weights[i+j] * a[j] for j=0..n-1

    # But we need: sum_{j=0}^{n-1} weights[i+j] * a[j] for i=0..n-1

    projected gradient on the probability simplex, and periodic
    exact max (hard) evaluations.
    """

    deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)
    
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(6121)
    parent = np.clip(parent, 0.0, None)
    n = parent.size
    if n == 0:
        n = 6121
        parent = np.ones(n)
    
    S = np.sum(parent)
    if S > 1e-10:
        a = parent / S
    else:
        a = np.ones(n) / n
    
    def evaluate(a):
        conv = np.convolve(a, a)
        return 2.0 * n * np.max(conv) / (np.sum(a)**2)
    
    def lse_obj(a, tau):
        conv = np.convolve(a, a)
        x = conv / tau
        x_max = np.max(x)
        return tau * (x_max + np.log(np.sum(np.exp(x - x_max))))
    
    def lse_grad(a, tau):
        conv = np.convolve(a, a)
        x = conv / tau
        x_max = np.max(x)
        w = np.exp(x - x_max)
        w /= np.sum(w)
        # Gradient: 2 * correlate(w, a, 'valid') / tau
        corr = np.correlate(w, a, mode='valid')
        return 2.0 * corr / tau
    
    def project_simplex(v, target=1.0):
        v = np.asarray(v, dtype=np.float64)
        n = len(v)
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u)
        rho = np.sum(u * np.arange(1, n+1) > (cssv - target))
        if rho == 0:
            return np.full(n, target / n)
        theta = (cssv[rho-1] - target) / rho
        return np.maximum(v - theta, 0.0)
    
    best_a = a.copy()
    best_score = evaluate(best_a)
    print(f"Initial score: {best_score:.6f}")

    tau_max = 0.1  # Start smooth
    tau_min = 0.001  # End sharp (close to max)

import numpy as np

def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.sum(u * np.arange(1, n + 1) > (cssv - target))
    if rho == 0:
        return np.full(n, target / n)
    theta = (cssv[rho - 1] - target) / rho
    return np.maximum(v - theta, 0.0)

def _conv(a):
    return np.convolve(a, a)

def _score(a):
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    return 2.0 * len(a) * np.max(_conv(a)) / (S * S)

def _lse_grad(a, tau):
    """Gradient of (1/tau)*log(sum(exp(conv(a,a)/tau))) w.r.t. a."""
    conv = _conv(a)
    x = conv / tau
    x_max = np.max(x)
    w = np.exp(x - x_max)
    w /= np.sum(w)
    corr = np.correlate(w, a, mode='valid')
    return 2.0 * corr / tau

def _subgrad_exact(a, top_k=10):
