    n = len(x)
    S = x.sum()
    if S < 1e-10:
        return 1e10, np.zeros(n)

    cmax = conv.max()

"""
Gradient-based optimization with log-sum-exp smoothing to minimize the peak
of the self-convolution of a non-negative step function.

Algorithm:
1. Initialize with multiple structured starting points (perturbed uniform,
   cosine, random, block patterns, and provided parent sequence).
2. Optimize using L-BFGS-B with a smoothed (log-sum-exp) approximation of
   the max function, gradually increasing the sharpness parameter beta.
3. Polish with local random perturbations.

Key insight: score = 2*n*max(conv)/sum^2 where conv = x*x (linear convolution).
The center of the convolution is sum_i x_i * x_{n-1-i} (NOT sum of squares
unless palindromic), so non-palindromic sequences can achieve score < 2.
"""
import numpy as np
from scipy.optimize import minimize
import time


def fast_score(f):
    """Compute score faithfully mirroring evaluate_sequence."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * n * conv.max() / (s * s))


def obj_and_grad(x, beta):
    """Compute smoothed objective value and gradient."""
    n = len(x)
    S = x.sum()
    if S < 1e-10:
        return 1e10, np.zeros(n)

    conv = np.convolve(x, x)  # length 2n-1
    cmax = conv.max()

    # Numerically stable log-sum-exp
    shifted = beta * conv - beta * cmax
    exp_s = np.exp(shifted)
    log_sum = np.log(exp_s.sum())
    M = cmax + log_sum / beta

    # Softmax weights p_k
    p = exp_s / exp_s.sum()

    # Gradient of M: grad_M[i] = 2 * sum_k p[k] * x[k-i] (valid k-i)
    # = 2 * cross_correlation(p, x)[i + n - 1]
    corr = np.correlate(p, x, mode='full')
    grad_M = 2.0 * corr[(n - 1):(2 * n - 1)]

    # Objective: f = 2*n*M / S^2
    f_val = 2.0 * n * M / (S * S)

    # Gradient: df/dx_i = 2*n*(grad_M_i * S - 2*M) / S^3
    grad_f = 2.0 * n * (grad_M * S - 2.0 * M) / (S * S * S)

    return f_val, grad_f


def optimize_with_lbgfgsb(x0, beta_schedule, max_iter_per_beta=200):
    """Run L-BFGS-B sequentially with increasing beta values."""
    n = len(x0)
    x = x0.copy()
    bounds = [(0.0, None)] * n

    for beta in beta_schedule:
        def funnel(u):
            v, g = obj_and_grad(u, beta)
            if not np.isfinite(v):
                return 1e10, np.zeros_like(u)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            return v, g

        try:
            res = minimize(funnel, x, method='L-BFGS-B', jac=True,
                           bounds=bounds,
                           options={'maxiter': max_iter_per_beta,
                                    'gtol': 1e-10, 'ftol': 1e-12})
            x = np.maximum(res.x, 0.0)
        except Exception:
            pass

        s = x.sum()
        if s > 1e-10:
            x = x / s
        else:
            x = np.ones(n) / n

    return x

    local_rng = np.random.default_rng(seed + seed_offset)

    # 1. Perturbed uniform

    # 2. Cosine taper (bell-shaped)
    x = np.linspace(0, np.pi, n)
    a = np.cos(x / 2) ** 2
    a = np.maximum(a, 0.01)
    inits.append(a / a.sum())

    # 3. Pure random
    a = rng.random(n) + 0.05
    inits.append(a / a.sum())

    # 4. Block pattern (alternating high/low blocks)
    nblocks = max(2, n // 64)
    block_len = n // nblocks
    a = np.ones(n)
    for i in range(nblocks):
        a[i * block_len:(i + 1) * block_len] = rng.uniform(0.3, 2.0)
    a = np.maximum(a, 0.01)
    inits.append(a / a.sum())

    # 5. Asymmetric ramp (encourages non-palindromic structure)
    a = np.linspace(0.3, 2.0, n) + rng.normal(0, 0.1, n)
    a = np.maximum(a, 0.01)
    inits.append(a / a.sum())

    # 6. Decay (exponential)
    a = np.exp(-np.linspace(0, 3, n)) + 0.05
    a = a + rng.normal(0, 0.05, n)
    a = np.maximum(a, 0.01)
    inits.append(a / a.sum())

    return inits


def local_search_polish(x, deadline, rng):
    """Local search with random multiplicative block perturbations."""
    n = len(x)
    best = x.copy()
    best_sc = fast_score(best)
    no_improve = 0
    max_no_improve = 500

    while time.monotonic() < deadline and no_improve < max_no_improve:
        w = int(rng.integers(1, max(2, n // 15 + 1)))
        start = int(rng.integers(0, n - w + 1))

        cand = best.copy()
        noise = np.exp(rng.normal(0, 0.08, w))
        cand[start:start + w] *= noise
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-10:
            no_improve += 1
            continue
        cand = cand / s

        sc = fast_score(cand)
        if sc < best_sc - 1e-10:
            best_sc = sc
            best = cand
            no_improve = 0
        else:
            no_improve += 1

    return best, best_sc
