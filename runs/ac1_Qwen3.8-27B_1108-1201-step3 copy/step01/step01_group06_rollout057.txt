Minimize 2*n*max_conv/sum^2 using graduated non-convexity with Adam.

Strategy:
- Normalize to sum=sqrt(2n) so objective equals max of self-convolution

- Exact gradient via FFT: grad = cross(c^(p-1), a) + conv(c^(p-1), a)

import numpy as np

from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def smooth_gradient(a, c, p):
    """Unnormalized gradient of sum(c[k]^p) w.r.t. a[i]."""
    n = len(a)
    c_max = c.max()
    if c_max <= 1e-15:
        return np.zeros(n)
    c_norm = np.clip(c / c_max, 0.0, 1.0)
    w = c_norm ** (p - 1)
    
    w_len = len(w)  # 2n-1
    fft_size = 1 << (w_len + n - 1 - 1).bit_length()
    
    W = fft(w, fft_size)
    A = fft(a, fft_size)
    
    cross = np.real(ifft(np.conj(W) * A))
    conv_wa = np.real(ifft(W * A))
    
    return cross[:n] + conv_wa[:n]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta_val = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta_val, 0.0)

def peak_shave(a, n, num_steps, rng):
    """Direct peak shaving: find max lag, reduce contributing elements."""
    S = np.sqrt(2.0 * float(n))
    a = normalize_seq(a, n)
    best_score = np.inf
    best_local = a.copy()
    
    for step in range(num_steps):
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()
        
        # Find lags where c is close to max
        k_star = int(np.argmax(c))
        
        # Contribution of a[i] to c[k_star]:
        # c[k_star] = sum_j a[j] * a[k_star - j]
        # For each i, the contribution is a[i] * a[k_star - i] + a[k_star - i] * a[i]
        # = 2 * a[i] * a[k_star - i]
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = 2.0 * a[i] * a[j]
        
        # Reduce the top contributors
        top_k = min(100, n // 10)
        top_idx = np.argsort(contrib)[::-1][:top_k]
        
        # Reduce top contributors and increase bottom
        bottom_k = min(100, n // 10)
        bottom_idx = np.argsort(contrib)[:bottom_k]
        
        delta = 0.01 * S / n
        a[top_idx] -= delta
        # Distribute reduction to bottom
        per_bottom = delta * len(top_idx) / max(1, len(bottom_idx))
        a[bottom_idx] += per_bottom
        
        a = np.maximum(a, 0.0)
        a = a * (S / max(a.sum(), 1e-15))
    
    return best_local, best_score

"""
Minimize 2*n*max_conv/sum^2 using graduated non-convexity with exact FFT-based gradients.

Key ideas:
- Normalize to sum=sqrt(2n) so objective equals max of self-convolution
- Use p-norm smooth surrogate with increasing p for graduated non-convexity
- Compute exact smooth gradient via FFT (cross-correlation + convolution)
- Adam optimizer with simplex projection for non-negativity + sum constraint
- Start from given best sequence; also try structured initializations
- Peak shaving as local refinement
- Polish best candidate with remaining budget using high p values
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def smooth_gradient(a, c, p):
    """Gradient of sum(c[k]^p) w.r.t. a (up to positive scalar).
    Returns cross(c^(p-1), a) + conv(c^(p-1), a), evaluated at indices 0..n-1.
    """
    n = len(a)
    c_max = c.max()
    if c_max <= 1e-15:
        return np.zeros(n)
    c_norm = np.clip(c / c_max, 0.0, 1.0)
    w = c_norm ** (p - 1)

    w_len = len(w)
    fft_size = 1 << (w_len + n - 1 - 1).bit_length()

    W = fft(w, fft_size)
    A = fft(a, fft_size)

    cross = np.real(ifft(np.conj(W) * A))
    conv_wa = np.real(ifft(W * A))

    return cross[:n] + conv_wa[:n]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta_val = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta_val, 0.0)


def normalize_seq(a, n):
    """Clip to [0,1000] and scale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.full(n, S / n, dtype=np.float64)
    return a * (S / s)


def run_graduated_adam(a_init, n, p_schedule, max_time, rng, lr_init, restart_patience=20000):
    """Adam with graduated p-norm non-convexity and periodic restarts."""
    a = normalize_seq(a_init, n)
    S = np.sqrt(2.0 * float(n))

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8

    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    best_t = 0

    start = time.monotonic()
    end_time = start + max_time
    total_iters = 0

    for p in p_schedule:
        if time.monotonic() >= end_time - 3.0:
            break

        p_stall = 0
        p_iter = 0

        while time.monotonic() < end_time - 3.0:
            t += 1
            p_iter += 1
            total_iters += 1

            c = fft_convolve_self(a)
            cur_max = c.max()
            if cur_max < best_score - 1e-12:
                best_score = cur_max
                best_local = a.copy()
                best_t = t
                p_stall = 0
            else:
                p_stall += 1

            grad = smooth_gradient(a, c, p)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            grad = grad / gnorm

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            lr = lr_init * (0.99995 ** p_iter)
            a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)

            # Restart if stuck for too long
            if p_stall > restart_patience and rng.random() < 0.1:
                # Inject moderate noise
                noise = rng.normal(0.0, 0.02 * S / n, n)
                a = project_simplex(a + noise, S)
                p_stall = 0
                m = np.zeros(n)
                v = np.zeros(n)

        print("      p=%3d: %d iters, best_max=%.8f, elapsed=%.1fs" %
              (p, p_iter, best_score, time.monotonic() - start))

    return best_local, best_score, total_iters


def peak_shave(a_init, n, max_time, rng):
    """Vectorized peak shaving: iteratively reduce elements contributing to max lag."""
    S = np.sqrt(2.0 * float(n))
    a = normalize_seq(a_init, n)

    best_score = np.inf
    best_local = a.copy()
    start = time.monotonic()

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()

        # Find top lags (broaden slightly to avoid over-focusing on one)
        top_lags = np.argsort(c)[::-1][:3]

        # Compute total contribution from each a[i] to top lags
        contrib = np.zeros(n)
        for k in top_lags:
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    contrib[i] += c[k] * 2.0 * a[i] * a[j]

        # Reduce top contributors, boost bottom
        n_reduce = max(10, n // 50)
        n_boost = max(10, n // 50)
        top_idx = np.argsort(contrib)[::-1][:n_reduce]
        bot_idx = np.argsort(contrib)[:n_boost]

        # Amount to shift
        total_reduce = contrib[top_idx].sum()
        if total_reduce < 1e-15:
            break
        fraction = 0.05  # shift 5% of the peak contribution
        delta = fraction * total_reduce / (2.0 * c[top_lags[0]] * max(n_reduce, 1))

        # Actually, let's just use a fixed small step
        step = 0.002 * S / n
        a[top_idx] -= step
        a[bot_idx] += step * n_reduce / max(n_boost, 1)

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-15:
            a = a * (S / s)

    return best_local, best_score
