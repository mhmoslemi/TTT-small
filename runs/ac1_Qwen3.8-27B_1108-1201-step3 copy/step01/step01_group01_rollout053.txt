"""
Improved multi-phase optimizer for minimizing the normalized autocorrelation peak
of a non-negative sequence. Uses soft-max gradient descent with Adam optimizer,
multiple diverse initializations, and coordinate descent polishing.
"""

import numpy as np
import time

def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _compute_conv_and_score(a):
    """Compute convolution and score for a sequence with sum=1."""
    n = len(a)
    conv = np.convolve(a, a)
    max_conv = np.max(conv)
    score = 2.0 * n * max_conv
    return conv, max_conv, score

def _grad_conv_k(a, k):
    """Gradient of conv[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * a[k - idx]
    return grad

def _soft_max_gradient(a, conv, t, M=50):
    """Compute gradient of soft-max approximation of max(conv)."""
    n = len(a)
    cur_max = np.max(conv)
    
    # Select top M values
    if M >= len(conv):
        top_idx = np.arange(len(conv))
    else:
        top_idx = np.argpartition(conv, -M)[-M:]
    
    top_vals = conv[top_idx]
    
    # Soft-max weights (numerically stable)
    shifted = t * (top_vals - cur_max)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    
    # Compute weighted gradient
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        if weights[i] < 1e-12:
            continue
        k = top_idx[i]
        w = weights[i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            grad[idx] += w * 2.0 * a[k - idx]
    
    return grad

def _adam_step(a, m, v, grad, lr, step, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad**2
    m_hat = m / (1 - beta1**step)
    v_hat = v / (1 - beta2**step)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    return a_new, m, v

def _smooth_optimize(a, deadline, rng, t_init=5.0, t_max=200.0, t_increase=1.03,
                     lr_init=0.05, M=50, perturb_every=200, perturb_scale=0.02):
    """
    Soft-max gradient descent with Adam to minimize max(conv(a,a)).
    Assumes a is normalized to sum=1.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = np.sum(a)
    if S < 1e-15:
        a = np.ones(n) / n
        return a, np.inf
    a /= S
    
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    t = t_init
    lr = lr_init
    
    conv, _, best_score = _compute_conv_and_score(a)
    best_a = a.copy()
    
    no_improve = 0
    
    while time.monotonic() < deadline:
        step += 1
        
        # Compute soft-max gradient
        grad = _soft_max_gradient(a, conv, t, M)
        
        # Center for sum constraint
        grad -= np.mean(grad)
        
        # Adam step
        a, m, v = _adam_step(a, m, v, grad, lr, step)
        
        # Project to simplex
        a = _project_simplex(a, 1.0)
        
        # Compute score
        conv = np.convolve(a, a)
        cur_score = 2.0 * n * np.max(conv)
        
        if cur_score < best_score - 1e-10:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Anneal temperature
        if step % 50 == 0:
            t = min(t * t_increase, t_max)
            lr *= 0.99
        
        # Perturb to escape local minima
        if step % perturb_every == 0 and no_improve > perturb_every // 2:
            a = a + perturb_scale * rng.standard_normal(n)
            a = _project_simplex(a, 1.0)
            no_improve = 0
            lr = lr_init * 0.5  # Reset learning rate after perturbation
        
        # Check time periodically (every 100 steps)
        if step % 100 == 0:
            if time.monotonic() >= deadline:
                break
    
    return best_a, best_score

def _make_two_level(n, eps, rng):
    """Two-level sequence with random threshold."""
    threshold = rng.uniform(0.3, 0.7)
    a = np.where(np.arange(n) / n < threshold, 1.0 + eps, 1.0 - eps)
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S

def _make_periodic(n, freq, eps, rng):
    """Periodic modulation: 1 + eps*cos(2*pi*freq*i/n + phase)."""
    phase = rng.uniform(0, 2 * np.pi)
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * i / n + phase)
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S

def _make_quadratic_residue(n, rng):
    """
    Use quadratic residues mod p (where p is the largest prime <= n)
    as a starting pattern.
    """
    # Find largest prime <= n
    def is_prime(k):
        if k < 2:
            return False
        for i in range(2, int(np.sqrt(k)) + 1):
            if k % i == 0:
                return False
        return True
    
    p = n
    while p > 3 and not is_prime(p):
        p -= 1
    
    if p < 5:
        return _make_two_level(n, 0.5, rng)
    
    if p % 4 != 1:
        # Use a random pattern instead
        mask = rng.random(n) < 0.5
        a = np.where(mask, 1.0, 0.3)
    else:
        # Quadratic residues mod p
        qr = set((i * i) % p for i in range(p))
        a = np.zeros(n)
        for i in range(n):
            a[i] = 1.0 if i in qr else 0.3
    
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S

def _project_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _compute_score(a):
    n = len(a)
    conv = np.convolve(a, a)
    return 2.0 * n * np.max(conv), conv

def _get_top_gradient(a, conv, t, M):
    """Compute gradient of soft-max of top M conv values."""
    n = len(a)
    cur_max = np.max(conv)
    if M >= len(conv):
        top_idx = np.arange(len(conv))
    else:
        top_idx = np.argpartition(conv, -M)[-M:]
    top_vals = conv[top_idx]
    shifted = t * (top_vals - cur_max)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        w = weights[i]
        if w < 1e-12:
            continue
        k = int(top_idx[i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            grad[idx] += w * 2.0 * a[k - idx]
    return grad

def _adam_update(params, state, grad, lr):
    """Adam update. params is the array to update, state contains (m, v, step)."""
    m, v, step = state
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step += 1
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * grad * grad
    m_hat = m / (1.0 - beta1 ** step)
    v_hat = v / (1.0 - beta2 ** step)
    update = lr * m_hat / (np.sqrt(v_hat) + eps)
    new_params = params - update
    return new_params, (m, v, step)
