Multi-method optimization of non-negative step function to minimize
2*n*max(f*f)/(sum(f))^2.

Combines:

- LP-based refinement for final polish
- Multi-start with diverse initializations
- Simulated annealing for exploration

import scipy.sparse as sp
from scipy.optimize import linprog
import time


def fast_score(f):
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))

    """Compute subgradient of max_k (f*f)[k] at the given argmax k_star."""
    n = len(f)
    sg = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n, k_star + 1)
    if lo < hi:
        indices = np.arange(lo, hi)
        sg[indices] = 2.0 * f[k_star - indices]

Multi-method optimization of non-negative step function to minimize

def fast_score(f):
    """Fast score computation (scale-invariant)."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))


def make_initial(n, rng, method='cosine'):
    """Create initial non-negative sequence normalized to sum=1."""
    x = np.arange(n, dtype=np.float64)
    if method == 'cosine':
        a = 0.5 * (1.0 + np.cos(np.pi * x / max(1, n - 1)))
    elif method == 'u_shape':

def subgradient_step(f, lr, top_k=1):
    """One multiplicative subgradient step using top-k peaks."""
    n = len(f)
    conv = np.convolve(f, f)
    # Get top-k indices
    if top_k >= len(conv):
        top_indices = np.arange(len(conv))
    else:
        top_indices = np.argsort(conv)[::-1][:top_k]
    sg = np.zeros(n)
    for k in top_indices:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            indices = np.arange(lo, hi)
            sg[indices] += f[k - indices]
    sg /= top_k
    if sg.max() < 1e-15:
        return f, conv.max()
    # Multiplicative update with scaling
    alpha = lr / sg.max()
    f_new = f * np.exp(-alpha * sg)
    f_new = np.maximum(f_new, 1e-15)
    s = f_new.sum()
    f_new = f_new / s
    return f_new, conv.max()


def run_subgradient(f, deadline, lr_init=2.0, lr_min=0.01, decay=0.999,
                    top_k=3, stall_limit=50):

    best_sc = fast_score(f)

    stall = 0
    it = 0
    rng = np.random.default_rng(it * 7919 + 3)

        f = f_new

                f = f / f.sum()
                stall = 0

def softmax_objective(x, n):
    """Objective and gradient for softmax parameterization."""
    # x has length n-1 (last component is 0 for normalization)
    x_full = np.append(x, 0.0)
    x_shifted = x_full - x_full.max()
    f = np.exp(x_shifted)
    f = f / f.sum()
    conv = np.convolve(f, f)
    k_star = int(np.argmax(conv))
    val = 2.0 * n * conv.max()
    # Subgradient of val w.r.t. f
    sg = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n, k_star + 1)
    if lo < hi:
        indices = np.arange(lo, hi)
        sg[indices] = 2.0 * n * 2.0 * f[k_star - indices]
    # Chain rule through softmax: dJ/dx_i = f_i * (dJ/df_i - sum_j f_j * dJ/df_j)
    mean_sgf = np.dot(f, sg)
    grad = f * (sg - mean_sgf)
    return val, grad[:n - 1]

def build_conv_matrix(n, f):
    """Sparse convolution matrix A (2n-1 x n)."""
    i_idx = np.tile(np.arange(n), n)
    j_idx = np.repeat(np.arange(n), n)
    row_idx = j_idx + i_idx
    data = f[i_idx]
    A = sp.coo_matrix((data, (row_idx, j_idx)), shape=(2 * n - 1, n)).tocsr()
    return A


def lp_step(f, time_limit=5.0):
    """One LP-based step."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(conv.max())
    if M < 1e-15:
        return f
    A = build_conv_matrix(n, f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                g = g / gs
                if np.linalg.norm(g - f) > 1e-10:
                    return g
    except Exception:
        pass
    return f

def block_perturb(f, rng, width_frac=0.1, noise=0.3):

    cand[start:stop] *= float(np.exp(rng.normal(0.0, noise)))
    cand = np.maximum(cand, 1e-15)
    s = cand.sum()

        print("[SG] n=%d method=%s t=%.0fs" % (n, method, per_config_time),
              flush=True)
        f_best, sc = run_subgradient(f, local_deadline)
        if np.isfinite(sc):
            update(f_best, sc)

    # Phase 2: L-BFGS refinement on best candidates (15% of time)
    phase2_time = 0.15 * budget_s
    phase2_deadline = time.monotonic() + phase2_time
    print("[L-BFGS] Refining best candidate (n=%d)" % (best_seq.size if best_seq is not None else 0),
          flush=True)

    if best_seq is not None:
        f_lbg, sc_lbg = run_lbfgs(best_seq, phase2_deadline, max_iter=100)
        if np.isfinite(sc_lbg):
            update(f_lbg, sc_lbg)
        # Also try L-BFGS from a few other starts
        for n, method in [(256, 'cos_mod'), (512, 'u_shape'), (384, 'cosine')]:
            if time.monotonic() > phase2_deadline:
                break
            f_init = make_initial(n, rng, method)
            f_lbg2, sc_lbg2 = run_lbfgs(f_init, min(phase2_deadline, time.monotonic() + 30),
                                        max_iter=100)
            if np.isfinite(sc_lbg2):
                update(f_lbg2, sc_lbg2)

    # Phase 3: LP polish (20% of time)
    phase3_time = 0.2 * budget_s
    phase3_deadline = time.monotonic() + phase3_time
    print("[LP] Polishing best candidate (n=%d)" % (best_seq.size if best_seq is not None else 0),
          flush=True)

    if best_seq is not None:
        f_lp, sc_lp = run_lp_phase(best_seq, phase3_deadline)
        if np.isfinite(sc_lp):
            update(f_lp, sc_lp)

    # Phase 4: Final random search + mini subgradient (5% of time)
    phase4_deadline = deadline
    print("[Polish] Final random search phase", flush=True)
    if best_seq is not None:
        rng4 = np.random.default_rng(seed + 999)
        for _ in range(200):
            if time.monotonic() > phase4_deadline:
                break
            cand = block_perturb(best_seq, rng4, width_frac=0.05, noise=0.2)
            # Quick subgradient refinement
            mini_deadline = min(phase4_deadline, time.monotonic() + 2.0)
            f_ref, sc_ref = run_subgradient(cand, mini_deadline, lr_init=1.0,
                                            lr_min=0.05, decay=0.999, top_k=3,
                                            stall_limit=20)
            if np.isfinite(sc_ref):
                update(f_ref, sc_ref)

    # Final validation
    if best_seq is None:
        best_seq = np.ones(256)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.size == 0 or not np.isfinite(best_seq).all() or best_seq.sum() < 0.01:
        best_seq = np.ones(256)

    final_score = float(evaluate_sequence(best_seq.tolist()))
    print("FINAL: score = %.6f (n=%d)" % (final_score, best_seq.size), flush=True)
    return [float(x) for x in best_seq]
