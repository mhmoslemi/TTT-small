import numpy as np

def _score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b) / (s * s)

def _project_simplex(v, target=1.0):
    """Project onto {x >= 0, sum(x) = target}"""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _subgrad_step(a, eta, M_peaks=20):
    """One subgradient step."""
    n = len(a)
    b = np.convolve(a, a)
    bm = np.max(b)
    # Get top M active lags
    if M_peaks >= len(b):
        k_idx = np.arange(len(b))
    else:
        k_idx = np.argpartition(b, -M_peaks)[-M_peaks:]
    # Weight by how close to peak
    b_sel = b[k_idx]
    wts = np.exp(-10.0 * (bm - b_sel) / max(bm, 1e-15))
    wts /= np.sum(wts)
    
    # Compute gradient
    grad = np.zeros(n)
    for ki, w in zip(k_idx, wts):

        k = ki
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] += w * 2.0 * a[k - idx]
    
    # Center
    grad -= np.mean(grad)
    
    # Normalize
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return a, bm
    a = a - eta * grad / gn
    a = _project_simplex(a, 1.0)

"""
Subgradient descent with momentum + LP-based polishing to minimize
the normalized autocorrelation peak of a non-negative sequence.
Uses adaptive step size, Nesterov-like momentum, and diverse restarts.
"""

import numpy as np
import time


def _compute_score_unit(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)


def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad(a, k_active, wts):
    """Compute weighted subgradient of max autocorrelation."""
    n = len(a)
    grad = np.zeros(n)
    for k, w in zip(k_active, wts):
        if w < 1e-12:
            continue
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            grad[idx] += 2.0 * w * a[k - idx]
    return grad


def _subgrad_momentum(a, deadline, rng, M=80, eta_init=0.05, momentum_init=0.5):
    """
    Subgradient descent with momentum and adaptive step size.
    Minimizes max_k conv(a,a)_k subject to sum(a)=1, a>=0.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a = a / S
    else:
        a = np.ones(n) / n

    best = a.copy()
    best_score = _compute_score_unit(best)
    eta = eta_init
    mu = momentum_init  # momentum
    velocity = np.zeros(n)
    no_improve = 0
    iter_count = 0

    while time.monotonic() < deadline:
        b = np.convolve(a, a)
        bm = np.max(b)
        cur_score = 2.0 * n * bm

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive step size: decrease when no improvement
        if no_improve % 100 == 0 and no_improve > 0:
            eta *= 0.85
            mu = max(0.1, mu * 0.95)

        if eta < 1e-12:
            break

        # Get active lags
        if M >= len(b):
            k_active = np.arange(len(b))
            wts = np.ones(len(b)) / len(b)
        else:
            k_active = np.argpartition(b, -M)[-M:]
            b_sel = b[k_active]
            spread = bm - b_sel.min()
            if spread > 1e-15:
                wts = np.exp(-8.0 * (bm - b_sel) / spread)
            else:
                wts = np.ones(M) / M
            wts = wts / np.sum(wts)

        # Compute gradient
        grad = _compute_grad(a, k_active, wts)
        grad -= np.mean(grad)  # Center for sum constraint

        # Momentum update
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad_normed = grad / gn

        velocity = momentum_init * velocity + (1 - momentum_init) * grad_normed
        # Actually, let me use standard Nesterov momentum:
        # velocity = mu * velocity + grad_normed
        # a = a - eta * velocity
        
        velocity = mu * velocity + grad_normed
        vn = np.linalg.norm(velocity)
        if vn < 1e-15:
            break
        velocity = velocity / vn

        a = a - eta * velocity
        a = _project_simplex(a, 1.0)

        iter_count += 1
        if iter_count % 10000 == 0:
            print(f"  iter {iter_count}: score={cur_score:.8f}, best={best_score:.8f}, eta={eta:.6f}")

    return best, best_score


def _lp_polish(a, deadline, max_steps=50):
    """
    LP-based polishing: at each step, find the best direction g
    that maximizes sum(g) subject to conv(a,g) <= current_peak, g >= 0.
    Then blend: a <- (1-t)*a + t*g.
    """
    try:
        import cvxpy as cp
    except ImportError:
        print("CVXPY not available, skipping LP polish")
        return a, _compute_score_unit(a)

    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a = a / S

    best = a.copy()
    best_score = _compute_score_unit(best)
    t_blend = 0.3  # Blend factor

    for step in range(max_steps):
        if time.monotonic() >= deadline:
            break

        b = np.convolve(a, a)
        M = np.max(b)

        # Select active lags (top 150)
        n_active = min(150, 2 * n - 1)
        k_active = np.argsort(b)[-n_active:]

        # Build LP
        g = cp.Variable(n, nonneg=True)
        obj = cp.Maximize(cp.sum(g))
        constraints = []
        for k in k_active:
            k = int(k)
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                idx = np.arange(j_lo, j_hi + 1)
                constraints.append(a[idx] @ g[k - idx] <= M)

        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-7)
        except Exception as e:
            print(f"  LP solve error: {e}")
            break

        if g.value is None:
            break

        g_val = np.maximum(g.value, 0)
        sg = np.sum(g_val)
        if sg < 1e-10:
            break
        g_val = g_val / sg  # Normalize to sum=1

        # Blend
        a_new = (1 - t_blend) * a + t_blend * g_val
        a_new = _project_simplex(a_new, 1.0)

        new_score = _compute_score_unit(a_new)
        if new_score < best_score - 1e-13:
            best = a_new.copy()
            best_score = new_score
            print(f"  LP step {step+1}: score={new_score:.8f}")
            a = a_new
        else:
            # Try smaller blend
            t_blend *= 0.5
            if t_blend < 0.01:
                break
            a_new = (1 - t_blend) * a + t_blend * g_val
            a_new = _project_simplex(a_new, 1.0)
            new_score = _compute_score_unit(a_new)
            if new_score < best_score:
                best = a_new.copy()
                best_score = new_score
                a = a_new
            else:
                break

    return best, best_score


def _make_start_two_level(n, eps):
    """Two-level anti-symmetric."""
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    a = np.maximum(a, 0)
    return a / np.sum(a)


def _make_start_cosine(n, freq, phase=0):
    """Cosine-modulated start."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + 0.5 * np.cos(2 * np.pi * freq * i / n + phase)
    a = np.maximum(a, 0)
    return a / np.sum(a)


def _make_start_edge_weighted(n, power=2.0):
    """Edge-weighted: higher at edges."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    dist = np.abs(i - center) / center
    a = 1.0 + power * dist
    return a / np.sum(a)


def _make_start_center_weighted(n, power=2.0):
    """Center-weighted: higher in middle."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    dist = np.abs(i - center) / center
    a = 1.0 + power * (1.0 - dist)
    a = np.maximum(a, 0)
    return a / np.sum(a)

"""
Multi-phase optimization combining subgradient descent with momentum
and LP-based direction finding to minimize the normalized autocorrelation
peak of a non-negative sequence. Uses adaptive step sizes, diverse
starting points, and cutting-plane LP polishing for final refinement.
"""

import numpy as np
import time


def _compute_score_unit(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)


def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_gradient(a, k_active, wts):
    """Compute weighted subgradient of max autocorrelation peak."""
    n = len(a)
    grad = np.zeros(n)
    for k, w in zip(k_active, wts):
        if w < 1e-12:
            continue
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            grad[idx] += 2.0 * w * a[k - idx]
    return grad


def _subgrad_momentum(a, deadline, rng, M=100, eta_init=0.04, mu=0.6):
    """
    Subgradient descent with Nesterov momentum and adaptive step size.
    Minimizes max_k conv(a,a)_k subject to sum(a)=1, a>=0.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a = a / S
    else:
        a = np.ones(n) / n

    best = a.copy()
    best_score = _compute_score_unit(best)
    eta = eta_init
    velocity = np.zeros(n)
    no_improve = 0
    iter_count = 0
    last_print = time.monotonic()

    while time.monotonic() < deadline:
        b = np.convolve(a, a)
        bm = np.max(b)
        cur_score = 2.0 * n * bm

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive step size
        if no_improve > 50 and no_improve % 50 == 0:
            eta *= 0.85
            mu = max(0.3, mu * 0.97)

        if eta < 1e-12:
            break

        # Get active lags with softmax weighting
        if M >= len(b):
            k_active = np.arange(len(b))
            wts = np.ones(len(b)) / len(b)
        else:
            k_active = np.argpartition(b, -M)[-M:]
            b_sel = b[k_active]
            spread = bm - b_sel.min()
            if spread > 1e-15:
                wts = np.exp(-6.0 * (bm - b_sel) / spread)
            else:
                wts = np.ones(M) / M
            wts = wts / np.sum(wts)

        # Compute subgradient
        grad = _compute_gradient(a, k_active, wts)
        grad -= np.mean(grad)

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn

        # Momentum update
        velocity = mu * velocity + (1.0 - mu) * grad
        vn = np.linalg.norm(velocity)
        if vn < 1e-15:
            break
        velocity = velocity / vn

        a = a - eta * velocity
        a = _project_simplex(a, 1.0)

        iter_count += 1
        if time.monotonic() - last_print > 30:
            print(f"    iter {iter_count}: score={cur_score:.8f}, eta={eta:.6f}, mu={mu:.3f}")
            last_print = time.monotonic()

    return best, best_score


def _lp_polish(a, deadline, max_steps=200, n_active_init=200):
    """
    LP-based direction finding with cutting plane.
    At each step: find g maximizing sum(g) s.t. conv(a,g) <= M, g >= 0.
    Blend: a <- (1-t)*a + t*g.
    """
    try:
        import cvxpy as cp
    except ImportError:
        print("  CVXPY not available, skipping LP polish")
        return a, _compute_score_unit(a)

    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a = a / S

    best = a.copy()
    best_score = _compute_score_unit(best)
    t_blend = 0.4
    n_active = n_active_init
    stagnant = 0

    for step in range(max_steps):
        if time.monotonic() >= deadline:
            break

        b = np.convolve(a, a)
        M = np.max(b)

        # Active lags
        k_active = np.argsort(b)[-n_active:]

        # Build and solve LP
        g = cp.Variable(n, nonneg=True)
        obj = cp.Maximize(cp.sum(g))
        constraints = []
        for k in k_active:
            k = int(k)
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                idx = np.arange(j_lo, j_hi + 1)
                constraints.append(a[idx] @ g[k - idx] <= M)

        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=300000, eps=1e-7)
        except Exception as e:
            print(f"  LP error at step {step}: {e}")
            break

        if g.value is None:
            break

        g_val = np.maximum(g.value, 0)
        sg = np.sum(g_val)
        if sg < 1e-10:
            break
        g_val = g_val / sg

        # Check if LP solution is good (cross-correlation bounded)
        fg = np.convolve(a, g_val)
        if np.max(fg) > M * 1.1:
            # LP wasn't tight enough, expand active set
            n_active = min(n_active + 100, 2 * n - 1)
            continue

        # Try blending
        improved = False
        for t in [t_blend, 0.2 * t_blend, 0.5 * t_blend]:
            a_new = (1.0 - t) * a + t * g_val
            a_new = _project_simplex(a_new, 1.0)
            new_score = _compute_score_unit(a_new)
            if new_score < best_score - 1e-14:
                best = a_new.copy()
                best_score = new_score
                a = a_new
                improved = True
                stagnant = 0
                print(f"  LP step {step+1}: score={new_score:.8f} (t={t:.3f})")
                break

        if not improved:
            stagnant += 1
            t_blend = max(0.05, t_blend * 0.7)
            if stagnant > 15:
                break

    return best, best_score


def _make_two_level(n, eps):
    """Two-level anti-symmetric sequence."""
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    a = np.maximum(a, 0)
    return a / np.sum(a)


def _make_cosine(n, freq, phase=0.0, amp=0.5):
    """Cosine-modulated sequence."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * i / n + phase)
    a = np.maximum(a, 0)
    return a / np.sum(a)


def _make_edge_weighted(n, power=2.0):
    """Edge-weighted: higher at edges."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    dist = np.abs(i - center) / max(center, 1)
    a = 1.0 + power * dist
    return a / np.sum(a)


def _make_center_weighted(n, power=2.0):
    """Center-weighted: higher in middle."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    dist = np.abs(i - center) / max(center, 1)
    a = 1.0 + power * (1.0 - dist)
    a = np.maximum(a, 0)
    return a / np.sum(a)


def _make_four_quarter(n, eps):
    """Four-quadrant pattern."""
    a = np.ones(n)
    q = n // 4
    a[:q] = 1.0 + eps
    a[q:2*q] = 1.0 - eps * 0.3
    a[2*q:3*q] = 1.0 - eps * 0.3
    a[3*q:] = 1.0 + eps
    a = np.maximum(a, 0)
    return a / np.sum(a)


def _perturb(a, rng, scale=0.05):
    """Add smooth random perturbation."""
    n = len(a)
    freqs = rng.integers(1, max(2, n // 8), size=min(8, n // 4))
    noise = np.zeros(n)
    for f in freqs:
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(-1, 1)
        noise += amp * np.cos(2 * np.pi * f * np.arange(n) / n + phase)
    noise /= (np.max(np.abs(noise)) + 1e-15)
    a_new = a + scale * noise
    return _project_simplex(a_new, 1.0)
