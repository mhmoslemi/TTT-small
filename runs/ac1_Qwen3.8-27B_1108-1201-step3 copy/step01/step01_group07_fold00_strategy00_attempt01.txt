        return np.inf

        if isinstance(x, bool) or not isinstance(x, (int, float)):

            return np.inf

def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    if n == 0:
        return np.convolve(a, a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

            h = np.linalg.solve(np.diag(1 - current_dir).astype(float), np.tile(1.0, len(a)))
            h = min(1.0, h.max()) * h  # Stabilize
            a = np.exp(a * h) * np.sqrt(guarantee_improvement)
        except:
            pass
        # Trust region updates
        for i in range(len(a)):
            a[i] += np.random.normal() * 1.0 / np.sqrt(len(a))
        a = np.minimum(a, 10.0)
        conv = fast_convolve(a)
        n_val = len(a)
        cur_obj = compute_objective(a)
        improvement = cur_obj - new_objective
        if improvement > 0 and iterations >= 10 and np.all(a > 0):
            # Slow down if not improving
            eta = min(eta * eta_factor, max_eta)
        if guarantee_improvement:
            # Reward small improvements
            cur_obj = cur_objective + improvement
        if last_improvement == iterations:
            break
        else:
            last_improvement = iterations
    return best_a, best_objective

def explore_sequence_lengths(low=64, high=2048, step=64, divisions=10):
    """Explore different sequence lengths for potential global optimum."""
    best_sequence = None
    best_objective = np.inf
    for length in range(low, high+1, step):
        # Only proceed if length is appropriate
        config = np.random.rand(length) * np.sqrt(2 * length / compute_objective([1]*length))
        best_run = do_optimize(config, time_limit=1000)
        cur_obj = evaluate_sequence(best_run.tolist())
        if cur_obj < best_objective:
            best_sequence = best_run.copy()
            best_objective = cur_obj
    return best_sequence
