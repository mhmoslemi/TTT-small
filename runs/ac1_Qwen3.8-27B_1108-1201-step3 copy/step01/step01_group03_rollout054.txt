def fast_score(f):
    """Compute score using FFT-based convolution."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    n = len(f)
    nfft = 1
    while nfft < 2 * n - 1:
        nfft *= 2
    F = np.fft.rfft(f, nfft)
    conv = np.fft.irfft(F * F, nfft)[:2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))


def compute_conv(f):
    """Full linear convolution via FFT."""
    f = np.asarray(f, dtype=np.float64)
    n = len(f)
    nfft = 1
    while nfft < 2 * n - 1:

    F = np.fft.rfft(f, nfft)
    return np.fft.irfft(F * F, nfft)[:2 * n - 1]


def compute_conv_cross(f, g):
    """Cross-convolution f*g via FFT (linear in g)."""
    f = np.asarray(f, dtype=np.float64)
    g = np.asarray(g, dtype=np.float64)

    max sum(g) s.t. conv(f,g)[k] <= M for k in top_k lags, g >= 0.

    else:

    cols = []

            rows.append(idx)
            cols.append(m)
            data.append(f[k - m])

        return None

                      shape=(len(top_indices), n)).tocsr()
    c = -np.ones(n)
    b = np.full(len(top_indices), M)
    bounds = [(0.0, None)] * n

                      options={'time_limit': time_limit, 'presolve': True})

            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:

    except Exception:
        pass
    return None


def subgradient_step(f, alpha, top_k=1):
    """One subgradient step on max conv(f,f)."""
    conv = compute_conv(f)
    n = len(f)

FFT-accelerated LP and subgradient optimization for minimizing the peak of
the linear self-convolution of a non-negative sequence.
Score = 2*n*max(conv(f,f))/(sum(f))^2. Scale invariant.

Key improvements over previous approach:
1. FFT-based convolution (O(n log n) vs O(n^2)) enabling larger n and more iterations
2. Reduced LP: only constrain top-k lags instead of all 2n-1
3. Adaptive line search for step size
4. Subgradient fallback when LP is infeasible/slow
5. Multiple diverse initializations and lengths
6. Spectral-shaping initialization for better starting points

import numpy as np
import scipy.sparse as sp

import time


def fft_convolve(a, b=None):
    """Full linear convolution using FFT. If b is None, compute a*a."""
    a = np.asarray(a, dtype=np.float64)
    if b is None:
        b = a
    b = np.asarray(b, dtype=np.float64)
    n = len(a)
    m = len(b)
    nfft = 1
    while nfft < n + m - 1:
        nfft *= 2
    FA = np.fft.rfft(a, nfft)
    FB = np.fft.rfft(b, nfft)
    return np.fft.irfft(FA * FB, nfft)[:n + m - 1]


def fast_score(f):
    """Compute score = 2*n*max(conv(f,f))/(sum(f))^2 using FFT."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    n = len(f)
    conv = fft_convolve(f)
    return float(2.0 * n * conv.max() / (s * s))


def reduced_lp_direction(f, M, top_k=100, time_limit=3.0):
    """
    Solve reduced LP: max sum(g) s.t. conv(f,g)[k] <= M for top-k lags, g >= 0.
    Returns normalized g (sum=1) or None.
    """
    n = len(f)
    conv_ff = fft_convolve(f)
    total_lags = 2 * n - 1
    if top_k >= total_lags:
        top_indices = np.arange(total_lags)
    else:
        top_indices = np.argsort(conv_ff)[::-1][:top_k]

    # Build sparse A (top_k x n): A[i, m] = f[k_i - m] if 0 <= k_i - m < n
    rows_list = []
    cols_list = []
    vals_list = []
    for idx, k in enumerate(top_indices):
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_hi >= m_lo:
            ms = np.arange(m_lo, m_hi + 1)
            rows_list.append(np.full(len(ms), idx))
            cols_list.append(ms)
            vals_list.append(f[k - ms])

    if not rows_list:
        return None

    rows_all = np.concatenate(rows_list)
    cols_all = np.concatenate(cols_list)
    vals_all = np.concatenate(vals_list)

    A = sp.coo_matrix((vals_all, (rows_all, cols_all)),
                      shape=(len(top_indices), n)).tocsr()
    c = -np.ones(n)
    b_vec = np.full(len(top_indices), M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def subgradient_step(f, alpha):
    """One subgradient descent step on max_k conv(f,f)[k]."""
    conv = fft_convolve(f)
    n = len(f)
    k_star = int(np.argmax(conv))
    # subgradient w.r.t. f[i] is 2*f[k_star - i] if 0 <= k_star - i < n
    s = np.zeros(n)
    valid_i = np.arange(n)
    vals = k_star - valid_i
    mask = (vals >= 0) & (vals < n)
    s[mask] = 2.0 * f[vals[mask]]
    f_new = f - alpha * s
    f_new = np.maximum(f_new, 0.0)
    s_sum = f_new.sum()
    if s_sum < 1e-10:
        return f
    return f_new / s_sum


def make_initial(n, rng, method='random'):
    """Generate a diverse non-negative initial sequence normalized to sum 1."""
    if method == 'uniform':
        a = np.ones(n)
    elif method == 'perturbed_uniform':
        a = 1.0 + rng.normal(0, 0.3, n)
    elif method == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
    elif method == 'sin2':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2
    elif method == 'random':
        a = rng.random(n)
    elif method == 'bimodal':
        x = np.arange(n, dtype=np.float64)
        sig = max(1.0, n / 12.0)
        a = np.exp(-0.5 * ((x - 0.25 * n) / sig) ** 2) + \
            np.exp(-0.5 * ((x - 0.75 * n) / sig) ** 2)
    elif method == 'alternating':
        a = np.ones(n)
        a[::2] = 0.4
    elif method == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
        a = np.maximum(a, 0.01)
    elif method == 'spectral_flat':
        # Design in frequency domain for flat |F|^2
        nfft = 1
        while nfft < 2 * n:
            nfft *= 2
        # Random phase, constant magnitude
        phase = rng.uniform(0, 2 * np.pi, nfft // 2 + 1)
        mag = np.ones(nfft // 2 + 1)
        mag[0] = 0  # zero DC to center
        F = mag * np.exp(1j * phase)
        F[0] = 0
        # Make it conjugate symmetric for real output
        F_full = np.zeros(nfft, dtype=np.complex128)
        F_full[:nfft // 2 + 1] = F
        for k in range(1, nfft // 2):
            F_full[nfft - k] = np.conj(F[k])
        a = np.fft.irfft(F_full, nfft)[:n].real
        a = np.maximum(a, 0.01)
    elif method == 'taylor_like':
        x = np.arange(n, dtype=np.float64) - n / 2.0
        a = 1.0 - 0.5 * (1 + np.tanh(4.0 * np.abs(x) / n))
        a = np.maximum(a, 0.02)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.005)
    a = a / a.sum()
    return a


def block_perturb(f, rng, width_max=None, sigma=0.3):
    """Random multiplicative block perturbation, renormalized."""
    n = len(f)
    cand = f.copy()
    if width_max is None:
        width_max = max(2, n // 10)
    w = int(rng.integers(1, min(width_max, n)))
    start = int(rng.integers(0, n - w + 1))
    cand[start:start + w] *= float(np.exp(rng.normal(0, sigma)))
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s < 1e-10:
        return f
    return cand / s


def full_perturb(f, rng, sigma=0.5):
    """Full multiplicative perturbation."""
    cand = f * np.exp(rng.normal(0, sigma, len(f)))
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s < 1e-10:
        return f
    return cand / s


def line_search_step(f, g, n_trials=6):
    """Find best interpolation t: f_new = (1-t)*f + t*g, minimize score."""
    best_t = 0.0
    best_sc = fast_score(f)
    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.75]:
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        sc = fast_score(f_new)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    return best_t, best_sc


def optimize_sequence(f_init, deadline, top_k=80, lp_time=2.0, verbose=True):
    """
    Main optimization loop: alternating LP steps, subgradient fallback,
    periodic perturbations. Returns (best_f, best_score).
    """
    f = f_init.copy()
    sc = fast_score(f)
    best_f = f.copy()
    best_sc = sc
    n = len(f)
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        conv = fft_convolve(f)
        M = float(conv.max())

        # Try LP
        g = reduced_lp_direction(f, M, top_k=top_k, time_limit=lp_time)

        if g is not None:
            t_best, sc_new = line_search_step(f, g)
            if t_best > 0:
                f_new = (1.0 - t_best) * f + t_best * g
                f_new = np.maximum(f_new, 0.0)
                s = f_new.sum()
                if s > 1e-10:
                    f_new = f_new / s
                    sc_new = fast_score(f_new)
                else:
                    f_new = f
                    sc_new = sc
            else:
                f_new = f
                sc_new = sc
        else:
            # Subgradient fallback
            alpha = 0.1 / (it + 10)
            f_new = subgradient_step(f, alpha)
            sc_new = fast_score(f_new)

        if sc_new < best_sc - 1e-9:
            best_sc = sc_new
            best_f = f_new.copy()
            stall = 0
            if verbose and it % 100 == 0:
                print("    it=%d score=%.6f best=%.6f" % (it, sc_new, best_sc),
                      flush=True)
        else:
            stall += 1

        f = f_new if np.isfinite(sc_new) else f

        # Periodic perturbation to escape stagnation
        if stall > 20:
            pert_type = rng_pert = np.random.default_rng(5000 + it)
            if stall > 50:
                f = full_perturb(f, pert_type, sigma=0.4)
            else:
                f = block_perturb(f, pert_type,
                                  width_max=max(3, n // 8), sigma=0.35)
            stall = 0

    return best_f, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Search for the best non-negative sequence minimizing
    score = 2*n*max(conv(f,f))/(sum(f))^2.
    Uses FFT-accelerated LP + subgradient optimization with diverse starts.
    """
    deadline = time.monotonic() + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(seed)

    # Initialize from parent if available
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64),
                         0.0, 1000.0)
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)

    best = np.maximum(parent, 0.0).copy()
    best_sum = best.sum()
    if best_sum > 0.01:
        best = best / best_sum
    bs = fast_score(best)
    if not np.isfinite(bs):
        bs = float(evaluate_sequence(best.tolist()))
    print("Initial: score=%.6f, n=%d" % (bs, best.size), flush=True)

    # === Phase 1: Optimize the parent (if reasonable size) ===
    if best.size <= 6000 and time.monotonic() < deadline - 60:
        p1_deadline = min(deadline - 60, time.monotonic() + 240)
        print("[Phase 1] LP optimize parent n=%d..." % best.size, flush=True)
        b1, s1 = optimize_sequence(best, p1_deadline,
                                   top_k=100, lp_time=2.5, verbose=True)
        if s1 < bs:
            bs = s1
            best = b1
            print("  Phase 1: score=%.6f" % bs, flush=True)

    # === Phase 2: Multi-start with diverse initializations ===
    lengths = [256, 512, 1024, 2048, 4096]
    methods = ['perturbed_uniform', 'random', 'hann', 'bimodal',
               'spectral_flat', 'alternating']

    # Filter out lengths already tried in phase 1
    if best.size in lengths:
        lengths.remove(best.size)

    # Sort by expected promise: moderate lengths first
    lengths.sort(key=lambda x: abs(x - 1024))

    # Time allocation
    remaining = deadline - time.monotonic() - 90  # leave 90s for final polish
    n_combos = min(len(lengths) * len(methods), 12)
    time_per = max(15.0, remaining / max(1, n_combos))

    combo_count = 0
    for n in lengths:
        for m in methods:
            if time.monotonic() > deadline - 90:
                break
            combo_count += 1
            if combo_count > n_combos:
                break
            start = make_initial(n, rng, m)
            t_budget = min(time_per, deadline - time.monotonic() - 90)
            if t_budget < 8:
                break
            print("[Phase 2] n=%d method=%s budget=%.0fs" %
                  (n, m, t_budget), flush=True)
            b2, s2 = optimize_sequence(start, time.monotonic() + t_budget,
                                       top_k=80, lp_time=1.5, verbose=False)
            if s2 < bs:
                bs = s2
                best = b2
                print("  ** New best: %.6f (n=%d, %s)" % (bs, n, m),
                      flush=True)
        if time.monotonic() > deadline - 90:
            break

    # === Phase 3: Final polish on the best sequence ===
    remaining = deadline - time.monotonic()
    if remaining > 30 and best.size <= 8000:
        print("[Phase 3] Final polish (%.0fs remaining, n=%d)" %
              (remaining, best.size), flush=True)
        p3_deadline = time.monotonic() + remaining * 0.7
        b3, s3 = optimize_sequence(best, p3_deadline,
                                   top_k=60, lp_time=1.0, verbose=False)
        if s3 < bs:
            bs = s3
            best = b3
            print("  Phase 3a: score=%.6f" % bs, flush=True)

        # Subgradient polish on remaining time
        if time.monotonic() < deadline - 5:
            f_polish = best.copy()
            sc_polish = bs
            n_pol = len(f_polish)
            it = 0
            while time.monotonic() < deadline - 3:
                it += 1
                alpha = 0.3 / (it + 50)
                f_new = subgradient_step(f_polish, alpha)
                sc_new = fast_score(f_new)
                if sc_new < sc_polish - 1e-10:
                    sc_polish = sc_new
                    f_polish = f_new
                    if sc_polish < bs:
                        bs = sc_polish
                        best = f_polish.copy()
                elif it % 200 == 0:
                    f_polish = block_perturb(f_polish,
                                             np.random.default_rng(it),
                                             width_max=max(2, n_pol // 25),
                                             sigma=0.2)
            if sc_polish < bs:
                bs = sc_polish
                best = f_polish

    # === Phase 4: Quick local search with block perturbations ===
    if time.monotonic() < deadline - 2:
        n_best = best.size
        f_ls = best.copy()
        sc_ls = bs
        rng_ls = np.random.default_rng(seed + 999)
        it = 0
        while time.monotonic() < deadline - 1:
            it += 1
            cand = block_perturb(f_ls, rng_ls,
                                 width_max=max(1, n_best // 30), sigma=0.15)
            sc_cand = fast_score(cand)
            if sc_cand < sc_ls - 1e-10:
                sc_ls = sc_cand
                f_ls = cand
                if sc_ls < bs:
                    bs = sc_ls
                    best = cand.copy()
            elif it % 500 == 0:
                f_ls = best.copy()

    # Final validation
    best = np.clip(np.asarray(best, dtype=np.float64), 0.0, 1000.0)
    if best.size == 0 or not np.isfinite(best).all() or best.sum() < 0.01:
        best = np.ones(256, dtype=np.float64)

    final_sc = float(evaluate_sequence(best.tolist()))
    if np.isfinite(final_sc):
        print("FINAL: score=%.6f, n=%d" % (final_sc, best.size), flush=True)
    else:
        print("FINAL (fallback): score=%.6f, n=%d" % (bs, best.size),
              flush=True)

    return [float(x) for x in best]
