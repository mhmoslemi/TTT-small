- After LP with new f: 2n * M* / (sum g*)^2

Where M* is the optimal value of the minimax LP.

If we normalize g* to sum 1, the new score is 2n * M*.

```python
def minimax_lp(f, M_init, time_limit=10):
    """Solve min M s.t. (f*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M_opt) or None."""
    n = len(f)
    # A: (2n-1) x n matrix, A[k,j] = f[k-j] if 0<=k-j<n
    # Constraints: A@ g <= M (but M is a variable too!)
    # This is: minimize M
    # subject to A@g - M*ones <= 0
    #            sum(g) = 1
    #            g >= 0
    #            M >= 0
    
    # Using scipy linprog:
    # Variables: [g_0, ..., g_{n-1}, M] (total n+1 variables)
    # Objective: minimize M -> c = [0,...,0,1]
    # Inequality: A@g - M <= 0 for each row -> [A | -I] @ [g; M] <= 0
    # Equality: sum(g) = 1 -> [ones | 0] @ [g; M] = 1
    # Bounds: g >= 0, M >= 0
    
    # Build A
    A = build_conv_matrix(n, f)  # (2n-1) x n
    
    # Full constraint matrix: (2n-1) x (n+1)
    # A_full[:, :-1] = A, A_full[:, -1] = -1
    A_full = sp.hstack([A, -sp.eye(2*n-1)]).tocsr()
    
    c = np.zeros(n+1)
    c[-1] = 1  # minimize M
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1
    b_eq = np.array([1.0])
    
    bounds = [(0, None)]*(n+1)
    
    res = linprog(c, A_ub=A_full, b_ub=np.zeros(2*n-1),
                  A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                  method='highs', options={'time_limit': time_limit})
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        M_opt = max(0, res.x[-1])
        return g, M_opt
    return None
```

"""
Hybrid optimizer for minimizing 2*n*max(f*f)/(sum(f))^2 over non-negative f.
Combines cyclic coordinate descent (exploiting the 1D convex structure of
each coordinate), LP-based global moves (minimax formulation), and structured
random perturbations for exploration.
Key insight: symmetric sequences cannot beat score 2 (by Cauchy-Schwarz),
so the search focuses on asymmetric structures.
"""
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog, minimize_scalar
import time


def fast_score(f):
    """Compute score = 2*n*max(conv)/sum^2."""
    f = np.asarray(f, dtype=np.float64)
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * n * conv.max() / (s * s))


def fft_convolve(f):
    """Compute full linear convolution using FFT."""
    n = len(f)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, N)
    return np.fft.irfft(F * F, N)[:size]


def build_conv_matrix(n, f):
    """Sparse (2n-1) x n matrix A with A[k,j] = f[k-j] if valid."""
    i_idx = np.tile(np.arange(n), n)
    j_idx = np.repeat(np.arange(n), n)
    row_idx = j_idx + i_idx
    data = f[i_idx]
    A = sp.coo_matrix((data, (row_idx, j_idx)), shape=(2*n-1, n)).tocsr()
    return A


def minimax_lp(f, time_limit=15.0):
    """Solve min M s.t. (f*g)[k] <= M, sum(g)=1, g>=0. Returns (g, M) or None."""
    n = len(f)
    A = build_conv_matrix(n, f)
    A_full = sp.hstack([A, -sp.eye(2*n-1)]).tocsr()
    c = np.zeros(n+1)
    c[-1] = 1.0
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n+1)
    try:
        res = linprog(c, A_ub=A_full, b_ub=np.zeros(2*n-1),
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                      method='highs', options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0.0)
            M_opt = max(0.0, float(res.x[-1]))
            return g, M_opt
    except Exception as e:
        print("  LP error: %s" % str(e))
    return None


def _coord_obj(x, conv_base_aff, c_aff, d_aff, max_unchanged, S, n):
    """Objective for 1D coordinate minimization."""
    vals = conv_base_aff + c_aff * x + d_aff * x * x
    max_total = max(max_unchanged, float(np.max(vals)))
    s = S + x
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_total / (s * s)


def coord_descent_pass(f, alpha=1.0, max_iter=40, tol=1e-8):
    """One pass of cyclic coordinate descent. Returns updated f and improvement."""
    n = len(f)
    f = f.copy()
    conv = fft_convolve(f)
    total_improve = 0.0
    
    for i in range(n):
        S = f.sum() - f[i]
        if S < 1e-10:
            continue
        old_ci = f[i]
        
        # Affected range
        lo = i
        hi = min(2*n-2, i + n - 1)
        aff_ks = np.arange(lo, hi+1)
        aff_js = aff_ks - i  # 0, 1, ..., len(aff_ks)-1
        
        # max over unchanged entries
        mask = np.ones(2*n-1, dtype=bool)
        mask[lo:hi+1] = False
        if mask.any():
            max_unchanged = float(np.max(conv[mask]))
        else:
            max_unchanged = 0.0
        
        # Build coefficient arrays
        m = len(aff_ks)
        conv_base_aff = np.empty(m)
        c_aff = np.empty(m)
        d_aff = np.empty(m)
        for idx in range(m):
            k = aff_ks[idx]
            j = aff_js[idx]
            if j == i:
                conv_base_aff[idx] = conv[k] - old_ci * old_ci
                c_aff[idx] = 0.0
                d_aff[idx] = 1.0
            else:
                conv_base_aff[idx] = conv[k] - 2.0 * old_ci * f[j]
                c_aff[idx] = 2.0 * f[j]
                d_aff[idx] = 0.0
        
        # Current value
        cur_val = _coord_obj(old_ci, conv_base_aff, c_aff, d_aff, max_unchanged, S, n)
        
        # Upper bound for x
        x_max = max(10.0 * old_ci + 1.0, 5.0 * S / n + 1.0)
        
        res = minimize_scalar(_coord_obj,
                              args=(conv_base_aff, c_aff, d_aff, max_unchanged, S, n),
                              bounds=(0.0, x_max), method='bounded',
                              options={'maxiter': max_iter, 'xatol': tol})
        
        if res.fun < cur_val - 1e-12:
            new_x = max(0.0, min(alpha * res.x + (1.0 - alpha) * old_ci, 1000.0))
            if abs(new_x - old_ci) > 1e-12:
                delta = new_x - old_ci
                # Update conv incrementally
                for idx in range(m):
                    k = aff_ks[idx]
                    j = aff_js[idx]
                    if j == i:
                        conv[k] += 2.0 * old_ci * delta + delta * delta
                    else:
                        conv[k] += 2.0 * delta * f[j]
                f[i] = new_x
                total_improve += cur_val - res.fun
    
    return f, total_improve


def make_initial(n, rng, method):
    """Generate initial sequence."""
    if method == 'uniform':
        a = np.ones(n)
    elif method == 'ramp':
        a = np.linspace(0.1, 1.0, n)
    elif method == 'ramp_rev':
        a = np.linspace(1.0, 0.1, n)
    elif method == 'exponential':
        a = np.exp(-np.arange(n, dtype=np.float64) / max(1, n // 3))
    elif method == 'exp_rev':
        a = np.exp(-np.arange(n-1, -1, -1, dtype=np.float64) / max(1, n // 3))
    elif method == 'two_camps':
        a = np.ones(n)
        a[:n//2] = 1.0
        a[n//2:] = 0.3
    elif method == 'two_camps_rev':
        a = np.ones(n)
        a[:n//2] = 0.3
        a[n//2:] = 1.0
    elif method == 'skewed':
        x = np.linspace(0, 2*np.pi, n)
        a = (0.5 + 0.5*np.cos(x))**2
    elif method == 'skewed2':
        x = np.linspace(0, 2*np.pi, n)
        a = (0.5 + 0.5*np.cos(x + 0.5))**2
    elif method == 'random':
        a = rng.random(n) + 0.01
    elif method == 'perturbed_uniform':
        a = np.ones(n) + rng.normal(0, 0.3, n)
        a = np.maximum(a, 0.01)
    elif method == 'cosine_asym':
        x = np.linspace(0, np.pi*0.7, n)
        a = 0.5 * (1 + np.cos(x))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.001)
    a = a / a.sum()
    return a


def block_perturb(f, rng, width=None):
    """Random multiplicative block perturbation."""
    n = len(f)
    cand = f.copy()
    if width is None:
        width = int(rng.integers(1, max(2, n // 10)))
    width = min(width, n)
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.3)))
    cand[start:start+width] *= factor
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s < 1e-10:
        return f
    return cand / s


def run_cd_phase(f, budget_s, n_passes_max=200, alpha=0.9):

    f = f.copy()
    best_f = f.copy()

    stall = 0

            best_sc = sc
            best_f = f.copy()
            stall = 0
        else:
            stall += 1
        if stall > 15:
            break
    
    return best_f, best_sc


def run_lp_cd_hybrid(f, budget_s, lp_time=15.0):
    """Alternate LP steps and CD passes."""
    f = f.copy()
    best_f = f.copy()
    best_sc = fast_score(f)
    
    while time.monotonic() < budget_s:

                # Relax
