"""
Optimizes a non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses soft-max smoothed gradient descent with momentum, FFT-based gradient computation,
and LP-based refinement.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def compute_gradient_hard(a, m_star):
    """Compute subgradient of c[m_star] w.r.t. a, projected to sum=0."""
    n = len(a)
    ks = np.arange(n)
    idxs = m_star - ks
    valid = (idxs >= 0) & (idxs < n)
    grad = np.zeros(n)
    grad[valid] = a[idxs[valid]] + a[ks[valid]]
    grad -= grad.mean()
    return grad


def compute_gradient_soft(a, w):
    """Compute gradient of soft-max using precomputed weights w over c."""
    n = len(a)
    # T1[k] = sum_j w[j+k] * a[j] = (w_rev * a)[2n-2-k]
    w_rev = w[::-1]
    T1_full = np.convolve(w_rev, a, 'full')
    T1 = T1_full[2*n-2 - np.arange(n)]
    
    # T2[k] = a[k] * sum_{m in window(k)} w[m]
    w_cs = np.concatenate([[0.0], np.cumsum(w)])
    lo = np.arange(n)
    hi = np.minimum(2*n-2, np.arange(n) + n - 1) + 1
    W_cum = w_cs[hi] - w_cs[lo]
    
    grad = T1 + a * W_cum
    grad -= grad.mean()
    return grad


def soft_vmax(c, tau):
    """Compute soft-max weights and value."""
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    soft_val = c.max() + np.log(w_sum) / tau
    return w / w_sum, soft_val


def normalize(a, target=None):
    if target is None:
        target = 1.0
    S = a.sum()
    if S < 1e-12:
        return a
    return a * (target / S)


def lp_solve_peak(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    if n > 1500:
        return None
    c_obj = -np.ones(n)
    # Build constraint matrix (banded)
    n_rows = 2 * n - 1
    # Use sparse-like approach: build row by row
    # For efficiency, build as dense for n <= 1500
    A = np.zeros((n_rows, n))
    for m in range(n_rows):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': 30})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def optimize_from_start(a0, deadline, max_iters=5000):
    """Optimize a sequence using soft-max gradient + hard-max gradient + LP."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = normalize(a, 1.0)
    
    best = a.copy()
    best_score = compute_obj(a)
    
    # Phase 1: Soft-max gradient with momentum
    # Anneal tau from low to high
    tau_start = 5.0
    tau_end = 50.0
    momentum = 0.9
    v = np.zeros(n)
    
    n_phase1 = max_iters // 2
    for it in range(n_phase1):
        if time.monotonic() > deadline - 30:
            break
        
        # Anneal tau
        frac = it / max(n_phase1 - 1, 1)
        tau = tau_start + (tau_end - tau_start) * frac
        
        c = np.convolve(a, a)
        w, _ = soft_vmax(c, tau)
        grad = compute_gradient_soft(a, w)
        
        # Momentum update
        v = momentum * v + grad
        # Adaptive step size
        lr = 0.5 / (np.linalg.norm(v) + 1e-8)
        lr = min(lr, 0.5)
        
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, 1.0)
        
        # Check improvement
        s = compute_obj(a_new)
        if s < best_score:
            best_score = s
            best = a_new.copy()
        a = a_new
    
    # Phase 2: Hard-max gradient (refine)
    n_phase2 = max_iters // 4
    for it in range(n_phase2):
        if time.monotonic() > deadline - 20:
            break
        
        c = np.convolve(a, a)
        # Find all peaks above 99th percentile
        threshold = np.percentile(c, 99)
        peak_mask = c >= threshold
        
        # Sum gradients from all active peaks
        grad = np.zeros(n)
        for m in np.where(peak_mask)[0]:
            grad += compute_gradient_hard(a, m)
        grad /= max(peak_mask.sum(), 1)
        
        lr = 0.1 / (np.linalg.norm(grad) + 1e-8)
        lr = min(lr, 0.3)
        
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, 1.0)
        
        s = compute_obj(a_new)
        if s < best_score:
            best_score = s
            best = a_new.copy()
        a = a_new
    
    # Phase 3: LP refinement
    if n <= 1024:
        n_lp = max_iters // 4
        for it in range(n_lp):
            if time.monotonic() > deadline - 10:
                break
            
            c = np.convolve(a, a)
            M = float(c.max()) * 1.001  # Slightly above current peak
            
            g = lp_solve_peak(a, M)
            if g is None:
                break
            
            # Blend
            for t in [0.7, 0.5, 0.3]:
                a_new = (1 - t) * a + t * g
                a_new = np.maximum(a_new, 0.0)
                a_new = normalize(a_new, 1.0)
                s = compute_obj(a_new)
                if s < best_score:
                    a = a_new
                    best = a_new.copy()
                    best_score = s
                    break
            else:
                # No improvement, try gradient
                c = np.convolve(a, a)
                m_star = c.argmax()
                grad = compute_gradient_hard(a, m_star)
                lr = 0.05 / (np.linalg.norm(grad) + 1e-8)
                a = np.maximum(a - lr * grad, 0.0)
                a = normalize(a, 1.0)
                s = compute_obj(a)
                if s < best_score:
                    best = a.copy()
                    best_score = s
    
    return best, best_score

Optimizes a non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: Multi-start soft-max gradient descent with Nesterov momentum,
adaptive step size, multi-peak subgradients, and LP-based refinement.
Includes restart mechanism to escape local minima.

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * c.max() / (S * S))


def normalize(a, target=1.0):
    S = a.sum()
    if S < 1e-12:
        return np.ones_like(a) * target / len(a)
    return a * (target / S)


def grad_hard(a, m_star):
    n = len(a)
    ks = np.arange(n)
    idxs = m_star - ks
    valid = (idxs >= 0) & (idxs < n)
    g = np.zeros(n)
    g[valid] = a[idxs[valid]] + a[ks[valid]]
    g -= g.mean()
    return g


def grad_multi_peak(a, c, pct=95):
    """Gradient from all peaks above the given percentile."""
    n = len(a)
    thresh = np.percentile(c, pct)
    peaks = np.where(c >= thresh)[0]
    if len(peaks) == 0:
        return grad_hard(a, int(c.argmax()))
    g = np.zeros(n)
    for m in peaks:
        g += grad_hard(a, m)
    g /= len(peaks)
    return g


def grad_soft(a, c, tau):
    """Soft-max gradient using FFT-based convolution."""
    n = len(a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)

                    best_score = s_new
                    improved = True
                    break
            if not improved:
                c = np.convolve(a, a)
                m_star = int(c.argmax())
                g = grad_hard_single(a, m_star)
                lr = 0.03 / (np.linalg.norm(g) + 1e-10)
                a = np.maximum(a - lr * g, 0.0)
                a = normalize(a, S_t)
                s = compute_obj(a)
                if s < best_score:
                    best = a.copy()
                    best_score = s

    return best, best_score


def optimize_with_restarts(a0, deadline, rng, n_restarts=3):
    """Try multiple restarts from a0 and return the best."""
    best_seq = None
    best_score = float('inf')
    n = len(a0)

    for r in range(n_restarts):
        if time.monotonic() > deadline - 30:
            break
        if r == 0:
            start = a0.copy()
        else:
            # Perturb the current best or original
            base = best_seq if best_seq is not None else a0
            scale = 0.15 * (r + 1) / n_restarts
            start = base + rng.standard_normal(n) * scale
            start = np.maximum(start, 0.0)
            start = normalize(start, np.sqrt(2.0 * n))

        seq, score = optimize_single(start, deadline - 10,
                                     n_iter=max(1000, int(3000 / (n_restarts))),
                                     rng=rng)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"    Restart {r}: score={score:.6f}")

    if best_seq is None:
        best_seq = a0
        best_score = compute_obj(a0)
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Explores multiple starting sequences of various lengths
    using soft-max gradient descent, multi-peak subgradients, and LP refinement.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 12.0)

    best_seq = None
    best_score = float('inf')

    print(f"Budget: {budget_s}s, deadline in {deadline - t_start:.0f}s")

    # Build list of (length, kind) to try
    configs = []

    # Try to use height_sequence_1
    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64)
        if hs.size >= 16 and np.all(np.isfinite(hs)):
            for L in [256, 384, 512, 768, min(1024, len(hs))]:
                if L <= len(hs):
                    configs.append(('hs', L, hs[:L].copy()))
    except NameError:
        pass

    # Various structured and random starts
    for n in [256, 384, 512, 768]:
        for kind in ['uniform', 'u_shape', 'random', 'edge_heavy']:
            configs.append((kind, n, None))

    # Sort by length (shorter first for quick wins)
    configs.sort(key=lambda x: x[1])

    # Time allocation: give more time to more promising configs
    # First pass: quick evaluation of all configs
    print(f"Trying {len(configs)} configurations...")

    for cfg in configs:
        if time.monotonic() > deadline - 45:
            break
        kind, n, prebuilt = cfg

        if prebuilt is not None:
            a0 = prebuilt
        else:
            a0 = make_start(n, kind, rng)
        a0 = normalize(a0)

        # Quick optimization (shorter budget for exploration)
        quick_deadline = min(deadline - 30, time.monotonic() + 120.0)
        seq, score = optimize_single(a0, quick_deadline, n_iter=2000, rng=rng)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  [{kind}/{n}] score={score:.6f}")

    # Second pass: intensive optimization of best candidates
    print(f"\nBest after exploration: {best_score:.6f} (n={len(best_seq) if best_seq is not None else 0})")

    if best_seq is not None and time.monotonic() < deadline - 30:
        # Intensive optimization of the best
        n_best = len(best_seq)
        print(f"Intensive optimization of best (n={n_best})...")
        seq, score = optimize_with_restarts(
            best_seq, deadline - 10, rng, n_restarts=3)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  After intensive: {best_score:.6f}")

        # Also try a fresh intensive run from a different start at same length
        if time.monotonic() < deadline - 40:
            a0 = make_start(n_best, 'u_shape', rng)
            a0 = normalize(a0)
            seq, score = optimize_single(a0, deadline - 20, n_iter=8000, rng=rng)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  Fresh u_shape ({n_best}): {best_score:.6f}")

        # Try a longer length if time permits
        if time.monotonic() < deadline - 60:
            n_long = min(1024, max(768, n_best + 128))
            a0 = make_start(n_long, 'uniform', rng)
            a0 = normalize(a0)
            seq, score = optimize_single(a0, deadline - 20, n_iter=5000, rng=rng)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  Long uniform ({n_long}): {score:.6f}")

    # Final refinement
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\nFinal refinement (score={best_score:.6f})...")
        n = len(best_seq)
        # A few more LP iterations
        a = best_seq.copy()
        S_t = np.sqrt(2.0 * n)
        for it in range(20):
            if time.monotonic() > deadline - 5:
                break
            c = np.convolve(a, a)
            M = float(c.max()) * 1.0001
            g = lp_direction(a, M, time_limit=10)
            if g is None:
                break
            for t in [0.5, 0.3, 0.1]:
                a_new = (1 - t) * a + t * g
                a_new = np.maximum(a_new, 0.0)
                a_new = normalize(a_new, S_t)
                s_new = compute_obj(a_new)
                if s_new < best_score:
                    a = a_new
                    best_seq = a_new.copy()
                    best_score = s_new
                    break
        print(f"  Final score: {best_score:.6f}")

    # Validate
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"\nFinal: n={len(best_seq)}, score={final_score:.6f}")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence: {eval_score:.6f}")
        except Exception as e:
            print(f"eval error: {e}")
        return [float(x) for x in best_seq]

    # Fallback
    print("Falling back to uniform 256.")
    return [1.0] * 256
