"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses LP-based iterative refinement (direct peak minimization), FISTA for
pre-processing, and combinatorial initializations.
"""

def _compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def _normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)

def _lp_minimize_peak(f):
    """
    Solve LP: minimize t s.t. conv(f,g) <= t (elementwise), sum(g) = sqrt(2n), g >= 0.
    Returns optimal g, or None if infeasible.
    """
    n = len(f)
    S_t = np.sqrt(2.0 * n)
    
    # Variables: [g_0, ..., g_{n-1}, t]
    # Objective: minimize t -> c = [0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: conv(f,g)[k] - t <= 0 for k = 0, ..., 2n-2
    # conv(f,g)[k] = sum_j f[k-j] * g[j] where 0 <= k-j < n and 0 <= j < n
    # So: sum_j f[k-j] * g[j] - t <= 0
    # In matrix form: [A_k, -1] * [g; t] <= 0
    
    m = 2 * n - 1
    A_ub = np.zeros((m, n + 1))
    for k in range(m):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A_ub[k, j] = f[k - j]
        A_ub[k, n] = -1.0  # coefficient of t
    
    b_ub = np.zeros(m)
    
    # Equality constraint: sum(g) = S_t
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S_t])
    
    # Bounds: g_i >= 0, t free
    bounds = [(0.0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n], res.x[n]
    except Exception:
        pass
    return None, None

def _fista_step(a, it, lr_init=0.1):
    """One FISTA step for minimizing peak autocorrelation with fixed sum."""
    n = len(a)
    c = np.convolve(a, a)
    m_star = int(c.argmax())
    
    # Subgradient of max(conv(a,a)) w.r.t. a
    # d/da_j max_k sum_i a[i]*a[k-i] = sum_i a[i] * 1[k = i+j] ... 
    # Actually: conv(a,a)[k] = sum_i a[i] * a[k-i]
    # d conv(a,a)[k] / d a[j] = a[k-j] + a[j-k] (but only for valid indices)
    # For the max, we just need the contribution from the argmax k:
    # d max / d a[j] = a[m_star - j] + a[j - m_star] (valid when indices are in range)
    
    grad = np.zeros(n)
    for j in range(n):
        idx1 = m_star - j
        idx2 = j - m_star
        if 0 <= idx1 < n:
            grad[j] += a[idx1]
        if 0 <= idx2 < n:
            grad[j] += a[idx2]
    
    # Project out the mean (to maintain fixed sum)
    grad -= grad.mean()
    
    # FISTA momentum
    # This is simplified; full FISTA needs momentum tracking
    return a - lr_init * grad

def _fista_optimize(a0, deadline, max_iters=2000):
    """FISTA-style optimization with momentum."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = _normalize(a)
    score = _compute_obj(a)
    best = a.copy()
    best_score = score
    S_t = np.sqrt(2.0 * n)
    
    # FISTA variables
    y = a.copy()
    t_k = 1.0
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 10.0:
            break
        
        # Learning rate (decreasing)
        lr = 1.0 / (it + 1.0)
        
        # Gradient step on y
        c = np.convolve(y, y)
        m_star = int(c.argmax())
        grad = np.zeros(n)
        for j in range(n):
            idx1 = m_star - j
            idx2 = j - m_star
            if 0 <= idx1 < n:
                grad[j] += y[idx1]
            if 0 <= idx2 < n:
                grad[j] += y[idx2]
        grad -= grad.mean()
        
        # Gradient step
        a_new = y - lr * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize(a_new, S_t)
        
        s_new = _compute_obj(a_new)
        if s_new < best_score:
            best_score = s_new
            best = a_new.copy()
        
        # FISTA update
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a_new + ((t_k - 1.0) / t_next) * (a_new - a)
        y = np.maximum(y, 0.0)
        y = _normalize(y, S_t)
        a = a_new
        t_k = t_next
    
    return best, best_score

def _lp_minimize_peak_sparse(f):
    """
    Solve LP: minimize t s.t. conv(f,g) <= t (elementwise), sum(g) = sqrt(2n), g >= 0.
    Uses sparse matrix for efficiency.
    Returns (optimal_g, optimal_t) or (None, None) if infeasible.
    """
    from scipy.sparse import csr_matrix
    
    n = len(f)
    S_t = np.sqrt(2.0 * n)
    m = 2 * n - 1
    
    # Build A_ub sparsely
    # Row k: [f[k-j] for valid j, ..., -1 (for t)]
    rows = []
    cols = []
    vals = []
    
    for k in range(m):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
        # t coefficient
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(m, n + 1))
    b_ub = np.zeros(m)
    
    # A_eq: sum(g) = S_t
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S_t])
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    bounds = [(0.0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n], res.x[n]
    except Exception as e:
        print(f"  LP error: {e}")
        pass
    return None, None

"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses LP-based iterative refinement (direct peak minimization via linear programming),
FISTA with momentum for pre-processing, combinatorial and structured initializations,
and adaptive blending with restart mechanism to escape local minima.

import numpy as np
import time
from scipy.optimize import linprog


def _compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def _build_conv_matrix(n):
    """Build the dense convolution constraint matrix for sequences of length n.
    A[k, j] = 1 if j is a valid index for conv[f,g][k], else 0.
    Actually returns the structure: for given f, A_ub[k,j] = f[k-j] if valid.
    We return the mask and offset arrays for efficient computation.
    """
    m = 2 * n - 1
    # For row k, valid columns j are max(0, k-n+1) to min(n-1, k)
    # The value at (k,j) is f[k-j]
    lo_idx = []
    hi_idx = []
    for k in range(m):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        lo_idx.append(lo)
        hi_idx.append(hi)
    return np.array(lo_idx), np.array(hi_idx), m


def _lp_minimize_peak(f, lo_idx, hi_idx, m):
    """
    Solve LP: minimize t s.t. conv(f,g)[k] <= t for all k,
    sum(g) = sqrt(2n), g >= 0.
    Returns (optimal_g, optimal_t) or (None, None).
    """
    n = len(f)
    S_t = np.sqrt(2.0 * n)

    # Build A_ub densely (for n <= 1024 this is manageable)
    A_ub = np.zeros((m, n + 1))
    for k in range(m):
        lo = lo_idx[k]
        hi = hi_idx[k]
        if lo <= hi:
            js = np.arange(lo, hi + 1)
            A_ub[k, js] = f[k - js]
        A_ub[k, n] = -1.0

    b_ub = np.zeros(m)

    # Equality: sum(g) = S_t
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S_t])

    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    bounds = [(0.0, None)] * n + [(None, None)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n], float(res.x[n])
    except Exception:
        pass
    return None, None


def _grad_peak(a):
    """Compute subgradient of max(conv(a,a)) with respect to a."""
    n = len(a)
    c = np.convolve(a, a)
    m_star = int(c.argmax())
    grad = np.zeros(n)
    for j in range(n):
        idx1 = m_star - j
        idx2 = j - m_star
        if 0 <= idx1 < n:
            grad[j] += a[idx1]
        if 0 <= idx2 < n:
            grad[j] += a[idx2]
    return grad


def _fista_optimize(a0, deadline, max_iters=1000):
    """FISTA with momentum for minimizing peak autocorrelation (fixed sum)."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = _normalize(a)
    score = _compute_obj(a)
    best = a.copy()
    best_score = score
    S_t = np.sqrt(2.0 * n)

    y = a.copy()
    t_k = 1.0
    lr_base = np.sqrt(2.0 * n)  # Scale learning rate

    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break

        lr = lr_base / (it + 10.0)
        lr = min(lr, 1.0)

        # Gradient at y
        grad = _grad_peak(y)
        grad -= grad.mean()

        # Gradient step
        a_new = y - lr * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize(a_new, S_t)

        s_new = _compute_obj(a_new)
        if s_new < best_score:
            best_score = s_new
            best = a_new.copy()

        # FISTA update
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        momentum = (t_k - 1.0) / t_next
        y = a_new + momentum * (a_new - a)
        y = np.maximum(y, 0.0)
        y = _normalize(y, S_t)
        a = a_new
        t_k = t_next

    return best, best_score


def _lp_iterative_refine(a0, deadline, max_iters=200):
    """
    LP-based iterative refinement:
    Given f, solve LP to find g minimizing peak of conv(f,g).
    Update f = (1-blend)*f + blend*g with adaptive blending.
    Include restart mechanism.
    """
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = _normalize(a)
    score = _compute_obj(a)
    best = a.copy()
    best_score = score
    S_t = np.sqrt(2.0 * n)

    lo_idx, hi_idx, m = _build_conv_matrix(n)

    no_improve_count = 0
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() > deadline - 3.0:
            break

        # Adaptive blend: start aggressive, become conservative
        progress = it / max(max_iters, 1)
        blend_max = max(0.3, 1.0 - 0.7 * progress)

        g, t_val = _lp_minimize_peak(a, lo_idx, hi_idx, m)
        if g is None:
            break

        # Try multiple blend factors
        improved = False
        blend_candidates = [blend_max, 0.7 * blend_max, 0.4 * blend_max, 0.2 * blend_max]
        for blend in blend_candidates:
            a_new = (1.0 - blend) * a + blend * g
            a_new = np.maximum(a_new, 0.0)
            a_new = _normalize(a_new, S_t)
            s_new = _compute_obj(a_new)
            if s_new < best_score:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                no_improve_count = 0
                break

        if not improved:
            no_improve_count += 1

            if no_improve_count >= 15:
                # Restart: add perturbation
                perturbation = rng.normal(0, 0.1, n)
                a = a + perturbation
                a = np.maximum(a, 0.0)
                a = _normalize(a, S_t)
                no_improve_count = 0
            else:
                # Try a gradient step
                grad = _grad_peak(a)
                grad -= grad.mean()
                a = a - 0.05 * grad
                a = np.maximum(a, 0.0)
                a = _normalize(a, S_t)
                s = _compute_obj(a)
                if s < best_score:
                    best = a.copy()
                    best_score = s

        if (it + 1) % 20 == 0:
            print(f"    LP iter {it+1}: best_score={best_score:.6f}")

    return best, best_score


def _combined_optimize(a0, deadline, max_fista=500, max_lp=150):
    """Combined FISTA + LP optimization."""
    # Phase 1: FISTA
    seq_fista, score_fista = _fista_optimize(a0, min(deadline, time.monotonic() + 60),
                                             max_iters=max_fista)
    print(f"    FISTA: score={score_fista:.6f}")

    # Phase 2: LP refinement
    remaining_deadline = deadline
    seq_lp, score_lp = _lp_iterative_refine(seq_fista, remaining_deadline,
                                            max_iters=max_lp)
    print(f"    LP:    score={score_lp:.6f}")

    return seq_lp, score_lp


def _generate_initializations(rng, n):
    """Generate diverse initial sequences of length n."""
    i = np.arange(n, dtype=np.float64)
    starts = []

    # All ones
    starts.append(np.ones(n))

    # Random smooth
    x = rng.random(n)
    x = np.convolve(x, np.ones(5) / 5.0, mode='same')
    starts.append(np.maximum(x, 0.01))

    # Cosine (various frequencies)
    for freq in [2, 3, 5, 7]:
        starts.append(0.5 + 0.5 * np.cos(2.0 * np.pi * i * freq / n))

    # Binary random
    starts.append(rng.integers(0, 2, n).astype(float) + 0.05)

    # Linear ramp
    starts.append(np.linspace(1.0, 0.1, n))

    # Symmetric bump
    center = (n - 1) / 2.0
    starts.append(np.exp(-0.5 * ((i - center) / (n * 0.25)) ** 2))

    # Triangular
    starts.append(1.0 - np.abs(i - center) / center * 0.8)
    starts.append(np.maximum(starts[-1], 0.05))

    # Quadratic
    starts.append(1.0 - 0.7 * ((i - center) / center) ** 2)
    starts.append(np.maximum(starts[-1], 0.05))

    # Wavelet-like
    starts.append(np.abs(np.sin(2.0 * np.pi * i * 3.0 / n)) *
                  np.exp(-0.5 * ((i - center) / (n * 0.3)) ** 2) + 0.05)

    return starts


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Uses FISTA and LP-based iterative refinement
    to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
    Explores multiple lengths and initializations within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 12.0)

    best_seq = None
    best_score = float('inf')

    def _update_best(seq, score, label=""):
        nonlocal best_seq, best_score
        if score < best_score and seq is not None:
            best_score = score
            best_seq = np.array(seq, dtype=np.float64).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:.0f}s] New best: {label} score={score:.6f}, n={len(seq)}")

    # --- Try the provided starting sequence ---
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 8 and np.all(np.isfinite(gs)):
            for L in [256, 512, min(1024, len(gs))]:
                if L <= len(gs) and time.monotonic() < deadline - 120.0:
                    t_sub = min(deadline - 120.0, time.monotonic() + 200.0)
                    seq, score = _combined_optimize(gs[:L].copy(), t_sub,
                                                    max_fista=300, max_lp=100)
                    _update_best(seq, score, f"h1_n{L}")
    except NameError:
        pass

    # --- Try various lengths and initializations ---
    # Focus on n=512 (proven good) and n=256, n=1024
    configs = [
        # (n, num_starts, max_fista, max_lp, time_per_start)
        (256, 4, 400, 100, 80),
        (512, 5, 400, 120, 120),
        (1024, 3, 300, 80, 100),
    ]

    for n, num_starts, mf, ml, tps in configs:
        if time.monotonic() > deadline - 60.0:
            break

        starts = _generate_initializations(rng, n)
        # Shuffle and take num_starts
        indices = rng.permutation(len(starts))[:num_starts]

        for idx in indices:
            if time.monotonic() > deadline - 30.0:
                break
            t_start_this = time.monotonic()
            t_sub = min(deadline - 30.0, t_start_this + tps)
            a0 = starts[idx]
            seq, score = _combined_optimize(a0, t_sub, max_fista=mf, max_lp=ml)
            _update_best(seq, score, f"n{n}_s{idx}")

    # --- Final intensive refinement on the best sequence ---
    if best_seq is not None and time.monotonic() < deadline - 15.0:
        print(f"  Final refinement on best (score={best_score:.6f}, n={len(best_seq)})...")

        # Multiple rounds of LP refinement with decreasing time allocations
        remaining = deadline - time.monotonic()
        for round_i in range(4):
            if time.monotonic() > deadline - 8.0:
                break
            t_alloc = min(remaining * 0.4, 120.0)
            t_sub = time.monotonic() + t_alloc
            seq, score = _lp_iterative_refine(best_seq, t_sub, max_iters=80)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"    Refine round {round_i}: score={best_score:.6f}")
                remaining = deadline - time.monotonic()
            else:
                break

        # One more FISTA pass
        if time.monotonic() < deadline - 10.0:
            t_sub = min(deadline - 8.0, time.monotonic() + 60.0)
            seq, score = _fista_optimize(best_seq, t_sub, max_iters=500)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"    Final FISTA: score={best_score:.6f}")

    # --- Final evaluation ---
    if best_seq is not None:
        final_score = _compute_obj(best_seq)
        print(f"  Final: n={len(best_seq)}, computed_score={final_score:.6f}")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"  evaluate_sequence score: {eval_score:.6f}")
        except Exception as e:
            print(f"  evaluate_sequence error: {e}")
        return [float(x) for x in best_seq]

    # Fallback
    print("  Falling back to uniform sequence.")
    return [1.0] * 256
