def fast_convolve(a):

    """
    Compute sg[j] = sum_m w[j+m] * a[m] for j=0..n-1 using FFT cross-correlation.
    w has length 2n-1, a has length n. Returns sg of length n.
    """
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(W * np.conj(A), N)
    return corr[:n]


def optimize_sequence(a0, time_limit, t_start=5.0, t_end=300.0,
                      eta_start=0.8, eta_end=0.02,
                      momentum_beta=0.8, patience_limit=80,
                      seed=0):
    """
    Multiplicative mirror descent with momentum and FFT-accelerated subgradient.
    Input a0 is normalized to sum=1 internally.
    Returns (best_a_normalized, best_obj) where obj = 2n * max(conv).
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    mom = np.zeros(n)
    beta = momentum_beta
    t0 = time.monotonic()
    patience = 0
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        it_count += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        # Adaptive parameters with smooth scheduling
        t = t_start * (t_end / t_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress

        max_conv = conv.max()
        if max_conv < 1e-18:
            break

        # Softmax weights (gradient of log-sum-exp / t)
        scaled = (t / max_conv) * conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws

        # FFT-accelerated subgradient
        sg = compute_subgradient_fft(w, a)
        sg_centered = sg - sg.mean()

        # Momentum update
        mom = beta * mom + (1.0 - beta) * sg_centered

        # Multiplicative (mirror descent) update
        lf = -eta * mom
        lf -= lf.max()
        exp_lf = np.exp(lf)
        a_new = a * exp_lf
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a.copy()
        else:
            a_new /= sn

        # Evaluate
        conv = fast_convolve(a_new)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1

        a = a_new

        # Periodic perturbation if stuck
        if patience >= patience_limit:
            patience = 0
            noise_scale = 0.2 + 0.3 * (patience_limit / 80.0)
            noise = rng.standard_normal(n) * noise_scale
            a_pert = a * np.exp(noise)
            a_pert = np.maximum(a_pert, 0.0)
            sp = a_pert.sum()
            if sp > 1e-15:
                a_pert /= sp
                conv = fast_convolve(a_pert)
                a = a_pert
            mom *= 0.3

    return best_a, best_obj


def get_parent_sequence():
    """Try to access the parent sequence from global scope."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p
    except Exception:
        pass
    return None


def resize_sequence(a, n_new):
    """Resize sequence to length n_new using linear interpolation."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def generate_initializations(n, rng):
    """Generate diverse initializations for a sequence of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits['uniform'] = np.ones(n)

    # Gaussian narrow
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.1 * n)) ** 2)
    inits['gauss_n'] = g + 1e-12

    # Gaussian wide
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.35 * n)) ** 2)
    inits['gauss_w'] = g + 1e-12

    # Bimodal (edges)
    bi = np.zeros(n)
    w = max(1, n // 4)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bimodal'] = bi

    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5) * n

    # Hanning window
    inits['hanning'] = np.hanning(n) + 1e-12

    # Cosine bump
    cos_b = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits['cosine'] = cos_b + 1e-12

    # Two offset Gaussians
    g1 = np.exp(-0.5 * ((x - 0.3 * n) / (0.12 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.7 * n) / (0.12 * n)) ** 2)
    inits['two_g'] = g1 + g2

    # Ramp
    inits['ramp'] = np.arange(1, n + 1, dtype=np.float64)

    # Flat middle, raised ends (anti-bump)
    anti = np.ones(n)
    w2 = max(1, n // 6)
    anti[:w2] = 2.0
    anti[-w2:] = 2.0
    inits['anti_bump'] = anti

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Uses FFT-accelerated multiplicative mirror descent
    with momentum to minimize 2n*max(conv)/(sum)^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 8.0)

    best_sequence = None
    best_score = float('inf')

    parent = get_parent_sequence()

    if parent is not None:
        print(f"Parent: len={len(parent)}, sum={parent.sum():.2f}")
        ps = float(evaluate_sequence(parent.tolist()))
        print(f"Parent score: {ps:.6f}")
        if np.isfinite(ps):
            best_score = ps
            best_sequence = parent.copy()

    # Determine sequence lengths to try
    lengths = [256, 512, 1024, 2048, 4096]
    if parent is not None:
        pl = len(parent)
        # Round to nearest power of 2 for FFT efficiency, min 128
        if pl not in lengths:
            lengths.append(pl)
    lengths = sorted(set(lengths))

    # Add large length if we have parent (likely good starting point)
    if parent is not None:
        if 8192 not in lengths:
            lengths.append(8192)

    print(f"Sequence lengths to try: {lengths}")

    # ===== Phase 1: Quick Scan =====
    print("\n=== Phase 1: Quick Scan ===")

    scan_items = []
    for n in lengths:
        inits = generate_initializations(n, rng)
        if parent is not None:
            pr = resize_sequence(parent, n)
            if pr.sum() > 1e-10:
                inits['parent'] = pr
        for name, arr in inits.items():
            arr = np.maximum(arr, 0.0)
            if arr.sum() > 1e-10:
                scan_items.append((n, name, arr))

    phase1_budget = max(30.0, (deadline - time.monotonic()) * 0.25)
    time_per_scan = max(3.0, phase1_budget / max(1, len(scan_items)))
    print(f"  {len(scan_items)} configs, ~{time_per_scan:.1f}s each, budget={phase1_budget:.0f}s")

    scan_results = []
    for i, (n, name, a0) in enumerate(scan_items):
        if time.monotonic() > deadline - 180:
            break
        t_alloc = min(time_per_scan, deadline - time.monotonic() - 180)
        if t_alloc < 2:
            break

        a_opt, _ = optimize_sequence(a0, time_limit=t_alloc,
                                      t_start=5.0, t_end=150.0,
                                      eta_start=0.8, eta_end=0.05,
                                      momentum_beta=0.85, patience_limit=50,
                                      seed=seed + i)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            scan_results.append((score, n, name, a_opt))
            marker = " *" if score < best_score else ""
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
            print(f"  [{i+1:3d}/{len(scan_items)}] n={n:5d} {name:12s}: {score:.6f}{marker}")

    # ===== Phase 2: Deep Optimization =====
    print(f"\n=== Phase 2: Deep Optimization (best: {best_score:.6f}) ===")

    scan_results.sort(key=lambda x: x[0])

    # Select top configs with length diversity
    top_configs = []
    seen_keys = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen_keys and len(top_configs) < 12:
            top_configs.append((n, name, a_opt))
            seen_keys.add(key)
    # Ensure parent-derived is included
    if parent is not None and best_sequence is not None:
        if not any(n == len(best_sequence) for n, _, _ in top_configs):
            top_configs.append((len(best_sequence), 'best', best_sequence.copy()))

    phase2_budget = max(60.0, (deadline - time.monotonic()) * 0.50)
    print(f"  {len(top_configs)} configs, budget={phase2_budget:.0f}s")

    # Parameter settings for deep optimization
    deep_settings = [
        # (t_start, t_end, eta_start, eta_end, momentum_beta, patience)
        (8.0, 400.0, 0.6, 0.02, 0.85, 80),
        (4.0, 200.0, 1.0, 0.03, 0.75, 60),
        (15.0, 800.0, 0.4, 0.01, 0.90, 100),
    ]

    for i, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 60:
            break
        remaining_here = min(phase2_budget / max(1, len(top_configs) - i),
                             deadline - time.monotonic() - 60)
        if remaining_here < 5:
            break

        t_per_setting = remaining_here / len(deep_settings)

        for ds in deep_settings:
            if time.monotonic() > deadline - 60:
                break
            t_single = min(t_per_setting, deadline - time.monotonic() - 60)
            if t_single < 3:
                break

            t_s, t_e, e_s, e_e, mom_b, pat = ds
            a_opt, _ = optimize_sequence(a0, time_limit=t_single,
                                          t_start=t_s, t_end=t_e,
                                          eta_start=e_s, eta_end=e_e,
                                          momentum_beta=mom_b, patience_limit=pat,
                                          seed=seed + 1000 + i * 10)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                marker = " *" if score < best_score else ""
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                print(f"  Deep [{i+1}/{len(top_configs)}] n={n:5d} {name:12s} "
                      f"t=[{t_s:.0f},{t_e:.0f}]: {score:.6f}{marker}")

    # ===== Phase 3: Final Polish =====
    print(f"\n=== Phase 3: Final Polish (best: {best_score:.6f}) ===")

    if best_sequence is not None and time.monotonic() < deadline - 5:
        remaining = deadline - 5 - time.monotonic()
        n = len(best_sequence)
        print(f"  Polishing n={n} sequence, remaining={remaining:.0f}s")

        polish_settings = [
            (50.0, 1500.0, 0.15, 0.005, 0.95, 200),
            (100.0, 3000.0, 0.08, 0.002, 0.96, 300),
            (200.0, 6000.0, 0.04, 0.001, 0.97, 500),
        ]

        t_per = remaining / len(polish_settings)
        for ps in polish_settings:
            if remaining < 3:
                break
            t_single = min(t_per, remaining)
            t_s, t_e, e_s, e_e, mom_b, pat = ps
            a_opt, _ = optimize_sequence(best_sequence, time_limit=t_single,
                                          t_start=t_s, t_end=t_e,
                                          eta_start=e_s, eta_end=e_e,
                                          momentum_beta=mom_b, patience_limit=pat,
                                          seed=seed + 5000)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                marker = " *" if score < best_score else ""
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                print(f"  Polish t=[{t_s:.0f},{t_e:.0f}] eta=[{e_s},{e_e}]: {score:.6f}{marker}")
            remaining = deadline - 5 - time.monotonic()

    # ===== Phase 4: Cross-length improvement =====
    # Try to transfer the best solution to nearby lengths
    print(f"\n=== Phase 4: Cross-length (best: {best_score:.6f}) ===")
    if best_sequence is not None and time.monotonic() < deadline - 5:
        remaining = deadline - 5 - time.monotonic()
        n_best = len(best_sequence)
        # Try nearby powers of 2
        nearby = []
        for n_try in [n_best // 2, n_best * 2]:
            if n_best // 2 * 2 == n_try or n_best * 2 == n_try:
                if 128 <= n_try <= 16384 and n_try != n_best:
                    nearby.append(n_try)
        # Also try n_best +/- some
        for n_try in [n_best + 128, n_best - 128, n_best + 256, n_best - 256]:
            if 128 <= n_try <= 16384 and n_try != n_best and n_try not in nearby:
                nearby.append(n_try)

        t_per = remaining / max(1, len(nearby))
        for n_try in nearby:
            if remaining < 10:
                break
            t_single = min(t_per, remaining)
            a_init = resize_sequence(best_sequence, n_try)
            a_opt, _ = optimize_sequence(a_init, time_limit=t_single,
                                          t_start=20.0, t_end=500.0,
                                          eta_start=0.3, eta_end=0.01,
                                          momentum_beta=0.9, patience_limit=100,
                                          seed=seed + 9000 + n_try)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                marker = " *" if score < best_score else ""
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                print(f"  Cross n={n_try}: {score:.6f}{marker}")
            remaining = deadline - 5 - time.monotonic()

    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f}")
    print(f"Sequence length: {len(best_sequence)}")
    print(f"{'='*50}")

    return [float(x) for x in best_sequence]
