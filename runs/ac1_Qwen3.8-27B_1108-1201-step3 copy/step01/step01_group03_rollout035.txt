**Even better idea: Use the first-order optimality condition**

At the optimum, for the problem min max_k (f*f)[k] subject to sum(f)=1, f>=0:
- By KKT, there exists a dual variable λ_k >= 0 for each constraint (f*f)[k] <= M
- The stationarity condition: for all i with f[i] > 0: sum_k λ_k * 2*f[k-i] = μ (the Lagrange multiplier for the sum constraint)
- For all i with f[i] = 0: sum_k λ_k * 2*f[k-i] <= μ

So at the optimum, the "weighted average" of f (weighted by the dual multipliers over the active lags) is constant on the support of f.

This gives us an iterative algorithm:
1. Start with some f (sum=1, f>=0)
2. Compute R = convolve(f, f), M = max(R)
3. Set λ_k = 1 if R[k] = M, else 0 (or use a soft version: λ_k = softmax(R/T))
4. Compute h[i] = sum_k λ_k * 2*f[k-i] (with validity)
5. Update: f_new[i] = f[i] - η * (h[i] - mean(h on support))
6. Project to f >= 0, renormalize to sum 1
7. Repeat

This is essentially a projected subgradient method with a twist: we only subtract the mean to preserve the sum constraint.

```python
def project_to_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0 : sum(x) = target_sum} in euclidean norm."""
    n = len(v)
    u = np.sort(v)[::-1]

    rho = (np.arange(1, n+1) * u - (cssv - target_sum)) > 0

    theta = (cssv[rho_hat-1] - target_sum) / rho_hat
```

```python
def adam_update(params, grad, lr, beta1=0.9, beta2=0.999, eps=1e-8, m=None, v=None, t=1):
    if m is None:
        m = np.zeros_like(grad)
    if v is None:
        v = np.zeros_like(grad)
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad**2
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    return params - lr * m_hat / (np.sqrt(v_hat) + eps), m, v, t + 1
```

```python
    return project_simplex(f_new, 1.0)

def make_inits(n, rng):
    """Generate diverse initial vectors of length n, sum=1, non-negative."""
    inits = []
    inits.append(np.ones(n) / n)
    x = np.linspace(0, np.pi, n)
    c = np.sin(x) ** 2
    c = np.maximum(c, 1e-8)
    inits.append(c / c.sum())
    a = np.ones(n) + 0.5 * rng.standard_normal(n)
    a = np.maximum(a, 0.01)
    inits.append(a / a.sum())
    x = np.arange(n, dtype=float)
    s = max(1, n // 16)
    a = np.exp(-0.5 * ((x - n / 4.0) / s) ** 2) + np.exp(-0.5 * ((x - 3.0 * n / 4.0) / s) ** 2)
    inits.append(a / a.sum())
    centers = [n * (k + 0.5) / 4.0 for k in range(4)]
    s = max(1, n // 20)
    a = sum(np.exp(-0.5 * ((x - c) / s) ** 2) for c in centers)
    inits.append(a / a.sum())
    a = np.ones(n)
    a[1::2] = 0.3
    inits.append(a / a.sum())
    a = np.ones(n)
    a[1::3] = 0.2
    a[2::3] = 0.5
    inits.append(a / a.sum())
    a = rng.random(n)
    a = np.maximum(a, 0.01)
    inits.append(a / a.sum())
    k = max(2, n // 16)
    a = np.zeros(n)
    positions = rng.choice(n, k, replace=False)
    a[positions] = 1.0
    inits.append(a / a.sum())
    return inits

def run_lse_opt(f_init, budget_s, alpha=1.0, T_start=0.05, T_end=0.002):
    """LSE exponential-gradient optimization."""
    f = f_init.copy()
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-10:
        return f_init, fast_score(f_init)
    f = f / s
    n = len(f)
    best_f = f.copy()
    best_score = fast_score(f)
    start = time.monotonic()
    stall = 0
    step = 0
    while time.monotonic() - start < budget_s:
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(budget_s, 1e-6))
        T = T_start * (T_end / T_start) ** frac
        _, grad = compute_lse_grad(f, T)
        a_eff = alpha * (1.0 + 0.5 * frac)
        f_new = exp_weights_step(f, grad, a_eff)
        sc = fast_score(f_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = f_new.copy()
            stall = 0
        else:
            stall += 1
        f = f_new
        step += 1
        if stall > 3000:
            f = best_f * (1.0 + 0.03 * np.random.randn(n))
            f = np.maximum(f, 0)
            ss = f.sum()
            if ss > 1e-10:
                f = f / ss
            stall = 0
    return best_f, best_score

def build_conv_matrix(n, f):
    """Sparse (2n-1 x n) matrix A with A[k,j] = f[k-j] if valid."""
    i_idx = np.tile(np.arange(n), n)
    j_idx = np.repeat(np.arange(n), n)
    row_idx = j_idx + i_idx
    data = f[i_idx]
    A = sp.coo_matrix((data, (row_idx, j_idx)), shape=(2 * n - 1, n)).tocsr()
    return A

def solve_lp(f, time_limit_s=8.0):
    """LP: max sum(g) s.t. (f*g) <= M, g >= 0. Returns normalized g or None."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(conv.max())
    if M < 1e-15:
        return None
    A = build_conv_matrix(n, f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit_s, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs < 1e-10:
                return None
            return g / gs
    except Exception:
        pass
    return None

def lp_polish(f, budget_s, t_init=0.3, t_min=0.05):
    """LP-based iterative polishing."""
    f = f.copy()
    f = f / f.sum()
    best_f = f.copy()
    best_score = fast_score(f)
    n = len(f)
    t = t_init
    start = time.monotonic()
    stall = 0
    while time.monotonic() - start < budget_s:
        g = solve_lp(f, time_limit_s=5.0)
        if g is None:
            f = block_perturb(f, np.random.default_rng(int(time.monotonic() * 1e6)), 2, max(5, n // 20))
            t = t_init
            continue
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-10:
            f = block_perturb(f, np.random.default_rng(42), 2, max(5, n // 20))
            continue
        f_new = f_new / s
        sc = fast_score(f_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = f_new.copy()
            stall = 0
        else:
            stall += 1
        f = f_new
        if stall > 30:
            f = best_f.copy()
            f = block_perturb(f, np.random.default_rng(int(stall * 7)), 1, max(3, n // 30))
            stall = 0
        t = max(t_min, t * 0.997)
    return best_f, best_score

def block_perturb(f, rng, low=1, high=20):
    """Multiplicative random block perturbation."""
    n = len(f)
    cand = f.copy()
    w = int(rng.integers(low, max(low + 1, min(high, n))))
    start = int(rng.integers(0, max(1, n - w + 1)))
    stop = min(n, start + w)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.2)))
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s < 1e-10:
        return f
    return cand / s

def final_local_search(f_best, budget_s):
    """Intensive local search around the best solution."""
    f = f_best.copy()
    f = f / f.sum()
    best_f = f_best.copy()
    best_score = fast_score(f_best)
    n = len(f)
    rng = np.random.default_rng(12345)
    start = time.monotonic()
    while time.monotonic() - start < budget_s:
        w = int(rng.integers(1, max(2, n // 10)))
        cand = block_perturb(f, rng, 1, max(3, n // 25))
        sc = fast_score(cand)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_f = cand.copy()
            f = cand
        else:
            if rng.random() < 0.1:
                f = best_f * (1.0 + 0.02 * rng.standard_normal(n))
                f = np.maximum(f, 0)
                ss = f.sum()
                if ss > 1e-10:
                    f = f / ss
    return best_f, best_score

def propose_candidate(seed=42, budget_s=1800):
    """Search for the best non-negative sequence minimizing the evaluation score."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)
    print("Starting search with budget %ds", flush=True)

    # Get parent sequence if available
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except Exception:
        parent = np.ones(256, dtype=np.float64)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)

    # Normalize parent
    parent = np.maximum(parent, 0)
    ps = parent.sum()
    if ps < 1e-10:
        parent = np.ones(256)
    else:
        parent = parent / ps

    best_f = parent.copy()
    best_score = fast_score(parent)
    print("Parent score: %.6f (n=%d)" % (best_score, parent.size), flush=True)

    def update(bf, bs):
        nonlocal best_f, best_score
        if np.isfinite(bs) and bs < best_score:
            best_score = bs
            best_f = np.maximum(bf, 0).copy()
            print("  New best: %.6f (n=%d)" % (bs, best_f.size), flush=True)

    # Phase 1: LSE gradient optimization with multiple initializations
    lens = [256, 512, 1024, 2048]
    if parent.size not in lens and 50 <= parent.size <= 4096:
        lens.append(int(parent.size))

    time_per_n = max(15.0, (budget_s * 0.4) / max(1, len(lens) * 4))
    print("Phase 1: LSE optimization, ~%.0fs per (n, init) combo" % time_per_n, flush=True)

    for n in lens:
        if time.monotonic() > deadline:
            break
        inits = make_inits(n, rng)
        if n == parent.size:
            inits.append(parent.copy())
        for idx, f0 in enumerate(inits):
            if time.monotonic() > deadline:
                break
            t_budget = min(time_per_n, max(0.0, (deadline - time.monotonic())))
            if t_budget < 5.0:
                break
            fb, sb = run_lse_opt(f0, t_budget, alpha=1.0, T_start=0.05, T_end=0.001)
            if np.isfinite(sb):
                update(fb, sb)

    print("After Phase 1: best = %.6f (n=%d)" % (best_score, best_f.size), flush=True)

    # Phase 2: LP polishing on best candidates
    lp_budget = min(400.0, max(0.0, (deadline - time.monotonic()) * 0.35))
    if lp_budget > 30.0:
        print("Phase 2: LP polishing, budget %.0fs" % lp_budget, flush=True)
        # Polish the current best
        fb, sb = lp_polish(best_f, lp_budget * 0.5, t_init=0.3, t_min=0.05)
        if np.isfinite(sb):
            update(fb, sb)
        # Also try polishing a few fresh initializations at the best length
        n_best = best_f.size
        inits2 = make_inits(n_best, np.random.default_rng(seed + 99))
        remaining_lp = max(0.0, lp_budget - (time.monotonic() - (deadline - lp_budget)))
        per_init = max(5.0, remaining_lp / max(1, len(inits2)))
        for f0 in inits2[:4]:
            t_rem = max(0.0, deadline - time.monotonic())
            if t_rem < 10.0:
                break
            t_b = min(per_init, t_rem * 0.3)
            if t_b < 5.0:
                break
            # Quick LSE warm-up then LP
            fb2, _ = run_lse_opt(f0, min(t_b * 0.3, 30.0), alpha=1.0, T_start=0.05, T_end=0.01)
            fb3, sb3 = lp_polish(fb2, min(t_b * 0.7, t_b), t_init=0.25, t_min=0.05)
            if np.isfinite(sb3):
                update(fb3, sb3)

    print("After Phase 2: best = %.6f (n=%d)" % (best_score, best_f.size), flush=True)

    # Phase 3: Final local search
    ls_budget = max(0.0, (deadline - time.monotonic()) * 0.6)
    if ls_budget > 10.0:
        print("Phase 3: Local search, budget %.0fs" % ls_budget, flush=True)
        fb, sb = final_local_search(best_f, ls_budget)
        if np.isfinite(sb):
            update(fb, sb)

    # Final verification
    final_score = evaluate_sequence(best_f.tolist())
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score
    print("FINAL: score = %.6f (n=%d)" % (best_score, best_f.size), flush=True)

    # Ensure validity
    best_f = np.clip(np.asarray(best_f, dtype=np.float64), 0.0, 1000.0)
    if best_f.size == 0 or not np.isfinite(best_f).all() or best_f.sum() < 0.01:
        best_f = np.ones(256, dtype=np.float64) / 256.0

    return [float(x) for x in best_f]
```
