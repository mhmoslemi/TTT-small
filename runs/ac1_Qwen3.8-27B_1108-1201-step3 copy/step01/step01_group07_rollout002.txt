"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth optimization
(logsumexp proxy + L-BFGS on softmax parameterization) with
multi-temperature schedule and diverse initializations.
Final polish with multiplicative mirror descent.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def smooth_obj_and_grad(w, t):
    """
    Compute smooth objective and gradient w.r.t. log-parameters w.
    p = softmax(w), obj = 2n * logsumexp(t * conv(p,p)) / t
    """
    n = len(w)
    # Softmax
    w_shifted = w - w.max()
    exp_w = np.exp(w_shifted)
    p = exp_w / exp_w.sum()
    
    # Convolution
    conv = fast_convolve(p)
    
    # Smooth max via logsumexp
    t_conv = t * conv
    t_conv_max = t_conv.max()
    exp_t = np.exp(t_conv - t_conv_max)
    logsumexp_val = t_conv_max + np.log(exp_t.sum())
    
    obj = 2.0 * n * logsumexp_val / t
    
    # Weights for the smooth max
    w_conv = exp_t / exp_t.sum()
    
    # Gradient w.r.t. p:
    # d_obj/dp[j] = (4n/t) * sum_{m=0}^{n-1} w_conv[j+m] * p[m]
    # This is cross-correlation of w_conv and p at lag j
    N_fft = 1 << (3 * n).bit_length()
    W = np.fft.rfft(w_conv, N_fft)
    P = np.fft.rfft(p, N_fft)
    grad_p = np.fft.irfft(W * np.conj(P), N_fft)[:n]
    grad_p *= 4.0 * n / t
    
    # Chain rule: d_obj/dw = grad_p * p - p * (grad_p . p)
    grad_w = grad_p * p - p * np.dot(grad_p, p)
    
    return obj, grad_w

def run_lbfgs(w0, t, maxiter=500, time_limit=None, t_start=time.time()):
    """Run L-BFGS optimization with given temperature."""
    n = len(w0)
    
    def fun(w):
        obj, grad = smooth_obj_and_grad(w, t)
        if time_limit is not None and time.time() - t_start > time_limit:
            raise KeyboardInterrupt
        return obj, grad
    
    try:
        result = minimize(fun, w0, method='L-BFGS-B', 
                         options={'maxiter': maxiter, 'maxfun': 10000,
                                  'ftol': 1e-12, 'gtol': 1e-10})
        return result.x, result.fun
    except (KeyboardInterrupt, Exception) as e:
        return w0, smooth_obj_and_grad(w0, t)[0]

def softmax(w):
    w_shifted = w - w.max()
    exp_w = np.exp(w_shifted)
    return exp_w / exp_w.sum()

def true_objective_normalized(p):
    """Compute 2n * max(conv(p,p)) for p summing to 1."""
    n = len(p)
    conv = fast_convolve(p)
    return 2.0 * n * conv.max()

def multiplicative_polish(p, time_limit, t_param=40.0, eta0=0.3):
    """Polish with multiplicative mirror descent on the true (non-smooth) objective."""
    p = np.maximum(p.astype(np.float64), 0.0)
    n = len(p)
    p_sum = p.sum()
    if p_sum < 1e-12:
        p = np.ones(n) / n
    else:
        p = p / p_sum
    
    conv = fast_convolve(p)
    best_obj = 2.0 * n * conv.max()
    best_p = p.copy()
    
    eta = eta0
    patience = 0
    max_patience = 60
    current_t = t_param
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break
        
        # Soft-max weights
        scaled = current_t * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        
        # Weighted subgradient
        sg = np.zeros(n)
        thresh = w.max() * 0.003
        active_idx = np.where(w > thresh)[0]
        
        for k_int in active_idx:
            k = int(k_int)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                sg[j_start:j_end + 1] += w[k] * p[k - j_end:k - j_start + 1][::-1]
        
        # Multiplicative update
        sg_centered = sg - sg.mean()
        log_factor = -eta * sg_centered
        log_factor -= log_factor.max()
        p = p * np.exp(log_factor)
        p = np.maximum(p, 0.0)
        p_sum = p.sum()
        if p_sum < 1e-12:
            p = np.ones(n) / n
        else:
            p = p / p_sum
        
        conv = fast_convolve(p)
        current_obj = 2.0 * n * conv.max()
        
        if current_obj < best_obj - 1e-9:
            best_obj = current_obj
            best_p = p.copy()
            patience = 0
        else:
            patience += 1
        
        if patience >= max_patience:
            patience = 0
            eta *= 0.6
            if eta < 0.005:
                current_t *= 2.0
                eta = 0.3
                max_patience = 40
            else:
                current_t = np.clip(current_t * 1.2, 5.0, 200.0)
    
    return best_p, best_obj

def generate_initializations(n, rng, parent=None):
    """Generate diverse initial probability vectors of length n."""
    x = np.arange(n, dtype=np.float64)
    inits = {}
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Off-center Gaussians
    for mu_frac in [0.3, 0.35, 0.25]:
        for sigma_frac in [0.15, 0.2, 0.25, 0.3]:
            mu = mu_frac * n
            sigma = sigma_frac * n
            g = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
            g /= g.sum()
            inits[f'gauss_mu{mu_frac:.2f}_s{sigma_frac:.2f}'] = g
    
    # Centered Gaussian
    g_center = np.exp(-0.5 * ((x - n * 0.5) / (0.2 * n)) ** 2)
    g_center /= g_center.sum()
    inits['gauss_center'] = g_center
    
    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp
    
    # Inverted ramp
    inv_ramp = np.arange(n, 0, -1, dtype=np.float64)
    inv_ramp /= inv_ramp.sum()
    inits['inv_ramp'] = inv_ramp
    
    # Bimodal (edges)
    bi = np.zeros(n)
    bi[:max(1, n // 5)] = 1.0
    bi[-max(1, n // 5):] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Sine bump (off-center)
    for phase in [0.3, 0.7]:
        s = np.maximum(0, np.sin(2 * np.pi * x / n + phase * 2 * np.pi))
        if s.sum() > 1e-10:
            s /= s.sum()
            inits[f'sine_{phase:.1f}'] = s
    
    # From parent if available
    if parent is not None:
        if parent.size >= n:
            pp = parent[:n].copy()
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent'] = pp
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent_interp'] = pp
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-temperature L-BFGS
on a logsumexp-smoothed objective with softmax parameterization,
combined with multiplicative mirror descent polish.
Explores diverse initializations (off-center Gaussians, ramps, random)
across multiple sequence lengths, then refines the best candidates.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def softmax_fn(w):
    """Stable softmax."""
    w_shifted = w - w.max()
    exp_w = np.exp(w_shifted)
    s = exp_w.sum()
    if s < 1e-300:
        return np.ones(len(w)) / len(w)
    return exp_w / s


def smooth_obj_and_grad(w, t):
    """
    Compute smooth objective and gradient w.r.t. log-parameters w.
    p = softmax(w), obj = 2n * logsumexp(t * conv(p,p)) / t
    """
    n = len(w)
    p = softmax_fn(w)

    conv = fast_convolve(p)

    # Logsumexp with stability
    t_conv = t * conv
    t_conv_max = t_conv.max()
    exp_t = np.exp(t_conv - t_conv_max)
    exp_sum = exp_t.sum()
    if exp_sum < 1e-300:
        # Degenerate: just use the max
        k_max = int(np.argmax(conv))
        obj = 2.0 * n * conv[k_max]
        # Subgradient at the max
        grad_p = np.zeros(n)
        for j in range(n):
            idx = k_max - j
            if 0 <= idx < n:
                grad_p[j] = 2.0 * p[idx]
        grad_p *= 2.0 * n
        grad_w = grad_p * p - p * np.dot(grad_p, p)
        return obj, grad_w

    logsumexp_val = t_conv_max + np.log(exp_sum)
    obj = 2.0 * n * logsumexp_val / t

    # Weights for smooth max
    w_conv = exp_t / exp_sum

    # Gradient w.r.t. p:
    # d_obj/dp[j] = (4n/t) * sum_{m=0}^{n-1} w_conv[j+m] * p[m]
    N_fft = 1 << (3 * n).bit_length()
    W_fft = np.fft.rfft(w_conv, N_fft)
    P_fft = np.fft.rfft(p, N_fft)
    grad_p = np.fft.irfft(W_fft * np.conj(P_fft), N_fft)[:n]
    grad_p *= 4.0 * n / t

    # Chain rule through softmax
    grad_w = grad_p * p - p * np.dot(grad_p, p)

    return obj, grad_w


def run_lbfgs(w0, t, maxiter=500, time_limit_s=30.0):
    """Run L-BFGS on the smoothed objective. Returns (w_opt, obj_opt)."""
    t_start = time.monotonic()
    timeout_flag = [False]

    def fun(w):
        if time.monotonic() - t_start > time_limit_s:
            timeout_flag[0] = True
            raise ValueError("timeout")
        return smooth_obj_and_grad(w, t)

    try:
        result = minimize(fun, w0, method='L-BFGS-B',
                          options={'maxiter': maxiter, 'maxfun': 5000,
                                   'ftol': 1e-14, 'gtol': 1e-10,
                                   'maxls': 50})
        w_opt = result.x
        obj_opt, _ = smooth_obj_and_grad(w_opt, t)
    except Exception:
        w_opt = w0
        obj_opt, _ = smooth_obj_and_grad(w0, t)

    return w_opt, obj_opt


def true_obj_normalized(p):
    """Compute 2n * max(conv(p,p)) for p summing to 1."""
    n = len(p)
    conv = fast_convolve(p)
    return 2.0 * n * conv.max()


def multiplicative_polish(p_init, time_limit, t_param=30.0, eta0=0.3):
    """Polish with multiplicative mirror descent on the true non-smooth objective."""
    p = np.maximum(p_init.astype(np.float64), 0.0)
    n = len(p)
    p_sum = p.sum()
    if p_sum < 1e-12:
        p = np.ones(n) / n
    else:
        p = p / p_sum

    conv = fast_convolve(p)
    best_obj = 2.0 * n * conv.max()
    best_p = p.copy()

    eta = eta0
    patience = 0
    max_patience = 60
    current_t = t_param
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break

        scaled = current_t * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

        sg = np.zeros(n)
        thresh = w.max() * 0.002
        active_idx = np.where(w > thresh)[0]

        for k_int in active_idx:
            k = int(k_int)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                sg[j_start:j_end + 1] += w[k] * p[k - j_end:k - j_start + 1][::-1]

        sg_centered = sg - sg.mean()
        log_factor = -eta * sg_centered
        log_factor -= log_factor.max()
        p = p * np.exp(log_factor)
        p = np.maximum(p, 0.0)
        p_sum = p.sum()
        if p_sum < 1e-12:
            p = np.ones(n) / n
        else:
            p = p / p_sum

        conv = fast_convolve(p)
        current_obj = 2.0 * n * conv.max()

        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_p = p.copy()
            patience = 0
        else:
            patience += 1

        if patience >= max_patience:
            patience = 0
            eta *= 0.65
            if eta < 0.004:
                current_t *= 2.5
                eta = 0.25
                max_patience = 35
            else:
                current_t = np.clip(current_t * 1.15, 5.0, 300.0)

    return best_p, best_obj


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial probability vectors of length n."""
    x = np.arange(n, dtype=np.float64)
    inits = {}

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Off-center Gaussians (key insight: asymmetry helps)
    for mu_frac in [0.25, 0.3, 0.35, 0.2]:
        for sigma_frac in [0.15, 0.2, 0.25, 0.3, 0.35]:
            mu = mu_frac * n
            sigma = sigma_frac * n
            g = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
            g /= g.sum()
            inits['g_{:.2f}_{:.2f}'.format(mu_frac, sigma_frac)] = g

    # Centered Gaussian (for comparison)
    g_c = np.exp(-0.5 * ((x - n * 0.5) / (0.22 * n)) ** 2)
    g_c /= g_c.sum()
    inits['gauss_center'] = g_c

    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp

    # Inverted ramp
    inv_ramp = np.arange(n, 0, -1, dtype=np.float64).astype(np.float64)
    inv_ramp /= inv_ramp.sum()
    inits['inv_ramp'] = inv_ramp

    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(2, n // 6)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # U-shape (high at edges, low in middle)
    u_shape = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    u_shape += 0.1
    u_shape /= u_shape.sum()
    inits['u_shape'] = u_shape

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['random2'] = rng.dirichlet(np.ones(n) * 2.0)

    # From parent if available
    if parent is not None:
        if parent.size >= n:
            pp = parent[:n].copy()
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent'] = pp
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent_interp'] = pp

    return inits
