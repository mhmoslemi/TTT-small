"""
Gradient-based optimization with softmax-smoothed convolution peak,
combined with LP polishing, to minimize 2*n*max(conv(f,f))/(sum(f))^2.

Key insight: The full linear convolution of a non-symmetric sequence with
itself is NOT symmetric, so the peak can be shifted away from center and
reduced below the naive Cauchy-Schwarz bound. The gradient of the smoothed
max-convolution is a weighted correlation of the sequence with itself,
computable in O(n log n) via FFT.

Method:
1. Softmax-smoothed gradient descent with adaptive learning rate
2. Multiple initializations (parent, exponential, chirp, random) at various lengths
3. LP-based polishing for final refinement
4. Local search with random perturbations
"""
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time


def fast_score(f):
    """Compute score = 2*n*max(conv(f,f))/sum(f)^2."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    n = len(f)
    conv = np.convolve(f, f)
    return float(2.0 * n * conv.max() / (s * s))


def compute_soft_gradient(f, T):
    """
    Compute the gradient of M_smooth = (1/T)*log(sum(exp(T*conv(f,f))))
    w.r.t. f. Uses the identity:
      dM/df[i] = 2 * sum_m w[i+m] * f[m]
    where w are softmax weights over the convolution values.
    Computed via convolution: g[i] = conv(w, reverse(f))[n-1+i]
    """
    n = len(f)
    conv = np.convolve(f, f)  # length 2n-1

    # Softmax weights with numerical stability
    log_w = T * conv
    log_w -= log_w.max()
    w = np.exp(log_w)
    w_sum = w.sum()
    w = w / w_sum

    # Gradient: g[i] = sum_m w[i+m] * f[m] = conv(w, reverse(f))[n-1+i]
    h = f[::-1]
    conv_wh = np.convolve(w, h)  # length (2n-1) + n - 1 = 3n-2
    g = conv_wh[n - 1:2 * n - 1]  # length n

    return 2.0 * g, conv, w


def compute_exact_subgradient(f, conv, n_peaks=5):
    """
    Compute exact subgradient using top-n peak locations.
    Returns gradient vector and the max value.
    """
    n = len(f)
    # Find top n_peaks lags
    top_indices = np.argsort(conv)[-n_peaks:][::-1]
    max_val = conv[top_indices].max()

    # Gradient: sum over peak lags k of 2*f[k-i] (where valid)
    grad = np.zeros(n)
    for k in top_indices:
        # d(conv[k])/df[i] = 2*f[k-i] if 0 <= k-i < n
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        if low <= high:
            indices = np.arange(low, high + 1)
            f_indices = k - indices
            grad[indices] += 2.0 * f[f_indices]
    return grad, max_val


def gradient_descent(f0, budget_s, T_init=5.0, T_final=50.0,
                     lr_init=0.005, lr_min=1e-6, lr_max=0.05,
                     n_peaks_mode='softmax', print_every=200):
    """
    Main gradient descent loop with adaptive learning rate and temperature.
    """
    n = len(f0)
    f = np.array(f0, dtype=np.float64)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-10:
        return f, np.inf
    f = f / s

    best_f = f.copy()
    best_sc = fast_score(f)
    lr = lr_init
    stall_count = 0
    it = 0
    t_start = time.monotonic()

    while time.monotonic() - t_start < budget_s:
        it += 1
        # Update temperature (log-linear schedule)
        progress = min(1.0, it / max(1, int(budget_s * 20)))
        T = T_init * (T_final / T_init) ** progress

        # Compute gradient
        if n_peaks_mode == 'softmax':
            grad, conv, _ = compute_soft_gradient(f, T)
        else:
            conv = np.convolve(f, f)
            grad, _ = compute_exact_subgradient(f, conv)

        # Project onto sum=0 subspace (maintain normalization)
        grad = grad - grad.mean()

        # Backtracking line search
        improved = False
        for trial in range(10):
            f_try = f - lr * grad
            f_try = np.maximum(f_try, 0.0)
            s_try = f_try.sum()
            if s_try < 1e-10:
                lr *= 0.5
                continue
            f_try = f_try / s_try
            sc_try = fast_score(f_try)
            if sc_try < best_sc - 1e-10 or sc_try < fast_score(f) - 1e-10:
                f = f_try
                improved = True
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_f = f.copy()
                    stall_count = 0
                break
            else:
                lr = max(lr_min, lr * 0.5)
                f_try = None

        if not improved:
            # Try perturbation to escape
            if stall_count > 0:
                rnd = np.random.default_rng(hash((n, it, stall_count)) & 0xFFFFFFFF)
                block_len = max(1, n // 20)
                start = int(rnd.integers(0, n - block_len))
                f[start:start + block_len] *= np.exp(rnd.normal(0, 0.3, block_len))
                f = np.maximum(f, 0.0)
                f = f / f.sum()
                sc = fast_score(f)
                if sc < best_sc:
                    best_sc = sc
                    best_f = f.copy()
                    stall_count = 0
                else:
                    f = best_f.copy()
            lr = max(lr_min, lr * 0.7)
            stall_count += 1

            # If severely stuck, restart from best with perturbation
            if stall_count > 50:
                rnd = np.random.default_rng(it)
                f = best_f.copy()
                mask = rnd.random(n) < 0.3
                f[mask] *= np.exp(rnd.normal(0, 0.5, mask.sum().astype(int) if mask.any() else 0))
                f = np.maximum(f, 0.0)
                s = f.sum()
                if s > 1e-10:
                    f = f / s
                lr = lr_init * 0.5
                stall_count = 0
        else:
            lr = min(lr_max, lr * 1.05)

        if it % print_every == 0:
            print("  GD it=%d T=%.1f lr=%.1e score=%.6f best=%.6f" %
                  (it, T, lr, fast_score(f), best_sc), flush=True)

    return best_f, best_sc


def build_conv_matrix(n, f):
    """Sparse A (2n-1 x n) with A[k,j] = f[k-j] if 0<=k-j<n."""
    i_idx = []
    j_idx = []
    data = []
    for j in range(n):
        row_start = j
        row_end = min(2 * n - 2, j + n - 1)
        for k in range(row_start, row_end + 1):
            diff = k - j
            if 0 <= diff < n:
                i_idx.append(k)
                j_idx.append(j)
                data.append(f[diff])
    A = sp.coo_matrix((data, (i_idx, j_idx)), shape=(2 * n - 1, n)).tocsr()
    return A


def lp_one_step(f, t=0.3, time_limit_s=5.0):
    """One LP-driven update step."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(conv.max())
    if M < 1e-15:
        return f

    A = build_conv_matrix(n, f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit_s, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs < 1e-10:
                return f
            g = g / gs
            if np.linalg.norm(g - f) < 1e-10:
                return f
            f_new = (1.0 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-10:
                return f
            return f_new / s
    except Exception:
        pass
    return f


def lp_polish(f0, budget_s, t_init=0.3, t_decay=0.99, lp_time=5.0):
    """Iterative LP polishing."""
    f = np.array(f0, dtype=np.float64)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-10:
        return f, np.inf
    f = f / s

    best_f = f.copy()
    best_sc = fast_score(f)
    t = t_init
    t_start = time.monotonic()
    it = 0

    while time.monotonic() - t_start < budget_s:
        it += 1
        f_new = lp_one_step(f, t, lp_time)
        if np.array_equal(f_new, f):
            # Perturb to escape
            rnd = np.random.default_rng(it + 1000)
            block = max(1, len(f) // 30)
            start = int(rnd.integers(0, len(f) - block))
            f[start:start + block] *= np.exp(rnd.normal(0, 0.2, block))
            f = np.maximum(f, 0.0)
            s = f.sum()
            if s > 1e-10:
                f = f / s
            continue
        f = f_new
        t = max(0.05, t * t_decay)
        sc = fast_score(f)
        if sc < best_sc:
            best_sc = sc
            best_f = f.copy()
            print("  LP it=%d score=%.6f" % (it, best_sc), flush=True)

    return best_f, best_sc


def make_exponential(n, alpha, reverse=False, rng=None):
    """Exponentially decaying initialization."""
    x = np.arange(n, dtype=np.float64)
    if reverse:
        x = x[::-1]
    f = np.exp(-alpha * x / max(1, n - 1))
    f = np.maximum(f, 1e-6)
    f = f / f.sum()
    return f


def make_chirp(n, beta, rng=None):
    """Quadratic phase (chirp) initialization."""
    x = np.arange(n, dtype=np.float64)
    f = 0.5 + 0.4 * np.abs(np.sin(np.pi * beta * x * x / (n * n)))
    f = np.maximum(f, 0.01)
    f = f / f.sum()
    return f


def make_sine_perturbed(n, k, amplitude, rng):
    """Uniform + sine perturbation."""
    f = np.ones(n) + amplitude * np.sin(2 * np.pi * k * np.arange(n) / n)
    f = np.maximum(f, 0.01)
    f = f / f.sum()
    return f


def make_parent_scaled(parent, n, rng):
    """Scale/crop/pad parent to length n."""
    parent = np.asarray(parent, dtype=np.float64)
    pn = len(parent)
    if pn == n:
        f = parent.copy()
    elif pn > n:
        f = parent[:n].copy()
    else:
        # Pad with edge values
        f = np.zeros(n)
        f[:pn] = parent
        f[pn:] = parent[-1]
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / s
    return f


def local_search(f0, budget_s, rng_seed=42):

    n = len(f0)
    f = np.array(f0, dtype=np.float64)
    f = np.maximum(f, 0.0)

    f = f / s

    rng = np.random.default_rng(rng_seed)
    t_start = time.monotonic()
    it = 0

    while time.monotonic() - t_start < budget_s:
        it += 1
        # Various perturbation strategies
        strategy = it % 4
        cand = f.copy()

        if strategy == 0:
            # Block scaling
            blen = max(1, int(rng.integers(1, max(2, n // 10))))
            start = int(rng.integers(0, n - blen))
            cand[start:start + blen] *= np.exp(rng.normal(0, 0.15, blen))
        elif strategy == 1:
            # Single element perturbation
            idx = int(rng.integers(0, n))
            cand[idx] *= np.exp(rng.normal(0, 0.3))
        elif strategy == 2:
            # Pair swap-like
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = rng.normal(0, 0.05 * f.mean())
                cand[i] += delta
                cand[j] -= delta
        else:
            # Smooth noise
            noise = rng.normal(0, 0.02, n)
            # Smooth it
            if n > 4:
                kernel = np.ones(4) / 4
                noise = np.convolve(noise, kernel, mode='same')
            cand = f + noise

        cand = np.maximum(cand, 0.0)
        s_c = cand.sum()
        if s_c < 1e-10:
            continue
        cand = cand / s_c
        sc = fast_score(cand)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_f = cand.copy()
            f = cand.copy()
            if it % 50 == 0:
                print("  LS it=%d score=%.6f" % (it, best_sc), flush=True)

    return best_f, best_sc

        # Allocate more time for better initial scores
        quality = max(0.1, min(3.0, best_score / max(sc0, 1e-10)))
        init_budget = min(time_per_init * quality, 120.0)
        init_budget = max(3.0, init_budget)

        remaining = deadline - reserve_lp - reserve_ls - time.monotonic()
        if remaining < 5:
            break
        init_budget = min(init_budget, remaining)

        print("[%d/%d] GD n=%d %s init_score=%.6f budget=%.0fs" %
              (idx + 1, n_inits, n, label, sc0, init_budget), flush=True)

        f_best, sc_best = gradient_descent(f0, init_budget,
                                           T_init=5.0, T_final=80.0,
                                           lr_init=0.003, lr_min=1e-7, lr_max=0.03)
        try_update(f_best, "GD_%s" % label)

    # Phase 2: LP polish on the best found
    print("\n=== Phase 2: LP Polish ===", flush=True)
    remaining = deadline - reserve_ls - time.monotonic()
    if remaining > 10:
        print("  Starting LP polish with budget=%.0fs, current best=%.6f (n=%d)" %
              (remaining, best_score, len(best_f)), flush=True)
        f_lp, sc_lp = lp_polish(best_f, remaining, t_init=0.3, t_decay=0.98, lp_time=5.0)
        try_update(f_lp, "LP_polish")

        # Also try LP from a few other good candidates
        if time.monotonic() < deadline - reserve_ls - 30:
            # Try LP from parent at its native length
            remaining2 = deadline - reserve_ls - time.monotonic()
            if remaining2 > 20:
                f_lp2, sc_lp2 = lp_polish(parent, min(remaining2, 60), t_init=0.2, lp_time=5.0)
                try_update(f_lp2, "LP_parent")

    # Phase 3: Local search
    print("\n=== Phase 3: Local Search ===", flush=True)
    remaining = deadline - time.monotonic()
    if remaining > 5:
        print("  Starting local search with budget=%.0fs" % remaining, flush=True)
        f_ls, sc_ls = local_search(best_f, remaining, rng_seed=seed)
        try_update(f_ls, "local_search")

    # Final verification
    best_f = np.clip(np.asarray(best_f, dtype=np.float64), 0.0, 1000.0)
    if best_f.size == 0 or not np.isfinite(best_f).all() or best_f.sum() < 0.01:
        best_f = np.ones(256, dtype=np.float64)

    # Confirm with authoritative evaluator
    final_sc = float(evaluate_sequence(best_f.tolist()))
    if not np.isfinite(final_sc):
        # Fallback
        best_f = np.ones(256, dtype=np.float64)
        final_sc = float(evaluate_sequence(best_f.tolist()))

    print("\nFINAL: score=%.6f, n=%d" % (final_sc, len(best_f)), flush=True)
    return [float(x) for x in best_f]
