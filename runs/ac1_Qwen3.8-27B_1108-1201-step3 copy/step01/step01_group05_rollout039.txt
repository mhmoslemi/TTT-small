  This is not smooth at the boundary of the simplex.
- Parameterize as p = softmax(θ), which automatically satisfies p > 0 and sum(p) = 1. But then the gradient through the softmax might be problematic.
- Work on the simplex directly with projected gradients (as the previous code does).

Actually, let me try a different approach: use the parametrization p[i] = exp(θ[i]) / sum(exp(θ[j])) (i.e., softmax), and minimize the smoothed max of conv(softmax(θ), softmax(θ)) with respect to θ using L-BFGS-B.

This avoids the projection step and gives a smooth optimization problem!

The gradient of the smoothed max with respect to θ can be computed via the chain rule:
dL/dθ[i] = sum_j (dL/dp[j]) * (dp[j]/dθ[i])
where dp[j]/dθ[i] = p[j] * (δ[i,j] - p[i])

So: dL/dθ[i] = p[i] * (dL/dp[i] - sum_j dL/dp[j] * p[j])
= p[i] * (G[i] - <G, p>)

where G = dL/dp is the gradient of the smoothed max with respect to p.

This is clean and avoids projections! Let me implement this.

"""
Step-function search for min max-autocorrelation ratio.
Uses softmax parametrization + L-BFGS-B on smoothed max surrogate,
with diverse initializations and adaptive temperature.
"""

import time
import numpy as np
from scipy.fft import fft, ifft, next_fast_len
from scipy.optimize import minimize

def _fft_len(n):
    return next_fast_len(max(1, 2 * n - 1))

def _grad_fft_len(n):
    return next_fast_len(max(1, 3 * n - 2))

def _conv_full(p):
    n = len(p)
    L = _fft_len(n)
    F = fft(p, L)
    return np.real(ifft(F * F))[:2*n-1]

def _score(p):
    n = len(p)
    s = float(p.sum())
    if s < 1e-12:
        return np.inf
    c = _conv_full(p)
    return 2.0 * n * float(c.max()) / (s * s)

def _smoothed_obj_and_grad(p, T):
    """
    Compute smoothed max of conv(p,p) and its gradient w.r.t. p.
    Returns (value, gradient).
    """
    n = len(p)
    c = _conv_full(p)
    cmax = float(c.max())
    if cmax <= 0:
        return 0.0, np.zeros(n)
    
    T = max(T, 1e-30)
    diff = (c - cmax) / T
    max_diff = diff.max()  # should be 0
    e = np.exp(diff - max_diff)
    raw = float(e.sum())
    if raw < 1e-30:
        # Degenerate: use exact max
        m_star = int(np.argmax(c))
        k = m_star - np.arange(n)
        valid = (k >= 0) & (k < n)
        grad = np.zeros(n)
        grad[valid] = 2.0 * p[k[valid]]
        return cmax, grad
    
    w = e / raw
    surr = cmax + T * np.log(raw)
    
    # Gradient: grad[j] = 2 * sum_m w[m] * p[m-j] (for valid m-j)
    # = 2 * conv(w[::-1], p)[j + 2n - 2]
    L2 = _grad_fft_len(n)
    Fw = fft(w[::-1], L2)
    Fp = fft(p, L2)
    h = np.real(ifft(Fw * Fp))
    grad = 2.0 * h[2*n-2 : 3*n-2]
    
    if len(grad) != n:
        tmp = np.zeros(n)
        m = min(n, len(grad))
        tmp[:m] = grad[:m]
        grad = tmp
    
    return surr, grad

def _softmax_obj_and_grad(theta, T):
    """
    Objective and gradient in log-space (softmax parametrization).
    p = softmax(theta)
    """
    # Stable softmax
    t = theta - theta.max()
    e = np.exp(t)
    p = e / e.sum()
    
    val, grad_p = _smoothed_obj_and_grad(p, T)
    
    # Chain rule: d(val)/d(theta[i]) = p[i] * (grad_p[i] - dot(grad_p, p))
    dp = np.dot(grad_p, p)
    grad_theta = p * (grad_p - dp)
    
    return val, grad_theta

def _make_notch(n, center, width, depth, rng):
    """Create a sequence with a Gaussian notch."""
    x = np.linspace(0, 1, n)
    p = np.ones(n)
    p -= depth * np.exp(-0.5 * ((x - center) / width) ** 2)
    return _normalize(np.maximum(p, 0.01))

def _make_two_peak(n, mu1, sig1, w1, mu2, sig2, w2):
    x = np.linspace(0, 1, n)
    p = w1 * np.exp(-0.5 * ((x - mu1) / sig1) ** 2) + w2 * np.exp(-0.5 * ((x - mu2) / sig2) ** 2)
    return _normalize(p)

def _make_taper(n, alpha):
    """Endpoint taper: p[i] ~ i^alpha * (n-1-i)^alpha (or similar)."""
    i = np.arange(n, dtype=float)
    x = i / max(1, n - 1)
    p = np.power(np.maximum(x, 1e-6), alpha) * np.power(np.maximum(1 - x, 1e-6), alpha)
    return _normalize(p)

def _make_random_smooth(n, rng):
    x = np.linspace(0, 1, n)
    freqs = rng.integers(1, max(2, n // 50), size=5)
    amps = rng.uniform(0.1, 0.8, size=5)
    phases = rng.uniform(0, 2*np.pi, size=5)
    p = np.ones(n)
    for f, a, ph in zip(freqs, amps, phases):
        p += a * np.sin(2*np.pi*f*x + ph)
    return _normalize(np.maximum(p, 0.05))

def _normalize(p):
    p = np.maximum(np.asarray(p, dtype=float), 0.0)
    s = p.sum()
    if s < 1e-12:
        return np.full(len(p), 1.0/len(p))
    return p / s
