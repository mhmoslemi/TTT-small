```python
"""
LP-based optimization with gradient polish for minimizing peak autocorrelation
of non-negative step functions. Uses scipy linprog for exact LP subproblems,
FFT-based convolution, and multi-start over sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def score_fast(a):
    """Compute the score without evaluate_sequence overhead."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if sum_a := np.sum(a) < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def build_conv_matrix(f, n):
    """Build the matrix A such that A @ b = conv(f, b)."""
    # conv(f, b)[k] = sum_j f[k-j] * b[j] (with valid indices)
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[idx]
    return A

def lp_direction(f, M):
    """Solve LP: maximize sum(b) s.t. conv(f,b)[k] <= M, b >= 0."""
    n = len(f)
    A = build_conv_matrix(f, n)
    c_obj = -np.ones(n)  # minimize -sum(b)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M), bounds=(0, None), method='highs')
    if res.success:
        return res.x
    return None

def optimize_lp_based(f_init, n_iters=200, t_schedule=None, verbose=False):
    """Iterate LP-based direction updates."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    
    for it in range(n_iters):
        c = fft_convolve(f, f)
        M = np.max(c)
        
        b = lp_direction(f, M)
        if b is None:
            break
        
        # Rescale b to have same sum as f
        S = np.sum(f)
        if np.sum(b) > 0:
            b = b * (S / np.sum(b))
        
        # Update
        t = t_schedule(it) if t_schedule else 0.05
        f_new = (1 - t) * f + t * b
        f_new = np.clip(f_new, 0.0, 1000.0)
        
        new_score = score_fast(f_new)
        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()
        
        f = f_new
    
    return best_f, best_score
```

"""
LP-based optimization for minimizing peak autocorrelation of non-negative
step functions. Primary method: iterative LP direction finding with line search.
Secondary: gradient-based polish with analytic subgradients.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n+m-1]

def score_fast(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))

def build_A(f):
    n = len(f)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def lp_step(f, M):
    n = len(f)
    A = build_A(f)
    res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M),
                  bounds=[(0, None)]*n, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if res.success:
        return res.x
    return None

def make_init_windowed_random(n, rng):
    """Random sequence with triangular (Bartlett) window."""
    base = rng.standard_normal(n) + 1.0
    w = np.linspace(0, 1, n//2)
    window = np.concatenate([w, w[::-1]]) if n % 2 == 0 else np.concatenate([w, [w[-1]], w[::-1][1:]])
    # Actually, let's use a proper Bartlett window
    window = 1.0 - np.abs(np.linspace(-1, 1, n))
    a = base * (0.5 + 0.5 * window)
    return np.clip(a, 0.01, None)

def make_init_hann(n, rng):
    """Hann window with random perturbation."""
    t = np.arange(n)
    a = 0.5 * (1 + np.cos(2*np.pi*t/(n-1)))
    a = a + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_init_chirp(n, rng):
    """Chirp-modulated window."""
    t = np.arange(n)
    window = 0.5 + 0.5 * np.cos(np.pi * t / (n-1))
    a = 1.0 + 0.3 * np.cos(2*np.pi * t*t/(2*n)) * window
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def optimize_lp(f_init, time_limit, t_base=0.05, verbose=False):
    """Iterate LP-based updates until time limit."""
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    it = 0
    stall_count = 0
    stall_best = best_score
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
            
        c = fft_convolve(f, f)
        M = float(np.max(c))
        
        b = lp_step(f, M)
        if b is None:
            break
        
        S = float(np.sum(f))
        Sb = float(np.sum(b))
        if Sb < 1e-10:
            break
        b = b * (S / Sb)
        
        # Adaptive step size
        t = t_base
        # Line search: try to find largest t that doesn't increase score
        for t_try in [t, t*0.5, t*0.25, t*0.1]:
            f_new = (1 - t_try) * f + t_try * b
            f_new = np.clip(f_new, 0.0, None)
            s_new = score_fast(f_new)
            if s_new <= best_score + 1e-6:
                t = t_try
                break
        
        f = (1 - t) * f + t * b
        f = np.clip(f, 0.0, None)
        s = score_fast(f)
        
        it += 1
        if s < best_score:
            best_score = s
            best_f = f.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count % 50 == 0:
                if verbose:
                    print(f"    it={it}, score={s:.6f}, best={best_score:.6f}, t={time.monotonic()-t0:.1f}s")
            if stall_count > 200:
                # Add noise to escape
                f += 0.01 * np.random.default_rng(it).standard_normal(len(f))
                f = np.clip(f, 0.01, None)
                stall_count = 0
                stall_best = best_score
        
        if verbose and it % 100 == 0:
            print(f"    it={it}, score={s:.6f}, best={best_score:.6f}, t={time.monotonic()-t0:.1f}s")
    
    return best_f, best_score

def gradient_polish(f_init, time_limit, lr=0.01, verbose=False):
    """Subgradient-based polish using analytic gradient of max convolution value."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    S = float(np.sum(f))
    # Normalize to sum = n (average = 1)
    f = f * (n / S)
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    it = 0
    momentum = np.zeros(n)
    beta = 0.9
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(f, f)
        # Find top-k maximizing lags for better gradient
        k_max = int(np.argmax(c))
        
        # Subgradient: g[j] = 2*f[k_max - j] if 0 <= k_max - j < n
        g = np.zeros(n)
        for j in range(n):
            idx = k_max - j
            if 0 <= idx < n:
                g[j] = 2.0 * f[idx]
        
        # Center the gradient to preserve sum
        g -= np.mean(g)
        
        # Momentum update
        momentum = beta * momentum + (1 - beta) * g
        f = f - lr * momentum
        
        # Project to non-negative and maintain sum
        f = np.clip(f, 0.01, None)
        f = f * (n / float(np.sum(f)))
        
        it += 1
        if it % 100 == 0:
            s = score_fast(f)
            if s < best_score:
                best_score = s
                best_f = f.copy()
            if verbose:
                print(f"    grad it={it}, score={s:.6f}, best={best_score:.6f}")
    
    s = score_fast(f)
    if s < best_score:
        best_score = s
        best_f = f.copy()
    
    return best_f, best_score

Actually, let me reconsider the whole approach. The key issue is that the LP gives a direction, but the update `(1-t)*f + t*b` might not decrease the score if `t` is too large or if `b` is very different from `f`.

A potentially better approach: **iterative proportional fitting (IPF) / Sinkhorn-like method**.

The idea: we want `conv(f, f)[k] ≈ P` for all k (flat autocorrelation). This is like finding f such that the "product" of f with itself (under convolution) is constant.

One iteration:
1. Compute `c = conv(f, f)`.
2. Compute the "correction": `r[k] = P / c[k]` (where P is the target, e.g., the mean of c).
3. Update: `f_new = f * something_involving_r`.

But this doesn't directly work because the convolution is bilinear, not linear.

Alternatively, think of it as: we want `f * g = P` (in the convolution sense), where `g` is another copy of `f`. If we alternate:
1. Fix `g = f`, find `f_new` such that `f_new * g ≈ P` (this is a deconvolution problem).
2. Fix `f = f_new`, find `g_new` such that `f_new * g_new ≈ P`.
3. Set `f = (f_new + g_new) / 2`.

But deconvolution in the presence of non-negativity constraints is tricky.

Let me stick with the LP approach but make it more robust.

Actually, here's another idea: **use the LP as a cutting plane**. Instead of doing the full LP at each iteration, we can:
1. Start with a small set of active constraints (the top few lags).
2. Solve the LP with only those constraints.
3. Check which new constraints are violated.
4. Add them to the active set.
5. Repeat.

This is the "revised simplex" or "column generation" approach. It would be faster when only a few constraints are active.

But for simplicity, let me just use the full LP. HiGHS is fast enough.

LP-based iterative optimization for minimizing peak aperiodic autocorrelation
of non-negative sequences. Uses scipy HiGHS for LP subproblems, FFT convolution,
analytic subgradient polish, and multi-start across sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def build_A(f):
    """Build matrix A where A@b = conv(f,b) (full)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp(f, M, timeout=5):
    """Solve: max sum(b) s.t. A@b <= M*1, b >= 0."""
    n = len(f)
    A = build_A(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

        f = f - lr * momentum
        f = np.clip(f, 0.01, None)
        f = f * (float(n) / float(np.sum(f)))

        it += 1
        if it % 300 == 0:
            s = score_fast(f)
            if s < best_score:
                best_score = s
                best_f = f.copy()
            if verbose:
                print(f"    [grad {time.monotonic()-t0:6.1f}s] it={it} score={s:.6f} best={best_score:.6f}")

    s = score_fast(f)
    if s < best_score:
        best_score = s
        best_f = f.copy()
    return best_f, best_score


def local_search(f_init, time_limit, verbose=True):
    """Perturbation-based local search for fine-tuning."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    temp = 0.002
    cooling = 0.9998

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break

        candidate = f.copy()
        r = rng.random()

        if r < 0.4:
            peak_idx = int(np.argmax(fft_convolve(f, f)))
            contribs = np.zeros(n)
            for j in range(n):
                idx = peak_idx - j
                if 0 <= idx < n:
                    contribs[j] = f[j] * f[idx]
            top_k = max(3, n // 12)
            top_mask = contribs >= np.sort(contribs)[-top_k]
            candidate[top_mask] *= np.exp(rng.normal(-0.03, 0.02, size=top_mask.sum()))
            other = ~top_mask
            if other.sum() > 0:
                inc_mask = np.zeros(n, dtype=bool)
                idxs = np.where(other)[0]
                chosen = rng.choice(idxs, size=min(6, len(idxs)), replace=False)
                inc_mask[chosen] = True
                candidate[inc_mask] *= np.exp(rng.normal(0.03, 0.02, size=inc_mask.sum()))
        elif r < 0.7:
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, max(1, n - width)))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.04, size=width))
        else:
            candidate *= np.exp(rng.normal(0.0, 0.015, size=n))

        candidate = np.clip(candidate, 0.01, 1000.0)
        new_score = score_fast(candidate)

        if new_score < best_score:
            best_score = new_score
            best_f = candidate.copy()
            f = candidate
        elif rng.random() < np.exp(-(new_score - score_fast(f)) / max(temp, 1e-15)):
            f = candidate

        temp *= cooling

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase: LP optimization + gradient polish + local search."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Evaluate provided sequence
    provided = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = score_fast(provided)
        print(f"Provided: n={len(provided)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = provided.copy()

    print("\n=== Phase 1: LP-based optimization ===")

    # Define tasks: (n, init_name)
    tasks = []
    if provided is not None:
        tasks.append((len(provided), 'provided'))

    for n in [256, 320, 384, 512]:
        tasks.append((n, 'hann'))
        tasks.append((n, 'bartlett'))
        tasks.append((n, 'random'))
    for n in [768, 1024]:
        tasks.append((n, 'hann'))
        tasks.append((n, 'chirp'))

    # Time allocation: smaller n gets more time (faster LP)
    time_weights = []
    for n, _ in tasks:
        time_weights.append(1.0 / (n / 128.0))
    total_w = sum(time_weights)

    for idx, (n, init_name) in enumerate(tasks):
        if time.monotonic() > deadline - 80:
            break

        # Generate init
        if init_name == 'provided' and provided is not None:
            if len(provided) == n:
                f0 = provided.copy()
            else:
                x_old = np.linspace(0, 1, len(provided))
                x_new = np.linspace(0, 1, n)
                f0 = np.interp(x_new, x_old, provided)
        elif init_name == 'hann':
            f0 = make_hann_init(n, rng)
        elif init_name == 'bartlett':
            f0 = make_bartlett_init(n, rng)
        elif init_name == 'chirp':
            f0 = make_chirp_init(n, rng)
        else:
            f0 = make_random_windowed(n, rng)

        t_alloc = max(10.0, (deadline - time.monotonic() - 80.0) * time_weights[idx] / total_w)
        t_alloc = min(t_alloc, 200.0)

        print(f"\n  n={n:4d} init={init_name:10s} alloc={t_alloc:.0f}s")
        f_best, s_best = optimize_lp_steps(f0, t_alloc, t_step=0.05, verbose=True)
        print(f"  => score={s_best:.6f} (best={best_score:.6f})")

        if s_best < best_score:
            best_score = s_best
            best_seq = f_best.copy()
            print(f"  *** NEW BEST: {best_score:.6f} ***")

    # Phase 2: Gradient polish
    if best_seq is not None and time.monotonic() < deadline - 40:
        pt = min(120.0, deadline - time.monotonic() - 30.0)
        print(f"\n=== Phase 2: Gradient polish ({pt:.0f}s) ===")
        f_pol, s_pol = gradient_polish(best_seq, pt, verbose=True)
        if s_pol < best_score:
            best_score = s_pol
            best_seq = f_pol.copy()
            print(f"  Polished: {best_score:.6f}")

    # Phase 3: Local search
    if best_seq is not None and time.monotonic() < deadline - 25:
        ls_t = min(80.0, deadline - time.monotonic() - 15.0)
        if ls_t > 10:
            print(f"\n=== Phase 3: Local search ({ls_t:.0f}s) ===")
            f_ls, s_ls = local_search(best_seq, ls_t, verbose=True)
            if s_ls < best_score:
                best_score = s_ls
                best_seq = f_ls.copy()
                print(f"  After local search: {best_score:.6f}")

    # Phase 4: Final LP refinement
    if best_seq is not None and time.monotonic() < deadline - 10:
        fl_t = min(60.0, deadline - time.monotonic() - 5.0)
        if fl_t > 10:
            print(f"\n=== Phase 4: Final LP ({fl_t:.0f}s) ===")
            f_fin, s_fin = optimize_lp_steps(best_seq, fl_t, t_step=0.02, verbose=False)
            if s_fin < best_score:
                best_score = s_fin
                best_seq = f_fin.copy()
                print(f"  Final: {best_score:.6f}")

    # Finalize
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'='*50}")
        return final_list

    return [1.0] * 256
