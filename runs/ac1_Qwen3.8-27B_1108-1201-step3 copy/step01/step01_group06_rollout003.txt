Minimize 2*n*max_conv/sum^2 over non-negative sequences using smooth
log-sum-exp objective with FFT-based gradient computation.

Key improvements over previous:
1. Log-sum-exp smoothing gives gradient information from ALL lags (weighted

3. Combined smooth optimization + subgradient polishing.
4. Population-based search with crossover for diversity.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_full(a, b):
    """Full linear convolution of two real sequences via FFT."""
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    return np.real(ifft(A * B))[:size]


def compute_smooth_grad(a, T):
    """Compute gradient of T*log(sum(exp(conv/T))) with respect to a.
    
    Returns (gradient, objective_value).
    """
    n = len(a)
    # Step 1: Compute self-convolution
    conv = fft_convolve_full(a, a)  # length 2n-1
    
    # Step 2: Compute softmax weights
    # Use log-sum-exp trick for numerical stability
    max_conv = conv.max()
    shifted = (conv - max_conv) / T
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    
    # Smooth objective value
    obj_val = max_conv + T * np.log(exp_shifted.sum())
    
    # Step 3: Compute gradient = 2 * cross_correlation(a, p)[0:n]
    # grad[m] = 2 * sum_j a[j] * p[m+j] = 2 * (flip(a) * p)[m + n - 1]
    flip_a = a[::-1]
    cross = fft_convolve_full(flip_a, p)  # length n + (2n-1) - 1 = 3n-2
    grad = 2.0 * cross[n-1 : 2*n-1]  # length n
    
    return grad, obj_val, conv, p, max_conv


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2*n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
        s = n
    return a * (S / s)


def run_smooth_adam(a_init, n, S, max_time, rng, T_start, T_end, 
                    lr_init, momentum_betas=(0.9, 0.999)):

    Returns (best_a, best_obj, best_maxc).
    """

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    beta1, beta2, eps = momentum_betas[0], momentum_betas[1], 1e-8
    start = time.monotonic()
    
    best_obj = np.inf
    best_maxc = np.inf
    best_local = a.copy()
    
    # Total time for temperature schedule
    # Use exponential decay of T
    T_total = max_time  # total time for the schedule
    
    while time.monotonic() - start < max_time:
        elapsed = time.monotonic() - start
        # Temperature schedule: exponential decay
        frac = min(1.0, elapsed / max_time)
        T = T_start * (T_end / T_start) ** frac  # geometric interpolation
        
        grad, obj_val, conv, p, maxc = compute_smooth_grad(a, T)
        
        # Track best based on actual max (not smooth approximation)
        if maxc < best_maxc:
            best_maxc = maxc
            best_obj = obj_val
            best_local = a.copy()
        
        # Adam update
        t += 1
        lr = lr_init / (1.0 + 0.002 * t)
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-12:
            grad = grad / grad_norm
        
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)
        
        # Occasionally add noise for exploration (early on)
        if frac < 0.3 and rng.random() < 0.05:
            noise = rng.normal(0, 0.05 * S / n, n)
            a = project_simplex(a + noise, S)
    
    return best_local, best_maxc


def run_subgradient_polish(a_init, n, S, max_time, rng, lr_init, top_k=10):
    """Polish with exact subgradient (top-k active constraints)."""
    a = safe_normalize(a_init, n)
    start = time.monotonic()
    t = 0
    best_maxc = np.inf
    best_local = a.copy()
    
    while time.monotonic() - start < max_time:
        t += 1
        conv = fft_convolve_full(a, a)
        maxc = conv.max()
        if maxc < best_maxc:
            best_maxc = maxc
            best_local = a.copy()
        
        # Top-k subgradient
        top_indices = np.argsort(conv)[::-1][:top_k]
        grad = np.zeros(n, dtype=np.float64)

        total = top_vals.sum()
        if total > 1e-15:

            for w_val, k in zip(weights, top_indices):
                k = int(k)
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                if lo <= hi:

        grad = grad / gnorm
        a = project_simplex(a - lr * grad, S)

    # Gradient via two-convolution identity
    flip_a = a[::-1]
    cross = fft_convolve_full(flip_a, p)
    grad = 2.0 * cross[n - 1: 2 * n - 1]
    
    return grad, smooth_obj, max_conv, conv


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2*n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n, dtype=np.float64) * (S / n)
    return a * (S / s)


def run_smooth_adam(a_init, n, S, max_time, rng, T_start, T_end, lr_init):
    """Adam optimizer with log-sum-exp smooth objective and adaptive temperature."""
    a = safe_normalize(a_init, n)
    m_buf = np.zeros(n, dtype=np.float64)
    v_buf = np.zeros(n, dtype=np.float64)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    start = time.monotonic()
    
    best_maxc = np.inf
    best_local = a.copy()
    no_improve_count = 0
    patience = 500
    
    while time.monotonic() - start < max_time:
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max_time)
        
        # Geometric temperature decay
        T = T_start * (T_end / T_start) ** frac
        
        grad, smooth_obj, maxc, conv = compute_smooth_grad(a, T)
        
        if maxc < best_maxc - 1e-9:
            best_maxc = maxc
            best_local = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        # Restart if stuck
        if no_improve_count > patience and frac < 0.9:
            a = project_simplex(a + rng.normal(0, 0.1 * S / n, n), S)
            no_improve_count = 0
            m_buf *= 0.5
            v_buf *= 0.5
        
        t += 1
        lr = lr_init * (1.0 + 0.5 * t) ** (-0.5)
        
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-12:
            grad = grad / gnorm
        
        m_buf = beta1 * m_buf + (1.0 - beta1) * grad
        v_buf = beta2 * v_buf + (1.0 - beta2) * (grad ** 2)
        m_hat = m_buf / (1.0 - beta1 ** t)
        v_hat = v_buf / (1.0 - beta2 ** t)
        
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)
    
    return best_local, best_maxc


def run_subgradient_polish(a_init, n, S, max_time, rng, lr_init, top_k=15):
    """Polish with exact subgradient from top-k active constraints."""
    a = safe_normalize(a_init, n)
    start = time.monotonic()
    t = 0
    best_maxc = np.inf
    best_local = a.copy()
    
    while time.monotonic() - start < max_time:
        t += 1
        conv = fft_convolve_full(a, a)
        maxc = conv.max()
        if maxc < best_maxc - 1e-10:
            best_maxc = maxc
            best_local = a.copy()
        
        # Weighted subgradient from top-k lags
        top_indices = np.argsort(conv)[::-1][:top_k]
        grad = np.zeros(n, dtype=np.float64)
        top_vals = conv[top_indices]
        total = top_vals.sum()
        if total > 1e-15:
            weights = top_vals / total
            for w_val, k in zip(weights, top_indices):
                k = int(k)
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                if lo <= hi:
                    seg = a[k - hi: k - lo + 1][::-1]
                    grad[lo: hi + 1] += w_val * 2.0 * seg
        
        gnorm = np.linalg.norm(grad)
        if gnorm <= 1e-12:
            break
        
        lr = lr_init * (1.0 + 0.02 * t) ** (-0.7)
        a = project_simplex(a - lr * grad / gnorm, S)
        
        if t % 2000 == 0 and rng.random() < 0.03:
            a = project_simplex(a + rng.normal(0, 0.003 * S / n, n), S)
    
    return best_local, best_maxc


def run_differential_evolution(a_init, n, S, max_time, rng, pop_size=12):
    """Differential evolution with smooth fitness."""
    a0 = safe_normalize(a_init, n)
    
    # Initialize population
    pop = [a0.copy()]
    for _ in range(pop_size - 1):
        # Mix of perturbations
        r = rng.random()
        if r < 0.4:
            individual = a0 * (1.0 + rng.normal(0, 0.4, n))
        elif r < 0.7:
            # Block perturbation
            block_size = n // 8
            blocks = rng.choice(n // block_size, size=2, replace=False)
            individual = a0.copy()
            for b in blocks:
                individual[b * block_size: (b + 1) * block_size] *= rng.uniform(0.3, 2.0)
        else:
            individual = a0 * rng.uniform(0.5, 1.5, n)
        pop.append(safe_normalize(individual, n))
    
    # Evaluate
    def get_fitness(ind):
        conv = fft_convolve_full(ind, ind)
        return conv.max()
    
    fitness = np.array([get_fitness(ind) for ind in pop])
    best_idx = np.argmin(fitness)
    best_a = pop[best_idx].copy()
    best_val = fitness[best_idx]
    
    start = time.monotonic()
    F = 0.8  # scaling factor
    CR = 0.9  # crossover rate
    
    while time.monotonic() - start < max_time:
        if time.monotonic() - start > max_time - 3.0:
            break
        
        # Update F and CR adaptively
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max_time)
        F_cur = F * (1.0 - 0.3 * frac)
        CR_cur = CR - 0.2 * frac
        
        # Generate trial vectors
        for i in range(pop_size):
            # Select two distinct individuals different from i
            candidates = list(range(pop_size))
            candidates.remove(i)
            r1 = candidates[rng.integers(len(candidates))]
            candidates.remove(r1)
            r2 = candidates[rng.integers(len(candidates))]
            
            # Differential mutation
            mutant = pop[r1] + F_cur * (pop[r2] - pop[i])
            
            # Binary crossover
            cross_mask = rng.random(n) < CR_cur
            cross_mask[rng.integers(n)] = True  # Ensure at least one from mutant
            trial = np.where(cross_mask, mutant, pop[i])
            trial = safe_normalize(trial, n)
            
            # Fitness evaluation
            trial_fit = get_fitness(trial)
            
            if trial_fit < fitness[i]:
                pop[i] = trial
                fitness[i] = trial_fit
                if trial_fit < best_val:
                    best_val = trial_fit
                    best_a = trial.copy()
        
        # Report
        if int(elapsed) % 30 < 2:
            print("    DE: gen~%d best=%.6f elapsed=%.0fs" % (
                int(elapsed / (max_time / 50)) * 50 // max_time * 50, best_val, elapsed))
    
    return best_a, best_val


def run_mirror_descent(a_init, n, S, max_time, rng, eta_init):
    """Mirror descent (multiplicative weights) with smooth objective."""
    a = safe_normalize(a_init, n)
    start = time.monotonic()
    t = 0
    best_maxc = np.inf
    best_local = a.copy()
    
    while time.monotonic() - start < max_time:
        t += 1
        frac = min(1.0, (time.monotonic() - start) / max_time)
        T = 0.3 * (0.05 / 0.3) ** frac
        
        grad, _, maxc, _ = compute_smooth_grad(a, T)
        if maxc < best_maxc - 1e-9:
            best_maxc = maxc
            best_local = a.copy()
        
        gmax = np.abs(grad).max()
        if gmax <= 1e-12:
            break
        
        eta = eta_init * (1.0 + 0.01 * t) ** (-0.5)
        a_new = a * np.exp(-eta * grad / gmax)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s <= 1e-15:
            a = np.full(n, S / n, dtype=np.float64)
        else:
            a = a_new * (S / s)
    
    return best_local, best_maxc


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function.
    
    Multi-phase strategy:
    1. Multi-start smooth Adam (55%)
    2. Differential evolution (20%)
    3. Subgradient polish (15%)
    4. Final smooth + mirror descent polish (10%)
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 20.0
    
    global_best = np.inf
    global_a = None
    
    # Load given sequence
    given_a = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given = np.clip(given, 0, 1000)
        if given.size >= 16 and given.sum() > 0:
            given_a = given
            print("Given sequence loaded: length=%d" % given.size)
    except Exception:
        pass
    
    # Determine lengths to try (prioritize the given length)
    lengths = []
    if given_a is not None:
        lengths.append(len(given_a))
    for L in [8192, 4096, 16384]:
        if L not in lengths:
            lengths.append(L)
    # Filter valid lengths
    lengths = [L for L in lengths if 1024 <= L <= 32768]
    
    # ============ PHASE 1: Multi-start smooth Adam (55%) ============
    phase1_frac = 0.55
    phase1_time = phase1_frac * (total_budget - 20)
    phase1_deadline = time.monotonic() + phase1_time
    
    n_inits_per_length = 3
    per_init_time = phase1_time / (len(lengths) * n_inits_per_length)
    
    print("=== Phase 1: Multi-start smooth optimization ===")
    
    for n in lengths:
        if time.monotonic() > phase1_deadline - 10:
            break
        S = np.sqrt(2.0 * float(n))
        
        # Build initializations
        inits = []
        if given_a is not None:
            if len(given_a) == n:
                inits.append(("given", given_a.copy()))
            else:
                from numpy import interp
                x_old = np.linspace(0, 1, len(given_a))
                x_new = np.linspace(0, 1, n)
                inits.append(("resampled", interp(x_new, x_old, given_a)))
        
        # Random init
        inits.append(("random", rng.random(n) * 0.5 + 0.25))
        
        # Cosine-bell init
        tt = np.linspace(0, 1, n)
        bell = 1.0 + 0.4 * np.cos(2 * np.pi * tt) + 0.15 * np.cos(4 * np.pi * tt)
        inits.append(("bell", np.maximum(bell, 0.05)))
        
        # Two-peak init
        peak1 = np.exp(-((tt - 0.3) ** 2) / 0.02)
        peak2 = np.exp(-((tt - 0.7) ** 2) / 0.02)
        two_peak = 0.3 + peak1 + peak2
        inits.append(("twopeak", two_peak))
        
        for name, init in inits:
            if time.monotonic() > phase1_deadline - 5:
                break
            
            remaining_for_this = phase1_deadline - time.monotonic() - 5
            if remaining_for_this < 10:
                break
            # Cap individual time
            t_alloc = min(per_init_time * 1.5, remaining_for_this)
            
            print("  n=%d init=%s alloc=%.0fs" % (n, name, t_alloc))
            
            # Run with different temperature schedules
            T_start = 0.3
            T_end = 0.005
            lr_base = 0.4 * S / n
            
            best_a, best_score = run_smooth_adam(
                init, n, S, t_alloc, rng,
                T_start=T_start, T_end=T_end,
                lr_init=lr_base
            )
            
            if best_score < global_best:
                global_best = best_score
                global_a = best_a.copy()
                print("  *** NEW BEST: %.6f (n=%d, init=%s) ***" % (best_score, n, name))
            else:
                print("  score=%.6f (n=%d, init=%s)" % (best_score, n, name))
    
    # ============ PHASE 2: Differential Evolution (20%) ============
    phase2_frac = 0.20
    phase2_time = phase2_frac * (total_budget - 20)
    phase2_deadline = time.monotonic() + phase2_time
    
    print("=== Phase 2: Differential Evolution ===")
    
    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * float(n_best))
        remaining = min(phase2_time, phase2_deadline - time.monotonic() - 10)
        if remaining > 15:
            print("  n=%d, time=%.0fs" % (n_best, remaining))
            best_a, best_score = run_differential_evolution(
                global_a, n_best, S_best, remaining, rng, pop_size=12
            )
            if best_score < global_best:
                global_best = best_score
                global_a = best_a.copy()
                print("  *** DE IMPROVED: %.6f ***" % best_score)
            else:
                print("  DE best: %.6f" % best_score)
    
    # ============ PHASE 3: Subgradient Polish (15%) ============
    phase3_frac = 0.15
    phase3_time = phase3_frac * (total_budget - 20)
    phase3_deadline = time.monotonic() + phase3_time
    
    print("=== Phase 3: Subgradient Polish ===")
    
    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * float(n_best))
        remaining = min(phase3_time, phase3_deadline - time.monotonic() - 10)
        if remaining > 10:
            print("  n=%d, time=%.0fs" % (n_best, remaining))
            best_a, best_score = run_subgradient_polish(
                global_a, n_best, S_best, remaining, rng,
                lr_init=0.3 * S_best / n_best, top_k=20
            )
            if best_score < global_best:
                global_best = best_score
                global_a = best_a.copy()
                print("  *** POLISH IMPROVED: %.6f ***" % best_score)
            else:
                print("  Polish best: %.6f" % best_score)
    
    # ============ PHASE 4: Final Smooth + Mirror Polish (10%) ============
    phase4_frac = 0.10
    phase4_time = phase4_frac * (total_budget - 20)
    phase4_deadline = time.monotonic() + phase4_time
    
    print("=== Phase 4: Final Polish ===")
    
    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * float(n_best))
        remaining = min(phase4_time, phase4_deadline - time.monotonic() - 5)
        
        if remaining > 10:
            # Split between smooth and mirror
            half = remaining / 2.0
            
            # Smooth with very low T
            best_a, best_score = run_smooth_adam(
                global_a, n_best, S_best, half, rng,
                T_start=0.01, T_end=0.001,
                lr_init=0.15 * S_best / n_best
            )
            if best_score < global_best:
                global_best = best_score
                global_a = best_a.copy()
                print("  Smooth polish: %.6f" % best_score)
            
            # Mirror descent
            remaining2 = phase4_deadline - time.monotonic() - 5
            if remaining2 > 5:
                best_a, best_score = run_mirror_descent(
                    global_a, n_best, S_best, remaining2, rng,
                    eta_init=0.2
                )
                if best_score < global_best:
                    global_best = best_score
                    global_a = best_a.copy()
                    print("  Mirror polish: %.6f" % best_score)
    
    # ============ FINAL VERIFICATION ============
    print("=== Final Verification ===")
    
    if global_a is not None:
        final_a = np.clip(global_a, 0.0, 1000.0)
        if final_a.sum() < 0.01:
            final_a = np.ones(256, dtype=np.float64)
        
        ver = evaluate_sequence(final_a.tolist())
        if np.isfinite(ver):
            print("evaluate_sequence = %.6f" % ver)
            if ver < global_best:
                global_best = ver
        else:
            print("evaluate_sequence = inf, using fallback")
            final_a = np.ones(256, dtype=np.float64)
            global_a = final_a
    else:
        global_a = np.ones(256, dtype=np.float64)
        ver = evaluate_sequence(global_a.tolist())
        print("Fallback evaluate_sequence = %.6f" % ver)
    
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
