import cvxpy as cp

def lp_direction(f, M):
    """Solve LP: max sum(b) s.t. b >= 0, f*b <= M (elementwise)."""
    n = len(f)
    b = cp.Variable(n, nonneg=True)
    constraints = [cp.sum(b) <= cp.Constant(np.sum(f))]  # normalize
    # Actually, we want to maximize sum(b) without the normalize constraint
    # The normalization is applied after
    
    # Convolution constraint: (f*b)[k] <= M for all k
    # (f*b)[k] = sum_j f[j] * b[k-j] (valid indices)
    # This is a linear constraint in b
    
    # Build the constraint matrix
    # For each k, the constraint is: sum_j f[j] * b[k-j] <= M
    # The coefficient of b[i] in constraint k is f[k-i] (if valid)
    
    # Actually, let's just compute f*b for symbolic b... that's expensive.
    # Better: use the FFT to compute the convolution for a given b,
    # but for the LP we need linear constraints.
    
    # Let's build the constraint matrix T where T[k,i] = f[k-i] if 0<=k-i<n, else 0
    # Then the constraint is T @ b <= M * ones(2n-1)
    
    # T is (2n-1) x n, which is fine for n up to ~10000.
    
    T = np.zeros((2*n-1, n))
    for i in range(n):
        for k in range(max(0, i), min(2*n-1, i+n)):
            # b[i] contributes to c[k] with coefficient f[k-i]
            j = k - i
            if 0 <= j < n:
                T[k, i] = f[j]
    
    # This is O(n^2) which is fine for n <= 1000.
    
    obj = cp.Maximize(cp.sum(b))
    constraints = [T @ b <= M]
    prob = cp.Problem(obj, constraints)
    prob.solve()
    
    return b.value
```

    n = len(a)

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid LP-based and mirror
descent optimization. The LP-based method solves a linear program to find
the optimal descent direction at each step. Multiple sequence lengths are
tried, with diverse initializations.
"""

import numpy as np
import time
import cvxpy as cp

def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T where (T @ b)[k] = (a*b)[k]."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for i in range(n):
        T[i:i + n, i] = a
    return T

def lp_direction(a, M):
    """Solve LP: max sum(b) s.t. b >= 0, a*b <= M (elementwise). Returns b normalized to sum 1."""
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=cp.ECOS, abs_tol=1e-9, rel_tol=1e-9)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, eps=1e-9)
        except Exception:
            return None
    if b_var.value is None:
        return None
    b = np.maximum(b_var.value, 0)
    b_sum = b.sum()
    if b_sum < 1e-12:
        return None
    return b / b_sum

def mirror_descent_step(a, t_param, eta):
    """One step of multiplicative mirror descent with soft-max subgradient."""
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    if M < 1e-15:
        return a, conv
    
    # Soft-max weights
    scaled = t_param * conv / M
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    w = w / w_sum
    
    # Subgradient: sg[i] = sum_k w[k] * a[k-i] (valid)
    sg = np.zeros(n)
    thresh = w.max() * 0.001
    active_idx = np.where(w > thresh)[0]
    for k_int in active_idx:
        k = int(k_int)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        sg[j_start:j_end + 1] += w[k] * a[k - j_end:k - j_start + 1][::-1]
    
    # Multiplicative update
    sg_centered = sg - sg.mean()
    log_factor = -eta * sg_centered
    log_factor -= log_factor.max()
    a_new = a * np.exp(log_factor)
    a_new = np.maximum(a_new, 0)
    a_sum = a_new.sum()
    if a_sum < 1e-12:
        a_new = np.ones(n) / n
    else:
        a_new = a_new / a_sum
    return a_new, conv


def run_lp_optimizer(a0, time_limit, t0=0.15):
    """LP-based optimization loop."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / a_sum
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t = t0
    patience = 0
    max_patience = 40
    t0_time = time.monotonic()
    iter_count = 0
    
    while time.monotonic() - t0_time < time_limit:
        iter_count += 1
        conv = fast_convolve(a)
        M = conv.max()
        current_obj = 2.0 * n * M
        
        # Get LP direction
        b = lp_direction(a, M)
        if b is None:
            break
        
        # Update
        a_new = (1 - t) * a + t * b
        a_new = np.maximum(a_new, 0)
        a_sum = a_new.sum()
        if a_sum < 1e-12:
            break
        a_new = a_new / a_sum
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            t = min(t * 1.02, 0.4)
            patience = 0
        else:
            patience += 1
            t *= 0.85
            if patience >= max_patience:
                if t < 0.02:
                    break
                patience = 0
                t = max(t0 * 0.5, 0.05)
    
    return best_a, best_obj

    """Mirror descent optimization loop."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64),
