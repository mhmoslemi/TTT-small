def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=0.0)

    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)

    if not np.any(cond):
        out = np.full(n, 1.0 / n, dtype=np.float64)
        return out

    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def safe_normalize(p):

        n = max(1, int(p.size))
        return np.full(n, 1.0 / n, dtype=np.float64)
    return p / s


def shape_from_theta(theta):
    """
    Build a positive vector from a Chebyshev polynomial plus endpoint log-taper.
    Uses global _SHAPE_N and _SHAPE_X.
    """
    n = int(_SHAPE_N)
    x = _SHAPE_X
    if x is None or x.size != n:
        x = np.linspace(0.0, 1.0, n)

    D = int(_SHAPE_DEG)
    theta = np.asarray(theta, dtype=np.float64)
    need = D + 2
    if theta.size < need:
        theta = np.concatenate((theta, np.zeros(need - int(theta.size), dtype=np.float64)))
    else:
        theta = theta[:need].copy()

    logp = np.zeros(n, dtype=np.float64)
    z = 2.0 * x - 1.0

    if D >= 1:
        T_nm = np.ones(n, dtype=np.float64)
        T_n = z.copy()
        logp += theta[0] * T_n
        for d in range(2, D + 1):
            T_np = 2.0 * z * T_n - T_nm
            logp += theta[d - 1] * T_np
            T_nm = T_n
            T_n = T_np

    eps = 1e-4
    logp += theta[D] * np.log(x + eps) + theta[D + 1] * np.log(1.0 - x + eps)

    if not np.all(np.isfinite(logp)):
        return np.full(n, 1.0 / n, dtype=np.float64)

    m = float(np.max(logp))
    if not np.isfinite(m):
        return np.full(n, 1.0 / n, dtype=np.float64)

    logp = logp - m
    p = np.exp(logp)
    return safe_normalize(p)

def make_uniform(n):
    n = int(max(1, n))

def make_linear(n, a, reverse=False):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    p = 1.0 + float(a) * x
    if reverse:

def make_power(n, a, reverse=False):
    n = int(max(2, n))
    x = np.linspace(1e-5, 1.0, n)
    p = np.power(x, float(a))
    if reverse:
        p = p[::-1]

def make_cup(n, a):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    p = 1.0 + float(a) * (2.0 * x - 1.0) ** 2
    return safe_normalize(p)


def make_cap(n, a):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    p = 1.0 - float(a) * (2.0 * x - 1.0) ** 2
    return safe_normalize(np.maximum(p, 0.0))


def make_geometric(n, s, reverse=False):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    p = np.exp(-float(s) * x)
    if reverse:
        p = p[::-1]
    return safe_normalize(p)


def make_beta(n, alpha, beta):
    n = int(max(2, n))
    x = np.linspace(1e-5, 1.0 - 1e-5, n)
    p = np.power(x, float(alpha)) * np.power(1.0 - x, float(beta))
    return safe_normalize(np.maximum(p, 0.0))


def make_skew_gauss(n, a, mu, b):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    dx = x - float(mu)
    p = np.exp(-float(a) * (dx * dx) + float(b) * dx)
    return safe_normalize(p)


def make_two_gauss(n, mu1, sig1, w1, mu2, sig2, w2):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    p = float(w1) * np.exp(-((x - mu1) ** 2) / (2.0 * max(1e-4, float(sig1)) ** 2))
    p += float(w2) * np.exp(-((x - mu2) ** 2) / (2.0 * max(1e-4, float(sig2)) ** 2))
    return safe_normalize(np.maximum(p, 0.0))

def lp_best_response(p, best_score, rng, budget_s):
    """
    Optional exact best-response LP:
        max sum(g)
        s.t. f * g <= M, g >= 0,
    where f = p + eps, M = max(p*p).
    """
