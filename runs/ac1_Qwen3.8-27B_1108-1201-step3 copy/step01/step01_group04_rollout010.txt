def fft_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

def fft_cross_corr(w, a):
    """
    Compute cross-correlation: result[i] = sum_j w[j+i] * a[j] for valid j.
    Uses FFT.
    """
    n = len(a)
    # As derived: result[i] = 2 * (reverse(w) * a)[2n-2-i] / 2
    # Actually: result[i] = (reverse(w) * a)[2n-2-i]

    return conv[idx]

def softmax_grad_step(a, tau, lr):
    n = len(a)
    c = fft_convolve(a)
    
    # Softmax weights
    c_shifted = c - c.max()
    exponents = tau * c_shifted
    exponents -= exponents.max()  # numerical stability
    w = np.exp(exponents)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(c.argmax())] = 1.0
    else:
        w /= w_sum
    
    # Gradient
    grad = 2.0 * fft_cross_corr(w, a)
    
    # Project (fix sum)
    grad -= grad.mean()
    
    # Step
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    
    # Normalize
    S_target = np.sqrt(2.0 * n)
    S = a_new.sum()
    if S < 1e-12:
        a_new = np.full(n, S_target / n)
    else:
        a_new *= S_target / S
    
    return a_new

def lp_refine(a, M, deadline, max_iters=50):
    """LP-based refinement: find g that maximizes sum(g) s.t. (a*g) <= M."""
    n = len(a)
    if n > 1024:
        return a
    
    S_target = np.sqrt(2.0 * n)
    best = a.copy()
    best_score = compute_obj(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break
        
        # Build LP
        c_obj = -np.ones(n)
        A = np.zeros((2*n-1, n))
        for m in range(2*n-1):
            lo = max(0, m - n + 1)
            hi = min(n-1, m)
            for j in range(lo, hi+1):
                A[m, j] = a[m-j]
        
        b = np.full(2*n-1, M)
        bounds = [(0, None)] * n

            if res.status != 0:
                break

        except:
            break
        
        # Blend
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1]:

            a_new = np.maximum(a_new, 0)

            s = compute_obj(a_new)

                best = a.copy()
                best_score = s
                improved = True
                break

            break

        c = fft_convolve(a)
        M = float(c.max())

    """
    Compute grad[i] = 2 * sum_j a[j] * w[i+j]
    using FFT.

"""
Optimizes non-negative sequences to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses FFT-accelerated softmax gradient descent with adaptive annealing,
noise injection for escaping local minima, and LP-based polish for final refinement.
Explores multiple sequence lengths and initializations within the time budget.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def fft_convolve(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_gradient(a, w):
    """
    Compute grad[i] = 2 * sum_j a[j] * w[i+j] using FFT.
    w has length 2n-1, a has length n.
    Uses: grad[i]/2 = (a * reverse(w))[2n-2-i]
    """
    n = len(a)
    rw = w[::-1]
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    R = np.fft.rfft(rw, fft_size)
    conv = np.fft.irfft(A * R, fft_size)[:conv_len]
    indices = 2 * n - 2 - np.arange(n)
    return 2.0 * conv[indices]


def normalize_seq(a, n):
    """Normalize sequence to sum sqrt(2n)."""
    S_target = np.sqrt(2.0 * n)
    S = a.sum()
    if S < 1e-12:
        return np.full(n, S_target / n)
    return a * (S_target / S)


def softmax_grad_step(a, tau, lr):
    """One projected gradient step using softmax approximation to max."""
    n = len(a)
    c = fft_convolve(a)

    # Softmax weights with numerical stability
    c_shifted = c - c.max()
    exponents = tau * c_shifted
    exponents -= exponents.max()
    w = np.exp(exponents)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum

    # Gradient via FFT cross-correlation
    grad = compute_gradient(a, w)

    # Project out mean (maintain sum constraint)
    grad -= grad.mean()

    # Step
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize_seq(a_new, n)

    return a_new


def optimize_softmax(a0, deadline, max_iters=10000, verbose=False):
    """
    Softmax gradient descent with adaptive annealing and noise injection.
    Returns (best_sequence, best_score).
    """
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0)
    if a.sum() < 1e-12:
        a = np.ones(n)
    a = normalize_seq(a, n)

    best = a.copy()
    best_score = compute_obj(a)
    last_improve_iter = 0

    # Annealing parameters
    tau_start = 3.0
    tau_end = 60.0
    lr_start = 0.4
    lr_end = 0.005
    anneal_frac = 0.6  # fraction of iterations for annealing

    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break

        frac = it / max(1.0, max_iters)

        # Anneal temperature: increase from tau_start to tau_end
        tau_frac = min(1.0, it / max(1, int(max_iters * anneal_frac)))
        tau = tau_start + (tau_end - tau_start) * tau_frac

        # Cosine decay learning rate
        lr = lr_end + (lr_start - lr_end) * 0.5 * (1.0 + np.cos(np.pi * frac))

        a = softmax_grad_step(a, tau, lr)

        # Noise injection when stuck
        if it - last_improve_iter > 400 and it > 100:
            noise_scale = 0.03 * a.mean()
            a = a + noise_scale * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a = normalize_seq(a, n)

        # Check improvement every 30 iterations
        if (it + 1) % 30 == 0:
            score = compute_obj(a)
            if score < best_score - 1e-10:
                best_score = score
                best = a.copy()
                last_improve_iter = it
            elif it - last_improve_iter > 800 and it > 200:
                # Stronger perturbation
                noise_scale = 0.1 * a.mean()
                a = a + noise_scale * np.random.randn(n)
                a = np.maximum(a, 0.0)
                a = normalize_seq(a, n)

    # Final check
    score = compute_obj(a)
    if score < best_score:
        best_score = score
        best = a.copy()

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    optimize_softmax: n={n}, iters={it+1}, score={best_score:.6f}, time={elapsed:.1f}s")

    return best, best_score


def lp_polish(a, deadline, max_iters=80):
    """
    LP-based refinement: iteratively solve LP to find best direction g,
    then blend. Returns (best_sequence, best_score).
    """
    n = len(a)
    if n > 1536:
        return a, compute_obj(a)

    S_target = np.sqrt(2.0 * n)
    best = a.copy()
    best_score = compute_obj(a)

    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break

        c = fft_convolve(a)
        M = float(c.max())

        # Build LP: maximize sum(g) s.t. (a*g)_k <= M, g >= 0
        # (a*g)_k = sum_j g[j] * a[k-j] for valid j
        # So constraint matrix A[m,j] = a[m-j] if 0<=m-j<n
        A = np.zeros((2 * n - 1, n))
        for m in range(2 * n - 1):
            lo = max(0, m - n + 1)
            hi = min(n - 1, m)
            for j in range(lo, hi + 1):
                A[m, j] = a[m - j]

        c_obj = -np.ones(n)
        b = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 20})
            if res.status != 0:
                break
            g = res.x
            if g is None or not np.all(np.isfinite(g)):
                break
        except Exception:
            break

        # Try blending
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            S = a_new.sum()
            if S < 1e-12:
                break
            a_new = a_new * (S_target / S)
            s = compute_obj(a_new)
            if s < best_score - 1e-10:
                a = a_new
                best = a.copy()
                best_score = s
                improved = True
                break

        if not improved:
            # Try a small gradient step to unstick
            c_cur = fft_convolve(a)
            k_star = int(np.argmax(c_cur))
            grad = np.zeros(n)
            lo = max(0, k_star - n + 1)
            hi = min(n - 1, k_star)
            for j in range(lo, hi + 1):
                grad[j] = 2.0 * a[k_star - j]
            grad -= grad.mean()
            a = a - 0.05 * grad
            a = np.maximum(a, 0.0)
            a = normalize_seq(a, n)
            s = compute_obj(a)
            if s < best_score:
                best = a.copy()
                best_score = s
                improved = True

        if not improved:
            break

    return best, best_score
