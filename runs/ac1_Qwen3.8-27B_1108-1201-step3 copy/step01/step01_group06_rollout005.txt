"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Strategy:
  - Start from the provided good sequence (height_sequence_1)
  - Use refined subgradient methods with exact active sets and adaptive step sizes
  - Try multiple sequence lengths (4096, 8192, 16384)
  - Periodic reheating to escape local minima
  - Polish with very fine step sizes at the end

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution of a (length 2*n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgradient_full(a, n, active_indices, c):
    """Subgradient using the full active set with proportional weights."""
    grad = np.zeros(n, dtype=np.float64)
    if len(active_indices) == 0:
        return grad
    top_vals = c[active_indices]
    total = top_vals.sum()
    if total <= 1e-15:
        return grad
    weights = top_vals / total
    for w_val, k in zip(weights, active_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w_val * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2*n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
    return a * (S / s)

    """Refined subgradient method with adaptive step and exact active set."""
    a = safe_normalize(a_init, n)
    S = np.sqrt(2.0 * float(n))
    F_lower = S * S / (2.0 * n - 1.0)  # theoretical lower bound
    
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    t = 0
    step = step_scale * S / n
    
    # Use backtracking
    patience = 0
    max_patience = 50
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        
        if cur < best_score - 1e-10:
            best_score = cur
            best_local = a.copy()
            patience = 0
        else:
            patience += 1

            patience = 0

            a = project_simplex(a + pert, S)

        # Active set: all lags within active_frac of the max
        threshold = cur - active_frac * cur
        active_mask = c >= threshold
        active_indices = np.where(active_mask)[0]
        
        if len(active_indices) == 0:

def lp_iteration(f, M, max_time_lp=30.0):
    """One LP-based iteration: find b maximizing sum(b) s.t. f*b <= M, b >= 0.
    Returns the optimal b (unnormalized) or None if the LP times out."""
    from scipy.optimize import linprog
    import scipy.sparse as sp
    
    n = len(f)
    # Constraint: for each k in 0..2n-2, sum_j f[j]*b[k-j] <= M
    # The (k,j) entry of the constraint matrix is f[j] if 0 <= k-j < n, else 0
    # This is a Toeplitz matrix.
    
    # For efficiency, use the structure: row k has f[n-1-k] at col 0, f[n-2-k] at col 1, etc.
    # Actually, let's build it row by row.
    # Row k: for j in max(0,k-n+1)..min(n-1,k), entry is f[k-j]
    
    # For large n, this matrix is too big for dense linprog. Let's use a smaller n.
    if n > 2048:
        return None
    
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = sp.coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()
    
    # Objective: maximize sum(b) -> minimize -sum(b)
    c_obj = -np.ones(n)
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M),
                         bounds=[(0, None)] * n, method='highs',
                         options={'time_limit': max_time_lp})
        if result.status == 0:
            return result.x
    except Exception:
        pass
    return None

The problem in log-space: minimize F(x) = max_k g_k(x) subject to h(x) = sum(exp(x_j)) - S = 0.

Using the method of Lagrange multipliers, at an optimum:
g*_j + lambda * exp(x*_j) = 0 for all j (where g* is the active subgradient)

This means: a*_j * (grad_a*_j / a*_j) + lambda * a*_j = 0
=> grad_a*_j + lambda * a*_j = 0
=> grad_a*_j = -lambda * a*_j

So at the optimum, the a-space subgradient is proportional to -a! This is a very strong condition.

The multiplicative update a_j <- a_j * exp(-alpha * a_j * grad_a_j) moves us in the direction that makes grad_a_j more negative (or less positive) for large a_j, which is the right direction.

Let me think about whether there's an even better update. The condition grad_a_j = -lambda * a_j suggests that we should equalize grad_a_j / a_j across all j. The multiplicative update does this approximately.

An alternative: we could use the KKT condition directly and solve for the optimal lambda at each step. This would be like a "one-step Newton" method.

At each iteration:
1. Compute the subgradient g = grad_a (for the active set)
2. Find lambda such that sum(exp(x - alpha*(g + lambda*exp(x)))) = S
   (this is the " projected" update that satisfies the constraint)
3. Update x <- x - alpha*(g + lambda*exp(x))

This is equivalent to:
a_new_j = a_j * exp(-alpha * (g_j + lambda * a_j))
Choose lambda so that sum(a_new) = S.

This is a 1D root-finding problem for lambda, which can be solved by bisection. This might converge faster than the simple multiplicative update.

        if given_a is not None and len(given_a) > 1:
            # Interpolate
            x_old = np.linspace(0, 1, len(given_a))
            x_new = np.linspace(0, 1, n)
            # Simple linear interpolation
            a_interp = np.interp(x_new, x_old, given_a)
            a_interp = np.maximum(a_interp, 0)
            return a_interp
        # Fallback: structured random
        tt = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) + 0.1 * np.cos(2 * np.pi * 7 * tt)
        a = np.maximum(a, 0) * (0.8 + 0.4 * rng.random())
        return a

    # Main optimization loop over multiple lengths
    lengths = [8192, 16384, 4096]

    for n in lengths:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_init = get_init_for_length(n, rng)
        a_init = safe_normalize(a_init, n)

        init_score = evaluate_sequence(a_init.tolist())
        print(f"n={n}, initial score: {init_score:.6f}")
        if np.isfinite(init_score) and init_score < global_best:
            global_best = init_score
            global_a = a_init.copy()

        # Run KKT method (main)
        remaining = deadline - time.monotonic()
        kkt_time = remaining * 0.45
        if kkt_time < 10:
            kkt_time = remaining * 0.7

        a1, s1 = run_kkt(a_init, n, kkt_time, rng, alpha_init=0.05, active_frac=0.05)
        ver1 = evaluate_sequence(a1.tolist())
        print(f"n={n}, KKT score: {s1:.6f}, verified: {ver1:.6f}")
        if np.isfinite(ver1) and ver1 < global_best:
            global_best = ver1
            global_a = a1.copy()

        # Run Euclidean subgradient (secondary)
        remaining = deadline - time.monotonic()
        if remaining > 30:
            eucl_time = remaining * 0.25
            a2, s2 = run_euclidean(a_init, n, eucl_time, rng, lr_init=1.0, active_frac=0.05)
            ver2 = evaluate_sequence(a2.tolist())
            print(f"n={n}, Euclidean score: {s2:.6f}, verified: {ver2:.6f}")
            if np.isfinite(ver2) and ver2 < global_best:
                global_best = ver2
                global_a = a2.copy()

        # Continue from best with smaller active fraction (more focused)
        remaining = deadline - time.monotonic()
        if remaining > 30 and global_a is not None:
            focus_time = remaining * 0.3
            a3, s3 = run_kkt(global_a, len(global_a), focus_time, rng,
                             alpha_init=0.02, active_frac=0.02, decay=0.9998)
            ver3 = evaluate_sequence(a3.tolist())
            print(f"Focus KKT: score={s3:.6f}, verified={ver3:.6f}")
            if np.isfinite(ver3) and ver3 < global_best:
                global_best = ver3
                global_a = a3.copy()

    # Final polish with very small alpha
    remaining = deadline - time.monotonic()
    if remaining > 20 and global_a is not None:
        n = len(global_a)
        polish_time = remaining * 0.8
        a_polish, s_polish = run_kkt(global_a, n, polish_time, rng,
                                     alpha_init=0.01, active_frac=0.01, decay=0.9999)
        ver_polish = evaluate_sequence(a_polish.tolist())
        print(f"Final polish: score={s_polish:.6f}, verified={ver_polish:.6f}")
        if np.isfinite(ver_polish) and ver_polish < global_best:
            global_best = ver_polish
            global_a = a_polish.copy()

    print(f"FINAL BEST: {global_best:.6f}, length={len(global_a) if global_a is not None else 0}")

    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    return [float(x) for x in global_a]
