```python
import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.fft import fft, ifft, next_fast_len

def internal_score(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.inf
    s = float(np.sum(p))
    if s <= 1e-12:
        return np.inf
    # Use FFT for convolution
    L = next_fast_len(2*n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))[:2*n-1]
    cmax = float(np.max(c))
    return float(2.0 * n * cmax / (s * s))

def safe_normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s <= 1e-12:
        return np.full(p.size, 1.0/p.size)
    return p / s

def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = (u * idx) > (cssum - 1.0)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / (k + 1)
    return np.maximum(v - theta, 0.0)

def lp_best_response(a):
    """
    Solve: min t
           s.t. (a*b)[k] <= t for all k
                sum(b) = 1
                b >= 0
    Returns (b_opt, t_opt).
    """
    n = len(a)
    # Build constraint matrix
    # (a*b)[k] <= t  <=>  sum_i a[i]*b[k-i] - t <= 0
    # Variables: b[0], ..., b[n-1], t
    # Total variables: n + 1
    
    # For constraint k: sum_{i: 0<=i<n, 0<=k-i<n} a[i]*b[k-i] - t <= 0
    # In matrix form: row k has a[i] in column (k-i) for valid i, and -1 in column n (the t column)
    
    rows = []
    cols = []
    data = []
    
    for k in range(2*n - 1):
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            j = k - i  # column index for b[j]
            if 0 <= j < n:
                rows.append(k)
                cols.append(j)
                data.append(a[i])
        # t coefficient
        rows.append(k)
        cols.append(n)  # t is the (n)-th variable (0-indexed)
        data.append(-1.0)
    
    A_ub = csr_matrix((data, (rows, cols)), shape=(2*n-1, n+1))
    b_ub = np.zeros(2*n - 1)
    
    # Objective: minimize t = maximize 0*b + 1*t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    # sum(b) = 1 is a constraint: sum_{i=0}^{n-1} b[i] = 1, 0*t = 0
    A_eq = csr_matrix((np.ones(n), (np.arange(n), np.arange(n))), shape=(1, n+1))
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]  # t >= 0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if not res.success:
        return None, None
    
    b_opt = res.x[:n]
    t_opt = res.x[n]
    return b_opt, t_opt
```

"""
LP-based best-response algorithm for minimizing max autocorrelation of non-negative sequences.
Uses the zero-sum game formulation: min_a max_b max_k (a*b)[k] s.t. a,b on simplex.
The LP finds the exact best response at each iteration.
"""
import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.fft import fft, ifft, next_fast_len

def internal_score(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.inf
    s = float(np.sum(p))
    if s <= 1e-12:
        return np.inf
    L = next_fast_len(2 * n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))[:2 * n - 1]
    cmax = float(np.max(c))
    return float(2.0 * n * cmax / (s * s))

def safe_normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s <= 1e-12:
        return np.full(max(1, p.size), 1.0 / max(1, p.size))
    return p / s

def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s <= 1e-300:
        return np.full(n, 1.0 / n)
    return out / s

def lp_best_response(a, n):
    """
    Solve: min t
           s.t. (a*b)[k] <= t for k = 0,...,2n-2
                sum(b) = 1
                b >= 0, t >= 0
    """
    # Build sparse constraint matrix
    rows = []
    cols = []
    data = []
    
    for k in range(2 * n - 1):
        # (a*b)[k] = sum_{i} a[i] * b[k-i] where 0<=i<n, 0<=k-i<n
        # b[j] appears when j = k-i, so i = k-j, valid when 0<=k-j<n -> k-n+1<=j<=k
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            i = k - j
            rows.append(k)
            cols.append(j)
            data.append(a[i])
        # t coefficient: -1
        rows.append(k)
        cols.append(n)
        data.append(-1.0)
    
    A_ub = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)
    
    # Objective: minimize t (variable index n)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Equality: sum(b) = 1
    A_eq_data = np.ones(n)
    A_eq = csr_matrix((A_eq_data, (np.arange(n), np.arange(n))), shape=(1, n + 1))
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'disp': False, 'time_limit': 60})
    except Exception:
        return None, None
    
    if res is None or not res.success:
        return None, None
    
    b_opt = np.asarray(res.x[:n], dtype=np.float64)
    t_opt = float(res.x[n])
    b_opt = np.maximum(b_opt, 0.0)
    s = b_opt.sum()
    if s <= 1e-12:
        return None, None
    b_opt = b_opt / s
    return b_opt, t_opt
