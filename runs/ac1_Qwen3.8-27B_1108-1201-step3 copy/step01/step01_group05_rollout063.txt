        out.append(normalize(x ** a))
        out.append(normalize((1.0 - x + 1e-5) ** a))

    for a in [0.2, 0.5, 1.0, 1.5, 2.0, 3.0]:
        x = np.linspace(0, 1, n)
        out.append(normalize(np.maximum(1.0 + a * np.cos(np.pi * x), 0.01)))
        out.append(normalize(np.maximum(1.0 - a * (2.0 * x - 1.0) ** 2, 0.01)))
        out.append(normalize(np.sin(np.pi * x) ** a))

    for al, be in [(0.5, 0.5), (1, 2), (2, 1), (0.5, 2), (2, 0.5),
                   (0.3, 1), (1, 0.3), (0.7, 0.7), (1.5, 1.5),
                   (0.4, 1.5), (1.5, 0.4), (0.2, 2), (2, 0.2)]:
        x = np.linspace(1e-5, 1 - 1e-5, n)
        out.append(normalize(x ** al * (1.0 - x) ** be))

    for m1, m2, w1, w2 in [(0.3, 0.7, 1, 1), (0.25, 0.75, 1.5, 1),
                           (0.4, 0.6, 1, 1.5), (0.2, 0.5, 1, 0.7),
                           (0.5, 0.8, 1.2, 0.8)]:
        x = np.linspace(0, 1, n)
        sig = max(0.03, 0.12)
        p = w1 * np.exp(-((x - m1) / sig) ** 2) + w2 * np.exp(-((x - m2) / (sig * 1.3)) ** 2)
        out.append(normalize(p))

    for f in [1, 2, 3, 4, 5, 7, 11]:
        x = np.linspace(0, 1, n)
        out.append(normalize(1.0 + 0.4 * np.sin(2 * np.pi * f * x)))
        out.append(normalize(1.0 + 0.3 * np.cos(2 * np.pi * f * x)))
        out.append(normalize(np.maximum(1.0 + 0.3 * np.sin(2 * np.pi * f * x)
                                        + 0.2 * np.sin(4 * np.pi * f * x), 0.01)))

    for _ in range(20):
        x = np.linspace(0, 1, n)
        f1 = int(rng.integers(1, max(2, n // 40)))
        f2 = int(rng.integers(1, max(2, n // 20)))
        f3 = int(rng.integers(1, max(2, n // 10)))
        p = (1.0
             + 0.4 * float(rng.uniform(0.5, 1.5)) * np.sin(2 * np.pi * f1 * x + rng.uniform(0, 2 * np.pi))
             + 0.25 * float(rng.uniform(0.5, 1.5)) * np.sin(2 * np.pi * f2 * x + rng.uniform(0, 2 * np.pi))
             + 0.1 * float(rng.uniform(0.5, 1.5)) * np.sin(2 * np.pi * f3 * x + rng.uniform(0, 2 * np.pi)))
        out.append(normalize(np.maximum(p, 0.01)))

    return out


def lp_best_response(p, budget_s):
    if not _HAVE_LP:
        return None
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n <= 0 or n > 600:
        return None
    if budget_s <= 1.0:
        return None
    c = conv_full(p)
    M = float(c.max()) if c.size else 0.0
    if not np.isfinite(M) or M <= 0:
        return None
    f = p + 1e-14
    fs = float(f.sum())
    f = f / fs
    try:
        mid = np.arange(n, dtype=np.int32)
        rows = (np.arange(n, dtype=np.int32)[:, None] + mid[None, :]).ravel().astype(np.int32)
        cols = np.repeat(np.arange(n, dtype=np.int32), n)
        data = np.tile(f, n).astype(np.float64)
        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0.0, None),
                      method='highs', options={'disp': False})
    except Exception:
        return None
    if res is None or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.maximum(np.where(np.isfinite(g), g, 0.0), 0.0)
    s = float(g.sum())
    if s < 1e-12:
        return None
    return g / s


def optimize_local(p0, rng, time_limit, label=""):
    p = normalize(np.asarray(p0, dtype=np.float64)).copy()
    n = p.size
    if n == 0:
        return p, np.inf
    best_p = p.copy()
    best_score = compute_score(p)
    cur_score = best_score
    prev_p = p.copy()

    end = time.monotonic() + max(0.0, time_limit)
    iter_no = 0
    stale = 0
    sa_temp = max(best_score * 0.015, 0.005)

    # Try LP at the start
    if _HAVE_LP and n <= 600 and time_limit > 10:
        g = lp_best_response(p, min(40.0, time_limit * 0.15))
        if g is not None:
            for t in [1.0, 0.7, 0.5, 0.3, 0.15]:
                q = project_simplex((1 - t) * p + t * g)
                sc = compute_score(q)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_p = q.copy()
                    p = q.copy()
                    prev_p = q.copy()
                    cur_score = sc

    while time.monotonic() < end:
        iter_no += 1
        accepted = False

        # Adaptive temperature
        T_frac = max(0.004, 0.08 * (0.9985 ** iter_no))
        if stale > 150:
            T_frac = max(0.003, T_frac * 0.85)

        # Gradient step
        grad, c, cmax, T_val, surr = smooth_max_and_grad(p, T_frac)
        if np.all(np.isfinite(grad)):
            gmax = float(np.max(np.abs(grad)))
            if gmax > 1e-15:
                base_lr = max(1.0 / n, 0.02 * float(p.max())) / gmax
                lrs = [base_lr * f for f in [0.35, 0.08, 0.02, 0.005, 0.001, 0.0003]]

                for lr in lrs:
                    if lr <= 0 or not np.isfinite(lr):
                        continue
                    v = p - lr * grad
                    # Add momentum
                    if iter_no > 50:
                        v = v + 0.3 * (p - prev_p)
                    pt = project_simplex(v)
                    sc = compute_score(pt)
                    if not np.isfinite(sc):
                        continue
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_p = pt.copy()
                        stale = 0
                    if sc < cur_score - 1e-12:
                        prev_p = p.copy()
                        p = pt.copy()
                        cur_score = sc
                        accepted = True
                        break
                    # SA acceptance for slightly worse moves
                    delta = sc - cur_score
                    if delta < sa_temp and rng.random() < np.exp(-delta / max(sa_temp, 1e-10)):
                        prev_p = p.copy()
                        p = pt.copy()
                        cur_score = sc
                        stale = max(0, stale - 1)
                        accepted = True
                        break

        # Peak shaving
        if not accepted and stale > 30:
            c = conv_full(p)
            k_peak = int(np.argmax(c))
            lo_i = max(0, k_peak - n + 1)
            hi_i = min(n, k_peak + 1)
            if hi_i > lo_i:
                valid_i = np.arange(lo_i, hi_i)
                contribs = np.array([p[i] * p[k_peak - i] for i in valid_i])
                i_star = valid_i[int(np.argmax(contribs))]
                # Choose transfer target: low value positions
                below_median = np.where(p < np.median(p))[0]
                if below_median.size > 0:
                    j = int(below_median[rng.integers(len(below_median))])
                    if j != i_star:
                        for frac in [0.15, 0.07, 0.03, 0.01]:
                            delta = frac * p[i_star]
                            if delta <= 0:
                                continue
                            pt = p.copy()
                            pt[i_star] -= delta
                            pt[j] += delta
                            pt = project_simplex(pt)
                            sc = compute_score(pt)
                            if not np.isfinite(sc):
                                continue
                            if sc < best_score - 1e-12:
                                best_score = sc
                                best_p = pt.copy()
                                stale = 0
                            if sc < cur_score - 1e-12:
                                prev_p = p.copy()
                                p = pt.copy()
                                cur_score = sc
                                accepted = True
                                break

        # Block perturbation
        if not accepted:
            block_len = max(2, n // 20)
            for _range in range(3):
                start = int(rng.integers(0, max(1, n - block_len)))
                end_b = min(n, start + block_len)
                pt = p.copy()
                pt[start:end_b] += rng.normal(0, 0.05 / n, end_b - start)
                pt = project_simplex(pt)
                sc = compute_score(pt)
                if not np.isfinite(sc):
                    continue
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_p = pt.copy()
                    stale = 0
                if sc < cur_score - 1e-12:
                    prev_p = p.copy()
                    p = pt.copy()
                    cur_score = sc
                    accepted = True
                    break

        # Global random perturbation
        if not accepted:
            for sigma in [0.06 / n, 0.015 / n, 0.003 / n]:
                pt = project_simplex(p + rng.normal(0, sigma, n))
                sc = compute_score(pt)
                if not np.isfinite(sc):
                    continue
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_p = pt.copy()
                    stale = 0
                if sc < cur_score - 1e-12:
                    prev_p = p.copy()
                    p = pt.copy()
                    cur_score = sc
                    accepted = True
                    break

        # Periodic LP
        if not accepted and _HAVE_LP and n <= 500 and iter_no % 3000 == 0:
            g = lp_best_response(p, 20.0)
            if g is not None:
                for t in [0.5, 0.3, 0.15, 0.05]:
                    q = project_simplex((1 - t) * p + t * g)
                    sc = compute_score(q)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_p = q.copy()
                        p = q.copy()
                        prev_p = q.copy()
                        cur_score = sc
                        stale = 0
                        break

        if not accepted:
            stale += 1
            if stale > 600:
                p = project_simplex(0.55 * best_p + 0.45 * p + rng.normal(0, 0.15 / n, n))
                cur_score = compute_score(p)
                prev_p = p.copy()
                stale = 0
                sa_temp = max(best_score * 0.01, 0.003)
            elif stale > 250:
                p = project_simplex(0.8 * best_p + 0.2 * p + rng.normal(0, 0.04 / n, n))
                cur_score = compute_score(p)
                prev_p = p.copy()

        # Decay SA temperature
        sa_temp = max(sa_temp * 0.9997, 0.001)

        if iter_no % 4000 == 0:
            print("%s iter=%d best=%.10f cur=%.10f stale=%d" % (label, iter_no, best_score, cur_score, stale), flush=True)

        if best_score < 1.4990 and stale > 8000:
            break

    if not np.all(np.isfinite(best_p)):
        best_p = project_simplex(p)
    else:
        best_p = project_simplex(best_p)
    return best_p, float(best_score) if np.isfinite(best_score) else np.inf


def search_shapes_global(n, rng, budget_s):
    D = 12
    dim = D + 2
    end = time.monotonic() + max(0.0, budget_s)
    best_score = np.inf
    best_p = None
    best_theta = None

    def try_theta(th):
        nonlocal best_score, best_p, best_theta
        p = make_shape_cheb(n, th)
        sc = compute_score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_p = p
            best_theta = np.asarray(th, dtype=np.float64).copy()

    # Initial evaluations
    starts = [np.zeros(dim)]
    starts.append(rng.uniform(-1.5, 1.5, dim))
    starts.append(rng.uniform(-1.5, 1.5, dim))
    t = np.zeros(dim)
    t[D] = 2.5
    t[D + 1] = 2.5
    starts.append(t.copy())
    t = np.zeros(dim)
    t[0] = 1.5
    t[1] = -0.8
    starts.append(t.copy())
    t = np.zeros(dim)
    for i in range(D):
        t[i] = 0.7 * (1 if i % 2 == 0 else -1)
    t[D] = 1.0
    t[D + 1] = 1.0
    starts.append(t.copy())
    t = np.zeros(dim)
    t[D] = 3.5
    t[D + 1] = 3.5
    t[0] = -1.0
    starts.append(t.copy())

    for s in starts:
        try_theta(s)

    if _HAVE_MIN and time.monotonic() < end - 30:
        bounds = [(-4.0, 4.0)] * D + [(-4.0, 6.0), (-4.0, 6.0)]
        for x0 in starts:
            if time.monotonic() > end - 15:
                break
            try:
                res = minimize(lambda th: compute_score(make_shape_cheb(n, th)),
                               np.asarray(x0, dtype=np.float64),
                               method='Nelder-Mead',
                               options={'maxiter': 250, 'xatol': 1e-5, 'fatol': 1e-7})
                if res is not None and np.isfinite(res.fun):
                    try_theta(res.x)
            except Exception:
                pass

    return best_p, best_score, best_theta


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 12.0)

    # Get parent
    parent = None
    try:
        pa = np.asarray(height_sequence_1, dtype=np.float64)
        pa = np.where(np.isfinite(pa), pa, 0.0)
        pa = np.maximum(pa, 0.0)
        if pa.ndim == 1 and pa.size >= 32 and pa.sum() > 0:
            parent = pa
    except NameError:
        parent = None
    except Exception:
        parent = None

    best_p = None
    best_score = np.inf

    def try_record(p, sc):
        nonlocal best_p, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64)
        if p.size == 0:
            return
        p = normalize(p)
        if not np.isfinite(sc):
            sc = compute_score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_p = p.copy()
            print("NEW BEST: score=%.10f n=%d" % (best_score, p.size), flush=True)

    # Determine lengths to try
    lengths = [400, 512, 700, 900, 1200, 1600, 2048, 3000, 4096]
    if parent is not None:
        pn = parent.size
        local_lengths = [l for l in lengths if abs(l - pn) / pn < 0.3]
        if not local_lengths:
            local_lengths = [pn]
        lengths = sorted(set(local_lengths + [400, 700, 1200, 2048, 4096]))

    # Phase 1: Quick global search across lengths
    phase1_end = start_time + min(250.0, budget_s * 0.15)
    print("Phase 1: Global search across lengths, budget=%.0fs" % (phase1_end - start_time), flush=True)

    candidates = []
    for n in lengths:
        if time.monotonic() > phase1_end - 20:
            break
        t0 = time.monotonic()

        # Shape search
        sh_budget = min(80.0, (phase1_end - time.monotonic()) * 0.5)
        if sh_budget > 15:
            sh_p, sh_sc, _ = search_shapes_global(n, rng, sh_budget)
            if sh_p is not None:
                candidates.append((sh_sc, n, sh_p, "shape"))
                try_record(sh_p, sh_sc)

        # Initials
        inits = gen_initials(n, rng, parent if parent is not None and parent.size == n else None)
        scored = []
        for ip in inits:
            ip = normalize(ip)
            sc = compute_score(ip)
            if np.isfinite(sc):
                scored.append((sc, ip))
        scored.sort(key=lambda x: x[0])
        for sc, ip in scored[:5]:
            candidates.append((sc, n, ip, "init"))
            try_record(ip, sc)

        elapsed = time.monotonic() - t0
        print("  n=%d best_local=%.6f time=%.1fs" % (n, scored[0][0] if scored else 99, elapsed), flush=True)

    # Phase 2: Deep local optimization on best candidates
    candidates.sort(key=lambda x: x[0])
    phase2_end = deadline - min(300.0, budget_s * 0.12)
    print("Phase 2: Local optimization, budget=%.0fs, %d candidates" % (
        phase2_end - time.monotonic(), len(candidates)), flush=True)

    # Deduplicate: keep at most 2 per length
    seen_lengths = {}
    selected = []
    for sc, n, p, src in candidates:
        if n not in seen_lengths or sc < seen_lengths[n]:
            seen_lengths[n] = sc
            selected.append((sc, n, p, src))
    # Keep top 6 overall
    selected = selected[:8]

    n_jobs = max(1, len(selected))
    for i, (sc, n, p, src) in enumerate(selected):
        if time.monotonic() > phase2_end - 30:
            break
        remaining = phase2_end - time.monotonic()
        if remaining < 30:
            break
        # Allocate time: more to better candidates
        frac = max(0.3, 1.0 - 0.1 * i)
        limit = min(remaining * frac, 400.0)
        limit = max(30.0, limit)
        print("  Optimize candidate %d (n=%d, score=%.6f, src=%s) with %.0fs" % (
            i, n, sc, src, limit), flush=True)
        p_best, sc_best = optimize_local(p, rng, limit, "opt%d" % i)
        try_record(p_best, sc_best)

    # Phase 3: Final polish on the global best
    if best_p is not None and time.monotonic() < deadline - 20:
        limit = max(10.0, deadline - time.monotonic() - 5.0)
        print("Phase 3: Final polish with %.0fs, current best=%.10f" % (limit, best_score), flush=True)
        p_best, sc_best = optimize_local(best_p, rng, limit, "FINAL")
        try_record(p_best, sc_best)

    # Fallback
    if best_p is None:
        best_p = np.full(700, 1.0 / 700, dtype=np.float64)
        best_score = compute_score(best_p)

    # Ensure valid output
    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    s = float(best_p.sum())
    if s < 0.01:
        best_p = np.full(700, 1.0 / 700, dtype=np.float64)
    best_p = np.minimum(best_p, 1000.0)

    final_score = compute_score(best_p)
    print("DONE: n=%d score=%.10f time=%.1fs" % (best_p.size, final_score, time.monotonic() - start_time), flush=True)

    return [float(x) for x in best_p]
