def normalize_f(f):

def lse_gradient(f, T):

    C = np.convolve(f, f)

    TC_max = TC.max()
    exp_vals = np.exp(TC - TC_max)

    if total < 1e-300:
        w = np.ones(len(C)) / len(C)
    else:
        w = exp_vals / total
    # B[i] = (f*w)[i] for i=0..n-1

    conv_frev_w = np.convolve(f_rev, w)

    grad = A + B
    return grad, C, w

def _is_prime(n):
    if n < 2:
        return False
    for i in range(2, int(n ** 0.5) + 1):
        if n % i == 0:

    return True


def _next_prime(n):
    n = max(n, 2)
    while not _is_prime(n):

def block_perturb(f, rng, min_w=1, max_w=None):
    """Multiplicative block perturbation, renormalized to sum 1."""
    n = len(f)
    if max_w is None:
        max_w = max(2, n // 8)
    w = int(rng.integers(min_w, max(2, min(max_w, n))))
    w = max(1, min(w, n))
    start = int(rng.integers(0, max(1, n - w + 1)))
    stop = min(n, start + w)
    cand = f.copy()
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.3)))
    cand = normalize_f(cand)
    return cand


def build_conv_matrix(n, f):
    """Sparse matrix A (2n-1 x n) with A[k,j] = f[k-j] for valid indices."""
    i_idx = np.tile(np.arange(n), n)
    j_idx = np.repeat(np.arange(n), n)
    row_idx = j_idx + i_idx
    data = np.where((row_idx >= 0) & (row_idx < 2 * n - 1), f[i_idx], 0.0)
    valid = (row_idx >= 0) & (row_idx < 2 * n - 1)
    A = sp.coo_matrix((data[valid], (row_idx[valid], i_idx[valid])),
                      shape=(2 * n - 1, n)).tocsr()
    return A

        M = float(conv.max())
        if M < 1e-15:
            break
        b = np.full(2 * n - 1, M)
        try:
            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                          options={'time_limit': lp_time, 'presolve': True})
            if res.status != 0:
                stall += 1
                if stall > 5:
                    break
                continue
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs < 1e-10:
                break
            g = g / gs
            if np.linalg.norm(g - f) < 1e-10:
                stall += 1
                if stall > 10:
                    break
                continue
            f_new = (1.0 - t) * f + t * g
            f_new = normalize_f(f_new)
            if not np.all(np.isfinite(f_new)):
                break
            f = f_new
            t = max(t_min, t * 0.99)

            sc = fast_score(f)
            if sc < best_sc:
                best_sc = sc
                best_f = f.copy()
                stall = 0
            else:
                stall += 1
                if stall > 30:
                    # Perturb and rebuild
                    rng = np.random.default_rng(hash(int(time.monotonic() * 1000)) % (2**31))
                    f = block_perturb(f, rng, 1, max(5, n // 10))
                    A = build_conv_matrix(n, f)
                    stall = 0
        except Exception:
            break

    return best_f, best_sc


def local_search(f, budget_s, rng):
    """Local search via block perturbations."""
    f = normalize_f(f.copy().astype(np.float64))
    best_f = f.copy()
    best_sc = fast_score(f)
    n = len(f)
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < budget_s:
        it += 1
        w = int(rng.integers(1, max(2, min(n, 30))))
        start = int(rng.integers(0, max(1, n - w + 1)))
        stop = min(n, start + w)
        cand = f.copy()
        cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.15)))
        cand = normalize_f(cand)
        sc = fast_score(cand)
        if sc < best_sc - 1e-10:
            best_sc = sc
            best_f = cand.copy()
            f = cand
        elif it % 500 == 0 and time.monotonic() - t0 > budget_s * 0.5:
            # Occasionally try larger perturbations
            w2 = int(rng.integers(1, max(2, n // 20)))
            start2 = int(rng.integers(0, max(1, n - w2 + 1)))
            cand2 = f.copy()
            cand2[start2:min(n, start2 + w2)] *= float(np.exp(rng.normal(0.0, 0.5)))
            cand2 = normalize_f(cand2)
            sc2 = fast_score(cand2)
            if sc2 < best_sc:
                best_sc = sc2
                best_f = cand2.copy()
                f = cand2

    return best_f, best_sc


def coordinate_descent(f, budget_s, rng):
    """Simple coordinate descent: try adjusting individual or pairs of coordinates."""
    f = normalize_f(f.copy().astype(np.float64))
    best_f = f.copy()
    best_sc = fast_score(f)
    n = len(f)
    t0 = time.monotonic()

    # Precompute for speed
    while time.monotonic() - t0 < budget_s:
        # Try a random set of coordinates to perturb
        n_try = min(n, 50)
        idx = rng.integers(0, n, n_try)
        for i in idx:
            orig = f[i]
            # Try a few multipliers
            for mult in [0.5, 0.8, 1.2, 1.5, 2.0, 0.1, 3.0]:
                f[i] = orig * mult
                sc = fast_score(f)
                if sc < best_sc:
                    best_sc = sc
                    best_f = f.copy()
                else:
                    f[i] = orig
            if time.monotonic() - t0 > budget_s:
                break
        # Reset f to best
        f = best_f.copy()

    return best_f, best_sc
