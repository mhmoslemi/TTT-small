def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1)
    return np.maximum(v - theta, 0)

"""
LSE-Gradient optimization for minimizing the peak autocorrelation of
non-negative sequences. Uses log-sum-exp smoothing of the max constraint,
projected gradient descent with Nesterov momentum, and multiple restarts
with asymmetric initializations.
"""
import numpy as np
import time
import sys

def fast_score(f):
    """Return the evaluation score quickly."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))

def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n+1) > (cssv - s))[0]
    if len(rho_candidates) == 0:
        return np.full(n, s/n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lse_gradient(f, beta):
    """Compute the LSE-smoothed gradient of max(f*f) w.r.t. f."""
    n = len(f)
    conv = np.convolve(f, f)
    # LSE weights
    conv_shifted = conv - conv.max()
    exp_vals = np.exp(beta * conv_shifted)
    w = exp_vals / exp_vals.sum()
    # Gradient component for each f[j]: 2 * sum_{k=j}^{j+n-1} w[k] * f[k-j]
    # = 2 * (w cross-correlate f)[j]
    # Using FFT for efficiency
    N = 1
    while N < 3 * n:
        N *= 2

"""
Optimized search for non-negative sequences minimizing peak autocorrelation.
Uses tapered initializations (small boundary values), LSE-smoothed gradient
descent with Nesterov momentum, spectral shaping, and multi-start strategy.
Key insight: small boundary values reduce the "triangular" artifact of the
linear autocorrelation, allowing the peak to be lowered below the uniform case.

import numpy as np
import time

def fast_score(f):
    f = np.asarray(f, dtype=np.float64)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))

def project_nonneg_simplex(v, s=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - s)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, s / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_lse_grad(f, beta):
    """Compute gradient of LSE approximation to max(f*f)."""
    n = len(f)
    conv = np.convolve(f, f)
    cmax = conv.max()
    shifted = conv - cmax
    exp_vals = np.exp(beta * shifted)
    w = exp_vals / exp_vals.sum()
    # Gradient: grad[j] = 2 * sum_{k=j}^{min(j+n-1, 2n-2)} w[k] * f[k-j]
    # = 2 * (w cross-correlate f)[j] for j = 0..n-1
    N = 1
    while N < 3 * n:
        N <<= 1
    W = np.fft.rfft(w, N)
    F = np.fft.rfft(f, N)
    corr = np.fft.irfft(W * np.conj(F), N)
    grad = 2.0 * corr[:n]
    return grad, cmax

def lse_optimize(f_init, beta_seq, step_sizes, momentum, deadline, max_total_iters=100000):
    """Run LSE gradient descent with given parameter schedule."""
    n = len(f_init)
    f = f_init.copy()
    f = f / f.sum()
    best = f.copy()
    best_sc = fast_score(f)
    velocity = np.zeros(n)
    
    it = 0
    while it < max_total_iters and time.monotonic() < deadline:
        # Interpolate beta and step size based on progress
        progress = it / max_total_iters
        beta = np.interp(progress, [0, 1], beta_seq)
        step = np.interp(progress, [0, 1], step_sizes)
        
        grad, cmax = compute_lse_grad(f, beta)
        
        # Nesterov momentum
        velocity = momentum * velocity - step * grad
        f = f + velocity
        f = project_nonneg_simplex(f, 1.0)
        
        it += 1
        if it % 10 == 0:
            sc = fast_score(f)
            if sc < best_sc:
                best_sc = sc
                best = f.copy()
    
    sc = fast_score(f)
    if sc < best_sc:
        best = f.copy()
        best_sc = sc
    return best, best_sc

def make_tapered(n, rng, taper_type='sine'):
    x = np.arange(n, dtype=np.float64)
    if taper_type == 'sine':
        a = np.sin(np.pi * (x + 1) / (n + 1))
    elif taper_type == 'cosine_sq':
        a = np.cos(np.pi * (x - (n-1)/2) / n) ** 2
    elif taper_type == 'parabolic':
        t = 2.0 * x / (n - 1) - 1.0
        a = 1.0 - t * t
    elif taper_type == 'raised_cosine':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
    elif taper_type == 'gaussian_center':
        center = (n - 1) / 2.0
        a = np.exp(-0.5 * ((x - center) / (n / 4.0)) ** 2)
    else:
        a = np.ones(n)
    # Add small noise for breaking symmetry
    a = a + rng.normal(0, 0.01, n)
    a = np.maximum(a, 1e-6)
    return a / a.sum()

def make_asymmetric(n, rng):
    """Asymmetric initialization with small boundaries."""
    x = np.arange(n, dtype=np.float64)
    # Tapered envelope
    env = np.sin(np.pi * (x + 1) / (n + 1))
    # Asymmetric modulation

    freq = rng.uniform(0.5, 3.0)
    mod = 1.0 + 0.3 * np.sin(2 * np.pi * freq * x / n + phase)
    a = env * mod + rng.normal(0, 0.02, n)
    a = np.maximum(a, 1e-6)
    return a / a.sum()

    f = f / f.sum()
    N = 1
    while N < 2 * n:
        N <<= 1
    
    for _ in range(n_iters):
        F = np.fft.rfft(f, N)
        # Boost DC, suppress others (makes time-domain more "peaked" which we don't want)
        # Actually, we want to SUPPRESS DC relative to others to make time-domain flatter
        # But f >= 0 constraint makes this hard.
        # Alternative: shape |F| to be more uniform (reduce peak at DC)
        mag = np.abs(F)
        phase = np.angle(F)
