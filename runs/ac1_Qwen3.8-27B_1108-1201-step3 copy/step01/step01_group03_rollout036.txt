Objective: score = 2*n*max(conv(a,a)) / (sum(a))^2, scale-invariant.

def fast_score(f):
    """Compute score = 2*n*max(conv)/sum^2."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))


def compute_gradient(a, alpha):
    """Compute the (sub)gradient of softmax-smoothed max-convolution.
    Returns g of same shape as a, where the update a * exp(-eta*g) decreases the score."""
    n = len(a)
    conv = np.convolve(a, a)
    # Softmax over conv entries
    conv_shifted = conv - conv.max()  # for numerical stability
    p = np.exp(alpha * conv_shifted)
    p = p / p.sum()
    # Gradient: g[j] = 2 * (p cross-correlate a)[j]
    # np.correlate(p, a, 'full')[j] = sum_k p[k] * a[k-j] for valid k, k-j
    cross = np.correlate(p, a, mode='full')
    g = 2.0 * cross[:n]
    return g, conv, p


def mirror_step(a, g, eta):
    """Multiplicative update: a_new = a * exp(-eta*g) / Z."""
    log_a = np.log(np.maximum(a, 1e-30))
    log_new = log_a - eta * g
    # Numerical stability
    log_new = log_new - log_new.max()
    new_weights = np.exp(log_new)
    s = new_weights.sum()
    if s < 1e-30:
        return a.copy()
    return new_weights / s


def build_conv_matrix(n, f):
    """Sparse matrix A (2n-1 x n) with A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    rows = []
    cols = []
    data = []
    for j in range(n):
        for k in range(2 * n - 1):
            m = k - j
            if 0 <= m < n:
                rows.append(k)
                cols.append(j)
                data.append(f[m])
    A = sp.coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A


def build_conv_matrix_fast(n, f):
    """Faster construction using vectorized operations."""
    i_idx = np.repeat(np.arange(n), n)  # column index j
    j_idx = np.tile(np.arange(n), n)    # offset within f
    row_idx = i_idx + j_idx             # row = k = j + offset
    # Filter valid rows
    valid = row_idx < (2 * n - 1)
    row_idx = row_idx[valid]
    col_idx = i_idx[valid]
    data = f[j_idx[valid]]
    A = sp.coo_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n)).tocsr()
    return A


def lp_kick(a, time_limit=5.0):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Return normalized g."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(conv.max())
    if M < 1e-15:
        return None
    A = build_conv_matrix_fast(n, a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None

def random_perturb(a, rng, scale=0.3):
    """Multiplicative random perturbation, renormalized."""
    w = max(1, int(rng.integers(1, max(2, len(a) // 4))))
    start = int(rng.integers(0, max(1, len(a) - w + 1)))
    cand = a.copy()
    cand[start:start+w] *= np.exp(rng.normal(0, scale, w))
    cand = np.maximum(cand, 0)
    s = cand.sum()
    if s < 1e-10:
        return a.copy()
    return cand / s

    """Compute score = 2*n*max(conv)/sum^2 quickly."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))


def compute_soft_gradient(a, alpha):
    """Compute gradient of softmax-smoothed max-convolution.
    Returns (g, conv) where g[j] is the subgradient component."""
    n = len(a)
    conv = np.convolve(a, a)
    conv_shifted = conv - conv.max()
    p = np.exp(alpha * conv_shifted)
    p_sum = p.sum()
    if p_sum < 1e-30:
        p = np.ones(len(conv)) / len(conv)
    else:
        p = p / p_sum
    cross = np.correlate(p, a, mode='full')
    g = 2.0 * cross[:n]
    return g, conv


def mirror_update(a, g, eta):
    """Multiplicative update: a_new = a * exp(-eta*g) / Z."""
    log_a = np.log(np.maximum(a, 1e-30))
    log_new = log_a - eta * g
    log_new = log_new - log_new.max()
    new_w = np.exp(log_new)
    s = new_w.sum()
    if s < 1e-30:
        return a.copy()
    return new_w / s


def build_conv_matrix_fast(n, f):
    """Sparse convolution matrix A (2n-1 x n), A[k,j] = f[k-j] if valid."""
    cols = np.repeat(np.arange(n), n)
    offsets = np.tile(np.arange(n), n)
    rows = cols + offsets
    valid = rows < (2 * n - 1)
    A = sp.coo_matrix((f[offsets[valid]], (rows[valid], cols[valid])),
                      shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp(a, time_limit=5.0):
    """Solve max sum(g) s.t. conv(a,g) <= M, g>=0. Return normalized g or None."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(conv.max())
    if M < 1e-15:
        return None
    A = build_conv_matrix_fast(n, a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def make_initial(n, rng, method='uniform'):
    """Create normalized non-negative initial sequence of length n."""
    if method == 'uniform':
        a = np.ones(n)
    elif method == 'gaussian':
        x = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        sigma = max(1.0, n / 5.0)
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    elif method == 'random':
        a = rng.random(n) + 0.1
    elif method == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1.0 + np.cos(x)) + 0.01
    elif method == 'bimodal':
        x = np.arange(n, dtype=np.float64)
        s = max(1, n // 10)
        a = (np.exp(-0.5 * ((x - n * 0.25) / s) ** 2) +
             np.exp(-0.5 * ((x - n * 0.75) / s) ** 2) + 0.01)
    elif method == 'triangular':
        x = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        a = np.maximum(1.0 - np.abs(x - center) / max(1, center), 0.01)
    elif method == 'exponential':
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-x / (n / 3.0)) + 0.01
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.001)
    a = a / a.sum()
    return a


def random_perturb(a, rng, scale=0.3):
    """Multiplicative random block perturbation, renormalized to sum 1."""
    n = len(a)
    w = max(1, int(rng.integers(1, max(2, n // 3))))
    start = int(rng.integers(0, max(1, n - w + 1)))
    cand = a.copy()
    cand[start:start + w] *= np.exp(rng.normal(0, scale, w))
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s < 1e-10:
        return a.copy()
    return cand / s


def hybrid_optimize(a, budget_s, use_lp=True, lp_every=100, lp_time=3.0,
                    alpha_start=5.0, alpha_end=30.0, seed=42):
    """Combined LP + gradient mirror descent optimizer.
    Returns (best_a, best_score)."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()

    rng = np.random.default_rng(seed)
    eta = n * 0.3
    alpha = alpha_start
    best_a = a.copy()
    best_sc = fast_score(a)
    stall = 0
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < budget_s:
        it += 1
        remaining = budget_s - (time.monotonic() - t0)

        # LP kick
        if use_lp and it % lp_every == 1 and remaining > lp_time + 2:
            g_lp = solve_lp(a, lp_time)
            if g_lp is not None:
                # Try several blend factors
                best_blend_sc = fast_score(a)
                best_blend_a = a.copy()
                for t_blend in [0.2, 0.4, 0.6, 0.8]:
                    cand = (1.0 - t_blend) * a + t_blend * g_lp
                    cand = np.maximum(cand, 0.0)
                    s = cand.sum()
                    if s > 1e-10:
                        cand = cand / s
                        sc = fast_score(cand)
                        if sc < best_blend_sc:
                            best_blend_sc = sc
                            best_blend_a = cand
                a = best_blend_a
                sc = fast_score(a)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                    stall = 0
                else:
                    stall += 10

        # Multiple gradient steps between LP calls
        n_grad = max(1, min(50, lp_every // 10))
        for _ in range(n_grad):
            if time.monotonic() - t0 > budget_s:
                break
            g, conv = compute_soft_gradient(a, alpha)
            a_new = mirror_update(a, g, eta)
            sc = fast_score(a_new)
            sc_old = fast_score(a)

            if np.isfinite(sc):
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_new.copy()
                    stall = 0
                    a = a_new
                    eta = min(eta * 1.03, n * 3.0)
                elif sc < sc_old:
                    a = a_new
                    eta = min(eta * 1.02, n * 3.0)
                    stall += 1
                else:
                    eta = max(eta * 0.8, n * 0.05)
                    stall += 1

        # Adapt alpha over time
        progress = min(1.0, it / max(1, int(budget_s / 0.1)))
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        # Random perturbation when stuck
        if stall > 300 and it % 20 == 0:
            a = random_perturb(a, rng, scale=0.4)
            stall = 0
        elif stall > 100 and it % 50 == 0:
            a = random_perturb(a, rng, scale=0.15)
            stall = 0

    return best_a, best_sc


def fine_polish(a, budget_s, seed=42):
    """Fine-grained gradient polish with very small steps."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = fast_score(a)
    eta = n * 0.05
    alpha = 20.0
    rng = np.random.default_rng(seed)
    stall = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < budget_s:
        g, conv = compute_soft_gradient(a, alpha)
        a_new = mirror_update(a, g, eta)
        sc = fast_score(a_new)

        if np.isfinite(sc) and sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.01, n * 0.5)
        else:
            sc_old = fast_score(a)
            if np.isfinite(sc) and sc < sc_old:
                a = a_new
            eta = max(eta * 0.97, n * 0.005)
            stall += 1

        if stall > 500:
            a = random_perturb(a, rng, scale=0.05)
            stall = 0

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """Search for the best non-negative sequence minimizing the score.
    Uses hybrid LP + gradient mirror descent with multi-start."""
    total_budget = max(0.0, float(budget_s) - 15.0)
    deadline = time.monotonic() + total_budget
    rng = np.random.default_rng(seed)

    # Get parent sequence
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        parent_raw = np.ones(256)
    if parent_raw.ndim != 1 or parent_raw.size == 0:
        parent_raw = np.ones(256)
    parent_raw = np.clip(parent_raw, 0.0, 1000.0)
    if parent_raw.sum() < 0.01:
        parent_raw = np.ones(256)

    best = parent_raw.copy()
    bs = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(bs):
        bs = fast_score(best)
    print("Initial: score=%.6f n=%d" % (bs, best.size), flush=True)

    # Define search tasks: (name, sequence, budget_fraction)
    # First optimize the parent, then try others
    tasks = []

    # Parent optimization (highest priority)
    tasks.append(('parent', parent_raw.copy(), 0.18))

    # Various inits at parent length
    n_p = len(parent_raw)
    inits_512 = ['uniform', 'gaussian', 'bimodal', 'random', 'cosine', 'exponential']
    for m in inits_512:
        tasks.append((('init_512_%s' % m), make_initial(512, rng, m), 0.08))

    # Longer sequences
    for n_long, meth in [(1024, 'uniform'), (1024, 'gaussian'),
                         (2048, 'uniform'), (2048, 'gaussian')]:
        tasks.append((('init_%d_%s' % (n_long, meth)),
                      make_initial(n_long, rng, meth), 0.08))

    # Shorter
    tasks.append(('init_256_gauss', make_initial(256, rng, 'gaussian'), 0.05))

    # Normalize budgets to fit total
    total_frac = sum(t[2] for t in tasks)
    polish_frac = 0.15
    scale = (1.0 - polish_frac) / total_frac

    for i, (name, a, frac) in enumerate(tasks):
        if time.monotonic() > deadline:
            break
        b = frac * scale * total_budget
        b = min(b, deadline - time.monotonic())
        if b < 3.0:
            break
        print("[%d/%d] %s n=%d budget=%.0fs" % (i + 1, len(tasks), name, len(a), b),
              flush=True)

        # Run hybrid optimizer
        a_opt, sc_opt = hybrid_optimize(a, b, use_lp=True,
                                        lp_every=80, lp_time=min(3.0, b * 0.05),
                                        alpha_start=5.0, alpha_end=40.0,
                                        seed=seed + i * 137)
        if np.isfinite(sc_opt):
            sc_verify = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc_verify):
                sc_opt = min(sc_opt, sc_verify)
            if sc_opt < bs:
                bs = sc_opt
                best = np.maximum(a_opt, 0.0).copy()
                print("  BEST: %.6f (n=%d)" % (bs, best.size), flush=True)
        else:
            print("  (no valid result)", flush=True)

    # Final polish phase
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print("Polish phase: %.0fs" % remaining, flush=True)
        a_pol = best.copy()
        a_pol = np.maximum(a_pol, 1e-10)
        a_pol = a_pol / a_pol.sum()
        a_pol, sc_pol = fine_polish(a_pol, remaining * 0.6, seed=seed + 77777)
        if np.isfinite(sc_pol) and sc_pol < bs:
            sc_v = float(evaluate_sequence(a_pol.tolist()))
            if np.isfinite(sc_v) and sc_v < bs:
                bs = sc_v
                best = np.maximum(a_pol, 0.0).copy()
            elif np.isfinite(sc_pol) and sc_pol < bs:
                bs = sc_pol
                best = np.maximum(a_pol, 0.0).copy()
            print("  After polish: %.6f" % bs, flush=True)

        # One more round of LP on the polished best
        remaining2 = deadline - time.monotonic()
        if remaining2 > 20 and len(best) <= 2048:
            print("LP polish: %.0fs" % remaining2, flush=True)
            a_lp = best.copy()
            a_lp = np.maximum(a_lp, 1e-10)
            a_lp = a_lp / a_lp.sum()
            a_lp, sc_lp = hybrid_optimize(a_lp, remaining2 * 0.7, use_lp=True,
                                          lp_every=30, lp_time=3.0,
                                          alpha_start=15.0, alpha_end=50.0,
                                          seed=seed + 88888)
            if np.isfinite(sc_lp):
                sc_v = float(evaluate_sequence(a_lp.tolist()))
                if np.isfinite(sc_v) and sc_v < bs:
                    bs = sc_v
                    best = np.maximum(a_lp, 0.0).copy()
                    print("  After LP polish: %.6f" % bs, flush=True)

    # Validate and return
    best = np.clip(np.asarray(best, dtype=np.float64), 0.0, 1000.0)
    if best.size == 0 or not np.isfinite(best).all() or best.sum() < 0.01:
        best = np.ones(256)

    final_sc = float(evaluate_sequence(best.tolist()))
    print("FINAL: score=%.6f n=%d" % (final_sc, best.size), flush=True)
    return [float(x) for x in best]
