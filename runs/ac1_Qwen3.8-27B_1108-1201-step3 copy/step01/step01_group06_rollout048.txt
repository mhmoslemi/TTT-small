    w = c[top_idx] ** weight_power
    total = w.sum()
    if total < 1e-30:
        return np.zeros(n, dtype=np.float64)

        grad = compute_dense_subgradient(a, c, n_active, weight_power)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad_normed = grad / gnorm
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_normed
        v = beta2 * v + (1.0 - beta2) * (grad_normed ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        # Learning rate with exponential decay
        lr = base_lr * (0.9995 ** t)
        
        # Take step
        a_candidate = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to simplex
        a_candidate = project_simplex(a_candidate, S)
        
        # Backtracking: check if we improved
        c_new = convolve_self(a_candidate)
        new_score = c_new.max()
        
        # Allow slight increase for exploration
        if new_score <= cur + 0.005 * best_score or t < 100:
            a = a_candidate
        else:
            # Halve the step
            a_candidate = a - 0.5 * lr * m_hat / (np.sqrt(v_hat) + eps)
            a_candidate = project_simplex(a_candidate, S)
            a = a_candidate
        
        # Periodic perturbation to escape local minima
        if t % 3000 == 0:
            sigma = 0.001 * S / np.sqrt(n)
            a = a + rng.normal(0.0, sigma, n)
            a = project_simplex(a, S)
    
    return best_a, best_score


def coordinate_descent_polish(a_init, budget_s, rng):
    """
    Coordinate descent: for each position, find the optimal value
    that minimizes the max convolution, given all other positions fixed.

    # Precompute the "background" convolution (without each element)
    # c[k] = a[m]*a[k-m] + rest for position m contributing to lag k

def optimize_phase2(a_init, budget_s, rng):
    """
    Phase 2: Fine-tuning with very small steps and dense subgradient.
    Uses a combination of Adam and direct gradient descent.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(a_init.astype(np.float64), 0.0, 1000.0)
    a = a * (S / a.sum())
    
    deadline = time.monotonic() + budget_s
    c = convolve_self(a)
    best_score = c.max()
    best_a = a.copy()
    
    t = 0
    # Very small learning rate for fine-tuning
    lr = 0.05 * S / n
    momentum = np.zeros(n, dtype=np.float64)
    beta = 0.95
    
    while time.monotonic() < deadline:
        t += 1
        c = convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
        
        # Use all lags above 95th percentile
        threshold = np.percentile(c, 95)
        active_mask = c >= threshold
        n_active = int(active_mask.sum())
        
        if n_active < 5:
            active_idx = np.argsort(c)[::-1][:10]
        else:
            active_idx = np.where(active_mask)[0]
        
        # Weighted subgradient
        w = c[active_idx] ** 2
        total = w.sum()
        if total < 1e-30:
            break
        w = w / total
        
        grad = np.zeros(n, dtype=np.float64)
        for wv, k in zip(w, active_idx):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi + 1] += wv * 2.0 * a[k - hi:k - lo + 1][::-1]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm
        
        # Momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        
        a_new = a - lr * momentum
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = np.full(n, S / n)
        else:
            a_new = a_new * (S / sn)
        
        # Accept if not too much worse
        c_new = convolve_self(a_new)
        if c_new.max() <= cur + 0.002 * best_score:
            a = a_new
        
        # Slowly decrease learning rate
        lr *= 0.9999
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(self_conv)/sum^2.
    
    Strategy:
    1. Initialize from given sequence (or random fallback).
    2. Phase 1: Adam with dense subgradient (top-50, quadratic weight) and backtracking.
    3. Phase 2: Fine-grained gradient with momentum and 95th-percentile active set.
    4. Phase 3: Very fine Adam polish.
    5. Return best verified sequence.
    """
    rng = np.random.default_rng(seed)
    overall_deadline = time.monotonic() + max(1.0, budget_s - 10.0)
    
    # --- Get initial sequences ---
    inits = []
    
    # Primary: given sequence
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).flatten()
        given = np.clip(given, 0.0, 1000.0)
        if given.size >= 32 and given.sum() > 0.01:
            inits.append(("given", given))
    except NameError:
        pass
    except Exception:
        pass
    
    if not inits:
        # Fallback initializations
        n = 8192
        tt = np.linspace(0.0, 1.0, n)
        inits.append(("smooth", np.maximum(
            1.0 + 0.3 * np.sin(2 * np.pi * 3.0 * tt) + 0.15 * np.cos(2 * np.pi * 7.0 * tt), 0.0)))
        inits.append(("rand", rng.random(n) + 0.1))
    
    # Create a perturbed variant of the best init
    if inits:
        base = inits[0][1].copy()
        n_b = len(base)
        pert = base + rng.normal(0.0, 0.01 * base.mean(), n_b)
        pert = np.maximum(pert, 0.0)
        inits.append(("perturbed", pert))
    
    # Create a "smoothed" variant
    if inits:
        base = inits[0][1].copy()
        # Light Gaussian smoothing
        kernel_size = max(3, len(base) // 1024)
        if kernel_size % 2 == 0:
            kernel_size += 1
        kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size // 2) ** 2 / (kernel_size // 4.0 + 1))
        kernel /= kernel.sum()
        from scipy.ndimage import uniform_filter1d
        smoothed = uniform_filter1d(base, size=min(kernel_size, len(base) // 10))
        smoothed = np.maximum(smoothed, 0.0)
        if smoothed.sum() > 0.01:
            inits.append(("smoothed", smoothed))
    
    # --- Optimize each initialization ---
    global_best = np.inf
    global_a = None
    
    time_left = overall_deadline - time.monotonic()
    n_inits = len(inits)
    # Give more time to the primary (given) initialization
    time_alloc = []
    for i, (name, _) in enumerate(inits):
        if i == 0:
            time_alloc.append(0.5 * time_left)
        else:
            time_alloc.append(0.5 * time_left / (n_inits - 1) if n_inits > 1 else 0.0)
    
    for i, (name, init) in enumerate(inits):
        if time.monotonic() > overall_deadline - 30.0:
            break
        
        budget = time_alloc[i]
        if budget < 5.0:
            budget = max(1.0, min(budget, overall_deadline - time.monotonic() - 5.0))
        
        print(f"[{name}] Starting optimization (n={len(init)}, budget={budget:.1f}s)")
        
        # Phase 1: Main optimization
        phase1_budget = budget * 0.6
        a_opt, score1 = optimize_adam(init, phase1_budget, rng,
                                       lr_scale=0.3, top_k_base=50, weight_power=2.0)
        
        # Phase 2: Fine-tuning
        phase2_budget = budget * 0.25
        if time.monotonic() < overall_deadline - 10.0:
            a_opt2, score2 = optimize_phase2(a_opt, phase2_budget, rng)
            if score2 < score1:
                a_opt = a_opt2
                score1 = score2
        
        # Phase 3: Final polish with very small steps
        phase3_budget = min(budget * 0.15, overall_deadline - time.monotonic() - 5.0)
        if phase3_budget > 1.0:
            a_opt3, score3 = optimize_adam(a_opt, phase3_budget, rng,
                                           lr_scale=0.1, top_k_base=30, weight_power=3.0)
            if score3 < score1:
                a_opt = a_opt3
                score1 = score3
        
        # Verify
        ver = evaluate_sequence(a_opt.tolist())
        print(f"[{name}] Phase scores: {score1:.7f}, verified: {ver:.7f}")
        
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_opt.copy()
            print(f"[{name}] NEW GLOBAL BEST: {ver:.7f}")
    
    # --- Final polish on the global best ---
    remaining = overall_deadline - time.monotonic()
    if remaining > 10.0 and global_a is not None:
        print(f"\nFinal polish: {remaining:.1f}s remaining")
        
        # Use the best lr and settings
        a_final, score_final = optimize_adam(global_a, remaining * 0.5, rng,
                                             lr_scale=0.1, top_k_base=80, weight_power=2.0)
        if time.monotonic() < overall_deadline - 3.0:
            a_final2, score_final2 = optimize_phase2(a_final, remaining * 0.3, rng)
            if score_final2 < score_final:
                a_final = a_final2
                score_final = score_final2
        
        ver_final = evaluate_sequence(a_final.tolist())
        print(f"Final polish score: {score_final:.7f}, verified: {ver_final:.7f}")
        
        if np.isfinite(ver_final) and ver_final < global_best:
            global_best = ver_final
            global_a = a_final.copy()
            print(f"FINAL NEW BEST: {ver_final:.7f}")
    
    # --- Fallback ---
    if global_a is None:
        print("WARNING: No valid sequence found, using fallback")
        global_a = np.ones(256, dtype=np.float64)
    
    # Final safety checks
    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)
    
    # Final verification
    final_score = evaluate_sequence([float(x) for x in global_a])
    print(f"\n{'='*50}")
    print(f"FINAL ANSWER: score={final_score:.7f}, length={len(global_a)}")
    print(f"Sum={global_a.sum():.4f}, Max={global_a.max():.4f}")
    print(f"{'='*50}")
    
    return [float(x) for x in global_a]
