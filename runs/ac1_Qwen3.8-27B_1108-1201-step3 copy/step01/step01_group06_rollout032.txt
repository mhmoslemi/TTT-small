```python
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize

def self_convolution(a):
    """Full self-convolution of a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def grad_conv_with_sm(a, sm):
    """Compute g[j] = 2 * sum_m sm[j+m] * a[m] for j=0..n-1."""
    n = len(a)
    size = 1 << (4 * n - 3).bit_length()
    e = np.zeros(size)
    e[:n] = a
    f = np.zeros(size)
    f[:2*n-1] = sm[::-1]
    result = np.real(ifft(fft(e, size) * fft(f, size)))
    return 2.0 * result[2*n-2 : n-2 : -1].copy()

def compute_smooth_obj_grad(u, beta, n, S, fsize_conv, fsize_grad):
    """Compute smooth objective and gradient w.r.t. u."""
    exp_u = np.exp(u - np.max(u))
    a = S * exp_u / np.sum(exp_u)
    
    # Self-convolution
    A = fft(a, fsize_conv)
    c = np.real(ifft(A * A))[:2*n-1]
    
    # Smooth objective
    beta_c = beta * c
    max_bc = np.max(beta_c)
    exp_bc = np.exp(beta_c - max_bc)
    sum_exp = np.sum(exp_bc)
    obj = (max_bc + np.log(sum_exp)) / beta
    sm = exp_bc / sum_exp
    
    # Gradient w.r.t. a
    g = grad_conv_with_sm(a, sm, fsize_grad)
    
    # Gradient w.r.t. u
    ga = np.dot(g, a)
    grad_u = a * (g - ga / S)
    
    return obj, grad_u
```

```python
1. Parameterize a = S * softmax(u) where S = sqrt(2n) (fixes the sum).
2. Use log-sum-exp smoothing of the max function for a smooth objective.
3. Optimize u with L-BFGS using a continuation method (increasing β).
4. Multiple initializations for diversity.
5. Polish the best result with very large β.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def self_conv(a):
    """Full self-convolution of a via FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def grad_via_fft(a, sm, fsize):
    """Compute 2 * sum_m sm[j+m] * a[m] for j=0..n-1 using FFT.
    sm has length 2n-1, a has length n.
    """
    n = len(a)
    e = np.zeros(fsize)
    e[:n] = a
    f = np.zeros(fsize)
    f[:2*n-1] = sm[::-1]
    result = np.real(ifft(fft(e, fsize) * fft(f, fsize)))
    # h[j] = result[2n-2-j]
    indices = 2*n-2 - np.arange(n)
    return 2.0 * result[indices]


def smooth_obj_grad(u, beta, n, S, fsize_conv, fsize_grad):
    """Compute log-sum-exp smooth objective and its gradient w.r.t. u."""
    # a = S * softmax(u)
    shifted = u - np.max(u)
    exp_u = np.exp(shifted)
    Z = np.sum(exp_u)
    a = S * exp_u / Z
    
    # Self-convolution
    A = fft(a, fsize_conv)
    c = np.real(ifft(A * A))[:2*n-1]
    
    # Smooth max: (1/β) log sum exp(β c)
    bc = beta * c
    bc_max = np.max(bc)
    exp_bc = np.exp(bc - bc_max)
    Z_bc = np.sum(exp_bc)
    obj = (bc_max + np.log(Z_bc)) / beta
    sm = exp_bc / Z_bc
    
    # Gradient w.r.t. a: g = 2 * corr(sm, a)
    g = grad_via_fft(a, sm, fsize_grad)
    
    # Gradient w.r.t. u: grad_u = a * (g - (g·a)/S)
    ga = np.dot(g, a)
    grad_u = a * (g - ga / S)
    
    return obj, grad_u
```
