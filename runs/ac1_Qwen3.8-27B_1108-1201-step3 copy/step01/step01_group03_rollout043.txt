```python
"""
Hybrid optimization for minimum autocorrelation peak step functions.
Uses smoothed gradient descent (FFT-accelerated) combined with LP-based
direction finding, followed by targeted local search.
"""
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time


def fast_score(f):
    """Fast score computation matching evaluate_sequence."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    N = 1
    while N < 2 * len(f) - 1:
        N *= 2
    conv = np.real(np.fft.ifft(np.fft.fft(f, N) ** 2))[:2 * len(f) - 1]
    return float(2.0 * len(f) * conv.max() / (s * s))


def compute_conv(f):
    """Compute full linear convolution f*f using FFT."""
    n = len(f)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    conv = np.real(np.fft.ifft(np.fft.fft(f, N) ** 2))[:2 * n - 1]
    return conv


def compute_gradient(conv, f, T):
    """Compute gradient of T*soft_max(conv) with respect to f.
    Returns gradient vector of length n."""
    n = len(f)
    log_weights = conv - conv.max()  # for numerical stability
    exp_w = np.exp(T * log_weights)
    W = exp_w.sum()
    if W < 1e-300:
        return np.zeros(n)
    weights = exp_w / W  # These are the softmax weights
    
    # grad[i] = 2 * sum_m f[m] * weights[i+m] for valid indices
    # This is cross-correlation of f and weights
    N = 1
    while N < 2 * n - 1:
        N *= 2
    F = np.fft.fft(f, N)
    Wf = np.fft.fft(weights, N)
    corr = np.real(np.fft.ifft(F * np.conj(Wf)))[:n]
    return 2.0 * corr


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    # Water-filling algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w / w.sum()


def smoothed_gradient_descent(f, time_budget, T_start=5.0, T_end=200.0,
                               lr_start=0.05, lr_end=0.005, target_iters=5000):
    """Run smoothed gradient descent for the given time budget."""
    f = project_simplex(f)
    n = len(f)
    best_f = f.copy()
    best_score = fast_score(f)
    start_time = time.monotonic()
    
    it = 0
    for it in range(target_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_budget:
            break
        progress = min(1.0, elapsed / max(time_budget, 1e-6))
        T = T_start + (T_end - T_start) * progress
        lr = lr_start + (lr_end - lr_start) * progress
        
        conv = compute_conv(f)
        M = conv.max()
        s = f.sum()
        score = 2.0 * n * M / (s * s)
        
        if score < best_score:
            best_score = score
            best_f = f.copy()
        
        grad = compute_gradient(conv, f, T)
        f_new = f - lr * grad
        f_new = project_simplex(f_new)
        
        # Check for progress
        new_score = fast_score(f_new)
        if new_score < score:
            f = f_new
        else:
            # Halve step size
            lr *= 0.5
            f_new = f - lr * grad
            f_new = project_simplex(f_new)
            f = f_new
    
    return best_f, best_score


def build_conv_matrix_sparse(n, f):
    """Build sparse matrix A (2n-1 x n) where A[k,j] = f[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            data.append(f[k - j])
    A = sp.coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A


def lp_min_peak(f, time_limit=30.0):
    """Solve LP: min M' s.t. (f*g)[k] <= M', sum(g)=1, g>=0.
    Returns optimal g and M'."""
    n = len(f)
    # Build constraint matrix
    A_conv = build_conv_matrix_sparse(n, f)
    
    # Full A_ub: (2n-1) x (n+1), last column is -1 for M'
    A_ub = sp.hstack([A_conv, sp.csc_matrix(-np.ones(2 * n - 1))]).tocsr()
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Objective: minimize M'
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    bounds = [(0.0, None)] * n + [(0.0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0.0)
            M_prime = max(0.0, res.x[n])
            gs = g.sum()
            if gs > 1e-10:
                g = g / gs
            return g, M_prime
    except Exception:
        pass
    return None, None


def lp_refine(f, time_budget, n_lp_steps=50, lp_time=15.0):
    """Iteratively apply LP-based refinement."""
    f = f.copy()
    s = f.sum()
    f = f / s  # normalize
    best_f = f.copy()
    best_score = fast_score(f)
    start_time = time.monotonic()
    
    for step in range(n_lp_steps):
        if time.monotonic() - start_time > time_budget:
            break
        
        g, M_prime = lp_min_peak(f, time_limit=lp_time)
        if g is None:
            break
        
        conv_f = compute_conv(f)
        M_f = conv_f.max()
        
        if M_prime >= M_f:
            # LP couldn't find improvement, break
            # Try with a smaller tolerance
            break
        
        # Interpolate
        t = 0.5
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()
        
        sc = fast_score(f_new)
        if sc < best_score:
            best_score = sc
            best_f = f_new.copy()
        f = f_new
    
    return best_f, best_score
```

```python
"""
Hybrid optimizer for minimum autocorrelation peak step functions.
Combines FFT-accelerated smoothed gradient descent with LP-based direction
finding and adaptive local search to minimize 2n*max(f*f)/(sum f)^2.
"""
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time


def fast_score(f):
    """Fast O(n log n) score computation using FFT."""
    f = np.asarray(f, dtype=np.float64)
    n = len(f)
    if n == 0:
        return np.inf
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    F = np.fft.fft(f, N)
    conv = np.real(np.fft.ifft(F * F))[:2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))


def compute_conv(f):
    """Compute full linear convolution f*f using FFT. O(n log n)."""
    n = len(f)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    F = np.fft.fft(f, N)
    return np.real(np.fft.ifft(F * F))[:2 * n - 1]


def compute_grad(conv, f, T):
    """Compute gradient of (1/T)*log(sum exp(T*conv)) w.r.t. f. O(n log n)."""
    n = len(f)
    # Numerically stable softmax
    shifted = conv - conv.max()
    exp_w = np.exp(T * shifted)
    W = exp_w.sum()
    if W < 1e-300:
        return np.zeros(n)
    weights = exp_w / W  # softmax(T * conv)

    # grad[i] = 2 * sum_m f[m] * weights[i+m]  (cross-correlation)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    F = np.fft.fft(f, N)
    Wf = np.fft.fft(weights, N)
    corr = np.real(np.fft.ifft(F * np.conj(Wf)))[:n]
    return 2.0 * corr


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_candidates = np.where(cond)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return w / s


def smooth_grad_descent(f, time_budget, T_start=3.0, T_end=150.0,
                        lr_init=0.02, lr_min=0.001, momentum=0.7):
    """Smoothed gradient descent with annealing and momentum."""
    f = project_simplex(f.copy())
    n = len(f)
    best_f = f.copy()
    best_score = fast_score(f)
    start_time = time.monotonic()
    lr = lr_init
    vel = np.zeros(n)
    stall = 0
    max_stall = 200

    while time.monotonic() - start_time < time_budget:
        elapsed = time.monotonic() - start_time
        progress = min(1.0, elapsed / max(time_budget, 1e-6))
        # Anneal T: smooth start, sharp finish
        T = T_start + (T_end - T_start) * progress ** 0.7
        # Decay learning rate
        lr_eff = max(lr_min, lr_init * (1.0 - 0.8 * progress))

        conv = compute_conv(f)
        score = 2.0 * n * conv.max() / (f.sum() ** 2)

        if score < best_score - 1e-12:
            best_score = score
            best_f = f.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            # Restart from perturbed best
            rng_restart = np.random.default_rng(int(time.monotonic() * 1e6) % 2**31)
            f = best_f + rng_restart.normal(0, 0.02, n)
            f = project_simplex(f)
            vel = np.zeros(n)
            lr = lr_init
            stall = 0
            continue

        grad = compute_grad(conv, f, T)
        # Momentum update
        vel = momentum * vel - lr_eff * grad
        f_new = f + vel
        f_new = project_simplex(f_new)

        # Adaptive: check if improvement
        new_score = fast_score(f_new)
        if new_score < score:
            f = f_new
            lr = min(lr * 1.05, lr_init)
        else:
            # Try smaller step
            lr = lr * 0.5
            vel = momentum * vel - lr * grad
            f_new = f + vel
            f_new = project_simplex(f_new)
            if fast_score(f_new) < score:
                f = f_new

    return best_f, best_score

def lp_min_peak_dir(f, time_limit=20.0):
    """LP: min M' s.t. (f*g)[k] <= M' for all k, sum(g)=1, g>=0.
    Returns (g, M_prime) or (None, None)."""
    n = len(f)
    A_conv = build_sp_conv_matrix(n, f)

    # A_ub is (2n-1) x (n+1): last column is -1 for M'
    neg_ones = -np.ones(2 * n - 1)
    A_ub = sp.hstack([A_conv, sp.csc_matrix(neg_ones, shape=(2 * n - 1, 1))]).tocsr()
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # Objective: minimize M'
    c = np.zeros(n + 1)
    c[n] = 1.0

    bounds = [(0.0, None)] * n + [(0.0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0.0)
            M_prime = max(0.0, float(res.x[n]))
            gs = g.sum()
            if gs > 1e-10:
                g = g / gs
            else:
                return None, None
            return g, M_prime
    except Exception:
        pass
    return None, None


def lp_refine(f, time_budget, max_steps=100):
    """Iterative LP-based refinement with adaptive step size."""
    f = f.copy()
    s = f.sum()
    if s < 1e-10:
        return f, fast_score(f)
    f = f / s
    best_f = f.copy()
    best_score = fast_score(f)
    start_time = time.monotonic()
    t = 0.5  # interpolation parameter

    for step in range(max_steps):
        if time.monotonic() - start_time > time_budget:
            break

        g, M_prime = lp_min_peak_dir(f, time_limit=15.0)
        if g is None:
            break

        conv_f = compute_conv(f)
        M_f = conv_f.max()

        if M_prime >= M_f * 0.999:
            # LP can't improve much
            t *= 0.7
            if t < 0.05:
                break
            # Try with current t anyway
            f_new = (1.0 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            sc = fast_score(f_new)
            if sc < best_score:
                best_score = sc
                best_f = f_new.copy()
                f = f_new
            else:
                # Try smaller t
                t *= 0.5
            continue

        # Try multiple step sizes and pick the best
        best_local = f
        best_local_sc = best_score
        for trial_t in [t, t * 0.5, t * 1.5]:
            trial_t = np.clip(trial_t, 0.05, 0.95)
            f_new = (1.0 - trial_t) * f + trial_t * g
            f_new = np.maximum(f_new, 0.0)
            fs = f_new.sum()
            if fs < 1e-10:
                continue
            f_new = f_new / fs
            sc = fast_score(f_new)
            if sc < best_local_sc:
                best_local_sc = sc
                best_local = f_new
                best_trial_t = trial_t

        if best_local_sc < best_score:
            best_score = best_local_sc
            best_f = best_local.copy()
            f = best_local
            t = max(0.1, best_trial_t * 1.1)
        else:
            t *= 0.6
            f = best_local  # still move to the best trial
            if t < 0.03:
                break

    return best_f, best_score


def make_initial(n, rng, method='random'):
    """Create a normalized non-negative initial sequence."""
    x = np.arange(n, dtype=np.float64)
    if method == 'random':
        a = rng.random(n)
    elif method == 'asym_exp':
        a = np.exp(-x / max(1.0, n / 5.0))
    elif method == 'cosine':
        a = 0.5 * (1.0 + np.cos(np.pi * x / max(1, n - 1)))
    elif method == 'two_peak':
        s = max(1, n // 12)
        a = (np.exp(-0.5 * ((x - n * 0.3) / s) ** 2) +
             np.exp(-0.5 * ((x - n * 0.7) / s) ** 2))
    elif method == 'single_peak':
        s = max(1, n // 8)
        a = np.exp(-0.5 * ((x - n * 0.5) / s) ** 2)
    elif method == 'ramp':
        a = x + 1.0
    elif method == 'uniform_perturb':
        a = 1.0 + rng.normal(0, 0.3, n)
    else:
        a = rng.random(n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def block_perturb(f, rng, max_block=None):
    """Multiplicative random block perturbation, renormalized."""
    n = len(f)
    if max_block is None:
        max_block = max(2, n // 25)
    w = int(rng.integers(1, max_block + 1))
    if w >= n:
        w = max(1, n // 5)
    start = int(rng.integers(0, max(1, n - w)))
    stop = min(n, start + w)
    factor = np.exp(rng.normal(0.0, 0.3))
    cand = f.copy()
    cand[start:stop] *= factor
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s < 1e-10:
        return f
    return cand / s


def local_search(f, time_budget, rng):
    """Targeted local search with block perturbations and coordinate tweaks."""
    f = f.copy()
    n = len(f)
    best_f = f.copy()
    best_score = fast_score(f)
    start_time = time.monotonic()
    no_improve = 0
    max_no_improve = 300

    while time.monotonic() - start_time < time_budget:
        # Mix of strategies
        strategy = rng.integers(0, 4)
        if strategy == 0:
            # Block perturbation (small block)
            cand = block_perturb(f, rng, max_block=max(2, n // 50))
        elif strategy == 1:
            # Block perturbation (medium block)
            cand = block_perturb(f, rng, max_block=max(3, n // 10))
        elif strategy == 2:
            # Gaussian noise (small)
            cand = f + rng.normal(0, 0.005, n)
            cand = np.maximum(cand, 0.0)
            cand = cand / cand.sum()
        else:
            # Swap energy between two random positions
            cand = f.copy()
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                amt = cand[i] * rng.uniform(0.05, 0.3)
                cand[i] = max(0, cand[i] - amt)
                cand[j] = cand[j] + amt
                s = cand.sum()
                if s > 1e-10:
                    cand = cand / s

        sc = fast_score(cand)
        if sc < best_score - 1e-12:
            best_score = sc
            best_f = cand.copy()
            f = cand
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                # Bigger kick to escape
                f = f + rng.normal(0, 0.05, n)
                f = np.maximum(f, 0.0)
                s = f.sum()
                if s > 1e-10:
                    f = f / s
                no_improve = 0

    return best_f, best_score

                                max_steps=40)
            if s4 < best_score:
                best_score = s4
                best_f = f4.copy()
                print(f"  Phase 4 best: {s4:.6f}", flush=True)

    # ========= Phase 5: Local search polish =========
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print(f"[Phase 5] Local search, {remaining:.0f}s", flush=True)
        rng_ls = np.random.default_rng(999)
        f5, s5 = local_search(best_f.copy(), min(remaining, 120.0), rng_ls)
        if s5 < best_score:
            best_score = s5
            best_f = f5.copy()
            print(f"  Phase 5 best: {s5:.6f}", flush=True)

    # Final validation
    best_f = np.clip(best_f, 0.0, 1000.0)
    if best_f.size == 0 or best_f.sum() < 0.01:
        best_f = np.ones(512) / 512

    try:
        final_check = float(evaluate_sequence(best_f.tolist()))
        if np.isfinite(final_check):
            print(f"FINAL (evaluated): {final_check:.6f}, n={len(best_f)}",
                  flush=True)
        else:
            print(f"FINAL (fast): {best_score:.6f}, n={len(best_f)}",
                  flush=True)
    except Exception:
        print(f"FINAL (fast): {best_score:.6f}, n={len(best_f)}", flush=True)

    return [float(x) for x in best_f]
```
