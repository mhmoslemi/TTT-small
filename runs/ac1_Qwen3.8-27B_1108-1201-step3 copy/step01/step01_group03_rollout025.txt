Let me think about the gradient descent approach more carefully.

The objective is: minimize max_k (a*a)[k] subject to sum(a) = 1, a >= 0.

This is a min-max problem. We can reformulate it as:
minimize t
subject to (a*a)[k] <= t for all k
sum(a) = 1
a >= 0

The Lagrangian (ignoring the inequality constraints a >= 0 for a moment) is:
L = t - sum_k lambda_k * ((a*a)[k] - t) - mu * (sum(a) - 1)

The KKT conditions give us an update rule. But this is complex.

A simpler approach: use the "smoothed" objective:
minimize sum_k exp(beta * (a*a)[k]) / beta  (log-sum-exp approximation to max)

The gradient of (a*a)[k] with respect to a[j] is:
d(a*a)[k]/da[j] = a[k-j] + a[j-k] (with appropriate bounds, i.e., only count terms where the index is valid)

More precisely:
d(a*a)[k]/da[j] = sum over l such that l + (k-l) = k and one of l, k-l equals j
= a[k-j] (if 0 <= k-j < n) + a[j-k] (if 0 <= j-k < n, and j-k != k-j)

If k-j = j-k (i.e., k = 2j), then we get 2*a[k-j].

So the gradient is:
grad[j] = sum_k (a*a)[k] * (1[k-j >= 0 and k-j < n] * a[k-j] + 1[j-k >= 0 and j-k < n] * a[j-k])

Let me define c_k = (a*a)[k] = sum_j a[j]*a[k-j] (valid j).

Then d(c_k)/da[j] = a[k-j] (if 0 <= k-j < n) + a[j-k] (if 0 <= j-k < n and j-k != k-j)
If j-k == k-j (i.e., 2j = k), then d(c_k)/da[j] = 2*a[k-j].

So the gradient of max_k c_k (subgradient) is:
For the k that achieves the max:
grad[j] = a[k-j] (if valid) + a[j-k] (if valid and different) or 2*a[k-j] if j = k/2

This is a subgradient method. It converges slowly but is simple.

Alternative: use the smooth approximation. Let me use softmax (log-sum-exp) with a large beta:
f(a) = (1/beta) * log(sum_k exp(beta * c_k))

This is smooth and close to max for large beta.

The gradient is:
grad_f[j] = (1/beta) * sum_k exp(beta * c_k) / sum_m exp(beta * c_m) * d(c_k)/da[j]
= sum_k p_k * d(c_k)/da[j]

```python
"""
Hybrid LSE gradient descent + LP-based optimization for minimizing
the normalized autocorrelation peak of a non-negative step function.

Score = 2*n*max(f*f)/(sum f)^2

Algorithm:
1. Generate diverse initial points
2. Fast LSE gradient descent to quickly reduce peaks
3. LP-based block coordinate descent with line search for final optimization
4. Structured perturbations to escape local minima
"""

import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time


def fast_score(f):
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))


def build_conv_matrix(n, f):
    col_idx = np.repeat(np.arange(n), n)
    row_idx = np.tile(np.arange(n), n) + np.repeat(np.arange(n), n)
    data = np.tile(f, n)
    A = sp.coo_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n)).tocsr()
    return A


def lp_step(f, M, time_limit=15.0):
    n = len(f)
    A = build_conv_matrix(n, f)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'time_limit': time_limit, 'presolve': True,
                           'mip_rel_gap': 0.0})
    if res.status == 0:
        return np.maximum(res.x, 0.0)
    return None


def lse_gradient_descent(f, n_steps=5000, beta=50.0, lr=0.01, 
                          deadline=None, verbose=False):
    """Minimize LSE of convolution via gradient descent."""
    f = f.copy().astype(np.float64)
    s = f.sum()
    if s < 1e-10:
        return f
    f = f / s  # normalize to sum 1
    
    best = f.copy()
    best_sc = fast_score(f)
    
    for i in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = np.convolve(f, f)
        mc = c.max()
        exp_bc = np.exp(beta * (c - mc))
        p = exp_bc / exp_bc.sum()
        
        # Compute gradient: grad[j] = 2 * sum_{m} p[j+m] * f[m]
        n = len(f)
        grad = np.zeros(n)
        for j in range(n):
            end = min(n, 2*n-1-j)
            if end > 0:
                grad[j] = 2.0 * np.dot(p[j:j+end], f[:end])
        
        # Projected gradient step
        f_new = f - lr * grad
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-10:
            break
        f_new = f_new / s_new
        
        sc = fast_score(f_new)
        if sc < best_sc:
            best_sc = sc
            best = f_new.copy()
        
        f = f_new
    
    return best
```

import time


def fast_score(f):
    """Fast evaluation of score = 2*n*max(conv)/sum^2."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))


def build_conv_matrix(n, f):
    """Build sparse convolution matrix A where A[k,j] = f[k-j] (if valid)."""
    col_idx = np.repeat(np.arange(n), n)
    row_idx = np.tile(np.arange(n), n) + np.repeat(np.arange(n), n)
    data = np.tile(f, n)
    A = sp.coo_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp(f, M, time_limit=20.0):
    """Solve max sum(g) s.t. A(f) @ g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(n, f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lse_gradient_step(f, beta=50.0):
    """One gradient step for LSE of convolution. Returns (f_new, grad, score)."""
    c = np.convolve(f, f)
    mc = c.max()
    # Avoid overflow
    exp_bc = np.exp(beta * (c - mc))
    p = exp_bc / exp_bc.sum()
    
    n = len(f)
    grad = np.zeros(n)
    for j in range(n):
        end = min(n, 2 * n - 1 - j)
        if end > 0:
            grad[j] = 2.0 * np.dot(p[j:j + end], f[:end])
    return grad, c, p


def lse_gradient_descent(f, n_steps=5000, beta=50.0, lr=0.005,
                          deadline=None, reheat_every=2000, verbose=False):
    """Run LSE gradient descent with periodic reheat."""
    f = f.copy().astype(np.float64)
    s = f.sum()
    if s < 1e-10:
        return f
    f = f / s
    
    best = f.copy()
    best_sc = fast_score(f)
    cur_lr = lr
    
    for i in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Reheat: increase beta and reset lr
        if i % reheat_every == 0 and i > 0:
            beta = min(beta * 1.5, 500.0)
            cur_lr = max(lr * 0.5, 0.0005)
        
        grad, c, p = lse_gradient_step(f, beta)
        
        f_new = f - cur_lr * grad
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-10:
            break
        f_new = f_new / s_new
        
        sc = fast_score(f_new)
        if sc < best_sc:
            best_sc = sc
            best = f_new.copy()
        
        f = f_new
        
        if verbose and i % 1000 == 0:
            print(f"    LSE step {i}: score={best_sc:.6f}")
    
    return best


def mirror_perturb(f, rng):
    """Mirror the sequence with random probability."""
    if rng.random() < 0.5:
        return f[::-1].copy()
    return f.copy()


def shift_perturb(f, rng):
    """Circular shift by random amount, then truncate (affects linear conv)."""
    n = len(f)
    k = int(rng.integers(1, n))
    return np.roll(f, k)


def power_perturb(f, rng):
    """Apply power transform."""
    p = float(rng.uniform(0.5, 2.0))
    f_new = f ** p
    s = f_new.sum()
    if s < 1e-10:
        return f.copy()
    return f_new / s


def block_perturb(f, rng, block_size_range=(5, 50)):
    """Multiplicative block perturbation."""
    n = len(f)
    f_new = f.copy()
    lo, hi = block_size_range
    bs = int(rng.integers(max(1, lo), min(hi, n)))
    start = int(rng.integers(0, max(1, n - bs + 1)))
    end = min(n, start + bs)
    f_new[start:end] *= float(np.exp(rng.normal(0.0, 0.3)))
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return f.copy()
    return f_new / s


def lp_optimize(f, deadline, lp_time=15.0, verbose=True):
    """Run LP-based optimization until deadline."""
    f = f.copy().astype(np.float64)
    s = f.sum()
    if s < 1e-10:
        return f
    f = f / s
    
    best = f.copy()
    best_sc = fast_score(f)
    it = 0
    stall = 0
    
    while time.monotonic() < deadline:
        it += 1
        conv = np.convolve(f, f)
        M = float(conv.max())
        
        g = solve_lp(f, M, time_limit=lp_time)
        if g is None or g.sum() < 1e-10:
            # LP failed, try perturbation
            rng = np.random.default_rng(hash((it, 42)) & 0xFFFFFFFF)
            pert_method = it % 4
            if pert_method == 0:
                f_new = mirror_perturb(f, rng)
            elif pert_method == 1:
                f_new = shift_perturb(f, rng)
            elif pert_method == 2:
                f_new = power_perturb(f, rng)
            else:
                f_new = block_perturb(f, rng)
            f_new = f_new / f_new.sum()
        else:
            g = g / g.sum()
            # Line search
            best_t = 0.0
            best_t_sc = best_sc
            for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
                f_try = (1 - t) * f + t * g
                f_try = np.maximum(f_try, 0.0)
                s_try = f_try.sum()
                if s_try < 1e-10:
                    continue
                f_try = f_try / s_try
                sc_try = fast_score(f_try)
                if sc_try < best_t_sc:
                    best_t_sc = sc_try
                    best_t = t
            if best_t > 0:
                f_new = (1 - best_t) * f + best_t * g
                f_new = np.maximum(f_new, 0.0)
                s_try = f_new.sum()
                if s_try < 1e-10:
                    f_new = f.copy()
                else:
                    f_new = f_new / s_try
            
            # Apply perturbation sometimes
            if it % 50 == 0:
                rng = np.random.default_rng(hash((it, 7)) & 0xFFFFFFFF)
                if rng.random() < 0.3:
                    f_new = block_perturb(f_new, rng)
                    f_new = f_new / f_new.sum()
        
        f = f_new
        sc = fast_score(f)
        
        if sc < best_sc:
            best_sc = sc
            best = f.copy()
            stall = 0
            if verbose and it % 10 == 0:
                print(f"      LP it {it}: score={best_sc:.6f}")
        else:
            stall += 1
            if stall > 30:
                # Reset to best and perturb
                f = best.copy()
                rng = np.random.default_rng(hash((stall, it)) & 0xFFFFFFFF)
                f = block_perturb(f, rng, block_size_range=(10, max(50, len(f)//10)))
                f = f / f.sum()
                stall = 0
    
    return best


def make_initial(n, method, rng):
    """Create initial sequence of length n."""
    if method == 'uniform':
        a = np.ones(n)
    elif method == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
    elif method == 'sine_sq':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2
    elif method == 'random':
        a = rng.random(n)
    elif method == 'bimodal':
        x = np.arange(n, dtype=np.float64)
        s = max(1, n // 10)
        a = np.exp(-0.5 * ((x - n//4) / s) ** 2) + np.exp(-0.5 * ((x - 3*n//4) / s) ** 2)
    elif method == 'ramp':
        a = np.linspace(0.1, 2.0, n)
    elif method == 'v_shape':
        mid = n // 2
        a = np.abs(np.arange(n) - mid) / mid + 0.1
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a
