"""
Hybrid optimization for minimizing normalized autocorrelation peak of non-negative sequences.
Combines smoothed gradient descent (log-sum-exp) with simulated annealing polishing,
using FFT-based gradient computation and O(n) per-step SA updates.
"""

import numpy as np
import time


def _compute_b(a):
    """Compute b = convolve(a, a)."""
    return np.convolve(a, a)


def _score_2n_maxb(n, b):
    """Compute 2n * max(b)."""
    return 2.0 * n * np.max(b)


def _project_simplex(v, target=1.0):
    """Project vector onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _smooth_max_and_grad(a, beta):
    """
    Compute smooth max of b_k and gradient w.r.t. a using log-sum-exp and FFT.
    Returns (smooth_val, grad_a, b).
    """
    n = len(a)
    b = np.convolve(a, a)
    b_max = np.max(b)
    shifted = beta * (b - b_max)
    log_sum = np.log(np.sum(np.exp(shifted)))
    val = log_sum / beta + b_max
    p = np.exp(shifted - log_sum)

    # Gradient: grad_a[j] = 2 * sum_m a[m] * p[m+j]
    # Computed via FFT cross-correlation
    N = 1
    while N < 2 * n:
        N <<= 1
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    grad_a = 2.0 * np.fft.irfft(np.conj(A) * P, N)[:n]

    return val, grad_a, b


def _make_sqrt_geo(n):
    """Truncated square root of geometric series: p_k = C(2k,k)/4^k."""
    p = np.zeros(n)
    p[0] = 1.0
    for k in range(1, n):
        p[k] = p[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    p /= np.sum(p)
    return p


def _make_hann(n):
    """Hann window normalized to sum 1."""
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * i / (n - 1)))
    w = np.maximum(w, 1e-10)
    w /= np.sum(w)
    return w


def _make_two_level(n, eps):
    """Two-level anti-symmetric sequence."""
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    a /= np.sum(a)
    return a


def _make_power_decay(n, alpha):
    """Power law decay: a_k ∝ (k+1)^(-alpha)."""
    k = np.arange(n, dtype=np.float64) + 1.0
    a = k ** (-alpha)
    a /= np.sum(a)
    return a


def _make_fourier(n):
    """Try a flat-top inspired window (non-negative version)."""
    i = np.arange(n, dtype=np.float64)
    x = 2.0 * np.pi * i / (n - 1)
    # Raised cosine to the 4th power (flatter top)
    w = (0.5 * (1.0 + np.cos(x))) ** 4 + 1e-6
    w /= np.sum(w)
    return w

def _run_sa(a0, deadline, rng, verbose=False):
    """
    Simulated annealing with two-position mass transfer moves.
    O(n) per step via vectorized b-update.
    """
    n = len(a0)
    a = a0.copy()
    s = np.sum(a)
    if s > 1e-10:
        a /= s

    b = _compute_b(a)
    b_max = np.max(b)
    best = a.copy()
    best_b_max_val = float(b_max)
    best_score = 2.0 * n * best_b_max_val

    # Temperature schedule
    # Typical |delta_b| ~ 2*t*mean(a) ~ 2*(0.05*max_a)*(1/n) ~ 0.1/n
    # delta_score ~ 2n * delta_b ~ 0.2
    # So T_init ~ 0.1
    T = 0.05
    min_T = 1e-10
    # Cool every K steps
    K = max(1, int(n / 50))
    gamma = 0.995

    step = 0
    accept_count = 0
    last_print = time.monotonic()
    last_best_print = time.monotonic()

    # Precompute index range
    max_b_true = 2.0 * n * b_max

    while time.monotonic() < deadline:
        # Pick two distinct positions
        j = rng.integers(n)
        l = rng.integers(n)
        if j == l:
            continue

        # Transfer amount
        t_max = min(a[j], a[l])
        if t_max < 1e-12:
            continue
        t = t_max * rng.uniform(0.01, 0.1)

        # Compute affected k range
        k_lo = min(j, l)
        k_hi = min(2 * n - 2, max(j, l) + n - 1)
        k_range = np.arange(k_lo, k_hi + 1)
        k_len = len(k_range)

        # a[k-j] and a[k-l] (zero where out of bounds)
        idx_j = k_range - j
        idx_l = k_range - l
        a_kj = np.zeros(k_len, dtype=np.float64)
        mask_j = (idx_j >= 0) & (idx_j < n)
        a_kj[mask_j] = a[idx_j[mask_j]]
        a_kl = np.zeros(k_len, dtype=np.float64)
        mask_l = (idx_l >= 0) & (idx_l < n)
        a_kl[mask_l] = a[idx_l[mask_l]]

        delta_b = 2.0 * t * (a_kl - a_kj)
        new_slice = b[k_lo:k_hi + 1] + delta_b
        new_local_max = np.max(new_slice)

        # Determine new global max
        if b_max <= new_local_max + 1e-15:
            # Old max is at or below new local max, or in affected range
            # New global max is at least new_local_max
            # Could be the old b_max if it's outside affected range and > new_local_max
            # For efficiency, just use new_local_max (may underestimate, but safe for acceptance)
            new_b_max = new_local_max
        else:
            # Old max is above new local max
            # If old max is outside affected range, it's still the global max
            new_b_max = b_max

        # Compute score change
        delta_score = 2.0 * n * new_b_max - 2.0 * n * b_max

        # Acceptance
        if delta_score < 0 or rng.random() < np.exp(-delta_score / T):
            # Accept
            a[j] -= t
            a[l] += t
            b[k_lo:k_hi + 1] += delta_b
            b_max = new_b_max
            accept_count += 1

            if 2.0 * n * b_max < best_score:
                best_score = 2.0 * n * b_max
                best = a.copy()
                best_b_max_val = float(b_max)
                if verbose and time.monotonic() - last_best_print > 5.0:
                    print(f"    SA new best: {best_score:.6f} (T={T:.6f})")
                    last_best_print = time.monotonic()
        else:
            # Reject: try reversed direction (swap j and l)
            delta_b_rev = -delta_b
            new_slice_rev = b[k_lo:k_hi + 1] + delta_b_rev
            new_local_max_rev = np.max(new_slice_rev)

            if b_max <= new_local_max_rev + 1e-15:
                new_b_max_rev = new_local_max_rev
            else:
                new_b_max_rev = b_max

            delta_score_rev = 2.0 * n * new_b_max_rev - 2.0 * n * b_max

            if delta_score_rev < 0 or rng.random() < np.exp(-delta_score_rev / T):

                a[l] -= t
                b[k_lo:k_hi + 1] += delta_b_rev
                b_max = new_b_max_rev
                accept_count += 1
                if 2.0 * n * b_max < best_score:
                    best_score = 2.0 * n * b_max
                    best = a.copy()
                    best_b_max_val = float(b_max)
                    if verbose and time.monotonic() - last_best_print > 5.0:
                        print(f"    SA new best (rev): {best_score:.6f} (T={T:.6f})")
                        last_best_print = time.monotonic()

        step += 1
        if step % K == 0:
            T *= gamma

        if verbose and time.monotonic() - last_print > 10.0:
            rate = accept_count / max(1, step)
            print(f"    SA step {step}: score={best_score:.6f}, T={T:.6f}, acc_rate={rate:.3f}")
            last_print = time.monotonic()

    return best, best_score

                a[l] -= t
                b[k_lo:k_hi + 1] += delta_b_rev
                b_max = new_b_max_rev
                accept_count += 1
                accepted = True

        if accepted and 2.0 * n * b_max < best_score:
            best_score = 2.0 * n * b_max
            best = a.copy()
            # Update peak tracking
            if time.monotonic() - last_best_print > 5.0:
                if verbose:
                    print(f"    SA2 new best: {best_score:.6f} (T={T:.6f}, step={step})")
                last_best_print = time.monotonic()
            # Refresh peak locations periodically
            if step % (20 * K) == 0:
                peak_k = np.where(b >= b_max * 0.99)[0]
                if len(peak_k) == 0:
                    peak_k = np.array([n - 1])

        step += 1
        if step % K == 0:
            T *= gamma
            if T < min_T:
                T = min_T

        if verbose and time.monotonic() - last_print > 10.0:
            rate = accept_count / max(1, step)
            print(f"    SA2 step {step}: score={best_score:.6f}, T={T:.6f}, acc={rate:.3f}")
            last_print = time.monotonic()

    return best, best_score


def _coordinate_descent_polish(a, deadline, rng, verbose=False):
    """
    Coordinate descent: for each coordinate j, find optimal a[j] given others fixed.
    Uses ternary search on the 1D convex subproblem (max of b_k as function of a[j]).
    """
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    if s > 1e-10:
        a /= s

    b = _compute_b(a)
    best_score = 2.0 * n * np.max(b)
    best = a.copy()

    # Precompute: for each j, the indices of b_k affected
    # b_k depends on a_j for k in [j, j+n-1] (intersected with [0, 2n-2])
    # When we change a_j to x, b_k changes by: 2*x*a[k-j] - 2*a[j]*a[k-j] = 2*(x-a[j])*a[k-j]
    # Plus we need to adjust the sum: if we set a[j]=x, we need to redistribute (a[j]-x) to others.
    # For simplicity, we'll use the approach: set a[j]=x, scale others to maintain sum=1.

    # Actually, a cleaner approach: parameterize as a[j] = x, sum_{i≠j} a[i] = 1-x.
    # Scale others proportionally: a'[i] = a[i] * (1-x) / (1-a[j]) for i≠j.
    # This changes ALL b_k, making the 1D problem hard.

    # Alternative: just do a small local search around the current a[j].
    # For each j, try a[j] + delta for a few deltas and pick the best.

    # Simpler approach: random coordinate perturbation with acceptance
    order = rng.permutation(n)
    it = 0

    while time.monotonic() < deadline:
        j = order[it % n]
        # Try a range of perturbations
        base_a_j = a[j]
        if base_a_j < 1e-12:
            it += 1
            continue

        # Try decreasing a[j] and giving the mass to a random other position
        # Actually, let's just try: increase a[j] by delta (steal from a random l)
        l = rng.integers(n)
        if l == j:
            l = (j + 1) % n

        delta_options = [0.01 * base_a_j, 0.05 * base_a_j, 0.1 * base_a_j,
                         0.01 * a[l], 0.05 * a[l]]

        best_delta = 0.0
        best_sc = 2.0 * n * np.max(b)

        for d in delta_options:
            d = min(d, 0.5 * min(base_a_j, a[l]) if base_a_j > a[l] else 0.5 * a[l])
            if d < 1e-12:
                continue
            # Try: a[j] += d, a[l] -= d
            a[j] += d
            a[l] -= d
            if a[l] < 0:
                a[j] -= d
                continue
            b_new = _compute_b(a)
            sc_new = 2.0 * n * np.max(b_new)
            if sc_new < best_sc:
                best_sc = sc_new
                best_delta = d
            a[j] -= d
            a[l] += d

        if best_delta > 0:
            a[j] += best_delta
            a[l] -= best_delta
            b = _compute_b(a)
            if 2.0 * n * np.max(b) < best_score:
                best_score = 2.0 * n * np.max(b)
                best = a.copy()

        it += 1

        if verbose and it % 1000 == 0:
            print(f"    CD iter {it}: score={best_score:.6f}")

    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Hybrid optimization: smoothed GD + intelligent SA + coordinate polish.
    Minimizes 2n*max(convolve(a,a))/sum(a)^2 over non-negative sequences.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(6121)
    parent = np.clip(parent, 0.0, None)
    if parent.size == 0:
        parent = np.ones(6121)
    n = parent.size
    print(f"Sequence length: n={n}")

    # Generate diverse starting points
    starts = []
    starts.append(("parent", parent / np.sum(parent)))
    starts.append(("sqrt_geo", _make_sqrt_geo(n)))
    starts.append(("hann", _make_hann(n)))
    starts.append(("power_0.5", _make_power_decay(n, 0.5)))
    starts.append(("power_0.3", _make_power_decay(n, 0.3)))
    starts.append(("power_0.7", _make_power_decay(n, 0.7)))
    starts.append(("two_level", _make_two_level(n, 0.5)))
    starts.append(("fourier4", _make_fourier(n)))
    # Random starts
    for _ in range(3):
        a_r = rng.random(n)
        a_r /= np.sum(a_r)
        starts.append((f"rand_{rng.integers(1000)}", a_r))

    # Allocate time
    total_time = deadline - time.monotonic()
    # Phase 1: GD on top candidates (60% of time, split among best starts)
    # Phase 2: SA on best result (30%)
    # Phase 3: Polish (10%)

    phase1_time = 0.55 * total_time
    phase2_time = 0.30 * total_time
    phase3_time = 0.15 * total_time

    global_best = None
    global_best_score = np.inf

    # Phase 1: Run GD on multiple starts
    # Give parent more time, split rest among others
    n_starts = len(starts)
    # Parent gets 40% of phase1, rest split evenly
    t_parent = 0.4 * phase1_time
    t_others = phase1_time - t_parent
    t_each = t_others / (n_starts - 1)

    print(f"\n=== Phase 1: Smoothed Gradient Descent ({phase1_time:.0f}s) ===")

    # First: parent (most promising)
    if time.monotonic() < deadline:
        d1 = min(time.monotonic() + t_parent, deadline)
        print(f"  GD on parent ({t_parent:.0f}s)...")
        best_a, best_s = _run_smoothed_gd(starts[0][1], d1, rng, beta_init=80.0)
        if best_s < global_best_score:
            global_best_score = best_s
            global_best = best_a.copy()
            print(f"    => {global_best_score:.6f}")

    # Then: analytical starts (more promising than random)
    analytical_starts = starts[1:8]  # sqrt_geo, hann, power_0.5, power_0.3, power_0.7, two_level, fourier4
    for name, a_start in analytical_starts:
        if time.monotonic() >= deadline:
            break
        d1 = min(time.monotonic() + t_each, deadline)
        remaining_for_this = d1 - time.monotonic()
        if remaining_for_this < 5:
            break
        print(f"  GD on {name} ({remaining_for_this:.0f}s)...")
        best_a, best_s = _run_smoothed_gd(a_start, d1, rng, beta_init=100.0)
        if best_s < global_best_score:
            global_best_score = best_s
            global_best = best_a.copy()
            print(f"    => NEW BEST: {global_best_score:.6f}")
        else:
            print(f"    => {best_s:.6f} (best={global_best_score:.6f})")

    # Random starts get whatever time is left in phase 1
    for name, a_start in starts[8:]:
        if time.monotonic() >= deadline:
            break
        d1 = min(time.monotonic() + 10, deadline)
        remaining_for_this = d1 - time.monotonic()
        if remaining_for_this < 3:
            break
        best_a, best_s = _run_smoothed_gd(a_start, d1, rng, beta_init=50.0)
        if best_s < global_best_score:
            global_best_score = best_s
            global_best = best_a.copy()
            print(f"  GD {name}: NEW BEST {global_best_score:.6f}")

    print(f"\nPhase 1 complete. Best score: {global_best_score:.6f}")

    if global_best is None:
        global_best = starts[0][1]
        global_best_score = _score_2n_maxb(n, _compute_b(global_best))

    # Phase 2: SA polishing
    if time.monotonic() < deadline:
        print(f"\n=== Phase 2: Simulated Annealing ({phase2_time:.0f}s) ===")
        d2 = min(time.monotonic() + phase2_time, deadline)
        print(f"  Starting SA from score={global_best_score:.6f}")

        # Split SA time: first half standard, second half intelligent
        mid2 = (time.monotonic() + d2) / 2.0

        best_a, best_s = _run_sa(global_best, mid2, rng, verbose=True)
        if best_s < global_best_score:
            global_best_score = best_s
            global_best = best_a.copy()
            print(f"  SA standard: {global_best_score:.6f}")

        if time.monotonic() < d2:
            best_a, best_s = _run_sa_intelligent(global_best, d2, rng, verbose=True)
            if best_s < global_best_score:
                global_best_score = best_s
                global_best = best_a.copy()
                print(f"  SA intelligent: {global_best_score:.6f}")

    # Phase 3: Final polish with more GD + short SA
    if time.monotonic() < deadline:
        print(f"\n=== Phase 3: Polish ({phase3_time:.0f}s) ===")
        remaining = deadline - time.monotonic()

        # GD refinement with lower beta
        t_gd3 = 0.5 * remaining
        d3 = min(time.monotonic() + t_gd3, deadline)
        best_a, best_s = _run_smoothed_gd(global_best, d3, rng, beta_init=200.0)
        if best_s < global_best_score:
            global_best_score = best_s
            global_best = best_a.copy()
            print(f"  GD polish: {global_best_score:.6f}")

        # Final SA
        if time.monotonic() < deadline:
            d4 = deadline
            best_a, best_s = _run_sa(global_best, d4, rng, verbose=True)
            if best_s < global_best_score:
                global_best_score = best_s
                global_best = best_a.copy()
                print(f"  Final SA: {global_best_score:.6f}")

    # Ensure non-negativity and compute final score
    global_best = np.maximum(global_best, 0.0)
    result = [float(x) for x in global_best]

    # Final verification
    final_score = float(evaluate_sequence(result))
    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {final_score:.6f}")
    print(f"Internal score: {global_best_score:.6f}")
    print(f"Sum: {np.sum(global_best):.6f}, Max: {np.max(global_best):.6f}")
    print(f"Min: {np.min(global_best):.6f}")
    print(f"{'='*50}")

    return result
