"""
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov-accelerated gradient descent with adaptive soft-max peak
approximation, LP-based alternating refinement, and multi-scale optimization.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def compute_conv(a):
    """Compute full convolution efficiently."""
    n = len(a)
    size = 2 * n - 1
    if n <= 256:
        return np.convolve(a, a)
    else:
        fft_size = 1 << (size - 1).bit_length()
        fa = np.fft.rfft(a, fft_size)
        return np.fft.irfft(fa * fa, fft_size)[:size]


def normalize(a, target=None):
    """Normalize to target sum (default sqrt(2n))."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def grad_exact_peak(a):
    """
    Compute gradient of max(conv(a,a)) using the exact peak location.
    grad[m] = 2*a[k-m] if 0 <= k-m < n, else 0, where k = argmax(conv(a,a)).
    """
    n = len(a)
    c = compute_conv(a)
    k = int(c.argmax())
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    grad = np.zeros(n)
    for j in range(lo, hi + 1):
        grad[j] = 2.0 * a[k - j]
    return grad, k


def grad_softmax(a, tau):
    """
    Compute softmax-weighted gradient.
    grad[m] = 2 * sum_i w[m+i] * a[i]
    """
    n = len(a)
    c = compute_conv(a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(c))
        w[int(c.argmax())] = 1.0
    else:
        w /= w_sum
    
    grad = np.zeros(n)
    for m in range(n):
        # grad[m] = 2 * sum_{i=0}^{n-1} a[i] * w[m+i] (where m+i < 2n-1)
        valid = min(n, 2 * n - 1 - m)
        if valid > 0:
            grad[m] = 2.0 * np.dot(a[:valid], w[m:m + valid])
    return grad


def nesterov_step(a, v, lr, tau, use_exact=False):
    """
    One Nesterov gradient step.
    a: current iterate
    v: momentum vector
    lr: learning rate
    tau: soft-max temperature (ignored if use_exact)
    """
    n = len(a)
    target = np.sqrt(2.0 * n)
    
    # Nesterov lookahead
    a_look = a + v
    
    if use_exact:
        grad = grad_exact_peak(a_look)[0]
    else:
        grad = grad_softmax(a_look, tau)
    
    # Project out mean (fixed-sum constraint)
    grad -= grad.mean()
    
    # Update momentum
    beta = 0.9
    v = beta * v - lr * grad
    
    # Update
    a_new = a + v
    
    # Project to non-negative
    a_new = np.maximum(a_new, 0.0)
    
    # Normalize
    a_new = normalize(a_new, target)
    
    return a_new, v


def lp_refine(f, M, n_max=1024):
    """
    Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0.
    Returns g or None.
    """
    n = len(f)
    if n > n_max:
        return None
    
    c_obj = -np.ones(n)
    
    # Build constraint matrix efficiently
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]
    
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

"""
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov momentum with adaptive softmax peak, FFT-accelerated gradients,
LP-based alternating refinement, and multi-phase time management.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_conv(a):
    """Compute full convolution using FFT for large n."""
    n = len(a)
    size = 2 * n - 1
    if n <= 256:
        return np.convolve(a, a)
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    return np.fft.irfft(fa * fa, fft_size)[:size]


def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    c = compute_conv(a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    """Normalize to target sum (default sqrt(2n))."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def grad_exact(a):
    """Gradient of max(conv(a,a)) at the exact peak. Returns (grad, peak_idx)."""
    n = len(a)
    c = compute_conv(a)
    k = int(c.argmax())
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    grad = np.zeros(n)
    grad[lo:hi + 1] = 2.0 * a[k - lo:k - lo + (hi - lo + 1)][::-1]
    return grad, k


def grad_softmax(a, tau):
    """Softmax-weighted gradient: grad[m] = 2 * sum_i a[i]*w[m+i]."""
    n = len(a)
    c = compute_conv(a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(c))
        w[int(c.argmax())] = 1.0
    else:
        w /= w_sum
    grad = np.zeros(n)
    for m in range(n):
        valid = min(n, 2 * n - 1 - m)
        if valid > 0:
            grad[m] = 2.0 * np.dot(a[:valid], w[m:m + valid])
    return grad


def momentum_step(a, v, lr, tau, use_exact, beta):
    """One momentum step with projection and normalization."""
    n = len(a)
    target = np.sqrt(2.0 * n)
    # Nesterov lookahead
    a_look = a + v
    if use_exact:
        grad = grad_exact(a_look)[0]
    else:
        grad = grad_softmax(a_look, tau)
    # Project out mean for fixed-sum constraint
    grad -= grad.mean()
    # Momentum update
    v_new = beta * v - lr * grad
    a_new = a + v_new
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize(a_new, target)
    return a_new, v_new


def lp_solve(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M elementwise, g >= 0."""
    n = len(f)
    if n > 1024:
        return None
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        if lo <= hi:
            A[m, lo:hi + 1] = f[m - lo:m - lo + (hi - lo + 1)]
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_alternate(a, deadline, n_iters=50):
    """Alternating LP refinement: fix one side, optimize the other."""
    n = len(a)
    if n > 768:
        return a
    S_t = np.sqrt(2.0 * n)
    c = compute_conv(a)
    M = float(c.max())
    f = a.copy()
    g = a.copy()
    best = a.copy()
    best_score = compute_obj(a)
    for it in range(n_iters):
        if time.monotonic() > deadline - 5.0:
            break
        # Solve for g given f
        g_new = lp_solve(f, M)
        if g_new is not None and g_new.sum() > 1e-10:
            g_new = normalize(g_new, S_t)
            # Blend
            for t_b in [0.6, 0.4, 0.2]:
                cand = (1.0 - t_b) * f + t_b * g_new
                cand = np.maximum(cand, 0.0)
                cand = normalize(cand, S_t)
                s = compute_obj(cand)
                if s < best_score:
                    best_score = s
                    best = cand.copy()
                    f = cand.copy()
                    c = compute_conv(f)
                    M = float(c.max())
                    break
        else:
            break
        # Solve for f given g
        f_new = lp_solve(g, M)
        if f_new is not None and f_new.sum() > 1e-10:
            f_new = normalize(f_new, S_t)
            for t_b in [0.6, 0.4, 0.2]:
                cand = (1.0 - t_b) * g + t_b * f_new
                cand = np.maximum(cand, 0.0)
                cand = normalize(cand, S_t)
                s = compute_obj(cand)
                if s < best_score:
                    best_score = s
                    best = cand.copy()
                    g = cand.copy()
                    c = compute_conv(g)
                    M = float(c.max())
                    break
            if f is not None and not np.any(np.abs(f_new) > 1e-15):

                for factor in [0.8, 0.9, 0.95]:
                    a_test = a_cd.copy()
                    a_test[j] *= factor
                    a_test = normalize(a_test, S_t)
                    s_test = compute_obj(a_test)
                    if s_test < best_score:
                        best_score = s_test
                        best = a_test.copy()
                        a_cd = a_test.copy()
                        break
            if not np.allclose(a_cd, a):
                a = a_cd
                v = np.zeros(n)

    s = compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()
    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function: multi-phase optimization with diverse initializations.
    Phase 1 (30%): Explore many starts with moderate gradient descent.
    Phase 2 (40%): Refine best candidates with LP and reheating.
    Phase 3 (30%): Polish with exact gradient, heavy LP, coordinate descent.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total = float(budget_s)
    deadline_final = t_start + total - 10.0

    best_seq = None
    best_score = float('inf')

    def update_best(seq, score):
        nonlocal best_seq, best_score
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  *** NEW BEST: score={best_score:.6f}, n={len(seq)}, "
                  f"t={time.monotonic()-t_start:.1f}s")

    # --- Generate diverse starting sequences ---
    starts = []

    # From provided sequence
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        gs = np.maximum(gs, 0.0)
        if gs.size >= 16 and np.all(np.isfinite(gs)):
            for L in [256, 384, 512, 640, 768]:
                if L <= len(gs):
                    starts.append(gs[:L].copy())
                else:
                    # Tile and truncate
                    reps = int(np.ceil(L / len(gs)))
                    tiled = np.tile(gs, reps)[:L]
                    starts.append(tiled.copy())
            if 16 <= len(gs) <= 1024:
                starts.append(gs.copy())
    except NameError:
        pass

    # Random starts with different structure
    for n in [256, 384, 512, 640]:
        starts.append(rng.random(n) * 0.5 + 0.25)
        # Bimodal random
        w = rng.random(n)
        w = w * (0.5 + 0.5 * np.cos(np.pi * np.arange(n) / n))
        starts.append(w + 0.1)

    # Structured starts exploiting center-peak structure
    for n in [256, 512]:
        i = np.arange(n, dtype=np.float64)
        # Anti-symmetric emphasis (large at edges, small at center)
        starts.append(0.3 + 0.7 * np.abs(np.cos(np.pi * i / n)))
        # Two-lobe
        starts.append(np.exp(-((i - n * 0.25) / (n * 0.2)) ** 2) +
                      np.exp(-((i - n * 0.75) / (n * 0.2)) ** 2) + 0.05)
        # Four-lobe
        starts.append(np.sum(np.exp(-((i - n * (k + 1) / (n_lobe + 1)) / (n * 0.12)) ** 2)
                              for k, n_lobe in [(0, 3), (1, 3), (2, 3)]).copy() + 0.05)
        # High-frequency oscillation
        starts.append(0.5 + 0.3 * np.sin(2 * np.pi * 11.0 * i / n))
        starts.append(0.5 + 0.3 * np.sin(2 * np.pi * 17.0 * i / n))
        # Gaussian center
        center = (n - 1) / 2.0
        starts.append(np.exp(-0.5 * ((i - center) / (n * 0.2)) ** 2) + 0.05)
        # Inverse Gaussian (edges)
        starts.append(1.0 - 0.8 * np.exp(-0.5 * ((i - center) / (n * 0.25)) ** 2))

    # Sort shorter first
    starts.sort(key=len)
    # Deduplicate (keep max 20)
    if len(starts) > 20:
        # Keep diversity: mix of lengths
        starts = starts[:8] + sorted(
            [s for s in starts[8:] if len(s) >= 384],
            key=len)[:12]

    print(f"Budget: {total}s | Starts: {len(starts)} sequences")

    # === PHASE 1: Exploration (30% of time) ===
    p1_deadline = t_start + total * 0.30
    print(f"\n=== Phase 1: Exploration (deadline={p1_deadline:.0f}s) ===")

    for a0 in starts:
        if time.monotonic() > p1_deadline - 5.0:
            break
        t0 = time.monotonic()
        seq, score = optimize_phase1(a0, p1_deadline)
        update_best(seq, score)
        print(f"  n={len(a0):4d} | score={score:.6f} | time={time.monotonic()-t0:6.1f}s | "
              f"best={best_score:.6f}")

    # === PHASE 2: Refinement (40% of time) ===
    p2_deadline = t_start + total * 0.70
    print(f"\n=== Phase 2: Refinement (deadline={p2_deadline:.0f}s) ===")

    if best_seq is not None:
        # Refine the current best
        t0 = time.monotonic()
        seq, score = optimize_phase2(best_seq, p2_deadline, rng)
        update_best(seq, score)
        print(f"  Refined best: n={len(seq)}, score={score:.6f}, "
              f"time={time.monotonic()-t0:.1f}s")

        # Perturb and re-refine (escape local minima)
        for trial in range(4):
            if time.monotonic() > p2_deadline - 20.0:
                break
            scales = [0.02, 0.05, 0.1, 0.2]
            pert_scale = scales[trial % len(scales)]
            perturbed = best_seq + pert_scale * rng.standard_normal(len(best_seq))
            perturbed = np.maximum(perturbed, 0.0)
            t0 = time.monotonic()
            seq, score = optimize_phase2(perturbed, p2_deadline, rng)
            update_best(seq, score)
            print(f"  Perturb trial {trial} (scale={pert_scale}): score={score:.6f}, "
                  f"best={best_score:.6f}, time={time.monotonic()-t0:.1f}s")

    # Try a couple more fresh random starts with full refinement
    for n_try in [512, 640]:
        if time.monotonic() > p2_deadline - 30.0:
            break
        a0 = rng.random(n_try) * 0.5 + 0.25
        t0 = time.monotonic()
        seq, score = optimize_phase2(a0, p2_deadline, rng)
        update_best(seq, score)
        print(f"  Fresh n={n_try}: score={score:.6f}, time={time.monotonic()-t0:.1f}s")

    # === PHASE 3: Polishing (remaining time) ===
    p3_deadline = deadline_final
    print(f"\n=== Phase 3: Polishing (deadline={p3_deadline:.0f}s) ===")

    if best_seq is not None:
        # Multiple polish passes
        for pass_i in range(6):
            if time.monotonic() > p3_deadline - 10.0:
                break
            t0 = time.monotonic()
            seq, score = optimize_phase3(best_seq, p3_deadline)
            update_best(seq, score)
            print(f"  Polish pass {pass_i}: score={score:.6f}, time={time.monotonic()-t0:.1f}s")
            if score >= best_score * (1 - 1e-9):
                # No improvement, try a small perturbation first
                if pass_i < 3:
                    perturbed = best_seq + 0.01 * rng.standard_normal(len(best_seq))
                    perturbed = np.maximum(perturbed, 0.0)
                    seq2, score2 = optimize_phase3(perturbed, p3_deadline)
                    update_best(seq2, score2)
                else:
                    break

    # Final validation
    if best_seq is not None:
        final = compute_obj(best_seq)
        print(f"\n{'='*60}")
        print(f"FINAL RESULT: n={len(best_seq)}, score={final:.6f}")
        print(f"Previous best: 1.517421 | Target: 1.5027")
        print(f"{'='*60}")
        try:
            ev = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence confirms: {ev:.6f}")
        except Exception as e:
            print(f"Warning: evaluate_sequence gave: {e}")
        return [float(x) for x in best_seq]

    # Fallback
    print("Warning: No valid sequence found, using fallback.")
    return [1.0] * 256
