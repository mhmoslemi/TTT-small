Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. LP-based best response iterations (finding g that maximizes sum subject to f*g <= M)
2. Gradient-based steps using soft-max weighted subgradients
3. Random perturbations for escaping local minima
4. Multiple sequence lengths and initializations

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def fast_cross_convolve(f, g):
    """Linear cross-convolution of f and g using FFT."""
    n1, n2 = len(f), len(g)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    G = np.fft.rfft(g, fft_size)
    result = np.fft.irfft(F * G, fft_size)
    return result[:size]


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = (u - (cssv - 1.0) / ind) > 0
    # Find the largest index where cond is True
    idx = np.where(cond)[0]
    if len(idx) == 0:
        theta = 0.0
    else:
        rho = idx[-1] + 1
        theta = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)


def build_A_ub(f):
    """Build the constraint matrix for the LP."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A[k, js:je + 1] = f[js:je + 1]
    return A


def lp_best_response(f, M):

    n = len(f)
    A_ub = build_A_ub(f)
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)

    if res.success and res.x is not None:
        g = np.maximum(res.x, 0)
        s = g.sum()
        if s > 1e-15:

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based best response iterations
combined with FFT-accelerated gradient steps, adaptive line search over blend
parameters, random restarts, and multiple sequence lengths. The LP finds a

descent direction. Gradient steps use a log-sum-exp smooth approximation of
the max, computed via FFT cross-correlation.

import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.ones(1)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = (u - (cssv - 1.0) / ind) > 0
    idx = np.where(cond)[0]
    if len(idx) == 0:
        theta = 0.0
    else:
        rho = idx[-1] + 1
        theta = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)

    k = np.arange(2 * n - 1)[:, None].astype(np.float64)

    """Build A_ub = pattern * f[j] for each row k."""
    return pattern * f[None, :]


def lp_best_response(f, pattern, M):
    """Solve LP: max sum(g) s.t. A_ub @ g <= M, g >= 0."""
    n = len(f)
    A_ub = build_A_ub(f, pattern)
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            g = np.maximum(res.x, 0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None


def compute_gradient_fft(a, conv, t):
    """Compute gradient of LSE(t * conv) w.r.t. a using FFT cross-correlation."""
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)
    scaled = t * (conv - max_c)
    e = np.exp(scaled)
    w = e / e.sum()
    total_len = len(w) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(W * np.conj(A), fft_size)[:total_len]
    return 2.0 * corr[:n]


def optimize_sequence(a0, time_limit, tau_candidates=None, lse_t=50.0, eta=0.3):
    """
    Optimize using alternating LP best-response and gradient steps.
    a0 must be non-negative, will be normalized to sum=1.
    Returns (best_a, best_normalized_obj).
    """
    if tau_candidates is None:
        tau_candidates = [0.05, 0.1, 0.2, 0.4]

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s

    n = len(a)
    pattern = precompute_pattern(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    iter_count = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        M = conv.max()

        # LP step with line search over tau
        g = lp_best_response(a, pattern, M)

        if g is not None:
            best_tau_obj = best_obj
            best_tau_a = a.copy()
            best_tau_val = tau_candidates[0]

            for tau in tau_candidates:
                a_trial = (1.0 - tau) * a + tau * g
                a_trial = project_simplex(a_trial)
                conv_trial = fast_convolve(a_trial)
                obj_trial = 2.0 * n * conv_trial.max()
                if obj_trial < best_tau_obj:
                    best_tau_obj = obj_trial
                    best_tau_a = a_trial.copy()
                    best_tau_val = tau

            if best_tau_obj < best_obj - 1e-12:
                a = best_tau_a
                conv = fast_convolve(a)
                best_obj = best_obj  # will update below
                # Recompute to be safe
                conv = fast_convolve(a)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_obj - 1e-10:
                    best_obj = cur_obj
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Gradient step (every 2 iterations or when LP stalls)
        if (iter_count % 2 == 1) or (no_improve >= 10):
            if time.monotonic() - t0 >= time_limit:
                break
            grad = compute_gradient_fft(a, conv, lse_t)
            # Center gradient to preserve sum constraint better
            grad_centered = grad - grad.mean()
            a_new = a - eta * grad_centered
            a = project_simplex(a_new)
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj - 1e-10:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        # Adaptive: if stuck, perturb
        if no_improve >= 50:
            noise_scale = 0.05 + 0.05 * (no_improve / 100.0)
            a = a + noise_scale * np.random.default_rng(iter_count * 7919).standard_normal(n)
            a = project_simplex(a)
            conv = fast_convolve(a)
            no_improve = 0

        iter_count += 1

    return best_a, best_obj


def generate_initializations(rng, n, parent=None):
    """Generate diverse initializations of length n."""
    x = np.arange(n, dtype=np.float64)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussians
    for sf in [0.15, 0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits.append(g)

    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits.append(ramp)

    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(1, n // 5)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits.append(bi)

    # Cosine bump
    cos_bump = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_bump /= cos_bump.sum()
    inits.append(cos_bump)

    # Random Dirichlet
    for alpha in [0.5, 1.0, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits.append(r)

    # Binary (random subset)
    for frac in [0.3, 0.5, 0.7]:
        mask = rng.random(n) < frac
        if mask.sum() > 0:
            b = mask.astype(np.float64)
            b /= b.sum()
            inits.append(b)

    # Symmetric random
    half = rng.random(n // 2 + 1)
    sym = np.concatenate([half[:n//2][::-1], half]) if n % 2 == 0 else \
          np.concatenate([half[:n//2][::-1], half])
    sym = sym[:n]
    sym = np.maximum(sym, 0)
    if sym.sum() > 1e-10:
        inits.append(sym / sym.sum())

    # Parent
    if parent is not None:
        if parent.size >= n:
            p = parent[:n].copy()
            p = np.maximum(p, 0)
            if p.sum() > 1e-10:
                inits.append(p / p.sum())
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            p = np.maximum(p, 0)
            if p.sum() > 1e-10:
                inits.append(p / p.sum())

    return inits


def get_parent_sequence():
    """Try to access the parent sequence from global scope."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Uses LP-based best response + gradient optimization
    with multiple initializations, lengths, and random restarts.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + budget_s - 30.0

    best_sequence = None
    best_score = float('inf')

    parent = get_parent_sequence()

    # lengths to try
    lengths = [64, 128, 256, 512, 1024]
    if parent is not None:
        pn = len(parent)
        # Include parent length if reasonable
        if 32 <= pn <= 4096 and pn not in lengths:
            lengths.append(pn)
            lengths.sort()

    print(f"Starting optimization. Budget: {budget_s}s. Parent length: {parent.size if parent is not None else 'N/A'}")
    print(f"Lengths to try: {lengths}")

    # Phase 1: Quick scan over lengths and initializations
    print("Phase 1: Quick scan...")
    phase1_fraction = 0.30
    phase1_budget = phase1_fraction * (total_deadline - time.monotonic())

    all_inits = []
    for n in lengths:
        inits = generate_initializations(rng, n, parent)
        for i, a0 in enumerate(inits):
            all_inits.append((n, i, a0))

    # Sort by length (prefer medium lengths first)
    order = {l: i for i, l in enumerate(sorted(set(l for l, _, _ in all_inits),
                                                   key=lambda x: -abs(x - 256)))}
    all_inits.sort(key=lambda x: order[x[0]])

    n_total = len(all_inits)
    time_per_scan = max(3.0, phase1_budget / n_total)
    time_per_scan = min(time_per_scan, 20.0)  # cap at 20s per scan

    scan_results = []
    for i, (n, idx, a0) in enumerate(all_inits):
        if time.monotonic() > total_deadline - 60:
            break
        t_alloc = min(time_per_scan, total_deadline - time.monotonic() - 60)
        if t_alloc < 2:
            break

        a_opt, obj_norm = optimize_sequence(a0, time_limit=t_alloc,
                                            tau_candidates=[0.1, 0.2, 0.4],
                                            lse_t=40.0, eta=0.3)
        score = float(evaluate_sequence(a_opt.tolist()))

        if np.isfinite(score):
            scan_results.append((score, n, a_opt, obj_norm))
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Scan: n={n}, init={idx}, score={score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best score: {best_score:.6f}")

    # Phase 2: Deep optimization
    print("Phase 2: Deep optimization...")
    phase2_budget = 0.55 * (total_deadline - time.monotonic())
    phase2_end = time.monotonic() + phase2_budget

    # Build list of starting points for deep optimization
    deep_starts = []
    # Add top scan results
    for score, n, a_opt, obj_norm in scan_results[:8]:
        deep_starts.append((n, a_opt))

    # Ensure the current best is included
    if best_sequence is not None:
        deep_starts.append((len(best_sequence), best_sequence.copy()))

    # Add some random restarts based on best
    if best_sequence is not None:
        bn = len(best_sequence)
        for _ in range(3):
            perturbed = best_sequence + 0.15 * rng.standard_normal(bn)
            perturbed = project_simplex(np.maximum(perturbed, 0))
            deep_starts.append((bn, perturbed))

    n_deep = len(deep_starts)
    time_per_deep = max(10.0, phase2_budget / n_deep)

    for i, (n, a0) in enumerate(deep_starts):
        if time.monotonic() > phase2_end - 10:
            break
        t_alloc = min(time_per_deep, phase2_end - time.monotonic() - 10)
        if t_alloc < 5:
            break

        # Try different parameter settings
        param_sets = [
            {'tau_candidates': [0.05, 0.1, 0.2], 'lse_t': 50.0, 'eta': 0.3},
            {'tau_candidates': [0.1, 0.3, 0.5], 'lse_t': 80.0, 'eta': 0.2},
            {'tau_candidates': [0.05, 0.15, 0.3], 'lse_t': 30.0, 'eta': 0.4},
        ]

        for ps in param_sets:
            if time.monotonic() > phase2_end - 10:
                break
            t_single = (phase2_end - time.monotonic() - 10) / (len(param_sets) - i * 0)
            t_single = min(t_single, t_alloc / len(param_sets))
            if t_single < 3:
                break

            a_opt, obj_norm = optimize_sequence(a0, time_limit=t_single, **ps)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Deep: n={n}, start={i}, params={ps['lse_t']:.0f}, score={score:.6f}")

    print(f"Phase 2 done. Best score: {best_score:.6f}")

    # Phase 3: Final polish
    print("Phase 3: Final polish...")
    remaining = total_deadline - time.monotonic()
    if best_sequence is not None and remaining > 20:
        n = len(best_sequence)
        # Run several polish iterations with increasing lse_t
        for lse_t in [50.0, 100.0, 200.0, 400.0]:
            if time.monotonic() > total_deadline - 5:
                break
            t_single = remaining / 4.0
            a_opt, obj_norm = optimize_sequence(best_sequence, time_limit=t_single,
                                                tau_candidates=[0.05, 0.1],
                                                lse_t=lse_t, eta=0.15)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish: lse_t={lse_t:.0f}, score={score:.6f}")
            remaining = total_deadline - time.monotonic()

        # One more gradient-only polish
        if time.monotonic() < total_deadline - 10:
            t_single = total_deadline - time.monotonic() - 5
            # Pure gradient refinement
            a = best_sequence.copy()
            s = a.sum()
            a = a / s
            nn = len(a)
            conv = fast_convolve(a)
            for _ in range(200):
                if time.monotonic() > total_deadline - 5:
                    break
                grad = compute_gradient_fft(a, conv, 200.0)
                grad_centered = grad - grad.mean()
                a = project_simplex(a - 0.05 * grad_centered)
                conv = fast_convolve(a)
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a.copy()
                print(f"  Grad polish: score={score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"FINAL SCORE: {best_score:.6f}, length={len(best_sequence)}")
    return [float(x) for x in best_sequence]
