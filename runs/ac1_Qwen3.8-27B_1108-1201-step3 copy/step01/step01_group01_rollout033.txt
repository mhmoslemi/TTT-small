```python
def mirror_descent_step(a, b, tau, eta):
    """
    One step of mirror descent to minimize smooth max of b = convolve(a,a).
    """
    n = len(a)
    
    # Compute softmax weights over b
    b_max = np.max(b)
    logits = (b - b_max) / tau
    weights = np.exp(logits)
    weights /= np.sum(weights)
    
    # Compute gradient: g_i = 2 * sum_k weights_k * a[k-i] (valid k)
    g = np.zeros(n)
    # For efficiency, only use top-weighted k's
    threshold = np.max(weights) * 1e-4
    active_k = np.where(weights > threshold)[0]
    
    for k in active_k:
        w = weights[k]
        # (P_k a)_i = a[k-i] if 0 <= k-i <= n-1
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # g[lo:hi+1] += w * 2 * a[k - lo : k - hi] (reversed)
            g[lo:hi+1] += 2.0 * w * a[k - hi : k - lo + 1][::-1]
    
    # Mirror descent update
    a_new = a * np.exp(-eta * g)
    # Handle potential underflow
    a_new = np.maximum(a_new, 1e-300)
    # Normalize
    a_new /= np.sum(a_new)
    
    return a_new
```

    """Compute the score for a sequence."""
    n = len(a)
    b = np.convolve(a, a)

def _make_logistic_step(n, sharpness, offset=0.5):
    """Create a logistic step profile."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (1.0 + np.exp(-sharpness * (x - offset)))
    a = np.maximum(a, 1e-10)

def _make_two_level(n, eps):

def _local_search(a, deadline, rng, n_iters=100000):

    n = len(a)
    a = a.copy()
    a /= np.sum(a)
    best_score = _score(a)

        if time.monotonic() > deadline:
            break
        
        # Pick two random positions
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:

        delta = rng.uniform(0.001, 0.05) * a[i]

        a[j] += delta

        else:

            a[i] += delta
