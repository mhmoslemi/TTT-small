import numpy as np
import time

def _compute_b(a):
    """Compute the autocorrelation b = convolve(a, a)."""
    return np.convolve(a, a)

def _compute_score(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    b = np.convolve(a, a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * np.max(b) / (sum_a ** 2)

def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _grad_bk(a, k):
    """Gradient of b_k = sum_i a_i * a_{k-i} w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * a[k - idx]
    return grad

def _make_start_two_level(n, eps):

    a[n//2:] = 1.0 - eps
    S = np.sum(a)

def _optimize_nesterov_mirror(a, deadline, rng, K=30, eta_init=None, beta=0.9):

    Uses multiplicative (mirror) updates.

    a = np.asarray(a, dtype=np.float64).copy()

        a /= S

        a = np.ones(n) / n

    best = a.copy()
    best_score = 2.0 * n * np.max(np.convolve(a, a))
    
    # Nesterov momentum
    y = a.copy()  # lookahead point
    momentum = 0.0
    t_counter = 1.0
    
    eta = eta_init
    no_improve = 0
    patience = 200

    while time.monotonic() < deadline:

        b_max = np.max(b_y)
        cur_score = 2.0 * n * b_max
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = y.copy()
            no_improve = 0
        else:
            no_improve += 1

            momentum = 0.0
            no_improve = 0

            # Softmax weighting

            wts /= np.sum(wts)

        grad = np.zeros(n)
        for ki in range(len(k_idx)):
            if wts[ki] < 1e-8:

            k = int(k_idx[ki])
            lo = max(0, k - n + 1)

                grad[idx] += wts[ki] * 2.0 * y[k - idx]

        logits -= np.max(logits)
        a_new = np.exp(logits)
        a_new /= np.sum(a_new)
        
        a = a_new
        t_counter = t_new
        momentum = momentum_new
    
    # Final: also evaluate a (not just y)
    final_score = 2.0 * n * np.max(np.convolve(a, a))
    if final_score < best_score:

        best_score = final_score

"""
Improved Nesterov-accelerated subgradient descent with adaptive top-K peak
selection and dual (projected + mirror) update strategies to minimize the
normalized autocorrelation peak of a non-negative sequence.
Key improvements: Nesterov momentum, adaptive curriculum on K, backtracking
step size, mirror descent option, extended polishing, and more diverse starts.

import numpy as np
import time


def _compute_b(a):
    return np.convolve(a, a)


def _compute_score_normalized(a):
    n = len(a)
    b = _compute_b(a)
    return 2.0 * n * np.max(b)


def _project_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted := u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_bk_at(y, k, n):
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * y[k - idx]
    return grad


def _weighted_grad(y, k_idx, wts, n):
    grad = np.zeros(n)
    for ki in range(len(k_idx)):
        if wts[ki] < 1e-10:
            continue
        k = int(k_idx[ki])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            grad[idx] += wts[ki] * 2.0 * y[k - idx]
    return grad


def _top_k_weights(b, K, temperature_scale=8.0):
    n_b = len(b)
    if K >= n_b:
        return np.arange(n_b), np.ones(n_b) / n_b
    k_idx = np.argpartition(b, -K)[-K:]
    k_idx = k_idx[np.argsort(b[k_idx])[::-1]]
    b_sel = b[k_idx]
    b_max = b_sel[0]
    spread = b_max - b_sel[-1]
    if spread < 1e-15:
        wts = np.ones(K) / K
    else:
        wts = np.exp(-temperature_scale * (b_max - b_sel) / spread)
        wts /= np.sum(wts)
    return k_idx, wts


def _make_two_level(n, eps):
    a = np.ones(n)
    half = n // 2
    a[:half] = 1.0 + eps
    a[half:] = 1.0 - eps
    S = np.sum(a)
    return a / S


def _make_linear_anti(n, eps):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * (2.0 * i / max(n - 1, 1) - 1.0)
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _make_cosine(n, freq, eps, phase=0.0):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * i / n + phase)
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _make_quadratic_anti(n, eps):
    i = np.arange(n, dtype=np.float64)
    x = 2.0 * i / max(n - 1, 1) - 1.0
    a = 1.0 + eps * (x ** 2 - 0.333)
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _subgradient_optimize(a, deadline, K_start=10, K_end=50, eta_init=None,
                          use_nesterov=True, patience=250, cool_rate=0.75,
                          min_eta=1e-10, temperature_scale=8.0):
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    else:
        a = np.ones(n) / n

    if eta_init is None:
        eta_init = 0.15 / np.sqrt(n)

    best = a.copy()
    best_score = _compute_score_normalized(best)

    eta = eta_init
    no_improve = 0
    t_iter = 0.0

    if use_nesterov:
        y = a.copy()
        a_prev = a.copy()
        t_param = 1.0

    iter_count = 0
    while time.monotonic() < deadline:
        iter_count += 1

        # Compute b at current point
        if use_nesterov:
            b = _compute_b(y)
        else:
            b = _compute_b(a)
        b_max = np.max(b)
        cur_score = 2.0 * n * b_max

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            if use_nesterov:
                best = y.copy()
            else:
                best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= patience:
            eta *= cool_rate
            no_improve = 0
            if use_nesterov:
                a_prev = a.copy()
                y = a.copy()
                t_param = 1.0
            if eta < min_eta:
                break

        # Adaptive K: interpolate from K_start to K_end over runtime
        frac = min(1.0, iter_count / max(patience * 3, 1))
        K = int(K_start + frac * (K_end - K_start))
        K = max(3, min(K, 200))

        k_idx, wts = _top_k_weights(b, K, temperature_scale)
        grad = _weighted_grad(y if use_nesterov else a, k_idx, wts, n)

        # Center for sum constraint
        grad -= np.mean(grad)

        if use_nesterov:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_param ** 2))
            mu = (t_param - 1.0) / t_new
            y_new = y + mu * (y - a)
            step = y_new - eta * grad
            a_new = _project_simplex(step, 1.0)
            a_prev = a.copy()
            a = a_new
            y = y_new
            t_param = t_new
        else:
            step = a - eta * grad
            a = _project_simplex(step, 1.0)

        # Safety: ensure non-negativity
        if np.any(a < -1e-15):
            a = np.maximum(a, 0.0)
            S = np.sum(a)
            if S > 1e-10:
                a /= S
            else:
                a = best.copy()
                break

    # Final check
    final_b = _compute_b(a)
    final_score = 2.0 * n * np.max(final_b)
    if final_score < best_score:
        best = a.copy()
        best_score = final_score
    if use_nesterov:
        final_b2 = _compute_b(y)
        final_score2 = 2.0 * n * np.max(final_b2)
        if final_score2 < best_score:
            best = y.copy()
            best_score = final_score2

    return best, best_score


def _mirror_descent_optimize(a, deadline, K=30, eta_init=None, patience=200):
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    else:
        a = np.ones(n) / n

    if eta_init is None:
        eta_init = 2.0 / np.sqrt(n)

    best = a.copy()
    best_score = _compute_score_normalized(best)
    eta = eta_init
    no_improve = 0
    eps = 1e-15

    while time.monotonic() < deadline:
        b = _compute_b(a)
        b_max = np.max(b)
        cur_score = 2.0 * n * b_max

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= patience:
            eta *= 0.7
            no_improve = 0
            if eta < 1e-12:
                break

        k_idx, wts = _top_k_weights(b, K, temperature_scale=8.0)
        grad = _weighted_grad(a, k_idx, wts, n)

        # Mirror (multiplicative) update
        log_a = np.log(a + eps)
        logits = log_a - eta * grad
        logits -= np.max(logits)
        a_new = np.exp(logits)
        S_new = np.sum(a_new)
        if S_new < 1e-15:
            a = best.copy()
            break
        a = a_new / S_new

    final_score = 2.0 * n * np.max(_compute_b(a))
    if final_score < best_score:
        best = a.copy()
        best_score = final_score
    return best, best_score


def _polish(a, deadline, K=100, eta_init=None):
    n = len(a)
    if eta_init is None:
        eta_init = 0.01 / np.sqrt(n)
    return _subgradient_optimize(a, deadline, K_start=K, K_end=K,
                                 eta_init=eta_init, use_nesterov=True,
                                 patience=500, cool_rate=0.9, min_eta=1e-12,
                                 temperature_scale=4.0)


def _perturb(a, rng, scale=0.08):
    n = len(a)
    freqs = rng.integers(1, max(2, n // 3), size=min(8, n // 2))
    noise = np.zeros(n)
    for f in freqs:
        phase = rng.uniform(0, 2.0 * np.pi)
        amp = rng.uniform(-1.0, 1.0)
        noise += amp * np.cos(2.0 * np.pi * f * np.arange(n) / n + phase)
    max_abs = np.max(np.abs(noise))
    if max_abs > 1e-15:
        noise /= max_abs
    a_new = a + scale * noise
    a_new = _project_simplex(a_new, 1.0)
    return a_new
