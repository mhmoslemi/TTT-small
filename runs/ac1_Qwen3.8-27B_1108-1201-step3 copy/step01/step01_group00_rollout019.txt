def flat_spectrum_init(n, rng, iterations=100):
    """Initialize with approximately flat power spectrum."""
    # Start with random phase
    F = np.exp(1j * 2 * np.pi * rng.random(n // 2 + 1))
    F[0] = 1.0  # DC
    
    for _ in range(iterations):
        a = np.fft.irfft(F, n)
        a = np.abs(a)  # enforce non-negativity
        a = np.maximum(a, 1e-6)  # avoid exact zeros
        F = np.fft.rfft(a)
        #Normalize magnitude to be flat (keep phase)
        mag = np.abs(F)

        F = F / mag  # unit magnitude, keep phase
    
    a = np.fft.irfft(F, n)
    a = np.abs(a)
    a = np.maximum(a, 1e-6)
    return a

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:

    v_v = np.zeros(n)

    best_score, _ = score_peak(a)

    stale = 0

        if score < best_score:
            best_score = score
            best_a = a.copy()
            stale = 0
        else:
            stale += 1
        
        g = peak_grad(a, k)
        g -= np.mean(g)  # project to sum=0 subspace
        
        step += 1
        m_v = beta1*m_v + (1-beta1)*g
        v_v = beta2*v_v + (1-beta2)*g**2

        v_hat = v_v / (1 - beta2**step)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)

                a = np.maximum(a, 0.0)
                a = normalize(a, S)

            print(f"    step={step}, score={score:.7f}, best={best_score:.7f}, lr={lr:.4f}, t={time.monotonic()-t0:.0f}s")
    
    return best_a, best_score

    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = np.sum(a)  # keep the actual sum, score is scale-invariant
    
    # Precompute conv
    conv = fft_convolve(a, a)
    current_score = 2.0 * n * np.max(conv) / S**2
    best_score = current_score
    best_a = a.copy()
    
    t0 = time.monotonic()
    iteration = 0
    
    # Precompute for each k, which j values are affected and the coefficients
    # For coordinate k, changing a[k] from old to new affects:
    #   conv[2k] by (new^2 - old^2) [quadratic]
    #   conv[j] by 2*(new-old)*a[j-k] for j where 0<=j-k<n, j!=2k [linear in (new-old)]
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        # Sweep through coordinates
        for k in range(n):
            old = a[k]
            if old < 1e-15:
                continue  # skip zero elements (for now)
            
            # Determine candidate values for a[k]
            # The peak of conv is at some index p
            p = int(np.argmax(conv))
            
            # If p is affected by a[k], we can reduce the peak by reducing a[k]
            # conv[p] = base[p] + contribution
            jk = p - k
            if 0 <= jk < n:
                if p == 2*k:
                    # quadratic: conv[2k] = base + a[k]^2
                    # To reduce: decrease a[k]
                    # New value: a[k] = sqrt(max(0, target - base))
                    base_p = conv[p] - old*old
                    # Try a few values
                    candidates = [0.0, old*0.8, old*0.5, old*0.2, old]
                else:
                    # linear: conv[p] = base + 2*a[k]*a[jk]
                    base_p = conv[p] - 2*old*a[jk]
                    # To match target: a[k] = (target - base_p) / (2*a[jk])
                    candidates = [0.0, old*0.8, old*0.5, old*0.2, old]
                    if a[jk] > 1e-15:
                        # Find second-highest peak
                        conv_sorted = np.sort(conv)[::-1]
                        target = conv_sorted[1] if len(conv_sorted) > 1 else 0
                        x_eq = max(0.0, (target - base_p) / (2.0 * a[jk]))
                        candidates.append(x_eq)
            
            # Also always try 0 and the current value
            candidates = list(set([max(0.0, c) for c in candidates]))
            
            best_x = old
            best_local_max = np.max(conv)
            
            for x in candidates:
                dx = x - old
                if abs(dx) < 1e-15:
                    continue

                # The elements that increase are those with a[j-k] > 0
                # The elements that decrease are... well, if dx < 0, the linear terms decrease
                # Let's just compute the new conv for the affected elements
                # This is O(n) per candidate
                
                # Actually, let's just compute the new max directly
                # The maximum can only come from:
                # 1. Elements of conv that are unaffected by a[k] (their value is unchanged)
                # 2. conv[2k] (quadratic change)
                # 3. conv[j] for valid j != 2k (linear change)
                
                # Unaffected max:
                affected = np.zeros(2*n-1, dtype=bool)
                affected[2*k] = True
                jk_valid = np.arange(max(0, k-(n-1)), min(2*n-2, k+n-1) + 1)
                for j in jk_valid:
                    if j-k >= 0 and j-k < n:
                        affected[j] = True
                
                unaffected_max = np.max(conv[~affected]) if np.any(~affected) else 0

                quadr = conv[2*k] + x*x - old*old
                # Linear terms
                # conv[j] + 2*dx*a[j-k] for j where 0<=j-k<n and j!=2k
                j_vals = np.arange(max(0, k-(n-1)), min(2*n-2, k+n-1) + 1)
                lin_vals = np.array([

                    for j in j_vals

                lin_max = np.max(lin_vals) if len(lin_vals) > 0 else 0
                
                new_max = max(unaffected_max, quadr, lin_max)

                    best_local_max = new_max

                a[k] = best_x
                conv[2*k] += best_x*best_x - old*old
                # Update linear terms
                for j in range(max(0, k-(n-1)), min(2*n-1, k+n) ):
                    jk = j - k
                    if 0 <= jk < n and j != 2*k:
                        conv[j] += 2*dx*a[jk]

        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
        
        iteration += 1

            conv = fft_convolve(a, a)  # recompute
            if verbose:
