def _fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def _compute_obj(a):

def _adam_step(a, grad, m, v, t, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """One Adam step."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad * grad
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    return a_new, m, v

def _lp_refine(f, M, max_iter=50):
    """LP-based refinement. Returns improved sequence or None."""
    n = len(f)
    if n > 1500:
        return None
    
    c_obj = -np.ones(n)

    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]

    bounds = [(0.0, None)] * n

        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

"""
Optimizes non-negative sequences to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses FFT-based Adam gradient descent with top-k peak focusing, multiple sequence
lengths, LP-based refinement, and iterative re-weighting for flat autocorrelation.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def _compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = _fft_convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _normalize(a, target):
    """Normalize a to have the given sum."""
    S = a.sum()
    if S < 1e-12:
        return a
    return a * (target / S)


def _compute_grad_topk(a, tau, k=None):
    """
    Compute gradient using top-k peak lags with softmax weighting.
    Returns (gradient, max_c, c_array).
    """
    n = len(a)
    c = _fft_convolve(a, a)
    max_c = float(c.max())

    if tau <= 1.0:
        # Hard max: single peak
        m_star = int(np.argmax(c))
        grad = np.zeros(n)
        lo = max(0, m_star - n + 1)
        hi = min(n - 1, m_star)
        for j in range(lo, hi + 1):
            grad[j] = 2.0 * a[m_star - j]
    else:
        # Soft-max weighted over all lags
        c_shifted = c - max_c
        w = np.exp(tau * c_shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            # Fallback to hard max
            m_star = int(np.argmax(c))
            w = np.zeros(len(c))
            w[m_star] = 1.0
        else:
            w /= w_sum

        # grad[j] = 2 * sum_k a[k] * w[j+k] where j+k < 2n-1
        # Use cross-correlation via FFT
        a_flip = a[::-1]
        cw = _fft_convolve(a_flip, w)
        grad = 2.0 * cw[n - 1:2 * n - 1]

    return grad, max_c, c


def _adam_update(a, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Standard Adam update step."""
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * grad * grad
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    return a - lr * m_hat / (np.sqrt(v_hat) + eps), m, v


def _optimize_adam(a0, deadline, max_iters=5000, tau_min=3.0, tau_max=300.0,
                   lr_init=0.05, lr_min=0.001):
    """
    Adam-based optimization with annealed soft-max temperature.
    """
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    S_t = np.sqrt(2.0 * n)
    a = _normalize(a, S_t)

    score = _compute_obj(a)
    best = a.copy()
    best_score = score

    m_opt = np.zeros(n)
    v_opt = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break

        progress = it / max_iters

        # Anneal temperature: start low (smooth), end high (sharp peak focus)
        if progress < 0.2:
            tau = tau_min
        elif progress < 0.6:
            tau = tau_min + (tau_max - tau_min) * (progress - 0.2) / 0.4
        else:
            tau = tau_max

        # Learning rate: cosine decay
        lr = lr_min + 0.5 * (lr_init - lr_min) * (1.0 + np.cos(np.pi * progress))

        # Compute gradient
        grad, max_c, c = _compute_grad_topk(a, tau)

        # Project out mean (preserve sum)
        grad -= grad.mean()

        # Adam step
        a_new, m_opt, v_opt = _adam_update(a, grad, m_opt, v_opt, it + 1, lr)
        a_new = np.maximum(a_new, 0.0)
        a = _normalize(a_new, S_t)

        # Periodic evaluation
        if (it + 1) % 300 == 0:
            s = _compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()

    # Final evaluation
    s = _compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()

    return best, best_score

def _lp_refine(f, M):
    """Solve LP: max sum(g) s.t. convolve(f,g) <= M, g >= 0."""
    n = len(f)
    if n > 1200:
        return None

    c_obj = -np.ones(n)

    # Build constraint matrix A[m,j] = f[m-j] if 0 <= m-j < n
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]

    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

    M = float(c.max())

    rng = np.random.default_rng(123)

    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break

        g = _lp_refine(a, M)
        if g is None:
            break

        improved = False
        for t in [0.95, 0.8, 0.6, 0.4, 0.2, 0.1]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = _normalize(a_new, S_t)
            s_new = _compute_obj(a_new)
            if s_new < best_score:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                break

        if not improved:
            # Perturb to escape local optimum
            noise = 0.02 * rng.randn(n)
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _normalize(a, S_t)

        c = _fft_convolve(a, a)
        M = float(c.max())

    return best, best_score


def _reweighting_method(a0, deadline, max_iters=200):
    """
    Iterative reweighting: adjust sequence to flatten autocorrelation.
    At each step, reweight based on current autocorrelation profile.
    """
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    S_t = np.sqrt(2.0 * n)
    a = _normalize(a, S_t)

    best = a.copy()
    best_score = _compute_obj(a)

    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break

        c = _fft_convolve(a, a)
        max_c = float(c.max())

        # Compute "excess" at each lag relative to target
        target = 1.0 / (2 * n - 1)  # ideal flat value (normalized to sum=1)
        # Excess lags are those above target
        excess = np.maximum(c - target, 0.0)

        # For each position j, compute its "contribution" to excess lags
        # contrib[j] = sum over excess lags m of a[m-j] (for valid m-j)
        contrib = np.zeros(n)
        for m in range(2 * n - 1):
            if excess[m] > 0:
                lo = max(0, m - n + 1)
                hi = min(n - 1, m)
                for j in range(lo, hi + 1):
                    contrib[j] += excess[m] * a[m - j]

        # Reduce positions with high contribution
        alpha = 0.1 * (1.0 + np.cos(np.pi * it / max_iters)) + 0.01
        a_new = a - alpha * contrib
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize(a_new, S_t)

        s = _compute_obj(a_new)
        if s < best_score:
            best_score = s
            best = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            a_new2 = a - 0.01 * contrib
            a_new2 = np.maximum(a_new2, 0.0)
            a_new2 = _normalize(a_new2, S_t)
            s2 = _compute_obj(a_new2)
            if s2 < best_score:
                best_score = s2
                best = a_new2.copy()
                a = a_new2
            else:
                # Stuck, try random perturbation
                a = a + 0.01 * np.random.default_rng(it).randn(n)
                a = np.maximum(a, 0.0)
                a = _normalize(a, S_t)

    return best, best_score


def _generate_initial_sequences(rng, lengths):
    """Generate diverse initial sequences for various lengths."""
    seqs = []
    for n in lengths:
        i = np.arange(n, dtype=np.float64)
        # Uniform with noise
        seqs.append(rng.random(n) * 0.3 + 0.5)
        # Raised cosine
        seqs.append(0.5 + 0.5 * np.cos(2 * np.pi * i / n))
        # Inverted (edge-heavy)
        seqs.append(0.5 + 0.5 * np.cos(np.pi * i / (n / 2 - 1)))
        # Gaussian bump
        center = n / 2.0
        seqs.append(np.exp(-0.5 * ((i - center) / (n * 0.3)) ** 2) + 0.05)
        # Chirp
        seqs.append(0.5 + 0.4 * np.abs(np.sin(4 * np.pi * i / n)))
        # Two peaks
        seqs.append(np.exp(-0.5 * ((i - n * 0.2) / (n * 0.15)) ** 2) +
                    np.exp(-0.5 * ((i - n * 0.8) / (n * 0.15)) ** 2) + 0.05)
    return seqs


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Uses multi-stage Adam optimization with soft-max
    peak annealing, LP-based refinement, and iterative reweighting.
    Explores multiple sequence lengths and starting points.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 12.0)

    best_seq = None
    best_score = float('inf')

    def update_best(seq, score, label=""):
        nonlocal best_seq, best_score
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: score={score:.6f}, "
                  f"n={len(seq)} {label}")

    # Get known good starting point
    base_seq = None
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 8 and np.all(np.isfinite(gs)) and np.all(gs >= 0):
            base_seq = gs
            print(f"Using height_sequence_1: n={len(gs)}")
    except NameError:
        print("No height_sequence_1 available, using random starts")
    except Exception:
        pass

    # Phase 1: Quick optimization of base sequence and nearby lengths
    print("Phase 1: Optimizing known good sequence and variants...")

    candidates = []

    if base_seq is not None:
        for n in [128, 256, 512, 1024]:
            if n <= len(base_seq):
                candidates.append(base_seq[:n].copy())
            else:
                # Interpolate/tile
                from scipy.interpolate import interp1d
                x_old = np.linspace(0, 1, len(base_seq))
                x_new = np.linspace(0, 1, n)
                f = interp1d(x_old, base_seq, kind='cubic')
                candidates.append(np.maximum(f(x_new), 0.0))

    # Add diverse random starts
    for n in [512, 1024, 2048]:
        candidates.append(rng.random(n) * 0.4 + 0.4)

    # Sort by length
    candidates.sort(key=len)

    # Time budget for Phase 1: ~40% of total
    phase1_deadline = t_start + 0.40 * (budget_s - 12)
    for idx, a0 in enumerate(candidates):
        if time.monotonic() > phase1_deadline - 30:
            break
        n = len(a0)
        print(f"  Candidate {idx}: n={n}")
        t_lim = min(time.monotonic() + 120, phase1_deadline - 20)
        seq, score = _optimize_multistage(a0, t_lim)
        update_best(seq, score, f"(phase1, start {idx})")

    # Phase 2: Extended optimization on the current best
    print(f"Phase 2: Extended optimization, current best={best_score:.6f}")
    phase2_deadline = t_start + 0.65 * (budget_s - 12)

    if best_seq is not None:
        # Multi-restart from best
        for restart in range(3):
            if time.monotonic() > phase2_deadline - 30:
                break
            a0 = best_seq.copy()
            if restart > 0:
                # Add noise for diversification
                a0 = a0 + 0.05 * rng.randn(len(a0))
                a0 = np.maximum(a0, 0.0)
                a0 = _normalize(a0, np.sqrt(2.0 * len(a0)))
            t_lim = min(time.monotonic() + 120, phase2_deadline - 20)
            seq, score = _optimize_multistage(a0, t_lim)
            update_best(seq, score, f"(phase2, restart {restart})")

        # Try LP refinement
        if len(best_seq) <= 1024:
            print("  LP refinement...")
            t_lim = min(time.monotonic() + 120, phase2_deadline - 10)
            seq, score = _lp_optimize(best_seq, t_lim, max_iters=60)
            update_best(seq, score, "(LP)")

    # Phase 3: Explore different lengths and methods
    print(f"Phase 3: Exploration, current best={best_score:.6f}")
    phase3_deadline = t_start + 0.80 * (budget_s - 12)

    # Try larger sequences
    for n in [1024, 2048, 4096]:
        if time.monotonic() > phase3_deadline - 30:
            break
        print(f"  Trying n={n}...")
        if base_seq is not None and len(base_seq) >= min(n, len(base_seq)):
            if n <= len(base_seq):
                a0 = base_seq[:n].copy()
            else:
                reps = n // len(base_seq) + 1
                a0 = np.tile(base_seq, reps)[:n]
        else:
            a0 = rng.random(n) * 0.3 + 0.5

        t_lim = min(time.monotonic() + 90, phase3_deadline - 20)
        seq, score = _optimize_multistage(a0, t_lim)
        update_best(seq, score, f"(phase3, n={n})")

    # Phase 4: Final intensive refinement
    print(f"Phase 4: Final refinement, current best={best_score:.6f}")

    if best_seq is not None:
        n_best = len(best_seq)

        # Intensive Adam with very sharp peak focus
        if time.monotonic() < deadline - 20:
            t_lim = min(time.monotonic() + 120, deadline - 20)
            seq, score = _optimize_adam(best_seq, t_lim, max_iters=8000,
                                         tau_min=200.0, tau_max=1000.0,
                                         lr_init=0.005, lr_min=0.0003)
            update_best(seq, score, "(phase4 sharp)")

        # Iterative reweighting
        if time.monotonic() < deadline - 20:
            t_lim = min(time.monotonic() + 90, deadline - 20)
            seq, score = _reweighting_method(best_seq, t_lim, max_iters=300)
            update_best(seq, score, "(reweighting)")

        # Another LP pass if applicable
        if len(best_seq) <= 1024 and time.monotonic() < deadline - 15:
            t_lim = min(time.monotonic() + 90, deadline - 15)
            seq, score = _lp_optimize(best_seq, t_lim, max_iters=80)
            update_best(seq, score, "(final LP)")

        # Final Adam polish
        if time.monotonic() < deadline - 10:
            t_lim = deadline - 10
            seq, score = _optimize_adam(best_seq, t_lim, max_iters=5000,
                                         tau_min=300.0, tau_max=1000.0,
                                         lr_init=0.003, lr_min=0.0002)
            update_best(seq, score, "(final polish)")

    # Final validation
    if best_seq is not None:
        final_score = _compute_obj(best_seq)
        print(f"\nFinal result: n={len(best_seq)}, score={final_score:.6f}")

        # Verify with evaluate_sequence
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence confirms: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

        return [float(x) for x in best_seq]

    # Fallback
    print("Falling back to uniform sequence.")
    return [1.0] * 512
