Uses multiplicative gradient descent (mirror descent on simplex),

def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    return 2.0 * n * float(c.max()) / (S * S)

def mirror_descent_step(a, eta, top_k=1):
    """One step of mirror descent (multiplicative) targeting top-k peaks."""
    n = len(a)
    c = np.convolve(a, a)
    
    # Find top-k peak positions
    if top_k == 1:
        peak_indices = [int(c.argmax())]
    else:
        peak_indices = list(np.argsort(c)[-top_k:])
    
    # Compute gradient-like update: for each peak at position m,
    # the "pressure" on a[j] is proportional to a[m-j] (for valid m-j)
    log_update = np.zeros(n)
    for m in peak_indices:
        for j in range(n):
            idx = m - j
            if 0 <= idx < n:
                log_update[j] += a[idx]
    
    # Multiplicative update: a[j] *= exp(-eta * log_update[j])
    a_new = a * np.exp(-eta * log_update)
    
    # Renormalize to maintain sum
    S_new = a_new.sum()
    if S_new > 0:
        S_old = a.sum()
        a_new = a_new * (S_old / S_new)
    
    return np.maximum(a_new, 0.0)

"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses mirror descent (multiplicative updates) with multi-peak softmax targeting,
LP-based refinement, simulated annealing escapes, and structured exploration.
"""
import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    return 2.0 * n * float(c.max()) / (S * S)


def normalize_to(a, target=None):
    """Normalize sequence to have a target sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a
    return a * (target / S)


def mirror_descent_step(a, eta, tau, top_k=1):
    """One mirror descent step targeting top peaks with softmax weighting."""
    n = len(a)
    c = np.convolve(a, a)

    if top_k == 1:
        peak_idx = int(c.argmax())
        w = np.zeros(2 * n - 1)
        w[peak_idx] = 1.0
    else:
        c_shifted = c - c.max()
        w = np.exp(tau * c_shifted)
        w /= w.sum()

    # Compute: for each j, pressure = sum_m w[m] * a[m-j] (valid m-j in [0,n-1])
    grad = np.zeros(n)
    for j in range(n):
        lo = j
        hi = min(2 * n - 1, j + n - 1)
        if lo <= hi:
            grad[j] = 2.0 * float(np.dot(w[lo:hi+1], a[lo-j:hi+1-j]))

    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 0.0)
    S_old = a.sum()
    S_new = a_new.sum()
    if S_new > 1e-12:
        a_new = a_new * (S_old / S_new)
    return a_new


def optimize_mirror_descent(a0, deadline, eta0=0.3, tau0=10.0, max_iters=10000):
    """Optimize using mirror descent with adaptive parameters."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = normalize_to(a)

    best_score = compute_obj(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline - 30:
            break

        progress = it / max_iters
        eta = eta0 * (1.0 - 0.95 * progress) + 0.005
        tau = tau0 + 80.0 * progress

        a_new = mirror_descent_step(a, eta, tau)
        a = a_new

        if (it + 1) % 50 == 0:
            s = compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            # If stuck, try a perturbation
            if no_improve > 20:
                rng = np.random.default_rng(it * 7919)
                a_pert = a * (1.0 + 0.05 * rng.standard_normal(n))
                a_pert = np.maximum(a_pert, 0)
                a_pert = normalize_to(a_pert)
                s_pert = compute_obj(a_pert)
                if s_pert < best_score * 1.05:
                    a = a_pert
                no_improve = 0

    return best, best_score


def build_lp_matrix(f):
    """Build the LP constraint matrix for conv(f, b) <= M."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]
    return A


def lp_solve(f, M):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0."""
    n = len(f)
    if n > 2048:
        return None

    c_obj = -np.ones(n)
    A = build_lp_matrix(f)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def optimize_lp(a0, deadline, max_iters=200):
    """LP-based iterative refinement."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = normalize_to(a)

    best = a.copy()
    best_score = compute_obj(a)
    S_target = np.sqrt(2.0 * n)

    c = np.convolve(a, a)
    M = float(c.max())

    for it in range(max_iters):
        if time.monotonic() > deadline - 10:
            break

        g = lp_solve(a, M)
        if g is None:
            break

        S_g = g.sum()
        if S_g < 1e-6:
            break

        # Normalize g to target sum
        g_norm = g * (S_target / S_g)

        improved = False
        for t in [0.5, 0.3, 0.7, 0.2, 0.1]:
            a_new = (1.0 - t) * a + t * g_norm
            a_new = np.maximum(a_new, 0.0)
            S_new = a_new.sum()
            if S_new < 1e-6:
                continue
            a_new = a_new * (S_target / S_new)
            s = compute_obj(a_new)
            if s < best_score:
                a = a_new
                best = a.copy()
                best_score = s
                improved = True
                break

        if not improved:
            # Try pure LP solution
            s_g = compute_obj(g_norm)
            if s_g < best_score:
                a = g_norm
                best = a.copy()
                best_score = s_g
            else:
                # Small gradient escape
                a_pert = a * (1.0 + 0.02 * np.random.default_rng(it * 31).standard_normal(n))
                a_pert = np.maximum(a_pert, 0)
                a_pert = normalize_to(a_pert)
                s_pert = compute_obj(a_pert)
                if s_pert < best_score * 1.02:
                    a = a_pert

        c = np.convolve(a, a)
        M = float(c.max())

    return best, best_score

def generate_structured_starts(n, rng):
    """Generate structured starting sequences."""
    starts = []
    i = np.arange(n, dtype=np.float64)

    # Uniform
    starts.append(np.ones(n))

    # Linear ramp
    starts.append(np.linspace(0.1, 1.0, n))

    # Reverse ramp
    starts.append(np.linspace(1.0, 0.1, n))

    # Cosine (single period)
    starts.append(0.5 + 0.5 * np.cos(2 * np.pi * i / n))

    # Two bumps
    for pos1, pos2 in [(0.2, 0.8), (0.1, 0.9), (0.25, 0.75)]:
        b = np.exp(-((i - n*pos1)/(n*0.1))**2) + np.exp(-((i - n*pos2)/(n*0.1))**2) + 0.01
        starts.append(b)

    # Gapped uniform (zero out middle)
    for gap_frac in [0.2, 0.3]:
        gap_start = int(n * (0.5 - gap_frac/2))
        gap_end = int(n * (0.5 + gap_frac/2))
        a = np.ones(n)
        a[gap_start:gap_end] = 0.0
        starts.append(a)

    # Alternating high-low
    starts.append(np.where(np.arange(n) % 2 == 0, 1.0, 0.2))
    starts.append(np.where(np.arange(n) % 3 == 0, 1.0, 0.1))

    # Random
    starts.append(rng.random(n) + 0.1)
    starts.append(rng.random(n) * 0.5 + 0.25)

    # Chirp-like
    starts.append(0.5 + 0.5 * np.cos(2 * np.pi * i * 3.0 / n))
    starts.append(0.5 + 0.5 * np.cos(2 * np.pi * i * 5.0 / n))

    # Triangular
    starts.append(1.0 - 2.0 * np.abs(i / n - 0.5))

    # Gaussian
    starts.append(np.exp(-0.5 * ((i - n/2) / (n * 0.3))**2))

    return starts
