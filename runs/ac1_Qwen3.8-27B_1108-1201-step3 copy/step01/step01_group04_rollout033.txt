```python
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a combination of:
1. Subgradient descent with momentum (Adam-like) on the peak autocorrelation
2. SCA (Successive Convex Approximation) via LP
3. Frequency-domain spectral flattening
4. Multi-start with diverse initializations
"""

import numpy as np
import time
from scipy.optimize import linprog

def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(np.max(c))
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def normalize(a, target=None):
    if target is None:
        target = 1.0
    S = a.sum()
    if S < 1e-12:
        return a
    return a * (target / S)

def subgradient_descent_step(a, lr, momentum=0.9):
    """One step of subgradient descent on the peak autocorrelation objective."""
    n = len(a)
    c = np.convolve(a, a)
    S = a.sum()
    
    # Find peak (use all peaks with small tolerance for robustness)
    max_c = c.max()
    # Soft peak selection
    tau = 10.0
    w = np.exp(tau * (c - max_c) / max(max_c, 1e-12))
    w /= w.sum()
    
    # Gradient of max_c w.r.t. a[j] = sum_k w[k] * (a[k-j] + a[j-k])
    # where a[i] = 0 for i outside [0, n-1]
    grad_M = np.zeros(n)
    for k in range(2*n - 1):
        if w[k] < 1e-10:
            continue
        # a[j] appears in conv[k] as a[j]*a[k-j] and a[k-j]*a[j]
        # So d conv[k] / d a[j] = a[k-j] + a[j-k] (with a[i]=0 for invalid i)
        for j in range(n):
            idx1 = k - j
            idx2 = j - k  # = -(k-j)
            term = 0.0
            if 0 <= idx1 < n:
                term += a[idx1]
            if 0 <= idx2 < n:
                term += a[idx2]
            grad_M[j] += w[k] * term
    
    # Full gradient of f = 2n*M/S^2
    # df/da[j] = 2n * (grad_M[j] * S - 2*M) / S^3
    grad_f = 2.0 * n * (grad_M * S - 2.0 * max_c) / (S ** 3)
    
    # Update
    a = a - lr * grad_f
    a = np.maximum(a, 0.0)
    a = normalize(a, 1.0)
    return a
```

```python
    """Compute grad_M = d(M)/d(a) using the formula grad_M[m] = 2*sum_k w[k]*a[k-m]*I_k(m).
    Uses top-k k values for efficiency."""
    n = len(a)
    grad = np.zeros(n)
    # Get top k values
    idx = np.argsort(w)[::-1][:top_k]
    for k in idx:
        wk = w[k]
        if wk < 1e-10:
            break
        # Valid m range: m in [max(0, k-n+1), min(n-1, k)]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[m] += 2 * wk * a[k-m] for m in [lo, hi]
        # Let i = k-m, then i ranges from k-hi to k-lo
        # grad[k-i] += 2*wk*a[i]
        i_lo = k - hi
        i_hi = k - lo
        grad[lo:hi+1] += 2.0 * wk * a[i_lo:i_hi+1][::-1]
    return grad
```

```python
"""
Optimizes non-negative step function heights to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses:
1. Analytical subgradient descent with Adam optimizer
2. SCA (Successive Convex Approximation) via LP for local refinement
3. Multi-start with diverse initializations
4. Spectral flattening in frequency domain
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(np.max(c))
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def normalize(a, target=1.0):
    S = a.sum()
    if S < 1e-15:
        return a.copy()
    return a * (target / S)

def compute_soft_weights(c, tau=5.0):
    """Compute softmax weights over the convolution values."""
    max_c = c.max()
    if max_c < 1e-15:
        return np.ones(len(c)) / len(c)
    shifted = c / max_c - 1.0  # <= 0
    w = np.exp(tau * shifted)
    total = w.sum()
    if total < 1e-15:
        # Fallback to hard max
        w = np.zeros(len(c))
        w[c == max_c] = 1.0 / np.sum(c == max_c)
        return w
    return w / total

def peak_gradient(a, w, top_k=50):
    """Compute d(max_c)/d(a) analytically.
    Formula: d(conv[k])/d(a[m]) = 2*a[k-m] if max(0,k-n+1)<=m<=min(n-1,k) else 0
    So grad[m] = 2*sum_k w[k]*a[k-m]*I_k(m)
    """
    n = len(a)
    grad = np.zeros(n)
    # Sort by weight, take top_k
    idx = np.argsort(w)[::-1][:top_k]
    for k in idx:
        wk = w[k]
        if wk < 1e-12:
            break
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        # grad[lo:hi+1] += 2*wk*a[k-hi : k-lo+1][::-1]
        grad[lo:hi+1] += 2.0 * wk * a[k-hi:k-lo+1][::-1]
    return grad

def adam_update(state, grad, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step. state = (m, v, t)."""
    m, v, t = state
    t_new = t + 1
    m_new = beta1 * m + (1 - beta1) * grad
    v_new = beta2 * v + (1 - beta2) * grad * grad
    m_hat = m_new / (1 - beta1 ** t_new)
    v_hat = v_new / (1 - beta2 ** t_new)
    update = lr * m_hat / (np.sqrt(v_hat) + eps)
    return (m_new, v_new, t_new), update

def subgradient_descent(a0, n_iters=5000, lr=0.01, tau_anneal=(2.0, 20.0), deadline=None, verbose=False):
    """Subgradient descent with Adam on the peak autocorrelation objective."""
    n = len(a0)
    a = normalize(a0.copy(), 1.0)
    best = a.copy()
    best_score = compute_obj(best)
    
    state = (np.zeros(n), np.zeros(n), 0)
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = np.convolve(a, a)
        max_c = c.max()
        
        # Anneal tau
        if tau_anneal is not None:
            frac = it / max(n_iters - 1, 1)
            tau = tau_anneal[0] + (tau_anneal[1] - tau_anneal[0]) * frac
        else:
            tau = 5.0
        
        w = compute_soft_weights(c, tau)
        grad_M = peak_gradient(a, w)
        
        # Full gradient of f = 2n*M/S^2, with S=1 (normalized)
        # f = 2n*M, grad_f = 2n*grad_M (since S=1)
        # But we also need to account for the normalization constraint.
        # Actually, since we normalize after each step, we just minimize M.
        grad = grad_M
        
        # Adam step
        state, update = adam_update(state, grad, lr)
        a = a - update
        a = np.maximum(a, 0.0)
        a = normalize(a, 1.0)
        
        # Check improvement every 100 steps
        if (it + 1) % 100 == 0:
            score = compute_obj(a)
            if score < best_score:
                best = a.copy()
                best_score = score
        if verbose and (it + 1) % 500 == 0:
            score = compute_obj(a)
            print(f"    SGD iter {it+1}: score={score:.6f}, best={best_score:.6f}")
    
    # Final check
    score = compute_obj(a)
    if score < best_score:
        best = a.copy()
        best_score = score
    
    return best, best_score

def sca_lp(a0, max_iters=50, deadline=None, verbose=False):
    """SCA: iteratively solve LP to minimize linearized peak."""
    n = len(a0)
    a = normalize(a0.copy(), 1.0)
    best = a.copy()
    best_score = compute_obj(best)
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Compute current convolution
        C = np.convolve(a, a)
        M = C.max()
        
        # Build LP:
        # minimize t
        # subject to: 2*conv(a, a_new)[k] - C[k] <= t for all k
        #              sum(a_new) = 1
        #              a_new >= 0
        
        # 2*conv(a, a_new)[k] = 2*sum_j a[k-j]*a_new[j] (valid j)
        # So constraint: 2*sum_j a[k-j]*a_new[j] - t <= C[k]
        
        # Variables: [a_new[0], ..., a_new[n-1], t]
        # Objective: minimize t → c_obj = [0,...,0, 1]
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        # Inequality constraints: A_ub @ x <= b_ub
        # For each k: 2*sum_j A[k,j]*a_new[j] - t <= C[k]
        # A_ub has 2n-1 rows
        n_constraints = 2 * n - 1
        A_ub = np.zeros((n_constraints, n + 1))
        b_ub = np.zeros(n_constraints)
        
        for k in range(n_constraints):
            b_ub[k] = C[k]
            A_ub[k, -1] = -1.0  # coefficient of t
            # 2*sum_j a[k-j]*a_new[j]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                A_ub[k, j] = 2.0 * a[k - j]
        
        # Equality: sum(a_new) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        bounds = [(0, None)] * n + [(None, None)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'presolve': True})
            if res.status == 0:
                a_new = res.x[:n]
                a_new = np.maximum(a_new, 0.0)
                a_new = normalize(a_new, 1.0)
                score = compute_obj(a_new)
                if score < best_score:
                    best = a_new.copy()
                    best_score = score
                a = a_new
            else:
                break
        except Exception:
            break
        
        if verbose and (it + 1) % 5 == 0:
            print(f"    SCA iter {it+1}: score={compute_obj(a):.6f}, best={best_score:.6f}")
    
    return best, best_score

def spectral_flatten(a0, n_iters=100, beta=2.0, deadline=None):
    """Frequency domain spectral flattening."""
    n = len(a0)
    a = normalize(a0.copy(), 1.0)
    best = a.copy()
    best_score = compute_obj(best)
    
    # Use FFT length >= 2n-1
    N = 1
    while N < 2 * n - 1:
        N *= 2
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # FFT
        A = np.fft.fft(a, N)
        P = np.abs(A) ** 2
        
        # Compute autocorrelation from power spectrum
        R = np.fft.ifft(P).real
        R = R[:2*n-1]  # Take the aperiodic part
        
        # Find peak
        max_R = R.max()
        mean_R = R.mean()
        
        # Modify spectrum to flatten autocorrelation
        # Reduce power at frequencies that contribute to the peak
        # Simple approach: P_new = P^alpha (with alpha < 1 to flatten)
        alpha = 1.0 - 0.1 * (it / n_iters)  # Start close to 1, gradually flatten
        # Actually, use a more targeted approach:
        # Reduce the peak by applying a transfer function
        
        # Alternative: multiplicative update
        # P_new = P * exp(-beta * (R - mean_R)^2 / max_R)
        # This reduces power where the autocorrelation is large
        
        R_shifted = R - R.mean()
        # Reshape R to be compatible with P (N points)
        R_full = np.zeros(N)
        R_full[:2*n-1] = R_shifted
        # Also add the wrapped part
        R_full[N-(n-1):] += R_shifted[1:n]  # approximate
        
        # Actually, let me use a simpler approach:
        # Just modify the spectrum to be more flat
        P_target = np.full(N, P.mean())
        P_new = P * (P_target / (P + 1e-10)) ** 0.1  # Gentle flattening
        
        # Or: P_new = P ** 0.9 (slight compression)
        P_new = P ** 0.95
        
        # Update A with new magnitude, keep phase
        A_new = A * (np.sqrt(P_new / (P + 1e-10)) / (np.abs(A) + 1e-10) * np.abs(A))
        # Simpler: A_new = A * sqrt(P_new/P)
        ratio = np.sqrt(P_new / (P + 1e-20))
        A_new = A * ratio
        
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, 1.0)
        
        score = compute_obj(a_new)
        if score < best_score:
            best = a_new.copy()
            best_score = score
        a = a_new
    
    return best, best_score

def combined_optimize(a0, deadline, verbose=False):
    """Run combined optimization: SGD then SCA then final SGD."""
    n = len(a0)
    a = normalize(a0.copy(), 1.0)
    best = a.copy()
    best_score = compute_obj(best)
    
    if verbose:
        print(f"  Starting: n={n}, score={best_score:.6f}")
    
    # Phase 1: Subgradient descent with Adam
    t1 = time.monotonic()
    remaining = deadline - t1

    if score < best_score:
        best = a.copy()
        best_score = score
    if verbose:

    # Phase 2: SCA LP refinement
    t2 = time.monotonic()
    a, score = sca_lp(a, max_iters=30, deadline=deadline - 30, verbose=verbose)

        best_score = score
```

```python
"""
Optimizes non-negative step function heights to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses analytical subgradient descent with Adam, SCA (Successive Convex Approximation) via LP,
spectral methods, and multi-start exploration. Key insight: the gradient of the peak
autocorrelation w.r.t. a[m] is 2*sum_k w[k]*a[k-m]*I_k(m), enabling efficient computation.
"""

import numpy as np
import time
from scipy.optimize import linprog

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(np.max(c))
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def normalize(a, target=1.0):
    """Normalize sequence to have a given sum."""
    S = a.sum()
    if S < 1e-15:
        return a.copy()
    return a * (target / S)

def compute_soft_weights(c, tau=5.0):
    """Compute softmax weights over convolution values for subgradient."""
    max_c = c.max()
    if max_c < 1e-15:
        return np.ones(len(c)) / len(c)
    shifted = c / max_c - 1.0
    w = np.exp(tau * shifted)
    total = w.sum()
    if total < 1e-15:
        w = np.zeros(len(c))
        peaks = (c >= max_c - 1e-12)
        w[peaks] = 1.0 / peaks.sum()
        return w
    return w / total

def peak_gradient(a, w, top_k=100):
    """Compute d(soft_max)/d(a) analytically.
    Formula: d(conv[k])/d(a[m]) = 2*a[k-m] if max(0,k-n+1)<=m<=min(n-1,k) else 0
    """
    n = len(a)
    grad = np.zeros(n)
    idx = np.argsort(w)[::-1][:top_k]
    for k in idx:
        wk = w[k]
        if wk < 1e-12:
            break
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        grad[lo:hi+1] += 2.0 * wk * a[k-hi:k-lo+1][::-1]
    return grad

def adam_update(state, grad, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step."""
    m, v, t = state
    t_new = t + 1
    m_new = beta1 * m + (1.0 - beta1) * grad
    v_new = beta2 * v + (1.0 - beta2) * grad * grad
    m_hat = m_new / (1.0 - beta1 ** t_new)
    v_hat = v_new / (1.0 - beta2 ** t_new)
    update = lr * m_hat / (np.sqrt(v_hat) + eps)
    return (m_new, v_new, t_new), update

def subgradient_descent(a0, n_iters=5000, lr=0.01, tau_start=3.0, tau_end=30.0,
                        deadline=None, verbose=False, check_interval=200):
    """Subgradient descent with Adam on the peak autocorrelation."""
    n = len(a0)
    a = normalize(a0.copy(), 1.0)
    best = a.copy()
    best_score = compute_obj(best)
    state = (np.zeros(n), np.zeros(n), 0)
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = np.convolve(a, a)
        max_c = c.max()
        frac = it / max(n_iters - 1, 1)
        tau = tau_start + (tau_end - tau_start) * frac
        w = compute_soft_weights(c, tau)
        grad = peak_gradient(a, w)
        
        # Anneal learning rate
        lr_eff = lr * (1.0 + np.cos(np.pi * frac)) * 0.5
        lr_eff = max(lr_eff, lr * 0.05)
        
        state, update = adam_update(state, grad, lr_eff)
        a = a - update
        a = np.maximum(a, 0.0)
        a = normalize(a, 1.0)
        
        if (it + 1) % check_interval == 0:
            score = compute_obj(a)
            if score < best_score:
                best = a.copy()
                best_score = score
            if verbose:
                print(f"    SGD iter {it+1}: current={score:.6f}, best={best_score:.6f}")
    
    score = compute_obj(a)
    if score < best_score:
        best = a.copy()
        best_score = score
    return best, best_score

def sca_lp_optimize(a0, max_iters=50, deadline=None, verbose=False):
    """SCA: iteratively solve LP minimizing linearized peak autocorrelation."""
    n = len(a0)
    a = normalize(a0.copy(), 1.0)
    best = a.copy()
    best_score = compute_obj(best)
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        C = np.convolve(a, a)
        M = C.max()
        
        # LP: min t s.t. 2*conv(a,a_new)[k] - C[k] <= t, sum(a_new)=1, a_new>=0
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        n_constraints = 2 * n - 1
        A_ub = np.zeros((n_constraints, n + 1))
        b_ub = np.zeros(n_constraints)
        
        for k in range(n_constraints):
            b_ub[k] = C[k]
            A_ub[k, -1] = -1.0
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                A_ub[k, j] = 2.0 * a[k - j]
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': max(1.0, (deadline - time.monotonic()) * 0.3 if deadline else 30.0)})
            if res.status == 0:
                a_new = np.maximum(res.x[:n], 0.0)
                a_new = normalize(a_new, 1.0)
                score = compute_obj(a_new)
                if score < best_score:
                    best = a_new.copy()
                    best_score = score
                a = a_new
            else:
                break
        except Exception:
            break
        
        if verbose and (it + 1) % 10 == 0:
            print(f"    SCA iter {it+1}: score={compute_obj(a):.6f}, best={best_score:.6f}")
    
    return best, best_score

def spectral_optimize(a0, n_iters=200, deadline=None, verbose=False):
    """Frequency-domain optimization: flatten power spectrum to reduce autocorrelation peak."""
    n = len(a0)
    a = normalize(a0.copy(), 1.0)
    best = a.copy()
    best_score = compute_obj(best)
    
    N = 1
    while N < 2 * n - 1:
        N *= 2
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        A = np.fft.fft(a, N)
        P = np.abs(A) ** 2
        
        # Compute aperiodic autocorrelation from spectrum
        R = np.fft.ifft(P).real[:2*n-1]
        
        # Target: make R more flat. Use iterative reweighting.
        max_R = R.max()
        if max_R < 1e-15:
            break
        
        # Reduce power at frequencies contributing most to the peak
        # The peak is at lag 0 of R (which is R[0] = sum a^2) or at center
        # Actually R[0] = sum(a^2), R[n-1] = sum(a[j]*a[n-1-j])
        # We want to reduce the overall peak of R
        
        # Apply spectral shaping: compress the dynamic range of P
        frac = it / max(n_iters - 1, 1)
        alpha = 1.0 - 0.3 * frac  # Start at 1.0 (no change), go to 0.7 (more compression)
        
        P_flat = P ** alpha
        P_flat /= P_flat.sum() * N  # Normalize to preserve total energy
        
        # Blend with original to avoid too aggressive changes
        blend = 0.3 + 0.4 * frac
        P_new = (1.0 - blend) * P + blend * P_flat * (P.sum() / (P_flat.sum() + 1e-15))
        
        # Update amplitude, preserve phase
        phase = np.angle(A)
        mag_new = np.sqrt(P_new)
        A_new = mag_new * np.exp(1j * phase)
        
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, 1.0)
        
        score = compute_obj(a_new)
        if score < best_score:
            best = a_new.copy()
            best_score = score
        a = a_new
    
    return best, best_score
```
