Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. Nesterov-accelerated LP alternating (main workhorse)
2. FFT-accelerated LogSumExp mirror descent (efficient gradient-based refinement)
3. Targeted coordinate descent (final micro-optimization)

Key innovations:
- Multiplicative (mirror) updates that respect non-negativity naturally
- FFT-based gradient computation in O(n log n)
- Structured exploration (mixing with sinusoidal patterns)
- Adaptive T-schedule for LogSumExp smoothing
- Pre-computed convolution index structures for fast LP matrix builds
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation (scale-invariant)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def compute_lse_grad(a, T):
    """
    Compute gradient of (2n/T) * logsumexp(T * conv(a,a)) wrt a.
    Returns (gradient, objective_value).
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp(T * (b - max_b))
    w = exp_b / exp_b.sum()
    obj = (2.0 * n / T) * (max_b + np.log(exp_b.sum()) / T)
    
    # grad_m = 4n * sum_j w_{m+j} * a_j
    # = 4n * (cross-correlation of a and w at lag m)
    # Computed via: rev(a) * w, extract elements [n-1 : 2n-2]
    
    n_fft = 1 << (3 * n).bit_length()
    a_rev = a[::-1]
    A_rev = rfft(a_rev, n_fft)
    W = rfft(w, n_fft)
    corr = irfft(A_rev * W, n_fft)
    
    grad = np.zeros(n)
    grad[:] = corr[n-1:2*n-1]
    grad *= 4.0 * n
    
    return grad, obj


def project_simplex(v, target_sum=1.0):
    """Project vector v onto {x >= 0 : sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.zeros(n) + target_sum / n
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix_data(a):
    """
    Build sparse convolution matrix T where T[k,i] = a[k-i] if 0<=k-i<n.
    Returns (row_ind, col_ind, data) for csr_matrix construction.
    T has shape (2n-1, n).
    """
    n = len(a)
    size = 2 * n - 1
    # Pre-allocate
    row_ind = np.zeros(n * n, dtype=np.int32)
    col_ind = np.zeros(n * n, dtype=np.int32)
    data = np.zeros(n * n, dtype=np.float64)
    
    idx = 0
    for i in range(n):
        # Column i: T[k, i] = a[k-i] for k = i, ..., i+n-1 (capped at 2n-2)
        k_start = i
        k_end = min(i + n, size)
        length = k_end - k_start
        row_ind[idx:idx+length] = range(k_start, k_end)
        col_ind[idx:idx+length] = i
        data[idx:idx+length] = a[:length]
        idx += length
    
    return row_ind[:idx], col_ind[:idx], data[:idx]

    for solver in [cp.ECOS, cp.CLARABEL, cp.SCS, cp.PDLP]:

    _lp_solver = cp.ECOS

def solve_lp_best_response(a, solver):
    """Given fixed a (sum=1), find b minimizing max conv(a,b)."""
    n = len(a)
    size = 2 * n - 1
    row_ind, col_ind, data = build_conv_matrix_data(a)
    T = csr_matrix((data, (row_ind, col_ind)), shape=(size, n))
    
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [T @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            for alt in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
                if alt == solver:
                    continue
                try:
                    prob.solve(solver=alt, verbose=False)
                    if prob.status in ('optimal', 'optimal_inaccurate'):
                        break
                except:
                    continue
    except:
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val = b_val / s
    return b_val


def nesterov_lp_refine(a0, time_limit, solver, max_iters=100000):
    """Nesterov LP alternating with structured exploration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    stall_threshold = 15
    rng = np.random.default_rng(42)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        # Nesterov step
        alpha = 2.0 / (np.sqrt(it + 1) + 1.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        # Weighted average update for a
        weight = alpha * (it + 2)
        a = ((it + 1) * a + weight * b_new) / (it + 1 + weight)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
            sa = 1.0
        a = a / sa
        
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        
        if (it + 1) % 25 == 0:
            stall += 1 if best_score > 2.0 * n * fast_convolve(y).max() - 1e-8 else 0
            # Actually let me track stall differently
            pass
        
        if (it + 1) % 30 == 0:
            if stall >= stall_threshold:
                # Structured exploration
                choice = rng.integers(0, 4)
                x = np.linspace(0, 1, n)
                if choice == 0:
                    # Mix with cosine
                    pattern = np.maximum(np.cos(np.pi * x * rng.integers(1, 5)), 0)
                elif choice == 1:
                    # Mix with sine squared
                    pattern = np.sin(np.pi * x * rng.integers(1, 4)) ** 2
                elif choice == 2:
                    # Mix with Gaussian bumps
                    centers = rng.uniform(0.1, 0.9, 2)
                    pattern = sum(np.exp(-20*(x-c)**2) for c in centers)
                else:
                    # Random perturbation
                    pattern = np.abs(rng.standard_normal(n))
                pattern = np.maximum(pattern, 0)
                ps = pattern.sum()
                if ps > 1e-15:
                    pattern /= ps
                mix = rng.uniform(0.2, 0.5)
                y = (1 - mix) * y + mix * pattern
                y = np.maximum(y, 0)
                y = y / y.sum()
                a = y.copy()
                stall = 0
                stall_threshold = min(40, stall_threshold + 5)
        
        if (it + 1) % 1000 == 0:
            print(f"    LP iter={it+1}: score={best_score:.8f}")
    
    return best_a, best_score
