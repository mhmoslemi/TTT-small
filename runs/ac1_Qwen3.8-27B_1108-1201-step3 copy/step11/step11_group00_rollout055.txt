```
def project_simplex(v):
    n = len(v)
    u = sorted(v, reverse=True)
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)
```

This is O(n log n) per iteration, which is fine for n=1600.

```python
def fista_lse(a0, time_limit, T_start=100, T_end=1, n_stages=20):
    """FISTA with LogSumExp smoothing for the convex problem."""
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_score = 2*n*fast_convolve(a).max()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    t0 = time.monotonic()
    
    for stage_T in temps:
        # Reset momentum for each stage
        y = a.copy()
        t = 1.0
        
        # Compute initial gradient for step size
        b = fast_convolve(y)
        Tb = stage_T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        g = compute_grad(y, w)
        
        # Lipschitz constant estimate (rough)
        # f is L-smooth where L ≈ 2*T*max(a) (rough bound)
        # Better: use the actual gradient norm
        L = 2.0 * stage_T * y.max() + 1e-10
        lr = 1.0 / L
        
        max_iters = 200  # per stage
        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_score
            
            # Compute gradient at y
            b = fast_convolve(y)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            g = compute_grad(y, w)
            
            # FISTA update
            a_new = y - lr * g
            a_new = project_simplex(a_new)
            
            t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
            y = a_new + ((t-1)/t_new) * (a_new - a)
            y = np.maximum(y, 0)
            if y.sum() > 0:
                y = y / y.sum()  # soft projection
            
            a = a_new
            t = t_new
            
            # Track best
            score = 2*n*fast_convolve(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
    
    return best_a, best_score
```

```python
"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. FFT-based FISTA with LogSumExp smoothing (primary fast method)
2. L-BFGS quasi-Newton refinement
3. Multi-length exploration via interpolation
4. Peak-shaving coordinate descent for final polish
Key: efficient O(n log n) gradients via FFT, adaptive temperature,
Nesterov momentum, and careful time management.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_grad(w, a):
    """
    Compute g[m] = sum_k w[k]*a[k-m] for m=0..n-1.
    Uses the identity: g = conv(a[::-1], w)[n-1 : n-1+n].
    """
    n = len(a)
    a_rev = a[::-1]
    conv_len = n + len(w) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2
    A_rev = rfft(a_rev, fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    return conv[n - 1:n - 1 + n]


def lse_obj_and_grad(a, T):
    """
    Compute LogSumExp objective value and gradient w.r.t. a.
    f(a) = (1/T) * log(sum_k exp(T * c_k)) where c = conv(a,a).
    grad[m] = sum_k w[k] * a[k-m] where w[k] = exp(T*c_k)/sum(exp(T*c)).
    """
    n = len(a)
    c = fast_convolve(a)
    Tc = T * c
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    s = exp_Tc.sum()
    if s < 1e-300:
        w = np.ones(len(c)) / len(c)
        obj = c.max()
    else:
        w = exp_Tc / s
        obj = (max_Tc + np.log(s)) / T
    grad = compute_grad(w, a)
    return obj, grad, w


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fista_lse(a0, time_limit, T_start=50.0, T_end=0.5, n_stages=15, max_iters_per_stage=500):
    """
    FISTA with LogSumExp smoothing and adaptive temperature.
    Minimizes max_k (a*a)_k subject to a in simplex.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_lse = 2.0 * n * fast_convolve(a).max()
    best_raw = best_lse  # track actual max (not LSE)

    temps = np.geomspace(T_start, T_end, n_stages)
    t_global = time.monotonic()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t_global > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t_global)
        if remaining < 2:
            break

        # Reset FISTA state per stage
        y = a.copy()
        momentum_t = 1.0

        # Estimate Lipschitz constant: L ~ 2*T*max(a) is rough
        # Use a safer initial step
        L = 4.0 * stage_T * (a.max() + 1e-10)
        lr = 1.0 / L

        iter_count = 0
        stage_t0 = time.monotonic()

        while iter_count < max_iters_per_stage:
            if time.monotonic() - t_global > time_limit:
                return best_a, best_raw
            if time.monotonic() - stage_t0 > max(1.0, remaining * 0.8 / (n_stages - si)):
                break

            # Compute gradient at y
            obj, grad, w = lse_obj_and_grad(y, stage_T)

            # Backtracking: reduce lr if function increases too much
            a_trial = y - lr * grad
            a_trial = project_simplex(a_trial)
            obj_trial, _, _ = lse_obj_and_grad(a_trial, stage_T)

            iters_bt = 0
            while obj_trial > obj + 0.5 * lr * np.dot(grad, a_trial - y) + 1e-12 and iters_bt < 10:
                lr *= 0.5
                a_trial = y - lr * grad
                a_trial = project_simplex(a_trial)
                obj_trial, _, _ = lse_obj_and_grad(a_trial, stage_T)
                iters_bt += 1

            if obj_trial <= obj + 1e-10:
                a_new = a_trial
                # Increase lr slightly if no backtracking needed
                if iters_bt == 0 and iteration_count_safe():
                    lr = min(lr * 1.05, 2.0 / L)
            else:
                a_new = a.copy()

            # FISTA momentum update
            momentum_t_new = (1.0 + np.sqrt(1.0 + 4.0 * momentum_t * momentum_t)) / 2.0
            y = a_new + ((momentum_t - 1.0) / momentum_t_new) * (a_new - a)
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy
            a = a_new
            momentum_t = momentum_t_new
            iter_count += 1

            # Track best using actual max (not LSE)
            if iter_count % 5 == 0 or iter_count == 1:
                raw_score = 2.0 * n * fast_convolve(a).max()
                if raw_score < best_raw:
                    best_raw = raw_score
                    best_a = a.copy()

    return best_a, best_raw


def iteration_count_safe():
    """Helper to allow lr increase (always True, for clarity)."""
    return True


def lbfgs_lse(a0, time_limit, T_start=20.0, T_end=0.3, n_stages=10):
    """L-BFGS refinement using LogSumExp smoothing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    z = np.log(np.maximum(a, 1e-30))
    best_a = a.copy()
    best_raw = 2.0 * n * fast_convolve(a).max()

    temps = np.geomspace(T_start, T_end, n_stages)
    t0 = time.monotonic()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func(z_in):
            a_in = softmax_vec(z_in)
            val, _, _ = lse_obj_and_grad(a_in, stage_T)
            return val

        def jac(z_in):
            a_in = softmax_vec(z_in)
            _, grad_a, _ = lse_obj_and_grad(a_in, stage_T)
            # Transform gradient to z-space
            S = np.dot(grad_a, a_in)
            grad_z = a_in * (grad_a - S)
            return grad_z

        try:
            maxiter = max(20, min(100, int(remaining * 0.7 / (n_stages - si) / 0.5)))
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxfun': maxiter * 500,
                         'disp': False}
            )
            z = np.clip(res.x, -50, 50)
            a_new = softmax_vec(z)
            score = 2.0 * n * fast_convolve(a_new).max()
            if score < best_raw:
                best_raw = score
                best_a = a_new.copy()
            a = a_new
        except Exception:
            pass

    return best_a, best_raw


def softmax_vec(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def peak_shaving(a0, time_limit, seed=123):

    Coordinate-descent peak shaving: directly reduce top autocorrelation peaks

        c = fast_convolve(a)
        top_k = np.argsort(c)[::-1][:10]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            if contrib.max() < 1e-30:
                continue

            src_local = np.argmax(contrib)
            src = idx[src_local]

            # Find peak areas
            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in peak_areas and j != src]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            candidates.sort(key=lambda j: a[j])
            dst_pool = candidates[:max(1, len(candidates) // 4)]
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04, 0.06]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturb
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 300:
                break

    return best_a, best_score
```
