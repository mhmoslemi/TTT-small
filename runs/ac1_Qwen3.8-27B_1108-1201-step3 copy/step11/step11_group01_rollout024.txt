4. **Better idea: Use the "active set" structure**

At the optimum, only a few positions in the convolution are at the maximum (the "active" positions). If we can identify these, we can solve a simpler problem.

For a given set of active positions A, the KKT conditions give:
- At the optimum, (a*a)[k] = M for k in A
- (a*a)[k] <= M for k not in A
- The gradient of (a*a)[k] for k in A should be non-negative (since a >= 0 is a constraint)

The gradient of (a*a)[k] with respect to a_j is:
d/d a_j [(a*a)[k]] = a_{k-j} (for valid indices)

So at the optimum, for each j:
sum_{k in A, k-j valid} a_{k-j} * λ_k + μ_j * (constraint for a_j >= 0) = ν (Lagrange multiplier for sum(a)=1)

where λ_k >= 0 are the multipliers for the binding constraints.

This is a system of equations that we can solve given the active set A.

**Algorithm:**
1. Start with a candidate active set A (e.g., the top M positions by convolution value)
2. Solve the KKT system for the given A
3. Check if the solution satisfies (a*a)[k] <= M for k not in A
4. If not, update A and repeat

        log_a = np.log(np.maximum(a, 1e-300))

        a_new = np.exp(log_a_new - log_a_new.max())

        a = a_new
        y = y_new
        
        # Check score
        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score

def project_simplex(v):
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0
    stagnation_count = 0
    prev_best = best_score

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            stagnation_count = 0
        else:
            sc_y = score_normalized(y)
            if sc_y < best_score:
                best_score = sc_y
                best_a = y.copy()
                stagnation_count = 0
            else:
                stagnation_count += 1

        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 5:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} M_lp={M_lp:.6f} stag={stagnation_count}")
                last_print = elapsed

    return best_a, best_score


def compute_subgradient_fft(y, active_indices):
    n = len(y)
    w = np.zeros(2 * n - 1)
    w[active_indices] = 1.0
    w /= w.sum()
    size = 1 << (3 * n - 2 - 1).bit_length()
    W = rfft(w, size)
    Y = rfft(y, size)
    corr = irfft(W * np.conj(Y), size)
    return corr[:n]


def mirror_nesterov(a0, time_limit, eta0=0.5, max_iters=10000000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0
    no_improve = 0

    t_val = 1.0
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_y = fast_convolve(y)
        max_b = b_y.max()

        thresh = max_b * 0.999
        active = np.where(b_y >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b_y))])

        if len(active) <= 50:
            g = np.zeros(n)
            for k in active:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                g[idx] += y[k - idx]
            g /= len(active)
        else:
            g = compute_subgradient_fft(y, active)

        eta = eta0 / np.sqrt(t_val + 1.0)

        a_old = a.copy()
        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - eta * g
        a_new = np.exp(log_a_new - log_a_new.max())
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 1e-300)
        a_new = a_new / a_new.sum()

        beta = (t_val - 1.0) / (t_val + 2.0)
        y_new = a_new + beta * (a_new - a_old)
        y_new = np.maximum(y_new, 1e-300)
        y_new = y_new / y_new.sum()

        a = a_new
        y = y_new
        t_val += 1.0

        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve = 0

            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    MirrorNest iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} eta={eta:.6f}")
        else:
            no_improve += 1

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 5:
                print(f"    MirrorNest iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} cur={sc:.8f} no_improve={no_improve}")
                last_print = elapsed

        if no_improve > 20000:
            a = best_a.copy()
            y = best_a.copy()
            no_improve = 0

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=50.0, T_end=0.01, n_stages=20):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = 50

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-25,
                         'gtol': 1e-18, 'maxfun': iters_per_stage * 300,
                         'disp': False}
            )
            if res.x is not None:
                z = res.x
                a_check = softmax_from_z(res.x)
                sc = score_normalized(a_check)
                if sc < best_score:
                    best_score = sc
                    best_a = a_check.copy()
        except Exception:
            pass

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-3 * x ** 2)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 4:
        a = np.ones(n) * (1.0 + 0.3 * np.cos(2 * np.pi * 5 * np.arange(n) / n))
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.15 * np.cos(7 * x)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * (1 + 0.3 * np.cos(4 * np.pi * x))
    else:
        a = rng.random(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def active_set_qp(a_init, active_positions, time_limit=30.0):
    n = len(a_init)
    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]
    size = 2 * n - 1
    for k in active_positions:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        q_a = a_var[idx]
        q_b = a_var[k - idx]
        qc = cp.sum(q_a * q_b)
        constraints.append(qc <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    a_var.value = a_init
    try:
        prob.solve(solver=cp.SCS, max_iters=10000, eps=1e-8)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        try:
            prob.solve(solver=cp.ECOS, max_iters=10000)
        except Exception:
            pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    a_val = np.asarray(a_var.value, dtype=np.float64)
    a_val = np.maximum(a_val, 0)
    s = a_val.sum()
    if s < 1e-15:
        return None
    return a_val / s


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        n_base = 1200
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    n_base = len(best_seq)

    # === Phase 1: Nesterov LP (35% of budget) ===
    remaining = deadline - time.monotonic()
    t1 = min(0.35 * (budget_s - 30), remaining * 0.6)
    print(f"\n=== Phase 1: Nesterov LP at n={n_base} (score={best_score:.8f}, budget={t1:.0f}s) ===")
    if t1 > 10:
        a_out, _ = nesterov_lp(best_seq, t1, solver=solver)
        try_update(a_out, "nesterov_lp_main")
    print(f"  Phase 1 done: best_score={best_score:.8f}")

    # === Phase 2: Mirror Nesterov (25% of budget) ===
    remaining = deadline - time.monotonic()
    t2 = min(0.25 * (budget_s - 30), remaining * 0.6)
    print(f"\n=== Phase 2: Mirror Nesterov (score={best_score:.8f}, budget={t2:.0f}s) ===")
    if t2 > 10:
        a_mn, _ = mirror_nesterov(best_seq, t2, eta0=0.3, max_iters=50000000)
        try_update(a_mn, "mirror_nesterov")
    print(f"  Phase 2 done: best_score={best_score:.8f}")

    # === Phase 3: L-BFGS log-sum-exp polish (15% of budget) ===
    remaining = deadline - time.monotonic()
    t3 = min(0.15 * (budget_s - 30), remaining * 0.6)
    print(f"\n=== Phase 3: LSE L-BFGS polish (score={best_score:.8f}, budget={t3:.0f}s) ===")
    if t3 > 10:
        a_lse, _ = lbgfs_logsumexp(best_seq, t3, T_start=50.0, T_end=0.01, n_stages=25)
        try_update(a_lse, "lse_polish")
    print(f"  Phase 3 done: best_score={best_score:.8f}")

    # === Phase 4: Second Nesterov LP round (15% of budget) ===
    remaining = deadline - time.monotonic()
    t4 = min(0.15 * (budget_s - 30), remaining * 0.6)
    print(f"\n=== Phase 4: Nesterov LP round 2 (score={best_score:.8f}, budget={t4:.0f}s) ===")
    if t4 > 10:
        a_out2, _ = nesterov_lp(best_seq, t4, solver=solver)
        try_update(a_out2, "nesterov_lp_r2")
    print(f"  Phase 4 done: best_score={best_score:.8f}")

    # === Phase 5: Multi-length exploration (8% of budget) ===
    remaining = deadline - time.monotonic()
    t5_total = min(0.08 * (budget_s - 30), remaining * 0.5)
    print(f"\n=== Phase 5: Length exploration (score={best_score:.8f}) ===")
    if t5_total > 20:
        n_options = [n_base - 200, n_base + 100, n_base + 300, n_base + 500]
        n_options = [n for n in n_options if n >= 100]
        t5_each = t5_total / max(1, len(n_options))
        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            t5 = min(t5_each, remaining * 0.4)
            if t5 < 5:
                break
            print(f"  Trying n={n_try}, budget={t5:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp(init_try, t5, solver=solver, max_iters=2000000)
            try_update(a_try, f"len_{n_try}")
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break
            t5b = min(t5 * 0.5, remaining * 0.3)
            if t5b > 5:
                a_mn2, _ = mirror_nesterov(a_try, t5b, eta0=0.2)
                try_update(a_mn2, f"mirror_len_{n_try}")
    print(f"  Phase 5 done: best_score={best_score:.8f}")

    # === Phase 6: Final mirror Nesterov + LSE polish (7% of budget) ===
    remaining = deadline - time.monotonic()
    t6_total = min(0.07 * (budget_s - 30), remaining * 0.7)
    print(f"\n=== Phase 6: Final polish (score={best_score:.8f}) ===")
    if t6_total > 10:
        t6a = t6_total * 0.6
        t6b = t6_total * 0.4
        a_final_mn, _ = mirror_nesterov(best_seq, t6a, eta0=0.1, max_iters=50000000)
        try_update(a_final_mn, "final_mirror")
        remaining = deadline - time.monotonic()
        t6b = min(t6b, remaining * 0.8)
        if t6b > 5:
            a_final_lse, _ = lbgfs_logsumexp(best_seq, t6b, T_start=20.0, T_end=0.005, n_stages=15)
            try_update(a_final_lse, "final_lse")
    print(f"  Phase 6 done: best_score={best_score:.8f}")

    # === Phase 7: Active set QP on best solution ===
    remaining = deadline - time.monotonic()
    if remaining > 5 and len(best_seq) <= 2000:
        print(f"\n=== Phase 7: Active set QP (score={best_score:.8f}) ===")
        b_best = fast_convolve(best_seq)
        max_b = b_best.max()
        active = np.where(b_best >= max_b * 0.95)[0]
        if len(active) <= 30:
            a_qp = active_set_qp(best_seq, active, time_limit=min(remaining * 0.5, 30.0))
            if a_qp is not None:
                try_update(a_qp, "active_qp")
                # Try with slightly larger active set
                active2 = np.where(b_best >= max_b * 0.9)[0]
                if len(active2) <= 50:
                    remaining = deadline - time.monotonic()
                    if remaining > 5:
                        a_qp2 = active_set_qp(best_seq, active2, time_limit=min(remaining * 0.5, 20.0))
                        if a_qp2 is not None:
                            try_update(a_qp2, "active_qp2")
    print(f"  Phase 7 done: best_score={best_score:.8f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
