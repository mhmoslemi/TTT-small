import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Full auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_ab(a, b):
    """Full convolution of a and b via FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence (scale-invariant version)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def simplex_proj(v, s=1.0):
    """Euclidean projection onto simplex {x: x>=0, sum(x)=s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * ind) > cssv
    if not np.any(cond):
        return np.full(n, s / n)
    rho = int(ind[cond][-1])
    theta = cssv[rho - 1] / rho
    return np.maximum(v - theta, 0.0)


def logsumexp_grad(a, T):
    """
    Compute LogSumExp smoothed objective value and gradient.
    Objective: (2n/T) * LogSumExp(conv(a,a)/T)
    Uses FFT for O(n log n) gradient computation.
    """
    n = len(a)
    b = fast_convolve(a)  # length 2n-1
    
    # Softmax weights
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T)
    w = exp_b / exp_b.sum()
    
    # Gradient: grad[i] = (4n/T) * sum_k w[k] * a[k-i]
    # = (4n/T) * (a * w_flip)[2n-2-i]
    # where w_flip = w[::-1]
    w_flip = w[::-1]
    conv_result = fast_convolve_ab(a, w_flip)  # length (2n-1) + n - 1 = 3n-2
    
    # grad[i] = (4n/T) * conv_result[2n-2-i] for i=0,...,n-1
    # Indices in conv_result: 2n-2, 2n-3, ..., n-1
    grad = (4.0 * n / T) * conv_result[n-1:2*n-1][::-1]
    
    # Objective value
    lse = max_b / T + np.log(exp_b.sum())
    obj = (2.0 * n / T) * lse
    
    return obj, grad


def logsumexp_lbgfs(a0, time_limit, T_stages=None):
    """L-BFGS on LogSumExp smoothed objective with adaptive temperature."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    t0 = time.monotonic()
    if T_stages is None:
        T_stages = [30.0, 20.0, 15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 1.5, 1.0, 0.6, 0.35, 0.2]
    
    best_a = a.copy()
    best_score = evaluate_local(a)
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit - 10:
            break
        
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 5:
            break
        
        # Determine max iterations based on time remaining
        maxiter = max(20, int(remaining * 50))  # ~50 iterations per second estimate
        
        def func(z):
            a_trial = np.maximum(z, 0)
            st = a_trial.sum()
            if st < 1e-15:
                return 1e15
            a_trial = a_trial / st
            obj, _ = logsumexp_grad(a_trial, T)
            return obj
        
        def jac(z):
            a_trial = np.maximum(z, 0)
            st = a_trial.sum()
            if st < 1e-15:
                return np.zeros_like(z)
            a_trial = a_trial / st
            _, grad = logsumexp_grad(a_trial, T)
            return grad
        
        try:
            res = minimize(fun=func, x0=a, jac=jac, method='L-BFGS-B',
                          options={'maxiter': maxiter, 'ftol': 1e-16,
                                   'gtol': 1e-10, 'maxfun': maxiter * 100,
                                   'disp': False})
            a = np.maximum(res.x, 0)
            sa = a.sum()
            if sa > 1e-15:
                a = a / sa
        except Exception as e:
            print(f"    LSE-LBFGS error at T={T}: {e}")
            break
        
        sc = evaluate_local(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"    LSE T={T:.2f} ({time.monotonic()-t0:.0f}s): NEW BEST {sc:.8f}")
        else:
            print(f"    LSE T={T:.2f} ({time.monotonic()-t0:.0f}s): score={sc:.8f}")
    
    return best_a, best_score

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Key improvements over baseline:
1. FFT-accelerated LogSumExp gradients (O(n log n) vs O(n^2))
2. Improved Nesterov LP alternating with FTRL averaging and adaptive exploration
3. Multiplicative weights (exponential formula) for entropy-based exploration
4. Systematic multi-n exploration with interpolation
5. Partitioned coordinate descent for final micro-optimization
6. Adaptive temperature annealing for LogSumExp L-BFGS

All helper functions are top-level (no closures/nesting).
"""

import numpy as np
import time
import math
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Full auto-convolution via FFT. Returns array of length 2*n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_ab(a, b):
    """Full convolution of a and b via FFT. Returns array of length len(a)+len(b)-1."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence score (scale-invariant)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def simplex_proj(v, s_val=1.0):
    """Euclidean projection onto simplex {x: x>=0, sum(x)=s_val}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s_val
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * ind) > cssv
    if not np.any(cond):
        return np.full(n, s_val / n)
    rho = int(ind[cond][-1])
    theta = cssv[rho - 1] / rho
    return np.maximum(v - theta, 0.0)


def logsumexp_obj_grad(a, T):
    """
    Compute LogSumExp smoothed objective and gradient using FFT.
    Objective: F(a) = (2n/T) * LogSumExp(conv(a,a) / T)
    Gradient: grad[i] = (4n/T) * sum_k w[k] * a[k-i]
    where w = softmax(conv(a,a)/T)
    
    The gradient is computed via: grad[i] = (4n/T) * (a * w_rev)[2n-2-i]
    where w_rev = w[::-1]
    """
    n = len(a)
    b = fast_convolve(a)  # length 2n-1

    max_b = b.max()
    exp_b = np.exp((b - max_b) / T)
    w = exp_b / exp_b.sum()

    # Compute cross-correlation via convolution with reversed w
    w_rev = w[::-1]  # length 2n-1
    conv_result = fast_convolve_ab(a, w_rev)  # length n + (2n-1) - 1 = 3n-2

    # grad[i] = (4n/T) * conv_result[2n-2-i] for i = 0, ..., n-1
    # conv_result indices: n-1, n, ..., 2n-2 (reversed)
    grad = (4.0 * n / T) * conv_result[n - 1:2 * n - 1][::-1]

    # Objective value
    lse = max_b / T + np.log(exp_b.sum())
    obj = (2.0 * n / T) * lse

    return obj, grad


def normalize_simplex(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    """Build sparse matrix T (2n-1 x n) such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    rows = []
    cols = []
    data = []
    for i in range(n):
        k_start = i
        k_end = min(i + n, size)
        length = k_end - k_start
        if length > 0:
            rows.extend(range(k_start, k_end))
            cols.extend([i] * length)
            data.extend(a[:length].tolist())
    rows_arr = np.array(rows, dtype=np.int32)
    cols_arr = np.array(cols, dtype=np.int32)
    data_arr = np.array(data, dtype=np.float64)
    return csr_matrix((data_arr, (rows_arr, cols_arr)), shape=(size, n))


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(a_fixed, T_sparse, solver):
    """Given fixed a (sum=1), find b minimizing max conv(a,b) s.t. b>=0, sum(b)=1."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [T_sparse @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)

    try:
        prob.solve(solver=solver)

                        break
                except Exception:
                    continue
    except Exception:
        return np.ones(n) / n

    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None or np.any(np.isnan(b_val)):
        return np.ones(n) / n
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return b_val / s


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=500000):
    """
    Nesterov LP alternating with FTRL averaging and adaptive exploration.
    The key iteration:
    1. Solve LP to get best response b to current y
    2. Update y via Nesterov extrapolation
    3. Update a via FTRL (follow-the-regularized-leader) averaging
    4. Periodic exploration to escape plateaus
    """
    n = len(a0)
    a = normalize_simplex(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    stall_threshold = 12
    prev_best = best_score
    rng = np.random.default_rng(99 + n % 1000)
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Build convolution matrix for current y
        T_sparse = build_conv_matrix(y)

        # Solve LP for best response
        b_new = solve_lp_best_response(y, T_sparse, solver)

        # Nesterov expiration step
        alpha = 2.0 / (np.sqrt(float(it + 1)) + 1.0)
        y_new = (1.0 - alpha) * y + alpha * b_new
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a.copy()
        else:
            y_new = y_new / sy
        y = y_new

        # FTRL-style weighted average
        weight_old = float(it + 1)
        weight_new = float(it + 2)
        a_new = (weight_old * a + weight_new * b_new) / (weight_old + weight_new)
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / sa
        a = a_new

        # Evaluate scores
        b_a = fast_convolve(a)
        score_a = 2.0 * n * b_a.max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0

        b_y = fast_convolve(y)
        score_y = 2.0 * n * b_y.max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0

        # Stall detection and exploration
        if (it + 1) % 30 == 0:
            if best_score >= prev_best - 1e-12:
                stall += 1
            else:
                stall = 0

                # Add random noise
                noise = rng.standard_normal(n) * magnitude
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                else:
                    y = a.copy()
                a = y.copy()
                stall = 0
                stall_threshold = min(40, stall_threshold + 3)

            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter={it+1} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def logsumexp_lbgfs(a0, time_limit, T_stages=None):
    """
    L-BFGS on LogSumExp smoothed objective with adaptive temperature annealing.
    Uses FFT-based gradient computation for O(n log n) per iteration.
    """
    n = len(a0)
    a = normalize_simplex(a0)
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [40.0, 25.0, 15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 1.8, 1.2, 0.8, 0.5, 0.3, 0.2, 0.1]

    best_a = a.copy()
    best_score = evaluate_local(a)

    for T in T_stages:
        if time.monotonic() - t0 > time_limit - 8:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        maxiter = max(30, int(remaining * 30))

        def func(z_inner):
            a_trial = np.maximum(z_inner, 0)
            st = a_trial.sum()
            if st < 1e-15:
                return 1e15
            a_trial = a_trial / st
            obj, _ = logsumexp_obj_grad(a_trial, T)
            return obj

        def jac(z_inner):
            a_trial = np.maximum(z_inner, 0)
            st = a_trial.sum()
            if st < 1e-15:
                return np.zeros_like(z_inner)
            a_trial = a_trial / st
            _, grad = logsumexp_obj_grad(a_trial, T)
            return grad

        try:
            res = minimize(fun=func, x0=a, jac=jac, method='L-BFGS-B',
                          options={'maxiter': maxiter, 'ftol': 1e-16,
                                   'gtol': 1e-10, 'maxfun': maxiter * 200,
                                   'disp': False})
            a = np.maximum(res.x, 0)
            sa = a.sum()
            if sa > 1e-15:
                a = a / sa
        except Exception:
            break

        sc = evaluate_local(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"    LSE T={T:.3f} ({time.monotonic()-t0:.0f}s): NEW BEST {sc:.8f}")
        else:
            print(f"    LSE T={T:.3f} ({time.monotonic()-t0:.0f}s): score={sc:.8f}")

    return best_a, best_score

def coordinate_descent_refine(a0, time_limit, n_steps=500000):
    """
    Partitioned coordinate descent for final polish.
    Groups coordinates by their contribution to the dominant peak
    and adjusts them in batches.
    """
    n = len(a0)
    a = normalize_simplex(a0)
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(314)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        peak_idx = int(np.argmax(b))

        # Find contributing indices
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        idx = np.arange(js, je + 1)
        if len(idx) == 0:
            idx = np.arange(n)

        contribs = a[idx] * a[peak_idx - idx]
        
        # Try reducing top contributors
        n_reduce = min(6, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:n_reduce]]

        improved = False
        for pos in top_positions:
            if a[pos] < 1e-14:
                continue
            for factor in [0.9995, 0.999, 0.998, 0.995, 0.99, 0.98]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] - a_trial[pos]
                if diff < 1e-17:
                    continue
                # Redistribute to other coordinates proportionally
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = normalize_simplex(a_trial)
                sc = evaluate_local(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 300:
                # Simultaneous small adjustment of multiple top contributors
                a_trial = a.copy()
                for pos in top_positions[:4]:
                    if a_trial[pos] > 1e-12:
                        a_trial[pos] *= 0.999
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = evaluate_local(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    stall = 0
                elif stall > 2000:
                    # Random restart
                    noise = rng.standard_normal(n) * 0.005
                    a = a + noise
                    a = normalize_simplex(a)
                    stall = 0
        else:
            stall = 0

        if (step + 1) % 50000 == 0:
            print(f"    CD ({time.monotonic()-t0:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate (or downsample) sequence to new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize_simplex(f)
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, np.append(f_norm, f_norm[0] * 0.1))
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n, endpoint=False)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-4.0 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-3.0 * (x - 0.5) ** 2 / 0.15)
    elif choice == 5:
        a = np.exp(-10 * (x - 0.3) ** 2) + np.exp(-10 * (x - 0.7) ** 2)
    elif choice == 6:
        a = np.exp(-6.0 * x) * (1 + 0.5 * np.sin(8 * np.pi * x))
        a = np.maximum(a, 0)
    else:
        # Polynomial with alternating coefficients
        a = 1.0 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(7 * np.pi * x)
        a = np.maximum(a, 0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: Nesterov LP alternating (main workhorse, ~40% time)
    Phase 2: LogSumExp L-BFGS with adaptive T (~25% time)
    Phase 3: Multi-n exploration (~15% time)
    Phase 4: Multiplicative weights (~10% time)
    Phase 5: Coordinate descent polish (~8% time)
    Phase 6: Final LP refinement (~2% time)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {budget_s:.0f}s")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Nesterov LP Alternating ===
    print(f"\n=== Phase 1: Nesterov LP (score={best_score:.8f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.40, 600.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=500000)
        try_update(a_lp, "LP_p1")

    # === Phase 2: LogSumExp L-BFGS with adaptive T ===
    print(f"\n=== Phase 2: LogSumExp L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages = [40.0, 25.0, 15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 1.8, 1.2, 0.8, 0.5, 0.3, 0.2, 0.15, 0.1]
        a_lse, _ = logsumexp_lbgfs(best_seq, t_alloc, T_stages=T_stages)
        try_update(a_lse, "LSE_BFGS")

    # === Phase 3: Multi-n Exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 30:
        n_best = len(best_seq)
        # Try several n values around the current best
        candidate_ns = []
        for delta in [-300, -150, 150, 300, 500]:
            n_try = n_best + delta
            if n_try >= 200 and n_try <= 5000:
                candidate_ns.append(n_try)
        # Also try very different sizes
        candidate_ns.extend([max(300, n_best // 2), min(5000, n_best * 2)])
        candidate_ns = sorted(set(candidate_ns))
        
        per_n = t_alloc / max(1, len(candidate_ns))
        print(f"  Trying n values: {candidate_ns}, ~{per_n:.0f}s each")
        for n_try in candidate_ns:
            if time.monotonic() > deadline - 20:
                break
            if n_try == n_best:
                continue
            a_interp = interpolate_sequence(best_seq, n_try)
            t_sub = min(per_n, deadline - time.monotonic() - 10)
            if t_sub < 15:
                continue
            print(f"  n={n_try}, budget={t_sub:.0f}s")
            a_sub, _ = nesterov_lp_refine(a_interp, t_sub, solver=solver, n_max_iters=100000)
            # Also try LSE on the result
            if time.monotonic() < deadline - 20 and t_sub > 30:
                a_sub2, _ = logsumexp_lbgfs(a_sub, min(30, deadline - time.monotonic() - 10),
                                            T_stages=[10.0, 5.0, 2.0, 1.0, 0.5, 0.3, 0.1])
                if evaluate_local(a_sub2) < evaluate_local(a_sub):
                    a_sub = a_sub2
            try_update(a_sub, f"multi_n_{n_try}")

    # === Phase 4: Multiplicative Weights ===
    print(f"\n=== Phase 4: Multiplicative Weights (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 180.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        # Run from current best
        a_mw, _ = multiplicative_weights_opt(best_seq, t_alloc, T0=15.0, T_min=0.3, n_stages=12)
        try_update(a_mw, "MW")
        # Also try from a different initial point
        if time.monotonic() < deadline - 20:
            a_init2 = structured_init(len(best_seq), rng)
            a_mw2, _ = multiplicative_weights_opt(a_init2, min(60, deadline - time.monotonic() - 15),
                                                  T0=20.0, T_min=0.5, n_stages=10)
            try_update(a_mw2, "MW_alt")

    # === Phase 5: Coordinate Descent Polish ===
    print(f"\n=== Phase 5: Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 120.0)
    if t_alloc > 10:
        a_cd, _ = coordinate_descent_refine(best_seq, t_alloc, n_steps=1000000)
        try_update(a_cd, "coord_desc")

    # === Phase 6: Final LP Refinement ===
    print(f"\n=== Phase 6: Final LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.02, 60.0)
    if t_alloc > 5:
        a_fin, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=50000)
        try_update(a_fin, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    # Safety: ensure valid output
    if best_seq is None:
        best_seq = np.ones(100) / 100

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
