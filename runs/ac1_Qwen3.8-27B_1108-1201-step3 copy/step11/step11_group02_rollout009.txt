    return best_a, best_score


def lse_func_wrap(z, params):
    """Top-level LSE objective for L-BFGS."""
    n, T, deadline, store = params
    if time.monotonic() > deadline:
        return 1e15
    a_val = softmax_fn(z)
    obj, _ = lse_obj_grad(a_val, T)
    store[0] = a_val.copy()
    store[1] = obj
    return obj


def lse_jac_wrap(z, params):
    """Top-level LSE gradient for L-BFGS."""
    n, T, deadline, store = params
    if time.monotonic() > deadline:
        return np.zeros(n)
    a_val = softmax_fn(z)
    _, grad_a = lse_obj_grad(a_val, T)
    # Transform to z-space: grad_z = a * (grad_a - a^T * grad_a)
    return a_val * (grad_a - np.dot(a_val, grad_a))


def lse_lbgfs_phase(a0, time_limit):
    """L-BFGS on LSE-smoothed objective with softmax parameterization."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    print(f"  LSE-LBFGS start: score={best_score:.8f}, n={n}")

    T_stages = [30.0, 15.0, 8.0, 5.0, 3.0, 2.0, 1.2, 0.8, 0.5, 0.3, 0.15, 0.08]
    store = [None, None]

    for T in T_stages:
        if time.monotonic() - t0 > time_limit - 8:
            break
        remaining = time_limit - (time.monotonic() - t0)
        stage_limit = min(remaining - 5, time_limit * 0.12)
        if stage_limit < 5:
            continue

        deadline = time.monotonic() + stage_limit
        params = (n, T, deadline, store)

        # Initialize z from current a
        z0 = np.log(np.maximum(a, 1e-300))

        maxiter = max(50, int(stage_limit * 200))
        try:
            res = minimize(fun=lse_func_wrap, x0=z0, args=params,
                           jac=lse_jac_wrap, method='L-BFGS-B',
                           options={'maxiter': maxiter, 'ftol': 1e-14,
                                    'gtol': 1e-10, 'maxfun': maxiter * 300,
                                    'disp': False})
            a_trial = softmax_fn(res.x)
            sc = eval_score(a_trial)
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a_trial.copy()
            a = a_trial
            print(f"  LSE-LB T={T:.2f}: score={sc:.8f}")
        except Exception as e:
            print(f"  LSE-LB T={T:.2f} failed: {e}")

    return best_a, best_score


def build_sparse_conv_matrix(a):
    """Build sparse matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    row_ind = []
    col_ind = []
    data = []
    for i in range(n):
        row_ind.extend(range(i, i + n))
        col_ind.extend([i] * n)
        data.extend(a)
    T = csr_matrix((data, (np.array(row_ind), np.array(col_ind))), shape=(size, n))
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver
