def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            no_improve += 1

        if no_improve >= 50:
            eta *= 0.85
            no_improve = 0
            if eta < 1e-5:
                break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Phases: init -> LP alternating -> active QCQP -> full QCQP attempt ->
    multi-n exploration -> final tight polish loop.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    print(f"Total budget: {budget_s:.0f}s, deadline in {budget_s - 30:.0f}s")

    best_seq = None
    best_score = float('inf')
    lp_solver, qp_solver = find_best_solver()
    print(f"LP solver: {lp_solver}, QCQP solver: {qp_solver}")

    # Load parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        # Multiple initializations, pick best
        inits = {
            'cosine': cosine_init(1200, 2.0),
            'gaussian': gaussian_init(1200, 0.35),
            'uniform': np.ones(1200) / 1200,
            'cos4': cosine_init(1200, 4.0),
        }
        for name, init in inits.items():
            sc = evaluate_local(init)
            print(f"  Init '{name}': score={sc:.10f}")
            if sc < best_score:
                best_score = sc
                best_seq = init.copy()
        print(f"Best init: score={best_score:.10f}")

    n_best = len(best_seq)

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: LP Alternating on best n ===
    print(f"\n=== Phase 1: LP Alternating n={n_best} (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 450.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        T_mat = build_conv_matrix(best_seq)
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, T_mat, lp_solver,
                                     n_max_iters=500000)
        update_best(a_lp, "lp_phase1")
        # Rebuild T after LP (sequence changed)
        T_mat = build_conv_matrix(best_seq)

    # === Phase 2: Active-set QCQP polish ===
    print(f"\n=== Phase 2: Active QCQP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_qcqp = min(rem * 0.10, 120.0)
        t0 = time.monotonic()
        a_qcqp, _ = solve_active_qcqp(best_seq, qp_solver, rel_tol=1e-5,
                                      max_active=40)
        update_best(a_qcqp, "qcqp_wide")
        # Tighter
        a_qcqp2, _ = solve_active_qcqp(best_seq, qp_solver, rel_tol=1e-7,
                                       max_active=25)
        update_best(a_qcqp2, "qcqp_tight")
        print(f"  QCQP time: {time.monotonic() - t0:.1f}s")

    # === Phase 3: LP round 2 from polished ===
    print(f"\n=== Phase 3: LP Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        T_mat = build_conv_matrix(best_seq)
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, T_mat, lp_solver,
                                      n_max_iters=300000)
        update_best(a_lp2, "lp_phase3")

    # === Phase 4: Full QCQP attempt ===
    print(f"\n=== Phase 4: Full QCQP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60 and n_best <= 1210:
        t0 = time.monotonic()
        a_full, t_val = solve_full_qcqp(best_seq, qp_solver, time_limit=min(rem * 0.15, 180.0))
        if a_full is not None:
            update_best(a_full, "full_qcqp")
            print(f"  Full QCQP t={t_val:.10f}")
        else:
            print(f"  Full QCQP failed (timed out or no solution)")
        print(f"  Full QCQP time: {time.monotonic() - t0:.1f}s")

    # === Phase 5: QCQP polish again ===
    print(f"\n=== Phase 5: QCQP Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        a_qcqp3, _ = solve_active_qcqp(best_seq, qp_solver, rel_tol=1e-8,
                                       max_active=30)
        update_best(a_qcqp3, "qcqp_round2")

    # === Phase 6: Multi-n exploration ===
    for n_try in [1199, 1201, 1200]:
        if time.monotonic() > deadline:
            break
        if n_try == n_best:
            continue
        rem = deadline - time.monotonic()
        if rem < 80:
            print(f"\n  Skipping n={n_try} (insufficient time: {rem:.0f}s)")
            continue
        print(f"\n=== Phase: n={n_try} (score={best_score:.10f}) ===")
        t_alloc = min(rem * 0.15, 180.0)
        print(f"  Budget: {t_alloc:.0f}s")

        init_try = interpolate_sequence(best_seq, n_try)
        T_try = build_conv_matrix(init_try)
        a_try, _ = nesterov_lp_refine(init_try, t_alloc * 0.7, T_try, lp_solver,
                                      n_max_iters=200000)
        update_best(a_try, f"lp_n{n_try}")

        rem = deadline - time.monotonic()
        if best_seq is not None and len(best_seq) == n_try and rem > 20:
            a_q, _ = solve_active_qcqp(best_seq, qp_solver, rel_tol=1e-6,
                                       max_active=20)
            update_best(a_q, f"qcqp_n{n_try}")

    # === Phase 7: Adaptive final polish loop ===
    print(f"\n=== Phase 7: Final Polish Loop (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        n_final = len(best_seq)
        T_final = build_conv_matrix(best_seq)
        polish_start = time.monotonic()
        polish_deadline = time.monotonic() + min(rem * 0.5, 200.0)

        for polish_iter in range(10):
            if time.monotonic() > polish_deadline:
                break

            # LP step
            T_final = build_conv_matrix(best_seq)
            a_lp_f, _ = nesterov_lp_refine(best_seq, min(polish_deadline - time.monotonic(), 30.0),
                                           T_final, lp_solver, n_max_iters=5000)
            if a_lp_f is not None:
                update_best(a_lp_f, f"polish_lp_{polish_iter}")

            # QCQP step
            rem_q = polish_deadline - time.monotonic()
            if rem_q > 15:
                a_qf, _ = solve_active_qcqp(best_seq, qp_solver,
                                            rel_tol=1e-8 * (polish_iter + 1),
                                            max_active=15)
                update_best(a_qf, f"polish_qcqp_{polish_iter}")

            if time.monotonic() > polish_deadline:
                break
            if polish_iter > 3 and best_score > evaluate_local(best_seq) - 1e-12:
                break

    # === Phase 8: Multiplicative weights final ===
    rem = deadline - time.monotonic()
    if rem > 10:
        a_ms, _ = mf_refine(best_seq, min(rem * 0.3, 30.0), n_steps=10000)
        update_best(a_ms, "mf_final")
        rem = deadline - time.monotonic()
        if rem > 15:
            a_qf, _ = solve_active_qcqp(best_seq, qp_solver, rel_tol=1e-9,
                                        max_active=10)
            update_best(a_qf, "final_qcqp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
