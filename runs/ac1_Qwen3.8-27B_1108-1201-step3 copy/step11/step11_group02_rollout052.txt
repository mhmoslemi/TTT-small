def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute the objective value."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)

    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def build_conv_matrix_sparse(a):
    """Build sparse matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    rows = []
    cols = []
    data = []
    # T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # For each j, T[k,j] is nonzero for k = j, j+1, ..., j+n-1
    for j in range(n):
        k_start = j
        k_end = j + n  # exclusive
        rows.extend(range(k_start, min(k_end, size)))
        cols.extend([j] * (min(k_end, size) - k_start))
        data.extend(a[:min(n, size - j)])
    
    return csr_matrix((data, (np.array(rows), np.array(cols))), shape=(size, n))

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def mm_algorithm(a0, time_limit, n_iters=5000, verbose=True):
    """
    Majorize-Minimize (Alternating Direction) approach.
    At each step, fix the "dual" variable (active peak distribution) and solve for a.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    
    # Dual variable: distribution over peak positions
    b = fast_convolve(a)
    max_b = b.max()
    # Initialize dual as concentrated on the max
    p = np.zeros(2 * n - 1)
    p[int(np.argmax(b))] = 1.0
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Given dual p (distribution over k), the "effective" gradient direction is:
        # g_j = 2 * sum_k p[k] * a[k-j]
        # We want to minimize g^T a (linear in a) subject to sum(a)=1, a>=0
        # But this is just: minimize sum_j a_j * g_j
        # The optimum is to put all mass on argmin g_j, which is degenerate.
        # Instead, use a smoothed version: minimize entropy-regularized objective
        
        # Better approach: use the dual to define a quadratic surrogate
        # f(a) = max_k (a*a)[k] <= sum_k p[k] (a*a)[k] for any p >= 0, sum(p) = 1
        # (a*a)[k] = sum_i a_i * a[k-i]
        # So sum_k p[k] (a*a)[k] = sum_i sum_k p[k] a_i a[k-i]
        # = sum_i a_i * (sum_k p[k] a[k-i])
        # = a^T P a where P is some matrix
        
        # The surrogate is: minimize a^T P a subject to sum(a) = 1, a >= 0
        # This is a QP! Let's solve it.
        
        # Build P: P[i,j] = sum_k p[k] * delta[k, i+j] ... 
        # Actually (a*a)[k] = sum_i a_i * a_{k-i}, so:
        # sum_k p[k] (a*a)[k] = sum_k p[k] sum_i a_i a_{k-i}
        # = sum_i sum_j a_i a_j * p[i+j]  (where k = i+j)
        # So P[i,j] = p[i+j] (with p extended by zeros)
        
        # This is a quadratic form a^T Q a where Q[i,j] = p[i+j]
        # But Q is not symmetric in general unless p is symmetric.
        # Actually Q[i,j] = p[i+j] and Q[j,i] = p[j+i] = p[i+j] = Q[i,j]. So it IS symmetric!
        
        # So we want: minimize a^T Q a subject to sum(a)=1, a>=0
        # where Q[i,j] = p[i+j] for i+j < 2n-1, else 0
        
        # This is a QP. Let's solve it with cvxpy.
        
        Q_mat = np.zeros((n, n))
        # Q[i,j] = p[i+j]
        for i in range(n):
            for j in range(n):
                k = i + j
                if k < 2 * n - 1:
                    Q_mat[i, j] = p[k]
        
        # Actually this O(n^2) loop is too slow for large n. 
        # Let's use a different approach: since Q = "Hankel matrix" of p,
        # we can compute a^T Q a = sum_k p[k] (sum_{i+j=k} a_i a_j) = sum_k p[k] (a*a)[k]
        # So the surrogate is just sum_k p[k] * (a*a)[k]
        
        # For the QP, we can use:
        # minimize a @ Q_mat @ a
        # s.t. sum(a) = 1, a >= 0
        
        # But building Q_mat is O(n^2) which is fine for n up to a few thousand.
        # Let's use a vectorized approach.
        
        # Actually, let me just use the subgradient approach differently.
        # The key insight: given p, the surrogate is linear in "a*a" which is quadratic in a.
        # We can solve the QP using a simple projected gradient on the quadratic.
        
        # Surrogate gradient: d/da (a^T Q a) = 2 * Q @ a
        # Q @ a has component i: sum_j Q[i,j] a_j = sum_j p[i+j] a_j
        
        # This is a convolution-like operation! (p * a)[i] but with index i+j not i-j.
        # Actually (Q @ a)[i] = sum_j p[i+j] a_j
        # If we define r = reverse(a), then (Q@a)[i] = sum_j p[i+j] a_j
        # Let m = i+j, so j = m-i: sum_m p[m] a[m-i] = (p * a')[m] evaluated at m=i... 
        # This is cross-correlation: (p ⋆ a)[i] where (p ⋆ a)[i] = sum_j p[i+j] a[j]
        
        # We can compute this via FFT!
        # (p ⋆ a)[i] = sum_j p[i+j] a[j]
        # Let's pad and use FFT.
        
        # Actually, let me just compute the gradient directly using the current p.
        # grad_surrogate[i] = 2 * sum_j p[i+j] * a[j]
        
        # This is O(n^2) per iteration but with n~1200, that's 1.4M ops per iter, fine.
        
        grad_s = np.zeros(n)
        for i in range(n):
            # sum_j p[i+j] * a[j] for j=0..n-1, i+j < 2n-1
            j_max = min(n, 2 * n - 1 - i)
            if j_max > 0:
                grad_s[i] = np.dot(p[i:i + j_max], a[:j_max])
        grad_s *= 2.0
        
        # Minimize a^T Q a s.t. sum(a)=1, a>=0 using projected gradient
        # Step: a_new = a - eta * grad_s, project to simplex
        
        # Use adaptive step size
        # The maximum eigenvalue of Q gives the Lipschitz constant
        # Q is a Hankel matrix, eigenvalues bounded by sum(p) * max(a) * n roughly
        # For safety, use small step
        eta_mm = 0.1 / (np.max(np.abs(grad_s)) + 1e-10)
        eta_mm = min(eta_mm, 1.0)
        
        a_trial = a - eta_mm * grad_s
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st
        
        # Update p based on current a (mirror descent on the dual)
        b_new = fast_convolve(a_trial)
        max_bn = b_new.max()
        # Exponential update for p
        T_p = 0.1 + 0.5 * (it / n_iters)  # annealing temperature
        p_new = np.exp(b_new / T_p - max_bn / T_p)
        p_sum = p_new.sum()
        if p_sum > 1e-30:
            p = p_new / p_sum
        # Mix with old p for stability
        p = 0.7 * p + 0.3 * (p_new / p_sum)
        
        a = a_trial
        score = score_of(a)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
        
        if verbose and (it + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MM iter={it+1} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute the objective value 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def max_conv_of(a):
    """Return max of auto-convolution (assumes a is normalized to sum=1)."""
    b = fast_convolve(a)
    return float(b.max())


def subgradient_max_conv(a, active_frac=0.01, min_active=1):
    """
    Compute subgradient of max_k (a*a)[k] at point a.
    Returns (gradient, active_indices, max_val).
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    threshold = max_b * (1.0 - active_frac)
    active = np.where(b >= threshold)[0]
    if len(active) < min_active:
        active = np.array([int(np.argmax(b))])

    grad = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += a[k - idx]
    grad *= 2.0 / len(active)

    return grad, active, max_b


def build_conv_matrix_sparse(a):
    """Build sparse matrix T such that T @ b = convolve(a, b). Shape (2n-1, n)."""
    n = len(a)
    size = 2 * n - 1
    rows = []
    cols = []
    data = []
    for j in range(n):
        k_start = j
        k_end = min(j + n, size)
        rows.extend(range(k_start, k_end))
        cols.extend([j] * (k_end - k_start))
        data.extend(a[:k_end - k_start])
    return csr_matrix((data, (np.array(rows), np.array(cols))), shape=(size, n))


_lp_solver_name = None


def get_lp_solver():
    """Find a working LP solver (cached)."""
    global _lp_solver_name
    if _lp_solver_name is not None:
        return _lp_solver_name
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_name = solver_name
                return _lp_solver_name
        except Exception:
            continue
    _lp_solver_name = cp.ECOS
    return _lp_solver_name


def lp_best_response(a_fixed, solver=None, timeout_s=30.0):
    """
    Solve LP: min t s.t. conv(a_fixed, b)[k] <= t, sum(b)=1, b>=0.
    Returns (b_opt, t_opt, dual_values) or (None, None, None) on failure.
    """
    n = len(a_fixed)
    if solver is None:
        solver = get_lp_solver()

    T = build_conv_matrix_sparse(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    conv_constraint = T @ b_var <= t_var
    constraints = [conv_constraint, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)

    t_start = time.monotonic()
    success = False
    for s in [solver, cp.ECOS, cp.CLARABEL, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                success = True
                break
        except Exception:
            continue
        if time.monotonic() - t_start > timeout_s:
            break

    if not success:
        return None, None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s
    t_val = float(t_var.value) if t_var.value is not None else max_conv_of(b_val)

    # Get dual values (lower bound on t for fixed a)
    dual_vals = None
    try:
        dual_vals = conv_constraint.dual.value
        if dual_vals is not None:
            dual_vals = np.maximum(np.asarray(dual_vals, dtype=np.float64), 0)
            ds = dual_vals.sum()
            if ds > 1e-15:
                dual_vals = dual_vals / ds
            else:
                dual_vals = None
    except Exception:
        dual_vals = None

    return b_val, t_val, dual_vals


def hybrid_subgrad_lp(a0, time_limit, solver=None, verbose=True,
                      lp_every=500, active_frac=0.005):
    """
    Hybrid: subgradient descent with periodic LP best-response targets.
    The subgradient method moves towards the LP target between LP calls.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    eta = 0.05
    no_improve = 0
    lp_target = None
    it = 0

    while time.monotonic() - t0 < time_limit:
        # Periodic LP solve
        if it % lp_every == 0:
            b_new, t_val, dual_vals = lp_best_response(a, solver=solver, timeout_s=60.0)
            if b_new is not None:
                lp_target = b_new.copy()
                # Check if b_new itself is good
                sc_b = score_of(b_new)
                if sc_b < best_score - 1e-14:
                    best_score = sc_b
                    best_a = b_new.copy()
                    no_improve = 0
                if verbose and it % (lp_every * 10) == 0:
                    elapsed = time.monotonic() - t0
                    print(f"    HS iter={it} ({elapsed:.0f}s): best={best_score:.8f}, "
                          f"LP_t={t_val:.8f}, score_b={sc_b:.8f}")

        # Subgradient step
        grad, active, max_b = subgradient_max_conv(a, active_frac=active_frac)
        score = 2.0 * n * max_b

        # Adaptive step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            eta = 0.01
        else:
            eta = min(eta * 1.001, 1.0 / (grad_norm * np.sqrt(it + 1) + 1))
            eta = max(eta, 1e-7)

        a_trial = a - eta * grad
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        # If we have an LP target, blend towards it
        if lp_target is not None:
            blend = 0.02  # small blend towards LP target
            a_trial = (1 - blend) * a_trial + blend * lp_target
            a_trial = np.maximum(a_trial, 0)
            st2 = a_trial.sum()
            if st2 > 1e-15:
                a_trial = a_trial / st2

        a = a_trial
        score_new = score_of(a)
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            eta *= 0.9
            no_improve = 0

        it += 1

    return best_a, best_score


def nesterov_game(a0, time_limit, solver=None, n_max_iters=100000, verbose=True):
    """
    Nesterov-accelerated game iteration.
    Alternates LP best-response with Nesterov momentum updates.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score
    rng = np.random.default_rng(777)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # LP best response for y
        b_new, t_val, dual_vals = lp_best_response(y, solver=solver, timeout_s=60.0)
        if b_new is None:
            continue

        # Check b_new as candidate
        sc_b = score_of(b_new)
        if sc_b < best_score:
            best_score = sc_b
            best_a = b_new.copy()

        # Nesterov update
        alpha = 2.0 / (np.sqrt(it + 1) + 1.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        # FTRL-style for a
        a = ((it + 1) * a + (it + 2) * alpha * b_new) / (it + 1 + (it + 2) * alpha)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
            sa = 1.0
        a = a / sa

        # Check scores
        score_a = score_of(a)
        score_y = score_of(y)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0

        # Stall detection
        if (it + 1) % 30 == 0:
            if best_score < prev_best - 1e-12:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= 6:
                # Structured perturbation
                b_cur = fast_convolve(y)
                peak_idx = int(np.argmax(b_cur))
                js = max(0, peak_idx - n + 1)
                je = min(n - 1, peak_idx)
                idx = np.arange(js, je + 1)
                contribs = y[idx] * y[peak_idx - idx]
                n_reduce = max(3, len(idx) // 15)
                top_pos = idx[np.argsort(contribs)[::-1][:n_reduce]]
                for pos in top_pos:
                    if y[pos] > 1e-10:
                        y[pos] *= 0.95
                noise = rng.standard_normal(n) * 0.01
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                a = y.copy()
                stall = 0

        if verbose and (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NG iter={it+1} ({elapsed:.0f}s): best={best_score:.8f}, "
                  f"LP_bound={2*n*t_val:.8f}")

    return best_a, best_score


def mm_algorithm(a0, time_limit, n_max_iters=10000, verbose=True):
    """
    Majorize-Minimize using dual distribution.
    Maintains a distribution p over convolution positions.
    At each step, minimizes the surrogate sum_k p[k]*(a*a)[k] (a QP),
    then updates p based on current convolution peaks.
    Uses projected gradient on the quadratic surrogate.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()

    # Initialize dual distribution
    b = fast_convolve(a)
    max_b = b.max()
    T_init = 0.5
    p = np.exp(b / T_init - max_b / T_init)
    p = p / p.sum()

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Surrogate gradient: grad[i] = 2 * sum_j p[i+j] * a[j]
        grad_s = np.zeros(n)
        for i in range(n):
            j_max = min(n, 2 * n - 1 - i)
            if j_max > 0:
                grad_s[i] = np.dot(p[i:i + j_max], a[:j_max])
        grad_s *= 2.0

        # Projected gradient step on the quadratic
        grad_norm = np.linalg.norm(grad_s)
        if grad_norm < 1e-15:
            break
        eta_mm = 0.5 / grad_norm
        eta_mm = min(eta_mm, 0.5)

        a_trial = a - eta_mm * grad_s
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        # Update p (exponential, annealed temperature)
        b_new = fast_convolve(a_trial)
        max_bn = b_new.max()
        T_p = max(0.05, 1.0 * (1.0 - it / float(n_max_iters)) + 0.05)
        p_new = np.exp(b_new / T_p - max_bn / T_p)
        p_sum = p_new.sum()
        if p_sum > 1e-30:
            p_new = p_new / p_sum
        p = 0.7 * p + 0.3 * p_new

        a = a_trial
        score = score_of(a)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()

        if verbose and (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MM iter={it+1} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=True):
    """Coordinate descent focusing on peak-contributing elements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    stall = 0

    for step in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.9998
        peaks = np.where(b >= threshold)[0]

        improved = False
        for peak_k in peaks[:3]:
            if improved:
                break
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_m = min(8, len(idx))
            top_pos = idx[np.argsort(contribs)[::-1][:top_m]]

            for pos in top_pos:
                if a[pos] < 1e-14:
                    continue
                for factor in [0.9998, 0.9995, 0.999, 0.998, 0.995, 0.99]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-18:
                        break
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    score = score_of(a_trial)
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 500:
                # Simultaneous reduction of top contributors
                peak_k = int(np.argmax(b))
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[peak_k - idx]
                top_pos = idx[np.argsort(contribs)[::-1][:4]]
                a_trial = a.copy()
                for pos in top_pos:
                    if a_trial[pos] > 1e-14:
                        a_trial[pos] *= 0.998
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-15:
                    a_trial = a_trial / st
                    score = score_of(a_trial)
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                if stall > 3000:
                    break
        else:
            stall = 0

        if verbose and (step + 1) % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def mirror_descent(a0, time_limit, eta0=0.03, verbose=True):
    """
    Mirror descent (exponential weights) on the max-convolution objective.
    Uses subgradient with multiplicative updates.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    no_improve = 0
    eta = eta0
    rng = np.random.default_rng(123)

    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        grad, active, max_b = subgradient_max_conv(a, active_frac=0.01)
        score = 2.0 * n * max_b

        # Mirror (multiplicative) descent
        log_a = np.log(a)
        g_shifted = grad - grad.min()
        log_a_new = log_a - eta * g_shifted
        log_a_new -= log_a_new.max()  # numerical stability
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-30:
            a_new = a.copy()
        else:
            a_new = a_new / a_new_sum

        a = a_new
        score_new = score_of(a)
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            eta *= 0.95
            no_improve = 0
            if eta < 1e-6:
                # Random restart
                noise = rng.standard_normal(n) * 0.02
                a = a + noise
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                eta = eta0
                no_improve = 0

        if verbose and (it + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={it+1} ({elapsed:.0f}s): best={best_score:.8f}, eta={eta:.6f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining hybrid subgradient+LP, Nesterov game,
    mirror descent, MM algorithm, multi-n exploration, and coordinate descent.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p = p / p.sum()
            p_score = score_of(p)
            print(f"Parent: n={len(p)}, score={p_score:.8f}")
            best_seq = p.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = np.cos(np.pi * x / 2) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = score_of(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = score_of(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Nesterov Game (main workhorse) ===
    print(f"\n=== Phase 1: Nesterov Game (score={best_score:.8f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_out, _ = nesterov_game(best_seq, t_alloc, solver=solver, n_max_iters=100000)
        try_update(a_out, "ng_p1")

    # === Phase 2: Hybrid Subgradient + LP ===
    print(f"\n=== Phase 2: Hybrid SGD+LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_out, _ = hybrid_subgrad_lp(best_seq, t_alloc, solver=solver, lp_every=300)
        try_update(a_out, "hybrid")

    # === Phase 3: Mirror Descent ===
    print(f"\n=== Phase 3: Mirror Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_out, _ = mirror_descent(best_seq, t_alloc, eta0=0.03)
        try_update(a_out, "mirror")

    # === Phase 4: MM Algorithm ===
    print(f"\n=== Phase 4: MM Algorithm (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_out, _ = mm_algorithm(best_seq, t_alloc, n_max_iters=20000)
        try_update(a_out, "mm")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 30:
        n_best = len(best_seq)
        candidate_ns = sorted(set([
            max(300, n_best - 400),
            max(500, n_best + 200),
            max(400, n_best + 500),
            max(600, n_best // 2),
        ]))
        per_n = t_alloc / max(1, len(candidate_ns))
        print(f"  n values: {candidate_ns}, ~{per_n:.0f}s each")
        for n_try in candidate_ns:
            if time.monotonic() > deadline - 10:
                break
            if n_try == n_best or n_try < 100 or n_try > 5000:
                continue
            a_interp = interpolate_seq(best_seq, n_try)
            t_sub = min(per_n, deadline - time.monotonic() - 5)
            if t_sub < 15:
                continue
            print(f"  n={n_try}, budget={t_sub:.0f}s")
            a_sub, _ = nesterov_game(a_interp, t_sub, solver=solver, n_max_iters=50000)
            try_update(a_sub, f"multi_n_{n_try}")

    # === Phase 6: Deep Nesterov Game ===
    print(f"\n=== Phase 6: Deep NG (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 15:
        a_out, _ = nesterov_game(best_seq, t_alloc, solver=solver, n_max_iters=50000)
        try_update(a_out, "deep_ng")

    # === Phase 7: Coordinate Descent Polish ===
    print(f"\n=== Phase 7: Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 70.0)
    if t_alloc > 10:
        a_out, _ = coordinate_descent(best_seq, t_alloc)
        try_update(a_out, "coord_desc")

    # Final verification
    if best_seq is not None:
        fs = score_of(best_seq)
        best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
