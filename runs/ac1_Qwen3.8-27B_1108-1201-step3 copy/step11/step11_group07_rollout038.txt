For the second term: we need k-i=j, i.e., i=k-j. We need i to be valid: max(0, k-n+1) <= k-j <= min(n-1, k), i.e., max(0, k-n+1) <= k-j and k-j <= min(n-1, k). The first gives j <= n-1, the second gives j >= 0 and j >= k-n+1 and k-j <= k (always true) and k-j <= n-1 (i.e., j >= k-n+1).

So both conditions are the same: max(0, k-n+1) <= j <= min(n-1, k).

Therefore: dc_k/da[j] = 2*a[k-j] if max(0, k-n+1) <= j <= min(n-1, k), else 0.

    Returns the optimal b (unnormalized).
    """

    # conv(a,b)[k] = sum_j a[k-j] * b[j] for valid j

    # This is a "toeplitz-like" matrix (actually it's the convolution matrix)
    
    # For efficiency, build it as a sparse matrix or use the structure
    # A_ub[k, j] = a[k-j] if |k-j| < n, i.e., k-n+1 <= j <= k and 0 <= j < n
    
    # Actually, let's think of it differently.
    # conv(a,b)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a[k-j] * b[j]
    
    # The matrix is (2n-1) x n. For row k:
    # - j ranges from max(0, k-n+1) to min(n-1, k)
    # - A[k,j] = a[k-j]
    
    # This is the "convolution matrix" of a, which has a banded structure.
    
    # For n=1200, this is 2399 x 1200 = ~2.9M entries (all non-zero in the band)
    # Actually, the total number of non-zeros is sum_k (min(n-1,k) - max(0,k-n+1) + 1)
    # = n + (n-1) + ... + 1 + 1 + ... + (n-1) + n = n^2
    # So it's a dense matrix with n^2 = 1.44M entries.
    
    # Let's build it efficiently using the Toeplitz structure.
    # Actually, np.lib.stride_tricks or a simple loop might work.
    
    # For speed, let's use the fact that A_ub is the "full convolution matrix"
    # A_ub = toeplitz-like
    
    # Row k has a[k], a[k-1], ..., a[k-(n-1)] starting at column max(0, k-n+1)
    # Equivalently: A_ub[k, j] = a[k-j] if 0 <= k-j <= n-1
    
    # This can be constructed as:
    # A_ub = np.zeros((2*n-1, n))
    # for k in range(2*n-1):
    #     for j in range(max(0, k-n+1), min(n, k+1)):
    #         A_ub[k, j] = a[k-j]
    
    # For n=1200, this is a tight double loop. Let's vectorize.
    
    k_idx = np.arange(2*n-1)[:, None]  # (2n-1, 1)
    j_idx = np.arange(n)[None, :]      # (1, n)
    diff = k_idx - j_idx               # (2n-1, n)
    mask = (diff >= 0) & (diff < n)
    A_ub = np.where(mask, a[diff * mask], 0.0)  # This creates a (2n-1, n) array
    
    # Actually, a[diff * mask] won't work directly. Let me use:
    A_ub = np.zeros((2*n-1, n))
    A_ub[mask] = a[diff[mask]]
    
    b_ub = np.full(2*n-1, M)
    
    # c = -ones(n) (we want to maximize sum(b), so minimize -sum(b))
    c = -np.ones(n)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    else:
        # Fallback: return proportional to a
        return a.copy()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based Hamiltonian descent
with spectral shaping and multi-scale optimization.
Key innovation: iterative LP solves to find optimal descent directions,
combined with frequency-domain filtering for spectral flatness.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_convolution_matrix(a):
    """Build the matrix C such that C @ b = conv(a, b).
    C has shape (2n-1, n), with C[k, j] = a[k-j] if 0 <= k-j < n.
    """
    n = len(a)
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    diff = k - j
    mask = (diff >= 0) & (diff < n)
    C = np.zeros((2 * n - 1, n))
    C[mask] = a[diff[mask]]
    return C

def lp_hamiltonian_step(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0.
    Returns normalized b.
    """
    n = len(a)
    C = build_convolution_matrix(a)
    c = -np.ones(n)  # minimize -sum(b)
    b_ub = np.full(2 * n - 1, M)
    
    res = linprog(c, A_ub=C, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True})
    
    if res.success:
        b = res.x
        s = b.sum()
        if s > 1e-15:
            return b / s
    return a.copy()

def lp_descent(a_init, time_limit, max_steps=5000, t0_init=0.1):
    """LP-based Hamiltonian descent with line search."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    t_param = t0_init  # mixing parameter
    
    C = build_convolution_matrix(a)  # Will be updated when a changes
    
    step = 0
    while time.monotonic() - t0 < time_limit:
        step += 1
        
        conv_a = fast_convolve(a)
        M = conv_a.max()
        
        # Solve LP
        b = lp_hamiltonian_step(a, M)
        
        # Line search for mixing parameter t
        improvement = False
        for t in [t_param, t_param * 2, t_param * 0.5, t_param * 0.25, t_param * 4]:
            if t <= 0 or t > 0.5:
                continue
            a_new = (1 - t) * a + t * b
            a_new = normalize(a_new)
            sc = evaluate_local(a_new)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_new.copy()
                improvement = True
            if sc <= best_score:
                a = a_new
                break
        else:
            # No improvement, reduce step size
            t_param *= 0.5
            if t_param < 1e-6:
                break
            # Try a small perturbation to escape
            rng = np.random.default_rng(step)
            a = normalize(a + rng.normal(0, 0.001, n))
            continue
        
        if step % 100 == 0:
            print(f"    LP step {step}: score={best_score:.8f}, t={t_param:.4f}")
    
    return best_a, best_score

def spectral_flatness_descent(a_init, time_limit, lr=0.01):
    """Flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    
    step = 0
    while time.monotonic() - t0 < time_limit:
        step += 1
        
        A = rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        
        # Direction to flatten P
        r = (P - P_mean) / (P_mean + 1e-15)
        
        H = np.zeros(fft_size // 2 + 1, dtype=np.complex128)
        valid = P > 1e-12
        H[valid] = -r[valid] * P_mean * A[valid] / (P[valid] + 1e-10)
        H[0] = 0.0
        h = irfft(H, fft_size)[:n]
        h -= h.mean()
        
        h_norm = np.linalg.norm(h)
        if h_norm < 1e-15:
            break
        h = h / h_norm
        
        for scale in [lr, lr * 0.5, lr * 0.1]:
            a_new = a + scale * h
            a_new = normalize(a_new)
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                break
        
        if step % 500 == 0:
            lr *= 0.8
            print(f"    Spectral step {step}: score={best_score:.8f}, lr={lr:.6f}")
    
    return best_a, best_score

def subgradient_descent(a_init, time_limit, lr=0.001):
    """Subgradient descent on the active set (conv peaks)."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    step = 0
    while time.monotonic() - t0 < time_limit:
        step += 1
        
        conv_a = fast_convolve(a)
        c_max = conv_a.max()
        
        # Active set: lags where conv is close to max
        threshold = c_max * (1 - 1e-4)
        active = np.where(conv_a >= threshold)[0]
        
        # Subgradient: average of gradients at active lags
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            g[js:je+1] += 2 * a[k - np.arange(js, je+1)][::-1]
        
        g /= len(active)
        
        # Projected gradient step (maintain sum=1, a>=0)
        a_new = a - lr * g
        a_new = a_new - a_new.mean()  # centered
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            lr *= 0.8
            if lr < 1e-10:
                break
        
        if step % 200 == 0:
            print(f"    Subgrad step {step}: score={best_score:.8f}, lr={lr:.8f}")
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based Hamiltonian descent
with spectral shaping and multi-phase optimization.
Key innovation: iterative LP solves for optimal descent directions in the
convolution constraint polytope, combined with frequency-domain spectral
flattening and adaptive restarts for escaping local minima.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_mat(a):
    """Build convolution matrix C where C @ b = conv(a, b)."""
    n = len(a)
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    diff = k - j
    mask = (diff >= 0) & (diff < n)
    C = np.zeros((2 * n - 1, n))
    C[mask] = a[diff[mask]]
    return C


def lp_step(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M elementwise, b >= 0.
    Returns normalized b or None on failure."""
    n = len(a)
    C = build_conv_mat(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if res.success:
            b = np.maximum(res.x, 0)
            s = b.sum()
            if s > 1e-12:
                return b / s
    except Exception:
        pass
    return None


def lp_descent(a_init, time_limit, t_init=0.05, max_steps=20000):
    """LP-based Hamiltonian descent with adaptive line search."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t_start = time.monotonic()
    t_param = t_init
    stagnant = 0

    for step in range(max_steps):
        if time.monotonic() - t_start > time_limit:
            break

        conv_a = fast_convolve(a)
        M = conv_a.max()
        if M < 1e-15:
            break

        b = lp_step(a, M)
        if b is None:
            break

        # Line search
        found = False
        for t in [t_param, t_param * 2.0, t_param * 0.5, t_param * 3.0, t_param * 0.25]:
            if t <= 0 or t > 0.8:
                continue
            a_new = normalize((1.0 - t) * a + t * b)
            sc = evaluate_local(a_new)
            if sc < best_score - 1e-13:
                best_score = sc
                best_a = a_new.copy()
                found = True
            if sc <= evaluate_local(a) + 1e-12:
                a = a_new
                if found:
                    stagnant = 0
                break
        else:
            stagnant += 1
            t_param *= 0.6
            if stagnant > 20:
                # Perturb to escape
                rng = np.random.default_rng(step * 7919)
                a = normalize(a + rng.normal(0, 0.002, n))
                stagnant = 0
            if t_param < 1e-7:
                break
            continue

        if step % 500 == 0 and step > 0:
            print(f"    LP[{step}]: score={best_score:.10f}, t={t_param:.6f}, M={M:.8f}")

    return best_a, best_score


def spectral_descent(a_init, time_limit, lr_init=0.003):
    """Flatten power spectrum to reduce convolution peaks."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t_start = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    lr = lr_init

    for step in range(50000):
        if time.monotonic() - t_start > time_limit:
            break

        A = rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-15:
            break

        r = (P - P_mean) / (P_mean + 1e-15)
        H = np.zeros(fft_size // 2 + 1, dtype=np.complex128)
        valid = P > 1e-12
        H[valid] = -r[valid] * P_mean * A[valid] / (P[valid] + 1e-10)
        H[0] = 0.0
        h = irfft(H, fft_size)[:n]
        h -= h.mean()

        hn = np.linalg.norm(h)
        if hn < 1e-15:
            break
        h = h / hn

        found = False
        for scale in [lr, lr * 0.5, lr * 0.1, lr * 2.0]:
            a_new = normalize(a + scale * h)
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                found = True
            if sc <= evaluate_local(a) + 1e-12:
                a = a_new
                break

        if step % 1000 == 0:
            lr *= 0.85
            if step > 0:
                print(f"    Spec[{step}]: score={best_score:.10f}, lr={lr:.7f}")

    return best_a, best_score


def subgrad_descent(a_init, time_limit, lr_init=0.0003):
    """Subgradient descent on active set (conv peaks)."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t_start = time.monotonic()
    lr = lr_init

    for step in range(100000):
        if time.monotonic() - t_start > time_limit:
            break

        conv_a = fast_convolve(a)
        c_max = conv_a.max()
        threshold = c_max * (1.0 - 1e-5)
        active = np.where(conv_a >= threshold)[0]

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            g[js:je + 1] += 2.0 * a[k - np.arange(js, je + 1)][::-1]
        g /= len(active)

        # Remove component along 1 (to maintain sum=1 after projection)
        g -= g.mean()

        a_new = a - lr * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            lr *= 0.7
            if lr < 1e-12:
                break

        if step % 2000 == 0 and step > 0:
            print(f"    SubGrad[{step}]: score={best_score:.10f}, lr={lr:.2e}")

    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)


def multiplicative_weights(a_init, time_limit, eta_init=0.03):
    """Multiplicative weights / mirror descent on the active set."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t_start = time.monotonic()
    eta = eta_init

    for step in range(50000):
        if time.monotonic() - t_start > time_limit:
            break

        conv_a = fast_convolve(a)
        c_max = conv_a.max()
        threshold = c_max * (1.0 - 1e-4)
        active = np.where(conv_a >= threshold)[0]

        # Subgradient (average over active peaks)
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            g[js:je + 1] += a[k - np.arange(js, je + 1)][::-1]
        g /= len(active)

        # Multiplicative update
        log_alpha = -eta * g
        log_alpha -= log_alpha.max()
        a_new = a * np.exp(log_alpha)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            eta *= 0.8
            if eta < 1e-8:
                break

        if step % 2000 == 0 and step > 0:
            print(f"    MW[{step}]: score={best_score:.10f}, eta={eta:.6f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: LP Hamiltonian descent + spectral shaping +
    subgradient polish + length exploration for minimizing
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={ps:.10f}")
            best_seq = parent.copy()
            best_score = ps
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        candidates = diverse_init(n_init, rng)
        scores = [evaluate_local(c) for c in candidates]
        best_idx = np.argmin(scores)
        best_seq = candidates[best_idx].copy()
        best_score = scores[best_idx]
        print(f"Init: n={len(best_seq)}, score={best_score:.10f}")

    n_best = len(best_seq)
    print(f"Starting optimization: n={n_best}, score={best_score:.10f}")

    # === Phase 1: LP Hamiltonian Descent (main) ===
    print(f"\n=== Phase 1: LP Hamiltonian (budget 40%) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.40, 700.0)
    if t_alloc > 30:
        print(f"  Alloc: {t_alloc:.0f}s")
        a1, _ = lp_descent(best_seq, t_alloc, t_init=0.03)
        try_update(a1, "lp_main")

    # === Phase 2: Spectral Shaping ===
    print(f"\n=== Phase 2: Spectral (budget 15%) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        a2, _ = spectral_descent(best_seq, t_alloc, lr_init=0.002)
        try_update(a2, "spectral")

    # === Phase 3: LP Refinement (fine) ===
    print(f"\n=== Phase 3: LP Refine (budget 25%) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 20:
        a3, _ = lp_descent(best_seq, t_alloc, t_init=0.01)
        try_update(a3, "lp_fine")

    # === Phase 4: Subgradient + MW Polish ===
    print(f"\n=== Phase 4: Polynomial (budget 10%) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 15:
        a4, _ = subgrad_descent(best_seq, t_alloc * 0.5, lr_init=0.0002)
        try_update(a4, "subgrad")
        rem2 = deadline - time.monotonic()
        if rem2 > 10:
            a5, _ = multiplicative_weights(best_seq, min(rem2, 60.0), eta_init=0.02)
            try_update(a5, "mw")

    # === Phase 5: Length Exploration ===
    print(f"\n=== Phase 5: Length Exploration ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for n_try in [n_best + 150, n_best - 150, int(n_best * 1.3), int(n_best * 0.8)]:
            if n_try < 100 or n_try > 4000:
                continue
            rem_sub = deadline - time.monotonic()
            if rem_sub < 40:
                break
            init = interpolate_seq(best_seq, n_try)
            t_sub = min(rem_sub * 0.35, 80.0)
            a_try, _ = lp_descent(init, t_sub, t_init=0.03)
            try_update(a_try, f"len_{n_try}")

    # === Phase 6: Final LP Polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    rem = deadline - time.monotonic()
    if rem > 15 and best_seq is not None:
        a_fin, _ = lp_descent(best_seq, min(rem * 0.6, 60.0), t_init=0.005)
        try_update(a_fin, "final_lp")
        rem2 = deadline - time.monotonic()
        if rem2 > 10:
            a_fin2, _ = spectral_descent(best_seq, min(rem2, 30.0), lr_init=0.001)
            try_update(a_fin2, "final_spec")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
