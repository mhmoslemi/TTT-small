```python
import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size-1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A*A, fft_size)[:size]

def build_conv_matrix(a):
    """T @ b = conv(a, b)"""
    n = len(a)
    size = 2*n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def solve_lp(a, solver, warm_start_b=None):
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    T = build_conv_matrix(a)
    prob = cp.Problem(cp.Minimize(M), [T @ b <= M, cp.sum(b) == 1])
    prob.solve(solver=solver, warm_start=True)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(b.value, 0)
    b_val /= b_val.sum()
    return b_val

def nesterov_lp_alt(a0, time_limit, solver, n_max_iter=None):
    n = len(a0)
    a = a0 / a0.sum()
    y = a.copy()
    best_a = a.copy()
    best_score = 2*n*fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Store all iterates for Nesterov averaging
    # Actually, to save memory, we can use the running average trick
    
    b_running_sum = np.zeros(n)
    b_count = 0
    
    for t in range(1, n_max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP
        b_star = solve_lp(y, solver)
        if b_star is None:
            continue
        
        b_running_sum += b_star
        b_count += 1
        
        # Nesterov weights
        alpha = 2.0 / (t + 2)
        y = (1 - alpha) * y + alpha * b_star
        y = np.maximum(y, 0)
        y /= y.sum()
        
        # Averaged iterate
        a_avg = b_running_sum / b_count
        a_avg = np.maximum(a_avg, 0)
        if a_avg.sum() > 1e-15:
            a_avg /= a_avg.sum()
        else:
            a_avg = y
        
        # Evaluate
        score_avg = 2*n*fast_convolve(a_avg).max()
        score_y = 2*n*fast_convolve(y).max()
        
        if score_avg < best_score:
            best_score = score_avg
            best_a = a_avg.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if t % 50 == 0:
            print(f"  Iter {t}: best={best_score:.8f}")
    
    return best_a, best_score
```

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    # More efficient construction using broadcasting
    idx = np.arange(2 * n - 1)[:, None] - np.arange(n)[None, :]
    # This gives a (2n-1) x n array where T[i,j] = 1 if 0 <= i-j <= n-1 else 0
    mask = (idx >= 0) & (idx < n)
    # T[i,j] = a[j] if mask[i,j] else 0
    # T[i,j] corresponds to: conv(a,b)[i] = sum_j a[j] * b[i-j]
    # So T[i,j] = a[j] if 0 <= i-j <= n-1 else 0
    T = np.zeros((size, n))
    rows, cols = np.where(mask)
    T[rows, cols] = a[cols]
    return T

def solve_lp_fixed_a(a, solver_name, top_k_constr=None):
    """
    Given fixed a (probability vector), solve:
    min M s.t. conv(a, b) <= M, sum(b) = 1, b >= 0
    Returns optimal b.
    """
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a)
    
    if top_k_constr is not None and top_k_constr < T.shape[0]:
        # Only include top-k constraints (those with largest T row sums)
        row_sums = T.sum(axis=1)
        idx = np.argpartition(row_sums, -top_k_constr)[-top_k_constr:]
        T_sub = T[idx, :]
    else:
        T_sub = T
    
    constraints = [T_sub @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver_name)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

def score_sequence(a):
    """Compute the objective value."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def nesterov_lp_alternating(a0, time_limit, solver_name, max_iter=None):
    """Nesterov-accelerated LP alternating."""
    n = len(a0)
    a = np.maximum(a0, 0)
    a = a / a.sum()
    
    y = a.copy()  # Nesterov lookahead
    best_a = a.copy()
    best_score = score_sequence(a)
    t0 = time.monotonic()
    
    # For Nesterov averaging, we track the running sum
    # a_avg = (1/t) * sum_{s=1}^{t} b_s
    b_sum = a.copy()  # Initialize with a_0
    t = 0
    
    lp_times = []
    stall_count = 0
    prev_best = best_score
    rng = np.random.default_rng(12345)
    
    if max_iter is None:
        max_iter = 100000
    
    for t in range(1, max_iter + 1):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        
        lp_start = time.monotonic()
        b_star = solve_lp_fixed_a(y, solver_name)
        lp_time = time.monotonic() - lp_start
        lp_times.append(lp_time)
        
        if b_star is None:
            continue
        
        b_sum += b_star
        
        # Nesterov update for y
        alpha = 2.0 / (t + 2)
        y = (1.0 - alpha) * y + alpha * b_star
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = best_a.copy()
        else:
            y = y / sy
        
        # Averaged iterate
        a_avg = b_sum / (t + 1)
        a_avg = np.maximum(a_avg, 0)
        sa = a_avg.sum()
        if sa > 1e-15:
            a_avg /= sa
        
        # Evaluate
        s_avg = score_sequence(a_avg)
        s_y = score_sequence(y)
        
        if s_avg < best_score:
            best_score = s_avg
            best_a = a_avg.copy()
            stall_count = 0
        if s_y < best_score:
            best_score = s_y
            best_a = y.copy()
            stall_count = 0
        
        # Stall detection
        if best_score >= prev_best - 1e-12:
            stall_count += 1
        else:
            prev_best = best_score
        
        if stall_count >= 30:
            # Perturb and restart
            noise = 0.02 * rng.normal(size=n)
            y = y + noise
            y = np.maximum(y, 0)
            y /= y.sum()
            stall_count = 0
            b_sum = y.copy()
            t = 0  # Reset counter (approximately)
        
        if t % 20 == 0:
            print(f"    LP-Alt: t={t} ({elapsed:.0f}s) best={best_score:.8f} lp_time={lp_time:.3f}s")
        
        # Check if LP is too slow
        if len(lp_times) >= 20:
            avg_lp = np.mean(lp_times[-20:])
            remaining_iters = (time_limit - elapsed) / avg_lp
            if remaining_iters < 5:
                print(f"    LP too slow, stopping. Avg={avg_lp:.2f}s, ~{remaining_iters:.0f} iters left")
                break
    
    return best_a, best_score

def subgradient_polish(a0, time_limit, n_steps=50000):
    """Subgradient descent on the max-conv objective with backtracking."""
    n = len(a0)
    a = np.maximum(a0, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_sequence(a)
    t0 = time.monotonic()
    
    eta = 0.01
    best_eta = eta
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Active set: peaks within 1% of max
        threshold = max_b * 0.99
        active = np.where(b >= threshold)[0]
        
        # Subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g *= 2.0 / len(active)
        
        # Projected subgradient step
        # We want to minimize in the direction g, projected to the simplex
        # Simple approach: a_new = a - eta * g, then project
        a_new = a - eta * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new /= s_new
        
        score = score_sequence(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            # Try to increase step size
            if eta < 0.1:
                eta *= 1.05
            best_eta = eta
        else:
            # Decrease step size
            eta *= 0.8
            if eta < 0.001:
                break
    
    return best_a, best_score

def zc_initializer(n, seed=42):
    """Initialize using Zadoff-Chu inspired pattern."""
    rng = np.random.default_rng(seed)
    
    # Choose a ZC length that divides n (or is close)
    # ZC sequences work best for prime lengths
    zc_len = 1
    for p in [n, n//2, n//3, n//4, 97, 89, 83, 79, 73, 67, 61, 59, 53, 47, 43, 41, 37, 31, 29, 23, 19, 17, 13, 11, 7]:
        if p >= 3 and n % p == 0:
            zc_len = p
            break
        if p >= 3 and n % p < n * 0.01:
            zc_len = p
            break
    
    k = np.arange(zc_len)
    zc = np.cos(np.pi * k * (k + 1) / zc_len)
    zc_nonneg = 1.0 + zc  # in [0, 2]
    
    # Tile to length n
    tiles = int(np.ceil(n / zc_len))
    a = np.tile(zc_nonneg, tiles)[:n]
    
    # Add graduated noise to break perfect periodicity
    a += 0.05 * rng.random(n)
    a = np.maximum(a, 0.001)
    a /= a.sum()
    return a

def cosine_initializer(n, seed=42):
    """Initialize with a cosine-like window."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    # Various cosine windows
    choice = rng.integers(0, 5)
    if choice == 0:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    elif choice == 1:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.2 * np.cos(2 * np.pi * 7 * x)
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 13 * x + 0.5)
    else:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.2 * np.cos(2 * np.pi * 5 * x + 1.0)
    
    a = np.maximum(a, 0.001)
    a += 0.01 * np.abs(rng.normal(size=n))
    a = np.maximum(a, 0)
    a /= a.sum()
    return a

def uniform_initializer(n):
    return np.ones(n) / n
