**Nesterov's accelerated subgradient method for `min f(a)` subject to `a in Delta` (probability simplex):**
- This is O(1/k^2) convergence rate
- At iteration k: compute subgradient at `y_k`, take gradient step, project onto simplex
- The projection onto the simplex can be done in O(n log n)

Let me code this up:

```python
def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    mu = 0
    sorted_v = np.sort(v)[::-1]
    cssv = np.cumsum(sorted_v)
    rho = np.where(sorted_v * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)
```

**Algorithm:**
```
a_0 = initial (in simplex)
y_0 = a_0
t_1 = 1
for k = 1, 2, ...:
    c = conv(y_k, y_k)
    K = argmax(c)
    w = uniform on K (or weighted by values)
    g = 2 * (w cross-correlate y_k)  # subgradient
    a_{k+1} = project_simplex(y_k - step_k * g)
    t_{k+1} = (1 + sqrt(1 + 4*t_k^2)) / 2
    y_{k+1} = a_{k+1} + (t_k - 1)/t_{k+1} * (a_{k+1} - a_k)
```

import numpy as np
import cvxpy as cp
import time

from scipy.optimize import minimize
from scipy.sparse import csr_matrix

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def cross_convolve(a, b):
    """Cross-convolution of a and b (both length n) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def get_lp_solver():
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x[0] <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def solve_active_lp(y, active_set, solver, n):
    """Solve LP: min M s.t. conv(y,b)[k] <= M for k in active_set, sum b = 1, b >= 0."""
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [cp.sum(b_var) == 1]
    # Build the active constraints
    for k in active_set:
        # conv(y,b)[k] = sum_j y[j] * b[k-j] for valid j
        # This is a linear function of b
        coeffs = []
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                coeffs.append((j, y[j]))
        if not coeffs:
            continue
        # Build the expression: sum_j y[j] * b[k-j]
        expr = cp.sum([y_j * b_var[j] for j, y_j in coeffs])
        constraints.append(expr <= M_var)
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    for s in [solver, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_b = b_val.sum()
                if s_b < 1e-15:
                    return None, None
                b_val = b_val / s_b
                M_val = float(max(0, M_var.value))
                return b_val, M_val
        except Exception:
            continue
    return None, None

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Cross-convolution of a and b (both length n) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            if stall >= 3:
                # Restart active set
                c_re = fast_convolve(y)
                active_list = list(np.argsort(c_re)[::-1][:P_init])
                stall = 0

        if (it + 1) % 10000 == 0 and (time.monotonic() - last_print) > 10:
            last_print = time.monotonic()
            print(f"    LP iter={it+1} ({time.monotonic()-t0:.0f}s): best={best_score:.8f} |A|={len(active_list)}")

    return best_a, best_score


def multiplicative_weights_fast(a0, time_limit, n_steps=200000, eta0=0.05):
    """Fast multiplicative weights with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 1e-3)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        g = compute_subgradient_vectorized(a, active)
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.02, 0.5)
        else:
            eta *= 0.85
            if eta < 1e-10:
                break
    return best_a, best_score


def nesterov_subgradient(a0, time_limit, n):
    """Nesterov accelerated subgradient method."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    y = a.copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_val = fast_convolve(a).max()
    t_nest = 1.0
    t0 = time.monotonic()

    for k in range(50000000):
        if time.monotonic() - t0 > time_limit:
            break
        c = fast_convolve(y)
        max_c = c.max()
        threshold = max_c * (1.0 - 5e-4)
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])

        g = compute_subgradient_vectorized(y, active)
        step = 1.0 / np.sqrt(k + 1)
        a_new = project_simplex(y - step * g)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest ** 2)) / 2.0
        y_new = a_new + (t_nest - 1.0) / t_new * (a_new - a_prev)
        y_new = project_simplex(y_new)

        a_prev = a_new
        a = a_new
        y = y_new
        t_nest = t_new

        val = fast_convolve(a).max()
        if val < best_val:
            best_val = val
            best_a = a.copy()

        if (k + 1) % 50000 == 0:
            print(f"    SubGrad k={k+1} ({time.monotonic()-t0:.0f}s): best={best_val:.8f}")

    return best_a, best_val


def lse_logsumexp_obj_grad(z, n, T):
    """Compute LogSumExp objective and gradient w.r.t. z for softmax parameterization."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        a = np.ones(n) / n
    else:
        a = exp_z / s
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T
    scaled_shifted = scaled - scaled.max()
    exp_terms = np.exp(scaled_shifted)
    lse = max_b + T * (scaled.max() + np.log(exp_terms.sum() + 1e-300))
    obj = 2.0 * n * lse
    w = exp_terms / (exp_terms.sum() + 1e-300)
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lse_lbfgs_refine(a0, time_limit, T_stages, maxiter_per_stage=500):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_in):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = lse_logsumexp_obj_grad(z_in, n, stage_T)
            return obj

        def jac(z_in):
            if time.monotonic() > deadline:
                return np.zeros(len(z_in))
            _, g = lse_logsumexp_obj_grad(z_in, n, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-25,
                         'gtol': 1e-18, 'maxfun': maxiter_per_stage * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return exp_z / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Fast local evaluation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 5 * x + 0.5)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = 1.0 / (1.0 + 3.0 * np.abs(x - 0.5))
    elif choice == 8:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 4
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def perturb_sequence(a, rng, strength=0.03):
    """Create a perturbed version with structured frequency components."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq1 = rng.integers(1, 8)
    freq2 = rng.integers(2, 12)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    pert = strength * (np.cos(freq1 * x + p1) + 0.5 * np.sin(freq2 * x + p2))
    a_pert = np.maximum(a + pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Main improvements: fast LP game with scipy HiGHS and adaptive active set,
    multi-length exploration, Nesterov subgradient, LSE L-BFGS polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_seq = None
    best_score = float('inf')
    print(f"Budget: {budget_s:.0f}s, deadline in {budget_s - 30:.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Quick MW warmup ===
    print(f"\n=== Phase 1: MW n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.03, 40.0)
    if t1 > 10:
        a_mw, _ = multiplicative_weights_fast(best_seq, t1, n_steps=100000, eta0=0.05)
        try_update(a_mw, "mw_warmup")

    # === Phase 2: Main LP Game (active set + HiGHS) ===
    print(f"\n=== Phase 2: LP Game n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.35, 550.0)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s")
        a_lp, _ = lp_game_fast(best_seq, t2, len(best_seq), P_init=15, P_max=50)
        try_update(a_lp, "lp_main")

    # === Phase 3: Nesterov Subgradient (backup / complement) ===
    rem = deadline - time.monotonic()
    if rem > 120:
        t3 = min(rem * 0.12, 180.0)
        print(f"\n=== Phase 3: SubGrad n={len(best_seq)} (score={best_score:.8f}) ===")
        a_sg, _ = nesterov_subgradient(best_seq, t3, len(best_seq))
        try_update(a_sg, "subgrad")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (score={best_score:.8f}) ===")
    lengths_to_try = [1500, 1800, 2000]
    for ln in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 150:
            break
        if len(best_seq) == ln:
            continue
        t_ln = min(rem * 0.18, 250.0)
        if t_ln < 40:
            continue
        print(f"  Trying n={ln}, budget={t_ln:.0f}s")
        init_ln = interpolate_sequence(best_seq, ln)
        # Quick MW
        a_ln_mw, _ = multiplicative_weights_fast(init_ln, min(t_ln * 0.15, 30.0),
                                                  n_steps=50000, eta0=0.05)
        try_update(a_ln_mw, f"mw_n{ln}")
        # LP Game
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.5, t_ln * 0.75)
        if t_lp > 30:
            start = a_ln_mw if a_ln_mw is not None else init_ln
            a_ln_lp, _ = lp_game_fast(start, t_lp, ln, P_init=12, P_max=35)
            try_update(a_ln_lp, f"lp_n{ln}")

    # === Phase 5: LSE L-BFGS polish ===
    print(f"\n=== Phase 5: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t5 = min(rem * 0.20, 300.0)
        T_stages = [8.0, 4.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        print(f"  {len(T_stages)} stages, budget={t5:.0f}s")
        a_lse = lse_lbfgs_refine(best_seq, t5, T_stages, maxiter_per_stage=400)
        try_update(a_lse, "lse_fine")

    # === Phase 6: Perturbation + re-optimize ===
    print(f"\n=== Phase 6: Perturbation escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        for trial in range(3):
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            a_pert = perturb_sequence(best_seq, rng, strength=0.01 + 0.005 * trial)
            t_sub = min(rem * 0.3, 80.0)
            a_opt, _ = lp_game_fast(a_pert, t_sub, len(a_pert), P_init=12, P_max=30)
            try_update(a_opt, f"pert{trial}")

    # === Phase 7: Final LP polish ===
    print(f"\n=== Phase 7: Final LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t7 = min(rem * 0.4, 150.0)
        a_fin, _ = lp_game_fast(best_seq, t7, len(best_seq), P_init=20, P_max=60,
                                check_interval=20)
        try_update(a_fin, "final_lp")

    # === Phase 8: Final MW polish ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t8 = min(rem * 0.5, 40.0)
        a_fin_mw, _ = multiplicative_weights_fast(best_seq, t8, n_steps=100000, eta0=0.02)
        try_update(a_fin_mw, "final_mw")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
