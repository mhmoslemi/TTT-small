import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def get_lp_solver():
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(a_fixed, solver, T_mat, K=None):
    """LP: min M s.t. T@b <= M, sum(b)=1, b>=0. Optional: only top-K constraints."""
    n = len(a_fixed)
    n_constraints = T_mat.shape[0]
    
    if K is not None and K < n_constraints:
        # Only use top-K constraints (by row norm of T)
        row_norms = np.sqrt((T_mat.multiply(T_mat)).sum(axis=1)).flatten()
        top_k = np.argsort(row_norms)[-K:]
        T_reduced = T_mat[top_k, :]
    else:
        T_reduced = T_mat
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_reduced @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers_to_try = [solver, cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                sb = b_val.sum()
                if sb < 1e-15:
                    return None, None
                b_val = b_val / sb
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def build_T_matrix(a):
    """Build dense Toeplitz-like matrix T where T@b = convolve(a,b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T


def multiplicative_weights_improved(a0, time_limit, n_steps, soft_T=0.01):
    """MW with soft-max weighting and adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = 0.05

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        # Soft-max weighting
        b_max = b.max()
        b_mean = b.mean()
        # Weight proportional to how much above mean
        excess = np.maximum(b - b_mean, 0)
        w = excess / (excess.sum() + 1e-30)
        
        # Compute gradient
        g = np.zeros(n)
        # Use FFT-based gradient computation for speed
        # g[j] = sum_k w[k] * a[k-j] (for valid k-j)
        # This is convolve(w, a_reversed)
        w_ext = np.zeros(2*n-1)
        w_ext[:len(w)] = w
        a_flip = a[::-1]
        g = np.convolve(w_ext, a_flip, mode='valid')[:n]
        
        # Multiplicative update
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.02, 0.2)
        else:
            eta *= 0.7
            if eta < 1e-10:
                break

    return best_a, best_score

def spectral_refine(a0, time_limit, n_steps=1000, gamma=0.1):
    """FFT-based spectral flattening."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    N = 1 << (2*n-2).bit_length()  # FFT size >= 2n-1
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, N)
        # Reduce non-DC components
        A_abs = np.abs(A)
        A_dc = A_abs[0]
        # Target: reduce |A[k]| for k>0 towards a fraction of DC
        target = 0.0  # We want A[k] -> 0 for k > 0
        # Smooth reduction
        for k in range(1, len(A)):
            if A_abs[k] > target:
                A[k] *= (1.0 - gamma * A_abs[k] / A_dc)
        
        a_new = np.fft.irfft(A, N)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
            score = 2.0 * n * fast_convolve(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                a = a_new
        # Decay gamma
        gamma *= 0.999
    
    return best_a, best_score


def coordinate_descent_refine(a0, time_limit, n_sweeps=50):
    """Simple coordinate descent: adjust one element at a time."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        for j in range(n):
            # Try a few multipliers
            old_val = a[j]
            a[j] = 0
            s_rest = a.sum()
            for mult in [0.0, 0.5, 0.8, 1.0, 1.2, 1.5, 2.0]:
                a[j] = old_val * mult
                s_new = a.sum()
                if s_new < 1e-15:
                    continue
                a_norm = a / s_new
                c = fast_convolve(a_norm)
                score = 2.0 * n * c.max()
                if score < best_score:
                    best_score = score
                    a = a_norm.copy()
            # Restore and normalize
            a[j] = old_val
            a = a / a.sum()
    
    return a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Fast local evaluation (assumes a >= 0, sum > 0)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_T_rows(a, rows):
    """Build rows of the convolution matrix T where T@b = convolve(a,b)."""
    n = len(a)
    T = np.zeros((len(rows), n))
    for i, k in enumerate(rows):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        T[i, js:je+1] = a[k - np.arange(js, je + 1)]
    return T


def solve_lp_core(a_fixed, rows):
    """Solve LP: min M s.t. T_rows @ b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    K = len(rows)

    T = build_T_rows(a_fixed, rows)

    # Variables: [b_0, ..., b_{n-1}, M]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    # Inequality: T @ b - M <= 0  =>  [T | -1] @ [b; M] <= 0
    A_ub = np.hstack([T, -np.ones((K, 1))])
    b_ub = np.zeros(K)

    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
    except Exception:
        return None, None

    if not result.success:
        return None, None

    b_val = np.maximum(result.x[:n], 0)
    sb = b_val.sum()
    if sb < 1e-15:
        return None, None
    b_val = b_val / sb
    M_val = float(max(0.0, result.x[n]))
    return b_val, M_val


def select_top_rows(a_fixed, K):
    """Select K most important constraint rows based on potential impact."""
    n = len(a_fixed)
    size = 2 * n - 1
    # Row k has entries a[k-j] for valid j. Impact ~ L1 norm of row.
    # Compute row norms efficiently
    row_norms = np.zeros(size)
    for j in range(n):
        row_norms[j:j+n] += np.abs(a_fixed)
    top_k = np.argsort(row_norms)[-K:]
    return top_k


def multiplicative_weights_v2(a0, time_limit, n_steps=500000):
    """MW with soft-max excess weighting and adaptive step size."""
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = 0.08
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        b_mean = b.mean()
        # Excess weighting: weight proportional to how much above mean
        excess = np.maximum(b - b_mean, 0)
        total_excess = excess.sum()
        if total_excess < 1e-30:
            w = np.ones(len(b)) / len(b)
        else:
            w = excess / total_excess

        # Compute gradient: g[j] = sum_k w[k] * a[k-j] (valid range)
        # This is a cross-correlation: g = correlate(w, a)
        g = np.zeros(n)
        # Use FFT for speed
        fft_size = 1 << (len(b) + n - 1).bit_length()
        W = rfft(w, fft_size)
        A_flip = rfft(a[::-1], fft_size)

def spectral_refine(a0, time_limit, gamma=0.03):
    """FFT-based spectral shaping to flatten autocorrelation."""
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    N = 1 << (2 * n - 2).bit_length()

    step = 0
    while time.monotonic() - t0 < time_limit:
        step += 1
        A = np.fft.rfft(a, N)
        A_dc = abs(A[0])
        if A_dc < 1e-30:
            break

        # Reduce non-DC components
        A_modified = A.copy()
        for k in range(1, len(A_modified)):
            A_modified[k] *= (1.0 - gamma * abs(A[k]) / A_dc)

        a_new = np.fft.irfft(A_modified, N)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            gamma *= 0.95
            if gamma < 1e-6:
                break
        if step > 50000:
            break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 5 * x + 0.5)
        a = np.maximum(a, 0)
    else:
        a = 1.0 / (1.0 + 3.0 * np.abs(x - 0.5))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses fast scipy-based LP, improved MW, spectral refinement,
    and multi-length exploration with adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_seq = None
    best_score = float('inf')
    print(f"Budget: {budget_s:.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(arr, label):
        nonlocal best_seq, best_score
        if arr is None:
            return
        arr = np.maximum(np.asarray(arr, dtype=np.float64), 0)
        s = arr.sum()
        if s < 0.01:
            return
        arr = arr / s
        sc = evaluate_local(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(arr)})")

    # === Phase 1: Fast MW ===
    print(f"\n=== Phase 1: MW n={len(best_seq)} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.04, 50.0)
    if t1 > 10:
        a_mw, _ = multiplicative_weights_v2(best_seq, t1, n_steps=300000)
        try_update(a_mw, "mw_1")

    # === Phase 2: LP alternating (main workhorse) ===
    print(f"\n=== Phase 2: LP n={len(best_seq)} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.40, 600.0)
    if t2 > 60:
        n_cur = len(best_seq)
        K_red = min(2 * n_cur - 1, 1500)
        print(f"  Budget: {t2:.0f}s, K={K_red}")
        a_lp, _ = lp_alternate_v2(best_seq, t2, K_reduce=K_red, n_max_iters=1000000)
        try_update(a_lp, "lp_1")

    # === Phase 3: Spectral refinement ===
    print(f"\n=== Phase 3: Spectral (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t3 = min(rem * 0.04, 30.0)
        a_spec, _ = spectral_refine(best_seq, t3, gamma=0.02)
        try_update(a_spec, "spectral")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (score={best_score:.8f}) ===")
    lengths_to_try = []
    n_best = len(best_seq)
    for ln in [1000, 1200, 1500, 2000]:
        if ln != n_best:
            lengths_to_try.append(ln)

    for ln in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 100:
            break
        t_ln = min(rem * 0.18, 200.0)
        if t_ln < 40:
            continue
        print(f"  Trying n={ln}, budget={t_ln:.0f}s")
        init_ln = interpolate_sequence(best_seq, ln)
        # Quick MW
        a_ln_mw, _ = multiplicative_weights_v2(init_ln, t_ln * 0.15, n_steps=50000)
        try_update(a_ln_mw, f"mw_n{ln}")
        # LP
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.55, t_ln * 0.65)
        if t_lp > 30:
            K_ln = min(2 * ln - 1, 1500)
            a_ln_lp, _ = lp_alternate_v2(a_ln_mw, t_lp, K_reduce=K_ln, n_max_iters=200000)
            try_update(a_ln_lp, f"lp_n{ln}")

    # === Phase 5: LP polish on current best ===
    print(f"\n=== Phase 5: LP polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t5 = min(rem * 0.35, 300.0)
        n_cur = len(best_seq)
        K_cur = min(2 * n_cur - 1, 2000)
        print(f"  Budget: {t5:.0f}s, K={K_cur}")
        a_lp2, _ = lp_alternate_v2(best_seq, t5, K_reduce=K_cur, n_max_iters=500000)
        try_update(a_lp2, "lp_polish")

    # === Phase 6: Final MW polish ===
    print(f"\n=== Phase 6: Final MW (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t6 = min(rem * 0.3, 40.0)
        a_mw2, _ = multiplicative_weights_v2(best_seq, t6, n_steps=100000)
        try_update(a_mw2, "mw_final")

    # === Phase 7: Spectral + MW combo ===
    print(f"\n=== Phase 7: Spectral+MW (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t7a = min(rem * 0.15, 20.0)
        t7b = min(rem * 0.25, 30.0)
        a_sp, _ = spectral_refine(best_seq, t7a, gamma=0.01)
        a_mw3, _ = multiplicative_weights_v2(a_sp, t7b, n_steps=50000)
        try_update(a_mw3, "spec_mw")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
