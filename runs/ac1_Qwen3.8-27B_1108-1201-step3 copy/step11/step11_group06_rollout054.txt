        return 1e15, np.zeros(len(z))
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        a = np.ones(n) / n
    else:
        a = exp_z / s
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T
    scaled_shifted = scaled - scaled.max()
    exp_terms = np.exp(scaled_shifted)
    w_sum = exp_terms.sum()
    if w_sum < 1e-300:
        lse = max_b + T * len(b) * np.log(len(b))
        w = np.ones(len(b)) / len(b)
    else:
        lse = max_b + T * (scaled.max() + np.log(w_sum))
        w = exp_terms / w_sum
    obj = 2.0 * n * lse
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage=500, seed=0):
    """L-BFGS with LogSumExp smoothing and temperature annealing + restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    z = np.log(np.maximum(a, 1e-300)).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    best_score = 2.0 * n * fast_convolve(a).max()
    best_z = z.copy()
    rng = np.random.default_rng(seed)

    for stage_idx, stage_T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        def func(z):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = lse_objective_and_grad(z, n, stage_T, deadline)
            return obj

        def jac(z):
            if time.monotonic() > deadline:
                return np.zeros(len(z))
            _, g = lse_objective_and_grad(z, n, stage_T, deadline)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-30,
                         'gtol': 1e-20, 'maxfun': maxiter_per_stage * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

        # Evaluate at current z
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a_curr = exp_z / exp_z.sum()
        a_curr = np.maximum(a_curr, 0)
        s_curr = a_curr.sum()
        if s_curr > 1e-15:
            a_curr = a_curr / s_curr
        score = 2.0 * n * fast_convolve(a_curr).max()
        if score < best_score:
            best_score = score
            best_z = z.copy()

        # Periodic restart to escape local minima at low temperatures
        if stage_T < 0.1 and stage_idx > 0:
            for restart in range(3):
                if time.monotonic() - t0 > time_limit:
                    break
                z_pert = best_z + rng.normal(0, 0.05, size=n)
                try:
                    res2 = minimize(
                        fun=func, x0=z_pert, jac=jac, method='L-BFGS-B',
                        options={'maxiter': 200, 'ftol': 1e-25,
                                 'gtol': 1e-15, 'maxfun': 40000,
                                 'disp': False}
                    )
                    z_pert = res2.x
                    z_shifted = z_pert - z_pert.max()
                    exp_z = np.exp(z_shifted)
                    a_pert = exp_z / exp_z.sum()
                    a_pert = np.maximum(a_pert, 0)
                    s_pert = a_pert.sum()
                    if s_pert > 1e-15:
                        a_pert = a_pert / s_pert
                    score_pert = 2.0 * n * fast_convolve(a_pert).max()
                    if score_pert < best_score:
                        best_score = score_pert
                        best_z = z_pert.copy()
                        z = z_pert.copy()
                except Exception:
                    pass

    z_shifted = best_z - best_z.max()
    exp_z = np.exp(z_shifted)
    a_out = exp_z / exp_z.sum()
    return np.maximum(a_out, 0)


def freq_domain_mw(a0, time_limit, max_iters=300000, lr0=0.005):
    """
    Frequency-domain multiplicative weights.
    Exploits the structure: gradient of LSE w.r.t. FFT coefficients is multiplicative.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    N_fft = 1 << (2 * n - 1).bit_length()
    conv_len = 2 * n - 1
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    T = 1.0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / (max_iters * 0.7))
        T = 1.0 * (0.005) ** progress
        lr = lr0 / (1.0 + 0.05 * it)

        # FFT of padded signal
        a_pad = np.zeros(N_fft)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)

        # Convolution via frequency domain
        C = A * A
        c_full = np.fft.ifft(C).real[:conv_len]
        max_c = c_full.max()

        # LSE weights
        scaled = (c_full - max_c) / T
        scaled -= scaled.max()
        exp_w = np.exp(scaled)
        w_sum = exp_w.sum()
        if w_sum < 1e-300:
            w = np.ones(conv_len) / conv_len
        else:
            w = exp_w / w_sum

        # Gradient direction: G = IFFT of padded weights (in complex domain)
        w_ext = np.zeros(N_fft)
        w_ext[:conv_len] = w
        G = np.fft.ifft(w_ext)

        # Multiplicative update in frequency domain
        # dLSE/dA[k] = 2 * A[k] * conj(G[k]) (for real objective)
        # Use: A_new = A * exp(-lr * 2 * real(A * conj(G)) / |A|) for stability
        # Simpler: A_new = A * exp(-lr * 2 * G.real) (works when A is real, e.g., symmetric)
        # For complex A, use: A_new = A * exp(-lr * 2 * conj(G) * |A|^2 / |A|)
        # Let's use the safe version: compute gradient properly
        grad_A = 2.0 * A * np.conj(G)
        # Real-valued gradient step: A_new = A - lr * Re(grad_A / A) * A (multiplicative)
        # For stability, use: A_new = A * exp(-lr * 2 * G.real)
        update = -2.0 * lr * G.real
        update -= update.max()  # normalize for stability
        A_new = A * np.exp(update)

        # Convert back to time domain
        a_new = np.fft.ifft(A_new).real[:n].copy()
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Blend
            a_new = 0.7 * a + 0.3 * a_new
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
                score = 2.0 * n * fast_convolve(a_new).max()
                if score < best_score:
                    best_score = score
                    best_a = a_new.copy()
                    a = a_new

        if it % 10000 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            print(f"    FreqMW {elapsed:.0f}s: score={best_score:.8f} T={T:.5f}")

    return best_a, best_score


def subspace_fourier_optimize(a0, time_limit, n_modes=20, seed=0):
    """
    Optimize in a low-dimensional Fourier subspace.
    Parameterize a[j] = (1/n) * |1 + sum_k c_k * exp(2*pi*i*k*j/n)|^2
    This ensures non-negativity automatically.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()

    # Get initial frequency coefficients from a0
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s_init = a_init.sum()
    if s_init < 1e-15:
        a_init = np.ones(n) / n
        s_init = 1.0
    a_init = a_init / s_init

    # Try different parameterizations
    # Option 1: a[j] = base * (1 + sum c_k cos(2*pi*k*j/n + phi_k))^2
    # Option 2: a[j] = |sum_k d_k exp(2*pi*i*k*j/n)|^2 / (n * sum|d_k|^2)

    # Use option 2: a[j] = |FFT_coeff|^2 normalized
    # Get initial FFT coefficients
    A_init = np.fft.rfft(a_init)

    # Keep first n_modes+1 bins (DC + n_modes harmonics)
    n_bins = min(n_modes + 1, n // 2 + 1)
    d = A_init[:n_bins].copy()

    def objective(params):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        # params: [Re(d0), Im(d1), Re(d1), Im(d2), Re(d2), ...]
        # Reconstruct d
        d_new = np.zeros(n_bins, dtype=np.complex128)
        d_new[0] = params[0]
        for k in range(1, n_bins):
            d_new[k] = params[2 * k - 1] + 1j * params[2 * k]
        # Reconstruct a via IFFT
        A_full = np.zeros(n, dtype=np.complex128)
        A_full[:n_bins] = d_new
        # Enforce Hermitian symmetry
        if n % 2 == 0:
            A_full[n - n_bins + 1:] = np.conj(d_new[1:n_bins][::-1])
        else:
            A_full[n - n_bins + 1:] = np.conj(d_new[1:n_bins][::-1])
        a_new = np.fft.ifft(A_full).real
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            return 1e15
        a_new = a_new / s
        b = fast_convolve(a_new)
        return 2.0 * n * b.max()

    # Initial parameter vector
    params0 = np.zeros(2 * n_bins - 1)
    params0[0] = d[0].real
    for k in range(1, n_bins):
        params0[2 * k - 1] = d[k].real
        params0[2 * k] = d[k].imag

    # Optimize
    best_params = params0.copy()
    best_obj = objective(params0)

    try:
        res = minimize(
            fun=objective, x0=params0, method='L-BFGS-B',
            options={'maxiter': 2000, 'ftol': 1e-20, 'gtol': 1e-15,
                     'maxfun': 100000, 'disp': False}
        )
        if objective(res.x) < best_obj:
            best_obj = objective(res.x)
            best_params = res.x
    except Exception:
        pass

    # Reconstruct from best params
    d_best = np.zeros(n_bins, dtype=np.complex128)
    d_best[0] = best_params[0]
    for k in range(1, n_bins):
        d_best[k] = best_params[2 * k - 1] + 1j * best_params[2 * k]
    A_full = np.zeros(n, dtype=np.complex128)
    A_full[:n_bins] = d_best
    if n % 2 == 0:
        A_full[n - n_bins + 1:] = np.conj(d_best[1:n_bins][::-1])
    else:
        A_full[n - n_bins + 1:] = np.conj(d_best[1:n_bins][::-1])
    a_out = np.fft.ifft(A_full).real
    a_out = np.maximum(a_out, 0)
    s_out = a_out.sum()
    if s_out < 1e-15:
        return a_init.copy(), best_obj
    a_out = a_out / s_out
    score = 2.0 * n * fast_convolve(a_out).max()
    return a_out, min(score, best_obj)


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_palindromic(a):
    """Make a sequence palindromic."""
    n = len(a)
    a_pal = (a + a[::-1]) / 2.0
    a_pal = np.maximum(a_pal, 0)
    s = a_pal.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_pal / s


def perturb_sequence(a, rng, strength=0.02):
    """Perturb with structured frequency components."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq1 = rng.integers(1, 10)
    freq2 = rng.integers(2, 15)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    pert = strength * (np.cos(freq1 * x + p1) + 0.5 * np.sin(freq2 * x + p2))
    a_pert = np.maximum(a + pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {budget_s:.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = np.cos(np.pi * x / 2) ** 2
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Quick MW warmup ===
    print(f"\n=== Phase 1: MW warmup n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.04, 50.0)
    if t1 > 10:
        a_mw, _ = multiplicative_weights(best_seq, t1, max_iters=300000, eta0=0.04)
        try_update(a_mw, "mw_warm")

    # === Phase 2: Frequency-domain MW (novel) ===
    print(f"\n=== Phase 2: FreqMW n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.08, 100.0)
    if t2 > 20:
        a_fm, _ = freq_domain_mw(best_seq, t2, max_iters=100000, lr0=0.003)
        try_update(a_fm, "freqmw")

    # === Phase 3: Subspace Fourier optimization ===
    print(f"\n=== Phase 3: Subspace Fourier n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.06, 60.0)
    if t3 > 15:
        for nm in [15, 30, 50]:
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            a_sub, _ = subspace_fourier_optimize(best_seq, min(t3 / 3, rem), n_modes=nm)
            try_update(a_sub, f"subspace_{nm}")

    # === Phase 4: Main LP alternating ===
    print(f"\n=== Phase 4: LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.35, 550.0)
    if t4 > 60:
        print(f"  Budget: {t4:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t4, solver, max_iters=500000)
        try_update(a_lp, "lp_main")

    # === Phase 5: LSE L-BFGS fine polish ===
    print(f"\n=== Phase 5: LSE L-BFGS n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t5 = min(rem * 0.18, 250.0)
        T_stages = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
        a_lse = lbgfs_refine(best_seq, t5, T_stages, maxiter_per_stage=600, seed=seed)
        try_update(a_lse, "lse_main")

    # === Phase 6: Palindromic refinement ===
    print(f"\n=== Phase 6: Palindromic LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t6 = min(rem * 0.12, 150.0)
        a_pal = make_palindromic(best_seq)
        a_pal_lp, _ = nesterov_lp_refine(a_pal, t6, solver, max_iters=200000)
        try_update(a_pal_lp, "pal_lp")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-length (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    for ln in [1500, 2000]:
        rem = deadline - time.monotonic()
        if rem < 120:
            break
        if len(best_seq) == ln:
            continue
        t_ln = min(rem * 0.18, 250.0)
        if t_ln < 40:
            continue
        print(f"  Trying n={ln}, budget={t_ln:.0f}s")
        init_ln = interpolate_sequence(best_seq, ln)
        # Quick MW
        a_ln_mw, _ = multiplicative_weights(init_ln, t_ln * 0.15, max_iters=50000, eta0=0.04)
        try_update(a_ln_mw, f"mw_n{ln}")
        # LP
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.4, t_ln * 0.6)
        if t_lp > 30:
            a_ln_lp, _ = nesterov_lp_refine(a_ln_mw if a_ln_mw is not None else init_ln, t_lp, solver, max_iters=100000)
            try_update(a_ln_lp, f"lp_n{ln}")
        # LSE
        rem = deadline - time.monotonic()
        if rem > 30:
            T_short = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05]
            a_ln_lse = lbgfs_refine(a_ln_lp if a_ln_lp is not None else init_ln, min(rem * 0.3, 60), T_short, maxiter_per_stage=300)
            try_update(a_ln_lse, f"lse_n{ln}")

    # === Phase 8: Perturbation escape + re-optimize ===
    print(f"\n=== Phase 8: Perturbation escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        for trial in range(5):
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            strength = 0.01 + 0.01 * trial
            a_pert = perturb_sequence(best_seq, rng, strength=strength)
            t_sub = min(rem * 0.2, 50.0)
            a_opt, _ = nesterov_lp_refine(a_pert, t_sub, solver, max_iters=50000)
            try_update(a_opt, f"pert{trial}")

    # === Phase 9: Final LP polish on global best ===
    print(f"\n=== Phase 9: Final LP polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t9 = min(rem * 0.25, 100.0)
        a_fin, _ = nesterov_lp_refine(best_seq, t9, solver, max_iters=100000)
        try_update(a_fin, "final_lp")

    # === Phase 10: Final LSE polish ===
    print(f"\n=== Phase 10: Final LSE (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t10 = min(rem * 0.3, 60.0)
        T_final = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        a_fin_lse = lbgfs_refine(best_seq, t10, T_final, maxiter_per_stage=400, seed=seed + 1)
        try_update(a_fin_lse, "final_lse")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
