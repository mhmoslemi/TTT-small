```python
import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_crosscorr(w, a):
    """Cross-correlation (w ⊛ a)[i] = sum_j w[i+j] * a[j] for i=0..n-1."""
    n = len(a)
    w_rev = w[::-1]  # Reverse w
    # Full convolution of w_rev (length 2n-1) and a (length n)
    conv_size = (2*n-1) + n - 1  # = 3n-2
    fft_size = 1 << (conv_size - 1).bit_length()
    W = rfft(w_rev, fft_size)
    A = rfft(a, fft_size)
    conv = irfft(W * A, fft_size)[:conv_size]
    # We need indices i + (2n-2) for i = 0, ..., n-1
    # i.e., indices from 2n-2 to 3n-3
    return conv[(2*n-2):(2*n-2)+n]

def simplex_project(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cv - 1))[0][-1]
    theta = (cv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def compute_grad(a, beta):
    """Compute gradient of (1/beta) log(sum exp(beta * c)) where c = conv(a,a)."""
    n = len(a)
    c = fast_conv(a)
    # Numerically stable softmax
    c_shifted = c - c.max()
    exp_c = np.exp(beta * c_shifted)
    w = exp_c / exp_c.sum()
    # Cross-correlation
    corr = fast_crosscorr(w, a)
    grad = 2.0 * corr
    return grad, c

def score_of(a):
    """Compute the objective value."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_norm = a / s
    n = len(a_norm)
    c = fast_conv(a_norm)
    return 2.0 * n * c.max()
```

        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - eta * g
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        a = a_new

        if no_improve > 10000:
            if time.monotonic() - t0 > time_limit - 10:
                break
            noise = rng.standard_normal(n) * 0.01
            a = a + noise
            a = simplex_project(a)
            g_sq_sum = 1e-10
            no_improve = 0

        if t_count % 10000 == 0:
            print(f"    MirDesc: {t_count} iters, beta={beta:.1f}, score={best_score:.8f} ({elapsed:.0f}s)")

    return best_a, best_score


def subgradient_opt(a0, time_limit, n_active=5):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    t_count = 0
    avg_g = np.zeros(n)
    lr = 0.01
    no_improve = 0
    rng = np.random.default_rng(11111)

    while time.monotonic() - t0 < time_limit - 2:
        t_count += 1
        c = fast_conv(a)
        max_c = c.max()
        threshold = max_c - 0.01 * max_c
        active = np.where(c >= threshold)[0]
        if len(active) > n_active:
            peaks = np.argsort(c)[::-1][:n_active]
        else:
            peaks = active

        g = np.zeros(n)
        for k in peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(peaks)

        eta = lr / (np.sqrt(t_count) + 1.0)
        avg_g = 0.9 * avg_g + 0.1 * g

        a_new = a - eta * avg_g
        a_new = simplex_project(a_new)

        sc = compute_score(a_new)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        a = a_new

        if no_improve > 5000:
            if time.monotonic() - t0 > time_limit - 10:
                break
            noise = rng.standard_normal(n) * 0.01
            a = a + noise
            a = simplex_project(a)
            no_improve = 0

        if t_count % 10000 == 0:
            print(f"    SubGrad: {t_count} iters, score={best_score:.8f} ({time.monotonic()-t0:.0f}s)")

    return best_a, best_score


def coord_descent_polish(a0, time_limit):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(77777)

    factor_list = [0.999, 0.998, 0.995, 0.99, 0.98, 0.97, 0.95, 1.01, 1.02, 1.03, 1.05]

    for step in range(500000):
        if time.monotonic() - t0 > time_limit - 1:
            break

        c = fast_conv(a)
        max_c = c.max()
        threshold = max_c * 0.9999
        peaks = np.where(c >= threshold)[0]

        improved = False
        for peak_k in peaks[:5]:
            if improved:
                break
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_m = min(12, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

            for pos in top_positions:
                if a[pos] < 1e-12:
                    continue
                for factor in factor_list:
                    a_trial = a.copy()
                    diff = a[pos] * (1.0 - factor)
                    a_trial[pos] *= factor
                    if abs(diff) < 1e-16:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] += diff * a_trial[mask] / others_sum
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        break
                    a_trial = a_trial / st
                    score = compute_score(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 500:
                c = fast_conv(a)
                peak_k = int(np.argmax(c))
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[peak_k - idx]
                top_positions = idx[np.argsort(contribs)[::-1][:6]]
                a_trial = a.copy()
                for pos in top_positions:
                    if a_trial[pos] > 1e-12:
                        a_trial[pos] *= 0.999
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-15:
                    a_trial = a_trial / st
                    score = compute_score(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                    else:
                        if stall > 3000:
                            noise = rng.standard_normal(n) * 0.005
                            a = a + noise
                            a = simplex_project(a)
                            stall = 0
                        elif stall > 10000:
                            break
                if st <= 1e-15:
                    break
            if stall % 2000 == 0:
                print(f"    CoordDesc: {step} steps, stall={stall}, score={best_score:.8f} ({time.monotonic()-t0:.0f}s)")
        else:
            stall = 0

        if (step + 1) % 10000 == 0:
            print(f"    CoordDesc: {step+1} steps, score={best_score:.8f} ({time.monotonic()-t0:.0f}s)")

    return best_a, best_score
