def fast_convolve(a):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver=None, max_iters=500000):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 15:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} M_lp={M_lp:.6f}")
                last_print = elapsed

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lsbfgs_lse_adaptive(a0, time_limit, T_start=80.0, T_min=0.3, n_stages=25, iters_per_stage=40):
    """Adaptive LogSumExp L-BFGS with progress-based temperature control."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    T = T_start
    stagnant = 0
    prev_obj = None

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 300,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)

            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()

            # Adaptive temperature update
            if prev_obj is not None:
                obj_ratio = (sc - best_score) / (best_score + 1e-15)
                if obj_ratio > 0.005:
                    stagnant += 1
                else:
                    stagnant = max(0, stagnant - 1)

            if stagnant >= 3:
                # Stuck: keep temperature or increase slightly
                T = max(T, T * 1.2)
                stagnant = 0
            else:
                # Progressing: decrease temperature
                T = max(T_min, T * 0.75)

            prev_obj = sc

            if (stage + 1) % 5 == 0:
                print(f"    LSE-Adaptive stage={stage+1}/{n_stages} T={T:.2f}: score={sc:.8f} best={best_score:.8f}")

        except Exception:
            T = max(T_min, T * 0.8)

    return best_a, best_score


def peak_subgradient(a0, time_limit, n_steps=200000):
    """Peak-targeted subgradient with Polyak step and iterate averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()

    best_a = a.copy()
    best_score = score_normalized(a)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)  # Theoretical lower bound

    # Polyak step tracking
    running_best = best_score
    t0 = time.monotonic()
    avg_a = a.copy()
    avg_count = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # Find active peaks (top 5% or at least the max)
        threshold = max_b * 0.98
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        if len(active) > 50:
            # Keep only top 50
            top_idx = np.argsort(b[active])[-50:]
            active = active[top_idx]

        # Compute subgradient: average of gradients at active peaks
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        # Project gradient to tangent space of simplex (for softmax parametrization)
        S = np.dot(g, a)
        grad_z = a * (g - S)

        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-15:
            break

        # Polyak step size
        gap = cur_score - lower_bound
        if gap < 1e-10:
            break
        eta = gap / (2.0 * grad_norm * grad_norm)
        eta = min(eta, 0.5)  # Cap step size

        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)
        sc_new = score_normalized(a_new)

        # Bisection line search if step is too large
        if sc_new > cur_score * 1.01:
            # Find a smaller step
            lo_e, hi_e = 0.0, eta
            for _ in range(5):
                mid_e = (lo_e + hi_e) / 2.0
                z_mid = z - mid_e * grad_z
                a_mid = softmax_from_z(z_mid)
                if score_normalized(a_mid) < cur_score:
                    lo_e = mid_e
                else:
                    hi_e = mid_e
            eta = lo_e
            z_new = z - eta * grad_z
            a_new = softmax_from_z(z_new)
            sc_new = score_normalized(a_new)

        z = z_new
        a = a_new

        # Maintain average (last 5000 iterates)
        avg_a = (avg_count * avg_a + a) / (avg_count + 1)
        avg_count += 1
        if avg_count > 5000:
            avg_count = 5000
            avg_a = (avg_a * 5000 + a) / 5001

        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()
            running_best = best_score

        # Also check the average
        sc_avg = score_normalized(avg_a)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_a.copy()

        if (step + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakSG step={step+1} ({elapsed:.0f}s): cur={cur_score:.8f} best={best_score:.8f} eta={eta:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x ** 2)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 4:
        a = np.ones(n) * (1.0 + 0.2 * np.cos(2 * np.pi * 3 * np.arange(n) / n))
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.15 * np.cos(7 * x)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 3) ** 2
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.5) ** 2) / 0.05) + 0.3 * np.exp(-((x - 0.2) ** 2) / 0.02) + 0.3 * np.exp(-((x - 0.8) ** 2) / 0.02)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def symmetric_constraint(a):
    """Enforce palindromic symmetry: a[i] = a[n-1-i]."""
    n = len(a)
    s = a[:n // 2 + 1].sum() + a[1:n // 2].sum() if n % 2 == 0 else a[:n // 2 + 1].sum() + a[1:n // 2 + 1].sum()
    half = (n + 1) // 2
    a_sym = np.zeros(n)
    for i in range(half):
        a_sym[i] = a[i]
        a_sym[n - 1 - i] = a[i]
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym /= ss
    return a_sym


def nesterov_lp_symmetric(a0, time_limit, solver=None, max_iters=300000):
    """Nesterov LP with palindromic symmetry constraint."""
    n = len(a0)
    half = (n + 1) // 2
    # Work in the symmetric half-space
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = symmetric_constraint(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        a = symmetric_constraint(a)
    a = a / a.sum()

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(symmetric_constraint(y), solver)
        if b_new is None:
            continue
        b_new = symmetric_constraint(b_new)

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa
        a_new = symmetric_constraint(a_new)

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy
        y_new = symmetric_constraint(y_new)

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

    return best_a, best_score


def final_propose_candidate(seed=42, budget_s=1800.0):
    """Main search function."""
    rng = np.random.default_rng(seed)
    global_deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Total budget: {budget_s}s, effective deadline: {global_deadline - time.monotonic():.0f}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    n_base = 1200

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic()
            remain = global_deadline - elapsed
            print(f"  ** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, remaining={remain:.0f}s)")

    # === Phase 1: Nesterov LP (unconstrained) - 35% time ===
    print(f"\n=== Phase 1: Nesterov LP at n={n_base} (score={best_score:.8f}) ===")
    t1 = 0.35 * (budget_s - 30)
    a_out, _ = nesterov_lp(best_seq, t1, solver=solver)
    try_update(a_out, "nesterov_lp_main")
    print(f"  Phase 1 done: best_score={best_score:.8f}")

    # === Phase 2: Symmetric Nesterov LP - 10% time ===
    print(f"\n=== Phase 2: Symmetric Nesterov LP (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t2 = min(0.10 * (budget_s - 30), remaining * 0.4)
    if t2 > 20:
        a_sym = symmetric_constraint(best_seq)
        a_out_sym, _ = nesterov_lp_symmetric(a_sym, t2, solver=solver)
        try_update(a_out_sym, "nesterov_sym")
        # Also try unsymmetric from symmetric init
        if remaining - t2 > 20:
            a_out_sym2, _ = nesterov_lp(a_out_sym, remaining * 0.3, solver=solver)
            try_update(a_out_sym2, "nesterov_from_sym")
    print(f"  Phase 2 done: best_score={best_score:.8f}")

    # === Phase 3: Adaptive LSE - 25% time ===
    print(f"\n=== Phase 3: Adaptive LSE (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t3 = min(0.25 * (budget_s - 30), remaining * 0.5)
    if t3 > 30:
        # Run multiple LSE passes with different initial temperatures
        a_lse1, _ = lsbfgs_lse_adaptive(best_seq, t3 * 0.6, T_start=100.0, T_min=0.2, n_stages=20, iters_per_stage=35)
        try_update(a_lse1, "lse_pass1")

        remaining = global_deadline - time.monotonic()
        if remaining > 20:
            a_lse2, _ = lsbfgs_lse_adaptive(best_seq, remaining * 0.5, T_start=20.0, T_min=0.1, n_stages=15, iters_per_stage=30)
            try_update(a_lse2, "lse_pass2")

        # LSE from a perturbed starting point
        remaining = global_deadline - time.monotonic()
        if remaining > 20:
            perturbed = best_seq + 0.05 * np.random.default_rng(123).normal(0, 1, len(best_seq))
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()
            a_lse3, _ = lsbfgs_lse_adaptive(perturbed, remaining * 0.5, T_start=50.0, T_min=0.2, n_stages=15, iters_per_stage=30)
            try_update(a_lse3, "lse_perturbed")
    print(f"  Phase 3 done: best_score={best_score:.8f}")

    # === Phase 4: Peak-targeted subgradient - 15% time ===
    print(f"\n=== Phase 4: Peak Subgradient (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t4 = min(0.15 * (budget_s - 30), remaining * 0.5)
    if t4 > 20:
        a_sg, _ = peak_subgradient(best_seq, t4, n_steps=500000)
        try_update(a_sg, "peak_sg")

        # Second pass from the result
        remaining = global_deadline - time.monotonic()
        if remaining > 15:
            a_sg2, _ = peak_subgradient(best_seq, remaining * 0.5, n_steps=300000)
            try_update(a_sg2, "peak_sg2")
    print(f"  Phase 4 done: best_score={best_score:.8f}")

    # === Phase 5: Multi-length exploration - 10% time ===
    print(f"\n=== Phase 5: Multi-length (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t5_total = min(0.10 * (budget_s - 30), remaining * 0.6)
    if t5_total > 30:
        lengths_to_try = [n_base + 200, n_base + 500, n_base - 200]
        for n_try in lengths_to_try:
            remaining = global_deadline - time.monotonic()
            if remaining < 25:
                break
            t5 = min(t5_total / 3.0, remaining * 0.45)
            if t5 < 15:
                break
            print(f"  Trying n={n_try}, budget={t5:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            # Quick Nesterov
            a_try, _ = nesterov_lp(init_try, t5 * 0.6, solver=solver, max_iters=100000)
            try_update(a_try, f"len_{n_try}_nesterov")
            # Quick LSE
            remaining = global_deadline - time.monotonic()
            if remaining > 15:
                a_try2, _ = lsbfgs_lse_adaptive(a_try, remaining * 0.5, T_start=50.0, T_min=0.3, n_stages=10, iters_per_stage=25)
                try_update(a_try2, f"len_{n_try}_lse")
    print(f"  Phase 5 done: best_score={best_score:.8f}")

    # === Phase 6: Final combined polish - 5% time ===
    print(f"\n=== Phase 6: Final polish (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    if remaining > 20:
        # Short Nesterov
        a_f1, _ = nesterov_lp(best_seq, remaining * 0.35, solver=solver, max_iters=50000)
        try_update(a_f1, "final_nesterov")

        remaining = global_deadline - time.monotonic()
        if remaining > 15:
            # Short LSE from low T
            a_f2, _ = lsbfgs_lse_adaptive(best_seq, remaining * 0.5, T_start=10.0, T_min=0.1, n_stages=10, iters_per_stage=25)
            try_update(a_f2, "final_lse")

        remaining = global_deadline - time.monotonic()
        if remaining > 10:
            # Short subgradient
            a_f3, _ = peak_subgradient(best_seq, remaining * 0.6, n_steps=100000)
            try_update(a_f3, "final_sg")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Time used: {time.monotonic() - (global_deadline - (budget_s - 30)):.0f}s")
    return [float(x) for x in best_seq]


def propose_candidate(seed=42, budget_s=1800.0):
    """Entry point for the search."""
    return final_propose_candidate(seed=seed, budget_s=budget_s)
