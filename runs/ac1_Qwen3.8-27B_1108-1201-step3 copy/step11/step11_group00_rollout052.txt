"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast Nesterov LP with top-K constraint restriction
2. FISTA on LogSumExp smooth approximation with FFT gradients
3. Multi-length exploration with PCHIP interpolation
4. Peak-shaving coordinate descent for final polish
Key improvements: top-K LP restriction for speed, FISTA instead of L-BFGS,
Blackman-Harris initialization, and adaptive time allocation.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.interpolate import PchipInterpolator


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_grad_lse_z(z, T):
    """Compute gradient of LSE objective w.r.t. z, using FFT."""
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()

    # grad[i] = 2 * sum_j w[i+j] * a[j]
    # This is a cross-correlation that can be computed via FFT
    conv_len = n + (2 * n - 1) - 1  # = 3n-2
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2

    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)

    grad = 2.0 * conv[n - 1:n - 1 + n]

    # Transform to z-space: grad_z[i] = a[i]*(grad[i] - dot(grad, a))
    S = np.dot(grad, a)
    grad_z = a * (grad - S)
    return grad_z, w


def lse_value_z(z, T):
    """Compute LSE value without gradient."""
    a = softmax_fn(z)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    return (max_Tb + np.log(exp_Tb.sum())) / T


def fista_lse(a0, time_limit, T_start=50.0, T_end=0.1, n_stages=8):
    """
    FISTA on LogSumExp-smoothed objective in softmax parameterization.
    Uses FFT-based gradients for O(n log n) per iteration.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a_init = a_init / a_init.sum()
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # FISTA parameters
        # Lipschitz constant estimate: roughly T * n (conservative)
        # Use backtracking for safety
        step = 1.0 / (T * n * 4.0)  # Initial step
        x = z.copy()  # FISTA iterate
        y = z.copy()  # FISTA extrapolate
        momentum = 1.0  # Initial momentum (will update as (1+sqrt(1+4*momentum^2))/2)

        for it in range(5000):
            if time.monotonic() - t0 > time_limit:
                break

            # Compute gradient at y
            grad_y, _ = compute_grad_lse_z(y, stage_T)

            # Backtracking line search
            s = step
            x_candidate = y - s * grad_y
            f_y = lse_value_z(y, stage_T)
            f_candidate = lse_value_z(x_candidate, stage_T)

            bs_iter = 0
            while f_candidate > f_y + 0.5 * s * np.dot(grad_y, x_candidate - y) and bs_iter < 20:
                s *= 0.5
                x_candidate = y - s * grad_y
                f_candidate = lse_value_z(x_candidate, stage_T)
                bs_iter += 1

            if bs_iter >= 20:
                # Accept anyway if we've done too many backtracks
                x_candidate = y - s * grad_y

            x = x_candidate
            step = s  # Update step for next iteration

            # FISTA momentum update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
            y = x + ((momentum - 1.0) / t_new) * (x - y)
            momentum = t_new

            # Check score
            a_trial = softmax_fn(x)
            score = 2.0 * n * fast_autocorr(a_trial).max()
            if score < best_score:
                best_score = score
                best_a = a_trial.copy()

            if it % 100 == 0:
                elapsed = time.monotonic() - t0
                print(f"        FISTA stage {si+1}/{n_stages} T={stage_T:.3f} it={it}: best={best_score:.6f}")

        z = x.copy()

    return best_a, best_score


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x_var = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x_var)), [cp.sum(x_var) == 1, x_var <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_topK(a_fixed, top_k_indices, solver, K):
    """
    Solve LP: min M s.t. (a_fixed * b)[k] <= M for k in top_k_indices,
    sum(b) = 1, b >= 0.
    Returns (b, M).
    """
    n = len(a_fixed)
    K = len(top_k_indices)
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [cp.sum(b_var) == 1]
    for k in top_k_indices:
        conv_k = np.zeros(n)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        conv_k[idx] = a_fixed[k - idx]
        constraints.append(np.dot(conv_k, b_var) <= M_var)
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_topK(a0, time_limit, solver, K=100):
    """
    Nesterov-accelerated LP best-response with top-K constraint restriction.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    y = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        # Find top K lags
        b_y = fast_autocorr(y)
        top_k = np.argsort(b_y)[::-1][:K]

        # Solve restricted LP
        b_star, M_val = solve_lp_topK(y, top_k, solver, K)
        if b_star is None:
            continue

        # Nesterov step
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s_new = y_new.sum()
        if s_new < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_new

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        y = y_new
        a = a_new

        # Check score
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Running average
        avg_a += a
        n_avg += 1
        if n_avg >= 50 and n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        if t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP-K t={t} ({elapsed:.0f}s): best={best_score:.6f}, LP_M={M_val:.8f}, K={K}")

    return best_a, best_score

                candidates = [j for j in range(n) if j != src]
            dst = candidates[rng.integers(len(candidates))]

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.005, 0.05)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= -1e-15:
                    a_trial = np.maximum(a_trial, 0)
                    s = a_trial.sum()
                    if s > 1e-15:
                        a_trial = a_trial / s
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0

            if stall > 100:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length using PCHIP."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    try:
        pchip = PchipInterpolator(old_x, f_norm)
        f_interp = pchip(new_x)
    except Exception:
        f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def blackman_harris_init(n):
    """Generate Blackman-Harris window as initialization."""
    m = np.arange(n)
    a = (0.35875 - 0.48829 * np.cos(2 * np.pi * m / (n - 1))
        + 0.14128 * np.cos(4 * np.pi * m / (n - 1))
        - 0.01168 * np.cos(6 * np.pi * m / (n - 1)))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 7)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 1.5
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.6 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    else:
        # Flat with small perturbation
        a = np.ones(n) + 0.1 * rng.standard_normal(n)
        a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
