```python
    return irfft(A * A, fft_size)[:size]


def score_fn(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2"""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def score_normalized(a):
    """Score assuming sum(a)=1: 2*max(conv(a,a))"""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def build_conv_matrix(a):
    """Build sparse matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    # T[k, i] = a[k-i] if 0 <= k-i < n, else 0
    # For each i, T[k, i] is nonzero for k in [i, i+n-1]
    row_ind = []
    col_ind = []
    data = []
    for i in range(n):
        row_ind.extend(range(i, i + n))
        col_ind.extend([i] * n)
        data.extend(a)
    return csr_matrix((data, (np.array(row_ind), np.array(col_ind))), shape=(size, n))


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max conv(a,b) s.t. b>=0, sum(b)=1."""
    n = len(a_fixed)
    T = build_conv_matrix(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [T @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    if solver is None:
        solver = get_lp_solver()
    
    try:
        prob.solve(solver=solver)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            for alt in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
                if alt == solver:
                    continue
                try:
                    prob.solve(solver=alt)
                    if prob.status in ('optimal', 'optimal_inaccurate'):
                        break
                except Exception:
                    continue
    except Exception:
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    t_val = float(t_var.value) if t_var.value is not None else float((a_fixed * 0 + 1).max())  # fallback
    
    # Verify
    actual_t = float(np.max(build_conv_matrix(a_fixed) @ b_val))
    t_val = min(t_val, actual_t)
    
    return b_val, t_val

    global _solver_cached
    if _solver_cached is not None:

    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:

def line_search_lp_alternating(a0, solver, time_limit, max_iters=50000):
    """
    LP alternating with line search for monotone decrease.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_score = score_normalized(a) / n  # This is max_conv for normalized a
    # Actually let me use the full score for clarity
    best_score = score_fn(a)
    t0 = time.monotonic()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP for best response
        b_star, lp_val = solve_lp_best_response(a, solver)
        if b_star is None:
            time.sleep(0.01)
            b_star, lp_val = solve_lp_best_response(a, solver)
            if b_star is None:
                continue
        
        # Line search: find alpha in [0,1] minimizing scorefn((1-alpha)*a + alpha*b_star)
        # Since the normalized score is convex in a, we can use golden section search
        best_alpha = 0.0
        best_trial_score = best_score
        
        # Try several alphas
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0, 0.4, 0.6]:
            a_trial = (1.0 - alpha) * a + alpha * b_star
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            sc = score_fn(a_trial)
            if sc < best_trial_score:
                best_trial_score = sc
                best_alpha = alpha
        
        if best_alpha > 0:
            a = (1.0 - best_alpha) * a + best_alpha * b_star
            a = np.maximum(a, 0)
            a = a / a.sum()
            
            sc = score_fn(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve >= 50:
            # Try a larger step or perturbation
            a_trial = a + 0.05 * b_star
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            sc = score_fn(a_trial)
            if sc < best_score:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            else:
                # Random perturbation
                rng = np.random.default_rng(it * 7 + 3)
                a_trial = a + rng.standard_normal(n) * 0.02
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = score_fn(a_trial)
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                no_improve = 0
        
        if (it + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP-LS iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}, alpha={best_alpha:.2f}")
    
    return best_a, best_score


def nesterov_lp_with_line_search(a0, solver, time_limit, max_iters=50000):
    """
    Nesterov LP alternating with line search restarts.
    Combines Nesterov momentum with periodic line search for stability.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    
    best_a = a.copy()
    best_score = score_fn(a)
    t0 = time.monotonic()
    prev_best = best_score
    rng = np.random.default_rng(42)
    
    t_nesterov = t0  # restart time
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute Nesterov parameter based on time since restart
        it_since_restart = it - int((t_nesterov - t0) * 100)  # approximate
        it_since_restart = max(0, it - int((t0 - t0) * 100))  # just use it
        alpha = 2.0 / (np.sqrt(it + 1) + 1.0)
        
        # Solve LP for best response to y
        b_star, lp_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue
        
        # Nesterov update
        y = (1.0 - alpha) * y + alpha * b_star
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        # FTRL-style update for a
        weight = (it + 1)
        a = (a * weight + alpha * (it + 2) * b_star) / (weight + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
            sa = 1.0
        a = a / sa
        
        # Evaluate
        score_a = score_fn(a)
        score_y = score_fn(y)
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        # Check for stall and restart
        if (it + 1) % 30 == 0:
            if best_score < prev_best - 1e-10:
                prev_best = best_score
                # Reset Nesterov counter (approximate by keeping going)
            else:
                # Line search from best_a toward b_star
                for alpha_ls in [0.3, 0.5, 0.7, 1.0]:
                    a_trial = (1.0 - alpha_ls) * best_a + alpha_ls * b_star
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = score_fn(a_trial)
                    if sc < best_score:
                        a = a_trial
                        y = a_trial.copy()
                        best_score = sc
                        best_a = a_trial.copy()
                        prev_best = best_score
                        break
        
        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  NLPM iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def logsumexp_lbgfs(a0, time_limit, T_stages=None, maxiter_per_stage=3000):
    """L-BFGS on LogSumExp smoothed objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [30.0, 20.0, 15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 1.5, 1.0, 0.6, 0.3, 0.15]
    
    best_a = a.copy()
    best_score = score_fn(a)
    n_stages = len(T_stages)
    iters_per_stage = max(30, maxiter_per_stage // n_stages)
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        def func(z_inner):
            if time.monotonic() - t0 > time_limit - 2:
                return 1e15
            a_trial = np.maximum(z_inner, 0)
            st = a_trial.sum()
            if st < 1e-15:
                return 1e15
            a_trial = a_trial / st
            b = fast_convolve(a_trial)
            max_b = b.max()
            exp_b = np.exp((b - max_b) / T)
            lse = max_b + T * np.log(exp_b.sum() / len(b))
            return (2.0 * n / (st * st)) * lse  # Hmm, need to be careful with normalization
        
        def jac(z_inner):
            if time.monotonic() - t0 > time_limit - 2:
                return np.zeros_like(z_inner)
            a_trial = np.maximum(z_inner, 0)
            st = a_trial.sum()
            if st < 1e-15:
                return np.zeros_like(z_inner)
            a_trial = a_trial / st
            b = fast_convolve(a_trial)
            max_b = b.max()
            exp_b = np.exp((b - max_b) / T)
            w = exp_b / exp_b.sum()
            # grad of lse wrt a_i = sum_k w[k] * a[k-i] (for valid k-i)
            grad = np.zeros(n)
            for i in range(n):
                js = max(0, i - n + 1)
                je = min(2*n-1, i + n - 1)
                # w[k] * a[k-i] for k where 0<=k-i<n
                # k ranges from i to i+n-1 (clamped to [0, 2n-2])
                k_start = max(0, i)
                k_end = min(2*n-1, i+n)
                # a[k-i] for k in [k_start, k_end-1]
                a_slice = a[k_start-i:k_end-i]
                w_slice = w[k_start:k_end]
                grad[i] = np.sum(w_slice * a_slice)
            # Full gradient of (2n/T) * LSE / st^2 -- but since st=1 (normalized), 
            # gradient of 2n * LSE wrt a is 2n * grad (of LSE)
            # Actually the objective is 2n * LSE(b) where b = conv(a,a)
            # dLSE/da_i = sum_k w[k] * db_k/da_i = sum_k w[k] * a[k-i]
            # So full grad = 2n * (that)
            grad *= 2.0 * n
            return grad
        
        try:
            res = minimize(fun=func, x0=a, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters_per_stage, 'ftol': 1e-14,
                                    'gtol': 1e-10, 'maxfun': iters_per_stage * 500,
                                    'disp': False})
            a_new = np.maximum(res.x, 0)
            sa = a_new.sum()
            if sa > 1e-15:
                a_new = a_new / sa
                sc = score_fn(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                a = a_new
        except Exception as e:
            print(f"  LSE-LBFGS error at T={T}: {e}")
        
        elapsed = time.monotonic() - t0
        print(f"  LSE-LB T={T:.1f} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a
```

```python
def cross_corr_lag_pos(a, w):
    """Compute c[i] = sum_j a[j] * w[j+i] for i = 0, ..., len(a)-1.
    a has length n, w has length 2n-1.
    """
    n = len(a)
    # Zero-pad to avoid circular effects
    # We need c[i] for i=0,...,n-1
    # c[i] = sum_{j=0}^{n-1} a[j] * w[j+i] where j+i < 2n-1
    # FFT size needs to be at least n + (2n-1) - 1 = 3n-2
    fft_size = 1 << (3*n).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    # The "correlation" sum_j a[j]*w[j+i]:
    # This is the same as the i-th element of the "anti-diagonal convolution"
    # Using FFT: ifft(conj(FFT(a)) * FFT(w)) gives sum_j conj(a[j]) * w[j+i] (circular)
    # For real a: = sum_j a[j] * w[j+i] (circular)
    # With zero padding, this gives the linear version
    corr = irfft(np.conj(A) * W, fft_size)
```

```python
def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_fn(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2"""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    if n == 0:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build sparse matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    row_ind = np.repeat(np.arange(n), n)
    col_ind = np.tile(np.arange(n), n)
    # T[k, i] = a[k-i] if 0 <= k-i < n
    # For column i, rows i to i+n-1 have values a[0], a[1], ..., a[n-1]
    data = np.tile(a, n)
    # Fix: for column i, the rows are i, i+1, ..., i+n-1 with values a[0], a[1], ..., a[n-1]
    # row_ind: for i=0: 0,1,...,n-1; for i=1: 1,2,...,n; ...
    row_ind = np.concatenate([np.arange(i, i + n) for i in range(n)])
    col_ind = np.repeat(np.arange(n), n)
    data = np.tile(a, n)
    return csr_matrix((data, (row_ind, col_ind)), shape=(size, n))

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max conv(a,b) s.t. b>=0, sum(b)=1."""
    n = len(a_fixed)
    T = build_conv_matrix(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [T @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)

    if solver is None:
        solver = get_lp_solver()

    status = None
    try:
        prob.solve(solver=solver)
        status = prob.status
    except Exception:
        pass

    if status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    status = prob.status
                    break
            except Exception:
                continue

    if status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None, None
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s

    t_val = float(t_var.value) if t_var.value is not None else None
    if t_val is None or np.isnan(t_val):
        t_val = float(np.max(T @ b_val))

    return b_val, t_val


def lp_alternating_nesterov(a0, solver, time_limit, max_iters=50000):
    """LP alternating with Nesterov momentum and periodic line search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = score_fn(a)
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve LP for best response to y
        b_star, lp_val = solve_lp_best_response(y, solver)
        if b_star is None:
            time.sleep(0.005)
            b_star, lp_val = solve_lp_best_response(y, solver)
            if b_star is None:
                continue

        # Nesterov update for y
        alpha = 2.0 / (np.sqrt(it + 1) + 1.0)
        y = (1.0 - alpha) * y + alpha * b_star
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        # FTRL-style update for a
        a = (a * (it + 1) + alpha * (it + 2) * b_star) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        # Evaluate
        score_a = score_fn(a)
        score_y = score_fn(y)

        improved = False
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            improved = True
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            improved = True

        # Periodic line search for stability
        if (it + 1) % 20 == 0:
            if improved:
                prev_best = best_score
            else:
                stall_count += 1

            if stall_count >= 3:
                # Line search from best_a toward b_star
                found = False
                for ls_alpha in [0.5, 0.7, 1.0, 0.3, 0.8, 0.9]:
                    a_trial = (1.0 - ls_alpha) * best_a + ls_alpha * b_star
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    sc = score_fn(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial
                        y = a_trial.copy()
                        best_score = sc
                        best_a = a_trial.copy()
                        stall_count = 0
                        prev_best = best_score
                        found = True
                        break
                if not found:
                    # Random perturbation
                    rng = np.random.default_rng(it * 13 + 7)
                    noise_scale = min(0.05, 0.01 * stall_count)
                    a_trial = best_a + rng.standard_normal(n) * noise_scale
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st > 1e-15:
                        a_trial = a_trial / st
                        sc = score_fn(a_trial)
                        if sc < best_score:
                            a = a_trial
                            y = a_trial.copy()
                            best_score = sc
                            best_a = a_trial.copy()
                    stall_count = 0

        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}, lp_val={lp_val:.6f}")

    return best_a, best_score


def lse_gradient(a, w):
    """Compute gradient of LSE: grad[i] = sum_j a[j] * w[i+j] via FFT.
    a has length n, w has length 2n-1.
    Returns gradient of length n.
    """
    n = len(a)
    fft_size = 1 << (3 * n).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(np.conj(A) * W, fft_size)
    return corr[:n]


def lse_obj_grad(a, T):
    """Compute LogSumExp smoothed objective and gradient.
    Objective: 2n * LSE(conv(a,a), T) where LSE(x, T) = T * log(sum(exp(x/T)))
    Returns (objective_value, gradient).
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T)
    w = exp_b / exp_b.sum()
    lse_val = max_b + T * np.log(exp_b.sum() / len(b))
    obj = 2.0 * n * lse_val

    # Gradient: grad[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i)
    # = 2 * (cross-correlation of w and a at appropriate lags)
    # grad[i] = 2 * sum_j a[j] * w[i+j] (reindexing)
    grad = lse_gradient(a, w) * 2.0

    return obj, grad


def logsumexp_lbgfs(a0, time_limit, T_stages=None, maxiter_per_stage=5000):
    """L-BFGS on LogSumExp smoothed objective with FFT-accelerated gradients."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [40.0, 25.0, 15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 1.5, 1.0, 0.6, 0.35, 0.2, 0.1]

    n_stages = len(T_stages)
    iters_per_stage = max(30, maxiter_per_stage // n_stages)
    best_a = a.copy()
    best_score = score_fn(a)

    for T in T_stages:
        if time.monotonic() - t0 > time_limit - 5:
            break

        def func(z_inner):
            if time.monotonic() - t0 > time_limit - 2:
                return 1e15
            a_trial = np.maximum(z_inner, 0)
            st = a_trial.sum()
            if st < 1e-15:
                return 1e15
            a_trial = a_trial / st
            obj, _ = lse_obj_grad(a_trial, T)
            return obj

        def jac(z_inner):
            if time.monotonic() - t0 > time_limit - 2:
                return np.zeros_like(z_inner)
            a_trial = np.maximum(z_inner, 0)
            st = a_trial.sum()
            if st < 1e-15:
                return np.zeros_like(z_inner)
            a_trial = a_trial / st
            _, grad = lse_obj_grad(a_trial, T)
            return grad

        try:
            res = minimize(fun=func, x0=a, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters_per_stage, 'ftol': 1e-14,
                                    'gtol': 1e-10, 'maxfun': iters_per_stage * 300,
                                    'disp': False})
            a_new = np.maximum(res.x, 0)
            sa = a_new.sum()
            if sa > 1e-15:
                a_new = a_new / sa
                sc = score_fn(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                a = a_new
        except Exception as e:
            pass

        elapsed = time.monotonic() - t0
        print(f"    LSE-LB T={T:.1f} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a


def coordinate_descent_refine(a0, time_limit):
    """Targeted coordinate descent focusing on peak-contributing elements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_fn(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.9999
        peaks = np.where(b >= threshold)[0]

        improved = False
        for peak_k in peaks[:3]:
            if improved:
                break
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_m = min(10, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

            for pos in top_positions:
                if a[pos] < 1e-12:
                    continue
                for factor in [0.999, 0.995, 0.99, 0.98, 0.97, 0.95]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-16:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask].copy()
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = score_fn(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 1000:
                # Simultaneous reduction of top contributors
                peak_k = int(np.argmax(b))
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[peak_k - idx]
                top_positions = idx[np.argsort(contribs)[::-1][:5]]
                a_trial = a.copy()
                for pos in top_positions:
                    if a_trial[pos] > 1e-12:
                        a_trial[pos] *= 0.995
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = score_fn(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    stall = 0
                elif stall > 3000:
                    break
        else:
            stall = 0

        if (int)((time.monotonic() - t0)) % 30 == 0 and time.monotonic() - t0 > 0:
            print(f"    CoordDesc ({time.monotonic()-t0:.0f}s): score={best_score:.8f}, stall={stall}")

    return best_a


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        a = np.maximum(a, 0)
    else:
        a = 1.0 / (1.0 + np.exp(-5 * (x - 0.5)))  # sigmoid
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Combines LP alternating, multi-n exploration, LogSumExp L-BFGS, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = score_fn(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = score_fn(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01 or len(a) == 0:
            return
        sc = score_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: LP Alternating (main workhorse) ===
    print(f"\n=== Phase 1: LP Alternating (score={best_score:.8f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = lp_alternating_nesterov(best_seq, solver, t_alloc, max_iters=100000)
        try_update(a_lp, "LP_p1")

    # === Phase 2: Multi-n Exploration ===
    print(f"\n=== Phase 2: Multi-n Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 40:
        n_best = len(best_seq)
        candidate_ns = sorted(set([
            max(400, n_best - 200),
            max(500, n_best + 200),
            max(400, n_best // 2),
            min(3000, n_best * 2)
        ]))
        # Filter out n_best itself
        candidate_ns = [n_try for n_try in candidate_ns if n_try != n_best]
        per_n = t_alloc / max(1, len(candidate_ns))
        print(f"  Trying n values: {candidate_ns}, {per_n:.0f}s each")
        for n_try in candidate_ns:
            if time.monotonic() > deadline - 30:
                break
            a_interp = interpolate_sequence(best_seq, n_try)
            t_sub = min(per_n, deadline - time.monotonic() - 10)
            if t_sub < 15:
                continue
            print(f"  n={n_try}, budget={t_sub:.0f}s")
            a_sub, _ = lp_alternating_nesterov(a_interp, solver, t_sub, max_iters=30000)
            try_update(a_sub, f"multi_n_{n_try}")

    # === Phase 3: LogSumExp L-BFGS ===
    print(f"\n=== Phase 3: LogSumExp L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        T_stages = [30.0, 20.0, 15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 1.5, 1.0, 0.6, 0.35, 0.2, 0.1]
        print(f"  Budget: {t_alloc:.0f}s, {len(T_stages)} stages")
        a_lb = logsumexp_lbgfs(best_seq, t_alloc, T_stages=T_stages, maxiter_per_stage=8000)
        try_update(a_lb, "lbgfs")

    # === Phase 4: Coordinate Descent Polish ===
    print(f"\n=== Phase 4: Coordinate Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        a_cd = coordinate_descent_refine(best_seq, t_alloc)
        try_update(a_cd, "coord_desc")

    # === Phase 5: Final LP Polish ===
    print(f"\n=== Phase 5: Final LP Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 15:
        a_fin, _ = lp_alternating_nesterov(best_seq, solver, t_alloc, max_iters=50000)
        try_update(a_fin, "final_lp")

    # === Phase 6: Quick coordinate descent on final best ===
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 60.0)
    if t_alloc > 10:
        a_cd2 = coordinate_descent_refine(best_seq, t_alloc)
        try_update(a_cd2, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = score_fn(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
```
