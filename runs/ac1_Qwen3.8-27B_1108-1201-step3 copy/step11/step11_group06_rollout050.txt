```python
import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp
from scipy.sparse import csr_matrix

def fft_convolve(a, b=None):
    """Linear convolution via FFT. If b is None, compute a*a."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    B = rfft(b.astype(np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

def fft_correlation(w, a):
    """Compute cross-correlation: g[j] = sum_m w[j+m] * a[m] for j=0..n-1
    where w has length 2n-1 and a has length n."""
    n = len(a)
    size_w = len(w)  # 2n-1
    size = size_w + n - 1  # 3n-2
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w.astype(np.float64), fft_size)
    A = rfft(a.astype(np.float64), fft_size)
    # Cross-correlation via FFT: IFFT(FFT(w) * conj(FFT(a)))
    G = irfft(W * np.conj(A), fft_size)[:n]
    return G
```

```python
def multiplicative_weights_fft(a0, time_limit, T0=5.0, T_min=0.01, eta0=0.1, n_steps=1000000):
    """Fast FFT-based multiplicative weights with LogSumExp smoothing."""
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    
    n = len(a)
    best_a = a.copy()
    best_score = 2.0 * n * fft_convolve(a).max()
    
    t0 = time.monotonic()
    eta = eta0
    T = T0
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute convolution
        b = fft_convolve(a)  # length 2n-1
        max_b = b.max()
        
        # Compute softmax weights
        scaled = (b - max_b) / T
        exp_scaled = np.exp(scaled)
        w = exp_scaled / exp_scaled.sum()
        
        # Compute gradient via FFT correlation
        g = fft_correlation(w, a)
        
        # Multiplicative update
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        # Evaluate
        b_new = fft_convolve(a_new)
        score_new = 2.0 * n * b_new.max()
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            eta = min(eta * 1.005, 0.5)
        else:
            no_improve += 1
            eta *= 0.95
            if eta < 1e-6:
                # Cool temperature and reset step size
                T = max(T * 0.7, T_min)
                eta = max(eta * 5, 0.01)
                if T <= T_min and eta < 0.005:
                    break
        
        # Periodic temperature cooling
        if step % 10000 == 0 and step > 0:
            T = max(T * 0.85, T_min)
        
        if step % 50000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    MW {elapsed:.1f}s step={step}: score={best_score:.8f} T={T:.4f} eta={eta:.6f}")
    
    return best_a, best_score
```

```python
import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp
from scipy.sparse import csr_matrix


def fft_convolve(a, b=None):
    """Linear convolution via real FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def fft_correlation(w, a):
    """Compute g[j] = sum_{m=0}^{n-1} w[j+m] * a[m] for j=0..n-1.
    w has length 2n-1, a has length n."""
    n = len(a)
    size = len(w) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    G = irfft(W * np.conj(A), fft_size)[:n]
    return G


def build_conv_matrix_sparse(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    nnz = 0
    for j in range(n):
        js = max(0, 0)
        je = min(n, size - j)
        nnz += (je - js)
    rows = np.zeros(nnz, dtype=np.int32)
    cols = np.zeros(nnz, dtype=np.int32)
    vals = np.zeros(nnz, dtype=np.float64)
    count = 0
    for j in range(n):
        je = min(n, size - j)
        for k in range(j, je):
            rows[count] = k
            cols[count] = j
            vals[count] = a[k - j]
            count += 1
    return csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(size, n))

def multiplicative_weights_fft(a0, time_limit, T0=8.0, T_min=0.005,
                                eta0=0.08, n_steps=5000000, focus_fraction=0.3):
    """
    Ultra-fast FFT-based multiplicative weights with LogSumExp smoothing.
    Each step is O(n log n) via FFT-based convolution and correlation.
    """
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s

    n = len(a)
    best_a = a.copy()
    best_score = 2.0 * n * fft_convolve(a).max()

    t0 = time.monotonic()
    eta = eta0
    T = T0
    stall_count = 0
    rng = np.random.default_rng(12345)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute convolution
        b = fft_convolve(a)
        max_b = b.max()

        # Compute focused softmax weights (top fraction of elements)
        b_range = max_b - b.min()
        if b_range < 1e-15:
            w = np.ones(len(b)) / len(b)
        else:
            threshold = max_b - b_range * (1.0 - focus_fraction)
            scaled = (b - max_b) / T
            exp_scaled = np.exp(scaled)
            # Optionally zero out far-from-peak elements for focus
            exp_scaled[b < threshold] *= 0.1
            w = exp_scaled / exp_scaled.sum()

        # Compute gradient via FFT correlation
        g = fft_correlation(w, a)
        g = np.maximum(g, 0)

        # Adaptive step size based on gradient magnitude
        g_mean = g.mean()
        if g_mean > 1e-15:
            eta_adapted = eta * (max_b / (g_mean + 1e-15))
        else:
            eta_adapted = eta
        eta_adapted = min(eta_adapted, 1.0)

        # Multiplicative update
        log_mult = -eta_adapted * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Evaluate
        b_new = fft_convolve(a_new)
        max_new = b_new.max()
        score_new = 2.0 * n * max_new

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            eta = min(eta * 1.002, 0.8)
        else:
            a = a_new  # still accept for exploration
            stall_count += 1
            eta *= 0.97
            if eta < 1e-5:
                # Cool and reset
                T = max(T * 0.6, T_min)
                eta = max(eta * 3, 0.02)
                if T <= T_min * 1.1 and eta < 0.01:
                    # Near convergence, do a small perturbation
                    if stall_count > 5000:
                        pert = 0.005 * rng.standard_normal(n)
                        a = np.maximum(a + pert, 0)
                        a = a / a.sum()
                        stall_count = 0
                        eta = 0.05

        # Temperature cooling schedule
        if step % 5000 == 0 and step > 0:
            T = max(T * 0.92, T_min)

        if step % 100000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    MW step={step} ({elapsed:.0f}s): score={best_score:.8f} T={T:.4f} eta={eta:.6f}")

    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=200000):

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fft_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(777)
    T_mat = build_conv_matrix_sparse(y)
    lp_times = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver, T_mat)
        lp_times.append(time.monotonic() - lp_t0)
        if b_new is None:
            continue
        if len(lp_times) > 20:
            avg_lp = np.mean(lp_times[-20:])
            if avg_lp > 10.0:
                print(f"    LP too slow ({avg_lp:.2f}s/iter), stopping at iter {it+1}")
                break

        alpha = 2.0 / (it + 2)
        y_old = y.copy()
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = y_old
            sy = y_old.sum()
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        T_mat = build_conv_matrix_sparse(y)
        score_a = 2.0 * n * fft_convolve(a).max()
        score_y = 2.0 * n * fft_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 4:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 8)
                phase = rng.uniform(0, 2 * np.pi)
                noise = 0.01 * np.cos(freq * x + phase)
                y = np.maximum(y + noise, 0)
                y = y / y.sum()
                T_mat = build_conv_matrix_sparse(y)
                stall_count = 0
            if (it + 1) % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def lse_logsumexp_obj_grad(z, n, T_temp):
    """Compute LSE objective and gradient in log-domain."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    b = fft_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T_temp
    scaled_shifted = scaled - scaled.max()
    exp_terms = np.exp(scaled_shifted)
    lse = max_b + T_temp * (scaled.max() + np.log(exp_terms.sum() + 1e-300))
    obj = 2.0 * n * lse
    w = exp_terms / exp_terms.sum()
    g = fft_correlation(w, a)
    S = np.dot(g, a)
    grad_z = a * (2.0 * n * (g - S))
    return obj, grad_z


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage=500):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(x):
            obj, _ = lse_logsumexp_obj_grad(x, n, stage_T)
            return obj

        def jac(x):
            _, g = lse_logsumexp_obj_grad(x, n, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': maxiter_per_stage * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

    z_shifted = z - z.max()
    a_final = np.exp(z_shifted)
    a_final = np.maximum(a_final, 0)
    s_f = a_final.sum()
    if s_f > 1e-15:
        a_final = a_final / s_f
    return a_final


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 5 * x + 0.5)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = 1.0 / (1.0 + 3.0 * np.abs(x - 0.5))
    elif choice == 8:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 7 * x) + 0.2 * np.cos(2 * np.pi * 11 * x + 1.0)
        a = np.maximum(a, 0)
    else:
        a = np.exp(-np.abs(x - 0.5) * 4) + 0.3 * np.exp(-np.abs(x - 0.2) * 6) + 0.3 * np.exp(-np.abs(x - 0.8) * 6)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturb_sequence(a, rng, strength=0.02):
    """Perturb with structured frequency components."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq1 = rng.integers(1, 10)
    freq2 = rng.integers(2, 15)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    pert = strength * (np.cos(freq1 * x + p1) + 0.5 * np.sin(freq2 * x + p2))
    a_pert = np.maximum(a + pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Primary: ultra-fast FFT-based MW (millions of O(n log n) steps).
    Secondary: LP alternating + LSE L-BFGS for fine-tuning.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {budget_s:.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Ultra-fast FFT-MW (main workhorse) ===
    print(f"\n=== Phase 1: FFT-MW n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = rem * 0.55
    print(f"  Budget: {t1:.0f}s")
    a_mw, _ = multiplicative_weights_fft(best_seq, t1, T0=8.0, T_min=0.003,
                                          eta0=0.08, n_steps=10000000,
                                          focus_fraction=0.3)
    try_update(a_mw, "fft_mw_1")

    # === Phase 2: SecondMW pass with different params ===
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.2, 200.0)
    if t2 > 30:
        print(f"\n=== Phase 2: FFT-MW pass 2 (score={best_score:.8f}) ===")
        print(f"  Budget: {t2:.0f}s")
        a_mw2, _ = multiplicative_weights_fft(best_seq, t2, T0=2.0, T_min=0.001,
                                               eta0=0.03, n_steps=10000000,
                                               focus_fraction=0.1)
        try_update(a_mw2, "fft_mw_2")

    # === Phase 3: LP refinement ===
    print(f"\n=== Phase 3: LP refinement (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.25, 400.0)
    if t3 > 60:
        print(f"  Budget: {t3:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t3, solver, n_max_iters=500000)
        try_update(a_lp, "lp_refine")

    # === Phase 4: LSE L-BFGS polish ===
    print(f"\n=== Phase 4: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t4 = min(rem * 0.2, 200.0)
        T_stages = [3.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
        print(f"  {len(T_stages)} stages, budget={t4:.0f}s")
        a_lse = lbgfs_refine(best_seq, t4, T_stages, maxiter_per_stage=300)
        try_update(a_lse, "lse")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    for ln in [1000, 1500]:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        if len(best_seq) == ln:
            continue
        t_ln = min(rem * 0.3, 200.0)
        if t_ln < 30:
            continue
        print(f"  Trying n={ln}, budget={t_ln:.0f}s")
        init_ln = interpolate_sequence(best_seq, ln)
        a_ln, _ = multiplicative_weights_fft(init_ln, t_ln * 0.7, T0=5.0, T_min=0.005,
                                              eta0=0.06, n_steps=5000000,
                                              focus_fraction=0.2)
        try_update(a_ln, f"mw_n{ln}")
        rem = deadline - time.monotonic()
        if rem > 30:
            a_ln_lp, _ = nesterov_lp_refine(a_ln, min(rem * 0.5, 100.0), solver, n_max_iters=50000)
            try_update(a_ln_lp, f"lp_n{ln}")

    # === Phase 6: Perturbation escape ===
    print(f"\n=== Phase 6: Perturbation (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        for trial in range(3):
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            a_pert = perturb_sequence(best_seq, rng, strength=0.01 + 0.005 * trial)
            t_sub = min(rem * 0.3, 60.0)
            a_opt, _ = multiplicative_weights_fft(a_pert, t_sub, T0=3.0, T_min=0.01,
                                                    eta0=0.04, n_steps=2000000,
                                                    focus_fraction=0.2)
            try_update(a_opt, f"pert{trial}")

    # === Phase 7: Final LP polish ===
    print(f"\n=== Phase 7: Final LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t7 = min(rem * 0.4, 80.0)
        a_fin, _ = nesterov_lp_refine(best_seq, t7, solver, n_max_iters=100000)
        try_update(a_fin, "final_lp")

    # === Phase 8: Final MW micro-polish ===
    rem = deadline - time.monotonic()
    if rem > 10:
        t8 = min(rem * 0.5, 30.0)
        a_fin2, _ = multiplicative_weights_fft(best_seq, t8, T0=0.5, T_min=0.001,
                                                eta0=0.01, n_steps=5000000,
                                                focus_fraction=0.1)
        try_update(a_fin2, "final_mw")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
```
