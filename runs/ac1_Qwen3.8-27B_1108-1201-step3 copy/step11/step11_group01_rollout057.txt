    """
    Active-set convex QP optimization.
    At each step:
    1. Compute all (a*a)[k]
    2. Find active set S = {k : (a*a)[k] >= max - threshold*max}
    3. Solve: min a^T C_S a s.t. sum(a)=1, a>=0
    4. Update a
    """

    a = np.maximum(a, 0).astype(np.float64)

    a = a / s

        active = np.where(b >= max_b * (1 - threshold))[0]
