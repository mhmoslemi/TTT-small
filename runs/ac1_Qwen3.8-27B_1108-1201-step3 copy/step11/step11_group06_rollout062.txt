For the uniform p_j = 1/n: max(p*p)_k = 1/n (at center), giving objective = 2.
The theoretical minimum is 2n/(2n-1) ≈ 1.

So we need to find p such that conv(p,p) is as flat as possible.

**Key insight**: The convolution of a non-negative sequence with itself is always non-negative, and its sum is 1 (when p is normalized). We want it to be as close to uniform (1/(2n-1) at each point) as possible.

This is equivalent to: find a non-negative sequence p of length n with sum 1, such that the linear convolution p*p is as close to constant as possible.

This is related to the concept of "Barker sequences" and "low autocorrelation sequences" in the discrete setting, but here we're free to choose continuous values.

**Let me think about a completely different approach: spectral methods.**

The condition that conv(p,p) is constant is equivalent to FFT(conv(p,p)) being a constant (in the frequency domain, up to the DC component).

If we pad p to length N = 2n-1 (or a power of 2 >= 2n-1), then:
FFT(conv(p,p)) = FFT(p)^2 (pointwise squaring)

We want FFT(p)^2 to be as "flat" as possible (i.e., its IFFT should be as constant as possible).

But FFT(p)^2 being "flat in the time domain" means its IFFT is flat, which means FFT(p)^2 has only a DC component... which would mean p is constant. That's the uniform case giving objective 2.

Wait, that's not right. IFFT of a constant is a delta function, not a constant. Let me be more careful.

If conv(p,p) = c (constant) for all k, then FFT(conv(p,p)) = c*N (at frequency 0) + 0 (at other frequencies) = c*N * delta_0.

But FFT(conv(p,p)) = FFT(p)^2. So FFT(p)^2 = c*N * delta_0, which means FFT(p) = sqrt(c*N) * delta_0, which means p is constant.

So the only way to get a perfectly flat convolution is with a constant sequence, which gives objective 2. For non-constant sequences, the convolution cannot be perfectly flat.

But we can get close to flat! The question is how close, and that depends on n.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Ultra-fast FFT-based gradient computation for LogSumExp-smoothed objective
2. Multi-stage Nesterov-accelerated gradient descent with adaptive step sizes
3. LP alternating with Nesterov dual averaging (game-theoretic approach)
4. Direct subgradient method on non-smooth objective
5. Multi-length exploration (n=1000 to n=4000)
Key innovation: O(n log n) gradient via cross-correlation FFT trick,
enabling millions of optimization steps within time budget.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_obj_grad_fast(a, t):
    """Compute 2n*LSA(conv(a,a), t) and its gradient. Returns (obj, grad, b, max_b)."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    shifted = (b - max_b) / t
    exp_b = np.exp(shifted)
    Z = exp_b.sum()
    p = exp_b / Z
    lse = max_b + t * np.log(Z)
    obj = 2.0 * n * lse
    # grad[m] = 2/t * sum_j p[m+j] * a[j] for m=0,...,n-1
    # using cross-correlation: IFFT(FFT(p) * conj(FFT(a)))
    N = 1 << (2 * n - 2).bit_length()
    P = rfft(p, N)
    A = rfft(a, N)
    corr = irfft(P * np.conj(A), N)
    grad = (2.0 / t) * corr[:n]
    return obj, grad, b, max_b


def subgradient_fast(a):
    """Compute subgradient of 2n*max(conv(a,a)) at a. Returns (grad, max_b)."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    threshold = max_b * (1.0 - 1e-5)
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(b)])
    p = np.zeros(len(b))
    p[active] = 1.0 / len(active)
    N = 1 << (2 * n - 2).bit_length()
    P = rfft(p, N)
    A = rfft(a, N)
    corr = irfft(P * np.conj(A), N)
    grad = 2.0 * corr[:n]
    return grad, max_b


def gd_nesterov_stages(a_start, t_stages, iters_per_stage, eta0, momentum, time_limit, deadline):
    """Multi-stage Nesterov GD on LSE-smoothed objective with simplex projection."""
    n = len(a_start)
    a = np.maximum(np.asarray(a_start, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    v = np.zeros(n)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t_lo = time.monotonic()
    t0 = time.monotonic()

    for stage_idx, t in enumerate(t_stages):
        if time.monotonic() - t_lo > time_limit:
            break
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 2:
            break
        stage_limit = min(time_limit - (time.monotonic() - t_lo), remaining)
        # Adaptive eta: scale with t
        eta = eta0 * max(t, 0.01)
        stage_t0 = time.monotonic()
        it = 0
        for it in range(iters_per_stage):
            if time.monotonic() - stage_t0 > stage_limit:
                break
            # Backtracking line search every 1000 steps
            if it % 1000 == 0 and it > 0:
                # Check if we should reduce eta
                obj_now, _, _, _ = lse_obj_grad_fast(a, t)
                if not np.isfinite(obj_now) or obj_now > best_score * 10:
                    eta *= 0.5
                    v *= 0.5

            # Nesterov lookahead
            y = a - momentum * v
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
                sy = a.sum()
            y = y / sy

            # Gradient at y
            _, grad, _, _ = lse_obj_grad_fast(y, t)

            # Update
            v = momentum * v + eta * grad
            a_new = y - v
            a_new = np.maximum(a_new, 0)
            sa = a_new.sum()
            if sa < 1e-15:
                a = a.copy()
            else:
                a = a_new / sa

            # Periodic evaluation of true objective
            if it % 200 == 0:
                score = 2.0 * n * fast_convolve(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    # Slightly increase momentum when improving
                    momentum = min(momentum * 1.001, 0.95)

        # End of stage: evaluate
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        elapsed = time.monotonic() - t0
        print(f"    LSE-GD stage {stage_idx+1}/{len(t_stages)} (t={t:.4f}, {it+1} iters, {elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score

def build_conv_matrix_sparse(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    rows = np.zeros(n * n, dtype=np.int32)
    cols = np.zeros(n * n, dtype=np.int32)
    vals = np.zeros(n * n, dtype=np.float64)
    count = 0
    for j in range(n):
        for k in range(j, min(j + n, size)):
            rows[count] = k
            cols[count] = j
            vals[count] = a[k - j]
            count += 1
    return csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(size, n))


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, T_mat):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]
    solved = False
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue
    if not solved:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(max(0, M_var.value))


def lp_nesterov(a0, time_limit, deadline, solver, n_max_iters=500000):
    """LP alternating with Nesterov dual averaging."""

    stall_count = 0
    rng = np.random.default_rng(42)
    lp_times = []
    T_mat = build_conv_matrix_sparse(y)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit or time.monotonic() > deadline:
            break

        lp_times.append(time.monotonic() - lp_t0)

        if len(lp_times) > 20:
            avg_lp = np.mean(lp_times[-20:])

                break
        alpha = 2.0 / (it + 2)
        y_old = y.copy()
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = y_old
            sy = y_old.sum()
        y = y / sy

        # Nesterov-style primal update
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()

            a = y.copy()

            a = a / sa

        score_y = 2.0 * n * fast_convolve(y).max()

        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

    return best_a, best_score


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 5 * x + 0.5)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = 1.0 / (1.0 + 3.0 * np.abs(x - 0.5))
    elif choice == 8:
        # Linear decay (inspired by n=3 optimum [0.5, 0.25, 0.25])
        a = (1 - x) ** 0.7
    else:
        # Bimodal
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0)
    a = np.maximum(a, 0)
    ss = a.sum()
    if ss < 1e-15:
        return np.ones(n) / n
    return a / ss


def make_palindromic(a):
    """Make a sequence palindromic."""
    n = len(a)
    a_pal = (a + a[::-1]) / 2.0
    a_pal = np.maximum(a_pal, 0)
    s = a_pal.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_pal / s
