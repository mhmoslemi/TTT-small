def proj_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA (accelerated projected gradient) on smoothed log-sum-exp objective

2. LP best-response alternating for final refinement
3. Multi-length exploration (n=1200 primary, n=1400-1600 secondary)
4. Multi-frequency structured initialization

Algorithm:
- Work in normalized space (sum(a) = 1)
- Use log-sum-exp with top-k focusing for stable gradient computation
- Decay temperature over time to approach exact max
- Use Nesterov momentum for acceleration
- Project onto simplex after each step
- Periodically switch to LP for refinement

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def proj_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    idx = rho[-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Normalize to fix any floating point issues
    s = w.sum()
    if s > 0:
        w = w / s
    else:
        w = np.ones(n) / n
    return w

def score(a):
    """Compute the objective value."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    n = len(a_norm)
    b = fast_convolve(a_norm)
    return 2.0 * n * b.max()

                   top_k_frac=0.2, backtracking=False):

    FISTA on smoothed log-sum-exp with adaptive temperature.
    """
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a_prev = a.copy()
    t_mom = 1.0
    tau = float(tau_init)
    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()
    last_tau_change = 0
    
    for it in range(5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        t_old = t_mom
        t_mom = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        mu = (t_old - 1.0) / t_mom
        y = a + mu * (a - a_prev)
        y = proj_simplex(y)
        
        # Compute smoothed gradient
        b = fast_convolve(y)
        total_k = 2 * n - 1
        top_k = max(10, int(top_k_frac * total_k))
        
        if top_k < total_k:
            idx = np.argpartition(b, -top_k)[-top_k:]
            b_top = b[idx]
            w = np.zeros(total_k)
            w_top = np.exp(tau * (b_top - b_top.max()))
            w_top /= w_top.sum()
            w[idx] = w_top
        else:
            w = np.exp(tau * (b - b.max()))
            w /= w.sum()
        
        # Gradient: g[i] = tau * sum_k w[k] * y[k-i] (valid)
        # = tau * np.correlate(w, y, 'full')[:n]
        g = np.correlate(w, y, mode='full')[:n] * tau
        
        # Step size
        L = tau * np.dot(y, y)
        step = 1.0 / max(L, 1e-15)

        a_new = proj_simplex(y - step * g)
        
        a_prev = a
        a = a_new
        
        # Evaluate periodically
        if it % 50 == 0 or it < 10:
            sc = score(a)
            if sc < best_score:

                best_a = a.copy()

            last_tau_change = it
            if it % 10000 == 0:

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def proj_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    idx = rho[-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 1e-15:
        w /= s
    else:
        w = np.ones(n) / n
    return w


def compute_score(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def smooth_score_grad(a, tau, top_k_frac=0.3):
    """
    Compute smoothed objective and gradient.
    f(a) = (1/tau) * log(sum_k exp(tau * (a*a)_k))
    Returns (f_value, gradient_wrt_a).
    """
    n = len(a)
    total_k = 2 * n - 1
    b = fast_convolve(a)
    top_k = max(20, int(top_k_frac * total_k))

    if top_k < total_k:
        idx = np.argpartition(b, -top_k)[-top_k:]
        b_top = b[idx]
        w = np.zeros(total_k)
        w_top = np.exp(tau * (b_top - b_top.max()))
        w_top /= w_top.sum()
        w[idx] = w_top
        lse = b_top.max() + (1.0 / tau) * np.log(w_top.sum() * top_k)
        # More accurate: use the full top_k contribution
        lse_approx = b_top.max() + (1.0 / tau) * np.log(np.sum(np.exp(tau * (b_top - b_top.max()))))
    else:
        w = np.exp(tau * (b - b.max()))
        w /= w.sum()
        lse_approx = b.max() + (1.0 / tau) * np.log(np.sum(np.exp(tau * (b - b.max()))))

    obj = (1.0 / tau) * lse_approx * tau  # = lse_approx * ... 
    # Actually f = (1/tau)*log(sum exp(tau*b)) = b_max + (1/tau)*log(sum exp(tau*(b-b_max)))
    # Let me just return the gradient, we track the true score separately.
    
    # Gradient: g[i] = sum_k w[k] * T[k,i] = sum_k w[k] * a[k-i] (valid)
    # = np.correlate(w, a, 'full')[:n]
    g = np.correlate(w, a, mode='full')[:n]
    return g, b.max()


def fista_optimize(a_init, time_limit, tau_schedule, top_k_frac=0.25,
                   use_backtracking=False, verbose=True):
    """
    FISTA on the log-sum-exp smoothed objective.
    tau_schedule: list of (tau, fraction_of_time) pairs.
    """
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a_prev = a.copy()
    t_mom = 1.0
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    deadline = t0 + time_limit
    
    if verbose:
        print(f"  FISTA start: n={n}, score={best_score:.8f}")
    
    total_frac = sum(f for _, f in tau_schedule)
    it = 0
    for tau, frac in tau_schedule:
        if time.monotonic() > deadline:
            break
        stage_deadline = deadline  # Use overall deadline, just respect ordering
        stage_start = time.monotonic()
        
        while time.monotonic() < deadline:
            # Nesterov momentum
            t_old = t_mom
            t_mom = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
            mu = (t_old - 1.0) / t_mom
            y = a + mu * (a - a_prev)
            y = proj_simplex(y)
            
            # Compute gradient
            g, max_b = smooth_score_grad(y, tau, top_k_frac)
            
            # Step size: 1/L where L ~ tau * ||y||_2^2
            L = tau * np.dot(y, y)
            step = 1.0 / max(L, 1e-12)
            
            # Gradient step + projection
            a_new = proj_simplex(y - step * g)
            
            a_prev = a
            a = a_new
            it += 1
            
            # Check score periodically
            if it % 100 == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                if verbose and it % 10000 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"    FISTA it={it} tau={tau:.2f} t={elapsed:.0f}s best={best_score:.8f}")
    
    # Final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"  FISTA done: {elapsed:.0f}s, score={best_score:.8f}")
    return best_a, best_score

def solve_lp_best_response(a_fixed, time_limit=30.0, solver=None):
    """
    Given fixed a (on simplex), find b on simplex minimizing max_k conv(a,b)[k].
    Returns (b, M_optimal).
    """
    if not HAS_CVXPY:
        return None, None
    
    n = len(a_fixed)
    total_k = 2 * n - 1
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # conv(a, b)[k] = sum_i a[k-i] * b[i] for valid (k-i, i)
    # Build constraint matrix: T[k, i] = a[k-i] if 0 <= k-i < n else 0
    # To avoid building the full matrix, use a few different approaches:
    
    # For moderate n (< 2000), build the matrix directly
    if n <= 2000:
        T = np.zeros((total_k, n))
        for j in range(n):
            T[j:j+n, j] = a_fixed
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    else:
        # Use a subset of constraints (every other one, plus the center)
        constraints = [cp.sum(b_var) == 1]
        # This is harder; for large n, skip LP
        return None, None
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers_to_try = []
    if solver:
        solvers_to_try.append(solver)
    solvers_to_try.extend([cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS, cp.SCS])
    
    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                break
        except Exception:
            continue
    else:
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    b_val[np.isnan(b_val)] = 0
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(max(0.0, M_var.value))


def lp_alternating(a_init, time_limit, solver=None, n_max_iters=10000):
    """
    LP best-response alternating with Nesterov dual averaging.
    """
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()  # Nesterov dual variable
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP: find best response b to y
        b_new, M_val = solve_lp_best_response(y, time_limit=max(1.0, time_limit * 0.1))
        if b_new is None:
            break
        
        # Nesterov dual averaging update
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        # Primal update (running average)
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        # Track best
        sc_a = compute_score(a)
        sc_y = compute_score(y)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        if (it + 1) % 20 == 0:
            if best_score < prev_best - 1e-12:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            
            if stall >= 15:
                # Perturb to escape
                rng = np.random.default_rng(it)
                noise = 0.02 * rng.standard_normal(n)
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                stall = 0
            
            if (it + 1) % 100 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LPalt it={it+1} t={elapsed:.0f}s best={best_score:.8f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        print(f"\n=== Phase 6: Try n=1400 (score={best_score:.8f}, t={t_n1400:.0f}s) ===")
        n_target = 1400
        init_1400 = interpolate_sequence(best_seq, n_target)
        
        tau_sched_1400 = [
            (100.0, 0.20),
            (50.0, 0.20),
            (25.0, 0.20),
            (12.0, 0.20),
            (7.0, 0.20),
        ]
        t_sub = t_n1400 * 0.7
        a_1400, _ = fista_optimize(init_1400, t_sub, tau_sched_1400, top_k_frac=0.2)
        try_update(a_1400, "fista_n1400")
        
        rem = deadline - time.monotonic()
        if rem > 30 and HAS_CVXPY and best_score < 1.503:
            # Only try LP if we're close
            n_cur = len(best_seq)
            if n_cur == n_target:
                a_1400_lp, _ = lp_alternating(best_seq, min(rem * 0.5, 60.0), solver, n_max_iters=5000)
                try_update(a_1400_lp, "lp_n1400")
    
    # === Phase 7: Try n=1600 if time ===
    rem = deadline - time.monotonic()
    if rem > 150 and len(best_seq) < 1550:
        t_n1600 = min(rem * 0.25, 200.0)
        print(f"\n=== Phase 7: Try n=1600 (score={best_score:.8f}, t={t_n1600:.0f}s) ===")
        n_target = 1600
        init_1600 = interpolate_sequence(best_seq, n_target)
        
        tau_sched_1600 = [
            (100.0, 0.25),
            (50.0, 0.25),
            (25.0, 0.25),
            (12.0, 0.25),
        ]
        a_1600, _ = fista_optimize(init_1600, t_n1600, tau_sched_1600, top_k_frac=0.2)
        try_update(a_1600, "fista_n1600")
    
    # === Final: one more FISTA pass on best ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t_fin = min(rem * 0.5, 30.0)
        tau_final = [(3.0, 0.3), (2.0, 0.3), (1.5, 0.4)]
        a_fin, _ = fista_optimize(best_seq, t_fin, tau_final, top_k_frac=0.1)
        try_update(a_fin, "final_fista")
    
    # Final verification
    if best_seq is None:
        best_seq = np.ones(1200) / 1200.0
    
    best_seq = np.maximum(best_seq, 0.0)
    final_local = compute_score(best_seq)
    print(f"\nFINAL (local): score={final_local:.8f}, n={len(best_seq)}")
    
    # Call evaluate_sequence for the true score
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        final_eval = evaluate_sequence(result)
        print(f"FINAL (evaluate_sequence): {final_eval:.8f}")
    except Exception as e:
        print(f"evaluate_sequence error: {e}")
    
    return result
