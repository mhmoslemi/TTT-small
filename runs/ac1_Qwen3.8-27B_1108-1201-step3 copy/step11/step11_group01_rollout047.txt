```python
def fast_convolve(a):

def softmax_from_z(z):
    """Convert log-space to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:

def z_from_a(a):

def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a_fixed):
    """Build the convolution matrix T such that T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_subgradient(a, k_star):
    """Compute subgradient of (a*a)[k_star] with respect to a.
    Result: 2*a[k_star - j] for valid j, else 0."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    for j in range(lo, hi + 1):
        g[j] = 2.0 * a[k_star - j]
    return g


def get_subgradient_fast(a, k_star):
    """Faster subgradient computation using slicing."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    # g[j] = 2*a[k_star - j] for j in [lo, hi]
    # k_star - j ranges from k_star - hi to k_star - lo
    src_lo = k_star - hi
    src_hi = k_star - lo
    g[lo:hi+1] = 2.0 * a[src_lo:src_hi+1][::-1]
    return g


def nesterov_subgrad(a0, time_limit, max_iters=10000000, alpha_param=0.01):
    """Nesterov accelerated subgradient on the probability simplex.
    Uses active set tracking for better subgradients."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
        a = np.maximum(a, 0)
        a = a / a.sum()
    
    y = a.copy()  # Nesterov look-ahead point
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0
    
    # Step size schedule: eta_t = 2 / (L * sqrt(t)) where L is Lipschitz constant
    # L = 4*n * max(a) approximately, but we'll use adaptive
    # Actually for f(a) = 2*n*max_k (a*a)[k] on the simplex,
    # L ~ 4*n (since ||∇((a*a)[k])||_2 <= 2*||a||_2 <= 2)
    # So L_f = 2*n * 2 = 4*n
    # But we're on a bounded domain, so we can use larger steps
    
    L = 4.0 * n  # Lipschitz constant (upper bound)
    
    active_set = None
    active_decay = 0.95  # How quickly to update active set
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        t_k = it + 1
        b = fast_convolve(y)
        max_b = b.max()
        k_star = int(np.argmax(b))
        
        # Track active set (positions where convolution is close to max)
        threshold = max_b * 0.99
        active = np.where(b >= threshold)[0]
        
        # Subgradient: average over active set for smoother gradient
        if len(active) <= 3:
            g = get_subgradient_fast(y, k_star)
        else:
            g = np.zeros(n)
            for k in active[:20]:  # Limit to top 20 for speed
                g += get_subgradient_fast(y, int(k))
            g /= len(active[:20])
        
        # Nesterov step size
        eta = alpha_param / np.sqrt(t_k + 1.0)
        
        # Gradient step
        a_old = a.copy()
        a_new = a - eta * g
        a_new = project_simplex(a_new)
        
        # Nesterov momentum
        momentum = (t_k - 1.0) / (t_k + 1.0)
        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy > 1e-15:
            y_new = y_new / sy
        else:
            y_new = a_new.copy()
        
        a = a_new
        y = y_new
        
        # Track best
        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        
        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 15:
                print(f"    NSubgrad iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} curr={sc_a:.8f}")
                last_print = elapsed
    
    return best_a, best_score


def solve_lp_best_response(a_fixed, solver=None):
    """Solve the LP best response problem."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:

def lse_objective_grad(z_in, T_val, n):
    """LogSumExp smoothed objective and gradient in log-space."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=50.0, T_end=0.05, n_stages=20):
    """L-BFGS optimization with LogSumExp smoothing and annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': 50, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxfun': 50 * 200,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(res.x)
            sc = score_normalized(a_check)
            if (si + 1) % 5 == 0:
                print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.2f}: score={sc:.8f}")
        except Exception:
            pass

    return softmax_from_z(z)


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
```

```python
def lse_grad_fast(a, w, n):
    """Compute grad_a = 2 * sum_k w[k] * shifted_a[k]"""
    grad_a = np.zeros(n)
    # For each k, grad_a[j] += 2*w[k]*a[k-j] for j in [max(0,k-n+1), min(n-1,k)]
    # This is equivalent to: for each k, take a[k::-1][:min(n,k+1)] and add 2*w[k]*that to grad_a
    for k in range(2*n-1):
        if w[k] < 1e-18:
            continue
        lo = max(0, k-(n-1))
        hi = min(n-1, k)
        # grad_a[lo:hi+1] += 2*w[k]*a[k-lo : k-hi+1 : -1]
        a_slice = a[k-hi:k-lo+1][::-1]  # length hi-lo+1
        grad_a[lo:hi+1] += 2.0 * w[k] * a_slice
    return grad_a
```
