Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated gradient optimization
with multi-stage LogSumExp smoothing, LP best-response interleaving, and coordinate descent.
Key innovation: O(n log n) gradient via FFT cross-correlation enables 100x more iterations.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_corr(w, a):
    """
    Compute cross-correlation R[i] = sum_k w[k] * a[k-i] for i = 0, ..., n-1.
    where w has length 2n-1 and a has length n.
    """
    n = len(a)
    size = 3 * n - 2  # length of w * a_rev
    fft_size = 1 << (size - 1).bit_length()
    a_rev = a[::-1]
    W = rfft(w, fft_size)
    A = rfft(a_rev, fft_size)
    conv = irfft(W * A, fft_size)[:size]
    # R[i] = conv[n-1+i] for i = 0, ..., n-1
    return conv[n-1:n-1+n]


def lse_objective_and_grad(a, T):
    """
    Compute LogSumExp smoothed objective and its gradient.
    Objective: (2n/T) * LSE(T * conv(a,a))
    Returns (objective, gradient) where gradient is d(obj)/da.
    """
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    lse_val = (Tb_max + np.log(exp_Tb.sum())) / T
    obj = 2.0 * n * lse_val
    
    # Softmax weights
    w = exp_Tb / exp_Tb.sum()
    
    # Gradient: (2n) * cross_corr(w, a)
    grad = 2.0 * n * fast_cross_corr(w, a)
    
    return obj, grad


def lse_objective_and_grad_softmax(z, T):
    """
    Objective and gradient in softmax parameterization.
    a = softmax(z), returns (obj, grad_z).
    """
    n = len(z)
    z_shift = z - z.max()
    a = np.exp(z_shift)
    a = a / a.sum()
    
    obj, grad_a = lse_objective_and_grad(a, T)
    
    # Chain rule for softmax: grad_z_i = a_i * (grad_a_i - sum_j a_j * grad_a_j)
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)
    
    return obj, grad_z

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_corr(w, a):
    """
    Compute R[i] = sum_k w[k] * a[k-i] for i = 0, ..., n-1.
    w has length 2n-1, a has length n.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    size = 3 * n - 2
    fft_size = 1 << (size - 1).bit_length()
    a_rev = a[::-1]
    W = rfft(w, fft_size)
    A = rfft(a_rev, fft_size)
    conv = irfft(W * A, fft_size)[:size]
    return conv[n - 1:n - 1 + n]

    """

    grad = 2n * cross_corr(softmax(T*b), a)

    obj = (2.0 * n / T) * (Tb_max + np.log(exp_sum))

        print(f"    LSE stage={stage_idx} T={T:.3f} ({elapsed:.0f}s): sc={score:.8f} best={best_score:.8f}")

        z = np.log(np.maximum(a_new, 1e-15))

    return best_a, best_score


def mirror_descent_fft(a0, time_limit, eta0=0.03):
    """
    Multiplicative mirror descent with FFT gradient.
    Update: a_i <- a_i * exp(-eta * g_i) where g is projected gradient.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    no_improve = 0
    T = 5.0
    eta = eta0
    rng = np.random.default_rng(99)

    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute gradient using current T
        _, grad = lse_objective_and_grad(a, T)
        # Project onto simplex tangent: subtract weighted mean
        grad_proj = grad - np.dot(a, grad)
        # Adaptive step
        eta_eff = eta / (1.0 + 0.01 * np.sqrt(it))

        # Multiplicative update
        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - eta_eff * grad_proj
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        score = evaluate_local(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        # Reduce T over time
        T = max(0.05, T * 0.9995)

        if no_improve > 300:
            if no_improve > 2000:
                break
            # Perturb
            if no_improve > 500:
                noise = rng.standard_normal(n) * 0.01
                a = a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
                T = max(T, 3.0)

        if (it + 1) % 20000 == 0:
            print(f"    MD iter={it+1} ({time.monotonic()-t0:.0f}s): best={best_score:.8f} T={T:.3f}")

    return best_a, best_score


def coord_descent_polish(a0, time_limit):
    """Coordinate descent: reduce top peak contributors, redistribute."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0

    for step in range(300000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Find all near-max peaks
        threshold = max_b * (1.0 - 1e-6)
        peaks = np.where(b >= threshold)[0]

        improved = False
        for pk in peaks[:4]:
            if improved:
                break
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            top_m = min(10, len(idx))
            top_pos = idx[np.argsort(contribs)[::-1][:top_m]]

            for pos in top_pos:
                if a[pos] < 1e-14:
                    continue
                for factor in [0.9995, 0.999, 0.998, 0.997, 0.995, 0.993, 0.99, 0.985]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-18:
                        break
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 1000:
                break
        else:
            stall = 0

        if (step + 1) % 50000 == 0:
            print(f"    CD ({time.monotonic()-t0:.0f}s): {best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate structured initial sequences."""
    x = np.linspace(0, 1, n)
    choice = rng.integers(0, 8)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-4.0 * x * (1 - x))
    elif choice == 3:
        a = 0.4 + 0.4 * np.cos(2 * np.pi * 1.7 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-3.0 * (x - 0.4) ** 2 / 0.15)
    elif choice == 5:
        # Asymmetric bimodal
        a = np.exp(-10 * (x - 0.25) ** 2) + 0.7 * np.exp(-8 * (x - 0.7) ** 2)
    elif choice == 6:
        # Rising ramp with bump
        a = x * (1 - x) * (0.5 + 0.5 * np.cos(4 * np.pi * x))
        a = np.maximum(a, 0.01)
    else:
        a = (1 - x) ** 0.7 + 0.3 * np.sin(3 * np.pi * x) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main optimization: combines LP alternating, FFT-accelerated L-BFGS,
    mirror descent, and coordinate descent for minimizing the convolution peak.
    """
    global _lp_solver
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: LP Refinement (25% time) ===
    print(f"\n=== Phase 1: LP ({best_score:.8f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.25, 400.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a_lp, _ = lp_alternate(best_seq, t1, solver, max_iters=40000)
        try_update(a_lp, "LP1")

    # === Phase 2: FFT L-BFGS Multi-stage (35% time) ===
    print(f"\n=== Phase 2: L-BFGS ({best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.35, 550.0)
    if t2 > 20:
        print(f"  Budget: {t2:.0f}s")
        T_stages = [50.0, 30.0, 18.0, 10.0, 6.0, 4.0, 2.5, 1.5, 0.9, 0.5, 0.3, 0.18, 0.1, 0.06, 0.04, 0.025]
        a_lb, _ = lbgfs_multistage(best_seq, t2, T_stages=T_stages)
        try_update(a_lb, "LBF1")

    # === Phase 3: LP interleave (10% time) ===
    print(f"\n=== Phase 3: LP2 ({best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.10, 150.0)
    if t3 > 20:
        print(f"  Budget: {t3:.0f}s")
        a_lp2, _ = lp_alternate(best_seq, t3, solver, max_iters=20000)
        try_update(a_lp2, "LP2")

    # === Phase 4: Mirror Descent (10% time) ===
    print(f"\n=== Phase 4: Mirror Descent ({best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 150.0)
    if t4 > 15:
        print(f"  Budget: {t4:.0f}s")
        a_md, _ = mirror_descent_fft(best_seq, t4, eta0=0.03)
        try_update(a_md, "MD")

    # === Phase 5: Multi-n Exploration (10% time) ===
    print(f"\n=== Phase 5: Multi-n ({best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.10, 150.0)
    if t5 > 20:
        n_best = len(best_seq)
        n_candidates = sorted(set([
            max(400, n_best - 100),
            max(400, n_best + 100),
            max(400, n_best + 200),
            n_best
        ]))
        n_candidates = [x for x in n_candidates if x != n_best]
        if n_candidates:
            per_n = t5 / len(n_candidates)
            for n_try in n_candidates:
                if time.monotonic() > deadline - 10:
                    break
                a_interp = interpolate_sequence(best_seq, n_try)
                t_sub = min(per_n, deadline - time.monotonic() - 5)
                if t_sub < 10:
                    continue
                print(f"  n={n_try}, budget={t_sub:.0f}s")
                # Quick L-BFGS at this n
                T_stages_short = [20.0, 10.0, 5.0, 3.0, 1.5, 0.8, 0.4, 0.2, 0.1]
                a_sub, _ = lbgfs_multistage(a_interp, t_sub, T_stages=T_stages_short)
                try_update(a_sub, f"n{n_try}")

    # === Phase 6: Final L-BFGS polish (5% time) ===
    print(f"\n=== Phase 6: Final L-BFGS ({best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.08, 100.0)
    if t6 > 10:
        T_final = [5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.15, 0.08, 0.04, 0.02]
        a_fin, _ = lbgfs_multistage(best_seq, t6, T_stages=T_final)
        try_update(a_fin, "LBFin")

    # === Phase 7: Coordinate Descent Polish (remaining time) ===
    print(f"\n=== Phase 7: Coord Descent ({best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t7 = max(5.0, rem - 5.0)
    if t7 > 5:
        print(f"  Budget: {t7:.0f}s")
        a_cd, _ = coord_descent_polish(best_seq, t7)
        try_update(a_cd, "CD")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: {best_score:.10f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
