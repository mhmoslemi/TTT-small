def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1

    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

    n = len(a)

    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)

    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)

        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def spectral_flatten(a, n_iters=50, blend=0.4, deadline=None):
    """Iterative spectral flattening."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)

        mags = np.abs(F)
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])

        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

    return a

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS, cp.GUROBI, cp.MOSEK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]
    solved = False
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue

    if not solved:
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(max(0, M_var.value))


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp, deadline=None):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def spectral_flatten(a, n_iters=50, blend=0.4, deadline=None):
    """Iterative spectral flattening (Gerchberg-Saxton style)."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)

        mags = np.abs(F)
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])

        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

    return a


def nesterov_lp_core(a0, time_limit, solver, n_max_iters=5000000):
    """
    Heavily optimized Nesterov LP alternating method.
    Uses adaptive restarts and tracks both primal and dual iterates.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    y = a.copy()
    prev_best = best_score
    stall_count = 0
    patience = 0
    rng = np.random.default_rng(777)
    lp_times = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)
        lp_time = time.monotonic() - lp_t0
        lp_times.append(lp_time)

        if b_new is None:
            continue

        # Monitor LP speed
        if len(lp_times) > 100 and np.mean(lp_times[-50:]) > 15.0:
            print(f"    LP too slow ({np.mean(lp_times[-50:]):.2f}s/iter), stopping at iter {it+1}")
            break

        # Nesterov update: alpha = 2/(t+1)
        t = it + 1
        alpha = 2.0 / (t + 1)

        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a = (1.0 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        # Check scores
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Stall detection every 100 iterations
        if (t + 1) % 100 == 0:
            if best_score < prev_best - 1e-14:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            # Adaptive restart
            if stall_count >= 10:
                strength = min(0.08, 0.005 * stall_count + 0.01)
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 12)
                phase = rng.uniform(0, 2 * np.pi)
                noise = strength * np.cos(freq * x + phase)
                a_new = a + noise
                a_new = np.maximum(a_new, 0)
                sa = a_new.sum()
                if sa > 1e-15:
                    a = a_new / sa
                    y = a.copy()
                stall_count = 0
                patience += 1
                if patience >= 8:
                    # Bigger restart
                    a = spectral_flatten(a, n_iters=5, blend=0.3)
                    a = np.maximum(a, 0)
                    sa = a.sum()
                    if sa > 1e-15:
                        a = a / sa
                        y = a.copy()
                    patience = 0

            if (t + 1) % 5000 == 0:
                elapsed = time.monotonic() - t0
                avg_lp = np.mean(lp_times[-100:]) * 1000 if lp_times else 0
                print(f"    NesterovLP iter={t+1} ({elapsed:.0f}s): score={best_score:.10f} lp={avg_lp:.1f}ms")

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage=10000):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    n_stages = len(T_stages)

    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        stages_left = n_stages - stage_i
        iters = max(20, min(maxiter_per_stage, int(remaining * 30 / max(1, stages_left))))

        def func(z_in):
            obj, _ = logsumexp_obj_grad(z_in, n, stage_T)
            return obj

        def jac(z_in):
            _, g = logsumexp_obj_grad(z_in, n, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-25,
                         'gtol': 1e-20, 'maxfun': iters * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def peak_shave(a, time_limit, n_steps=100000):
    """Directly target the maximum of the autocorrelation using multiplicative updates."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return None, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.03
    scale = 1.0
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        threshold = max_b * (1.0 - 2e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        eta_eff = eta * scale
        log_scale = -eta_eff * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 40:
            scale *= 0.7
            no_improve = 0
            if scale < 0.005:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturbed_sequence(a, rng, strength=0.03):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 10)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s

        local_update(a_lb, "lbgfs_fine")

    # === Phase 3: Peak shaving (5% of time) ===
    rem = deadline - time.monotonic()
    t_peak = rem * 0.05
    print(f"=== Phase 3: Peak Shaving (score={best_score:.10f}) ===")
    if t_peak > 10:
        a_pk, _ = peak_shave(best_seq, t_peak, n_steps=200000)
        local_update(a_pk, "peak_shave")

    # === Phase 4: Second LP round on best found (8% of time) ===
    rem = deadline - time.monotonic()
    t_lp2 = rem * 0.08
    print(f"=== Phase 4: LP Round 2 (score={best_score:.10f}) ===")
    if t_lp2 > 15:
        print(f"  Budget: {t_lp2:.0f}s")
        a_lp2, _ = nesterov_lp_core(best_seq, t_lp2, solver, n_max_iters=2000000)
        local_update(a_lp2, "nesterov_lp_2")

    # === Phase 5: Try larger n (5% of time) ===
    rem = deadline - time.monotonic()
    if rem > 60:
        print(f"=== Phase 5: Try larger n (score={best_score:.10f}) ===")
        for target_n in [1500, 2000]:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_sub = min(rem * 0.5, 80.0)
            if t_sub < 20:
                break
            print(f"  Trying n={target_n}, budget={t_sub:.0f}s")
            init_n = interpolate_sequence(best_seq, target_n)
            init_n = spectral_flatten(init_n, n_iters=20, blend=0.3)
            a_n, _ = nesterov_lp_core(init_n, t_sub * 0.8, solver, n_max_iters=1000000)
            local_update(a_n, f"lp_n{target_n}")

    # === Phase 6: Fizz perturbations (remaining time) ===
    rem = deadline - time.monotonic()
    if rem > 30:
        print(f"=== Phase 6: Fizz (score={best_score:.10f}) ===")
        n_fizz = min(int(rem / 15), 10)
        for trial in range(n_fizz):
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            a_pert = perturbed_sequence(best_seq, rng, strength=0.01 + 0.005 * trial)
            t_sub = min(rem - 5, 20.0)
            if t_sub < 5:
                break
            a_opt, _ = nesterov_lp_core(a_pert, t_sub, solver, n_max_iters=200000)
            local_update(a_opt, f"fizz_{trial}")

    # === Phase 7: Final L-BFGS polish (last bit of time) ===
    rem = deadline - time.monotonic()
    if rem > 10:
        print(f"=== Phase 7: Final Polish (score={best_score:.10f}) ===")
        t_fin = min(rem * 0.8, 30.0)
        if t_fin > 5:
            T_stages_fin = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
            a_fin = lbgfs_refine(best_seq, t_fin, T_stages_fin, maxiter_per_stage=3000)
            local_update(a_fin, "final_lbgfs")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)}) at {elapsed:.1f}s")
        test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            eval_score = evaluate_sequence(test_seq)
            print(f"Verified with evaluate_sequence: {eval_score:.10f}")
        except Exception as e:
            print(f"Verification error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
