def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_q_sparse(k, n):
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)

def build_gp_sparse(k, n):
    """Build gradient of a^T Q_k a w.r.t. a, which is 2*Q_k*a.
    Returns a sparse matrix G such that G @ a = 2*Q_k*a."""
    G = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        G[j, partner] = 1.0
    return csc_matrix(G)

    n = len(a_fixed)
    size = 2 * n - 1
    rows = []
    cols = []
    data = []

            cols.append(j)
            data.append(a_fixed[j])

def iterative_active_qp(a0, solver, time_limit, max_active=40):

    Iterative active-set method for:
    min t s.t. a^T Q_k a <= t for all k, sum(a)=1, a>=0

    - Add constraints that are violated (or near-violation)
    - Remove constraints with dual variable near zero
    - Stop when KKT conditions are satisfied to tolerance

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    t0 = time.monotonic()
    
    # Initialize active set
    b = fast_convolve(a)
    max_b = b.max()
    threshold = max_b * (1.0 - 1e-3)
    active_set = set(np.where(b >= threshold)[0].tolist())
    
    if not active_set:
        active_set = {int(np.argmax(b))}

        if time.monotonic() - t0 > time_limit:
            break

        a_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()

        for k in sorted(active_set):
            Q = build_q_sparse(int(k), n)
            Q_list.append((k, Q))
            constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

                    solved = True
                    break
            except Exception:
                continue

            break
        
        a_new = np.asarray(a_var.value, dtype=np.float64)

                pass

        for k in violated:
            active_set.add(int(k))
        
        # Remove constraints with zero dual (and not the max)

        for k in list(active_set):

                    removable.append(k)

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def build_conv_matrix_fast(a):
    """Fast vectorized construction of (2n-1) x n matrix T where T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    j_idx = np.arange(n)
    rows = []
    cols = []
    data = []
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, size - 1)
        k_vals = np.arange(k_start, k_end + 1)
        rows.extend(k_vals.tolist())
        cols.extend([j] * len(k_vals))
        data.extend([a[j]] * len(k_vals))
    return csc_matrix((data, (rows, cols)), shape=(size, n))


def get_lp_solver():

def evaluate_local(a):

def solve_lp_best_response(a_fixed, solver):

                b_val = np.maximum(b_val, 0)

        alpha = 2.0 / (it + 2.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)

            y = a.copy()
        else:
            y = y / sy

        denom = (it + 1.0) + alpha * (it + 2.0)

        a = np.maximum(a, 0)

            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

            best_score = score_a
            best_a = a.copy()

def multiplicative_weights_refine(a0, time_limit, n_steps=100000):
    """Fast multiplicative weights / mirror descent local search (no solver needed)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.05
    no_improve = 0
    decay = 0.95

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Subgradient: weighted average of active peaks
        threshold = max_b * (1.0 - 1e-3)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        a_new = a * np.exp(-eta * g)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 50:
            eta *= decay
            no_improve = 0
            if eta < 0.001:
                break

    return best_a, best_score

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    # Initialize active set
    b = fast_convolve(a)
    max_b = b.max()
    threshold = max_b * (1.0 - 5e-4)
    active_set = set(np.where(b >= threshold)[0].tolist())
    if not active_set:
        active_set = {int(np.argmax(b))}

    for iteration in range(300):
        if time.monotonic() - t0 > time_limit:
            break

        # Build and solve QP
        a_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1]
        k_sorted = sorted(active_set)

        for k in k_sorted:
            Q = build_q_sparse(int(k), n)
            constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

        prob = cp.Problem(cp.Minimize(t_var), constraints)

        solved = False
        solvers_to_try = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.ECOS]
        seen = set()
        for s_solver in solvers_to_try:
            if s_solver is None or s_solver in seen:
                continue
            seen.add(s_solver)
            try:
                prob.solve(solver=s_solver)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    solver_used = s_solver
                    break
            except Exception:
                continue

        if not solved:
            break

        a_new = np.asarray(a_var.value, dtype=np.float64)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()

        a = a_new

        # Update active set
        b_new = fast_convolve(a)
        max_new = b_new.max()

        # Add significantly violated constraints
        violated = np.where(b_new > max_new * (1.0 + 1e-8))[0]
        for k in violated:
            active_set.add(int(k))

        # Add near-active constraints (within 0.1% of max)
        near = np.where(b_new > max_new * (1.0 - 1e-3))[0]
        for k in near:
            active_set.add(int(k))

        # Remove constraints that are far from active
        to_remove = []
        for k in list(active_set):
            if b_new[k] < max_new * (1.0 - 5e-3):
                to_remove.append(k)

        # Keep at least 3 constraints and the top ones
        keep_count = max(3, len(active_set) - len(to_remove))
        if len(to_remove) < len(active_set):
            for k in to_remove:
                active_set.discard(k)

        # Cap size
        if len(active_set) > max_active:
            active_list = list(active_set)
            active_list.sort(key=lambda k: -b_new[k])
            active_set = set(active_list[:max_active])

        # Convergence check
        if (b_new <= max_new * (1.0 + 1e-8)).all():
            # Check if we've stabilized (score not changing)
            if iteration > 5:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lse_objective(z, n, T_temp):
    """Compute LogSumExp-smoothed objective."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    return 2.0 * n * lse


def lse_gradient(z, n, T_temp):
    """Compute gradient of LogSumExp-smoothed objective w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    w = exp_terms / exp_terms.sum()

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S_val = np.dot(grad_a, a)
    grad_z = a * (grad_a - S_val)
    return 2.0 * n * grad_z


def lbgfs_refine(a0, time_limit, T_stages=None, maxiter_per_stage=2000):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    if T_stages is None:
        T_stages = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]

    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    n_stages = len(T_stages)
    iters_per_stage = max(20, maxiter_per_stage // n_stages)

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def fun(x):
            return lse_objective(x, n, stage_T)

        def jac(x):
            return lse_gradient(x, n, stage_T)

        try:
            res = minimize(
                fun=fun, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-10, 'maxfun': iters_per_stage * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LP alternating, active-set QCQP, multiplicative weights,
    multi-n exploration, and L-BFGS polish.
    """
    rng = np.random.default_rng(seed)
    safety_margin = 30.0
    deadline = time.monotonic() + max(1.0, float(budget_s) - safety_margin)
    total_budget = float(budget_s) - safety_margin

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Total budget: {total_budget:.0f}s, deadline in {deadline - time.monotonic():.0f}s")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = np.cos(np.pi * x / 2) ** 2
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n_best = len(best_seq)

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: LP alternating for n=1200 (main workhorse, 35% budget) ===
    print(f"\n=== Phase 1: LP Alternating n={n_best} (score={best_score:.10f}) ===")
    t_alloc = min((deadline - time.monotonic()) * 0.35, 550.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        t0 = time.monotonic()
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=1000000)
        update_best(a_lp, "lp_phase1")
        print(f"  Phase 1 done in {time.monotonic()-t0:.1f}s, score={best_score:.10f}")

    # === Phase 2: Multiplicative weights fast local search (5% budget) ===
    print(f"\n=== Phase 2: Multiplicative Weights (score={best_score:.10f}) ===")
    t_alloc = min((deadline - time.monotonic()) * 0.05, 80.0)
    if t_alloc > 10:
        a_mw, _ = multiplicative_weights_refine(best_seq, t_alloc, n_steps=200000)
        update_best(a_mw, "mult_weights")

    # === Phase 3: Active-set QCQP polish (10% budget) ===
    print(f"\n=== Phase 3: Active-Set QCQP (score={best_score:.10f}) ===")
    t_alloc = min((deadline - time.monotonic()) * 0.10, 150.0)
    if t_alloc > 20:
        a_qp, _ = iterative_active_qp(best_seq, solver, t_alloc, max_active=40)
        update_best(a_qp, "active_qp_1")

    # === Phase 4: Second LP round from polished (15% budget) ===
    print(f"\n=== Phase 4: LP Round 2 (score={best_score:.10f}) ===")
    t_alloc = min((deadline - time.monotonic()) * 0.15, 250.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        t0 = time.monotonic()
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=500000)
        update_best(a_lp2, "lp_phase4")
        print(f"  Phase 4 done in {time.monotonic()-t0:.1f}s, score={best_score:.10f}")

    # === Phase 5: Active-set QCQP again (5% budget) ===
    print(f"\n=== Phase 5: QCQP Round 2 (score={best_score:.10f}) ===")
    t_alloc = min((deadline - time.monotonic()) * 0.05, 80.0)
    if t_alloc > 15:
        a_qp2, _ = iterative_active_qp(best_seq, solver, t_alloc, max_active=30)
        update_best(a_qp2, "active_qp_2")

    # === Phase 6: Multi-n exploration (20% budget) ===
    print(f"\n=== Phase 6: Multi-n Exploration (score={best_score:.10f}) ===")
    n_candidates = [1197, 1198, 1199, 1201, 1202, 1203]
    # Remove current best n
    n_best_curr = len(best_seq)
    n_candidates = [n for n in n_candidates if n != n_best_curr]

    rem = deadline - time.monotonic()
    t_total_n = min(rem * 0.20, 300.0)
    if t_total_n > 60:
        t_per_n = t_total_n / len(n_candidates)
        print(f"  Total budget: {t_total_n:.0f}s, per n: {t_per_n:.0f}s")

        for n_try in n_candidates:
            if time.monotonic() > deadline - 60:
                print("  Running out of time, stopping n exploration")
                break

            print(f"  --- Trying n={n_try} (time_left={deadline-time.monotonic():.0f}s) ---")
            init_n = interpolate_sequence(best_seq, n_try)

            # Short LP alternating
            t_lp = min(t_per_n * 0.5, 60.0)
            if t_lp > 10:
                a_n, _ = nesterov_lp_refine(init_n, t_lp, solver, n_max_iters=100000)
            else:
                a_n = init_n

            # Active-set polish
            t_qp = min(t_per_n * 0.4, 30.0)
            if t_qp > 8:
                a_n2, _ = iterative_active_qp(a_n, solver, t_qp, max_active=25)
                if a_n2 is not None:
                    a_n = a_n2

            # Multiplicative weights
            t_mw = min(t_per_n * 0.1, 10.0)
            if t_mw > 3:
                a_n3, _ = multiplicative_weights_refine(a_n, t_mw, n_steps=50000)
                if a_n3 is not None:
                    a_n = a_n3

            update_best(a_n, f"n_{n_try}")

    # === Phase 7: L-BFGS polish on best (5% budget) ===
    print(f"\n=== Phase 7: L-BFGS Polish (score={best_score:.10f}) ===")
    t_alloc = min((deadline - time.monotonic()) * 0.05, 100.0)
    if t_alloc > 20:
        a_lb = lbgfs_refine(best_seq, t_alloc,
                            T_stages=[2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01],
                            maxiter_per_stage=3000)
        update_best(a_lb, "lbgfs")

    # === Phase 8: Final active-set polish (remaining time) ===
    print(f"\n=== Phase 8: Final QCQP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_final = min(rem * 0.6, 120.0)
        a_final, _ = iterative_active_qp(best_seq, solver, t_final, max_active=35)
        update_best(a_final, "final_qp")

    # === Phase 9: One more multiplicative weights pass ===
    rem = deadline - time.monotonic()
    if rem > 15:
        a_mw_f, _ = multiplicative_weights_refine(best_seq, min(rem * 0.5, 30.0), n_steps=50000)
        update_best(a_mw_f, "final_mw")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        print(f"Time used: {time.monotonic() + safety_margin - (deadline):.1f}s / {budget_s:.0f}s")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
