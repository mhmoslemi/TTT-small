    """Project v onto {x: x >= 0, sum(x) = s}"""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated adaptive
subgradient descent with multi-n exploration and peak-splitting heuristics.

Key differences from previous approach:
- Replaces expensive LP oracle with fast FFT-based subgradients
- Uses RMSProp adaptive learning rate for efficient convergence
- Multi-start with random perturbations to escape local optima
- Peak-splitting heuristic to specifically target the dominant peak
- Coarse-to-fine n exploration
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation (scale-invariant)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: x >= 0, sum(x) = target_sum} using Euclidean projection."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    # Find largest rho such that u[rho-1] - (cssv[rho-1] - target_sum) / rho > 0
    vals = u * rho_candidates - (cssv - target_sum)
    valid = vals > 0
    if not valid.any():
        return np.full(n, target_sum / n)
    rho = np.max(rho_candidates[valid])
    theta = (cssv[rho - 1] - target_sum) / rho
    return np.maximum(v - theta, 0)


def compute_subgrad_fft(a, n_peaks=1):
    """
    Compute subgradient of max(conv(a,a)) w.r.t. a.
    Returns (gradient, max_value, peak_positions).
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    
    # Find peak positions (top n_peaks or all within 99.99% of max)
    # Use a tolerance to capture "flat" peaks
    tol = max_b * 0.9999
    candidates = np.where(b >= tol)[0]
    
    if len(candidates) > n_peaks:
        # Take the top n_peaks by value
        order = np.argsort(b[candidates])[::-1][:n_peaks]
        peaks = candidates[order]
    else:
        peaks = candidates
    
    if len(peaks) == 0:
        peaks = np.array([int(np.argmax(b))])
    
    # Compute subgradient
    # grad[j] = 2 * sum_{k in peaks} w_k * a[k-j] (for valid k-j)
    # Equal weights for simplicity
    grad = np.zeros(n)
    w = 1.0 / len(peaks)
    
    for k in peaks:
        # a[k-j] is nonzero when 0 <= k-j <= n-1, i.e., k-(n-1) <= j <= k
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        # For j in [js, je]: a[k-j]
        # k-js, k-js-1, ..., k-je are the indices into a
        # k-je = k - min(n-1, k)
        # k-js = k - max(0, k-n+1)
        # These are consecutive: from k-je to k-js
        lo = k - je  # = max(0, k-(n-1)) = max(0, k-n+1)
        hi = k - js  # = min(n-1, k)
        # a[lo], a[lo+1], ..., a[hi]
        # But we need a[k-j] for j=js, js+1, ..., je
        # j=js: a[k-js] = a[hi]
        # j=je: a[k-je] = a[lo]
        # So it's a[hi], a[hi-1], ..., a[lo] = a[lo:hi+1][::-1]
        grad[js:je+1] += 2.0 * w * a[lo:hi+1][::-1]
    
    return grad, max_b, peaks


def rmsprop_optimize(a0, time_limit, lr=0.01, beta=0.9, eps=1e-8,
                     n_peaks=1, peak_tol=0.9999, restart_every=2000,
                     noise_scale=0.01, name=""):
    """
    RMSProp optimization of -max(conv(a,a)) on the simplex.
    a0 should be non-negative with sum=1.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = evaluate_local(a)
    print(f"  [{name}] Start: score={best_score:.8f}, n={n}")
    
    t0 = time.monotonic()
    v = np.zeros(n)  # RMSProp second moment
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute subgradient
        grad, max_b, peaks = compute_subgrad_fft(a, n_peaks=n_peaks)
        
        # RMSProp update
        v = beta * v + (1 - beta) * grad * grad
        step = lr / (np.sqrt(v) + eps)
        a_new = a - step * grad
        
        # Project to simplex
        a_new = project_simplex(a_new, 1.0)
        
        # Evaluate
        score = 2.0 * n * fast_convolve(a_new).max()
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        it += 1
        
        # Periodic restart
        if no_improve >= restart_every:
            # Add noise and restart
            rng = np.random.default_rng(it * 7 + 13)
            noise = rng.standard_normal(n) * noise_scale
            a = a + noise
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            else:
                a = best_a.copy()
            v = np.zeros(n)
            no_improve = 0
            noise_scale = max(0.001, noise_scale * 0.5)
        
        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [{name}] iter={it+1} ({elapsed:.0f}s): best={best_score:.8f}, cur={score:.8f}")
    
    return best_a, best_score / (1.0 * 1.0)  # score = 2n * max_conv (since sum=1)


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.arange(old_n) / max(1, old_n - 1)
    new_x = np.arange(new_n) / max(1, new_n - 1)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def peak_splitting(a, n_reduce=5, reduce_factor=0.95, name=""):
    """
    Peak splitting heuristic: reduce top contributors to the main peak,
    redistribute mass to low-convolution positions.
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    peak_k = int(np.argmax(b))
    
    # Find contributing positions
    js = max(0, peak_k - n + 1)
    je = min(n - 1, peak_k)
    idx = np.arange(js, je + 1)
    contribs = a[idx] * a[peak_k - idx]
    
    # Reduce top contributors
    n_reduce_actual = min(n_reduce, len(idx))
    top_pos = idx[np.argsort(contribs)[::-1][:n_reduce_actual]]
    
    a_new = a.copy()
    total_removed = 0.0
    for pos in top_pos:
        if a_new[pos] > 1e-12:
            removed = a_new[pos] * (1 - reduce_factor)
            a_new[pos] -= removed
            total_removed += removed
    
    if total_removed < 1e-15:
        return a_new
    
    # Redistribute to positions with low contribution to any peak
    # Find positions that don't contribute much to the top peaks
    top_peaks = np.argsort(b)[::-1][:3]
    contrib_to_peaks = np.zeros(n)
    for pk in top_peaks:
        pjs = max(0, pk - n + 1)
        pje = min(n - 1, pk)
        for j in range(pjs, pje + 1):
            contrib_to_peaks[j] += b[pk] * a[j] * a[pk - j]
    
    # Prefer positions with low contribution and currently high value
    # (to spread mass more evenly)
    weights = 1.0 / (contrib_to_peaks + 1e-10)
    weights = weights * a_new  # proportional to current mass
    w_sum = weights.sum()
    if w_sum < 1e-15:
        weights = np.ones(n) / n
    else:
        weights = weights / w_sum
    
    a_new += total_removed * weights
    a_new = np.maximum(a_new, 0)
    a_new = a_new / a_new.sum()
    
    return a_new


def coordinate_descent_polish(a0, time_limit, name=""):
    """Coordinate descent: try reducing each top-contributing element."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))
        
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_m = min(10, len(idx))
        top_pos = idx[np.argsort(contribs)[::-1][:top_m]]
        
        improved = False
        for pos in top_pos:
            if a[pos] < 1e-14:
                continue
            for factor in [0.999, 0.997, 0.995, 0.99, 0.98]:
                a_trial = a.copy()
                removed = a_trial[pos] * (1 - factor)
                a_trial[pos] -= removed
                if removed < 1e-16:
                    break
                # Redistribute proportionally to others
                others_sum = a_trial.sum()
                if others_sum < 1e-15:
                    break
                a_trial += removed * a_trial / others_sum
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        it += 1
        if not improved and it > 50000:
            break  # Give up if no improvement for a long time
    
    print(f"  [{name}] CoordDesc done: score={best_score:.8f}")
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated Adam optimization
with multi-n exploration, peak-splitting heuristics, and coordinate descent polish.

Key differences from previous LP-based approach:
- Replaces expensive LP oracle (O(n^2) per iteration) with FFT-based subgradients (O(n log n))
- Enables 100-1000x more iterations in the same time budget
- Uses Adam optimizer with adaptive per-coordinate step sizes
- Multi-start with diverse perturbation strategies
- Peak-splitting heuristic to specifically target the dominant convolution peak
- Coarse-to-fine exploration across different sequence lengths

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation (scale-invariant)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x: x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    vals = u * rho_candidates - (cssv - target_sum)
    valid = vals > 0
    if not valid.any():
        return np.full(n, target_sum / n)
    rho = np.max(rho_candidates[valid])
    theta = (cssv[rho - 1] - target_sum) / rho
    return np.maximum(v - theta, 0)


def compute_subgrad(a, n_peaks=1, peak_tol=0.9999):
    """
    Compute subgradient of max(conv(a,a)) w.r.t. a.
    Returns (gradient, max_value).
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()

    tol = max_b * peak_tol
    candidates = np.where(b >= tol)[0]

    if len(candidates) > n_peaks:
        order = np.argsort(b[candidates])[::-1][:n_peaks]
        peaks = candidates[order]
    else:
        peaks = candidates

    if len(peaks) == 0:
        peaks = np.array([int(np.argmax(b))])

    grad = np.zeros(n)
    w = 1.0 / len(peaks)

    for k in peaks:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        lo = k - je
        hi = k - js
        grad[js:je + 1] += 2.0 * w * a[lo:hi + 1][::-1]

    return grad, max_b


def compute_smooth_grad(a, T):
    """
    Compute gradient of LogSumExp(conv(a,a)/T) w.r.t. a.
    This is a smoothed version of the max subgradient.
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp(b / T - max_b / T)
    w = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        lo = k - je
        hi = k - js
        grad[js:je + 1] += 2.0 * w[k] * a[lo:hi + 1][::-1]

    return grad, max_b


def adam_optimize(a0, time_limit, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8,
                  n_peaks=1, peak_tol=0.9999, restart_every=2000,
                  noise_scale=0.01, smooth_T=None, name=""):
    """
    Adam optimization of max(conv(a,a)) on the simplex (minimize).
    a0 should be non-negative with sum=1.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    print(f"  [{name}] Start: score={best_score:.8f}, n={n}")

    t0 = time.monotonic()
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        # Batch multiple iterations for speed
        batch_size = max(1, min(50, int((time_limit - (time.monotonic() - t0)) * 2000)))
        
        for _ in range(batch_size):
            if time.monotonic() - t0 > time_limit:
                break

            if smooth_T is not None:
                grad, max_b = compute_smooth_grad(a, smooth_T)
            else:
                grad, max_b = compute_subgrad(a, n_peaks=n_peaks, peak_tol=peak_tol)

            t_count += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad * grad
            m_hat = m / (1 - beta1 ** t_count)
            v_hat = v / (1 - beta2 ** t_count)

            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = project_simplex(a, 1.0)

        score = 2.0 * n * fast_convolve(a).max()

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        it += 1

        if no_improve >= restart_every:
            rng_local = np.random.default_rng(it * 7 + 13 + n)
            noise = rng_local.standard_normal(n) * noise_scale
            a = a + noise
            a = np.maximum(a, 0)
            ss = a.sum()
            if ss > 1e-15:
                a = a / ss
            else:
                a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            noise_scale = max(0.001, noise_scale * 0.7)

        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [{name}] iter={it} ({elapsed:.0f}s): best={best_score:.8f}, cur={score:.8f}")

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.arange(old_n, dtype=np.float64)
    new_x = np.linspace(0, old_n - 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def peak_splitting(a, n_reduce=5, reduce_factor=0.95):
    """
    Peak splitting: reduce top contributors to main peak,
    redistribute to low-convolution positions.
    """
    n = len(a)
    b = fast_convolve(a)
    peak_k = int(np.argmax(b))

    js = max(0, peak_k - n + 1)
    je = min(n - 1, peak_k)
    idx = np.arange(js, je + 1)
    contribs = a[idx] * a[peak_k - idx]

    n_reduce_actual = min(n_reduce, len(idx))
    top_pos = idx[np.argsort(contribs)[::-1][:n_reduce_actual]]

    a_new = a.copy()
    total_removed = 0.0
    for pos in top_pos:
        if a_new[pos] > 1e-12:
            removed = a_new[pos] * (1.0 - reduce_factor)
            a_new[pos] -= removed
            total_removed += removed

    if total_removed < 1e-15:
        return a_new

    # Redistribute to positions with low total peak contribution
    top_peaks = np.argsort(b)[::-1][:5]
    contrib_to_peaks = np.zeros(n)
    for pk in top_peaks:
        pjs = max(0, pk - n + 1)
        pje = min(n - 1, pk)
        lo = pk - pje
        hi = pk - pjs
        contrib_to_peaks[pjs:pje + 1] += b[pk] * a[lo:hi + 1][::-1]

    weights = 1.0 / (contrib_to_peaks + 1e-10)
    weights = weights * a_new
    w_sum = weights.sum()
    if w_sum < 1e-15:
        weights = np.ones(n) / n
    else:
        weights = weights / w_sum

    a_new += total_removed * weights
    a_new = np.maximum(a_new, 0)
    a_new = a_new / a_new.sum()
    return a_new


def coordinate_descent_polish(a0, time_limit, name=""):
    """Coordinate descent: try reducing each top-contributing element."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_m = min(12, len(idx))
        top_pos = idx[np.argsort(contribs)[::-1][:top_m]]

        improved = False
        for pos in top_pos:
            if a[pos] < 1e-14:
                continue
            for factor in [0.9995, 0.999, 0.998, 0.995, 0.99, 0.98]:
                a_trial = a.copy()
                removed = a_trial[pos] * (1.0 - factor)
                a_trial[pos] -= removed
                if removed < 1e-16:
                    break
                others_sum = a_trial.sum()
                if others_sum < 1e-15:
                    break
                a_trial += removed * a_trial / others_sum
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()

                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        it += 1
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20000:
                break

        if it % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [{name}] CD iter={it} ({elapsed:.0f}s): score={best_score:.8f}")

    print(f"  [{name}] CoordDesc done: score={best_score:.8f}")
    return best_a, best_score

    print(f"\n=== Phase 1: n-screening (budget={t_p1:.0f}s) ===")

    candidate_ns = [n_parent]
    for dn in [-200, -100, -50, 50, 100, 200, 400]:
        n_try = n_parent + dn
        if 300 <= n_try <= 2500 and n_try != n_parent:
            candidate_ns.append(n_try)
    candidate_ns = sorted(set(candidate_ns))

    per_n = t_p1 / len(candidate_ns)
    print(f"  Trying n values: {candidate_ns}, {per_n:.0f}s each")

    for n_try in candidate_ns:
        if time.monotonic() > deadline - 60:
            break
        t_sub = min(per_n, deadline - time.monotonic() - 30)
        if t_sub < 8:
            continue

        if n_try == n_parent:
            a_try = parent.copy()
        else:
            a_try = interpolate_sequence(parent, n_try)

        print(f"  n={n_try}, budget={t_sub:.1f}s, init_score={evaluate_local(a_try):.8f}")
        a_res, score = adam_optimize(a_try, t_sub, lr=0.008, n_peaks=1,
                                      restart_every=1000, noise_scale=0.01, name=f"scr_n{n_try}")
        if score < best_score:
            best_score = score
            best_seq = a_res.copy()
            print(f"  NEW BEST: n={n_try}, score={score:.8f}")

    n_best = len(best_seq)
    print(f"\nAfter Phase 1: best n={n_best}, score={best_score:.8f}")

    # === Phase 2: Deep optimization at best n (600s) ===
    rem = deadline - time.monotonic()
    t_p2 = min(rem * 0.38, 600.0)
    print(f"\n=== Phase 2: Deep Adam (budget={t_p2:.0f}s) ===")

    if t_p2 > 30:
        # First: multi-peak subgradient
        t_sub1 = t_p2 * 0.4
        a_res, score = adam_optimize(best_seq, t_sub1, lr=0.004, n_peaks=3,
                                      peak_tol=0.999, restart_every=2000,
                                      noise_scale=0.005, name="deep_3pk")
        if score < best_score:
            best_score = score
            best_seq = a_res.copy()
            print(f"  NEW BEST (3-peak): score={score:.8f}")

        # Second: single peak, fine lr
        if time.monotonic() < deadline - 30:
            t_sub2 = min(t_p2 * 0.3, deadline - time.monotonic() - 30)
            if t_sub2 > 10:
                a_res, score = adam_optimize(best_seq, t_sub2, lr=0.002, n_peaks=1,
                                              peak_tol=0.9999, restart_every=3000,
                                              noise_scale=0.003, name="deep_1pk_fine")
                if score < best_score:
                    best_score = score
                    best_seq = a_res.copy()
                    print(f"  NEW BEST (1-peak fine): score={score:.8f}")

        # Third: smoothed objective (LogSumExp)
        if time.monotonic() < deadline - 30:
            t_sub3 = min(t_p2 * 0.3, deadline - time.monotonic() - 30)
            if t_sub3 > 10:
                a_res, score = adam_optimize(best_seq, t_sub3, lr=0.003, n_peaks=1,
                                              smooth_T=0.5, restart_every=1500,
                                              noise_scale=0.005, name="smooth_T0.5")
                if score < best_score:
                    best_score = score
                    best_seq = a_res.copy()
                    print(f"  NEW BEST (smooth): score={score:.8f}")

    print(f"\nAfter Phase 2: best n={len(best_seq)}, score={best_score:.8f}")

    # === Phase 3: Multi-start exploration (400s) ===
    rem = deadline - time.monotonic()
    t_p3 = min(rem * 0.28, 400.0)
    print(f"\n=== Phase 3: Multi-start (budget={t_p3:.0f}s) ===")

    if t_p3 > 30:
        n_starts = max(4, int(t_p3 / 50))
        per_start = t_p3 / n_starts
        print(f"  {n_starts} restarts, ~{per_start:.0f}s each")

        for si in range(n_starts):
            if time.monotonic() > deadline - 30:
                break

            pert_type = si % 5
            if pert_type == 0:
                # Gaussian noise
                noise = rng.standard_normal(n_best) * 0.008
                a_start = best_seq + noise
            elif pert_type == 1:
                # Peak splitting (aggressive)
                a_start = peak_splitting(best_seq, n_reduce=10, reduce_factor=0.88)
            elif pert_type == 2:
                # Peak splitting (mild) + noise
                a_start = peak_splitting(best_seq, n_reduce=4, reduce_factor=0.95)
                noise = rng.standard_normal(n_best) * 0.005
                a_start = a_start + noise
            elif pert_type == 3:
                # Interpolate to different n and back
                delta_n = rng.integers(-80, 80)
                n_alt = n_best + delta_n
                if n_alt < 200:
                    n_alt = n_best + 100
                a_interp = interpolate_sequence(best_seq, n_alt)
                a_start = interpolate_sequence(a_interp, n_best)
            else:
                # Random coordinate transfer
                a_start = best_seq.copy()
                n_transfers = rng.integers(3, 10)
                for _ in range(n_transfers):
                    i = rng.integers(0, n_best)
                    j = rng.integers(0, n_best)
                    if i != j and a_start[i] > 1e-8:
                        amount = a_start[i] * rng.uniform(0.1, 0.4)
                        a_start[i] -= amount
                        a_start[j] += amount

            a_start = np.maximum(a_start, 0)
            ss = a_start.sum()
            if ss > 1e-15:
                a_start = a_start / ss
            else:
                a_start = best_seq.copy()

            t_sub = min(per_start, deadline - time.monotonic() - 10)
            if t_sub < 5:
                continue

            a_res, score = adam_optimize(a_start, t_sub, lr=0.005, n_peaks=1,
                                          restart_every=500, noise_scale=0.01,
                                          name=f"rs_{si}")
            if score < best_score:
                best_score = score
                best_seq = a_res.copy()
                print(f"  NEW BEST (restart {si}, type={pert_type}): score={score:.8f}")

    print(f"\nAfter Phase 3: best n={len(best_seq)}, score={best_score:.8f}")

    # === Phase 4: Peak-split + optimize loop (250s) ===
    rem = deadline - time.monotonic()
    t_p4 = min(rem * 0.18, 250.0)
    print(f"\n=== Phase 4: Peak-split loop (budget={t_p4:.0f}s) ===")

    if t_p4 > 20:
        t0_p4 = time.monotonic()
        split_idx = 0
        while time.monotonic() - t0_p4 < t_p4 and time.monotonic() < deadline - 20:
            factors = [0.90, 0.93, 0.95, 0.97, 0.98]
            n_reduces = [8, 5, 4, 3, 2]
            fi = split_idx % len(factors)
            
            a_split = peak_splitting(best_seq, n_reduce=n_reduces[fi], reduce_factor=factors[fi])
            score_split = 2.0 * len(best_seq) * fast_convolve(a_split).max()
            
            t_sub = min(25.0, deadline - time.monotonic() - 10)
            if t_sub < 5:
                break

            a_res, score = adam_optimize(a_split, t_sub, lr=0.005, n_peaks=1,
                                          restart_every=400, noise_scale=0.005,
                                          name=f"split_{split_idx}")
            if score < best_score:
                best_score = score
                best_seq = a_res.copy()
                print(f"  NEW BEST (split {split_idx}, f={factors[fi]}): score={score:.8f}")
            
            split_idx += 1
            if split_idx >= 15:
                break

    # === Phase 5: Coordinate descent polish ===
    rem = deadline - time.monotonic()
    t_p5 = min(rem * 0.12, 100.0)
    print(f"\n=== Phase 5: Coord Descent (budget={t_p5:.0f}s) ===")

    if t_p5 > 8:
        a_cd, score = coordinate_descent_polish(best_seq, t_p5, name="CD")
        if score < best_score:
            best_score = score
            best_seq = a_cd.copy()
            print(f"  NEW BEST (CD): score={score:.8f}")

    # Final quick Adam pass to squeeze out last improvements
    rem = deadline - time.monotonic()
    if rem > 10:
        a_final, score = adam_optimize(best_seq, min(rem * 0.5, 50.0), lr=0.001,
                                        n_peaks=1, peak_tol=0.99999,
                                        restart_every=5000, noise_scale=0.002,
                                        name="final")
        if score < best_score:
            best_score = score
            best_seq = a_final.copy()

    # Final verification
    final_score = evaluate_local(best_seq)
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\nFINAL: n={len(best_seq)}, score={best_score:.8f}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
