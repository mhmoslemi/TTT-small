"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated Frank-Wolfe with LP oracle (main workhorse)
2. Multi-stage LogSumExp L-BFGS fine-tuning with aggressive annealing
3. Periodic restarts and random perturbations for escape
4. Multi-length exploration
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

# Global deadline
_deadline = 0.0

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def lse_obj_grad(z, T, n):
    """Compute LSE objective and its gradient."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = np.clip(Tb.max(), -500, 500)
    exp_Tb = np.exp(Tb - max_Tb)
    sum_exp = exp_Tb.sum()
    if sum_exp < 1e-300:
        obj = 2.0 * n * Tb.max()
        grad = np.zeros(n)
        return obj, grad
    w = exp_Tb / sum_exp
    log_sum_exp = max_Tb + np.log(sum_exp)
    obj = (2.0 * n / T) * log_sum_exp
    
    # Compute c_j = sum_m a[m] * w[m+j]
    idx = np.arange(n)
    indices = idx[:, None] + idx[None, :]  # (n,n) array, entry (j,m) = m+j
    c_matrix = w[indices]  # c_matrix[j,m] = w[m+j]
    c = c_matrix @ a  # c[j] = sum_m w[m+j] * a[m]
    
    d = np.dot(w, b)
    grad_z = 4.0 * n * a * (c - d)
    
    return obj, grad_z

def nesterov_lp(a0, time_limit, solver=None, max_iters=500000):
    """Nesterov-accelerated Frank-Wolfe with LP best response."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0
    
    # Pre-build the constraint matrix structure (it changes each iter since y changes)
    # But the sparsity pattern is fixed
    
    for it in range(max_iters):
        if time.monotonic() > _deadline:
            break

        # LP: minimize M s.t. conv(y, b)_k <= M, sum(b) = 1, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        
        # Build T matrix: T[k, j] = y[k-j] if 0 <= k-j < n
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = y
        
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            if solver == cp.PDLP:
                prob.solve(solver=cp.PDLP, time_limit=10)
            else:
                prob.solve(solver=solver, max_iters=10000)
        except Exception:
            prob.solve(solver=cp.SCS, max_iters=10000, eps=1e-6)
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_new = np.asarray(b_var.value, dtype=np.float64)
        b_new = np.maximum(b_new, 0)
        s_b = b_new.sum()
        if s_b < 1e-15:
            continue
        b_new = b_new / s_b

        # Nesterov update
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        # Restart every 5000 iterations
        if (it + 1) % 5000 == 0:
            y = a.copy()
        
        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 20:
                print(f"      NW iter={it+1} ({elapsed:.0f}s): score={best_score:.10f}")
                last_print = elapsed

    return best_a, best_score

def lse_lbgfs(a0, time_limit, T_start=300.0, T_end=0.01, n_stages=30):
    """Multi-stage LSE L-BFGS with geometric temperature schedule."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() > _deadline:
            break
        remaining = _deadline - time.monotonic()
        if remaining < 3:
            break
        
        def func(z_val):
            obj, _ = lse_obj_grad(z_val, stage_T, n)
            return obj
        
        def jac(z_val):
            _, grad = lse_obj_grad(z_val, stage_T, n)
            return grad
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': 100, 'ftol': 1e-25,
                         'gtol': 1e-14, 'maxfun': 500,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass
        
        a_check = softmax_from_z(z)
        sc = score_normalized(a_check)
        if (si + 1) % 10 == 0:
            print(f"      LSE stage={si+1}/{n_stages} T={stage_T:.3f}: score={sc:.10f}")
    
    return softmax_from_z(z)

def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x ** 2)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 4:
        a = np.ones(n) * (1.0 + 0.2 * np.cos(2 * np.pi * 3 * np.arange(n) / n))
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.15 * np.cos(7 * x)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 4
    else:
        x = np.linspace(0, 1, n)
        a = 1.0 / (0.5 + np.exp(10 * (x - 0.5)))  # sigmoid-like
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def get_lp_solver():
    for solver_name in [cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated Frank-Wolfe with LP oracle (main workhorse)
2. Multi-stage LogSumExp L-BFGS with aggressive geometric annealing (key improvement)
3. Random perturbations with LSE re-optimization for escape from local minima
4. Multi-length exploration via interpolation
Key: much finer LSE temperature schedule (30+ stages) with more iterations at low T.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

_deadline = 0.0

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def lse_obj_grad(z, T, n):
    """Compute LSE objective and gradient. F_T = (2n/T)*log(sum exp(T*b_k))."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = np.clip(Tb.max(), -500.0, 500.0)
    exp_Tb = np.exp(Tb - max_Tb)
    sum_exp = exp_Tb.sum()
    if sum_exp < 1e-300:
        return 2.0 * n * b.max(), np.zeros(n)
    w = exp_Tb / sum_exp
    log_sum_exp = max_Tb + np.log(sum_exp)
    obj = (2.0 * n / T) * log_sum_exp

    # c_j = sum_m a[m] * w[m+j], for j=0..n-1
    # w has length 2n-1
    idx = np.arange(n)
    col_idx = idx[None, :]  # (1, n): m values
    row_idx = idx[:, None]   # (n, 1): j values
    indices = row_idx + col_idx  # (n, n): entry (j,m) = m+j
    c_vals = w[indices]  # (n, n): c_vals[j, m] = w[m+j]
    c = c_vals @ a  # c[j] = sum_m w[m+j] * a[m]

    d = np.dot(w, b)
    grad_z = 4.0 * n * a * (c - d)
    return obj, grad_z

def nesterov_lp(a0, time_limit, solver=None, max_iters=500000, restart_every=5000):
    """Nesterov-accelerated FW with LP best response."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0

    for it in range(max_iters):
        if time.monotonic() > _deadline:
            break

        # LP oracle: min M s.t. conv(y,b)_k <= M, sum(b)=1, b>=0
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        T_mat = np.zeros((2 * n - 1, n))
        for j in range(n):
            T_mat[j:j + n, j] = y
        constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            if solver is not None:
                prob.solve(solver=solver, time_limit=15)
            else:
                prob.solve(solver=cp.SCS, max_iters=20000, eps=1e-7)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=20000, eps=1e-7)
            except Exception:
                continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_new = np.asarray(b_var.value, dtype=np.float64)
        b_new = np.maximum(b_new, 0)
        s_b = b_new.sum()
        if s_b < 1e-15:
            continue
        b_new = b_new / s_b

        # Nesterov update
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if (it + 1) % restart_every == 0:
            y = a.copy()

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 25:
                print(f"      NW iter={it+1} ({elapsed:.0f}s): score={best_score:.10f}")
                last_print = elapsed

    return best_a, best_score

def lse_lbgfs(a0, time_limit, T_start=300.0, T_end=0.01, n_stages=30, iters_per_stage=80):
    """Multi-stage LSE L-BFGS with geometric temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    best_z = z.copy()
    best_sc = score_normalized(softmax_from_z(z))

    for si, stage_T in enumerate(temps):
        if time.monotonic() > _deadline:
            break
        remaining = _deadline - time.monotonic()
        if remaining < 3:
            break

        # Determine iterations for this stage (more at low T)
        frac = si / max(1, n_stages - 1)
        n_iters = int(iters_per_stage * (1.0 + 2.0 * frac))  # up to 3x at low T

        def func_staged(z_val):
            obj, _ = lse_obj_grad(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_obj_grad(z_val, stage_T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': n_iters, 'ftol': 1e-30,
                         'gtol': 1e-16, 'maxfun': n_iters * 300,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

        a_check = softmax_from_z(z)
        sc = score_normalized(a_check)
        if sc < best_sc:
            best_sc = sc
            best_z = z.copy()

        if (si + 1) % 5 == 0:
            print(f"      LSE stage={si+1}/{n_stages} T={stage_T:.4f}: score={sc:.10f}")

    return softmax_from_z(best_z)

def structured_init(n, rng):

    elif choice == 1:

        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x ** 2)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 4:
        a = np.ones(n) * (1.0 + 0.2 * np.cos(2 * np.pi * 3 * np.arange(n) / n))
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.15 * np.cos(7 * x)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 4
    else:
        x = np.linspace(0, 1, n)
        a = 1.0 / (0.5 + np.exp(10 * (x - 0.5)))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(
