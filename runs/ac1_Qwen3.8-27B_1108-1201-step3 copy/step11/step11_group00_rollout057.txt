    Solve: max sum(b) s.t. (A@b)_k <= M for all k, b >= 0.
    A is the convolution matrix: A[k,j] = a[k-j] if 0<=k-j<=n-1 else 0.
    """
    n = len(a)
    # A is (2n-1) x n
    # c = -1 (we minimize -sum(b) to maximize sum(b))
    # A_ub = A, b_ub = M (for each of 2n-1 constraints)
    # bounds: (0, None) for each b_j
    
    # Build A efficiently (it's banded)
    # A[k,j] = a[k-j] if 0 <= k-j <= n-1 else 0
    # For row k: A[k, max(0,k-n+1):min(n,k+1)] = a[k - max(0,k-n+1) : k - max(0,k-n+1) + (min(n,k+1)-max(0,k-n+1))]
    
    # For efficiency, use sparse matrix
    from scipy.sparse import coo_matrix
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        j_start = max(0, k-n+1)
        j_end = min(n-1, k)
        for j in range(j_start, j_end+1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k-j])
    
    A_sparse = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    c = -np.ones(n)  # minimize -sum(b)
    res = linprog(c, A_ub=A_sparse, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.status == 0:
        b = res.x
        b = np.maximum(b, 0)
        s = b.sum()
        b = b / s
        return b
    return None

def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


class Convolver:
    """Precomputes convolution matrix structure for fast best-response LP."""
    def __init__(self, n):
        self.n = n
        # Precompute sparsity pattern
        rows = []
        cols = []
        for k in range(2*n-1):
            j_start = max(0, k-n+1)
            j_end = min(n-1, k)
            for j in range(j_start, j_end+1):
                rows.append(k)
                cols.append(j)
        self.rows = np.array(rows)
        self.cols = np.array(cols)
        self.nnz = len(rows)
    
    def get_matrix(self, a):
        """Build sparse matrix A where A[k,j] = a[k-j]."""
        vals = a[self.rows - self.cols]  # a[k-j] for each (k,j) pair
        return coo_matrix((vals, (self.rows, self.cols)), shape=(2*self.n-1, self.n))


def lse_obj_grad_z(z, T):
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T
    w = exp_Tb / exp_Tb.sum()
    
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad_a = 2.0 * conv[n - 1:n - 1 + n]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def nesterov_lp_optimize(a0, time_limit, convolver, max_iters=50000):
    """
    Nesterov-accelerated LP best-response for minimizing max_k (a*a)_k
    subject to sum(a)=1, a>=0.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    avg_a = np.zeros(n)
    n_avg = 0
    lp_times = []
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP: max sum(b) s.t. (y*b)_k <= M for all k, b >= 0
        M_val = fast_autocorr(y).max()
        
        # Build constraint matrix
        A_sparse = convolver.get_matrix(y)
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M_val)
        
        try:
            res = linprog(c, A_ub=A_sparse, b_ub=b_ub, 
                         bounds=[(0, None)]*n, method='highs',
                         options={'time_limit': 5, 'presolve': True})
        except Exception:
            break
        
        if res.status != 0:
            break
        
        b_star = np.maximum(res.x, 0)
        s_b = b_star.sum()
        if s_b < 1e-15:
            b_star = a.copy()
        else:
            b_star = b_star / s_b
        
        # Nesterov step
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        
        # Nesterov extrapolation
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a
        
        y = y_new
        a = a_new
        
        # Check score
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        # Running average
        avg_a += a
        n_avg += 1
        if n_avg >= 10 and n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
        
        if t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score
