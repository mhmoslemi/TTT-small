def fast_convolve(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def lse_obj_grad(z, T, n):

def solve_lp_best_response(a_fixed, solver=None):
    """Solve LP: min M s.t. conv_matrix(a_fixed) @ b <= M, b >= 0, sum(b) = 1."""

    M_var = cp.Variable()

    # This is T @ b <= M where T is the convolution matrix
    # Instead of building the full matrix, use constraints efficiently
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers = [solver] if solver else [cp.PDLP, cp.ECOS, cp.SCS]
    if solver is None:
        solvers = [cp.PDLP, cp.ECOS, cp.SCS]
    
    for s in solvers:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s_val = b_val.sum()
    if s_val < 1e-15:

    b_val = b_val / s_val

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-stage LSE L-BFGS with fine temperature schedule (main polishing)
2. Nesterov-accelerated LP best-response for larger structural moves
3. Basin hopping (random perturb + re-optimize) to escape local minima
4. Length exploration around the parent length
5. Final ultra-fine LSE polish

Key improvements over previous:
- Finer LSE temperature schedule (30+ stages)
- Basin hopping with adaptive perturbation scale
- Better time allocation based on observed improvement
- Multiple independent LSE passes at different starting points

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Compute full auto-convolution of a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely evaluate the sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_from_z(z):
    """Convert z-vector to probability simplex via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert simplex vector to log-space."""
    return np.log(np.maximum(a, 1e-300))


def lse_obj_grad(z, T, n):
    """Compute LSE-smoothed objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp

    # grad_a[j] = sum_k 2*w[k]*a[k-j] for valid indices
    size = len(b)
    grad_a = np.zeros(n)
    for k in range(size):
        wk = w[k]
        if wk < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * wk * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_lbfgs(a0, time_limit, T_start=50.0, T_end=0.05, n_stages=25, iters_per_stage=40):
    """Multi-stage LSE optimization with L-BFGS-B."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func(z_val):
            obj, _ = lse_obj_grad(z_val, stage_T, n)
            return obj

        def jac(z_val):
            _, grad = lse_obj_grad(z_val, stage_T, n)
            return grad

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-22,
                         'gtol': 1e-18, 'maxfun': iters_per_stage * 500,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)
            if (si + 1) % 5 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.4f}: score={sc:.10f} ({elapsed:.0f}s)")
        except Exception as e:
            print(f"    LSE stage {si+1} failed: {e}")

    return softmax_from_z(z)


def solve_lp_best_response(a_fixed, solver=None):
    """Solve LP: min M s.t. conv_matrix(a_fixed)@b <= M, b>=0, sum(b)=1."""
    import cvxpy as cp
    n = len(a_fixed)
    size = 2 * n - 1

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers = [solver] if solver else [cp.PDLP, cp.ECOS, cp.SCS]
    for s in solvers:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s_val = b_val.sum()
    if s_val < 1e-15:
        return None, None
    b_val = b_val / s_val
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver=None, max_iters=200000):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_improve = t0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            last_improve = time.monotonic()

        if (it + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f} M={M_lp:.6f}")

        # Stall detection: if no improvement for > 120s, break
        if time.monotonic() - last_improve > 120 and it > 1000:
            break

    return best_a, best_score


def basin_hopping(a0, time_limit, rng, max_hops=10):
    """Basin hopping: random perturb + LSE re-optimize, keep best."""
    n = len(a0)
    best_a = a0.copy()
    best_score = score_normalized(a0)
    t0 = time.monotonic()

    for i in range(max_hops):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 30:
            break

        # Adaptive perturbation scale: larger early, smaller late
        scale = 0.15 * (1.0 - 0.5 * i / max_hops)
        z = z_from_a(a0) + scale * rng.standard_normal(n)
        z = z - z.mean()  # center to avoid drift
        a_perturbed = softmax_from_z(z)

        # Quick LSE optimization on perturbed point
        time_per = min(remaining * 0.6 / (max_hops - i), 120.0)
        a_opt = lse_lbfgs(a_perturbed, time_per, T_start=30.0, T_end=0.1,
                          n_stages=8, iters_per_stage=25)

        sc = score_normalized(a_opt)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"    Hop {i}: new best {best_score:.10f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    for s_name in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s_name
        except Exception:
            continue
    return cp.ECOS

            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)

    except Exception:

        best_seq = np.sin(np.pi * x) ** 2

        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.10f}")

        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic()
            print(f"  *** NEW BEST [{label}]: {sc:.10f} (n={len(a)}) ***")

    # === Phase 1: LSE L-BFGS fine polish on parent (35% budget) ===
    t1 = 0.35 * total_budget
    remaining = deadline - time.monotonic()
    t1 = min(t1, remaining * 0.9)
    print(f"\n=== Phase 1: LSE L-BFGS polish (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out = lse_lbfgs(best_seq, t1, T_start=80.0, T_end=0.03,
                          n_stages=30, iters_per_stage=35)

    print(f"  Phase 1 done: best={best_score:.10f}")

    t2 = 0.25 * total_budget
    remaining = deadline - time.monotonic()
    t2 = min(t2, remaining * 0.8)
    print(f"\n=== Phase 2: Nesterov LP (budget={t2:.0f}s) ===")

    # === Phase 3: LSE polish on Phase 2 result (15% budget) ===
    t3 = 0.15 * total_budget
    remaining = deadline - time.monotonic()
    t3 = min(t3, remaining * 0.7)
    print(f"\n=== Phase 3: LSE polish round 2 (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out = lse_lbfgs(best_seq, t3, T_start=30.0, T_end=0.02,
                          n_stages=20, iters_per_stage=35)
        try_update(a_out, "lse_phase3")
    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Basin hopping (15% budget) ===
    t4 = 0.15 * total_budget
    remaining = deadline - time.monotonic()

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    print(f"{'='*60}")
    return [float(x) for x in best_seq]
