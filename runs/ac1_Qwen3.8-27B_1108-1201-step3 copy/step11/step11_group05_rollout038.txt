```python
"""
IceShard Optimization with MRF Directional Refinement.

Strategy:
1. IceShard initialization: structured initial sequences with sharp features
2. Nesterov-accelerated log-sum-exp optimization with adaptive temperature
3. LP alternating with Nesterov dual averaging (warm-started)
4. MRF (Maximum Residual Function) Directional Refinement: active-set focused
   subgradient with backtracking line search
5. Multi-length exploration

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
import math


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_grad_lse(a, p):
    """Compute gradient of LSE(conv(a,a)) w.r.t. a using FFT.
    Returns grad[i] = 2 * sum_k p[k] * a[k-i]
    """
    n = len(a)
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    W = rfft(p, fft_size)
    A_rev = rfft(a[::-1], fft_size)
    conv_result = irfft(W * A_rev, fft_size)[:3 * n - 2]
    return 2.0 * conv_result[n - 1:2 * n - 1]


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(a_fixed, solver):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    for s in [solver, cp.PDLP, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                break
        except Exception:
            continue
    else:
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    b_val[np.isnan(b_val)] = 0
    sv = b_val.sum()
    if sv < 1e-15:
        return None, None
    b_val = b_val / sv
    return b_val, float(max(0.0, M_var.value))


def nesterov_lse_optimize(a0, time_limit, T_stages, lr_init, lr_decay):
    """Nesterov-accelerated LSE optimization in log-space."""
    n = len(a0)
    z = z_from_a(a0).copy()
    z_prev = z.copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    
    t_nest = 1.0
    total_stages = len(T_stages)
    
    for stage_idx, T_stage in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        steps_per_stage = max(2000, int(time_limit * 30000 / total_stages))
        
        for step in range(steps_per_stage):
            if time.monotonic() > deadline:
                break
            
            # Nesterov lookahead
            beta = (t_nest - 1.0) / (t_nest + 1.0)
            y = z + beta * (z - z_prev)
            t_nest = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * t_nest * t_nest))
            z_prev = z
            
            a = softmax_from_z(y)
            b = fast_convolve(a)
            
            # Compute LSE and gradient
            max_b = b.max()
            exp_b = np.exp((b - max_b) / T_stage)
            p = exp_b / exp_b.sum()
            lse_val = max_b + T_stage * np.log(exp_b.sum())
            
            score = 2.0 * n * lse_val
            if 2.0 * n * max_b < best_score:
                best_score = 2.0 * n * max_b
                best_a = a.copy()
            
            # Gradient in a-space
            grad_a = (4.0 * n / T_stage) * compute_grad_lse(a, p)
            
            # Gradient in z-space
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            
            # Step
            lr = lr_init * (lr_decay ** stage_idx)
            z = y - lr * grad_z
            z -= z.mean()  # Centering
    
    # Final evaluation
    a_final = softmax_from_z(z)
    score_final = 2.0 * n * fast_convolve(a_final).max()
    if score_final < best_score:
        best_score = score_final
        best_a = a_final.copy()
    
    return best_a, best_score


def mrf_directional_refine(a0, time_limit, n_steps=50000):
    """MRF Directional Refinement: active-set subgradient with backtracking."""
    n = len(a0)

def ice_shard_init(n, rng):
    """Create an 'ice shard' structured initialization."""
    x = np.linspace(0, 1, n)
    choice = rng.integers(0, 6)
    
    if choice == 0:
        # Sharp central peak with oscillating sides
        w = rng.uniform(0.1, 0.3)
        a = np.maximum(0, 1.0 - np.abs(x - 0.5) / w) ** rng.uniform(1, 3)
    elif choice == 1:
        # Raised cosine
        f = rng.integers(1, 4)
        a = 0.5 + 0.4 * np.cos(2.0 * np.pi * f * (x - 0.5))
        a = np.maximum(a, 0.01)
    elif choice == 2:
        # Gaussian
        sigma = rng.uniform(0.1, 0.3)
        a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    elif choice == 3:
        # Double peak
        sigma = rng.uniform(0.05, 0.15)
        a = np.exp(-0.5 * ((x - 0.3) / sigma) ** 2) + np.exp(-0.5 * ((x - 0.7) / sigma) ** 2)
    elif choice == 4:
        # Asymmetric shard
        a = np.where(x < 0.5, 
                     np.maximum(0, 1 - (0.5 - x) / rng.uniform(0.1, 0.3)),
                     np.maximum(0, 1 - (x - 0.5) / rng.uniform(0.2, 0.4)))
    else:
        # Polyphase-like
        f1, f2 = rng.integers(2, 5), rng.integers(5, 10)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * x) + 0.2 * np.cos(2 * np.pi * f2 * x + 0.5)
        a = np.maximum(a, 0.01)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
```

```python
def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_grad_lse(a, p):
    """Compute grad_a[i] = 2 * sum_k p[k] * a[k-i] using FFT.
    p has length 2n-1, a has length n. Returns array of length n.
    """
    n = len(a)
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    W = rfft(p, fft_size)
    A_rev = rfft(a[::-1], fft_size)
    conv_result = irfft(W * A_rev, fft_size)[:3 * n - 2]
    return 2.0 * conv_result[n - 1:2 * n - 1]


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""

    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    for s in [solver, cp.PDLP, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                break
        except Exception:
            continue
    else:
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    b_val[np.isnan(b_val)] = 0
    sv = b_val.sum()
    if sv < 1e-15:
        return None, None
    b_val = b_val / sv
    return b_val, float(max(0.0, M_var.value))


def adam_nesterov_lse(a0, time_limit, T_stages, lr_init, lr_decay, top_k_frac=0.2):
    """Adam with Nesterov lookahead on LSE objective in log-space."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit

    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step_count = 0

    total_stages = len(T_stages)
    steps_per_stage = max(3000, int(time_limit * 40000 / total_stages))

    for stage_idx, T_stage in enumerate(T_stages):
        if time.monotonic() > deadline:
            break

        top_k = max(10, int(top_k_frac * (2 * n - 1)))
        lr = lr_init * (lr_decay ** stage_idx)

        for step in range(steps_per_stage):
            if time.monotonic() > deadline:
                break

            # Nesterov lookahead
            z_nest = z + beta1 * m
            a = softmax_from_z(z_nest)
            b = fast_convolve(a)
            max_b = b.max()
            score = 2.0 * n * max_b
            step_count += 1

            if score < best_score:
                best_score = score
                best_a = a.copy()

            # Compute LSE gradient
            if top_k < len(b):
                idx = np.argpartition(b, -top_k)[-top_k:]
                b_top = b[idx]
                max_top = b_top.max()
                exp_top = np.exp((b_top - max_top) / T_stage)
                w_top = exp_top / exp_top.sum()
                p_full = np.zeros(len(b))
                p_full[idx] = w_top
            else:
                exp_all = np.exp((b - max_b) / T_stage)
                p_full = exp_all / exp_all.sum()

            # Gradient in a-space: (4n/T) * compute_grad_lse(a, p_full)
            grad_a = (4.0 * n / T_stage) * compute_grad_lse(a, p_full)

            # Gradient in z-space
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * grad_z ** 2
            m_hat = m / (1.0 - beta1 ** (step + 1))
            v_hat = v / (1.0 - beta2 ** (step + 1))
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            z -= z.mean()

    # Final evaluation
    a_final = softmax_from_z(z)
    score_final = 2.0 * n * fast_convolve(a_final).max()
    if score_final < best_score:
        best_score = score_final
        best_a = a_final.copy()

    return best_a, best_score


def mrf_directional_refine(a0, time_limit, n_steps=100000):
    """MRF Directional Refinement: active-set subgradient with backtracking."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.03
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Active set: top peaks with soft weighting
        k_active = max(5, min(100, int(0.03 * (2 * n - 1))))
        idx = np.argpartition(b, -k_active)[-k_active:]
        b_active = b[idx]
        max_active = b_active.max()

        # Soft weights (closer to max get more weight)
        T_w = max(0.005 * max_active, 1e-10)
        w = np.exp((b_active - max_active) / T_w)
        w = w / w.sum()

        # Compute subgradient direction
        p_full = np.zeros(2 * n - 1)
        p_full[idx] = w
        grad = compute_grad_lse(a, p_full)

        # Project onto tangent space of simplex
        S = np.dot(grad, a)
        grad_tangent = grad - S

        # Backtracking line search
        target = 2.0 * n * max_b
        step_taken = False
        for trial in range(15):
            a_new = a + eta * grad_tangent
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
            score_new = 2.0 * n * fast_convolve(a_new).max()
            if score_new < target - 1e-14:
                a = a_new
                step_taken = True
                break
            eta *= 0.5

        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 40:
            eta *= 0.7
            no_improve = 0
            if eta < 1e-7:
                break

        # Occasionally try increasing eta if stuck for very long
        if no_improve == 20:
            eta = max(eta * 1.5, 0.01)

    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver):
    """Nesterov dual averaging LP alternating with adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    prev_best = best_score
    stall_iters = 0
    lp_count = 0

    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        lp_count += 1

        # Nesterov dual averaging
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Stall detection
        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-10:
                stall_iters = 0
            else:
                stall_iters += 1
            prev_best = best_score

            if stall_iters >= 15:
                # Perturb y to escape
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 8)
                noise = 0.03 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y_new = y + noise
                y_new = np.maximum(y_new, 0)
                y_new = y_new / y_new.sum()
                y = y_new
                stall_iters = 0

            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def ice_shard_init(n, rng):
    """Create an 'ice shard' structured initialization with various patterns."""
    x = np.linspace(0, 1, n)
    choice = rng.integers(0, 8)

    if choice == 0:
        # Sharp central peak
        w = rng.uniform(0.15, 0.35)
        p = rng.uniform(1.0, 3.0)
        a = np.maximum(0, 1.0 - np.abs(x - 0.5) / w) ** p
    elif choice == 1:
        # Raised cosine with random frequency
        f = rng.integers(2, 6)
        a = 0.5 + 0.4 * np.cos(2.0 * np.pi * f * (x - 0.5))
        a = np.maximum(a, 0.01)
    elif choice == 2:
        # Gaussian
        sigma = rng.uniform(0.1, 0.25)
        a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    elif choice == 3:
        # Double peak
        sigma = rng.uniform(0.05, 0.15)
        a = np.exp(-0.5 * ((x - 0.3) / sigma) ** 2) + np.exp(-0.5 * ((x - 0.7) / sigma) ** 2)
    elif choice == 4:
        # Asymmetric shard
        w1 = rng.uniform(0.1, 0.25)
        w2 = rng.uniform(0.2, 0.4)
        a = np.where(x < 0.5,
                     np.maximum(0, 1 - (0.5 - x) / w1),
                     np.maximum(0, 1 - (x - 0.5) / w2))
    elif choice == 5:
        # Polyphase-like
        f1, f2 = rng.integers(2, 5), rng.integers(5, 12)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * x) + 0.15 * np.cos(2 * np.pi * f2 * x + 0.7)
        a = np.maximum(a, 0.01)
    elif choice == 6:
        # Exponential decay from center
        a = np.exp(-3.0 * np.abs(x - 0.5))
    else:
        # Uniform (baseline)
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturbed_sequence(a, rng, strength=0.02):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 10)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def local_update(best_seq, best_score, a, label):
    """Check if a is better than current best, update if so."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = evaluate_local(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}) ***")
    return best_seq, best_score
```
