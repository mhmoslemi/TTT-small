```python
import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def mat_vec_My_a(y, a):
    """Compute M(y) @ a where M(y)_{j,l} = y_{j+l}."""
    # This is the cross-correlation of y and a
    return np.correlate(y, a, mode='valid')

def proj_simplex(v, cap=1.0):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def softmax_temperature(x, T):
    """Compute softmax(x/T) stably."""
    x_shift = (x - x.max()) / T
    exp_x = np.exp(x_shift)
    s = exp_x.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return exp_x / s

def logsumexp(x, T):
    """Compute (1/T) * log(sum(exp(T*x)))."""
    m = x.max()
    return m + np.log(np.exp(T * (x - m)).sum()) / T

def mirror_descent_ls(a_init, T, n_iters, eta=None, deadline=None, verbose=False):
    """
    Mirror descent (multiplicative weights) on the log-sum-exp objective.
    Minimizes (1/T) log sum_k exp(T * (a*a)_k) subject to a in simplex.
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    if eta is None:
        eta = 1.0 / np.sqrt(n)
    
    no_improve = 0
    scale = 1.0
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        b = fast_convolve(a)
        p = softmax_temperature(b, T)  # p_k = exp(T*b_k) / sum(exp(T*b_l))
        
        # Gradient: 2 * M(p) @ a (where M(p)_{j,l} = p_{j+l})
        grad = 2.0 * mat_vec_My_a(p, a)
        
        # Mirror descent update (multiplicative weights)
        # a_new_j = a_j * exp(-eta * grad_j) / sum
        eta_eff = eta * scale
        log_update = -eta_eff * grad
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 100:
            scale *= 0.8
            no_improve = 0
            if scale < 0.01:
                break
        
        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MD-LS iter={it}, T={T:.4f}, score={best_score:.8f}, scale={scale:.4f}, time={elapsed:.1f}s")
    
    return best_a, best_score

def primal_dual_mirror(a_init, y_init, n_iters, eta_a=None, eta_y=None, deadline=None, verbose=False):
    """
    Primal-dual mirror descent for min_a max_y a^T M(y) a.
    """

        a = a / s

    if sy < 1e-15:
        y = np.ones(2*n-1) / (2*n-1)
    else:
        y = np.maximum(y, 0)
        y = y / y.sum()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    if eta_a is None:
        eta_a = 1.0 / np.sqrt(n)
    if eta_y is None:

    scale_a = 1.0
    scale_y = 1.0

def spectral_flatten(a, n_iters=100, blend=0.4, deadline=None):
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)
        
        mags = np.abs(F)
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        
        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])
        
        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new
    
    return a

def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage, deadline=None):
    n = len(a0)
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit
    n_stages = len(T_stages)
    
    # Work in log domain for better conditioning
    z = np.log(np.maximum(a0, 1e-300))
    
    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 2:
            break
        
        iters = max(20, maxiter_per_stage // n_stages * 2)
        
        def func(z_in):
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            return 2.0 * n * logsumexp(b, stage_T)
        
        def jac(z_in):
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            p = softmax_temperature(b, stage_T)
            grad_a = 2.0 * mat_vec_My_a(p, a)
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            return 2.0 * n * grad_z
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20, 'gtol': 1e-15, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    a_final = softmax_from_z(z)
    return a_final

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def peak_shave(a, time_limit, n_steps=100000):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return None, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    eta = 0.03
    scale = 1.0
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        threshold = max_b * (1.0 - 2e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Weight by proximity to max
        weights = (b[active] - threshold) / max(1e-30, max_b - threshold)
        if weights.sum() < 1e-15:
            weights = np.ones(len(active))
        weights = weights / weights.sum()
        
        grad_a = np.zeros(n)
        for i, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += weights[i] * a[k - idx]
        
        eta_eff = eta * scale
        log_update = -eta_eff * grad_a
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 50:
            scale *= 0.7
            no_improve = 0
            if scale < 0.005:
                break
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
