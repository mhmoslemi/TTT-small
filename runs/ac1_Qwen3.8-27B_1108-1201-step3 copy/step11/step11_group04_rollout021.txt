"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_autoconvolve(a):
    """Compute linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_local(a):
    """Compute score for a normalized (sum=1) non-negative sequence."""
    n = len(a)
    b = fast_autoconvolve(a)
    return 2.0 * n * b.max()


def normalize(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def init_geometric(n, r, rng=None):
    """Geometric sequence a[i] = r^i, normalized."""
    if r <= 1:
        r = 1.0 + 1e-6
    a = r ** np.arange(n, dtype=np.float64)
    # Clip to avoid overflow
    max_val = a.max()
    if max_val > 1e10:
        a = a / max_val
    return normalize(a)


def init_powerlaw(n, alpha, rng=None):
    """Power-law sequence a[i] = (i+1)^alpha, normalized."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** alpha
    return normalize(a)


def init_random(n, rng):
    """Random non-negative sequence."""
    return normalize(rng.random(n))


def init_bimodal(n, rng):
    """Sequence with mass concentrated at two locations."""
    a = np.zeros(n)
    c1 = rng.integers(0, n // 3)
    c2 = rng.integers(2 * n // 3, n)
    w1 = rng.random() * n // 5 + n // 20
    w2 = rng.random() * n // 5 + n // 20
    a[c1:c1+w1] = 1.0
    a[c2:c2+w2] = 1.0
    return normalize(a)


def peak_shave_iter(a, n_iters=10000, eta0=0.05, decay=0.999, top_k=5, deadline=None):
    """
    Iterative peak shaving: reduce the elements that contribute most to the max of autoconvolution.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = score_local(a)
    eta = eta0
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = fast_autoconvolve(a)
        max_c = c.max()
        
        # Find top-k peaks
        if top_k >= len(c):
            top_indices = np.arange(len(c))
            weights = np.ones(len(top_indices))
        else:
            top_indices = np.argsort(c)[-top_k:]
            top_vals = c[top_indices]
            # Weight by how close to max
            weights = np.maximum(top_vals / max_c - 0.9, 0.1)
            weights = weights / weights.sum()
        
        # Compute subgradient: g[j] = weighted sum over peaks of a[k-j]
        g = np.zeros(n)
        for i, k in enumerate(top_indices):
            w = weights[i]
            # c[k] = sum_j a[j] * a[k-j]
            # dc[k]/da[j] = a[k-j] (if 0 <= k-j < n)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            js = np.arange(j_start, j_end + 1)
            g[js] += w * a[k - js]
        
        # Multiplicative update
        scale = -eta * g
        scale -= scale.max()
        a_new = a * np.exp(scale)
        a_new = normalize(a_new)
        
        s_new = score_local(a_new)
        if s_new < best_score:
            best_score = s_new
            best_a = a_new.copy()
            a = a_new
        
        eta *= decay
    
    return best_a, best_score


def logsumexp_score_grad(a, T):
    """
    Compute LogSumExp-smoothed score and its gradient w.r.t. a.
    Score = 2n * LSE(c) / ... but since sum(a)=1, we work with 2n*LSE(c).
    Actually we want to minimize max c, so use LSE as smooth approximation.
    """
    n = len(a)
    c = fast_autoconvolve(a)
    max_c = c.max()
    # LSE approximation of max
    exp_c = np.exp((c - max_c) / T)
    lse = max_c + T * np.log(exp_c.sum())
    
    # Gradient of lse w.r.t. a
    # d(lse)/da[j] = (1/T) * sum_k exp_c[k] * dc[k]/da[j]
    # dc[k]/da[j] = a[k-j] if valid
    w = exp_c / exp_c.sum()  # This is the softmax weight
    # Actually: d(max_c + T*log(sum exp((c-max_c)/T)))/da[j]
    # = sum_k (exp((c[k]-max_c)/T) / sum(exp((c[k']-max_c)/T'))) * dc[k]/da[j]
    # = sum_k w[k] * a[k-j] (for valid k-j)
    
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-15:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1)
        g[js] += w[k] * a[k - js]
    
    score = 2.0 * n * lse
    grad = 2.0 * n * g
    return score, grad


class LBFGSWrapper:
    """Wrapper for L-BFGS-B on log-domain variables."""
    def __init__(self, T, deadline):
        self.T = T
        self.deadline = deadline
    
    def obj(self, z):
        if time.monotonic() > self.deadline:
            return 1e15
        a = self._z_to_a(z)
        c = fast_autoconvolve(a)
        max_c = c.max()
        exp_c = np.exp((c - max_c) / self.T)
        lse = max_c + self.T * np.log(exp_c.sum())
        return 2.0 * len(a) * lse
    
    def grad(self, z):
        if time.monotonic() > self.deadline:
            return np.zeros_like(z)
        a = self._z_to_a(z)
        c = fast_autoconvolve(a)
        max_c = c.max()
        exp_c = np.exp((c - max_c) / self.T)
        w = exp_c / exp_c.sum()
        
        g_a = np.zeros(len(a))
        for k in range(len(c)):
            if w[k] < 1e-15:
                continue
            j_start = max(0, k - len(a) + 1)
            j_end = min(len(a) - 1, k)
            js = np.arange(j_start, j_end + 1)
            g_a[js] += w[k] * a[k - js]
        
        # Gradient w.r.t. z: a_g = a * (g_a - sum(g_a * a))
        S = np.dot(g_a, a)
        g_z = a * (g_a - S)
        return 2.0 * len(a) * g_z
    
    def _z_to_a(self, z):
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        s = exp_z.sum()
        if s < 1e-300:
            return np.ones(len(z)) / len(z)
        return exp_z / s


def lbgfs_refine(a0, T_stages, maxiter_per_stage, deadline):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    
    for T in T_stages:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 2:
            break
        
        wrapper = LBFGSWrapper(T, deadline)
        iters = max(30, min(maxiter_per_stage, int(remaining * 100)))
        
        try:
            res = minimize(
                fun=wrapper.obj, x0=z, jac=wrapper.grad, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxfun': iters * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    a_final = LBFGSWrapper(1.0, deadline)._z_to_a(z)
    return a_final


def spectral_flatten_autoconv(a, n_iters=100, blend=0.3, deadline=None):
    """
    Spectral flattening targeting flat autoconvolution.
    In the frequency domain, autoconvolution spectrum = |A(f)|^2.
    We want the time-domain autoconvolution to be flat, which means
    its DFT should be concentrated at DC. So we want |A(f)|^2 to be
    concentrated at f=0, i.e., |A(f)| to be a sinc-like function.
    
    Alternative: iteratively adjust A(f) to make the IFFT of |A(f)|^2 flat.
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    a = normalize(a.copy())
    
    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Compute autoconvolution in frequency domain
        A = rfft(a, fft_size)
        C = irfft(A * A, fft_size)[:size]  # autoconvolution
        
        # Target: make C as flat as possible
        # C_target = mean(C) * ones(size)
        target = C.mean()
        
        # In frequency domain: C = IFFT(|A|^2)
        # We want IFFT(|A_new|^2) to be closer to flat
        # Adjust |A|^2 toward a value that gives flat C
        # If C is flat, then |A|^2 should be delta at f=0 (i.e., A is delta at f=0)
        # But A[0] = sum(a) = 1 (fixed), so |A|^2[0] = 1
        # For flat C: C[k] = 1/(2n-1) for all k
        # DFT(C)[f] = 1/(2n-1) * sum_k exp(-2πikf/fft_size)
        # This is a delta at f=0 with value 1
        
        # So we want |A(f)|^2 to be as close to delta(f) as possible
        # i.e., minimize |A(f)| for f != 0
        # But A[0] = 1 is fixed
        
        # Instead, let's use a different approach:
        # Want C to be flat. C = IFFT(|A|^2).
        # Gradient of max(C) w.r.t. |A|^2 is related to the DFT of the indicator of argmax(C).
        # Simpler: just project C toward flat in the time domain, then back-propagate.
        
        # Time-domain projection: C_new = (1-blend)*C + blend*target (component-wise)
        C_new = (1 - blend) * C + blend * target
        C_new = np.maximum(C_new, 0)  # PSD constraint
        
        # Convert back: |A|^2 = FFT(C_new)
        # But C_new has length `size` (2n-1), and we need it to be PSD
        C_padded = np.zeros(fft_size)
        C_padded[:size] = C_new
        A2 = np.fft.fft(C_padded)  # This should be |A|^2 (real and non-negative for valid C)
        
        # A2 may have small negative or imaginary parts due to numerical issues
        A2 = np.maximum(A2.real, 0)
        
        # A = sqrt(A2) * exp(i*phase(A))
        # Keep the phase of the original A
        phases = np.angle(A)
        mags = np.sqrt(A2)
        
        # For real signal, A is conjugate-symmetric
        A_new = mags * np.exp(1j * phases)
        # Enforce conjugate symmetry
        A_new[1:] = A_new[1:fft_size//2 + 1].copy()
        # Actually, rfft only stores non-negative frequencies
        # Let's just use the real part of ifft
        
        a_new = irfft(A_new, fft_size)[:n]
        a_new = normalize(a_new)
        
        if a_new.sum() < 1e-15:
            break
        a = a_new
    
    return a


def mirrored_peak_shave(a, n_iters=5000, eta0=0.03, deadline=None):
    """
    Peak shaving that accounts for the fact that reducing a[j] affects
    multiple convolution positions.
    Uses a more sophisticated update that considers the full impact.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = score_local(a)
    eta = eta0
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        c = fast_autoconvolve(a)
        max_c = c.max()
        
        # Find all positions where c is within 1% of max
        threshold = max_c * 0.99
        active = np.where(c >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        
        # For each active position k, the gradient is a[k-j]
        # We want to reduce all active peaks simultaneously
        # Use weighted average of gradients
        w = c[active] / c[active].sum()
        
        g = np.zeros(n)
        for i, k in enumerate(active):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            js = np.arange(j_start, j_end + 1)
            g[js] += w[i] * a[k - js]
        
        # Multiplicative update
        scale = -eta * g
        scale -= scale.max()
        a_new = a * np.exp(scale)
        a_new = normalize(a_new)
        
        s = score_local(a_new)
        if s < best_score:
            best_score = s
            best_a = a_new.copy()
            a = a_new
        else:
            eta *= 0.95
            if eta < 1e-6:
                break
    
    return best_a, best_score


def asymmetric_sweep(a_base, n, rng, n_trials=10, deadline=None):
    """
    Try different asymmetry levels by scaling the left vs right half.
    """
    best_a = a_base.copy()
    best_score = score_local(a_base)
    mid = n // 2
    
    for trial in range(n_trials):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Try different asymmetry ratios
        ratio = rng.uniform(0.1, 10.0)
        a = a_base.copy()
        a[:mid] *= ratio
        a = normalize(a)
        
        # Quick peak shave
        a, s = peak_shave_iter(a, n_iters=200, eta0=0.02, top_k=3, deadline=deadline)
        if s < best_score:
            best_score = s
            best_a = a.copy()
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    
    Key ideas:
    1. Exploit asymmetry: the autoconvolution (not autocorrelation) allows
       non-symmetric sequences to achieve lower scores.
    2. Direct peak shaving targets the actual maximum of the convolution.
    3. LogSumExp L-BFGS provides smooth gradient-based refinement.
    4. Multiple initializations with different shapes and lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)
    t_start = time.monotonic()
    
    best_seq = None
    best_score = float('inf')
    best_n = 0
    
    def update_best(a, label):
        nonlocal best_seq, best_score, best_n
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = score_local(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            best_n = len(a)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST [{label}]: score={sc:.8f}, n={len(a)} (t={elapsed:.0f}s)")
    
    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = score_local(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()

    except Exception:
        pass

        n_init = 1200

        best_n = n_init
        print(f"Default init: n={n_init}, score={best_score:.8f}")
