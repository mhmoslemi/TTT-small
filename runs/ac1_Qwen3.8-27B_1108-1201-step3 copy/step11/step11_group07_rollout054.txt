```python
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_conv_matrix(a):
    """Build the convolution matrix A where (A @ g)_k = conv(a, g)_k"""
    n = len(a)

    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # For each j, A[k, j] = a[k-j] for k-j in [0, n-1], i.e., k in [j, j+n-1]
        k_start = j
        k_end = min(j + n, 2*n-1)
        A[k_start:k_end, j] = a[:k_end-j]
    return A

def frank_wolfe_lp(a, M):
    """Solve the LP to find the Frank-Wolfe direction."""
    n = len(a)
    A = build_conv_matrix(a)
    # max 1^T g s.t. A g <= M, g >= 0
    c = -np.ones(n)  # minimize -1^T g
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    if res.status != 0:
        return None
    return res.x

def frank_wolfe_iteration(a, alpha=1.0):
    """One Frank-Wolfe iteration with LP direction."""
    n = len(a)
    b = fast_convolve(a)
    M = b.max()
    
    g = frank_wolfe_lp(a, M)
    if g is None:
        return a, M
    
    s = g.sum()
    if s < 1e-15:
        return a, M
    g = g / s  # normalize to simplex
    
    # Line search
    best_t = 0.0
    best_score = evaluate_local(a)
    for t in [alpha, alpha*0.5, alpha*0.25, alpha*0.1]:
        a_new = (1-t)*a + t*g
        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_t = t
    
    if best_t > 0:
        a = (1-best_t)*a + best_t*g
    return a, best_score
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-spread Frank-Wolfe with convolution-constrained directions
2. FFT-based subgradient with active-set refinement and momentum
3. LogSumExp-smoothed L-BFGS with aggressive temperature annealing
4. Multi-start with diverse initializations and length exploration
5. Spectral domain power-flattening as complementary direction
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_subgradient(a, fraction=0.05):
    """Compute subgradient using active set (top fraction of lags)."""
    n = len(a)
    b = fast_convolve(a)
    M = b.max()
    n_active = max(1, int(len(b) * fraction))
    idx = np.argsort(b)[-n_active:]
    I = np.zeros(len(b))
    I[idx] = 1.0
    N = 1 << (3 * n - 1).bit_length()
    H = rfft(I, N)
    A = rfft(a, N)
    corr = irfft(np.conj(H) * A, N)[:n]
    s = 2.0 * corr / n_active
    return s, b, M


def subgradient_step(a, eta, momentum=0.0):
    """One subgradient step with optional momentum."""
    s, b, M = compute_subgradient(a)
    j_min = np.argmin(s)
    v = np.zeros(len(a))
    v[j_min] = 1.0
    # Soft direction: spread to near-min coordinates
    s_shifted = s - s.min() + 1e-8
    soft = 1.0 / s_shifted
    soft = soft / soft.sum()
    # Blend vertex and soft direction
    v = 0.5 * v + 0.5 * soft
    v = v / v.sum()
    return v, M


def lp_spread_direction(a, M, timeout_s=2.0):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g."""
    n = len(a)
    if n > 2500:
        return None
    # Build constraint matrix A_conv where (A_conv @ g)[k] = conv(a,g)[k]
    A_conv = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A_conv[k_start:k_end, j] = a[:k_end - j]
    c = -np.ones(n)
    t_start = time.monotonic()
    try:
        res = linprog(c, A_ub=A_conv, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': timeout_s, 'presolve': True})
    except Exception:
        return None
    if res.status != 0 or time.monotonic() - t_start > timeout_s:
        return None
    g = res.x
    s = g.sum()
    if s < 1e-10:
        return None
    return g / s


def frank_wolfe_lp_loop(a, time_limit, lp_interval=5):
    """Frank-Wolfe with LP spread direction every lp_interval iterations."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    it = 0
    lp_count = 0
    momentum_v = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        M = b.max()
        score = 2.0 * n * M

        # Every lp_interval iterations, use LP direction
        v = None
        if it % lp_interval == 1:
            v = lp_spread_direction(a, M, timeout_s=1.5)
            if v is not None:
                lp_count += 1
                if it % 100 == 1:
                    print(f"    FW-LP it={it}, score={score:.8f}, lp_solves={lp_count}")

        # Fall back to subgradient
        if v is None:
            s, _, _ = compute_subgradient(a, fraction=0.03)
            j_min = np.argmin(s)
            v = np.zeros(n)
            v[j_min] = 1.0
            # Soft component
            s_shifted = np.maximum(s - s.min() + 1e-10, 1e-10)
            soft = 1.0 / s_shifted
            soft = soft / soft.sum()
            v = 0.4 * v + 0.6 * soft

        # Momentum
        momentum_v = 0.8 * momentum_v + 0.2 * v
        v = momentum_v / momentum_v.sum()

        # Line search
        best_t = 0.0
        best_sc = best_score
        a_new = (1 - best_t) * a + best_t * v
        for t in [0.3, 0.5, 0.7, 0.2, 0.4, 0.6]:
            cand = (1 - t) * a + t * v
            sc = evaluate_local(cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t
                best_a = cand.copy()

        if best_t > 0:
            a = (1 - best_t) * a + best_t * v
        else:
            # Try smaller steps
            improved = False
            for t in [0.1, 0.05, 0.02]:
                cand = (1 - t) * a + t * v
                sc = evaluate_local(cand)
                if sc < best_sc:
                    best_sc = sc
                    best_t = t
                    best_a = cand.copy()
                    a = cand
                    improved = True
                    break
            if not improved:
                break

        if best_sc < best_score - 1e-10:
            best_score = best_sc
            best_a = a.copy()

    return best_a, best_score


def lse_objective_grad(a, T):
    """LogSumExp-smoothed objective and gradient."""
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    exp_sum = exp_c.sum()
    lse = c_max + T * np.log(exp_sum)
    obj = 2.0 * n * lse

    w = exp_c / exp_sum
    N = 1 << (len(w) + n - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    corr = irfft(W * np.conj(A), N)[:n]
    g = 2.0 * corr
    idx = 2 * np.arange(n)
    valid = idx < (2 * n - 1)
    g[valid] -= a[valid] * w[idx[valid]]
    return obj, g


def lse_lbfgs(a0, time_limit, T_list):
    """L-BFGS with LSE smoothing and temperature annealing."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    def func_and_grad(a_in):
        a_in = np.maximum(a_in, 0)
        s = a_in.sum()
        if s < 1e-15:
            return 0, np.zeros(n)
        a_in = a_in / s
        cur_T = T_list[min(T_idx[0], len(T_list) - 1)]
        obj, grad = lse_objective_grad(a_in, cur_T)
        return obj, grad

    T_idx = [0]

    for Ti, T in enumerate(T_list):
        T_idx[0] = Ti
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def obj_fn(a_in):
            a_in = np.maximum(a_in, 0)
            s = a_in.sum()
            if s < 1e-15:
                return 0.0
            a_in = a_in / s
            o, _ = lse_objective_grad(a_in, T)
            return o

        def jac_fn(a_in):
            a_in = np.maximum(a_in, 0)
            s = a_in.sum()
            if s < 1e-15:
                return np.zeros(n)
            a_in = a_in / s
            _, g = lse_objective_grad(a_in, T)
            return g

        maxiter = max(20, int(remaining * 50))
        try:
            res = minimize(
                fun=obj_fn, x0=a, jac=jac_fn, method='L-BFGS-B',
                bounds=[(0, None)] * n,
                options={'maxiter': maxiter, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxfun': maxiter * 100}
            )
            a = normalize(res.x)
            sc = evaluate_local(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception:
            pass

    return best_a, best_score


def spectral_flatten_step(a, lr=0.01):
    """One step of spectral flattening (reduce PSD variance)."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)
    P = np.abs(A) ** 2
    P_mean = P.mean() + 1e-15

    # Direction to flatten: h = IFFT(-(P - P_mean) * conj(A) / (P + eps))
    H = np.zeros(fft_size // 2 + 1, dtype=np.complex128)
    valid = P > 1e-12
    H[valid] = -(P[valid] - P_mean[valid]) * np.conj(A[valid]) / (P[valid] + 1e-10)
    H[0] = 0.0
    h = irfft(H, fft_size)[:n]
    h -= h.mean()
    h_norm = np.linalg.norm(h)
    if h_norm < 1e-15:
        return a
    h = h / h_norm
    a_new = a + lr * h
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s


def spectral_optimize(a0, time_limit, n_iters=5000):
    """Iterative spectral flattening with line search."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    lr = 0.02

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        h_dir = spectral_flatten_step(a, lr)
        improved = False
        for scale in [1.0, 0.5, 0.25, 0.1]:
            cand = normalize(h_dir if scale == 1.0 else a + scale * 0.02 * (h_dir - a))
            # Actually just use the direction properly
            break
        # Redo: use h as direction
        fft_size = 1 << (2 * n - 1).bit_length()
        A = rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean() + 1e-15
        H = np.zeros(fft_size // 2 + 1, dtype=np.complex128)
        valid = P > 1e-12
        H[valid] = -(P[valid] - P_mean[valid]) * np.conj(A[valid]) / (P[valid] + 1e-10)
        H[0] = 0.0
        h = irfft(H, fft_size)[:n]
        h -= h.mean()
        h_norm = np.linalg.norm(h)
        if h_norm < 1e-15:
            break
        h = h / h_norm

        for scale in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            cand = a + scale * h
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-15:
                continue
            cand = cand / s
            sc = evaluate_local(cand)
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
                a = cand.copy()
                improved = True
            break

        if not improved:
            lr *= 0.8
            if lr < 1e-6:
                break

    return best_a, best_score


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    candidates = []
    x = np.linspace(0, 1, n)

    candidates.append(normalize(np.exp(-6 * (x - 0.5) ** 2)))
    candidates.append(normalize(0.5 * (1 + np.cos(2 * np.pi * x))))
    candidates.append(normalize(np.sin(np.pi * x) ** 2))
    candidates.append(normalize((x * (1 - x)) ** 0.3))
    candidates.append(normalize(np.exp(-8 * (x - 0.25) ** 2) + np.exp(-8 * (x - 0.75) ** 2)))

    # Modulated cosine
    a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
    candidates.append(normalize(np.maximum(a, 0)))

    # Random smooth
    coeffs = rng.normal(0, 1, 8)
    a = np.zeros(n)
    for i, c in enumerate(coeffs):
        a += c * np.cos(2 * np.pi * (i + 1) * x + rng.uniform(0, 2 * np.pi))
    candidates.append(normalize(np.maximum(a + 0.5, 0)))

    # Uniform
    candidates.append(np.ones(n) / n)

    # Edge concentration
    candidates.append(normalize(np.exp(-3 * x) + np.exp(-3 * (1 - x))))

    # Triangular
    candidates.append(normalize(np.minimum(x, 1 - x) + 0.1))

    return candidates


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return normalize(np.maximum(f_interp, 0))


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LP-spread Frank-Wolfe, LSE-LBFGS, spectral flattening,
    and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        candidates = diverse_init(n_init, rng)
        scores = [evaluate_local(c) for c in candidates]
        best_idx = np.argmin(scores)
        best_seq = candidates[best_idx].copy()
        best_score = scores[best_idx]
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.8f}")

    # === Phase 1: LP-Spread Frank-Wolfe (main workhorse) ===
    print(f"\n=== Phase 1: LP-Spread Frank-Wolfe (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 540.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, n={n_best}")
        a_fw, _ = frank_wolfe_lp_loop(best_seq, t_alloc, lp_interval=3)
        local_update(a_fw, "FW_LP_1")

    # === Phase 2: Spectral flattening ===
    print(f"\n=== Phase 2: Spectral Flattening (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        a_spec, _ = spectral_optimize(best_seq, t_alloc, n_iters=3000)
        local_update(a_spec, "spectral_1")

    # === Phase 3: LSE + L-BFGS ===
    print(f"\n=== Phase 3: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        T_stages = [10, 5, 3, 2, 1, 0.7, 0.5, 0.3, 0.2, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002]
        print(f"  T stages: {len(T_stages)}, budget={t_alloc:.0f}s")
        a_lb, _ = lse_lbfgs(best_seq, t_alloc, T_stages)
        local_update(a_lb, "LSE_1")

    # === Phase 4: Second round FW-LP (with lower lp_interval for more LPs) ===
    print(f"\n=== Phase 4: FW-LP Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        a_fw2, _ = frank_wolfe_lp_loop(best_seq, t_alloc, lp_interval=1)
        local_update(a_fw2, "FW_LP_2")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Length Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for n_try in [n_best + 200, n_best - 200, n_best + 400, n_best + 600]:
            if n_try < 200 or n_try > 4000:
                continue
            rem_sub = deadline - time.monotonic()
            if rem_sub < 30:
                break
            init = interpolate_sequence(best_seq, n_try)
            T_short = [3, 1, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
            t_sub = min(rem_sub * 0.3, 60.0)
            a_try, _ = lse_lbfgs(init, t_sub, T_short)
            local_update(a_try, f"len_{n_try}")

    # === Phase 6: Final LSE polish ===
    print(f"\n=== Phase 6: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        T_final = [1, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01, 0.005, 0.002, 0.001]
        a_fin, _ = lse_lbfgs(best_seq, min(rem * 0.6, 60.0), T_final)
        local_update(a_fin, "final_LSE")

    # === Phase 7: Ultra-fine spectral + subgradient ===
    print(f"\n=== Phase 7: Ultra-fine (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        a_sp2, _ = spectral_optimize(best_seq, min(rem * 0.4, 30.0), n_iters=2000)
        local_update(a_sp2, "spectral_final")
        rem = deadline - time.monotonic()
        if rem > 10:
            a_fw3, _ = frank_wolfe_lp_loop(best_seq, min(rem * 0.5, 20.0), lp_interval=2)
            local_update(a_fw3, "FW_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
```
