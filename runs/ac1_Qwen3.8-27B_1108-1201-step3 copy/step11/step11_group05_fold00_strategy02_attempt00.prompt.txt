    """

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp, top_k=None):
    """Compute LogSumExp objective and gradient w.r.t. z using top-k focusing."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    
    if top_k is not None and top_k < len(b):
        # Focus on top-k peaks
        idx = np.argpartition(b, -top_k)[-top_k:]
        b_top = b[idx]
        max_b_top = b_top.max()
        exp_terms = np.exp((b_top - max_b_top) / T_temp)
        w_top = exp_terms / exp_terms.sum()
        # Reconstruct full weight vector (sparse)
        w = np.zeros(len(b))
        w[idx] = w_top
        lse = max_b_top + T_temp * np.log(exp_terms.sum())
    else:
        max_b = b.max()
        exp_terms = np.exp((b - max_b) / T_temp)
        w = exp_terms / exp_terms.sum()
        lse = max_b + T_temp * np.log(exp_terms.sum())
    
    obj = 2.0 * n * lse

    # Gradient: g_a[i] = (4n/T) * sum_k w[k] * a[k-i]
    # = (4n/T) * correlate(w, a)[n-1 + i]
    # Use FFT for efficiency
    fft_size = 1 << (2 * n - 1).bit_length()
    W = rfft(w, fft_size)
    A_rev = rfft(a[::-1], fft_size)
    g_a = irfft(W * A_rev, fft_size)[:n]
    # Actually let me just compute it directly via correlation
    # g_a[i] = sum_k w[k] * a[k-i] for valid k-i
    # This is np.correlate(w, a, 'full')[n-1:2*n-1]
    # But for speed, use FFT:
    # correlate(w, a)[i] = sum_k w[k+i] * a[k] (numpy convention)
    # I want: sum_k w[k] * a[k-i] = sum_j w[j+i] * a[j] = correlate(w, a) at lag i
    
    # Using np.correlate directly (it's C-optimized)
    g_a_raw = np.correlate(w, a, mode='full')
    g_a = g_a_raw[n - 1:2 * n - 1] * 2.0  # factor of 2 from derivative
    
    # Total gradient of objective w.r.t. a
    grad_a = (2.0 * n / T_temp) * g_a
    
    # Convert to z-space gradient
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, 2.0 * n * grad_z / (2.0 * n)  # The 2n factor is already in grad_a


def adam_logsumexp(a0, time_limit, T_stages, lr_init, lr_decay, top_k_frac=0.1):
    """Adam optimization in log-space with temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step_count = 0
    
    total_stages = len(T_stages)
    steps_per_stage = max(5000, int(time_limit * 50000 / total_stages))
    
    for stage_idx, T_stage in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 1:
            break
        
        top_k = max(5, int(top_k_frac * (2 * n - 1)))
        
        for step in range(steps_per_stage):
            if time.monotonic() > deadline:
                break
            if (step + 1) % 5000 == 0 and time.monotonic() - t0 > 30:
                # Check time
                if time.monotonic() > deadline:
                    break
            
            a = softmax_from_z(z)
            b = fast_convolve(a)
            max_b = b.max()
            score = 2.0 * n * max_b
            step_count += 1
            
            if score < best_score:
                best_score = score
                best_a = a.copy()
            
            # Compute gradient
            if T_stage < 0.01:
                # Use pure subgradient on active set
                threshold = max_b * (1.0 - 1e-4)
                active = np.where(b >= threshold)[0]
                if len(active) == 0:
                    active = np.array([np.argmax(b)])
                w = np.zeros(len(b))
                w[active] = 1.0 / len(active)
            else:
                # LogSumExp with top-k
                if top_k < len(b):
                    idx = np.argpartition(b, -top_k)[-top_k:]
                    b_top = b[idx]
                    max_top = b_top.max()
                    exp_top = np.exp((b_top - max_top) / T_stage)
                    w_top = exp_top / exp_top.sum()
                    w = np.zeros(len(b))
                    w[idx] = w_top
                else:
                    max_b_all = b.max()
                    exp_all = np.exp((b - max_b_all) / T_stage)
                    w = exp_all / exp_all.sum()
            
            # Gradient in a-space
            g_a_raw = np.correlate(w, a, mode='full')
            g_a = g_a_raw[n - 1:2 * n - 1] * 2.0
            
            # Gradient in z-space
            S = np.dot(g_a, a)
            grad_z = a * (g_a - S)
            
            # Adam update
            lr = lr_init * (lr_decay ** stage_idx) / (1.0 + 0.1 * step / steps_per_stage)
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** (step + 1))
            v_hat = v / (1 - beta2 ** (step + 1))
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Keep z bounded to prevent numerical issues
            z -= z.mean()  # Centering (doesn't change softmax)
            
            # Progress print
            if (step + 1) % 20000 == 0:
                elapsed = time.monotonic() - t0
                cur_score = 2.0 * n * fast_convolve(softmax_from_z(z)).max()
                print(f"    Adam stage {stage_idx} (T={T_stage:.3f}) step {step+1} ({elapsed:.0f}s): best={best_score:.8f} cur={cur_score:.8f}")
    
    return best_a, best_score

def multiplicative_subgradient(a0, time_limit, n_steps=100000):
    """Mirror descent (multiplicative weights) targeting active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta_base = 0.03
    no_improve = 0
    scale = 1.0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Adaptive active set
        threshold = max_b * (1.0 - max(1e-5, 1e-3 / (1 + step / 1000)))
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([0])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        eta = eta_base * scale
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 50:
            scale *= 0.8
            no_improve = 0
            if scale < 0.005:
                break

        if step % 20000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    MultiSubGrad {elapsed:.0f}s: score={best_score:.8f} eta_scale={scale:.4f}")

    return best_a, best_score


def structured_init(n, rng):

        a = (1 - x) ** 1.5 + 0.1

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()

        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturbed_sequence(a, rng, strength=0.02):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 10)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s
