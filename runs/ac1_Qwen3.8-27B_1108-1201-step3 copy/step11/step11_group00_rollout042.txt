"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Asymmetric (ramp/Beta) initialization for fast convergence
2. FFT-based LogSumExp multi-stage annealing (main workhorse)
3. LP Nesterov-accelerated best-response as supplement
4. Peak-shaving coordinate descent for final polish
5. Multi-length interpolation and exploration
Key insight: asymmetric sequences (ramp-like) achieve ~4/3 vs 2 for uniform,
so we initialize with highly asymmetric shapes and let the optimizer refine.
"""

import numpy as np
import time
import math
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_autocorr(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    out = irfft(A * A, fft_size)[:size]
    return out


def cross_corr(a, w):
    """Compute C[m] = sum_j a[j]*w[j+m] for m=0..n-1 via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    W = rfft(np.asarray(w, dtype=np.float64), fft_size)
    conv = irfft(A * W, fft_size)[:size]
    # C[m] = conv[n-1-m], so C = conv[0:n] reversed
    return conv[0:n][::-1]


def stable_softmax(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64)
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_objective_grad(z, T):
    """
    Compute (1/T)*log(sum exp(T*(a*a)_k)) and its gradient w.r.t. z,
    where a = softmax(z). Returns (value, grad_z).
    """
    a = stable_softmax(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum = max_Tb + np.log(exp_Tb.sum())
    val = log_sum / T
    w = exp_Tb / exp_Tb.sum()
    C = cross_corr(a, w)
    grad_a = 2.0 * C
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return val, grad_z


def evaluate_direct(a):
    """Compute the objective directly: 2*n*max(conv(a,a))/(sum a)^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    max_b = b.max()
    n = len(a)
    return float(2.0 * n * max_b / (s * s))


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def logsumexp_optimize(a0, time_limit, T_start=200.0, T_end=0.05, n_stages=20):
    """
    Multi-stage LogSumExp annealing with L-BFGS-B.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    a_init = a_init / a_init.sum()
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = evaluate_direct(a_init)

    for si in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        stage_T = temps[si]
        iters = max(20, min(100, int(50 + 30 * si / n_stages)))

        def func(z_in):
            v, _ = lse_objective_grad(z_in, stage_T)
            return v

        def jac(z_in):
            _, g = lse_objective_grad(z_in, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters * 100,
                         'disp': False}
            )
            z = res.x
            a = stable_softmax(z)
            score = evaluate_direct(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception:
            pass

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"        LSE stage {si}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. conv(a_fixed, b)_k <= M, sum(b)=1, b>=0."""
    try:
        import cvxpy as cp
    except ImportError:
        return None, None

    n = len(a_fixed)
    if n > 3000:
        return None, None

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=solver)
    except Exception:
        prob = None

    if prob is None or prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            try:
                prob = cp.Problem(cp.Minimize(M_var), constraints)
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

    if prob is None or prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=5000):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_direct(a)
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        y = y_new
        a = a_new

        score_a = evaluate_direct(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = evaluate_direct(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        avg_a += a
        n_avg += 1
        if n_avg >= 10 and n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            score_avg = evaluate_direct(avg_check)
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"        NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """Coordinate descent: reduce top autocorrelation peaks."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = evaluate_direct(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:5]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            src_local = np.argmax(contrib)
            src = idx[src_local]

            # Find a low-value target
            candidates = np.argsort(a)[:max(3, n // 10)]
            dst = candidates[rng.integers(len(candidates))]
            if dst == src:
                dst = (dst + 1) % n

            for frac in [0.005, 0.01, 0.02, 0.04, 0.06]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s = a_trial.sum()
                a_trial = a_trial / s
                score = evaluate_direct(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = evaluate_direct(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            stall += 1
            if stall > 500:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length usingubic."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def ramp_init(n, alpha=2.0):
    """Generate Beta(alpha, 1)-like initialization: a_j ∝ ((j+1)/n)^(alpha-1)."""
    x = np.arange(n, dtype=np.float64) + 1.0
    x_norm = x / n
    a = x_norm ** (alpha - 1.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def get_lp_solver():
    """Find a working LP solver."""
    try:
        import cvxpy as cp
        for solver_name in [cp.ECOS, cp.SCS, cp.PDLP]:
            try:
                x = cp.Variable(5, nonneg=True)
                prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
                prob.solve(solver=solver_name)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    return solver_name
            except Exception:
                continue
        return cp.ECOS
    except ImportError:
        return None
