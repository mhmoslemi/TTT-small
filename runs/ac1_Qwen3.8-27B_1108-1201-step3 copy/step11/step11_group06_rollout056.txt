    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    n = len(a)
    m = len(b)

    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix_sparse(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    counts = np.minimum(n, size - np.arange(n))
    total = int(counts.sum())
    cols = np.repeat(np.arange(n), counts)
    offsets = np.concatenate([np.arange(c) for c in counts])
    rows = cols + offsets
    vals = a[offsets].astype(np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(size, n))

            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]

    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based Nesterov dual averaging with vectorized sparse matrices (main workhorse)
2. Fast multiplicative weights with adaptive step for dual tracking
3. Multi-length exploration (n=1200, 1500, 1800)
4. Multi-stage L-BFGS with LogSumExp smoothing and fine temperature cooling
5. Periodic diversification and best-of-primal/dual tracking

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import csr_matrix


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix_sparse(a):
    n = len(a)
    size = 2 * n - 1
    counts = np.minimum(n, size - np.arange(n))
    total = int(counts.sum())
    cols = np.repeat(np.arange(n), counts)
    offsets = np.concatenate([np.arange(c) for c in counts])
    rows = cols + offsets
    vals = a[offsets].astype(np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(size, n))


def get_lp_solver():
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, T_mat):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_val = b_val.sum()
                if s_val < 1e-15:
                    return None, None
                return b_val / s_val, max(0.0, float(M_var.value))
        except Exception:
            continue
    return None, None


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lse_func(z, n, T_temp, deadline):
    if time.monotonic() > deadline:
        return 1e15
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T_temp
    shifted = scaled - scaled.max()
    exp_terms = np.exp(shifted)
    lse = max_b + T_temp * (scaled.max() + np.log(exp_terms.sum() + 1e-300))
    return 2.0 * n * lse


def lse_jac(z, n, T_temp, deadline):
    if time.monotonic() > deadline:
        return np.zeros(len(z))
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T_temp
    shifted = scaled - scaled.max()
    exp_terms = np.exp(shifted)
    w = exp_terms / exp_terms.sum()
    T_mat = build_conv_matrix_sparse(a)
    grad_a = 2.0 * (T_mat.T @ w)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return 2.0 * n * grad_z


def multiplicative_weights(a0, time_limit, n_steps, eta0=0.05):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 5e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        w_active = b[active] / b[active].sum()
        g = np.zeros(n)
        for w, k in zip(w_active, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w * a[k - idx]
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.005, 0.25)
        else:
            eta *= 0.9
            if eta < 1e-9:
                break
    return best_a, best_score

def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage=500):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        try:
            res = minimize(
                fun=lambda zz: lse_func(zz, n, stage_T, deadline),
                x0=z,
                jac=lambda zz: lse_jac(zz, n, stage_T, deadline),
                method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-25,
                         'gtol': 1e-18, 'maxfun': maxiter_per_stage * 300,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass
    return softmax_from_z(z)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
