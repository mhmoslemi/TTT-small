The triangle has area `1` (since `sum conv = (sum a)^2 = 1`) and height `1/n`. A flat function with the same area over `2n-1` points would have height `1/(2n-1)`. So the theoretical minimum is `1/(2n-1)`, giving a score of `2n/(2n-1) ≈ 1`.

The question is: how close can we get to a flat convolution while keeping `a` non-negative?

This is related to the "flat-top window" problem in signal processing. The ideal solution would be a sequence whose linear autocorrelation is as flat as possible.

For the purpose of this problem, the algorithm needs to:
1. Find a non-negative sequence `a` with `sum(a) = 1`
2. Minimize `max_k conv(a,a)_k`
3. The score is `2n * max_k conv(a,a)_k`

Now, the LP-based best response approach works as follows:
- Fix `y`, find `a` that minimizes `max_k (y * a)_k` subject to `a >= 0, sum(a) = 1`.
- This is valid because at a local minimum of the max-min problem, each coordinate is a best response to the other.

The convergence of this method (with Nesterov acceleration) should be O(1/√k) for the gap, or O(1/k) with proper acceleration.

Let me now focus on the implementation. I'll do the following:

1. **Primary method**: Nesterov-accelerated LP best response, with:
   - Heuristic restart (reset momentum if no improvement for K iterations)
   - Adaptive step size
   - Multiple n values (try 1500, 2000, 2500)
   
2. **Secondary method**: LogSumExp smoothing + L-BFGS for final polish

3. **Tertiary method**: Simulated annealing with random perturbations for exploration

Let me also think about the LP formulation more carefully. The LP at each step is:
```
min M
s.t. T @ a <= M * ones
     ones @ a = 1
     a >= 0
```

where T is a (2n-1) x n matrix with T[k,j] = y[k-j] (with y indexed from 0 to n-1, and y[m]=0 for m<0 or m>=n).

This LP has a very special structure: T is a "hankel" matrix (or more precisely, it's constructed from a fixed vector y). The constraints are `sum_j y[k-j] * a_j <= M` for each k.

For PDLP, we should specify the constraint matrix efficiently. Since the matrix is dense (each row has up to n non-zeros), we'll use the dense formulation.

Actually, for n=2000, the matrix is 3999 x 2000 = ~8 million entries. That's manageable for PDLP.

import numpy as np

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def evaluate_safe(a):

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def build_conv_matrix(y):
    """Build the (2n-1) x n convolution matrix for fixed y."""
    n = len(y)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        # T[k, j] = y[k - j] if 0 <= k-j < n
        k_start = j  # k - j >= 0 => k >= j
        k_end = min(size, j + n)  # k - j < n => k < j + n
        T[k_start:k_end, j] = y[:k_end - j]
    return T

_lp_solver = None

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver

def solve_lp_best_response(y, solver=None):
    """Solve min M s.t. T_y @ a <= M, sum(a)=1, a>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(y)
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(y)
    constraints = [T @ a_var <= M_var, cp.sum(a_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    a_val = np.asarray(a_var.value, dtype=np.float64)
    a_val = np.maximum(a_val, 0)
    s = a_val.sum()
    if s < 1e-15:
        return None, None
    a_val = a_val / s
    return a_val, float(M_var.value)

def nesterov_lp(a0, time_limit, solver=None, max_iters=1000000, heuristic_restart=True):
    """Nesterov-accelerated LP best response with heuristic restart."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    # Heuristic restart parameters
    no_improve_count = 0
    restart_threshold = 500  # restart after this many non-improving iterations
    score_tol = 1e-9
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) ** 2)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score - score_tol:
            best_score = sc_a
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        sc_y = score_normalized(y)
        if sc_y < best_score - score_tol:
            best_score = sc_y
            best_a = y.copy()
            no_improve_count = 0

        # Heuristic restart
        if heuristic_restart and no_improve_count >= restart_threshold:
            y = a.copy()
            no_improve_count = 0
            # Optionally: partially reset a
            # a = 0.5 * a + 0.5 * best_a
            # a = np.maximum(a, 0)
            # s = a.sum()
            # if s > 1e-15: a = a / s

    return best_a, best_score

def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    # Use log-sum-exp approximation of max
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    # Gradient with respect to a
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def lbgfs_logsumexp(a0, time_limit, T_start=100.0, T_end=0.05, n_stages=15):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = 50

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxfun': iters_per_stage * 200,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(res.x)
            sc = score_normalized(a_check)
            if (si + 1) % 5 == 0:
                print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.2f}: score={sc:.8f}")
        except Exception:
            pass

    return softmax_from_z(z)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x ** 2)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 4:
        a = np.ones(n) * (1.0 + 0.2 * np.cos(2 * np.pi * 3 * np.arange(n) / n))
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.15 * np.cos(7 * x)
    elif choice == 6:
        # Raised cosine window
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    else:
        # Hann window
        x = np.arange(n)
        a = 0.5 * (1 - np.cos(2 * np.pi * x / n))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def project_simplex(v):
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    n = len(a)

def nesterov_lp(a0, time_limit, solver=None, max_iters=1000000, restart_interval=400):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    no_improve = 0
    it_offset = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1 + it_offset) ** 2)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        improved = False
        if sc_a < best_score - 1e-10:
            best_score = sc_a
            best_a = a.copy()
            improved = True
        sc_y = score_normalized(y)
        if sc_y < best_score - 1e-10:
            best_score = sc_y
            best_a = y.copy()
            improved = True

        if improved:
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= restart_interval:
            y = a.copy()
            no_improve = 0
            it_offset += restart_interval

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=80.0, T_end=0.05, n_stages=12):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def _func(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj

        def _jac(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        try:
            res = minimize(
                fun=_func, x0=z, jac=_jac, method='L-BFGS-B',
                options={'maxiter': 60, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxfun': 60 * 200,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(res.x)
            sc = score_normalized(a_check)
            if (si + 1) % 4 == 0:
                print(f"    LSE stage={si + 1}/{n_stages} T={stage_T:.2f}: score={sc:.8f}")
        except Exception:
            pass

    return softmax_from_z(z)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        best_seq = None
        print("No parent available, using structured init")

    n_parent = len(best_seq) if best_seq is not None else 1200

    # Define n values to try, prioritizing larger n for better scores
    # Time per n: enough for Nesterov to make progress
    n_trials = []
    if best_seq is not None:
        n_trials.append(n_parent)
    n_trials.extend([1500, 2000, 2500])
    # Remove duplicates while preserving order
    seen = set()
    n_unique = []
    for nv in n_trials:
        if nv not in seen and 500 <= nv <= 4000:
            seen.add(nv)
            n_unique.append(nv)
    n_trials = n_unique

    print(f"Trying n values: {n_trials}")

    total_budget = budget_s - 30.0

    # Allocate time: give more time to larger n (they have more potential)
    # But ensure each gets at least 60s
    weights = [max(0.8, nv / 1500.0) for nv in n_trials]
    total_w = sum(weights)
    time_alloc = [w / total_w * total_budget * 0.85 for w in weights]

    for idx, n_try in enumerate(n_trials):
        remaining = deadline - time.monotonic()
        t_alloc = min(time_alloc[idx], remaining * 0.9)
        if t_alloc < 30:
            break

        print(f"\n=== n={n_try}, alloc={t_alloc:.0f}s, remaining={remaining:.0f}s ===")

        # Initialize
        if best_seq is not None:
            init = interpolate_sequence(best_seq, n_try)
        else:
            init = structured_init(n_try, rng)

        # Phase A: Nesterov LP (70% of allocation)
        t_nest = t_alloc * 0.70
        a_out, sc_out = nesterov_lp(init, t_nest, solver=solver,
                                     max_iters=500000, restart_interval=300)
        if a_out is not None:
            a_out = np.maximum(a_out, 0)
            s = a_out.sum()
            if s > 1e-15:
                a_out = a_out / s
                sc = evaluate_safe(a_out)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out.copy()
                    print(f"  NEW BEST [nesterov n={n_try}]: {sc:.8f}")

        # Phase B: LSE polish (20% of allocation)
        remaining = deadline - time.monotonic()
        t_lse = min(t_alloc * 0.20, remaining * 0.5)
        if t_lse > 10 and best_seq is not None:
            a_lse = lbgfs_logsumexp(best_seq, t_lse,
                                     T_start=60.0, T_end=0.05, n_stages=10)
            a_lse = np.maximum(a_lse, 0)
            s = a_lse.sum()
            if s > 1e-15:
                a_lse = a_lse / s
                sc = evaluate_safe(a_lse)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_lse.copy()
                    print(f"  NEW BEST [lse n={len(a_lse)}]: {sc:.8f}")

        # Phase C: Nesterov round 2 from best (10% of allocation)
        remaining = deadline - time.monotonic()
        t_r2 = min(t_alloc * 0.10, remaining * 0.4)
        if t_r2 > 10 and best_seq is not None:
            a_r2, _ = nesterov_lp(best_seq, t_r2, solver=solver,
                                   max_iters=300000, restart_interval=200)
            if a_r2 is not None:
                a_r2 = np.maximum(a_r2, 0)
                s = a_r2.sum()
                if s > 1e-15:
                    a_r2 = a_r2 / s
                    sc = evaluate_safe(a_r2)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_seq = a_r2.copy()
                        print(f"  NEW BEST [nesterov2 n={len(a_r2)}]: {sc:.8f}")

    # Final: random restarts at best length
    remaining = deadline - time.monotonic()
    if remaining > 40 and best_seq is not None:
        n_best = len(best_seq)
        print(f"\n=== Random restarts at n={n_best}, remaining={remaining:.0f}s ===")
        n_restarts = min(4, int(remaining / 25))
        for r in range(n_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            init_r = structured_init(n_best, rng)
            blend = 0.4 * init_r + 0.6 * (best_seq / best_seq.sum())
            blend = np.maximum(blend, 0)
            sb = blend.sum()
            if sb > 1e-15:
                blend = blend / sb
            t_r = remaining * 0.35
            a_r, _ = nesterov_lp(blend, t_r, solver=solver,
                                  max_iters=200000, restart_interval=200)
            if a_r is not None:
                a_r = np.maximum(a_r, 0)
                s = a_r.sum()
                if s > 1e-15:
                    a_r = a_r / s
                    sc = evaluate_safe(a_r)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_seq = a_r.copy()
                        print(f"  NEW BEST [restart_{r}]: {sc:.8f}")

    # Final LSE fine-tune
    remaining = deadline - time.monotonic()
    if remaining > 15 and best_seq is not None:
        t_final = min(remaining * 0.5, 45.0)
        if t_final > 5:
            a_final = lbgfs_logsumexp(best_seq, t_final,
                                       T_start=20.0, T_end=0.02, n_stages=8)
            a_final = np.maximum(a_final, 0)
            s = a_final.sum()
            if s > 1e-15:
                a_final = a_final / s
                sc = evaluate_safe(a_final)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_final.copy()
                    print(f"  NEW BEST [final_lse]: {sc:.8f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
        return [float(x) for x in best_seq]

    # Fallback
    fallback = structured_init(1200, rng)
    sc_f = evaluate_safe(fallback)
    print(f"\nFALLBACK: score={sc_f:.8f}, n=1200")
    return [float(x) for x in fallback]
