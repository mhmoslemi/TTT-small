def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix_sparse(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = lil_matrix((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return csc_matrix(T)


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS, cp.GUROBI, cp.MOSEK]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix_sparse(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS, cp.GUROBI, cp.MOSEK]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    return None, None
                b_val = b_val / s_sum
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def solve_active_qcqp(a, solver, rel_tol=1e-5, max_active=20):
    """Solve active-set QCQP."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    threshold = max_b * (1.0 - rel_tol)
    active = np.where(b >= threshold)[0]
    if len(active) > max_active:
        active = active[np.argsort(b[active])[-max_active:]]
    if len(active) == 0:
        active = np.array([np.argmax(b)])

    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]
    for k in active:
        Q = build_q_sparse(int(k), n)
        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

    prob = cp.Problem(cp.Minimize(t_var), constraints)
    solvers_to_try = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:

            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()

                    return None, None
                a_val = a_val / s_sum
                return a_val, float(max(0, t_var.value))
        except Exception:
            continue
    return None, None

            constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

        solvers_to_try = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.ECOS, cp.SCS]

def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def gradient_of_max_conv(a, top_k=10):
    """Compute gradient of max conv value w.r.t. a, averaged over top-k peaks."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    
    # Use softmax weighting over top peaks
    scores = (b - max_b) / (max_b * 1e-4 + 1e-10)
    w = np.exp(scores)
    w /= w.sum()
    
    grad = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-12:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[k] * a[k - idx]
    
    return grad, float(max_b)

    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    eta = 0.01
    no_improve = 0
    patience = 50

            eta = min(eta * 1.01, 0.05)
        else:
            no_improve += 1
            if no_improve >= patience:
                eta *= 0.7
                no_improve = 0
                if eta < 1e-5:

        if (step + 1) % 5000 == 0:

def fista_lp_alternate(a0, time_limit, solver, n_max_iters=500000):
    """
    LP alternating with FISTA (accelerated proximal gradient) approach.
    The "prox" here is identity (since we're on the simplex and the LP
    best response automatically stays on the simplex).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(123)
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute best response to y
        b_new, M_val = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = b_new + ((t_k - 1.0) / t_new) * (b_new - a)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        a = b_new
        t_k = t_new
        
        # Track best
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        # Adaptive restart
        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            
            if stall_count >= 5:
                # Restart with small perturbation
                noise = 0.01 * rng.standard_normal(n)
                a = a + noise
                a = np.maximum(a, 0)
                sa = a.sum()
                if sa > 1e-15:
                    a = a / sa
                y = a.copy()
                t_k = 1.0
                stall_count = 0
        
        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA-LP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f}")
    
    return best_a, best_score


def lse_lbgfs_refine(a0, time_limit, deadline):

    n = len(a0)

    return softmax_from_z(z)


def _lse_obj(z, n, T, deadline):
    if time.monotonic() > deadline:
        return 1e15
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T)
    lse = max_b + T * np.log(exp_terms.sum())
    return 2.0 * n * lse


def _lse_grad(z, n, T, deadline):
    if time.monotonic() > deadline:
        return np.zeros_like(z)
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T)
    w = exp_terms / exp_terms.sum()
    
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    S_val = np.dot(grad_a, a)
    grad_z = a * (grad_a - S_val)
    return 2.0 * n * grad_z


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    if s < 0.01:

    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import lil_matrix, csc_matrix, coo_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix_co
o(a):
    """Build COO sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for j in range(n):
        rows.extend(range(j, j + n))
        cols.extend([j] * n)
        vals.extend(a.tolist())
    T = coo_matrix((vals, (rows, cols)), shape=(size, n))
    return T.tocsr()


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS, cp.GUROBI, cp.MOSEK]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix_coo(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS, cp.GUROBI, cp.MOSEK]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    return None, None
                b_val = b_val / s_sum
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)

        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum < 1e-15:
                    return None, None
                a_val = a_val / s_sum
                return a_val, float(max(0, t_var.value))
        except Exception:
            continue
    return None, None


def solve_qcqp_with_active_set(a, solver, active_set):
    """Solve QCQP with a specific active set."""
    n = len(a)
    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]
    for k in active_set:
        Q = build_q_sparse(int(k), n)
        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

    prob = cp.Problem(cp.Minimize(t_var), constraints)
    solvers_to_try = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum < 1e-15:
                    return None, None
                a_val = a_val / s_sum
                return a_val, float(max(0, t_var.value))
        except Exception:
            continue
    return None, None


def kkt_guided_qcqp(a, solver, max_iters=5, max_active_total=25):
    """
    KKT-guided active set QCQP:
    - Start with top peaks
    - Solve QCQP
    - Check for violated constraints
    - Add most violated
    - Repeat until convergence
    """
    n = len(a)
    b = fast_convolve(a)
    sorted_peaks = np.argsort(b)[::-1]
    active_list = list(sorted_peaks[:min(20, len(sorted_peaks))])

    best_a = a.copy()
    best_score = 2.0 * n * b.max()

    for iteration in range(max_iters):
        if len(active_list) > max_active_total:
            active_list = list(sorted_peaks[:max_active_total])

        a_new, t_val = solve_qcqp_with_active_set(a, solver, np.array(active_list))
        if a_new is None:
            break

        b_new = fast_convolve(a_new)
        current_max = b_new.max()

        # Check KKT: find constraints violated at the solution
        violated = []
        for k in range(len(b_new)):
            if k in active_list:
                continue
            if b_new[k] > t_val * (1.0 + 1e-6) and b_new[k] > best_score / (2.0 * n) * 0.99:
                violated.append((k, b_new[k]))

        if not violated:
            best_a = a_new
            break

        # Add most violated constraints
        violated.sort(key=lambda x: x[1], reverse=True)
        for k, _ in violated[:3]:
            if k not in active_list:
                active_list.append(k)

        if a_new is not None and 2.0 * n * current_max < best_score:
            best_score = 2.0 * n * current_max
            best_a = a_new.copy()

    final_score = 2.0 * n * fast_convolve(best_a).max()
    return best_a, final_score


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def _lse_obj_func(z, n, T_temp, deadline):
    """Top-level LSE objective."""
    if time.monotonic() > deadline:
        return 1e15
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    return 2.0 * n * lse


def _lse_grad_func(z, n, T_temp, deadline):
    """Top-level LSE gradient."""
    if time.monotonic() > deadline:
        return np.zeros_like(z)
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    w = exp_terms / exp_terms.sum()

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S_val = np.dot(grad_a, a)
    grad_z = a * (grad_a - S_val)
    return 2.0 * n * grad_z


def fista_lp_alternate(a0, time_limit, solver, n_max_iters=500000):
    """
    LP alternating with FISTA acceleration and adaptive restarts.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(777)
    lp_durations = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_start = time.monotonic()
        b_new, M_val = solve_lp_best_response(y, solver)
        lp_durations.append(time.monotonic() - lp_start)

        if b_new is None:
            continue

        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = b_new + ((t_k - 1.0) / t_new) * (b_new - a)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        a = b_new.copy()
        t_k = t_new

        # Track best
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Adaptive restart based on stalling
        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-11:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 10:
                # Perturb and restart
                noise_scale = 0.005 * (1.0 + 0.5 * np.sin(0.1 * it))
                noise = noise_scale * rng.standard_normal(n)
                a = a + noise
                a = np.maximum(a, 0)
                sa = a.sum()
                if sa > 1e-15:
                    a = a / sa
                y = a.copy()
                t_k = 1.0
                stall_count = 0

            # Check LP speed
            if len(lp_durations) > 200:
                avg_lp = np.mean(lp_durations[-200:])
                if avg_lp > 3.0:
                    print(f"    LP too slow ({avg_lp:.2f}s), stopping at iter {it+1}")
                    break

        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            avg_lp = np.mean(lp_durations[-50:]) * 1000 if len(lp_durations) >= 50 else 0
            print(f"    FISTA-LP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f} avg_lp={avg_lp:.1f}ms")

    return best_a, best_score


def mirror_descent_polish(a0, time_limit, n_steps=200000):
    """Mirror descent with adaptive step size and softmax-weighted gradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.005
    no_improve = 0
    patience = 100

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Softmax-weighted gradient
        temp = max_b * 1e-4
        scores = (b - max_b) / temp
        w = np.exp(scores)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(b)
            w[np.argmax(b)] = 1.0
        else:
            w /= w_sum

        grad = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w[k] * a[k - idx]

        # Mirror descent update
        log_update = -eta * grad
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            eta = min(eta * 1.005, 0.03)
        else:
            no_improve += 1
            if no_improve >= patience:
                eta *= 0.6
                no_improve = 0
                if eta < 1e-6:
                    break

        if (step + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD step {step+1} ({elapsed:.0f}s): score={best_score:.10f}, eta={eta:.6f}")

    return best_a, best_score


def lse_lbgfs_refine(a0, time_limit, deadline):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    T_stages = [5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]
    n_stages = len(T_stages)

    for idx, T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break

        iters = max(10, int(remaining * 25 / (n_stages - idx)))
        iters = min(iters, 200)

        try:
            res = minimize(
                fun=_lse_obj_func,
                x0=z,
                jac=_lse_grad_func,
                args=(n, T, deadline),
                method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-25,
                         'gtol': 1e-18, 'maxfun': iters * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def generate_cosine_init(n, freq=1.0, amplitude=0.5):
    """Generate cosine-modulated initialization."""
    x = np.arange(n)
    a = 1.0 + amplitude * np.cos(2.0 * np.pi * freq * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def generate_fehe_init(n):
    """Generate FEH-inspired initialization (oscillatory)."""
    x = np.arange(n)
    # Use a multi-frequency sinusoid
    f1 = 1.5
    f2 = 3.7
    a = 1.0 + 0.4 * np.cos(2.0 * np.pi * f1 * x / n) + 0.2 * np.cos(2.0 * np.pi * f2 * x / n)
    a = np.maximum(a, 0.001)
    s = a.sum()
    a = a / s
    return a

        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
