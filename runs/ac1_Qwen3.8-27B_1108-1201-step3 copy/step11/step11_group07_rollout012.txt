```python
def solve_lp_direction(a, M):
    """Solve LP: max sum(b) s.t. b >= 0, conv(a,b) <= M."""
    n = len(a)
    # conv(a,b)[k] = sum_i a[i]*b[k-i] <= M for k=0,...,2n-2
    # In matrix form: A @ b <= M * ones(2n-1)
    # where A[k, i] = a[i] if 0 <= k-i <= n-1, else 0
    #           = a[k-i] if max(0,k-n+1) <= i <= min(n-1,k), else 0
    
    # Build constraint matrix (sparse)
    from scipy.sparse import lil_matrix
    A_mat = lil_matrix((2*n-1, n))
    for k in range(2*n-1):
        for i in range(max(0, k-n+1), min(n, k+1)):
            A_mat[k, i] = a[i]
    A_mat = A_mat.tocsr()
    
    # Objective: maximize sum(b) = -minimize -sum(b)
    c = -np.ones(n)
    
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A_mat, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding.
The key idea: at each iteration, solve an LP to find a non-negative direction b
that keeps the cross-convolution conv(a,b) bounded by the current peak M.
Interpolating a -> (1-t)*a + t*b (rescaled) reduces the peak while preserving
non-negativity and the normalization.
Combined with LSE-smoothed gradient polishing and multi-length exploration.
"""

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def cross_convolve(a, b):
    """Cross-convolution a * b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

def evaluate_local(a):
    """Evaluate score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_lp_matrix(a):
    """Build the constraint matrix A where conv(a,b)[k] = (A@b)[k]."""
    n = len(a)
    A_mat = lil_matrix((2*n-1, n))
    for k in range(2*n-1):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            A_mat[k, i] = a[i]
    return csr_matrix(A_mat)

def solve_direction_lp(a, M, upper_bound=10.0):
    """Solve LP: max sum(b) s.t. b >= 0, b <= upper_bound, conv(a,b) <= M."""
    n = len(a)
    A_mat = csr_matrix((2*n-1, n))
    # Build sparse matrix
    rows, cols, vals = [], [], []
    for k in range(2*n-1):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            rows.append(k)
            cols.append(i)
            vals.append(a[i])
    A_mat = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    c = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, upper_bound)] * n
    
    res = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    
    if res.success and res.x is not None:
        return np.maximum(res.x, 0)
    return None

def line_search_conv(a, b, M, n_trials=20):

    C_ab = cross_convolve(a, b)
    C_bb = fast_convolve(b)

        if m < best_max:

            if m < best_max:

def lse_optimize(a0, time_limit, T0=10.0, T_min=0.01, n_stages=15):
    """LSE-smoothed gradient descent with temperature annealing."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    T_values = np.geomspace(T0, T_min, n_stages)
    
    for stage, T in enumerate(T_values):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        max_iters = max(100, int(remaining * 5))
        lr = 0.1 / (n * T)
        
        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                break
            if it % 20 == 0 and it > 0:
                lr *= 0.95
            
            a_new, sc = lse_gradient_step(a, T, lr)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
    
    return best_a, best_score
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based peak shaving.
Key innovation: At each iteration, solve an LP to find the optimal non-negative
direction b that keeps cross-convolution conv(a,b) <= M (current peak).
Interpolating a -> (1-t)*a + t*b (rescaled) is guaranteed to reduce the peak

import numpy as np

from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Cross-convolution a * b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def evaluate_local(a):
    """Evaluate score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def precompute_lp_pattern(n):
    """Precompute sparsity pattern for LP constraint matrix."""
    rows, cols = [], []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def build_lp_matrix(n, a, pattern):
    """Build LP constraint matrix using precomputed pattern. A[k,j] = a[k-j]."""
    rows, cols = pattern
    vals = a[cols]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_direction_lp(n, a, M, pattern, upper_bound=50.0):
    """Solve LP: max sum(b) s.t. b >= 0, b <= ub, conv(a,b) <= M elementwise."""
    A_mat = build_lp_matrix(n, a, pattern)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, upper_bound)] * n
    
    res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 5.0})
    
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def line_search_interp(a, b, n_trials=25):
    """Find best t in [0,1] minimizing max(conv((1-t)*a + t*b, (1-t)*a + t*b))."""
    C_aa = fast_convolve(a)
    C_ab = cross_convolve(a, b)
    C_bb = fast_convolve(b)
    
    best_t = 0.0
    best_val = C_aa.max()
    
    for t in np.linspace(0, 1, n_trials):
        C_t = (1 - t) ** 2 * C_aa + 2 * t * (1 - t) * C_ab + t ** 2 * C_bb
        v = C_t.max()
        if v < best_val:
            best_val = v
            best_t = t
    
    # Refine
    if 0.01 < best_t < 0.99:
        lo = max(0, best_t - 0.04)
        hi = min(1, best_t + 0.04)

            v = C_t.max()
            if v < best_val:
                best_val = v
                best_t = t
    
    return best_t, best_val


def lse_gradient_step(a, T, lr):
    """One gradient step on LSE-smoothed objective. Returns (new_a, score)."""
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    exp_c = np.exp(np.clip((c - c_max) / T, -50, 50))
    w = exp_c / exp_c.sum()
    
    # Gradient: grad[m] = 2 * sum_k w[k] * a[k-m] (valid k-m)
    # = 2 * cross_correlation(a, w) at lag m
    N = 1 << (2 * n - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    # cross-correlation: sum_j a[j] * w[j+m] = ifft(conj(A) * W)[m]... 
    # Actually: we want sum_k w[k] * a[k-m] for k in [0, 2n-2], k-m in [0, n-1]
    # Let j = k-m: sum_j w[j+m] * a[j] = correlation of w with a at lag m
    # = ifft(conj(FFT(a)) * FFT(w)) evaluated at position m
    R = irfft(np.conj(A) * W, N)
    grad = 2.0 * R[:n]
    
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a, float('inf')
    a_new = a_new / s
    return a_new, evaluate_local(a_new)

    # Asymmetric smooth
    a = np.exp(-2 * (x - 0.4) ** 2) + 0.3 * np.exp(-8 * (x - 0.7) ** 2)
    candidates.append(normalize(a))
    
    return candidates


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    return normalize(f_interp)


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    
    Phase 1: LP-based peak shaving (primary method)
    Phase 2: LSE gradient polishing
    Phase 3: LP round 2
    Phase 4: Fine LSE + LP polish
    Phase 5: Length exploration + final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 5.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 1200
        candidates = diverse_init(n_init, rng)
        scores = [evaluate_local(c) for c in candidates]
        best_idx = np.argmin(scores)
        best_seq = candidates[best_idx].copy()
        best_score = scores[best_idx]
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")
    
    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.8f}")
    
    # Precompute LP pattern for current best length
    pattern_best = precompute_lp_pattern(n_best)
    
    # === Phase 1: LP-based optimization (primary, 40% budget) ===
    print(f"\n=== Phase 1: LP Peak Shaving ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.40, 700.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, n={n_best}")
        a_lp, _ = lp_phase(best_seq, t_alloc, pattern_best, verbose=True)
        try_update(a_lp, "lp_1")
    
    # === Phase 2: LSE gradient polishing (20% budget) ===
    print(f"\n=== Phase 2: LSE Gradient Polish ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 350.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lse, _ = lse_optimize(best_seq, t_alloc, T0=5.0, T_min=0.003, n_stages=12)
        try_update(a_lse, "lse_1")
    
    # === Phase 3: LP round 2 on potentially improved sequence (20% budget) ===
    print(f"\n=== Phase 3: LP Round 2 ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        n_cur = len(best_seq)
        pattern_cur = precompute_lp_pattern(n_cur) if n_cur != n_best else pattern_best
        print(f"  Budget: {t_alloc:.0f}s, n={n_cur}")
        a_lp2, _ = lp_phase(best_seq, t_alloc, pattern_cur, verbose=True)
        try_update(a_lp2, "lp_2")
    
    # === Phase 4: Fine LSE polish (10% budget) ===
    print(f"\n=== Phase 4: Fine LSE ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 10:
        a_lse2, _ = lse_optimize(best_seq, t_alloc, T0=1.0, T_min=0.001, n_stages=10)
        try_update(a_lse2, "lse_fine")
    
    # === Phase 5: Final LP + LSE (remaining budget) ===
    print(f"\n=== Phase 5: Final Polish ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        n_cur = len(best_seq)
        pattern_cur = precompute_lp_pattern(n_cur) if n_cur != n_best else pattern_best
        t_lp = min(rem * 0.5, 60.0)
        t_lse = rem - t_lp - 5
        if t_lp > 5:
            a_fin1, _ = lp_phase(best_seq, t_lp, pattern_cur, verbose=False)
            try_update(a_fin1, "lp_final")
        if t_lse > 5:
            a_fin2, _ = lse_optimize(best_seq, max(5, t_lse), T0=0.5, T_min=0.001, n_stages=8)
            try_update(a_fin2, "lse_final")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
```
