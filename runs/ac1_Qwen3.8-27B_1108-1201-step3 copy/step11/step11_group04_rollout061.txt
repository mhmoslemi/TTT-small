Actually, each conv[k] = sum_j a[j]*a[k-j] is a quadratic form in a (specifically, a^T Q_k a where Q_k is a permutation-like matrix). So f(a) = max_k a^T Q_k a is the max of quadratic forms, which is convex (max of convex functions).

Wait, is a^T Q_k a convex? Q_k for the k-th convolution lag... let's see.

conv[k] = sum_j a[j]*a[k-j] = sum_j a[j]*a[k-j]

This can be written as a^T P_k a where (P_k)_{j,l} = 1 if l = k-j, else 0. But this is not a symmetric matrix in general, and a^T P_k a = sum_{j,l} a[j]*P_k[j,l]*a[l] = sum_j a[j]*a[k-j].

For convexity, we need P_k to be positive semi-definite. But P_k is a permutation matrix (or a submatrix thereof), and permutation matrices are not generally PSD.

Actually, a^T P_k a is not the same as the quadratic form a^T (P_k + P_k^T)/2 a unless P_k is symmetric. Let's check:
a^T P_k a = sum_{j,l} a[j] P_k[j,l] a[l]
a^T P_k^T a = sum_{j,l} a[j] P_k^T[j,l] a[l] = sum_{j,l} a[j] P_k[l,j] a[l] = sum_{l,j} a[l] P_k[j,l] a[j] = a^T P_k a

So a^T P_k a = a^T P_k^T a, meaning the quadratic form is the same for P_k and P_k^T. The effective quadratic form matrix is (P_k + P_k^T)/2, which IS symmetric.

Is (P_k + P_k^T)/2 PSD? P_k is a "shifted reversal" matrix. For k = n-1 (the middle), P_k is the reversal matrix (anti-identity), and (P_k + P_k^T)/2 = P_k = reversal matrix. The eigenvalues of the reversal matrix are ±1, so it's not PSD.

So conv[k] is NOT a convex function of a in general! The objective f(a) = max_k conv[k] is NOT convex.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Over-relaxed spectral flattening (Gerchberg-Saxton with adaptive targets)
2. Nesterov-accelerated LP alternating with periodic restarts
3. Ultra-fine LogSumExp L-BFGS (30+ temperature stages down to 0.0003)
4. Backtracking peak-shaving mirror descent

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.GUROBI]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue

    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None, None
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(max(0, M_var.value))


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def spectral_flatten_v2(a, n_iters=100, blend=0.4, overrelax=1.2, deadline=None):
    """
    Improved Gerchberg-Saxton with over-relaxation and adaptive target.
    Uses geometric mean target and optional over-relaxation factor.
    """
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)

        mags = np.abs(F)
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend * overrelax) * mags[k] + blend * overrelax * target
            new_mag = max(0, new_mag)
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])

        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new

    return best_a


def spectral_flatten_flat(a, n_iters=80, deadline=None):
    """Spectral flattening: target exactly 1 for all non-DC frequencies."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)

        mags = np.abs(F)
        dc_val = mags[0]
        non_dc_mags = mags[1:L//2+1]
        if non_dc_mags.min() < 1e-30:
            scale = dc_val / max(non_dc_mags.max(), 1e-30)
        else:
            scale = dc_val / np.mean(non_dc_mags)
            scale = min(scale, 10.0)

        phases = np.angle(F)
        F_new = np.zeros(L, dtype=complex)
        F_new[0] = dc_val
        for k in range(1, L // 2 + 1):
            new_mag = mags[k] * scale
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])

        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def lbgfs_objective_and_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lbgfs_refine(a0, time_limit, T_stages, deadline=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit
    n_stages = len(T_stages)

    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 2:
            break

        stages_left = n_stages - stage_i
        iters = max(20, min(200, int(remaining * 30 / max(1, stages_left))))

        def func(z_in):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = lbgfs_objective_and_grad(z_in, n, stage_T)
            return obj

        def jac(z_in):
            if time.monotonic() > deadline:
                return np.zeros_like(z_in)
            _, g = lbgfs_objective_and_grad(z_in, n, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-30,
                         'gtol': 1e-20, 'maxfun': iters * 1000, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def peak_shave_bt(a0, time_limit, n_steps=200000):
    """Backtracking peak shaving with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return None, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.03
    scale = 1.0
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        threshold = max_b * (1.0 - 2e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Backtracking: try current step, halve if no improvement
        eta_try = eta * scale
        improved = False
        for bt in range(15):
            log_scale = -eta_try * g
            log_scale -= log_scale.max()
            a_new = a * np.exp(log_scale)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s_new

            score = 2.0 * n * fast_convolve(a_new).max()
            if score < best_score - 1e-15:
                best_score = score
                best_a = a_new.copy()
                a = a_new
                improved = True
                no_improve = 0
                break
            eta_try *= 0.5

        if not improved:
            no_improve += 1
            if no_improve >= 20:
                scale *= 0.6
                no_improve = 0
                if scale < 0.003:
                    break

    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=1000000):
    """Nesterov accelerated LP alternating with stall detection and restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)

        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(99999)
    lp_times = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)
        lp_dt = time.monotonic() - lp_t0
        lp_times.append(lp_dt)

        if b_new is None:
            continue

        if len(lp_times) >= 30:
            avg_lp = np.mean(lp_times[-30:])
            if avg_lp > 8.0 and len(lp_times) > 100:
                break

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-12:
                stall_count = 0
            else:
                stall_count += 1

                for freq in [rng.integers(1, 10) for _ in range(3)]:

                        best_score = sc_pert
                        best_a = y_pert.copy()
                stall_count = 0

            if (it + 1) % 2000 == 0:

                print(f"    NL iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} avg_lp={np.mean(lp_times[-30:]):.3f}s")

    return best_a, best_score


def simulated_annealing(a0, time_limit, n_iterations=50000):
    """Simulated annealing in log-domain."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a)
    best_z = z.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(77777)

    T_start = 0.002
    T_end = 0.00005
    current_T = T_start
    cooling = (T_end / T_start) ** (1.0 / max(1, n_iterations))

    x = np.linspace(0, 2 * np.pi, n)
    freqs = list(range(1, 15))

    for it in range(n_iterations):
        if time.monotonic() - t0 > time_limit:
            break

        current_T *= cooling

        # Perturb: add structured noise in log domain
        freq = rng.choice(freqs)
        phase = rng.uniform(0, 2 * np.pi)
        amplitude = 0.05 * (1 + 2 * current_T * 500)
        z_pert = z + amplitude * np.cos(freq * x + phase) * rng.standard_normal(n) * 0.1
        a_pert = softmax_from_z(z_pert)

        score_pert = 2.0 * n * fast_convolve(a_pert).max()

        if score_pert < best_score:
            best_score = score_pert
            best_a = a_pert.copy()
            best_z = z_pert.copy()
            z = z_pert
        elif np.log(rng.random() + 1e-30) < -(score_pert - best_score) / max(current_T, 1e-10):
            z = z_pert
            a = a_pert

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        center = n // 2
        t = (np.arange(n) - center) / center
        a = np.sinc(t)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = 1.0 - np.abs(2 * x - 1)
    elif choice == 8:
        # Raised cosine window
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    else:
        # Blackman-like
        a = 0.42 + 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def perturbed_sequence(a, rng, strength=0.03):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 12)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def full_optimize(a0, time_limit, solver, deadline):
    """Run a full optimization pipeline on a sequence."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    def try_update(a_cand, label):
        nonlocal best_a, best_score
        if a_cand is None:
            return
        a_cand = np.maximum(np.asarray(a_cand, dtype=np.float64), 0)
        sc = 2.0 * n * fast_convolve(a_cand).max()
        if np.isfinite(sc) and sc * (len(a_cand) * 2.0) / (a_cand.sum() ** 2) / (a_cand.sum() ** 2) < 0:
            return
        # Just use evaluate_local for correctness
        sc2 = evaluate_local(a_cand)
        if np.isfinite(sc2) and sc2 < best_score:
            best_score = sc2
            best_a = a_cand.copy()
            print(f"      [{label}] new best: {best_score:.8f}")

    rem = deadline - time.monotonic()
    if rem < 5:
        return best_a, best_score

    # Spectral flattening
    t_spec = min(rem * 0.1, 30.0)
    if t_spec > 3:
        a_sf = spectral_flatten_v2(a, n_iters=50, blend=0.4, overrelax=1.1,
                                   deadline=time.monotonic() + t_spec)
        try_update(a_sf, "spectral")
        a_sf2 = spectral_flatten_flat(a, n_iters=40, deadline=time.monotonic() + min(rem, 15))
        try_update(a_sf2, "spectral_flat")
        a = best_a

    # LP alternating
    rem = deadline - time.monotonic()
    t_lp = min(rem * 0.35, 200.0)
    if t_lp > 10:
        a_lp, _ = nesterov_lp_refine(a, t_lp, solver)
        try_update(a_lp, "lp")
        a = best_a

    # L-BFGS LogSumExp
    rem = deadline - time.monotonic()
    t_lb = min(rem * 0.4, 300.0)
    if t_lb > 10:
        T_stages = [5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.18,
                    0.12, 0.08, 0.05, 0.035, 0.025, 0.018, 0.012, 0.008,
                    0.005, 0.003, 0.002, 0.0012, 0.0007, 0.0004, 0.0002,
                    0.0001, 0.00006, 0.00003]
        a_lb = lbgfs_refine(a, t_lb, T_stages, deadline=time.monotonic() + t_lb)
        try_update(a_lb, "lbgfs")
        a = best_a

    # Peak shaving
    rem = deadline - time.monotonic()
    t_ps = min(rem * 0.15, 60.0)
    if t_ps > 5:
        a_ps, _ = peak_shave_bt(a, t_ps, n_steps=100000)
        try_update(a_ps, "peak")
        a = best_a

    # Second LP round
    rem = deadline - time.monotonic()
    t_lp2 = min(rem * 0.2, 80.0)
    if t_lp2 > 10:
        a_lp2, _ = nesterov_lp_refine(a, t_lp2, solver, n_max_iters=200000)
        try_update(a_lp2, "lp2")
        a = best_a

    # Final L-BFGS polish
    rem = deadline - time.monotonic()
    t_fin = min(rem * 0.3, 40.0)
    if t_fin > 5:
        T_fin = [2.0, 1.0, 0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005,
                 0.002, 0.001, 0.0005, 0.0002, 0.0001]
        a_fin = lbgfs_refine(a, t_fin, T_fin, deadline=time.monotonic() + t_fin)
        try_update(a_fin, "final_lb")

    return best_a, best_score
