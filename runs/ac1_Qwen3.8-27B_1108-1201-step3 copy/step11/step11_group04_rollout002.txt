"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

2. Top-K LogSumExp L-BFGS polishing with ultra-fine temperature

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(x):
    """Build (2n-1) x n matrix T such that T @ b = convolve(x, b)."""
    n = len(x)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = x
    return T


def get_lp_solver():
    """Find the fastest working LP solver."""
    import time as _t
    results = []
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.CLARABEL, cp.CBC, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(100, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            t0 = _t.monotonic()
            prob.solve(solver=solver_name)
            elapsed = _t.monotonic() - t0
            if prob.status in ('optimal', 'optimal_inaccurate'):
                results.append((elapsed, solver_name))
        except Exception:
            continue
    if not results:
        return cp.SCS
    results.sort()
    return results[0][1]


def solve_lp_best_response(x_fixed, solver, fallback_solvers=None):
    """Given fixed x (non-negative, sum=1), find y minimizing max_k conv(x,y)[k]."""
    n = len(x_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(x_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    if fallback_solvers is None:
        fallback_solvers = [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]
    solvers_to_try = [solver] + [s for s in fallback_solvers if s != solver]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    continue
                b_val = b_val / s_sum
                return b_val, float(M_var.value)
        except Exception:
            continue
    return None, None


def softmax_from_z(z):
    """Convert log-domain z to probability vector (sums to 1)."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lse_obj_grad_topk(z, n, T_temp, K):
    """
    Compute top-K LogSumExp objective and gradient w.r.t. z.
    Only top K peaks contribute.
    Returns (obj, grad_z).
    """
    a = softmax_from_z(z)
    b = fast_convolve(a)
    # Get top K indices
    if K < len(b):
        top_k = np.argpartition(b, -K)[-K:]
    else:
        top_k = np.arange(len(b))
    b_top = b[top_k]
    max_bt = b_top.max()
    exp_terms = np.exp((b_top - max_bt) / T_temp)
    lse = max_bt + T_temp * np.log(exp_terms.sum())
    obj = lse  # We'll multiply by 2*n externally

    # Gradient of lse w.r.t. a
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for i, k in enumerate(top_k):
        if w[i] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[i] * a[k - idx]

    # Chain rule: grad_z = a * (grad_a - sum(grad_a * a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_full_obj(z, n, T_temp, K, deadline):
    """Full objective for L-BFGS (includes 2n factor)."""
    if time.monotonic() > deadline:
        return 1e15
    obj, _ = lse_obj_grad_topk(z, n, T_temp, K)
    return 2.0 * n * obj


def lbgfs_full_grad(z, n, T_temp, K, deadline):
    """Full gradient for L-BFGS."""
    if time.monotonic() > deadline:
        return np.zeros_like(z)
    _, g = lse_obj_grad_topk(z, n, T_temp, K)
    return 2.0 * n * g


def lbgfs_polish(a0, time_limit, T_list, K, maxiter_total, deadline):
    """
    L-BFGS polishing with top-K LogSumExp and temperature schedule.
    """
    n = len(a0)
    z = z_from_a(a0).copy()
    iters_per_stage = max(30, maxiter_total // len(T_list))

    for stage_T in T_list:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 2:
            break

        def func(z_in):
            return lbgfs_full_obj(z_in, n, stage_T, K, deadline)

        def jac(z_in):
            return lbgfs_full_grad(z_in, n, stage_T, K, deadline)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-30,
                         'gtol': 1e-20, 'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def lp_alternating(a0, time_limit, solver, deadline, print_interval=500):
    """
    Nesterov-accelerated LP alternating method.
    Returns best (a, score) found.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # Nesterov: maintain momentum variable y
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(999)

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # Solve LP: best response to y
        b_new, M_val = solve_lp_best_response(y, solver)
        if b_new is None:
            # Fallback: use a as best response approximation
            b_new = a.copy()

        # Nesterov update
        alpha = 2.0 / (it + 2)
        # Update y (dual variable)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        # Update a (primal)
        t_nesterov = (it + 1) / (it + 1 + alpha * (it + 2))
        a_new = (it + 1) * a + alpha * (it + 2) * b_new
        a_new = a_new / (it + 1 + alpha * (it + 2))
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a_new / sa

        # Track best
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Stall detection and perturbation
        if it % 200 == 0:
            if best_score < prev_best - 1e-12:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 15:
                # Perturb y to escape
                x_lin = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 10)
                noise = 0.02 * np.cos(freq * x_lin + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                stall_count = 0

            if it % print_interval == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP Alt iter={it} ({elapsed:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, n_steps=500000):
    """
    Nesterov-accelerated subgradient method targeting the max peak.
    Uses the exact subgradient of max_k conv(a,a)_k.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()  # Nesterov momentum
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    gamma0 = 0.5  # Initial step size parameter
    L_bound = 4.0 * n  # Lipschitz constant estimate (rough)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute subgradient at y
        b_y = fast_convolve(y)
        max_by = b_y.max()
        # Active set: peaks within small epsilon of max
        eps = max_by * 1e-6 + 1e-15
        active = np.where(b_y >= max_by - eps)[0]
        # Uniform weights on active set
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * y[k - idx]
        g /= len(active)

        # Nesterov step size
        t_k = step + 1
        gamma_t = 2.0 * gamma0 / (L_bound * t_k)

        # Update y (lookahead)
        tau = (1.0 + np.sqrt(1.0 + 4.0 * (t_k - 1) ** 2)) / 2.0
        y_new = a - gamma_t * g
        y_new = np.maximum(y_new, 0)
        s_new = y_new.sum()
        if s_new < 1e-15:
            y_new = a.copy()
        else:
            y_new = y_new / s_new

        # Update a
        a = (t_k * a + (tau - 1.0) * y_new) / tau
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y_new.copy()
        else:
            a = a / sa

        y = y_new

        # Track best (use averaged iterate)
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (step + 1) % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad step={step+1} ({elapsed:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def spectral_flatten(a, n_iters=50, blend=0.4, deadline=None):
    """Gerchberg-Saxton style spectral flattening."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)

        mags = np.abs(F)
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])

        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

    return a


def evaluate_local(a):
    """Evaluate sequence score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        t_fin = min(rem * 0.5, 40)
        print(f"  Budget: {t_fin:.0f}s")
        # Ultra-fine: very small T, many iterations
        T_ultra = [1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02,
                   0.01, 0.008, 0.005, 0.004, 0.003, 0.002, 0.0015, 0.001]
        a_fin = lbgfs_polish(best_seq, t_fin, T_ultra, 15,
                             maxiter_total=50000,
                             deadline=time.monotonic() + t_fin)
        local_update(a_fin, "final_ultra")

    # === Phase 7: Final LP micro-round ===
    rem = deadline - time.monotonic()
    if rem > 10:
        a_flp, _ = lp_alternating(best_seq, min(rem, 15), solver,
                                  deadline=time.monotonic() + min(rem, 15))
        local_update(a_flp, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)}) at {elapsed:.1f}s")

        # Verify with evaluate_sequence
        test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            eval_score = evaluate_sequence(test_seq)
            print(f"Verified with evaluate_sequence: {eval_score:.10f}")
        except Exception as e:
            print(f"Verification error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
