```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response alternating with exponential weights (mirror descent)
2. Adam optimization in log-space with adaptive temperature
3. Multi-phase strategy with careful time management
4. Nesterov-style extrapolation for LP alternating
5. Multi-length exploration (n=1200 primary)

Key improvements over previous:
- More efficient gradient computation via FFT
- Better time allocation (more LP alternating)
- Exponential weights for LP update (avoids negative values)
- Adaptive step size based on duality gap
- Cleaner code structure
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_two(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def evaluate_local(a):
    """Evaluate score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def build_T_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver):
    """
    Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]
    subject to b >= 0, sum(b) = 1.
    Returns (b_value, M_value) or (None, None) on failure.
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    T = build_T_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers_to_try = [solver]
    for s in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        if s not in solvers_to_try:
            solvers_to_try.append(s)
    
    for s in solvers_to_try:
        try:
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                b_val = np.asarray(b_var.value, dtype=np.float64).copy()
                b_val = np.maximum(b_val, 0)
                b_val[np.isnan(b_val)] = 0
                s_val = b_val.sum()
                if s_val < 1e-15:
                    continue
                b_val = b_val / s_val
                M_val = float(max(0.0, M_var.value))
                return b_val, M_val
        except Exception:
            continue
    
    return None, None


def compute_gradient(a, b_conv):
    """
    Compute a subgradient of max_k conv(a,a)_k at a.
    Uses the active set (where b_conv is near maximum).
    
    Parameters:
    - a: sequence of length n (non-negative)
    - b_conv: auto-convolution of a, length 2n-1
    
    Returns: gradient vector of length n
    """
    n = len(a)
    max_b = b_conv.max()
    
    # Use a soft threshold on the active set
    threshold = max_b * 0.999
    active = np.where(b_conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(b_conv)])
    
    # Weight uniformly over active set
    w = np.zeros(2 * n - 1)
    w[active] = 1.0 / len(active)
    
    # Compute gradient: grad[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]
    # Use FFT for efficiency
    w_rev = w[::-1]  # length 2n-1
    conv_size = (2 * n - 1) + n - 1  # = 3n-2
    fft_size = 1 << (conv_size - 1).bit_length()
    W_rev = rfft(w_rev, fft_size)
    A = rfft(a, fft_size)
    conv_result = irfft(W_rev * A, fft_size)[:conv_size]
    # grad[j] = 2 * conv_result[2n-2-j], for j=0,...,n-1
    grad = 2.0 * conv_result[n-1:2*n-1][::-1]
    
    return grad


def compute_gradient_lse(a, b_conv, temperature):
    """
    Compute gradient using LogSumExp approximation.
    Uses soft weights based on temperature.
    """
    n = len(a)
    max_b = b_conv.max()
    exp_terms = np.exp((b_conv - max_b) / temperature)
    w = exp_terms / exp_terms.sum()
    
    # Same FFT-based gradient computation
    w_rev = w[::-1]
    conv_size = (2 * n - 1) + n - 1
    fft_size = 1 << (conv_size - 1).bit_length()
    W_rev = rfft(w_rev, fft_size)
    A = rfft(a, fft_size)
    conv_result = irfft(W_rev * A, fft_size)[:conv_size]
    grad = 2.0 * conv_result[n-1:2*n-1][::-1]
    
    return grad


def lp_alternating_exp_weights(a0, time_limit, solver, eta_init=0.1):
    """
    LP alternating with exponential weights (mirror descent).
    At each step:
    1. Solve LP to get best response b*
    2. Update a using exponential weights in direction b*
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    # Nesterov momentum
    y = a.copy()
    prev_score = best_score
    stall = 0
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Solve LP: find b* minimizing max_k conv(y, b)_k
        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            break
        
        # Current score of y
        score_y = evaluate_local(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        # Score of a
        score_a = evaluate_local(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        # Nesterov dual averaging update
        alpha = 2.0 / (it + 2)
        
        # Update y (the "lookahead" variable)
        y = (1.0 - alpha) * y + alpha * b_star
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        # Update a (the "average" variable)
        a = y.copy()  # Simple: set a = y (no momentum)
        # Or use Nesterov: a = ((it+1)*a + alpha*(it+2)*b_star) / (it+1 + alpha*(it+2))
        # Let's use the exponential weights version:
        # a_new[j] = a[j] * exp(eta * (b_star[j] - a[j])) / normalization
        # This is more like mirror descent
        
        # Actually, let's use a blend: linear interpolation with shrinking step
        t_blend = min(1.0, alpha * 2)
        a_new = (1.0 - t_blend) * a + t_blend * b_star
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a = a.copy()
        else:
            a = a_new / sa
        
        # Check for stall
        if it % 50 == 0:
            if best_score < prev_score - 1e-10:
                stall = 0
                prev_score = best_score
            else:
                stall += 1
            
            if stall >= 10:
                # Perturb to escape
                rng = np.random.default_rng(it)
                noise = 0.02 * rng.standard_normal(n)
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                stall = 0
            
            if it % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter={it} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def adam_logspace(a0, time_limit, T_stages, lr_init=0.005, top_k_frac=0.1):
    """
    Adam optimization in log-space with temperature annealing.
    """
    n = len(a0)
    z = z_from_a(np.asarray(a0, dtype=np.float64)).copy()
    t0 = time.monotonic()
    
    best_a = a0.copy()
    best_score = evaluate_local(a0)
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step_count = 0
    
    for stage_idx, T_stage in enumerate(T_stages):
        if time.monotonic() > t0 + time_limit:
            break
        
        steps_in_stage = max(1000, int(time_limit * 8000 / len(T_stages)))
        
        for step in range(steps_in_stage):
            if time.monotonic() > t0 + time_limit:
                break
            
            a = softmax_from_z(z)
            b = fast_convolve(a)
            max_b = b.max()
            score = 2.0 * n * max_b
            step_count += 1
            
            if score < best_score:
                best_score = score
                best_a = a.copy()
            
            # Compute gradient
            if T_stage < 0.01:
                threshold = max_b * (1.0 - 1e-4)
                active = np.where(b >= threshold)[0]
                if len(active) == 0:
                    active = np.array([np.argmax(b)])
                w = np.zeros(len(b))
                w[active] = 1.0 / len(active)
            else:
                if top_k_frac < 1.0:
                    top_k = max(5, int(top_k_frac * len(b)))
                    idx = np.argpartition(b, -top_k)[-top_k:]
                    b_top = b[idx]
                    max_top = b_top.max()
                    exp_top = np.exp((b_top - max_top) / T_stage)
                    w_top = exp_top / exp_top.sum()
                    w = np.zeros(len(b))
                    w[idx] = w_top
                else:
                    exp_all = np.exp((b - max_b) / T_stage)
                    w = exp_all / exp_all.sum()
            
            # Gradient via FFT
            w_rev = w[::-1]
            conv_size = 3 * n - 2
            fft_size = 1 << (conv_size - 1).bit_length()
            W_rev = rfft(w_rev, fft_size)
            A = rfft(a, fft_size)
            conv_result = irfft(W_rev * A, fft_size)[:conv_size]
            g_a = 2.0 * conv_result[n-1:2*n-1][::-1]
            
            # Chain rule to z-space
            S = np.dot(g_a, a)
            grad_z = a * (g_a - S)
            
            # Adam update
            lr = lr_init / (1.0 + 0.1 * step / steps_in_stage)
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** (step + 1))
            v_hat = v / (1 - beta2 ** (step + 1))
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            z -= z.mean()  # Centering
            
            if (step + 1) % 10000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Adam stage {stage_idx} (T={T_stage:.3f}) step {step+1} ({elapsed:.0f}s): best_score={best_score:.8f}")
    
    return best_a, best_score


def multiplicative_subgradient(a0, time_limit, n_steps=100000):
    """Mirror descent (multiplicative weights) targeting active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    eta = 0.05
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Active set
        threshold = max_b * (1.0 - max(1e-4, 1e-3 / (1 + step / 500)))
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Gradient
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)
        
        # Compute gradient
        w_rev = w[::-1]
        conv_size = 3 * n - 2
        fft_size = 1 << (conv_size - 1).bit_length()
        W_rev = rfft(w_rev, fft_size)
        A = rfft(a, fft_size)
        conv_result = irfft(W_rev * A, fft_size)[:conv_size]
        g_a = 2.0 * conv_result[n-1:2*n-1][::-1]
        
        # Multiplicative update
        log_update = -eta * g_a
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        score = evaluate_local(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 30:
            eta *= 0.8
            no_improve = 0
            if eta < 0.003:
                break
        
        if step % 20000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    MultiSubGrad {elapsed:.0f}s: score={best_score:.8f} eta={eta:.4f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
```

The dual LP:

```
max sum_k y_k
s.t. sum_k y_k * a[k-i] <= 1 for i=0,...,n-1
     y_k >= 0
```

This has 2n-1 variables (y_0,...,y_{2n-2}) and n constraints. For n=1200, that's 2399 variables and 1200 constraints. The constraint matrix is also banded Toeplitz (with a as the generating vector).

The dual objective value is a lower bound on `max_k conv(a,a)_k`. If the dual value is close to `max_k conv(a,a)_k`, we're near optimal.

```python
def solve_lp_dual(a_fixed, solver):
    """
    Solve the dual LP: max sum(y) s.t. T^T y <= 1, y >= 0
    where T^T has entries T^T[i,k] = a[k-i] (if valid).
    This gives a lower bound on max_k conv(a,a)_k.
    """
    n = len(a_fixed)
    m = 2 * n - 1  # number of y variables
    
    y_var = cp.Variable(m, nonneg=True)
    
    # Constraint: sum_k y_k * a[k-i] <= 1 for each i
    # The matrix is T^T where T[i,k] = a[k-i] (if 0 <= k-i < n)
    # So T^T[k,i] = a[k-i]
    # The constraint is: T^T @ y <= 1
    # But T^T is n x (2n-1), and T is (2n-1) x n
    
    # Build T^T directly
    Tt = np.zeros((n, m))
    for k in range(m):
        for i in range(max(0, k-n+1), min(n, k+1)):
            Tt[i, k] = a_fixed[k-i]
    
    constraints = [Tt @ y_var <= 1]
    prob = cp.Problem(cp.Maximize(cp.sum(y_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            return float(prob.value)
    except Exception:
        pass
    return None
```
