"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based Frank-Wolfe conditional gradients (key innovation)
2. LSE-smoothed gradient descent with adaptive temperature
3. Spectral flatness optimization
4. Population-based diversity and interpolation

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def lse_obj_grad(a, T):
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    lse = c_max + T * np.log(exp_c.sum())
    obj = 2.0 * n * lse
    w = exp_c / exp_c.sum()
    N = 1 << (len(w) + n - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    R = irfft(W * np.conj(A), N)
    b = R[:n]
    g = 2.0 * b
    idx = 2 * np.arange(n)
    valid = idx < (2 * n - 1)
    g[valid] -= a[valid] * w[idx[valid]]
    return obj, g

def lse_descent(a0, time_limit, T_stages, maxiter_per_stage=5000):
    from scipy.optimize import minimize
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func(z_in):
            z_shifted = z_in - z_in.max()
            exp_z = np.exp(z_shifted)
            a = exp_z / exp_z.sum()
            obj, grad_a = lse_obj_grad(a, stage_T)
            return obj
        
        def jac(z_in):
            z_shifted = z_in - z_in.max()
            exp_z = np.exp(z_shifted)
            a = exp_z / exp_z.sum()
            obj, grad_a = lse_obj_grad(a, stage_T)
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        
        iters = min(maxiter_per_stage, max(10, int(remaining * 20)))
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': iters, 'ftol': 1e-16,
                                   'gtol': 1e-14, 'maxfun': iters * 100, 'disp': False})
            z = res.x
        except Exception:
            pass
    
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()

def diverse_init(n, rng):
    candidates = []
    x = np.linspace(0, 1, n)
    
    a = np.exp(-8 * (x - 0.5) ** 2)
    candidates.append(normalize(a))
    
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    candidates.append(normalize(a))
    
    a = np.sin(np.pi * x) ** 2
    candidates.append(normalize(a))
    
    a = (x * (1 - x)) ** 0.3
    candidates.append(normalize(a))
    
    # Exponential taper
    a = np.exp(-3 * x) * np.exp(-3 * (1-x))
    candidates.append(normalize(a))
    
    # Power law taper
    a = (x * (1-x)) ** 1.5
    candidates.append(normalize(a))
    
    # Multi-scale
    a = np.ones(n)
    taper = 0.2
    for i in range(n):
        d = min(i, n-1-i) / n
        a[i] = 1.0 if d > taper else (d / taper) ** 0.5
    candidates.append(normalize(a))
    
    # Random smooth
    coeffs = rng.normal(0, 1, 15)
    a = np.zeros(n)
    for i, c in enumerate(coeffs):
        a += c * np.cos(2 * np.pi * (i + 1) * x + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a + 1.0, 0)
    candidates.append(normalize(a))
    
    return candidates

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    return normalize(f_interp)

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LSE-smoothed gradient descent with Nesterov acceleration
2. Spectral flatness refinement (iterative PSD equalization)
3. LP-based conditional gradient steps for escaping local minima
4. Multi-scale initialization and final polish

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate the objective locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):

    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based conditional gradient (Frank-Wolfe) for global direction finding
2. LSE-smoothed gradient descent with Nesterov acceleration and adaptive annealing
3. Iterative PSD equalization (spectral flatness)
4. Multi-start with diverse initializations and length exploration
5. Active-set subgradient for final polish

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def lse_obj_grad(a, T):
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    lse = c_max + T * np.log(exp_c.sum())
    obj = 2.0 * n * lse
    w = exp_c / exp_c.sum()
    fft_size = 1 << (len(w) + n - 1 - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    R = irfft(W * np.conj(A), fft_size)
    g = 2.0 * R[:n]
    return obj, g


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lbgfs_optimize(a0, time_limit, T_stages, maxiter_per_stage=3000):
    from scipy.optimize import minimize
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_in):
            z_shifted = z_in - z_in.max()
            exp_z = np.exp(z_shifted)
            a = exp_z / exp_z.sum()
            obj, _ = lse_obj_grad(a, stage_T)
            return obj

        def jac(z_in):
            z_shifted = z_in - z_in.max()
            exp_z = np.exp(z_shifted)
            a = exp_z / exp_z.sum()
            _, g = lse_obj_grad(a, stage_T)
            S = np.dot(g, a)
            return a * (g - S)

        iters = min(maxiter_per_stage, max(10, int(remaining * 25)))
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': iters, 'ftol': 1e-18,
                                   'gtol': 1e-16, 'maxfun': iters * 200, 'disp': False})
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def lse_descent_direct(a0, time_limit, T_stages=None):
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [15, 10, 7, 5, 3, 2, 1.5, 1, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.007, 0.004, 0.002]

    lr = 0.008
    beta = 0.0
    v = np.zeros(n)

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break

        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit:

            _, g = lse_obj_grad(a, T)
            g = g - g.mean()
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-12:
                break

            v = beta * v + lr * (g / g_norm)
            a_new = a + v
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                v = np.zeros(n)

            a_new = a_new / s

def spectral_refine(a0, time_limit, n_iters=5000):

def lp_frank_wolfe(a0, time_limit, n_iters=100, t_init=0.08):
    import cvxpy as cp
    from scipy.sparse import csr_matrix

    n = len(a0)
    a = normalize(a0.copy())
    sum_target = np.sqrt(2.0 * n)
    a_work = a * sum_target

    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    t = t_init

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - t0 > time_limit - 5:
            break

        c = fast_convolve(a_work)
        M = c.max()
        if M < 1e-10:
            break

        # Build sparse constraint matrix
        # Constraint k: sum_j a_work[k-j] * g[j] <= M for valid k-j
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            for j in range(j_lo, j_hi + 1):
                i = k - j
                if a_work[i] > 1e-15:
                    rows.append(k)
                    cols.append(j)
                    vals.append(a_work[i])

        A_mat = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

        g_var = cp.Variable(n)
        objective = cp.Maximize(cp.sum(g_var))
        constraints = [A_mat @ g_var <= M, g_var >= 0]

        prob = cp.Problem(objective, constraints)
        g_val = None
        for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver, verbose=False)
                g_val = g_var.value
                break
            except Exception:
                continue

        if g_val is None:
            break

        g_val = np.maximum(g_val, 0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            break
        g_val = g_val * (sum_target / g_sum)

        # Line search
        best_t = 0
        best_sc = best_score
        for trial_t in [t, t * 0.5, t * 0.25, t * 0.1]:
            a_new = (1 - trial_t) * a_work + trial_t * g_val
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s * sum_target
            sc = evaluate_local(a_new)
            if sc < best_sc:
                best_sc = sc
                best_t = trial_t

        if best_t > 0:
            a_work = (1 - best_t) * a_work + best_t * g_val
            a_work = np.maximum(a_work, 0)
            s = a_work.sum()
            a_work = a_work / s * sum_target
            if best_sc < best_score:
                best_score = best_sc
                best_a = normalize(a_work)
                t = min(0.3, t * 1.05)
            else:
                t *= 0.7
        else:
            t *= 0.7

        if it % 20 == 0 and it > 0:
            print(f"    LP-FW it={it}: score={best_score:.8f} t={t:.4f}")

    return best_a

def diverse_init(n, rng):
    candidates = []
    x = np.linspace(0, 1, n)

    a = np.exp(-6 * (x - 0.5) ** 2)
    candidates.append(normalize(a))

    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    candidates.append(normalize(a))

    a = np.sin(np.pi * x) ** 2
    candidates.append(normalize(a))

    a = (x * (1 - x)) ** 0.4
    candidates.append(normalize(a))

    # Smooth flat top
    edge = 0.15
    a = np.ones(n)
    for i in range(n):
        d = min(x[i], 1 - x[i])
        if d < edge:
            a[i] = 0.3 + 0.7 * (d / edge) ** 2
    candidates.append(normalize(a))

    # Tukey-like
    r = 0.25
    a = np.ones(n)
    for i in range(n):
        d = min(x[i], 1 - x[i])
        if d < r / 2:
            a[i] = 0.5 * (1 + np.cos(np.pi * d / (r / 2)))
    candidates.append(normalize(a))

    # Exponential taper
    a = np.exp(-4 * x) * np.exp(-4 * (1 - x))
    candidates.append(normalize(a))

    # Power law
    a = (x * (1 - x)) ** 1.0
    candidates.append(normalize(a))

    # Random smooth
    coeffs = rng.normal(0, 1, 20)
    a = np.zeros(n)
    for i, c in enumerate(coeffs):
        a += c * np.cos(2 * np.pi * (i + 1) * x + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a + 1.0, 0.01)
    candidates.append(normalize(a))

    return candidates


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    return normalize(f_interp)


def perturb_sequence(a, rng, strength=0.02):
    n = len(a)
    method = rng.integers(0, 4)
    x = np.linspace(0, 2 * np.pi, n)
    if method == 0:
        a_new = a + rng.normal(0, strength, n)
    elif method == 1:
        freq = rng.integers(2, 12)
        phase = rng.uniform(0, 2 * np.pi)
        a_new = a + strength * np.cos(freq * x + phase)
    elif method == 2:
        a_new = a.copy()
        bs = rng.integers(5, 30)
        i = rng.integers(0, n - bs)
        j = rng.integers(0, n - bs)
        if abs(i - j) > bs:
            a_new[i:i+bs], a_new[j:j+bs] = a_new[j:j+bs].copy(), a_new[i:i+bs].copy()
    else:
        a_new = a + rng.normal(0, strength, n)
    return normalize(a_new)


def population_search(a_init, time_limit, pop_size=6, rng=None):
    if rng is None:
        rng = np.random.default_rng(789)

    n = len(a_init)
    population = [normalize(a_init.copy())]
    for i in range(pop_size - 1):
        population.append(perturb_sequence(a_init, rng, 0.02 + 0.01 * (i % 3)))

    scores = [evaluate_local(p) for p in population]
    best_idx = np.argmin(scores)
    best_a = population[best_idx].copy()
    best_score = scores[best_idx]

    t0 = time.monotonic()
    gen = 0
    strength = 0.02

    while time.monotonic() - t0 < time_limit:
        gen += 1
        if time.monotonic() - t0 > time_limit - 10:
            break

        order = np.argsort(scores)
        population = [population[i] for i in order]
        scores = [scores[i] for i in order]

        if scores[0] < best_score:
            best_score = scores[0]
            best_a = population[0].copy()

        keep = max(2, pop_size // 2)
        elites = population[:keep]
        new_pop = list(elites)

        for i in range(pop_size - len(new_pop)):
            parent = elites[rng.integers(0, keep)]
            child = perturb_sequence(parent, rng, strength)
            new_pop.append(child)

        if len(new_pop) < pop_size and len(elites) >= 2:
            p1, p2 = elites[0], elites[1]
            t = rng.uniform(0.2, 0.8)
            new_pop.append(normalize(t * p1 + (1 - t) * p2))

        population = new_pop[:pop_size]
        scores = [evaluate_local(p) for p in population]

        if gen % 30 == 0:
            if scores[0] >= best_score - 1e-10:
                strength *= 0.8
            else:
                strength = min(0.05, strength * 1.3)

        if gen % 100 == 0:
            print(f"    Pop gen={gen}: best={best_score:.8f}")

    return best_a, best_score
