            elapsed = time.monotonic() - t0
            if elapsed - last_print > 30:
                print(f"    CD pass {pass_count} ({elapsed:.0f}s): score={best_score:.8f}")
                last_print = elapsed

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = min(Tb.max(), 700)
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=100.0, T_end=0.05, n_stages=15):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = 30

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x ** 2)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 4:
        a = np.ones(n) * (1.0 + 0.2 * np.cos(2 * np.pi * 3 * np.arange(n) / n))
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.15 * np.cos(7 * x)
    elif choice == 6:
        # Geometric-decay inspired
        r = 0.95 + 0.05 * rng.random()
        a = r ** (np.arange(n) * 0.3)
        a = a + r ** ((n - 1 - np.arange(n)) * 0.3)
    else:
        # Two-hump
        x = np.arange(n) / n
        a = np.exp(-10 * (x - 0.2) ** 2) + np.exp(-10 * (x - 0.8) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturb_sequence(a, rng, scale=0.05):
    """Add multiplicative noise to explore nearby solutions."""
    noise = np.exp(rng.normal(0, scale, size=len(a)))
    a_pert = a * noise
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate(seed=42, budget_s=1800.0):
    global _lp_solver
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Get parent sequence or default init
    best_seq = None
    best_score = float('inf')
    n_base = 1200

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Quick multi-start exploration (10% time) ===
    print(f"\n=== Phase 1: Multi-start exploration ===")
    t_phase1 = min(0.10 * (float(budget_s) - 30), deadline - time.monotonic())
    if t_phase1 > 10:
        t_per_init = t_phase1 / 4.0
        for trial in range(4):
            if time.monotonic() > deadline:
                break
            if trial == 0:
                init = best_seq.copy()
            else:
                init = structured_init(n_base, rng)
            a_try, _ = nesterov_lp(init, t_per_init, solver=solver, max_iters=20000)
            try_update(a_try, f"init_{trial}")

    # === Phase 2: Main LP Nesterov (35% time) ===
    print(f"\n=== Phase 2: Main LP Nesterov (score={best_score:.8f}) ===")
    t_phase2 = min(0.35 * (float(budget_s) - 30), deadline - time.monotonic())
    if t_phase2 > 30:
        a_out, _ = nesterov_lp(best_seq, t_phase2, solver=solver)
        try_update(a_out, "lp_main")

    # === Phase 3: Coordinate Descent (30% time) ===
    print(f"\n=== Phase 3: Coordinate Descent (score={best_score:.8f}) ===")
    t_phase3 = min(0.30 * (float(budget_s) - 30), deadline - time.monotonic())
    if t_phase3 > 20:
        a_cd, _ = coordinate_descent(best_seq, t_phase3, seed=seed, verbose=True)
        try_update(a_cd, "cd_main")

    # === Phase 4: Perturbation + CD (10% time) ===
    print(f"\n=== Phase 4: Perturbation + CD (score={best_score:.8f}) ===")
    t_phase4 = min(0.10 * (float(budget_s) - 30), deadline - time.monotonic())
    if t_phase4 > 15:
        n_pert = max(2, int(t_phase4 / 30))
        t_per_pert = t_phase4 / n_pert
        for pi in range(n_pert):
            if time.monotonic() > deadline:
                break
            scale = 0.03 + 0.02 * (pi / n_pert)
            a_pert = perturb_sequence(best_seq, rng, scale=scale)
            a_cd2, _ = coordinate_descent(a_pert, t_per_pert, seed=seed + 100 + pi, verbose=False)
            try_update(a_cd2, f"pert_cd_{pi}")

    # === Phase 5: LP Round 2 (8% time) ===
    print(f"\n=== Phase 5: LP Round 2 (score={best_score:.8f}) ===")
    t_phase5 = min(0.08 * (float(budget_s) - 30), deadline - time.monotonic())
    if t_phase5 > 15:
        a_out2, _ = nesterov_lp(best_seq, t_phase5, solver=solver, max_iters=100000)
        try_update(a_out2, "lp_r2")

    # === Phase 6: Final CD Polish (4% time) ===
    print(f"\n=== Phase 6: Final CD Polish (score={best_score:.8f}) ===")
    t_phase6 = min(0.04 * (float(budget_s) - 30), deadline - time.monotonic())
    if t_phase6 > 10:
        a_cd3, _ = coordinate_descent(best_seq, t_phase6, seed=seed + 999, verbose=False)
        try_update(a_cd3, "cd_final")

    # === Phase 7: LSE Polish (remaining time) ===
    print(f"\n=== Phase 7: LSE Polish (score={best_score:.8f}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        t_lse = min(remaining * 0.7, 120.0)
        if t_lse > 10:
            a_lse = lbgfs_logsumexp(best_seq, t_lse, T_start=50.0, T_end=0.03, n_stages=12)
            try_update(a_lse, "lse_polish")

    # === Final: one more CD with remaining time ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_cd4, _ = coordinate_descent(best_seq, min(remaining, 60.0), seed=seed + 7777, verbose=False)
        try_update(a_cd4, "cd_ultimate")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
