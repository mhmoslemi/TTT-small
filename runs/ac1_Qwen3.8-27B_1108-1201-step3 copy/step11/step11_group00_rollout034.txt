import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Full linear autocorrelation a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def proj_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Algorithm from Duchi et al. 2008
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.find_last((u * np.arange(1, n + 1) - (cssv - 1)) > 0)
    if rho is None or rho < 0:
        return np.ones(n) / n
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def subgrad_autocorr(a, k, top_k=1):
    """
    Compute the (sub)gradient of max_j (a*a)_j at a, using the top_k active lags.
    Returns the average subgradient vector.
    """
    n = len(a)
    b = fast_autocorr(a)
    # Find top_k lags
    idx = np.argsort(b)[::-1][:top_k]
    # Weight by how close to max
    max_b = b[idx[0]]
    min_b = b[idx[-1]]
    if max_b - min_b < 1e-10:
        weights = np.ones(top_k) / top_k
    else:
        # Use exponential weighting
        w = np.exp(-100 * (max_b - b[idx]) / (max_b - min_b))
        weights = w / w.sum()
    
    grad = np.zeros(n)
    for wi, ki in zip(weights, idx):
        # d/d(a_j) [(a*a)_k] = 2*a[k-j] if 0 <= k-j < n
        # For j in range(max(0, k-n+1), min(n-1, k)+1):
        #   grad[j] += 2 * weight * a[k-j]
        js = max(0, ki - n + 1)
        je = min(n - 1, ki)
        if js <= je:
            grad[js:je+1] += wi * 2.0 * a[ki - np.arange(js, je + 1)]
    return grad

def accelerated_subgrad_init(a, time_limit, top_k=5, verbose=True):
    """
    Nesterov-accelerated projected subgradient method for min max_k (a*a)_k
    on the simplex {a >= 0, sum(a) = 1}.
    """
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Nesterov variables
    a_curr = a.copy()
    a_prev = a.copy()
    y = a.copy()  # extrapolation point
    
    # Running average
    avg_a = a.copy()
    n_avg = 1
    
    T = 1
    D = np.sqrt(2.0)  # diameter of simplex
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        
        # Compute subgradient at y
        g = subgrad_autocorr(y, 0, top_k=top_k)
        
        # Step size for accelerated subgradient: eta = 2*D / (t + 3)
        # But we need to scale properly. The subgradient norm is about 2*||a||_2 <= 2.
        # Use eta_t = C / (t + 3) with C chosen for good performance.
        eta = 4.0 * D / (T + 3.0)
        
        # Nesterov extrapolation
        if T == 1:
            y_new = a_curr.copy()
        else:
            momentum = (T - 1.0) / (T + 2.0)
            y_new = a_curr + momentum * (a_curr - a_prev)
        
        # Gradient step + projection
        a_new = proj_simplex(y_new - eta * g)
        
        a_prev = a_curr
        a_curr = a_new
        y = a_new  # For subgradient, we evaluate at the new point
        
        # Running average
        avg_a += a_new
        n_avg += 1
        
        # Check score every few iterations
        if T % 50 == 0 or T < 10:
            score = 2.0 * n * fast_autocorr(a_curr).max()
            if score < best_score:
                best_score = score
                best_a = a_curr.copy()
            
            if n_avg >= 10 and T % 100 == 0:
                avg_check = avg_a / n_avg
                score_avg = 2.0 * n * fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
            
            if verbose and T % 5000 == 0:
                print(f"  AS t={T} ({elapsed:.0f}s): best={best_score:.8f}")
        
        T += 1
    
    # Final check
    score = 2.0 * n * fast_autocorr(a_curr).max()
    if score < best_score:
        best_score = score
        best_a = a_curr.copy()
    
    avg_check = avg_a / n_avg
    score_avg = 2.0 * n * fast_autocorr(avg_check).max()
    if score_avg < best_score:
        best_score = score_avg
        best_a = avg_check.copy()
    
    return best_a, best_score

def fast_autocorr(a):
    """Full linear autocorrelation a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:

        sc = evaluate_sequence(a_list)

def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) - (cssv - 1) > 0
    rho = np.find(cond)  # first True index (0-based)
    # find_last is not standard; use a different approach
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]  # last True index
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

Actually, for the simplex constraint, the natural mirror is the KL divergence, and the mirror descent step is:
  a_new = a * exp(-eta * grad) / sum(a * exp(-eta * grad))

This is the "exponential weighting" or "Hedge" algorithm. It might work better for this problem because it naturally keeps a positive (and the simplex projection is just normalization).

**Mirror descent (Hedge) algorithm:**
```
a = initial (sum = 1, a > 0)
for t = 1, 2, ...:
    g = subgrad(f, a)  # gradient of max_k (a*a)_k at a
    a = a * exp(-eta * g)
    a = a / a.sum()  # normalize
```

The step size eta = O(1/sqrt(t)) for the standard rate.

With Nesterov acceleration:
```
y_0 = a_0
for t = 1, 2, ...:
    g = subgrad(f, y_{t-1})
    a_t = y_{t-1} * exp(-eta_t * g)
    a_t = a_t / a_t.sum()
    y_t = a_t * (a_t / a_{t-1})^{beta_t}  # Nesterov extrapolation in "exponential" space
```

The Nesterov extrapolation in exponential space is:
  log(y_t) = log(a_t) + beta_t * (log(a_t) - log(a_{t-1}))
  y_t = a_t * (a_t / a_{t-1})^{beta_t} / normalization

This is the "accelerated multiplicative weight update" or "accelerated Hedge".

This might converge faster than the Euclidean projection version because:
1. No projection needed (just normalization)
2. Naturally stays in the interior of the simplex (a > 0)
3. The "geometry" matches the problem better (autocorrelation is multiplicative)
