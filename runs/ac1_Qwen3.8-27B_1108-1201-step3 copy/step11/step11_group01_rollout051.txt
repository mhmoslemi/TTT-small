def fast_convolve(a):
    """Fast autocorrelation using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    """Score for normalized sequence (sum=1). Returns 2n*max(b)."""
    n = len(a)

def evaluate_safe(a):

def peak_directed_descent(a0, time_limit, n_steps_max=500000, restart_every=5000,
                           k_threshold=0.90, mom_start=0.5, mom_end=0.9):
    """
    Peak-directed gradient descent in log-space with momentum and periodic restarts.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a)
    
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    # Momentum
    v = np.zeros(n)
    mom = mom_start
    
    # Adaptive learning rate
    eta = 0.1
    
    for step in range(n_steps_max):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Update momentum schedule
        progress = min(1.0, step / max(1, n_steps_max))
        mom = mom_start + (mom_end - mom_start) * progress
        
        # Compute gradient
        grad_z, active, max_b = compute_multi_lag_gradient(a, k_threshold)
        
        # Momentum update
        v = mom * v - eta * grad_z
        z_new = z + v
        a_new = softmax_from_z(z_new)
        
        sc_new = score_normalized(a_new)
        
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            z = z_new
            a = a_new
            # Increase step size slightly
            eta = min(eta * 1.01, 0.5)
        else:
            # Decrease step size
            eta = max(eta * 0.98, 0.001)
            # Don't update z, revert
            v = mom * v  # keep momentum direction
        
        # Periodic restart: add small random perturbation
        if restart_every > 0 and (step + 1) % restart_every == 0:
            perturbation_scale = 0.02 * (1.0 - progress * 0.5)
            z += perturbation_scale * np.random.randn(n)
            a = softmax_from_z(z)
            sc_re = score_normalized(a)
            if sc_re > best_score * 1.01:
                # Perturbation was too bad, revert
                z = z_from_a(best_a)
                a = best_a.copy()
        
    return best_a, best_score

def coarse_to_fine(parent, target_n, time_budget):
    """
    Coarse-to-fine: downsample parent, optimize, upsample, refine.
    """
    # Downsample to n/4
    coarse_n = max(20, target_n // 4)
    x_old = np.linspace(0, 1, len(parent))
    x_coarse = np.linspace(0, 1, coarse_n)
    parent_norm = parent / parent.sum()
    coarse = np.interp(x_coarse, x_old, parent_norm)
    coarse = np.maximum(coarse, 0)
    coarse = coarse / coarse.sum()
    
    # Optimize at coarse resolution
    t_coarse = time_budget * 0.3
    coarse_out, _ = peak_directed_descent(coarse, t_coarse, n_steps_max=200000,
                                           restart_every=2000)
    
    # Upsample to full resolution
    x_full = np.linspace(0, 1, target_n)
    upscale = np.interp(x_full, np.linspace(0, 1, coarse_n), coarse_out)
    upscale = np.maximum(upscale, 0)
    upscale = upscale / upscale.sum()
    
    # Refine at full resolution
    t_fine = time_budget * 0.7
    fine_out, fine_score = peak_directed_descent(upscale, t_fine, n_steps_max=500000,
                                                  restart_every=5000)
    
    return fine_out, fine_score

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:

2. Peak-directed gradient descent in log-space with multi-lag softmax and adaptive step

from scipy.optimize import minimize


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):

        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lse_objective_and_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):

        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

            obj, _ = lse_objective_and_grad(z_val, stage_T, n)
            return obj

            _, grad = lse_objective_and_grad(z_val, stage_T, n)
            return grad

        if (si + 1) % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.2f}: score={sc:.8f} ({elapsed:.0f}s)")

    return softmax_from_z(z)


def peak_descend(a0, time_limit, n_steps_max=400000, k_thresh=0.90,
                 eta_init=0.08, eta_min=0.002, eta_max=0.4,
                 mom_init=0.3, mom_end=0.75, restart_every=8000,
                 restart_scale=0.03, rng=None):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a)

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    v = np.zeros(n)
    eta = eta_init
    patience = 0
    patience_limit = 5000

    for step in range(n_steps_max):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, step / max(1, n_steps_max))
        mom = mom_init + (mom_end - mom_init) * progress

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * k_thresh
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        w_active = b[active]
        w_active = w_active / w_active.sum()

        grad_a = np.zeros(n)
        for idx in range(len(active)):
            k = active[idx]
            w = w_active[idx]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idxs = np.arange(lo, hi + 1)
            grad_a[idxs] += 2.0 * w * a[k - idxs]

        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)

        v = mom * v - eta * grad_z
        z_new = z + v
        a_new = softmax_from_z(z_new)

        sc_new = score_normalized(a_new)

        if sc_new < best_score - 1e-12:
            best_score = sc_new
            best_a = a_new.copy()
            z = z_new
            a = a_new
            eta = min(eta * 1.008, eta_max)
            patience = 0
        elif sc_new < best_score:
            z = z_new
            a = a_new
            patience += 1
        else:
            eta = max(eta * 0.97, eta_min)
            v = mom * v
            patience += 1

        if patience > patience_limit and restart_every > 0:
            pert_scale = restart_scale * (1.0 - 0.5 * progress)
            if rng is not None:
                z += pert_scale * rng.standard_normal(n)
            else:
                z += pert_scale * np.random.standard_normal(n)
            a = softmax_from_z(z)
            v *= 0.5
            patience = 0
            if (step + 1) % restart_every < patience_limit:
                pass

        if (step + 1) % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakDescend step={step+1} ({elapsed:.0f}s): best={best_score:.8f} eta={eta:.4f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def sinusoidal_init(n, rng, n_harmonics=5):
    """Initialize with a baseline plus a few sinusoidal harmonics."""
    a = np.ones(n)
    for m in range(1, n_harmonics + 1):
        freq = rng.integers(1, 20)
        amp = rng.uniform(0.02, 0.15) * (1.0 / m)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def structured_init(n, rng):
    choice = rng.integers(0, 7)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
    elif choice == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-3 * x ** 2)
    elif choice == 3:
        a = np.ones(n) * (1.0 + 0.2 * np.cos(2 * np.pi * 3 * np.arange(n) / n)
                          + 0.1 * np.cos(2 * np.pi * 7 * np.arange(n) / n))
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2 * x) + 0.15 * np.cos(5 * x)
    elif choice == 5:
        a = sinusoidal_init(n, rng, 4)
    else:
        a = np.ones(n) + 0.1 * rng.standard_normal(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    n_base = 1200
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
        else:
            raise ValueError("Invalid parent")
    except Exception:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            remaining = deadline - time.monotonic()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)}, remaining={remaining:.0f}s)")

    # === Phase 1: LSE multi-stage L-BFGS (25% budget) ===
    if time.monotonic() < deadline:
        t1 = min(0.25 * (budget_s - 30), (deadline - time.monotonic()) * 0.30)
        print(f"\n=== Phase 1: LSE L-BFGS ({t1:.0f}s) ===")
        a_out = lse_lbfgs_multistage(best_seq, t1, T_start=60.0, T_end=0.5, n_stages=20)
        try_update(a_out, "lse_lbfgs")

    # === Phase 2: Peak-directed descent (30% budget) ===
    if time.monotonic() < deadline:
        t2 = min(0.30 * (budget_s - 30), (deadline - time.monotonic()) * 0.35)
        print(f"\n=== Phase 2: Peak-directed descent ({t2:.0f}s) ===")
        a_out, _ = peak_descend(best_seq, t2, n_steps_max=300000, k_thresh=0.90,
                                eta_init=0.06, mom_init=0.3, mom_end=0.7,
                                restart_every=8000, restart_scale=0.025, rng=rng)
        try_update(a_out, "peak_descend_1")

    # === Phase 3: Length exploration (20% budget) ===
    if time.monotonic() < deadline:
        t3_total = min(0.20 * (budget_s - 30), (deadline - time.monotonic()) * 0.35)
        print(f"\n=== Phase 3: Length exploration ({t3_total:.0f}s) ===")
        lengths_to_try = [n_base, n_base - 100, n_base + 100,
                          n_base - 200, n_base + 200, n_base + 400]
        lengths_to_try = [l for l in lengths_to_try if 200 <= l <= 5000]
        # Remove duplicates
        seen = set()
        unique_lengths = []
        for l in lengths_to_try:
            if l not in seen:
                seen.add(l)
                unique_lengths.append(l)
        unique_lengths.remove(n_base)  # Already at n_base

        n_trials = min(len(unique_lengths), 4)
        t3_each = t3_total / max(1, n_trials)

        for trial_i in range(n_trials):
            if time.monotonic() >= deadline:
                break
            n_try = unique_lengths[trial_i]
            t3 = min(t3_each, (deadline - time.monotonic()) * 0.35)
            if t3 < 10:
                break
            print(f"  Trying n={n_try} (budget={t3:.0f}s)")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = peak_descend(init_try, t3, n_steps_max=150000, k_thresh=0.90,
                                    eta_init=0.05, mom_init=0.3, mom_end=0.6,
                                    restart_every=5000, restart_scale=0.02, rng=rng)
            try_update(a_try, f"n_{n_try}")

    # === Phase 4: Random restarts with perturbation (15% budget) ===
    if time.monotonic() < deadline:
        t4_total = min(0.15 * (budget_s - 30), (deadline - time.monotonic()) * 0.35)
        print(f"\n=== Phase 4: Random restarts ({t4_total:.0f}s) ===")
        n_restarts = 5
        t4_each = t4_total / n_restarts
        for ri in range(n_restarts):
            if time.monotonic() >= deadline:
                break
            t4 = min(t4_each, (deadline - time.monotonic()) * 0.3)
            if t4 < 8:
                break

            # Different perturbation strategies
            if ri == 0:
                a_pert = best_seq * (1.0 + 0.05 * rng.standard_normal(len(best_seq)))
            elif ri == 1:
                a_pert = best_seq * (1.0 + 0.10 * rng.standard_normal(len(best_seq)))
            elif ri == 2:
                # Frequency-domain perturbation
                x = np.linspace(0, 1, len(best_seq))
                freq = rng.integers(2, 50)
                a_pert = best_seq * (1.0 + 0.03 * np.sin(2 * np.pi * freq * x))
            elif ri == 3:
                a_pert = structured_init(len(best_seq), rng)
                # Blend with current best
                a_pert = 0.7 * best_seq + 0.3 * a_pert
            else:
                # Local random walk
                a_pert = best_seq.copy()
                n_perturb = len(best_seq) // 10
                idxs = rng.choice(len(best_seq), n_perturb, replace=False)
                a_pert[idxs] *= (1.0 + 0.2 * rng.standard_normal(n_perturb))

            a_pert = np.maximum(a_pert, 0)
            s = a_pert.sum()
            if s > 1e-10:
                a_pert = a_pert / s

            a_out, _ = peak_descend(a_pert, t4, n_steps_max=100000, k_thresh=0.88,
                                    eta_init=0.05, mom_init=0.2, mom_end=0.5,
                                    restart_every=3000, restart_scale=0.02, rng=rng)
            try_update(a_out, f"restart_{ri}")

    # === Phase 5: Ultra-fine tuning (10% budget) ===
    if time.monotonic() < deadline:
        t5 = min(0.10 * (budget_s - 30), (deadline - time.monotonic()) * 0.50)
        print(f"\n=== Phase 5: Ultra-fine tuning ({t5:.0f}s) ===")
        # Very tight threshold, small steps
        a_out, _ = peak_descend(best_seq, t5, n_steps_max=200000, k_thresh=0.97,
                                eta_init=0.02, eta_min=0.001, eta_max=0.1,
                                mom_init=0.5, mom_end=0.8,
                                restart_every=20000, restart_scale=0.01, rng=rng)
        try_update(a_out, "ultra_fine")

    # === Phase 6: Final LSE polish if time remains ===
    if time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            t6 = min(remaining * 0.6, 120.0)
            print(f"\n=== Phase 6: Final LSE polish ({t6:.0f}s) ===")
            a_out = lse_lbfgs_multistage(best_seq, t6, T_start=20.0, T_end=0.2, n_stages=10)
            try_update(a_out, "final_lse")
            # One more fine descent
            if time.monotonic() < deadline:
                t6b = min(30.0, (deadline - time.monotonic()) * 0.5)
                if t6b > 5:
                    a_out2, _ = peak_descend(best_seq, t6b, n_steps_max=50000,
                                             k_thresh=0.95, eta_init=0.01,
                                             eta_min=0.0005, eta_max=0.05,
                                             mom_init=0.5, mom_end=0.7,
                                             restart_every=50000, restart_scale=0.005, rng=rng)
                    try_update(a_out2, "final_peak")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
