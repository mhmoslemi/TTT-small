        return float('inf')
    n = len(a)
    return float(2.0 * n * fast_convolve(a).max() / (s * s))


def build_T(a):
    """Sparse convolution matrix: T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    # For each column i, T[k, i] = a[k-i] for k in [i, i+n-1]
    cols = np.repeat(np.arange(n), n)
    rows = np.concatenate([np.arange(i, i+n) for i in range(n)])
    data = np.tile(a, n)
    return csr_matrix((data, (rows, cols)), shape=(size, n))


def solve_best_response(a, T, solver):
    """Solve: min t s.t. T@b <= t, sum(b)=1, b>=0. Returns (b, t)."""
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    ones = cp.Parameter(n + 1)  # No, just use np.ones
    constraints = [T @ b <= t, cp.sum(b) == 1]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None, None
    except Exception:
        return None, None
    b_val = np.asarray(b.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    sb = b_val.sum()
    if sb < 1e-15:
        return None, None
    b_val /= sb
    t_val = float(t.value)
    if np.isnan(t_val) or np.isinf(t_val):
        t_val = float((T @ b_val).max())
    return b_val, t_val


def solve_dual(a, T, solver):
    """Solve dual: min max_i (T^T @ lambda)_i
    s.t. sum(lambda)=1, lambda>=0
    Returns (lambda, t_dual)."""
    n = len(a)
    size = 2 * n - 1
    lam = cp.Variable(size, nonneg=True)
    t_d = cp.Variable()
    constraints = [T.T @ lam <= t_d, cp.sum(lam) == 1]
    prob = cp.Problem(cp.Minimize(t_d), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None, None
    except Exception:
        return None, None
    lam_val = np.asarray(lam.value, dtype=np.float64)
    lam_val = np.maximum(lam_val, 0)
    sl = lam_val.sum()
    if sl < 1e-15:
        return None, None
    lam_val /= sl
    t_d_val = float(t_d.value)
    if np.isnan(t_d_val) or np.isinf(t_d_val):
        t_d_val = float((T.T @ lam_val).max())
    return lam_val, t_d_val


def nesterov_lp(a0, time_budget, solver, verbose=True):

def coord_descent(a0, time_budget, verbose=True):

    n = len(a0)

    t0 = time.monotonic()
    
    if verbose:
        print(f"  CD start: score={best_score:.8f}")
    
    it = 0
    while time.monotonic() - t0 < time_budget:
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        
        # Find top contributors
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Try reducing each top contributor
        top_m = min(10, len(idx))
        top_pos = idx[np.argsort(contribs)[::-1][:top_m]]
        
        improved = False
        for pos in top_pos:
            if a[pos] < 1e-14:
                continue
            for factor in [0.999, 0.995, 0.99, 0.98, 0.97, 0.95]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] - a_trial[pos]
                if diff < 1e-16:
                    continue
                # Redistribute to others proportionally
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others_sum = a[mask].sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] = a[mask] + diff * a[mask] / others_sum
                a_trial /= a_trial.sum()
                
                sc = eval_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

            # Try small random perturbation
            if it % 100 == 0:
                pos = np.random.randint(n)
                a_trial = a.copy()
                a_trial[pos] *= 0.98
                diff = a[pos] - a_trial[pos]

                    if others_sum > 1e-15:

    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    n = len(a)
    return float(2.0 * n * fast_convolve(a).max() / (s * s))


def build_T(a):
    """Sparse matrix T (2n-1 x n) such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    cols = np.repeat(np.arange(n), n)
    rows = np.concatenate([np.arange(i, i + n) for i in range(n)])
    data = np.tile(a, n)
    return csr_matrix((data, (rows, cols)), shape=(size, n))


def solve_primal_lp(a, T, solver):
    """Solve: min t s.t. T@b <= t*1, sum(b)=1, b>=0. Returns (b, t)."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [T @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            for alt in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
                if alt == solver:
                    continue
                try:
                    prob.solve(solver=alt, verbose=False)
                    if prob.status in ('optimal', 'optimal_inaccurate'):
                        break
                except Exception:
                    continue
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                return None, None
    except Exception:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    sb = b_val.sum()
    if sb < 1e-15:
        return None, None
    b_val /= sb
    t_val = float(t_var.value)
    if t_val is None or np.isnan(t_val) or np.isinf(t_val):
        t_val = float((T @ b_val).max())
    return b_val, t_val


def solve_dual_lp(a, T, solver):
    """Solve dual: min t s.t. T^T@lam <= t*1, sum(lam)=1, lam>=0."""
    n = len(a)
    size = 2 * n - 1
    lam_var = cp.Variable(size, nonneg=True)
    t_var = cp.Variable()
    constraints = [T.T @ lam_var <= t_var, cp.sum(lam_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None, None
    except Exception:
        return None, None
    lam_val = np.asarray(lam_var.value, dtype=np.float64)
    lam_val = np.maximum(lam_val, 0)
    sl = lam_val.sum()
    if sl < 1e-15:
        return None, None
    lam_val /= sl
    t_val = float(t_var.value)
    if t_val is None or np.isnan(t_val) or np.isinf(t_val):
        t_val = float((T.T @ lam_val).max())
    return lam_val, t_val


def nesterov_lp_optimize(a0, time_budget, solver, verbose=True):
    """Nesterov-accelerated LP alternating optimization."""
    n = len(a0)
    a = np.maximum(np.array(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    a /= a.sum()

    y = a.copy()
    it = 0
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    stall = 0
    max_stall = 40
    rng = np.random.default_rng(7)

    if verbose:
        print(f"    Start: n={n}, score={best_score:.10f}")

    while time.monotonic() - t0 < time_budget:
        it += 1
        if time.monotonic() - t0 > time_budget:
            break

        T = build_T(y)
        b_new, t_primal = solve_primal_lp(y, T, solver)
        if b_new is None:
            if it % 100 == 0:
                print(f"    iter={it}: LP failed")
            continue

        # Nesterov step
        alpha = 2.0 / (it + 2)
        y_new = (1.0 - alpha) * y + alpha * b_new
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a.copy()
            sy = 1.0
        y_new /= sy
        y = y_new

        # Weighted average
        a = (it * a + alpha * (it + 1) * b_new) / (it + alpha * (it + 1))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
            sa = 1.0
        a /= sa

        # Evaluate
        sc_a = eval_score(a)
        sc_y = eval_score(y)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            stall = 0
        elif sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        # Periodic checks and exploration
        if it % 25 == 0:
            # Dual gap check
            lam_val, t_dual = solve_dual_lp(y, T, solver)
            if verbose and it % 250 == 0:
                gap = abs(t_dual - t_primal) if t_dual is not None else -1
                print(f"    iter={it}, t_p={t_primal:.8f}, t_d={t_dual:.8f}, "
                      f"gap={gap:.2e}, score={best_score:.10f}")

            # Exploration on stall
            if stall >= max_stall:
                b_cur = fast_convolve(y)
                peak_k = int(np.argmax(b_cur))
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                idx = np.arange(js, je + 1)
                contribs = y[idx] * y[peak_k - idx]
                n_red = max(3, len(idx) // 10)
                top_pos = idx[np.argsort(contribs)[::-1][:n_red]]
                for pos in top_pos:
                    y[pos] *= (1.0 - 0.03 * np.sqrt(stall / max_stall))
                # Also try adding to a random non-peak position
                other_idx = np.setdiff1d(np.arange(n), idx)
                if len(other_idx) > 0:
                    add_pos = other_idx[rng.integers(0, len(other_idx))]
                    y[add_pos] += 0.01 * y.sum() / n
                # Random noise
                y += rng.standard_normal(n) * 0.003
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y /= sy
                a = y.copy()
                stall = 0
                max_stall = min(60, max_stall + 5)
                if verbose:
                    print(f"    iter={it}: EXPLORE")

        if verbose and it % 500 == 0:
            print(f"    iter={it}, t={time.monotonic()-t0:.0f}s, score={best_score:.10f}")

    return best_a, best_score


def subgradient_polish(a0, time_budget, verbose=True):
    """Subgradient method with Nesterov for fine-tuning (no LP, faster iterations)."""
    n = len(a0)
    a = np.maximum(np.array(a0, dtype=np.float64), 0)
    s = a.sum()
    a /= s
    y = a.copy()
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    eta = 0.1
    eta_decay = 0.9999
    stall = 0
    rng = np.random.default_rng(99)

    if verbose:
        print(f"    SG start: score={best_score:.10f}")

    it = 0
    while time.monotonic() - t0 < time_budget:
        it += 1
        b = fast_convolve(y)
        # Use all peaks above 99.9% of max
        max_b = b.max()
        threshold = max_b * 0.999
        peaks = np.where(b >= threshold)[0]
        if len(peaks) == 0:
            peaks = np.array([int(np.argmax(b))])

        # Subgradient: average of conv(y, e_k) for active peaks
        g = np.zeros(n)
        for k in peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += y[k - idx]
        g /= len(peaks)

        # Nesterov update
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * (y - eta * g)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y /= sy

        a = (it * a + alpha * (it + 1) * (y - eta * g)) / (it + alpha * (it + 1))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
            sa = 1.0
        a /= sa

        eta *= eta_decay

        sc_a = eval_score(a)
        sc_y = eval_score(y)
        if min(sc_a, sc_y) < best_score:
            best_score = min(sc_a, sc_y)
            best_a = a.copy() if sc_a < sc_y else y.copy()
            stall = 0
        else:
            stall += 1

        if stall > 200:
            y += rng.standard_normal(n) * 0.005
            y = np.maximum(y, 0)
            y /= y.sum()
            a = y.copy()
            eta = 0.1
            stall = 0

        if verbose and it % 2000 == 0:
            print(f"    SG iter={it}, t={time.monotonic()-t0:.0f}s, score={best_score:.10f}")

    return best_a, best_score


def coord_descent_polish(a0, time_budget, verbose=True):
    """Coordinate descent focusing on peak-contributing elements."""
    n = len(a0)
    a = np.maximum(np.array(a0, dtype=np.float64), 0)
    s = a.sum()
    a /= s
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    stall = 0

    if verbose:
        print(f"    CD start: score={best_score:.10f}")

    it = 0
    while time.monotonic() - t0 < time_budget:
        it += 1
        b = fast_convolve(a)
        max_b = b.max()
        # Consider top peaks
        threshold = max_b * 0.9999
        peaks = np.where(b >= threshold)[0][:3]
        if len(peaks) == 0:
            peaks = np.array([int(np.argmax(b))])

        improved = False
        for peak_k in peaks:
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_m = min(12, len(idx))
            top_pos = idx[np.argsort(contribs)[::-1][:top_m]]

            for pos in top_pos:
                if a[pos] < 1e-14:
                    continue
                for factor in [0.9995, 0.999, 0.998, 0.995, 0.99, 0.985, 0.98, 0.97, 0.95]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-16:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others_sum = a[mask].sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = a[mask] + diff * a[mask] / others_sum
                    a_trial /= a_trial.sum()
                    sc = eval_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 200:
                # Try simultaneous small adjustments
                peak_k = int(np.argmax(fast_convolve(a)))
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[peak_k - idx]
                top_pos = idx[np.argsort(contribs)[::-1][:5]]
                a_trial = a.copy()
                for pos in top_pos:
                    a_trial[pos] *= 0.999
                a_trial = np.maximum(a_trial, 0)
                a_trial /= a_trial.sum()
                sc = eval_score(a_trial)
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    stall = 0
                elif stall > 2000:
                    break
        else:
            stall = 0

        if verbose and it % 2000 == 0:
            print(f"    CD iter={it}, t={time.monotonic()-t0:.0f}s, score={best_score:.10f}")

    return best_a, best_score


def interpolate_seq(a, new_n):
    """Interpolate sequence to new length."""
    if len(a) == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, len(a))
    new_x = np.linspace(0, 1, new_n)
    out = np.interp(new_x, old_x, a_norm)
    out = np.maximum(out, 0)
    ss = out.sum()
    if ss > 1e-15:
        out /= ss
    return out


def get_solver():
    """Find a working LP solver."""
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS


def structured_init(n, seed=0):
    """Create a structured initial sequence."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 5)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    else:
        a = np.exp(-6 * (x - 0.3) ** 2) + np.exp(-6 * (x - 0.7) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Phases:
    1. Load parent or structured init
    2. Nesterov LP (main)
    3. Subgradient polish
    4. Multi-n exploration
    5. Deep LP on best
    6. Coordinate descent
    """
    deadline = time.monotonic() + max(1.0, budget_s - 15.0)
    solver = get_solver()
    print(f"Solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_score(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.10f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = structured_init(1200, seed=seed)
        best_score = eval_score(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.10f}")

    def do_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        sc = eval_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  >>> NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    n0 = len(best_seq)

    # === Phase 1: Main Nesterov LP ===
    print(f"\n=== Phase 1: Nesterov LP (n={n0}, score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.45, 650.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a1, _ = nesterov_lp_optimize(best_seq, t1, solver, verbose=True)
        do_update(a1, "LP1")

    # === Phase 2: Subgradient polish ===
    print(f"\n=== Phase 2: Subgradient (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.15, 200.0)
    if t2 > 15:
        print(f"  Budget: {t2:.0f}s")
        a2, _ = subgradient_polish(best_seq, t2, verbose=True)
        do_update(a2, "SG")

    # === Phase 3: Multi-n exploration ===
    print(f"\n=== Phase 3: Multi-n (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 280.0)
    if t3 > 30:
        n_best = len(best_seq)
        n_candidates = sorted(set([
            max(400, n_best - 200),
            max(400, n_best + 200),
            max(400, n_best + 400),
            max(400, int(n_best * 0.7)),
            max(400, int(n_best * 1.4)),
        ]))
        per_n = t3 / max(1, len(n_candidates))
        print(f"  Candidates: {n_candidates}, {per_n:.0f}s each")
        for n_try in n_candidates:
            if time.monotonic() > deadline - 10:
                break
            if n_try == n_best:
                continue
            t_sub = min(per_n, deadline - time.monotonic() - 5)
            if t_sub < 15:
                continue
            a_interp = interpolate_seq(best_seq, n_try)
            sc_i = eval_score(a_interp)
            print(f"  n={n_try}: init={sc_i:.8f}, budget={t_sub:.0f}s")
            a_sub, _ = nesterov_lp_optimize(a_interp, t_sub, solver, verbose=False)
            do_update(a_sub, f"n_{n_try}")

    # === Phase 4: Deep LP ===
    print(f"\n=== Phase 4: Deep LP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.30, 350.0)
    if t4 > 20:
        print(f"  Budget: {t4:.0f}s")
        a4, _ = nesterov_lp_optimize(best_seq, t4, solver, verbose=True)
        do_update(a4, "LP4")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coord Descent (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.25, 200.0)
    if t5 > 10:
        print(f"  Budget: {t5:.0f}s")
        a5, _ = coord_descent_polish(best_seq, t5, verbose=True)
        do_update(a5, "CD")

    # === Phase 6: Final quick LP ===
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.4, 80.0)
    if t6 > 10:
        print(f"\n=== Phase 6: Final LP (budget={t6:.0f}s) ===")
        a6, _ = nesterov_lp_optimize(best_seq, t6, solver, verbose=False)
        do_update(a6, "LP6")

    # Final
    if best_seq is not None:
        fs = eval_score(best_seq)
        print(f"\nFINAL SCORE: {fs:.10f} (n={len(best_seq)})")
        # Verify with evaluate_sequence
        try:
            es = evaluate_sequence([float(x) for x in best_seq])
            print(f"evaluate_sequence: {es:.10f}")
        except Exception as e:
            print(f"eval check error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
