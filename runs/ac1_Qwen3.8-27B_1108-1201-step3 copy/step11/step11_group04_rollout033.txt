"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral flattening for initialization

4. Ultra-fine LogSumExp L-BFGS for final polishing

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]
    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_val = b_val.sum()
                if s_val < 1e-15:
                    return None
                return b_val / s_val
        except Exception:
            continue
    return None


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse
    
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def spectral_flatten(a, n_iters=50, blend=0.4, deadline=None):
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)
        mags = np.abs(F)
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])
        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new
    return a

    prev_best = best_score

        max_b = b.max()

        S = set(np.where(b >= max_b - tol)[0].tolist())

            S = {int(np.argmax(b))}
        
        P_list = sorted(P)
        if len(P_list) == 0:
            break

        sums = P_arr[:, None] + P_arr[None, :]
        S_arr = np.zeros(2 * n - 1, dtype=bool)
        for k in S:
            if 0 <= k < 2 * n - 1:

        A_sys = np.zeros((m + 1, m + 1))
        A_sys[:m, :m] = C
        A_sys[:m, m] = -len(S)
        A_sys[m, :m] = 1.0
        b_sys = np.zeros(m + 1)
        b_sys[m] = 1.0
        
        try:
            sol = np.linalg.solve(A_sys, b_sys)
        except np.linalg.LinAlgError:

            try:
                sol = np.linalg.solve(A_sys + 1e-12 * np.eye(m + 1), b_sys)
            except:
                break

        nu = sol[m]

            # Remove the most negative from P
            worst = P_list[np.argmin(a_new_P)]
            P.discard(worst)
            if len(P) == 0:
                break
            continue
        
        # Clip small negatives to zero
        a_new_P = np.maximum(a_new_P, 0)
        s_new_P = a_new_P.sum()
        if s_new_P < 1e-15:
            break
        a_new_P = a_new_P / s_new_P  # Renormalize (the system should give sum=1 already)
        
        # Build full a
        a_new = np.zeros(n)
        for i, j in enumerate(P_list):
            a_new[j] = a_new_P[i]
        
        # Check KKT for indices NOT in P
        # For j not in P: (C_full @ a_new)[j] / |S| <= nu
        # C_full[j, :] @ a = sum_{l in P} C[j,l] a[l] = sum_{l in P} 2*1[j+l in S] a[l]

            if j in P:
                continue
            # Compute (C @ a)[j] = sum_{l: j+l in S} 2*a[l]
            val = 0.0
            for l in P_list:
                if (j + l) in S:
                    val += 2.0 * a_new[l]
            if val / len(S) > nu + 1e-8:
                # Add j to P
                P.add(j)
                violated = True
                break
        
        if not violated:
            # Check if S needs update
            b_new = fast_convolve(a_new)
            max_new = b_new.max()
            tol_new = max(1e-8, 1e-6 * max_new)
            S_new = set(np.where(b_new >= max_new - tol_new)[0].tolist())
            
            if S_new != S:
                S = S_new
                # Don't break, re-solve with new S
                a = a_new
                if 2.0 * n * max_new / (a.sum()**2) < best_score:
                    best_score = 2.0 * n * max_new / (a.sum()**2)
                    best_a = a.copy()
                continue

                best_score = score
                best_a = a.copy()

            best_score = 2.0 * n * fast_convolve(a).max() / (a.sum()**2)
            best_a = a.copy()
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral flattening (Gerchberg-Saxton) for fast initialization
2. Nesterov-accelerated LP alternating method
3. Active-set KKT linear system solver for precise convergence (new)
4. Ultra-fine LogSumExp L-BFGS with temperatures down to 0.0005
5. Peak-shaving multiplicative updates
6. Multi-n exploration and final polishing

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate score locally (fast, no function call overhead)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]
    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_val = b_val.sum()
                if s_val < 1e-15:
                    return None
                return b_val / s_val
        except Exception:
            continue
    return None


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse
    w = exp_terms / exp_terms.sum()
    # Only iterate over significant weights
    active_k = np.where(w > 1e-12)[0]
    grad_a = np.zeros(n)
    for k in active_k:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def spectral_flatten(a, n_iters=50, blend=0.4, deadline=None):
    """Iterative spectral flattening (Gerchberg-Saxton)."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)
        mags = np.abs(F)
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])
        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new
    return a


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=1000000):
    """Nesterov dual averaging LP alternating method."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)
    lp_times = []
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        lp_t0 = time.monotonic()
        b_new = solve_lp_best_response(y, solver)
        lp_times.append(time.monotonic() - lp_t0)
        if b_new is None:
            continue
        if len(lp_times) >= 50:
            avg_lp = np.mean(lp_times[-50:])
            if avg_lp > 8.0:
                break
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-12:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 6:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 7)
                noise = 0.02 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                stall_count = 0
    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    deadline = time.monotonic() + time_limit
    for stage_T in T_stages:
        if time.monotonic() > deadline:
            break
        if time.monotonic() > deadline - 3:
            break
        remaining = deadline - time.monotonic()
        iters = max(50, min(maxiter_per_stage, int(remaining * 15)))
        func_closed = _make_lbgfs_func(n, stage_T, deadline)
        jac_closed = _make_lbgfs_jac(n, stage_T, deadline)
        try:
            res = minimize(
                fun=func_closed, x0=z, jac=jac_closed, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-30, 'gtol': 1e-22,
                         'maxfun': iters * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    return softmax_from_z(z)


def _make_lbgfs_func(n, T_stage, deadline):
    """Create objective function for L-BFGS (top-level to avoid closures)."""
    def func(z_in):
        if time.monotonic() > deadline:
            return 1e15
        obj, _ = logsumexp_obj_grad(z_in, n, T_stage)
        return obj
    return func


def _make_lbgfs_jac(n, T_stage, deadline):
    """Create gradient function for L-BFGS."""
    def jac(z_in):
        if time.monotonic() > deadline:
            return np.zeros_like(z_in)
        _, g = logsumexp_obj_grad(z_in, n, T_stage)
        return g
    return jac


def active_set_refine(a0, max_iters=2000, deadline=None):
    """
    Active-set KKT method: solves the linear system from KKT conditions.
    This is the key new component for precise convergence.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    # Initial support: indices with a > threshold
    P_mask = a > 1e-8
    P = set(np.where(P_mask)[0].tolist())
    if not P:
        P = set(range(n))

    prev_obj = float('inf')
    no_improve_count = 0

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Active set: peaks within tolerance of max
        # Use adaptive tolerance
        tol = max(1e-10, 5e-7 * max_b)
        S_indices = np.where(b >= max_b - tol)[0]
        if len(S_indices) < 2:
            # Widen tolerance
            tol2 = max(1e-8, 1e-4 * max_b)
            S_indices = np.where(b >= max_b - tol2)[0]
        S = set(S_indices.tolist())

        P_list = sorted(P)
        if len(P_list) == 0:
            break
        P_arr = np.array(P_list, dtype=np.int64)
        m = len(P_arr)

        # Build C matrix: C[i,j] = 2 * 1[P_arr[i] + P_arr[j] in S]
        S_arr = np.zeros(2 * n - 1, dtype=bool)
        for k in S:
            if 0 <= k < 2 * n - 1:
                S_arr[k] = True
        sums_mat = P_arr[:, None] + P_arr[None, :]
        C = (2.0 * S_arr[sums_mat]).astype(np.float64)

        # Solve [C, -|S|*1; 1^T, 0] [a_P; nu] = [0; 1]
        A_sys = np.zeros((m + 1, m + 1))
        A_sys[:m, :m] = C
        A_sys[:m, m] = -float(len(S))
        A_sys[m, :m] = 1.0
        b_sys = np.zeros(m + 1)
        b_sys[m] = 1.0

        try:
            # Check condition number
            cond = np.linalg.cond(A_sys)
            if cond > 1e14:
                sol = np.linalg.lstsq(A_sys, b_sys, rcond=None)[0]
            else:
                sol = np.linalg.solve(A_sys, b_sys)
        except np.linalg.LinAlgError:
            try:
                sol = np.linalg.lstsq(A_sys + 1e-10 * np.eye(m + 1), b_sys, rcond=None)[0]
            except Exception:
                break

        a_new_P = sol[:m]
        nu = sol[m]

        # Check for significantly negative values - remove from support
        if np.any(a_new_P < -1e-6):
            # Remove the most negative
            worst_idx = np.argmin(a_new_P)
            worst_j = P_list[worst_idx]
            P.discard(worst_j)
            if len(P) == 0:
                break
            continue

        # Clip small negatives
        a_new_P = np.maximum(a_new_P, 0.0)
        s_new_P = a_new_P.sum()
        if s_new_P < 1e-15:
            break
        a_new_P = a_new_P / s_new_P

        # Build full vector
        a_new = np.zeros(n)
        for i, j in enumerate(P_list):
            a_new[j] = a_new_P[i]

        # Check KKT for indices NOT in P (using vectorized computation)
        # val[j] = sum_{l in P} 2*1[j+l in S] * a_new_P[l] = (C_full_row @ a_new_P)
        # sums_check[j, i] = j + P_arr[i]
        not_in_P = np.ones(n, dtype=bool)
        not_in_P[P_arr] = False
        check_indices = np.where(not_in_P)[0]

        if len(check_indices) > 0:
            checks_sums = check_indices[:, None].astype(np.int64) + P_arr[None, :]
            checks_mask = S_arr[checks_sums].astype(np.float64)
            vals = checks_mask @ a_new_P  # shape (len(check_indices),)
            threshold = nu + 1e-10
            violations = np.where(vals / float(len(S)) > threshold)[0]
            if len(violations) > 0:
                # Add the most violated index to P
                worst_violation = violations[np.argmax(vals[violations])]
                P.add(int(check_indices[worst_violation]))
                a = a_new
                sc = evaluate_local(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                continue

        # Check S consistency
        b_new = fast_convolve(a_new)
        max_new = b_new.max()
        tol_new = max(1e-10, 5e-7 * max_new)
        S_new = set(np.where(b_new >= max_new - tol_new)[0].tolist())

        if S_new != S:
            S = S_new
            a = a_new
            sc = evaluate_local(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            # Re-solve with new S (don't break)
            prev_obj = sc
            no_improve_count = 0
            continue

        # Check convergence: objective hasn't changed
        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        if abs(sc - prev_obj) < 1e-14:
            no_improve_count += 1
        else:
            no_improve_count = 0
        prev_obj = sc

        if no_improve_count >= 3:
            break

        a = a_new

    return best_a, best_score


def peak_shave(a0, time_limit, n_steps=200000):
    """Peak-shaving multiplicative updates targeting top peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = 0.015
    scale = 1.0
    no_improve = 0
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 5e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        eta_eff = eta * scale
        log_scale = -eta_eff * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 40:
            scale *= 0.75
            no_improve = 0
            if scale < 0.003:
                break
    return best_a, best_score


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    else:
        center = n // 2
        t = (np.arange(n) - center) / center
        a = np.sinc(t)
        a = np.maximum(a, 0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
