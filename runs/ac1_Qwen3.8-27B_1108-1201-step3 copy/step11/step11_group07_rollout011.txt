def gs_phase_retrieval(a0, n_iters=10000):
    n = len(a0)
    fft_size = 1 << (2*n - 1).bit_length()
    a = a0.copy()
    target_mag = np.ones(fft_size//2 + 1) / np.sqrt(fft_size/2 + 1)  # flat
    
    for it in range(n_iters):
        # Forward DFT
        A = rfft(a, fft_size)
        # Replace magnitude with target
        phase = A / (np.abs(A) + 1e-30)
        A_new = target_mag * phase
        # Inverse DFT
        a_new = irfft(A_new, fft_size)[:n]
        # Project: non-negative and sum=1
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        a = a_new
    return a
```

def subgradient_optimize(a0, t_max):
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = evaluate_local(a)
    
    momentum = np.zeros(n)
    lr = 0.01
    gamma = 0.9  # momentum coefficient
    
    t0 = time.monotonic()
    step = 0
    
    while time.monotonic() - t0 < t_max:
        c = fast_convolve(a)
        c_max = c.max()
        A = np.where(c >= c_max * (1 - 1e-8))[0]
        
        # Compute subgradient
        g = np.zeros(n)
        for k in A:
            for m in range(n):
                idx = k - m
                if 0 <= idx < n:
                    g[m] += 2 * a[idx]
        g /= len(A)
        
        # Project onto simplex tangent: remove mean
        g -= g.mean()
        
        # Momentum
        momentum = gamma * momentum + (1-gamma) * g
        g_use = momentum
        g_use -= g_use.mean()
        
        # Step
        a_new = a - lr * g_use
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            lr *= 0.95
        
        if step % 1000 == 0:
            lr = max(0.01 * (0.99 ** (step//1000)), 0.0001)
        step += 1
    
    return best_a, best_score
```

import time
from scipy.fft import rfft, irfft

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    return irfft(A * A, fft_size)[:size]

    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)

        return np.ones(len(a)) / len(a)
    return a / s

def build_conv_matrix_sparse(a):
    """Build sparse Toeplitz matrix for conv(a, b) <= M constraints."""
    n = len(a)
    rows = np.array([j + i for j in range(n) for i in range(n)])
    cols = np.array([j for j in range(n) for _ in range(n)])
    vals = np.tile(a, n)

            s = b.sum()
            if s > 1e-10:
                return b / s

        print(f"LP error: {e}")
    return None

def subgradient_optimize(a0, time_limit, momentum_decay=0.9):
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    
    v = np.zeros(n)  # momentum
    t = 0.01  # step size
    t0 = time.monotonic()
    step = 0
    patience = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        c_max = c.max()
        
        # Active set: positions where c is close to max
        threshold = c_max * (1 - 1e-6)
        active_mask = (c >= threshold).astype(float)
        n_active = active_mask.sum()
        if n_active == 0:
            active_mask[np.argmax(c)] = 1
            n_active = 1
        
        # Compute subgradient via correlation
        # g[m] = 2 * sum_k active_mask[k] * a[k-m] / n_active
        # This is 2/n_active * cross_correlate(active_mask, a)[m]
        # cross_correlate(u, a)[m] = sum_j u[j+m] * a[j]
        # numpy: np.correlate(u, a, 'valid') gives this for u longer than a

        g = 2.0 * np.correlate(u, a, 'valid')

        g -= g.mean()
        
        # Momentum
        v = momentum_decay * v + (1 - momentum_decay) * g
        v -= v.mean()
        
        # Step
        a_new = a - t * v
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            t *= 0.5
            continue
        a_new = a_new / s
        
        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            patience = 0
        else:
            patience += 1
            if patience % 50 == 0:
                t *= 0.9
                v *= 0.5
        
        step += 1
        if step % 500 == 0:
            t = max(0.01 * (0.99 ** (step // 500)), 0.0001)
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based iterative refinement (game-theoretic approach)
2. Momentum subgradient on exact max objective
3. L-BFGS with LogSumExp smoothing and temperature annealing
4. Multi-length exploration
5. Simulated annealing for escaping local minima
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    if n <= 1:
        return np.array([a[0]**2]) if n == 1 else np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_conv_matrix_sparse(a):
    n = len(a)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # Column j has nonzeros at rows j, j+1, ..., j+n-1 with values a[0], a[1], ..., a[n-1]
    n_rows = 2 * n - 1
    rows = np.concatenate([np.arange(j, j+n) for j in range(n)])
    cols = np.repeat(np.arange(n), n)
    vals = np.tile(a, n)
    return csr_matrix((vals, (rows, cols)), shape=(n_rows, n))

def solve_lp(a, M, time_limit=3.0):
    n = len(a)
    A = build_conv_matrix_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success:
            b = res.x
            s = b.sum()
            if s > 1e-10:
                return b / s
    except Exception:
        pass
    return None

def lp_refinement(a0, time_limit, max_iters=50):
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit * 0.8:
            break
        c = fast_convolve(a)
        M = c.max()
        
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        lp_time = min(remaining * 0.3, 5.0)
        b = solve_lp(a, M, time_limit=lp_time)
        if b is None:
            break
        
        # Try different step sizes
        improved = False
        for t in [0.1, 0.05, 0.02, 0.01, 0.005]:
            a_new = (1-t) * a + t * b
            a_new = normalize(a_new)
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                improved = True
                break
        if not improved:
            break
        
        if it % 10 == 0:
            print(f"    LP iter {it}: score={best_score:.8f}")
    
    return best_a, best_score

def subgradient_momentum(a0, time_limit):
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    
    v = np.zeros(n)
    lr = 0.02
    mu = 0.7
    t0 = time.monotonic()
    step = 0
    patience = 0

        c_max = c.max()

            active[idx] = True
            n_act = 1

        g = 2.0 * np.correlate(u, a, 'valid')

        s = a_new.sum()

            continue

        if sc < best_score - 1e-12:

            best_a = a_new.copy()
            a = a_new.copy()
            patience = 0
        else:
            patience += 1

                v *= 0.3
                if lr < 1e-6:
                    break
        
        step += 1

            lr = max(0.02 * (0.98 ** (step // 1000)), 1e-5)

def lse_objective_and_grad(a, T):
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    shifted = (c - c_max) / T
    exp_c = np.exp(shifted)
    lse = c_max + T * np.log(exp_c.sum())
    obj = 2.0 * n * lse
    
    w = exp_c / exp_c.sum()
    # Gradient: g[m] = 2 * sum_k w[k] * a[k-m] - a[m] * w[2m]
    N = 1 << (2*n).bit_length()
    W = rfft(w, N)
    A_fft = rfft(a, N)
    R = irfft(W * np.conj(A_fft), N)
    b = R[:n]
    
    g = 2.0 * b
    idx = 2 * np.arange(n)
    valid = idx < (2*n - 1)
    g[valid] -= a[valid] * w[idx[valid]]
    
    return obj, g

def lbgfs_refine(a0, time_limit, T_list):
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    z = np.log(np.maximum(a, 1e-300))
    z -= z.max()
    
    t0 = time.monotonic()
    
    def func_z(z_in):
        a_in = np.exp(z_in - z_in.max())
        a_in = a_in / a_in.sum()
        obj, _ = lse_objective_and_grad(a_in, T)
        return obj
    
    def jac_z(z_in):
        a_in = np.exp(z_in - z_in.max())
        a_in = a_in / a_in.sum()
        obj, grad_a = lse_objective_and_grad(a_in, T)
        S = np.dot(grad_a, a_in)
        grad_z = a_in * (grad_a - S)
        return grad_z
    
    for T in T_list:
        if time.monotonic() - t0 > time_limit * 0.9:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(func_z, z, jac=jac_z, method='L-BFGS-B',
                          options={'maxiter': max(10, int(remaining * 20)),
                                   'ftol': 1e-16, 'gtol': 1e-14, 'disp': False})
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = evaluate_local(a_new)

                best_score = sc

            pass

def simulated_annealing(a0, time_limit, rng):
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    
    T_start = 0.05
    T_end = 0.001
    t0 = time.monotonic()
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        progress = (time.monotonic() - t0) / time_limit
        T = T_start * (T_end / T_start) ** progress
        
        # Random perturbation
        perturbation = rng.normal(0, T, n)
        a_new = a + perturbation
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        
        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
        elif sc < best_score + T * best_score:
            # Accept with probability exp(-(sc - current) / T)
            if rng.random() < np.exp(-(sc - evaluate_local(a)) / (T + 1e-10)):
                a = a_new.copy()
        
        step += 1
        if step % 5000 == 0:
            print(f"    SA step {step}: T={T:.5f}, best={best_score:.8f}")
    
    return best_a, best_score

def diverse_init(n, rng):
    candidates = []
    x = np.linspace(0, 1, n)
    candidates.append(normalize(np.exp(-8 * (x - 0.5)**2)))
    candidates.append(normalize(0.5 * (1 + np.cos(2*np.pi*x))))
    candidates.append(normalize(np.sin(np.pi*x)**2))
    candidates.append(np.ones(n) / n)
    # Smooth random
    coeffs = rng.normal(0, 1, 20)
    a = np.zeros(n)
    for i, c in enumerate(coeffs):
        a += c * np.cos(2*np.pi*(i+1)*x + rng.uniform(0, 2*np.pi))
    a = np.maximum(a + 1.0, 0)
    candidates.append(normalize(a))
    return candidates

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return normalize(np.maximum(f_interp, 0))

  out[n] = sum_m u[m] * v[m-n+...] ... actually let me look at the docs.

From numpy docs:
`c_k(a, b) = \sum_n a_{n+k} \overline{b_n}`

So `np.correlate(u, a, 'full')[k] = sum_n u[n+k] * conj(a[n])` (for valid n).

With 'valid' mode, it returns only the values where there's no partial overlap. If len(u) >= len(a), it returns len(u) - len(a) + 1 values.

For u of length 2n-1 and a of length n:
'valid' output length = (2n-1) - n + 1 = n.
out[m] = sum_{n=0}^{n-1} u[n+m] * a[n] for m = 0, 1, ..., n-1.

This is exactly what we want: out[m] = sum_{j=0}^{n-1} u[j+m] * a[j] = sum_k u[k] * a[k-m] (where k=j+m).
