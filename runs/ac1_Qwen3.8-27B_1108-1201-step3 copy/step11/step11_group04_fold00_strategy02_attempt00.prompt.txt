def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()

        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):

def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z

def spectral_flatten(a, n_iters=50, blend=0.4, deadline=None):
    """
    Iterative spectral flattening (Gerchberg-Saxton style).
    Flattens the power spectrum of a while maintaining non-negativity.
    """
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)

        # Target: flatten magnitudes (keep DC component)
        mags = np.abs(F)
        # Use geometric mean of non-DC magnitudes as target
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        # Blend magnitudes toward target
        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])

        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

    return a


def peak_shave(a, time_limit, n_steps=100000):
    """
    Directly target the maximum of the autocorrelation using multiplicative updates.
    Focus on the top few peaks to reduce them fastest.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return None, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.02
    scale = 1.0
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Find top peaks (within 0.1% of max)
        threshold = max_b * (1.0 - 1e-3)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Compute weighted subgradient from active peaks
        weights = (b[active] - max_b * (1 - 1e-3)) / max(1e-30, max_b - max_b * (1 - 1e-3))
        weights = np.maximum(weights, 0)
        if weights.sum() < 1e-15:
            weights = np.ones(len(active))

        g = np.zeros(n)
        for i, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += weights[i] * a[k - idx]
        g /= max(1.0, weights.sum())

        # Multiplicative update
        eta_eff = eta * scale
        log_scale = -eta_eff * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 30:
            scale *= 0.7
            no_improve = 0
            if scale < 0.005:
                break

        if step % 20000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave {elapsed:.0f}s: score={best_score:.8f} scale={scale:.4f}")

    return best_a, best_score

def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        # Sinc-like
        center = n // 2
        t = (np.arange(n) - center) / center
        a = np.sinc(t)
        a = np.maximum(a, 0)
    else:
        # Triangle
        a = 1.0 - np.abs(2 * x - 1)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""

        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturbed_sequence(a, rng, strength=0.03):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 8)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s
