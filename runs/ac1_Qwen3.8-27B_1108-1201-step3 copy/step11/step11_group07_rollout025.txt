"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE-smoothed L-BFGS with aggressive multi-stage temperature annealing
2. Projected subgradient with Polyak-Ruppert averaging for global progress
3. Peak-targeted line search for final refinement
4. Spectral equalization (flatten power spectrum to reduce autocorr peaks)
5. Multi-length exploration with interpolation
6. Diverse restarts with structured perturbations
Key innovations: exact gradient for LSE objective, combined convex+nonconvex
approach, and aggressive final polishing for sub-0.1% improvements.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate score locally using FFT."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Log of a (for log-parameterization)."""
    return np.log(np.maximum(a, 1e-300))


def simplex_project(v):
    """Project onto probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    mu = np.sort(v)[::-1]
    cv = np.cumsum(mu)
    rho = np.arange(1, n + 1)
    cond = mu - (cv - 1.0) / rho > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho_star = np.max(rho[cond])
    theta = (cv[rho_star - 1] - 1.0) / rho_star
    return np.maximum(v - theta, 0)


def lse_grad_z(z, T):
    """Compute LSE objective and gradient in log-parameterization.
    a = softmax(z), objective = 2n * LSE(conv(a,a), T).
    Returns (obj, grad_z).
    """
    n = len(z)
    a = softmax_from_z(z)
    c = fast_convolve(a)  # length 2n-1

    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    exp_sum = exp_c.sum()
    w = exp_c / exp_sum  # softmax weights over convolution values
    lse = c_max + T * np.log(exp_sum)
    obj = 2.0 * n * lse

    # Gradient w.r.t. a: d(obj)/da[m] = 2n * 2 * sum_k w[k]*a[k-m] (valid indices)
    # This is 4n * (w cross-correlate a)[m]
    N = 1 << (len(c) + n - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    R = irfft(W * np.conj(A), N)
    grad_a = 4.0 * n * R[:n]

    # Transform to z-space: grad_z = a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def lbgfs_optimize(z0, time_limit, T_stages, maxiter_per_stage=5000):
    """L-BFGS with temperature annealing in log-space."""
    n = len(z0)
    z = z0.copy()
    t0 = time.monotonic()

    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        def func(z_in):
            obj, _ = lse_grad_z(z_in, stage_T)
            return obj

        def jac(z_in):
            _, g = lse_grad_z(z_in, stage_T)
            return g

        iters = min(maxiter_per_stage, max(20, int(remaining * 50)))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20,
                         'gtol': 1e-18, 'maxfun': iters * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        if stage_i % 5 == 0:
            a_check = softmax_from_z(z)
            sc = evaluate_local(a_check)
            print(f"    LSE stage {stage_i} (T={stage_T}): score={sc:.8f}")

    return softmax_from_z(z)


def subgradient_method(a_init, time_limit, n_iters=200000, alpha_init=0.5):
    """Projected subgradient with Polyak-Ruppert averaging."""
    n = len(a_init)
    a = normalize(a_init.copy())
    a_avg = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    alpha = alpha_init

    for t in range(1, n_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        c_max = c.max()

        # Find active peaks
        threshold = c_max * (1.0 - 5e-4)
        peak_indices = np.where(c >= threshold)[0]
        if len(peak_indices) == 0:
            peak_indices = np.array([np.argmax(c)])

        # Subgradient: g[m] = (1/|S|) sum_{k in S} 2*a[k-m] (valid)
        g = np.zeros(n)
        for k in peak_indices:
            km = k - np.arange(n)
            valid = (km >= 0) & (km < n)
            g[valid] += 2.0 * a[km[valid]]
        g /= len(peak_indices)

        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-20:
            break

        # Step size: eta = c_max / ||g||^2 (scaled)
        eta = alpha * c_max / (g_norm_sq + 1e-15)
        eta = min(eta, 0.05)

        a_new = a - eta * g
        a_new = simplex_project(a_new)

        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        # Polyak-Ruppert
        a_avg = (1.0 - 1.0 / t) * a_avg + (1.0 / t) * a_new
        a = a_new

        # Adaptive alpha decay
        if t % 10000 == 0:
            alpha *= 0.7

        if t % 50000 == 0:
            sc_avg = evaluate_local(a_avg)
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = a_avg.copy()
            print(f"    Subgrad iter {t}: best={best_score:.8f}, avg={sc_avg:.8f}")

    sc_avg = evaluate_local(a_avg)
    if sc_avg < best_score:
        best_score = sc_avg
        best_a = a_avg.copy()

    return best_a, best_score


def peak_targeted_refine(a_init, time_limit, n_iters=50000):
    """Targeted refinement: reduce the specific peak(s) of the auto-convolution."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    lr = 0.01

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        c_max = c.max()
        peak_idx = np.argmax(c)

        # Direction to reduce c[peak_idx]: negative subgradient
        km = peak_idx - np.arange(n)
        valid = (km >= 0) & (km < n)
        d = np.zeros(n)
        d[valid] = -a[km[valid]]

        # Also check nearby peaks and average their gradients
        nearby = np.argsort(c)[::-1][:5]
        d_avg = np.zeros(n)
        for k in nearby:
            km_k = k - np.arange(n)
            v_k = (km_k >= 0) & (km_k < n)
            d_avg[v_k] -= a[km_k[v_k]]
        d = d_avg / len(nearby)

        # Line search
        improved = False
        for scale in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            a_new = a + scale * d
            a_new = simplex_project(a_new)
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                improved = True
                break

        if not improved:
            lr *= 0.8
            if lr < 1e-8:
                break
        elif it % 100 == 0:
            lr = min(lr * 1.05, 0.05)

        if it % 1000 == 0 and it > 0:
            print(f"    PeakRefine iter {it}: score={best_score:.8f}, lr={lr:.6f}")

    return best_a, best_score


def spectral_equalize(a_init, time_limit, n_iters=10000):

    best_a = a.copy()

        if time.monotonic() - t0 > time_limit:

        P_mean = P.mean()

        H = np.zeros(fft_size // 2 + 1, dtype=np.complex128)
        valid = P > 1e-15
        H[valid] = -(P[valid] - P_mean[valid]) * np.conj(A[valid]) / (P[valid] + 1e-10)
        H[0] = 0.0
        h = irfft(H, fft_size)[:n]
        h -= h.mean()
        h_norm = np.linalg.norm(h)
        if h_norm < 1e-15:
            break
        h /= h_norm

        # Line search
        improved = False
        for scale in [lr, lr * 0.5, lr * 0.25]:
            a_new = a + scale * h
            a_new = simplex_project(a_new)
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                improved = True
                break

        if not improved:
            lr *= 0.85
            if lr < 1e-9:
                break

        if it % 2000 == 0 and it > 0:
            print(f"    Spectral iter {it}: score={best_score:.8f}, lr={lr:.6f}")

    return best_a, best_score


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    candidates = []
    x = np.linspace(0, 1, n)

    # Uniform
    candidates.append(np.ones(n) / n)

    # Hann window
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    candidates.append(normalize(a))

    # Blackman
    a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    a = np.maximum(a, 0)
    candidates.append(normalize(a))

    # Gaussian center
    a = np.exp(-4 * (x - 0.5) ** 2)
    candidates.append(normalize(a))

    # Edge-peaked
    a = np.exp(-6 * x) + np.exp(-6 * (1 - x))
    candidates.append(normalize(a))

    # Two bumps
    a = np.exp(-12 * (x - 0.3) ** 2) + np.exp(-12 * (x - 0.7) ** 2)
    candidates.append(normalize(a))

    # Random smooth (low-freq Fourier)
    coeffs = rng.normal(0, 1, 15)
    phases = rng.uniform(0, 2 * np.pi, 15)
    a = np.ones(n) * 0.5
    for i, (c, p) in enumerate(zip(coeffs, phases)):
        a += c * 0.1 * np.cos(2 * np.pi * (i + 1) * x + p)
    a = np.maximum(a, 0)
    candidates.append(normalize(a))

    # Exponential random (high dynamic range)
    a = rng.exponential(1.0, n)
    candidates.append(normalize(a))

    # Binary-like (m-sequence inspired)
    # Use a simple LFSR-like pattern
    bit = np.zeros(n, dtype=int)
    state = rng.integers(1, 256)
    for i in range(n):
        bit[i] = state & 1
        new_bit = ((state >> 7) ^ (state >> 3) ^ (state >> 2) ^ (state >> 1)) & 1
        state = ((state << 1) | new_bit) & 0xFF
    a = 0.3 + 0.7 * bit.astype(float)
    candidates.append(normalize(a))

    # Sine-modulated
    a = 0.5 + 0.4 * np.sin(2 * np.pi * 3 * x) + 0.1 * np.sin(2 * np.pi * 7 * x)
    a = np.maximum(a, 0)
    candidates.append(normalize(a))

    return candidates


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    return normalize(f_interp)


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Combines LSE+L-BFGS, subgradient, peak-targeting, spectral equalization,
    and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        candidates = diverse_init(n_init, rng)
        scores = [evaluate_local(c) for c in candidates]
        best_idx = np.argmin(scores)
        best_seq = candidates[best_idx].copy()
        best_score = scores[best_idx]
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting optimization: n={n_best}, score={best_score:.8f}")

    # === Phase 1: Subgradient for global progress ===
    print(f"\n=== Phase 1: Subgradient (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_sub, _ = subgradient_method(best_seq, t_alloc, n_iters=300000, alpha_init=0.3)
        local_update(a_sub, "subgrad")

    # === Phase 2: Aggressive LSE L-BFGS ===
    print(f"\n=== Phase 2: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 450.0)
    if t_alloc > 50:
        T_stages = [100, 50, 30, 20, 15, 10, 7, 5, 4, 3, 2.5, 2, 1.5, 1, 0.8, 0.6, 0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.08, 0.06, 0.05, 0.04, 0.03, 0.02, 0.01, 0.008, 0.005, 0.003, 0.002, 0.001]
        z0 = z_from_a(best_seq)
        print(f"  T_stages ({len(T_stages)}), budget={t_alloc:.0f}s")
        a_lb = lbgfs_optimize(z0, t_alloc, T_stages, maxiter_per_stage=5000)
        local_update(a_lb, "lbgfs_1")

    # === Phase 3: Peak-targeted refinement ===
    print(f"\n=== Phase 3: Peak Targeting (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_peak, _ = peak_targeted_refine(best_seq, t_alloc, n_iters=100000)
        local_update(a_peak, "peak_target")

    # === Phase 4: Spectral equalization ===
    print(f"\n=== Phase 4: Spectral Equalization (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_spec, _ = spectral_equalize(best_seq, t_alloc, n_iters=20000)
        local_update(a_spec, "spectral")

    # === Phase 5: Fine LSE polish ===
    print(f"\n=== Phase 5: Fine LSE Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 250.0)
    if t_alloc > 30:
        T_fine = [5, 3, 2, 1.5, 1, 0.7, 0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.08, 0.06, 0.05, 0.04, 0.03, 0.02, 0.015, 0.01, 0.008, 0.005, 0.003, 0.002, 0.001, 0.0005]
        z0 = z_from_a(best_seq)
        print(f"  T_stages ({len(T_fine)}), budget={t_alloc:.0f}s")
        a_lb2 = lbgfs_optimize(z0, t_alloc, T_fine, maxiter_per_stage=3000)
        local_update(a_lb2, "lbgfs_fine")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_restarts = 4
        t_per_restart = min(rem / n_restarts, 60.0)
        for r in range(n_restarts):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 20:
                break
            # Create perturbed version
            perturbation = rng.normal(0, 0.03, n_best)
            a_pert = normalize(best_seq + perturbation)
            # Short LSE optimization
            T_short = [10, 5, 3, 2, 1, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01]
            z0 = z_from_a(a_pert)
            t_sub = min(rem_sub * 0.4, 40.0)
            a_r = lbgfs_optimize(z0, t_sub, T_short, maxiter_per_stage=2000)
            local_update(a_r, f"restart_{r}")

    # === Phase 7: Length exploration ===
    print(f"\n=== Phase 7: Length Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for n_try in [n_best + 200, n_best - 200, n_best + 400, n_best + 100]:
            if n_try < 200 or n_try > 5000:
                continue
            rem_sub = deadline - time.monotonic()
            if rem_sub < 30:
                break
            init = interpolate_sequence(best_seq, n_try)
            t_sub = min(rem_sub * 0.3, 50.0)
            T_med = [20, 10, 5, 3, 2, 1, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]
            z0 = z_from_a(init)
            a_try = lbgfs_optimize(z0, t_sub, T_med, maxiter_per_stage=2000)
            local_update(a_try, f"length_{n_try}")

    # === Phase 8: Ultra-fine final polish ===
    print(f"\n=== Phase 8: Ultra-Fine Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        T_ultra = [1, 0.7, 0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.08, 0.06, 0.05, 0.04, 0.03, 0.025, 0.02, 0.015, 0.01, 0.008, 0.006, 0.005, 0.004, 0.003, 0.002, 0.0015, 0.001]
        z0 = z_from_a(best_seq)
        t_sub = min(rem * 0.6, 60.0)
        a_fin = lbgfs_optimize(z0, t_sub, T_ultra, maxiter_per_stage=3000)
        local_update(a_fin, "ultra_fine")

    # === Phase 9: Peak targeting on final best ===
    rem = deadline - time.monotonic()
    if rem > 20:
        a_pt, _ = peak_targeted_refine(best_seq, min(rem * 0.5, 40.0), n_iters=50000)
        local_update(a_pt, "final_peak")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
