   import numpy as np
   import time
   from scipy.fft import rfft, irfft

   def fast_convolve(a):
       n = len(a)
       size = 2 * n - 1
       fft_size = 1 << (size - 1).bit_length()
       A = rfft(a, fft_size)
       return irfft(A * A, fft_size)[:size]

   def simplex_project(v):
       """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
       n = len(v)
       if n == 0:
           return v
       u = np.sort(v)[::-1]
       cssv = np.cumsum(u)
       rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0]
       if len(rho) == 0:
           return np.ones(n) / n
       rho = rho[-1]
       theta = (cssv[rho] - 1) / (rho + 1)
       return np.maximum(v - theta, 0)

   def subgradient_at(a, k):
       """Compute subgradient of 2n*max_b at lag k. Result is 4n * a[k::-1] (truncated)."""
       n = len(a)
       g = np.zeros(n)
       # g[i] = 4n * a[k - i] for valid k-i
       for i in range(n):
           m = k - i
           if 0 <= m < n:
               g[i] = 4 * n * a[m]
       return g

   # Vectorized version:
   def subgradient_at_fast(a, k, n):
       g = np.zeros(n)
       js = max(0, k - n + 1)
       je = min(n - 1, k)
       idx = np.arange(js, je + 1)
       g[idx] = 4.0 * n * a[k - idx]
       return g

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = next_fast_len(size)
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def evaluate_local(a):

    n = len(a)

    return float(2.0 * n * b.max() / (s * s))

    """Find a working LP solver."""

            if prob.status in ('optimal', 'optimal_inaccurate'):

def build_conv_matrix(a):
    """Build dense convolution matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        for k in range(j, min(j + n, size)):
            T[k, j] = a[k - j]
    return T


def solve_lp_best_response(a_fixed, T_mat, solver, max_solvers=3):
    """LP: min M s.t. T@b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]
    for s in solvers_to_try[:max_solvers]:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                sv = b_val.sum()
                if sv < 1e-15:
                    return None
                return b_val / sv
        except Exception:
            continue
    return None

        for wi, k in zip(w_act, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += wi * 2.0 * y[k - idx]

        # Scale gradient for MW (objective is 2n*max_b, so grad is 2n * 2*y[...] = 4n*y[...])
        g *= 2.0 * n

        # Nesterov step size
        step = eta / t
        log_mult = -step * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Nesterov extrapolation for y
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = t / t_new - 1.0
        y_new = a_new + momentum * (a_new - a)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-300:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        # Evaluate
        b_new = fast_convolve(a_new)
        score_new = 2.0 * n * b_new.max()

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            stall = 0
            # Slight eta increase on improvement
            last_eta = min(last_eta * 1.005, 0.5)
        else:
            stall += 1
            if stall > 50000:
                last_eta *= 0.9
                stall = 0

        eta = last_eta
        a = a_new
        y = y_new
        t = t_new

        if (it + 1) % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW it={it+1} ({elapsed:.0f}s): score={best_score:.8f} eta={eta:.6f} t={t:.0f}")

    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver, palindromic=False):
    """LP alternating with Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)
    lp_times = []
    T_mat = build_conv_matrix(y)

    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new = solve_lp_best_response(y, T_mat, solver, max_solvers=3)
        lp_elapsed = time.monotonic() - lp_t0
        lp_times.append(lp_elapsed)
        if b_new is None:
            continue

        if len(lp_times) > 20:
            avg_lp = np.mean(lp_times[-20:])
            if avg_lp > 5.0:
                print(f"    LP too slow ({avg_lp:.2f}s/iter), stopping at iter {it+1}")
                break

        alpha = 2.0 / (it + 2)
        y_old = y.copy()
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = y_old
        else:
            y = y / sy
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / ((it + 1) + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        T_mat = build_conv_matrix(y)

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-12:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 8:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 6)
                phase = rng.uniform(0, 2 * np.pi)
                noise = 0.015 * np.cos(freq * x + phase)
                y = np.maximum(y + noise, 0)
                y = y / y.sum()
                T_mat = build_conv_matrix(y)
                stall_count = 0
            if (it + 1) % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter={it+1} ({elapsed:.0f}s, {lp_elapsed:.3f}s/solve): score={best_score:.8f}")

    return best_a, best_score


def lse_logsumexp_grad(z, n, T):
    """Compute LSE objective and gradient w.r.t. z (log-domain parameterization)."""
    a_shifted = z - z.max()
    exp_z = np.exp(a_shifted)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T
    scaled_shifted = scaled - scaled.max()
    exp_terms = np.exp(scaled_shifted)
    lse = max_b + T * (scaled.max() + np.log(exp_terms.sum() + 1e-300))
    obj = 2.0 * n * lse
    w = exp_terms / exp_terms.sum()
    # Gradient w.r.t. a
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * 2.0 * a[k - idx]
    grad_a *= 2.0 * n
    # Chain rule to z
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


class LSEFunc:
    def __init__(self, n, T, deadline):
        self.n = n
        self.T = T
        self.deadline = deadline

    def __call__(self, z):
        if time.monotonic() > self.deadline:
            return 1e15
        obj, _ = lse_logsumexp_grad(z, self.n, self.T)
        return obj


class LSEJac:
    def __init__(self, n, T, deadline):
        self.n = n
        self.T = T
        self.deadline = deadline

    def __call__(self, z):
        if time.monotonic() > self.deadline:
            return np.zeros(len(z))
        _, g = lse_logsumexp_grad(z, self.n, self.T)
        return g


def lse_lbfgs_refine(a0, time_limit, T_stages=None, maxiter_per_stage=300):
    """L-BFGS with LogSumExp smoothing and temperature annealing in log-domain."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        func = LSEFunc(n, stage_T, deadline)
        jac = LSEJac(n, stage_T, deadline)
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': maxiter_per_stage, 'ftol': 1e-30,
                                    'gtol': 1e-20, 'maxfun': maxiter_per_stage * 200,
                                    'disp': False})
            z = res.x
        except Exception:
            pass
    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out


def make_palindromic(a):
    """Make a sequence palindromic."""
    n = len(a)
    a_pal = (a + a[::-1]) / 2.0
    a_pal = np.maximum(a_pal, 0)
    s = a_pal.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_pal / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturb_sequence(a, rng, strength=0.02):
    """Create a perturbed version with structured frequency components."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq1 = rng.integers(1, 10)
    freq2 = rng.integers(3, 15)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    pert = strength * (np.cos(freq1 * x + p1) + 0.5 * np.sin(freq2 * x + p2))
    a_pert = np.maximum(a + pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses accelerated MW, LP alternating, multi-length exploration,
    and LSE L-BFGS for final polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {budget_s:.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = np.ones(n_init) / n_init
        # Add structured perturbation
        x = np.linspace(0, 1, n_init)
        best_seq = best_seq * (1.0 + 0.3 * np.cos(2 * np.pi * 3 * x))
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            n_best = len(a)
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={n_best}) ***")

    # === Phase 1: Accelerated MW on parent (20% time) ===
    print(f"\n=== Phase 1: Accel MW n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.20, 300.0)
    if t1 > 20:
        print(f"  Budget: {t1:.0f}s")
        a_mw, _ = accelerated_mw_topk(best_seq, t1, top_k=8, eta_init=0.03)
        try_update(a_mw, "mw_accel")
        # Second MW pass with smaller eta
        rem = deadline - time.monotonic()
        if rem > 20:
            t1b = min(rem * 0.10, 120.0)
            a_mw2, _ = accelerated_mw_topk(best_seq, t1b, top_k=4, eta_init=0.01)
            try_update(a_mw2, "mw_fine")

    # === Phase 2: LP alternating (main workhorse, 30% time) ===
    print(f"\n=== Phase 2: LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.30, 450.0)
    if t2 > 60:
        print(f"  Budget: {t2:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t2, solver)
        try_update(a_lp, "lp_main")

    # === Phase 3: Palindromic LP (10% time) ===
    print(f"\n=== Phase 3: Palindromic LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t3 = min(rem * 0.10, 150.0)
        a_pal = make_palindromic(best_seq)
        a_pal_lp, _ = nesterov_lp_refine(a_pal, t3, solver)
        try_update(a_pal_lp, "pal_lp")

    # === Phase 4: Multi-length exploration (25% time) ===
    print(f"\n=== Phase 4: Multi-length (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    lengths_to_try = [1000, 1500, 2000]
    for ln in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 120:
            break
        if len(best_seq) == ln:
            continue
        t_ln = min(rem * 0.15, 200.0)
        if t_ln < 40:
            continue
        print(f"  Trying n={ln}, budget={t_ln:.0f}s")
        init_ln = interpolate_sequence(best_seq, ln)
        # Quick MW
        t_mw = t_ln * 0.4
        a_ln_mw, _ = accelerated_mw_topk(init_ln, t_mw, top_k=8, eta_init=0.03)
        try_update(a_ln_mw, f"mw_n{ln}")
        # Then LP
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.3, t_ln * 0.5)
        if t_lp > 30:
            a_ln_lp, _ = nesterov_lp_refine(a_ln_mw, t_lp, solver)
            try_update(a_ln_lp, f"lp_n{ln}")

    # === Phase 5: LSE L-BFGS on best (15% time) ===
    print(f"\n=== Phase 5: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t5 = min(rem * 0.15, 250.0)
        T_stages = [30.0, 15.0, 8.0, 4.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
        print(f"  {len(T_stages)} stages, budget={t5:.0f}s")
        a_lse = lse_lbfgs_refine(best_seq, t5, T_stages, maxiter_per_stage=200)
        try_update(a_lse, "lse_fine")

    # === Phase 6: Perturbation escape + re-optimize (10% time) ===
    print(f"\n=== Phase 6: Perturbation escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        for trial in range(3):
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            a_pert = perturb_sequence(best_seq, rng, strength=0.01 + 0.008 * trial)
            t_sub = min(rem * 0.25, 80.0)
            # Quick MW then LP
            a_p_mw, _ = accelerated_mw_topk(a_pert, t_sub * 0.4, top_k=6, eta_init=0.02)
            try_update(a_p_mw, f"pert{trial}_mw")
            rem = deadline - time.monotonic()
            if rem > 30:
                a_p_lp, _ = nesterov_lp_refine(a_p_mw, min(rem * 0.4, t_sub * 0.5), solver)
                try_update(a_p_lp, f"pert{trial}_lp")

    # === Phase 7: Final LP polish (5% time) ===
    print(f"\n=== Phase 7: Final LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t7 = min(rem * 0.3, 100.0)
        a_fin_lp, _ = nesterov_lp_refine(best_seq, t7, solver)
        try_update(a_fin_lp, "final_lp")

    # === Phase 8: Final MW micro-polish ===
    print(f"\n=== Phase 8: Final MW (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t8 = min(rem * 0.4, 50.0)
        a_fin_mw, _ = accelerated_mw_topk(best_seq, t8, top_k=3, eta_init=0.005)
        try_update(a_fin_mw, "final_mw")

    # === Phase 9: Very fine LSE ===
    rem = deadline - time.monotonic()
    if rem > 30:
        t9 = min(rem * 0.5, 60.0)
        T_stages_fine = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
        a_fine = lse_lbfgs_refine(best_seq, t9, T_stages_fine, maxiter_per_stage=150)
        try_update(a_fine, "lse_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
