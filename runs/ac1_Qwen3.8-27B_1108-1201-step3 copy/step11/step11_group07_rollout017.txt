Actually, I realize there might be an issue with the L-BFGS approach when using the log-parameterization. The gradient in the z-space is:
d(obj)/dz[j] = a[j] * (d(obj)/da[j] - sum_k a[k] * d(obj)/da[k])

This ensures the gradient is tangent to the simplex.

Let me also think about the "exact peak tracking" approach more carefully:

At each step:
1. Compute c = conv(a, a)
2. Find the set S of indices where c[k] = max(c) (within some tolerance)
3. Compute the subgradient: g[j] = (1/|S|) * sum_{k in S} 2*a[k-j] (for valid k-j)
4. Take a step: a_new = a - lr * g, then project to the simplex (or use multiplicative update)
5. Restore normalization

The multiplicative update would be:
a_new[j] = a[j] * exp(-lr * g[j])
Then normalize.

This is essentially mirror descent on the simplex with the negative entropy barrier.

For the line search, I can try a few step sizes and pick the best one.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Aggressive LSE annealing with L-BFGS in log-space
2. Exact peak-tracking mirror descent polish
3. Spectral domain auxiliary optimization
4. Multi-length exploration with interpolation
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def evaluate_local(a):
    """Evaluate the objective for a non-negative sequence."""
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def lse_obj_z(z, T):
    """LSE-smoothed objective in log-space. z are log-values, a = softmax(z)."""
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    c = fast_convolve(a)
    c_max = c.max()
    shifted = (c - c_max) / T
    shifted -= shifted.max()
    exp_shifted = np.exp(shifted)
    lse = c_max + T * (shifted.max() + np.log(exp_shifted.sum()))
    obj = 2.0 * n * lse
    return obj, a, lse


def lse_grad_z(z, T):
    """Gradient of LSE-smoothed objective w.r.t. z (log-space)."""
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    c = fast_convolve(a)
    c_max = c.max()
    shifted = (c - c_max) / T
    shifted -= shifted.max()
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    
    # grad_a[j] = 4n * sum_m w[m+j] * a[m]
    # Using FFT: grad_a = 4n * IFFT(conj(FFT(a)) * FFT(w))[:n]
    N = 1 << (2 * n).bit_length()
    A = rfft(a, N)
    W = rfft(w, N)
    grad_a = 4.0 * n * irfft(np.conj(A) * W, N)[:n]
    
    # Transform to z-space: grad_z = a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return grad_z, a


def lbgfs_aggressive(a0, time_limit, T_stages):
    """L-BFGS optimization with LSE smoothing and aggressive annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    
    for stage_idx, T in enumerate(T_stages):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        remaining = time_limit - elapsed
        if remaining < 5:
            break
        
        def func(z_in):
            obj, _, _ = lse_obj_z(z_in, T)
            return obj
        
        def jac(z_in):
            gz, _ = lse_grad_z(z_in, T)
            return gz
        
        # Scale iterations based on remaining time
        maxiter = max(50, int(remaining * 20))
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={
                    'maxiter': maxiter,
                    'maxfun': maxiter * 500,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'disp': False
                }
            )
            z = res.x
            _, a_new, _ = lse_obj_z(z, T)
            sc = evaluate_local(a_new)
            if stage_idx % 3 == 0:
                print(f"    LSE T={T}: score={sc:.10f}, iters={res.nit}")
        except Exception as e:
            print(f"    LSE T={T} failed: {e}")
            break
    
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    return a


def exact_peak_polish(a0, time_limit, n_steps=50000, eta0=0.1):
    """Mirror descent (multiplicative weights) using exact peak subgradient."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = eta0
    stuck_count = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        c_max = c.max()
        
        # Find active set (peaks within tolerance of max)
        tol = max(1e-8, c_max * 1e-6)
        active = np.where(c >= c_max - tol)[0]
        
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        
        # Compute subgradient: g[j] = (1/|S|) * sum_{k in S} 2*a[k-j] (valid k-j)
        g = np.zeros(n)
        for k in active:
            # g[j] += 2*a[k-j] for j where 0 <= k-j < n
            # i.e., j = k - m for m = 0, ..., n-1 (where j is valid)
            m_vals = np.arange(n)
            j_vals = k - m_vals
            valid = (j_vals >= 0) & (j_vals < n)
            g[valid] += 2.0 * a[m_vals[valid]]
        g /= len(active)
        
        # Multiplicative update with line search
        improved = False
        for scale in [1.0, 0.5, 0.25, 0.1, 0.05]:
            log_update = -eta * scale * g
            log_update -= log_update.max()  # numerical stability
            a_new = a * np.exp(log_update)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                improved = True
                stuck_count = 0
            break
        
        if not improved:
            stuck_count += 1
            if stuck_count % 50 == 0:
                eta *= 0.8
                stuck_count = 0
            if eta < 1e-8:
                break
        
        if step % 5000 == 0 and step > 0:
            print(f"    PeakPolish step={step}: score={best_score:.10f}, eta={eta:.6f}")
    
    return best_a, best_score


def spectral_flatten(a0, time_limit, n_iters=5000):
    """Flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    t0 = time.monotonic()
    lr = 0.003
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = rfft(a, fft_size)
        P = np.abs(A) ** 2
        # We want |FFT(a)|^2 to be flat => autocorrelation is as concentrated at lag 0 as possible
        # But we want to MINIMIZE the peak of autocorrelation, not maximize it!
        # So we want |FFT(a)|^2 to be... hmm.
        # 
        # autocorrelation = IFFT(|FFT(a)|^2)
        # If |FFT(a)|^2 is flat, autocorrelation is a delta => peak is high!
        # If |FFT(a)|^2 is concentrated at low frequencies, autocorrelation is spread out.
        # 
        # So we actually want |FFT(a)|^2 to be concentrated at LOW frequencies!
        # This makes the autocorrelation spread out (more like a smooth bump).
        
        # Alternative: directly minimize the variance of the autocorrelation
        c = fast_convolve(a)
        c_mean = c.mean()
        
        # Gradient of (c - c_mean)^2 sum w.r.t. a
        # This is equivalent to minimizing the variance of c
        # d(sum (c[k]-mean)^2)/da = 2 * sum_k (c[k]-c_mean) * dc[k]/da
        # dc[k]/da[j] = 2*a[k-j] (for valid)
        # So grad[j] = 2 * sum_k (c[k]-c_mean) * 2*a[k-j] = 4 * sum_k (c[k]-c_mean) * a[k-j]
        
        # Using FFT: grad = 4 * IFFT(FFT(c - c_mean) * conj(FFT(a)))... 
        # Actually, sum_k (c[k]-c_mean) * a[k-j] is a cross-correlation.
        
        r = c - c_mean
        R = rfft(r, fft_size)
        A_fft = rfft(a, fft_size)
        var_grad = 4.0 * irfft(R * np.conj(A_fft), fft_size)[:n]
        
        # Also add a term to reduce the max (using LSE with very low T)
        T = 0.01
        c_max = c.max()
        shifted = (c - c_max) / T
        shifted -= shifted.max()
        w = np.exp(shifted)
        w /= w.sum()
        W = rfft(w, fft_size)
        max_grad = 4.0 * n * irfft(np.conj(A_fft) * W, fft_size)[:n]
        
        total_grad = var_grad + 2.0 * max_grad
        total_grad -= total_grad.mean()  # preserve normalization direction
        
        g_norm = np.linalg.norm(total_grad)
        if g_norm < 1e-15:
            break
        total_grad /= g_norm
        
        improved = False
        for scale in [lr, lr * 0.5, lr * 0.1]:
            a_new = a - scale * total_grad
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new /= s
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                improved = True
            break
        
        if it % 500 == 0:
            lr *= 0.85
    
    return best_a, best_score


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    candidates = []
    x = np.linspace(0, 1, n)
    
    # Various smooth profiles
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        a = (x * (1 - x)) ** alpha
        candidates.append(normalize(a))
    
    # Cosine-based
    for freq in [1, 2, 3]:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * freq * x)
        a = np.maximum(a, 0)
        if a.sum() > 0:
            candidates.append(normalize(a))
    
    # Gaussian bumps
    for center in [0.3, 0.5, 0.7]:
        for width in [0.05, 0.1, 0.2]:
            a = np.exp(-((x - center) / width) ** 2)
            candidates.append(normalize(a))
    
    # Harmonic (from strategy)
    for omega in [2, 3, 5, 8, 13]:
        a = np.abs(np.sin(omega * np.pi * x))
        candidates.append(normalize(a))
    
    # Random smooth
    for _ in range(5):
        coeffs = rng.normal(0, 1, 8)
        a = np.zeros(n)
        for i, c_val in enumerate(coeffs):
            a += c_val * np.cos(2 * np.pi * (i + 1) * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a + 0.3, 0)
        candidates.append(normalize(a))
    
    # Uniform
    candidates.append(np.ones(n) / n)
    
    return candidates


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    return normalize(f_interp)
