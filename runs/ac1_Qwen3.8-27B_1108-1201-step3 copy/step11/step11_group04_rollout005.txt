    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraint_obj = T_mat @ b_var <= M_var
    constraints = [constraint_obj, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS, cp.GUROBI, cp.MOSEK]
    solved = False
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue

    if not solved:
        return None, None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s

    M_val = float(max(0, M_var.value)) if M_var.value is not None else float(fast_convolve(a_fixed) @ b_val).max()

    y_dual = None
    try:
        if constraint_obj.dual_value is not None:
            y_dual = np.asarray(constraint_obj.dual_value, dtype=np.float64)
            y_dual = np.maximum(y_dual, 0)
            y_sum = y_dual.sum()
            if y_sum > 1e-15:
                y_dual = y_dual / y_sum
            else:
                y_dual = None
    except Exception:
        y_dual = None

    return b_val, M_val, y_dual


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T_temp
    scaled -= scaled.max()
    exp_terms = np.exp(scaled)
    exp_sum = exp_terms.sum()
    lse = max_b + T_temp * (scaled.max() + np.log(exp_sum))
    obj = 2.0 * n * lse

    w = exp_terms / exp_sum
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (2.0 * n * (grad_a - S))
    return obj, grad_z


def spectral_flatten(a, n_iters=50, blend=0.4, deadline=None):
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)

        mags = np.abs(F)
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])

        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

    return a


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage, deadline=None):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit
    n_stages = len(T_stages)
    iters_per_stage = max(30, maxiter_per_stage // n_stages)
    current_best = float('inf')

    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break

        stages_left = max(1, n_stages - stage_i)
        iters = max(20, min(iters_per_stage * 2, int(remaining * 30 / stages_left)))

        def func_scoped_z(z_in):
            obj, _ = logsumexp_obj_grad(z_in, n, stage_T)
            return obj

        def jac_scoped_z(z_in):
            _, g = logsumexp_obj_grad(z_in, n, stage_T)
            return g

        try:
            res = minimize(
                fun=func_scoped_z, x0=z, jac=jac_scoped_z, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-25,
                         'gtol': 1e-20, 'maxfun': iters * 200, 'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = 2.0 * n * fast_convolve(a_check).max()
            if sc < current_best:
                current_best = sc
        except Exception:
            pass

    return softmax_from_z(z)


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=1000000, use_dual=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(98765)
    lp_times = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new, M_b, y_dual = solve_lp_best_response(y, solver)
        lp_times.append(time.monotonic() - lp_t0)

        if b_new is None:
            continue

        if len(lp_times) >= 30:
            avg_lp_time = np.mean(lp_times[-30:])
            if avg_lp_time > 8.0 and it > 200:
                print(f"    LP too slow ({avg_lp_time:.2f}s/iter), stopping at iter {it+1}")
                break

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 6:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 5)
                noise = 0.02 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                if use_dual and y_dual is not None:
                    shift = y_dual[:n] * 0.1
                    y = y + shift * (y > 0).astype(float)
                    y = np.maximum(y, 0)
                    sy = y.sum()
                    if sy > 1e-15:
                        y = y / sy
                stall_count = 0

            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def peak_shave(a, time_limit, n_steps=100000):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return None, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.03
    scale = 1.0
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        threshold = max_b * (1.0 - 5e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        eta_eff = eta * scale
        log_scale = -eta_eff * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 40:
            scale *= 0.75
            no_improve = 0
            if scale < 0.003:
                break

        if step % 20000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave {elapsed:.0f}s: score={best_score:.8f} scale={scale:.4f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        center = n // 2
        t = (np.arange(n) - center) / center
        a = np.sinc(t)
        a = np.maximum(a, 0)
    else:
        a = 1.0 - np.abs(2 * x - 1)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturbed_sequence(a, rng, strength=0.03):
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 8)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def dual_guided_refine(a0, time_limit, solver, n_max_iters=200000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b, y_dual = solve_lp_best_response(a, solver)
        if b_new is None:
            break

        if y_dual is not None:
            shift_n = min(n, len(y_dual))
            y_shifted = np.zeros(n)
            y_shifted[:shift_n] = y_dual[:shift_n]
            y_shifted_sum = y_shifted.sum()
            if y_shifted_sum > 1e-15:
                y_shifted = y_shifted / y_shifted_sum
                blend = 0.3
                a_new = (1 - blend) * a + blend * (b_new * (1 + y_shifted))
                a_new = np.maximum(a_new, 0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a_new = a_new / s_new
                    score = 2.0 * n * fast_convolve(a_new).max()
                    if score < best_score:
                        best_score = score
                        best_a = a_new.copy()
                    a = a_new

        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()
            a = b_new.copy()

        if (it + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DualGuide iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    rem = deadline - time.monotonic()
    print(f"Total budget: {rem:.0f}s, starting score: {best_score:.8f}")

    # Phase 0: Spectral initialization
    print(f"\n=== Phase 0: Spectral Init (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        t_spec0 = min(rem * 0.02, 25.0)
        a_spec0 = spectral_flatten(best_seq, n_iters=100, blend=0.35,
                                   deadline=time.monotonic() + t_spec0)
        local_update(a_spec0, "spec_init")
        a_spec0b = spectral_flatten(best_seq, n_iters=50, blend=0.5,
                                    deadline=time.monotonic() + min(t_spec0, 10))
        local_update(a_spec0b, "spec_init_strong")

    # Phase 1: LogSumExp L-BFGS with adaptive temperature
    print(f"=== Phase 1: LogSumExp L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lbfgs1 = min(rem * 0.18, 250.0)
    if t_lbfgs1 > 20:
        print(f"  Budget: {t_lbfgs1:.0f}s")
        T_stages_1 = [5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.15,
                      0.1, 0.07, 0.05, 0.035, 0.025, 0.015, 0.01, 0.007, 0.005, 0.003]
        a_lb1 = lbgfs_refine(best_seq, t_lbfgs1, T_stages_1, maxiter_per_stage=15000,
                             deadline=time.monotonic() + t_lbfgs1)
        local_update(a_lb1, "lbgfs_1")

    # Phase 2: Nesterov LP (main workhorse)
    print(f"=== Phase 2: Nesterov LP n={len(best_seq)} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lp1 = min(rem * 0.28, 450.0)
    if t_lp1 > 30:
        print(f"  Budget: {t_lp1:.0f}s")
        a_lp1, _ = nesterov_lp_refine(best_seq, t_lp1, solver, n_max_iters=1000000, use_dual=True)
        local_update(a_lp1, "nesterov_lp_1")

    # Phase 3: Peak shaving
    print(f"=== Phase 3: Peak Shaving (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_peak = min(rem * 0.06, 80.0)
    if t_peak > 10:
        a_pk, _ = peak_shave(best_seq, t_peak, n_steps=100000)
        local_update(a_pk, "peak_shave_1")

    # Phase 4: Dual-guided refinement
    print(f"=== Phase 4: Dual-Guided (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_dual = min(rem * 0.12, 150.0)
    if t_dual > 20:
        print(f"  Budget: {t_dual:.0f}s")
        a_dual, _ = dual_guided_refine(best_seq, t_dual, solver, n_max_iters=100000)
        local_update(a_dual, "dual_guided")

    # Phase 5: LP round 2
    print(f"=== Phase 5: Nesterov LP Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lp2 = min(rem * 0.18, 250.0)
    if t_lp2 > 20:
        print(f"  Budget: {t_lp2:.0f}s")
        a_pert = perturbed_sequence(best_seq, rng, strength=0.01)
        a_lp2, _ = nesterov_lp_refine(a_pert, t_lp2, solver, n_max_iters=500000, use_dual=True)
        local_update(a_lp2, "nesterov_lp_2")

    # Phase 6: Try different n values
    print(f"=== Phase 6: Different n (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        for n_try in [2000, 1000]:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_n = min(rem * 0.25, 150.0)
            if t_n < 30:
                continue
            print(f"  Trying n={n_try}, budget={t_n:.0f}s")
            init_n = interpolate_sequence(best_seq, n_try)
            a_sp_n = spectral_flatten(init_n, n_iters=30, blend=0.3)
            a_n, _ = nesterov_lp_refine(a_sp_n, t_n * 0.75, solver, n_max_iters=100000)
            local_update(a_n, f"lp_n{n_try}")

            rem = deadline - time.monotonic()
            if best_seq is not None and len(best_seq) == n_try and rem > 20:
                T_stages_n = [3.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
                a_n2 = lbgfs_refine(best_seq, min(rem, 50.0), T_stages_n,
                                     maxiter_per_stage=5000, deadline=time.monotonic() + min(rem, 50.0))
                local_update(a_n2, f"lbgfs_n{n_try}")

    # Phase 7: Final polish
    print(f"=== Phase 7: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.3, 50.0)
        if t_fin > 10:
            T_stages_fin = [2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003]
            a_fin = lbgfs_refine(best_seq, t_fin, T_stages_fin, maxiter_per_stage=8000,
                                 deadline=time.monotonic() + t_fin)
            local_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_ps = min(rem * 0.3, 30.0)
        if t_ps > 5:
            a_ps, _ = peak_shave(best_seq, t_ps, n_steps=50000)
            local_update(a_ps, "final_peak")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_flp = min(rem * 0.5, 30.0)
        if t_flp > 5:
            a_flp, _ = nesterov_lp_refine(best_seq, t_flp, solver, n_max_iters=30000)
            local_update(a_flp, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}) at {elapsed:.1f}s")
        test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            eval_score = evaluate_sequence(test_seq)
            print(f"Verified with evaluate_sequence: {eval_score:.8f}")
        except Exception as e:
            print(f"Verification error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
