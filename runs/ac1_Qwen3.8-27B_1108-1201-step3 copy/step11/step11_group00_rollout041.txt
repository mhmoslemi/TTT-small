def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1

    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def hankel_mv(p, a):

def project_simplex(v):
    """Project vector v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def game_optimize(a0, time_limit, verbose=True):
    """
    Zero-sum game optimization:
    min_a max_k (a*a)_k s.t. a in simplex.
    Uses alternating best responses with running averages.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Initial p: uniform
    p = np.full(2*n-1, 1.0/(2*n-1))
    
    best_a = a.copy()
    best_obj = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Running average
    avg_a = np.zeros(n)
    n_avg = 0
    
    # Inner solve parameters
    inner_iters = 30  # Number of FW iterations for inner QP
    
    for outer in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # --- Inner solve: min_a a^T H_p a s.t. a in simplex ---
        # Use Frank-Wolfe method
        a_in = a.copy()
        for it in range(inner_iters):
            grad = 2.0 * hankel_mv(p, a_in)
            # FW: pick s = argmin grad
            s_idx = np.argmin(grad)
            # FW step size: t = 2/(it+2)
            t_fw = 2.0 / (it + 2.0)
            a_new = (1 - t_fw) * a_in + t_fw * np.zeros(n)
            a_new[s_idx] += t_fw
            a_in = a_new
        
        a = a_in
        
        # --- Update p: best response to a ---
        C_a = fast_autocorr(a)
        # p = delta at argmax
        best_k = np.argmax(C_a)
        p = np.zeros(2*n-1)
        p[best_k] = 1.0
        
        # --- Track running average ---
        avg_a += a
        n_avg += 1
        
        # --- Evaluate periodically ---
        if outer % 10 == 0:
            obj_a = C_a.max()
            if obj_a < best_obj:
                best_obj = obj_a
                best_a = a.copy()
            
            if n_avg >= 5:
                avg_check = avg_a / n_avg
                obj_avg = fast_autocorr(avg_check).max()
                if obj_avg < best_obj:
                    best_obj = obj_avg
                    best_a = avg_check.copy()
        
        if verbose and outer % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Game iter {outer}: best={best_obj:.8f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_obj

def projected_gradient_lse(a0, time_limit, T_start=50.0, T_end=1.0, n_stages=10):

    """
    n = len(a0)

        a = a / s

    t0 = time.monotonic()

        if remaining < 1:
            break
        # Scale iterations by remaining time

            grad = lse_smooth_grad_a(a, T)

                if obj < best_obj:

        max_iters = max(10, min(100, int(20 + 3 * si)))
        
        def func_z(z_in):
            a_in = softmax_z(z_in)
            C = fast_autocorr(a_in)
            TC = T * C
            m = TC.max()
            return (m + np.log(np.sum(np.exp(TC - m)))) / T
        
        def jac_z(z_in):
            a_in = softmax_z(z_in)
            C = fast_autocorr(a_in)
            TC = T * C
            m = TC.max()
            exp_TC = np.exp(TC - m)
            w = exp_TC / exp_TC.sum()
            grad_a = 2.0 * hankel_mv(w, a_in)
            S = np.dot(grad_a, a_in)
            grad_z = a_in * (grad_a - S)
            return grad_z
        
        try:
            res = minimize(
                fun=func_z, x0=z, jac=jac_z, method='L-BFGS-B',

            z = res.x
            a = softmax_z(z)
            obj = fast_autocorr(a).max()
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
        except Exception:
            pass

def softmax_z(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def peak_shaving(a0, time_limit, seed=123):
    """Coordinate-descent peak shaving."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        C = fast_autocorr(a)
        top_k = np.argsort(C)[::-1][:5]
        improved = False
        
        for k in top_k:
            # Find the main contributor to C[k]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]
            
            # Find a good destination
            dst_pool = np.argsort(a)[:max(1, n // 4)]
            dst = dst_pool[rng.integers(len(dst_pool))]
            
            for frac in [0.002, 0.005, 0.01, 0.02]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                obj = fast_autocorr(a_trial).max()
                if obj < best_obj - 1e-15:
                    a = a_trial
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 50:
                # Random perturbation
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.01, 0.05)
                    a[i] -= amt
                    a[j] += amt
                    a = a / a.sum()
                    obj = fast_autocorr(a).max()
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a.copy()
                        stall = 0
            if stall > 200:
                break
    
    return best_a, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def structured_init(n, rng):
    """Generate structured initializations."""
    choice = rng.integers(0, 5)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 2*np.pi, n)
        a = 0.5 + 0.3*np.cos(2*x) + 0.2*np.cos(5*x)
    else:
        a = np.ones(n)
        # Create a "bump" structure
        center = n // 2
        sigma = n // 4
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def game_optimize_fw(a0, time_limit, inner_iters=50, verbose=True):
    """
    Improved game optimization with:

    - Mixture strategy for p (not just delta)
    - Better frequency for tracking
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Start with uniform p
    p = np.full(2*n-1, 1.0/(2*n-1))
    
    best_a = a.copy()
    best_obj = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    avg_a = np.zeros(n)
    n_avg = 0
    
    # For better convergence, use a soft update for p
    p_soft = p.copy()
    gamma = 0.9  # Soft update parameter
    
    for outer in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Inner solve: min_a a^T H_p a s.t. a in simplex (Frank-Wolfe)
        a_in = a.copy()
        for it in range(inner_iters):
            grad = 2.0 * hankel_mv(p, a_in)
            s_idx = np.argmin(grad)
            t_fw = 2.0 / (it + 2.0)
            a_new = (1 - t_fw) * a_in
            a_new[s_idx] += t_fw
            a_in = a_new
        a = a_in

        C_a = fast_autocorr(a)

        best_k = np.argmax(C_a)
        p_delta = np.zeros(2*n-1)
        p_delta[best_k] = 1.0
        p = (1 - gamma) * p + gamma * p_delta
        
        # Track running average
        avg_a += a
        n_avg += 1
        
        # Evaluate
        if outer % 5 == 0:
            if C_a.max() < best_obj:
                best_obj = C_a.max()
                best_a = a.copy()

            print(f"      GameFW iter {outer}: best={best_obj:.8f}, elapsed={elapsed:.0f}s")

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Convex zero-sum game with Frank-Wolfe QP inner solve (main workhorse)
2. FFT-based O(n log n) Hankel matrix-vector products
3. L-BFGS with LogSumExp smoothing for smooth refinement
4. Multi-length exploration (n=1600, 2400, 3200)
5. Running average tracking for improved game solutions
6. Peak shaving coordinate descent for final polish
Key insight: The problem is convex in a for fixed n, so the game approach
converges to the global optimum. FFT enables fast iterations.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def _fast_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def _hankel_mv(p, a):
    """
    Compute (H_p @ a)[j] = sum_l p[j+l] * a[l] for j=0..n-1.
    H_p is the Hankel matrix with H_p[j,l] = p[j+l].
    Uses FFT cross-correlation: result = conv(p, a_rev)[n-1 : n-1+n].
    """
    n = len(a)
    a_rev = a[::-1]
    conv_len = len(p) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    P = rfft(p, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(P * A_rev, fft_size)
    return conv[n - 1: n - 1 + n]


def _eval_safe(a):

def _project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _softmax_z(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:

    return e / s


def _lse_grad_a(a, T):
    """
    Gradient of (1/T)*log(sum_k exp(T*C[k])) w.r.t. a.
    Returns (grad, w) where w = softmax(T*C).
    """
    n = len(a)
    C = _fast_conv(a)
    TC = T * C
    m = TC.max()
    exp_TC = np.exp(TC - m)
    w = exp_TC / exp_TC.sum()
    grad = 2.0 * _hankel_mv(w, a)
    return grad, w

    return best_a, best_obj


def _lbfgs_lse(a0, time_limit, T_start=50.0, T_end=0.5, n_stages=10):
    """
    L-BFGS optimization of LSE-smoothed max autocorrelation in softmax z-space.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    z = np.log(np.maximum(a, 1e-30))

    best_a = a.copy()
    best_obj = _fast_conv(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        max_iters = max(10, min(150, int(30 + 10 * si)))

        def _func(z_in, T=T):
            a_in = _softmax_z(z_in)
            C = _fast_conv(a_in)
            TC = T * C
            m = TC.max()
            return (m + np.log(np.sum(np.exp(TC - m)))) / T

        def _jac(z_in, T=T):
            a_in = _softmax_z(z_in)
            grad_a, _ = _lse_grad_a(a_in, T)
            S = np.dot(grad_a, a_in)
            return a_in * (grad_a - S)

        try:
            res = minimize(
                fun=_func, x0=z, jac=_jac, method='L-BFGS-B',
                options={'maxiter': max_iters, 'ftol': 1e-18,
                         'gtol': 1e-14, 'disp': False}
            )
            z = res.x
            a = _softmax_z(z)
            obj = _fast_conv(a).max()
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
            elapsed = time.monotonic() - t0
            print(f"        LSE stage {si+1}/{n_stages} (T={T:.2f}): obj={obj:.8f} ({elapsed:.0f}s)")
        except Exception:
            pass

    return best_a, best_obj


def _proj_grad_lse(a0, time_limit, T_start=100.0, T_end=1.0, n_stages=8):
    """
    Projected gradient with LSE smoothing and geometric annealing.
    Works directly in the simplex (no softmax parametrization).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = _fast_conv(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        # Each iteration: 1 FFT for gradient + projection
        # Estimate: ~0.2ms per iter for n=1600, so in 10s we get ~50000 iters
        max_iters = max(50, int(remaining * 30000))
        max_iters = min(max_iters, 200000)

        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                break
            grad, _ = _lse_grad_a(a, T)
            # Step size: 2/(it + 2) scaled
            alpha = 2.0 / (it + 2.0)
            a_new = a - alpha * grad
            a_new = _project_simplex(a_new)

            if it % 100 == 0:
                obj = _fast_conv(a_new).max()
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_new.copy()
            a = a_new

        print(f"        PG-LSE stage {si+1}/{n_stages} (T={T:.2f}): best={best_obj:.8f}")

    return best_a, best_obj


def _peak_shave(a0, time_limit, seed=123):
    """
    Coordinate descent peak shaving: directly reduce top autocorrelation peaks.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = _fast_conv(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        C = _fast_conv(a)
        top_k = np.argsort(C)[::-1][:6]
        improved = False

        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]

            # Find low-value positions as destinations
            sorted_positions = np.argsort(a)
            dst_pool = sorted_positions[:max(1, n // 3)]
            dst = int(dst_pool[rng.integers(len(dst_pool))])

            for frac in [0.002, 0.005, 0.01, 0.02, 0.03]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                obj = _fast_conv(a_trial).max()
                if obj < best_obj - 1e-15:
                    a = a_trial
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 30:
                # Random exploration
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.01, 0.04)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    obj = _fast_conv(a).max()
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a.copy()
                        stall = 0
            if stall > 300:
                break

    return best_a, best_obj


def _interpolate(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def _structured_init(n, rng):
    """Diverse structured initializations."""
    choice = int(rng.integers(0, 6))
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2.5)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2 * x) + 0.15 * np.cos(5 * x) + 0.05 * np.cos(9 * x)
    elif choice == 4:
        x = np.arange(n)
        center = n // 2
        sigma = n / 5.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    else:
        x = np.linspace(0, 1, n)
        a = 0.4 * np.exp(-8 * (x - 0.25) ** 2) + 0.6 * np.exp(-8 * (x - 0.75) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def _game_fw_pure(a0, time_limit, inner_iters=50, verbose=True):
    """
    Game optimization with pure (delta) updates for p.
    Often gives faster convergence to the equilibrium.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = _fast_conv(a).max()
    t0 = time.monotonic()

    # Use a single k index (pure strategy) for the maximizer
    p_k = n - 1  # start at center

    avg_a = np.zeros(n)
    n_avg = 0

    for outer in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        # Build p as delta at p_k
        # H_p @ a where p = delta_{p_k}: (H_p @ a)[j] = a[p_k - j] (if valid)
        # Actually: (H_p @ a)[j] = sum_l delta_{p_k, j+l} * a[l] = a[p_k - j] if 0<=p_k-j<n
        # So grad = 2 * a[p_k - j] (shifted and possibly truncated)
        if p_k >= n:
            # (H_p @ a)[j] = a[p_k - j] for j such that 0 <= p_k - j < n, i.e., p_k - n + 1 <= j <= p_k
            # But j is only 0..n-1, so j ranges from max(0, p_k-n+1) to min(n-1, p_k)
            js = max(0, p_k - n + 1)
            je = min(n - 1, p_k)
            idx = np.arange(js, je + 1)
            grad = np.zeros(n)
            grad[idx] = 2.0 * a[p_k - idx]
        else:
            # (H_p @ a)[j] = a[p_k - j] for j such that 0 <= p_k - j < n, i.e., 0 <= j <= p_k
            je = p_k
            idx = np.arange(0, je + 1)
            grad = np.zeros(n)
            grad[idx] = 2.0 * a[p_k - idx]

        # Frank-Wolfe inner solve using this grad structure
        # Actually for a fixed p (delta), the QP min a^T H_p a has a specific structure
        # But FW still works: at each step, pick argmin of current gradient
        a_in = a.copy()
        for it in range(inner_iters):
            # Recompute gradient for current a_in
            if p_k >= n:
                js = max(0, p_k - n + 1)
                je = min(n - 1, p_k)
                idx = np.arange(js, je + 1)
                g_in = np.zeros(n)
                g_in[idx] = 2.0 * a_in[p_k - idx]
            else:
                je = p_k
                idx = np.arange(0, je + 1)
                g_in = np.zeros(n)
                g_in[idx] = 2.0 * a_in[p_k - idx]

            s_idx = np.argmin(g_in)
            t_fw = 2.0 / (it + 2.0)
            a_new = (1.0 - t_fw) * a_in
            a_new[s_idx] += t_fw
            a_in = a_new
        a = a_in

        # Update p_k: argmax of C_a
        C_a = _fast_conv(a)
        p_k = int(np.argmax(C_a))

        # Running average
        avg_a += a
        n_avg += 1

        if outer % 5 == 0:
            obj_a = C_a.max()
            if obj_a < best_obj:
                best_obj = obj_a
                best_a = a.copy()

            if n_avg >= 10:
                avg_check = avg_a / n_avg
                obj_avg = _fast_conv(avg_check).max()
                if obj_avg < best_obj:
                    best_obj = obj_avg
                    best_a = avg_check.copy()

        if verbose and outer % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"        GamePure[{n}] iter={outer}: best={best_obj:.8f} ({elapsed:.0f}s)")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: multi-phase optimization combining game theory, gradient methods,
    and coordinate descent at multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    # Track best
    best_seq = None
    best_score = float('inf')
    parent = None

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = _eval_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_base = 1600
        best_seq = _structured_init(n_base, rng)
        best_score = _eval_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_parent = len(best_seq)

    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = _eval_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Game FW (soft) at parent length ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 1: GameFW soft at n={n_parent} (budget={min(remaining*0.30, 450):.0f}s) ===")
    t1 = min(remaining * 0.30, 450.0)
    if t1 > 30:
        a_out, _ = _game_fw(best_seq, t1, inner_iters=40, gamma=0.85)
        _try_update(a_out, f"game_fw_soft_n{n_parent}")

    # === Phase 2: Game FW (pure) at parent length ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 2: GameFW pure at n={n_parent} (budget={min(remaining*0.25, 350):.0f}s) ===")
    t2 = min(remaining * 0.25, 350.0)
    if t2 > 30:
        a_out, _ = _game_fw_pure(best_seq, t2, inner_iters=50)
        _try_update(a_out, f"game_fw_pure_n{n_parent}")

    # === Phase 3: L-BFGS LSE refinement at parent length ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 3: L-BFGS LSE at n={n_parent} (budget={min(remaining*0.20, 250):.0f}s) ===")
    t3 = min(remaining * 0.20, 250.0)
    if t3 > 15:
        a_out, _ = _lbfgs_lse(best_seq, t3, T_start=30.0, T_end=0.3, n_stages=10)
        _try_update(a_out, f"lbfgs_n{n_parent}")

    # === Phase 4: Game at longer length (1.5x) ===
    n_long1 = int(n_parent * 1.5)
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 4: GameFW at n={n_long1} (budget={min(remaining*0.25, 400):.0f}s) ===")
    t4 = min(remaining * 0.25, 400.0)
    if t4 > 30:
        init4 = _interpolate(best_seq, n_long1)
        a_out, _ = _game_fw(init4, t4, inner_iters=35, gamma=0.9)
        _try_update(a_out, f"game_fw_n{n_long1}")

    # === Phase 5: L-BFGS at longer length ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 5: L-BFGS at n={n_long1} (budget={min(remaining*0.20, 250):.0f}s) ===")
    t5 = min(remaining * 0.20, 250.0)
    if t5 > 15:
        a_out, _ = _lbfgs_lse(best_seq, t5, T_start=20.0, T_end=0.2, n_stages=8)
        _try_update(a_out, f"lbfgs_n{n_long1}")

    # === Phase 6: Even longer (2x) ===
    n_long2 = int(n_parent * 2.0)
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 6: GameFW at n={n_long2} (budget={min(remaining*0.25, 350):.0f}s) ===")
    t6 = min(remaining * 0.25, 350.0)
    if t6 > 30:
        init6 = _interpolate(best_seq, n_long2)
        a_out, _ = _game_fw(init6, t6, inner_iters=30, gamma=0.9)
        _try_update(a_out, f"game_fw_n{n_long2}")

    # === Phase 7: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 7: Peak shaving (budget={min(remaining*0.15, 120):.0f}s) ===")
    t7 = min(remaining * 0.15, 120.0)
    if t7 > 5:
        a_out, _ = _peak_shave(best_seq, t7, seed=456)
        _try_update(a_out, "peak_shave_1")

    # === Phase 8: Final LSE polish ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 8: Final LSE polish (budget={min(remaining*0.25, 200):.0f}s) ===")
    t8 = min(remaining * 0.25, 200.0)
    if t8 > 10:
        a_out, _ = _lbfgs_lse(best_seq, t8, T_start=5.0, T_end=0.1, n_stages=8)
        _try_update(a_out, "final_lse_polish")

    # === Phase 9: Final peak shave ===
    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 10:
        t9 = min(remaining * 0.5, 100.0)
        if t9 > 5:
            a_out, _ = _peak_shave(best_seq, t9, seed=789)
            _try_update(a_out, "peak_shave_2")

    # Final verification
    if best_seq is not None:
        fs = _eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
