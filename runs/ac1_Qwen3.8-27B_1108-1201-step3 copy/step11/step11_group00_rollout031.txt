Let me reconsider. The standard approach for the "min-max" problem in the context of sequence design is:

**Fictitious Play / Mirror Descent:**
1. Maintain a "virtual opponent" distribution `b` over lags.
2. At each step, update `a` to minimize the expected autocorrelation: `min_a sum_k b_k (a*a)_k`.
3. Update `b` to be concentrated on the lag(s) that maximize `(a*a)_k`.

This is simpler and doesn't require solving an LP at each step. The LP approach is a more sophisticated version where step 3 is replaced by solving the full best-response LP.

def get_lp_solver():

def solve_lp_best_response(a, rows, cols, solver):
    n = len(a)
    b_var = cp.Variable(2 * n - 1, nonneg=True)
    M_var = cp.Variable()

    W = cp.SparseMatrix((vals, (rows, cols)), shape=(n, 2 * n - 1))
    constraints = [W @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers = [solver, cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]
    seen = set()
    for s in solvers:
        if s is None or s in seen:
            continue

            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break

            continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val /= s
    return b_val, float(M_var.value)

def coordinate_descent_polish(a0, time_limit, seed=42, verbose=True):
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0)
    a /= a.sum()
    best_a = a.copy()

    t0 = time.monotonic()
    stall = 0
    iter_count = 0
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:5]
        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src_local = np.argmax(contrib)
            src = idx[src_local]
            peak_lags = set(int(x) for x in top_k)
            src_appears_in = set()
            for kl in peak_lags:
                js2 = max(0, kl - n + 1)
                je2 = min(n - 1, kl)
                if js2 <= src <= je2:
                    src_appears_in.add(kl)
            candidates = []
            for j in range(n):
                if j == src:
                    continue
                j_appears_in = set()
                for kl in peak_lags:
                    js2 = max(0, kl - n + 1)
                    je2 = min(n - 1, kl)
                    if js2 <= j <= je2:
                        j_appears_in.add(kl)
                overlap = len(src_appears_in & j_appears_in)
                if overlap < len(src_appears_in):
                    candidates.append((j, overlap, a[j]))
            if not candidates:
                candidates = [(j, 0, a[j]) for j in range(n) if j != src]
            candidates.sort(key=lambda x: (x[1], x[2]))
            top_candidates = candidates[:8]
            for dst, _, _ in top_candidates:
                for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial /= a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            stall += 1
            if stall % 50 == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial /= a_trial.sum()
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
            if stall > 500:

        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD iter={iter_count} ({elapsed:.0f}s): best={best_score:.6f}, stall={stall}")
    return best_a, best_score

def logsumexp_optimize(a0, time_limit, T_start=50.0, T_end=0.1, n_stages=10, verbose=True):
    n = len(a0)
    a_init = np.maximum(a0.astype(np.float64), 0)
    a_init /= a_init.sum()
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def lse_func(z_in):
            a_tmp = np.exp(z_in - z_in.max())
            a_tmp /= a_tmp.sum()
            b_tmp = fast_autocorr(a_tmp)
            Tb_tmp = stage_T * b_tmp
            return Tb_tmp.max() + np.log(np.sum(np.exp(Tb_tmp - Tb_tmp.max()))) / stage_T

        def lse_jac(z_in):
            a_tmp = np.exp(z_in - z_in.max())
            a_tmp /= a_tmp.sum()
            b_tmp = fast_autocorr(a_tmp)
            Tb_tmp = stage_T * b_tmp
            max_Tb = Tb_tmp.max()
            exp_Tb = np.exp(Tb_tmp - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            risk = compute_risk(a_tmp, w)
            risk *= stage_T
            S = np.dot(risk, a_tmp)
            grad_z = a_tmp * (risk - S)
            return grad_z

        try:
            res = minimize(fun=lse_func, x0=z, jac=lse_jac, method='L-BFGS-B',
                           options={'maxiter': 50, 'ftol': 1e-20, 'gtol': 1e-14, 'disp': False})
            z = res.x
            a = np.exp(z - z.max())
            a /= a.sum()
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
                if verbose:
                    print(f"    LSE stage {si+1}/{n_stages} (T={stage_T:.2f}): {score:.6f}")
        except Exception:
            pass
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp

    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2 + 0.3
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.7 + 0.2 * np.cos(2 * x) + 0.1 * np.cos(5 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.4) ** 2) + 0.6 * np.exp(-5 * (x - 0.6) ** 2)
    elif choice == 6:
        a = np.ones(n)
        a[:n//4] *= 0.5
        a[-(n//4):] *= 0.5
    else:
        a = np.random.default_rng(choice).dirichlet(np.ones(n) * 0.5)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def multi_start_search(n, time_limit, solver, verbose=True):
    rng = np.random.default_rng(123)
    t0 = time.monotonic()
    best_a = None
    best_score = float('inf')
    n_starts = max(2, int(time_limit / 30))
    time_per_start = time_limit / n_starts
    for si in range(n_starts):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        a_init = structured_init(n, rng)
        a_out, sc = nesterov_lp_optimize(a_init, min(remaining * 0.7, time_per_start * 0.6), solver, verbose=False)
        if a_out is not None and sc < best_score:
            best_score = sc
            best_a = a_out.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    MultiStart {si+1}/{n_starts} ({elapsed:.0f}s): {sc:.6f}")
    if best_a is None:
        best_a = structured_init(n, rng)
        best_score = 2.0 * n * fast_autocorr(best_a).max()
    return best_a, best_score

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_init = 1600
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_base = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global_start)

    print(f"\n=== Phase 1: Quick FP at n={n_base} ===")
    t1 = min(remaining * 0.05, 60.0)
    if t1 > 5:
        a_out, _ = fictitious_play(best_seq, t1, verbose=True)
        try_update(a_out, f"fp_n{n_base}")

    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: LP Nesterov at n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, f"lp_n{n_base}")

    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: LSE gradient at n={n_base} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 10:
        a_out, _ = logsumexp_optimize(best_seq, t3, T_start=30.0, T_end=0.1, n_stages=8, verbose=True)
        try_update(a_out, f"lse_n{n_base}")

    # Try longer lengths
    lengths_to_try = []
    for factor in [1.25, 1.5, 1.75, 2.0]:
        nl = int(round(n_base * factor))
        if nl != n_base and nl not in lengths_to_try and nl <= 3500:
            lengths_to_try.append(nl)

    for nl in lengths_to_try[:3]:
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 30:
            break
        print(f"\n=== Phase: LP Nesterov at n={nl} (remaining={remaining:.0f}s) ===")
        t_lp = min(remaining * 0.15, 250.0)
        if t_lp > 20:
            init_nl = interpolate_sequence(best_seq, nl)
            a_out, _ = nesterov_lp_optimize(init_nl, t_lp, solver, verbose=True)
            try_update(a_out, f"lp_n{nl}")

        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 15:
            break
        t_cd = min(remaining * 0.08, 80.0)
        if t_cd > 5:
            a_out, _ = coordinate_descent_polish(best_seq, t_cd, seed=seed + nl, verbose=False)
            try_update(a_out, f"cd_n{nl}")

    # Try shorter lengths
    for nl in [int(n_base * 0.85), int(n_base * 0.7)]:
        if nl < 500:
            continue
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 20:
            break
        print(f"\n=== Phase: LP at n={nl} (remaining={remaining:.0f}s) ===")
        t_lp = min(remaining * 0.12, 150.0)
        if t_lp > 15:
            init_nl = interpolate_sequence(best_seq, nl)
            a_out, _ = nesterov_lp_optimize(init_nl, t_lp, solver, verbose=True)
            try_update(a_out, f"lp_n{nl}")

    # Final polishing on the best solution
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase: Coordinate Descent Polish (remaining={remaining:.0f}s) ===")
    t_cd = min(remaining * 0.25, 300.0)
    if t_cd > 10:
        a_out, _ = coordinate_descent_polish(best_seq, t_cd, seed=789, verbose=True)
        try_update(a_out, "cd_final_1")

    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase: LSE polish (remaining={remaining:.0f}s) ===")
    t_lse = min(remaining * 0.15, 100.0)
    if t_lse > 8:
        a_out, _ = logsumexp_optimize(best_seq, t_lse, T_start=10.0, T_end=0.05, n_stages=8, verbose=True)
        try_update(a_out, "lse_final")

    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase: Final CD polish (remaining={remaining:.0f}s) ===")
    t_cd2 = min(remaining * 0.3, 200.0)
    if t_cd2 > 10:
        a_out, _ = coordinate_descent_polish(best_seq, t_cd2, seed=1010, verbose=True)
        try_update(a_out, "cd_final_2")

    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 30:
        print(f"\n=== Phase: Second LP refinement (remaining={remaining:.0f}s) ===")
        t_lp2 = min(remaining * 0.4, 200.0)
        if t_lp2 > 15:
            a_out, _ = nesterov_lp_optimize(best_seq, t_lp2, solver, verbose=True)
            try_update(a_out, "lp_second")

    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 20:
        t_cd3 = min(remaining * 0.5, 100.0)
        if t_cd3 > 5:
            a_out, _ = coordinate_descent_polish(best_seq, t_cd3, seed=9999, verbose=False)
            try_update(a_out, "cd_final_3")

    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        best_seq = np.ones(100) / 100

    return [float(x) for x in best_seq]
