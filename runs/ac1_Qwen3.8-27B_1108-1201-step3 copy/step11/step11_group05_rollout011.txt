Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

3. Adam-based LogSumExp optimization with temperature annealing

- Better Nesterov step sizes

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_response(a_fixed, method='highs'):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k].
    
    LP: min M s.t. T_a @ b <= M*1, sum(b)=1, b>=0, M>=0
    """
    n = len(a_fixed)
    size = 2 * n - 1
    T = build_conv_matrix(a_fixed)
    
    # Variables: [b_0, ..., b_{n-1}, M]
    # c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # A_ub: [T, -1] @ [b; M] <= 0
    # T is size x n, so A_ub is size x (n+1)
    A_ub = np.hstack([T, -np.ones((size, 1))])
    b_ub = np.zeros(size)
    
    # A_eq: [1, ..., 1, 0] @ [b; M] = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # bounds: b_i >= 0, M free (but we can set M >= 0)
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method=method)
    
    if res.status != 0:
        return None, None
    
    b_val = np.asarray(res.x[:n], dtype=np.float64)
    M_val = float(max(0.0, res.x[n]))
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, M_val

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, lp_method='highs'):
    n = len(a_fixed)
    size = 2 * n - 1
    T = build_conv_matrix(a_fixed)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([T, -np.ones((size, 1))])
    b_ub = np.zeros(size)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method=lp_method)
        if res.status != 0:
            return None, None
        b_val = np.asarray(res.x[:n], dtype=np.float64)
        M_val = float(max(0.0, res.x[n]))
        b_val = np.maximum(b_val, 0)
        s = b_val.sum()
        if s < 1e-15:
            return None, None
        return b_val / s, M_val
    except Exception:
        return None, None


def evaluate_local(a):

    if s < 0.01:

    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def adam_logsumexp(a0, time_limit, T_stages, lr_init, lr_decay, top_k_frac=0.1):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    z = np.log(np.maximum(a, 1e-300))
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step_count = 0
    
    total_stages = len(T_stages)
    
    for stage_idx, T_stage in enumerate(T_stages):
        if time.monotonic() > t0 + time_limit:
            break
        remaining = t0 + time_limit - time.monotonic()
        if remaining < 1:
            break
        
        steps_per_stage = max(1000, int(remaining * 50000 / (total_stages - stage_idx)))
        top_k = max(5, int(top_k_frac * (2 * n - 1)))
        
        for step in range(steps_per_stage):
            if time.monotonic() > t0 + time_limit:
                break
            
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a = exp_z / exp_z.sum()
            
            b = fast_convolve(a)
            max_b = b.max()
            score = 2.0 * n * max_b
            step_count += 1
            
            if score < best_score:
                best_score = score
                best_a = a.copy()
            
            if top_k < len(b):
                idx = np.argpartition(b, -top_k)[-top_k:]
                b_top = b[idx]
                max_top = b_top.max()
                if T_stage < 0.01:
                    w_top = np.ones(len(b_top)) / len(b_top)
                else:
                    exp_top = np.exp((b_top - max_top) / T_stage)
                    w_top = exp_top / exp_top.sum()
                w = np.zeros(len(b))
                w[idx] = w_top
            else:
                max_b_all = b.max()
                if T_stage < 0.01:
                    w = np.ones(len(b)) / len(b)
                else:
                    exp_all = np.exp((b - max_b_all) / T_stage)
                    w = exp_all / exp_all.sum()
            
            g_a_raw = np.correlate(w, a, mode='full')
            g_a = g_a_raw[n - 1:2 * n - 1] * 2.0
            
            S = np.dot(g_a, a)
            grad_z = a * (g_a - S)
            
            lr = lr_init * (lr_decay ** stage_idx)
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** (step + 1))
            v_hat = v / (1 - beta2 ** (step + 1))
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            z -= z.mean()
    
    return best_a, best_score


def coordinate_descent(a0, time_limit, n_inner=10):

            break
        
        # Try to optimize a[coord]

        # We want to minimize max_k (a*a)[k]
        
        # The value of (a*a)[k] as a function of a[coord] = t:
        # (a*a)[k] = original[k] + t * (d/dt (a*a)[k] at t=0) + t^2 * (second order)
        # For k where coord + (k-coord) = k, the term involving a[coord] is:
        # a[coord] * a[k-coord] (appears once or twice)
        
        # Let me just do a 1D golden section search
        old_val = a[coord]
        if old_val < 1e-15:
            # Try small values
            candidates = [0, 0.001, 0.01, 0.05, 0.1]
        else:
            candidates = [0, old_val * 0.5, old_val * 0.8, old_val, old_val * 1.2, old_val * 1.5]
        
        best_candidate = old_val
        best_candidate_score = best_score

            a[coord] = max(0, c)
            s_new = a.sum()
            if s_new < 1e-15:
                a[coord] = old_val
                continue

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
