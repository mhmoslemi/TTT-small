    obj = 2.0 * n * lse_val

    w = exp_c / (exp_c.sum() + 1e-300)

    # Gradient: grad[m] = 2 * sum_k w[k] * a[k-m] (with boundary handling)
    # This is 2 * cross-correlation of w and a
    fft_size = 1 << (len(w) + n - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # cross-correlation: IFFT(W * conj(A))
    R = irfft(W * np.conj(A), fft_size)
    g = 2.0 * R[:n]

    # Correction: when k=2m, the term a[m]*a[m] should only be counted once
    # But actually in the formula sum_k w[k]*a[k-m], when k=2m we get w[2m]*a[m]
    # which is correct (it's just one term in the sum over k)
    # The gradient of c[k] w.r.t. a[m] is:
    #   a[k-m] if 0<=k-m<n, times (1 if k!=2m, 2 if k==2m)... 
    # Wait, let me reconsider.
    # c[k] = sum_i a[i]*a[k-i]
    # d(c[k])/d(a[m]) = a[k-m] (from i=m term) + a[m] (from i=k-m term) if k-m != m
    #                  = 2*a[m] if k = 2m
    # So d(c[k])/d(a[m]) = a[k-m] + (a[m] if k-m != m and 0<=k-m<n else 0)
    #                      = a[k-m] + a[k-m] (if k-m != m) = 2*a[k-m] (if valid and k!=2m)
    #                      = 2*a[m] (if k=2m)
    # In all cases where 0<=k-m<n, it's 2*a[k-m] (since when k=2m, a[k-m]=a[m]).
    # So the formula is actually just: grad[m] = 2 * sum_{k: 0<=k-m<n} w[k]*a[k-m]
    # Which is exactly 2 * (w cross-correlated with a)[m] where a is zero-padded.
    # My FFT computation above already handles this correctly!

    return obj, g


def lse_obj_grad_z(z, T):
    """LSE objective and gradient in log-parameterization."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    n = len(a)

    obj, grad_a = lse_obj_grad(a, T)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lbgfs_optimize(a0, time_limit, T_stages, maxiter_per_stage=5000):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_in):
            obj, _ = lse_obj_grad_z(z_in, stage_T)
            return obj

        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, stage_T)
            return g

        iters = min(maxiter_per_stage, max(5, int(remaining * 50)))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-16,
                         'maxfun': iters * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def multiplicative_weights_adaptive(a0, time_limit, n_steps=100000, eta0=0.1):
    """Adaptive multiplicative weights with soft active set and backtracking."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = eta0
    T = max(0.01, best_score / (2.0 * n) * 5.0)  # Initial temperature based on peak
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        c_max = c.max()

        # Soft active set: exponential weights
        exp_c = np.exp((c - c_max) / T)
        w = exp_c / (exp_c.sum() + 1e-300)

        # Compute weighted subgradient
        fft_size = 1 << (len(w) + n - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        R = irfft(W * np.conj(A), fft_size)
        g = 2.0 * R[:n]

        # Subtract mean for scale invariance
        g_mean = np.dot(g, a)
        g_centered = g - g_mean * a  # Project to tangent space of simplex

        # Normalize for step size
        g_norm = np.sqrt(np.sum(a * g ** 2) + 1e-30)
        if g_norm < 1e-15:
            break

        # Multiplicative update with backtracking line search
        accepted = False
        for scale in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            effective_eta = eta * scale / g_norm
            log_alpha = -effective_eta * g
            log_alpha -= log_alpha.max()
            a_new = a * np.exp(log_alpha)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                accepted = True
                no_improve = 0
            break

        if not accepted:
            no_improve += 1
            eta *= 0.8
            T *= 1.3  # Widen the active set when stuck
            if no_improve >= 30:
                eta = max(eta * 0.5, 1e-6)
                T = max(T * 1.5, 0.01)
                no_improve = 0
                if eta < 1e-8:
                    break
        else:
            T = max(T * 0.95, 0.005)  # Slowly narrow
            eta = min(eta * 1.02, 1.0)

        if step % 5000 == 0 and step > 0:
            c_now = fast_convolve(a)
            print(f"    MW step={step}: score={best_score:.8f}, T={T:.4f}, eta={eta:.6f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, n_iters=50000):
    """Directly target and reduce individual convolution peaks."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        c_max = c.max()
        peak_k = int(np.argmax(c))

        # Identify which positions contribute to this peak
        # c[peak_k] = sum_i a[i]*a[peak_k - i]
        # To reduce c[peak_k], reduce a[i] for i that have large a[peak_k - i]
        
        i_start = max(0, peak_k - n + 1)
        i_end = min(n - 1, peak_k)
        
        if i_start > i_end:
            break

        # Contribution of each a[i] to the peak
        contributions = np.zeros(n)
        for i in range(i_start, i_end + 1):
            j = peak_k - i
            contributions[i] = a[j]  # d(c[peak_k])/d(a[i]) includes a[j]
            if i != j:
                contributions[j] += a[i]
            else:
                contributions[i] = a[i]  # a[i]^2 contributes 2*a[i], already counted once above
                # Actually for i==j: d(a[i]*a[i])/d(a[i]) = 2*a[i]
                # But we already added a[j]=a[i] once, so we need to add a[i] more
                contributions[i] = 2.0 * a[i]

        # Find the "expensive" positions (high contribution) and "cheap" positions
        # Move mass from expensive to cheap positions
        median_contrib = np.median(contributions[i_start:i_end + 1])
        expensive_mask = np.zeros(n, dtype=bool)
        cheap_mask = np.zeros(n, dtype=bool)
        expensive_mask[i_start:i_end + 1] = contributions[i_start:i_end + 1] > median_contrib
        cheap_mask[i_start:i_end + 1] = contributions[i_start:i_end + 1] < median_contrib

        # Also consider positions outside the peak's influence as cheap
        for i in range(n):
            if i < i_start or i > i_end:
                cheap_mask[i] = True

        if not np.any(expensive_mask) or not np.any(cheap_mask):
            break

        exp_total = a[expensive_mask].sum()
        cheap_total = a[cheap_mask].sum()
        if exp_total < 1e-15 or cheap_total < 1e-15:
            break

        # Move a small fraction of mass from expensive to cheap
        for frac in [0.01, 0.005, 0.002, 0.001]:
            delta = frac * exp_total
            # Reduce expensive proportionally
            a_new = a.copy()
            if a_new[expensive_mask].sum() > 0:
                a_new[expensive_mask] -= a_new[expensive_mask] * (delta / a_new[expensive_mask].sum())
            # Add to cheap proportionally (weighted by inverse contribution)
            cheap_weights = np.ones(n)
            cheap_weights[cheap_mask] = 1.0 / (contributions[cheap_mask] + 1e-10)
            cheap_weights[~cheap_mask] = 0
            cw = cheap_weights[cheap_mask].sum()
            if cw > 0:
                a_new[cheap_mask] += delta * cheap_weights[cheap_mask] / cw
            else:
                a_new[cheap_mask] += delta / cheap_mask.sum()

            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
            break

        if it % 5000 == 0 and it > 0:
            print(f"    PeakShave it={it}: score={best_score:.8f}, peak_k={peak_k}, c_max={c_max:.6f}")

    return best_a, best_score


def spectral_equalize(a0, time_limit, n_iters=10000):
    """Equalize the power spectrum to reduce autocorrelation peaks.
    Uses proper gradient of variance of autocorrelation w.r.t. a.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    t0 = time.monotonic()
    lr = 0.01

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = rfft(a, fft_size)
        # Periodic autocorrelation (circular)
        P = np.abs(A) ** 2
        P_mean = P.mean()

        # We want to minimize the variance of P (flat spectrum => flat autocorrelation)
        # But actually we want to minimize max of linear autocorrelation.
        # As a proxy, minimize the top peaks of P.
        # Use top-k weighting
        k_top = max(3, len(P) // 10)
        P_sorted = np.sort(P)[::-1]
        threshold = P_sorted[k_top - 1] if k_top < len(P_sorted) else 0
        P_weight = np.where(P > threshold, (P - threshold) ** 2, 0.0)
        P_weight /= (P_weight.sum() + 1e-30)

        # Gradient: d(P_weight)/d(a)
        # P[m] = |A[m]|^2, d(P[m])/da = 2*Re(A[m] * conj(dA[m]/da))
        # dA[m]/da = conj(FFT(delta_m)) where delta_m is the m-th basis vector
        # So d(P[m])/da = 2*Re(A[m] * conj(conj(FFT(e_m)))) = 2*Re(A[m] * FFT(e_m))
        # = 2*Re(A[m] * exp(2*pi*i*m*k/N)) for k-th element of a
        # This is: h[k] = 2*Re(sum_m P_weight[m] * A[m] * exp(2*pi*i*m*k/N))
        # = 2*Re(IFFT(P_weight * A))  [with appropriate scaling]

        H = np.zeros(fft_size // 2 + 1, dtype=np.complex128)
        H[:] = P_weight * A  # length fft_size//2+1
        h = irfft(H * 2, fft_size)[:n]  # factor 2 for conjugate symmetry
        h -= h.sum() * a  # Project to tangent space (preserve sum)

        h_norm = np.linalg.norm(h)
        if h_norm < 1e-15:
            break
        h = h / h_norm

        # Line search
        accepted = False
        for scale in [lr, lr * 0.5, lr * 0.1, lr * 0.01]:
            a_new = a + scale * h
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                accepted = True
            break

        if not accepted:
            lr *= 0.7
            if lr < 1e-8:
                break

        if it % 2000 == 0 and it > 0:
            print(f"    SpectralEq it={it}: score={best_score:.8f}, lr={lr:.6f}")

    return best_a, best_score


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    candidates = []
    x = np.linspace(0, 1, n)

    # Uniform
    candidates.append(np.ones(n) / n)

    # Gaussian bump
    a = np.exp(-8 * (x - 0.5) ** 2)
    candidates.append(normalize(a))

    # Raised cosine
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    candidates.append(normalize(a))

    # Sine squared
    a = np.sin(np.pi * x) ** 2
    candidates.append(normalize(a))

    # Power law
    a = (x * (1 - x)) ** 0.3
    candidates.append(normalize(a))

    # Two bumps
    a = np.exp(-10 * (x - 0.3) ** 2) + np.exp(-10 * (x - 0.7) ** 2)
    candidates.append(normalize(a))

    # Edge-peaked
    a = np.exp(-3 * x) + np.exp(-3 * (1 - x))
    candidates.append(normalize(a))

    # Modulated
    a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.2 * np.cos(2 * np.pi * 7 * x + 0.5)
    a = np.maximum(a, 0)
    candidates.append(normalize(a))

    # Random smooth
    coeffs = rng.normal(0, 1, 15)
    a = np.zeros(n)
    for i, c in enumerate(coeffs):
        a += c * np.cos(2 * np.pi * (i + 1) * x + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a + 1.0, 0)
    candidates.append(normalize(a))

    # Fermi-like step
    a = 1.0 / (1.0 + np.exp(-20 * (x - 0.5)))
    candidates.append(normalize(a))

    return candidates


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    return normalize(f_interp)


def perturb_sequence(a, rng, strength=0.02):
    """Perturb a sequence."""
    n = len(a)
    method = rng.integers(0, 4)
    if method == 0:
        a_new = a + rng.normal(0, strength, n)
    elif method == 1:
        freq = rng.integers(1, 10)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 2 * np.pi, n)
        a_new = a + strength * np.cos(freq * x + phase)
    elif method == 2:
        a_new = a.copy()
        bs = min(n // 10, 50)
        i = rng.integers(0, max(1, n - 2 * bs))
        j = i + bs + rng.integers(0, max(1, n - 2 * i - bs))
        if j + bs <= n:
            a_new[i:i+bs], a_new[j:j+bs] = a[j:j+bs].copy(), a[i:i+bs].copy()
    else:
        a_new = a + strength * rng.randint(-1, 2, n).astype(float) * (1.0 - 2.0 * np.abs(np.linspace(-1, 1, n)))
    return normalize(a_new)


def population_search(a_init, time_limit, pop_size=12, rng=None):
    """Population-based search with adaptive mutations and LSE polish."""
    if rng is None:
        rng = np.random.default_rng(789)

    n = len(a_init)
    population = [normalize(a_init.copy())]
    for i in range(pop_size - 1):
        population.append(perturb_sequence(a_init, rng, strength=0.01 + 0.01 * (i % 3)))

    scores = [evaluate_local(p) for p in population]
    best_idx = np.argmin(scores)
    best_a = population[best_idx].copy()
    best_score = scores[best_idx]

    t0 = time.monotonic()
    gen = 0
    adaptive_strength = 0.03
    T_lbgfs = 1.0

    while time.monotonic() - t0 < time_limit:
        gen += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 15:
            break

        order = np.argsort(scores)
        population = [population[i] for i in order]
        scores = [scores[i] for i in order]

        if scores[0] < best_score:
            best_score = scores[0]
            best_a = population[0].copy()

        # Keep top half
        keep = max(3, pop_size // 2)
        elites = population[:keep]

        # Generate offspring
        new_pop = list(elites)

        for i in range(pop_size - len(new_pop)):
            parent = elites[rng.integers(0, keep)]
            child = perturb_sequence(parent, rng, strength=adaptive_strength * (1 + 0.3 * gen % 5))
            new_pop.append(child)

        # Crossover
        if len(elites) >= 2 and len(new_pop) < pop_size:
            p1, p2 = elites[rng.integers(0, keep)], elites[rng.integers(0, keep)]
            t = rng.uniform(0.2, 0.8)
            child = normalize(t * p1 + (1 - t) * p2)
            new_pop.append(child)

        population = new_pop[:pop_size]
        scores = [evaluate_local(p) for p in population]

        # Periodically do LSE polish on the best
        if gen % 10 == 0 and remaining > 30:
            T_short = [T_lbgfs, T_lbgfs * 0.5, T_lbgfs * 0.25]
            a_polished = lbgfs_optimize(population[0], min(remaining * 0.1, 30.0), T_short, maxiter_per_stage=500)
            sc = evaluate_local(a_polished)
            if sc < scores[0]:
                population[0] = a_polished
                scores[0] = sc
            T_lbgfs = max(T_lbgfs * 0.8, 0.1)

        # Adapt strength
        if gen % 20 == 0:
            if scores[0] >= best_score - 1e-10:
                adaptive_strength *= 0.85
            else:
                adaptive_strength = min(0.08, adaptive_strength * 1.1)

        if gen % 20 == 0:
            print(f"    Pop gen={gen}: best={best_score:.8f}, strength={adaptive_strength:.4f}")

    return best_a, best_score


def hybdrid_optimize(a0, time_limit, rng):
    """Cycle through different optimization methods to escape local minima."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    methods = ['mw', 'lbgfs', 'peak', 'spectral']
    method_idx = 0
    no_improve_global = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 20:
            break

        method = methods[method_idx % len(methods)]
        t_sub = min(remaining * 0.15, 120.0)
        if t_sub < 10:
            t_sub = min(remaining, 30.0)

        prev_score = best_score

        if method == 'mw':
            a, _ = multiplicative_weights_adaptive(a, t_sub, n_steps=50000, eta0=0.1)
        elif method == 'lbgfs':
            T_stages = [5, 2, 1, 0.5, 0.2, 0.1, 0.05, 0.02]
            a = lbgfs_optimize(a, t_sub, T_stages, maxiter_per_stage=2000)
        elif method == 'peak':
            a, _ = peak_shaving(a, t_sub, n_iters=20000)
        elif method == 'spectral':
            a, _ = spectral_equalize(a, t_sub, n_iters=5000)

        sc = evaluate_local(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve_global = 0
        else:
            no_improve_global += 1
            # If stuck, try a perturbation
            if no_improve_global >= 3:
                a = perturb_sequence(best_a, rng, strength=0.02 + 0.01 * no_improve_global)
                no_improve_global = 0

        method_idx += 1
        if gen := method_idx // len(methods) > 0 and gen % 3 == 0:
            print(f"    Hybrid cycle={gen}: score={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase hybrid optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Combines multiplicative weights, L-BFGS+LSE, peak shaving, and spectral
    equalization in a cycling fashion, with population search and length
    adaptation for global exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (prev={prev:.8f}, n={len(a)})")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        candidates = diverse_init(n_init, rng)
        scores = [evaluate_local(c) for c in candidates]
        best_idx = np.argmin(scores)
        best_seq = candidates[best_idx].copy()
        best_score = scores[best_idx]
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.8f}")

    # === Phase 1: Adaptive multiplicative weights (long run) ===
    print(f"\n=== Phase 1: Multiplicative Weights ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_mw, _ = multiplicative_weights_adaptive(best_seq, t_alloc, n_steps=200000, eta0=0.08)
        local_update(a_mw, "mw_1")

    # === Phase 2: L-BFGS with aggressive annealing ===
    print(f"\n=== Phase 2: L-BFGS LSE (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        T_stages = [20, 10, 5, 3, 2, 1, 0.7, 0.5, 0.3, 0.2, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001]
        print(f"  T_stages ({len(T_stages)}), budget={t_alloc:.0f}s")
        a_lb = lbgfs_optimize(best_seq, t_alloc, T_stages, maxiter_per_stage=5000)
        local_update(a_lb, "lbgfs_1")

    # === Phase 3: Peak shaving ===
    print(f"\n=== Phase 3: Peak Shaving (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 20:
        a_pk, _ = peak_shaving(best_seq, t_alloc, n_iters=30000)
        local_update(a_pk, "peak_shave")

    # === Phase 4: Spectral equalization ===
    print(f"\n=== Phase 4: Spectral Equalization (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 15:
        a_sp, _ = spectral_equalize(best_seq, t_alloc, n_iters=10000)
        local_update(a_sp, "spectral")

    # === Phase 5: Hybrid cycling ===
    print(f"\n=== Phase 5: Hybrid Cycling (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 30:
        a_hyb, _ = hybdrid_optimize(best_seq, t_alloc, rng)
        local_update(a_hyb, "hybrid")

    # === Phase 6: Fine L-BFGS polish ===
    print(f"\n=== Phase 6: Fine L-BFGS Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 15:
        T_fine = [1, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001]
        a_fine = lbgfs_optimize(best_seq, t_alloc, T_fine, maxiter_per_stage=3000)
        local_update(a_fine, "lbgfs_fine")

    # === Phase 7: Length exploration ===
    print(f"\n=== Phase 7: Length Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        length_tries = []
        # Try lengths around current best
        for delta in [100, -100, 200, -200, 300]:
            n_try = n_best + delta
            if 200 <= n_try <= 3000 and n_try not in length_tries:
                length_tries.append(n_try)
        # Also try some standard lengths
        for n_std in [800, 1000, 1500, 2000]:
            if n_std not in length_tries:
                length_tries.append(n_std)

        for n_try in length_tries[:5]:
            rem_sub = deadline - time.monotonic()
            if rem_sub < 40:
                break
            print(f"  Trying n={n_try}...")
            init = interpolate_sequence(best_seq, n_try)
            t_sub = min(rem_sub * 0.2, 60.0)
            if t_sub < 15:
                continue
            # Quick optimization
            T_short = [5, 2, 1, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
            a_try = lbgfs_optimize(init, t_sub, T_short, maxiter_per_stage=2000)
            local_update(a_try, f"len_{n_try}")

            if evaluate_local(best_seq) == best_score:
                # Also try MW on the best so far at this length
                t_sub2 = min(deadline - time.monotonic() - 20, 30.0)
                if t_sub2 > 10:
                    a_try2, _ = multiplicative_weights_adaptive(best_seq, t_sub2, n_steps=50000, eta0=0.05)
                    local_update(a_try2, f"len_{n_try}_mw")

    # === Phase 8: Population search for diversity ===
    print(f"\n=== Phase 8: Population Search (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 30:
        a_pop, _ = population_search(best_seq, t_alloc, pop_size=12, rng=rng)
        local_update(a_pop, "population")

    # === Phase 9: Final polish ===
    print(f"\n=== Phase 9: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        # Final L-BFGS with very low temperature
        T_final = [0.3, 0.1, 0.05, 0.03, 0.02, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001, 0.0005]
        a_fin = lbgfs_optimize(best_seq, min(rem * 0.3, 40.0), T_final, maxiter_per_stage=2000)
        local_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        # Final MW push
        a_mw2, _ = multiplicative_weights_adaptive(best_seq, min(rem * 0.4, 40.0), n_steps=100000, eta0=0.03)
        local_update(a_mw2, "final_mw")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = min(best_score, fs)
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
