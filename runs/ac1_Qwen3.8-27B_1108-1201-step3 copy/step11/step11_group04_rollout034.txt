```python
import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a, n):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    T = np.zeros((2*n-1, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s

def solve_lp_best_response(a_fixed, solver_name=None):
    """Solve the LP: min M s.t. conv(a_fixed, b)_k <= M, b >= 0, sum(b) = 1"""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed, n)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers = [solver_name, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    if solver_name is None:
        solvers = [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    
    for s in solvers:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                ss = b_val.sum()
                if ss < 1e-15:
                    continue
                b_val = b_val / ss
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None
```
