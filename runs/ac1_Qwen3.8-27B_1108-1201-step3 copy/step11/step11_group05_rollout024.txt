def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_gradient(a, w):
    """Compute gradient of max(conv) w.r.t. a using weights w on the convolution.
    grad[j] = 2 * sum_k w[k] * a[k-j]
    """
    n = len(a)
    # grad[j] = 2 * correlate(w, a, 'full')[j]
    c = np.correlate(w, a, mode='full')
    return 2.0 * c[:n]


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def evaluate_local(a):

    if s < 0.01:

    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):

    s = a.sum()

def structured_init(n, rng):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():

    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):

        except Exception:

    return None


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None or not HAS_CVXPY:
        return None
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s > 1e-15:
                return b_val / s
    except Exception:
        pass
    return None


def adam_logsumexp(a0, time_limit, T_stages, lr_init, lr_decay, top_k_frac=0.1, nesterov=False):
    """Adam optimization in log-space with temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step_count = 0
    
    total_stages = len(T_stages)
    steps_per_stage = max(2000, int(time_limit * 30000 / total_stages))
    
    for stage_idx, T_stage in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        
        top_k = max(5, int(top_k_frac * (2 * n - 1)))
        
        for step in range(steps_per_stage):
            if time.monotonic() > deadline:
                break
            
            a = softmax_from_z(z)
            b = fast_convolve(a)
            max_b = b.max()
            score = 2.0 * n * max_b
            step_count += 1
            
            if score < best_score:
                best_score = score
                best_a = a.copy()
            
            # Compute weights
            if T_stage < 0.01:
                threshold = max_b * (1.0 - 1e-4)
                active = np.where(b >= threshold)[0]
                if len(active) == 0:
                    active = np.array([np.argmax(b)])
                w = np.zeros(len(b))
                w[active] = 1.0 / len(active)
            else:
                if top_k < len(b):
                    idx = np.argpartition(b, -top_k)[-top_k:]
                    b_top = b[idx]
                    max_top = b_top.max()
                    exp_top = np.exp((b_top - max_top) / T_stage)
                    w_top = exp_top / exp_top.sum()
                    w = np.zeros(len(b))
                    w[idx] = w_top
                else:
                    max_b_all = b.max()
                    exp_all = np.exp((b - max_b_all) / T_stage)
                    w = exp_all / exp_all.sum()
            
            # Gradient in a-space
            g_a = compute_gradient(a, w)
            
            # Gradient in z-space
            S = np.dot(g_a, a)
            grad_z = a * (g_a - S)
            
            # Adam update
            lr = lr_init * (lr_decay ** stage_idx)
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** (step + 1))
            v_hat = v / (1 - beta2 ** (step + 1))
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            z -= z.mean()
            
            if (step + 1) % 50000 == 0:
                elapsed = time.monotonic() - t0

def direct_subgradient_polish(a0, time_limit, n_iters=500000):
    """Direct subgradient method with backtracking line search on the max."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    eta = 0.1
    patience = 0
    max_patience = 100
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        score = 2.0 * n * max_b
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > max_patience:
            eta *= 0.7
            patience = 0
            if eta < 1e-8:
                break
        
        # Find active set
        threshold = max_b * (1.0 - 1e-5)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Gradient
        w = np.zeros(len(b))
        w[active] = 1.0 / len(active)
        g = compute_gradient(a, w)
        
        # Project gradient onto tangent space of simplex
        g_proj = g - np.mean(g)
        
        # Backtracking line search
        a_new = a - eta * g_proj
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        score_new = 2.0 * n * fast_convolve(a_new).max()
        
        if score_new < score:
            a = a_new
        else:
            # Halve step size
            a_new = a - 0.5 * eta * g_proj
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
            a = a_new
            eta *= 0.99
        
        if it % 100000 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad iter {it} ({elapsed:.0f}s): best={best_score:.8f} eta={eta:.6f}")
    
    return best_a, best_score


def nesterov_lp_alternating(a0, time_limit, solver, n_max_iters=500000):
    """Nesterov-accelerated LP alternating method."""
    n = len(a0)
    a = normalize(a0.copy())
    
    y = a.copy()  # Nesterov interpolation point
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # LP best response
        b_new = solve_lp_best_response(y, solver)
        if b_new is None:
            if it > 100:
                break
            continue
        
        # Nesterov update
        alpha = 2.0 / (it + 2)
        y_new = (1.0 - alpha) * y + alpha * b_new
        y_new = normalize(y_new)
        
        a_new = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a_new = normalize(a_new)
        
        score_a = 2.0 * n * fast_convolve(a_new).max()
        score_y = 2.0 * n * fast_convolve(y_new).max()
        
        if score_a < best_score:
            best_score = score_a
            best_a = a_new.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y_new.copy()
        
        a = a_new
        y = y_new
        
        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            
            if stall_count >= 30:
                # Perturb to escape
                noise = 0.01 * rng.standard_normal(n)
                y = y + noise
                y = normalize(y)
                stall_count = 0
            
            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_gradient(a, w):
    """Compute gradient: grad[j] = 2 * sum_k w[k] * a[k-j] using cross-correlation."""
    n = len(a)

    return 2.0 * c[:n]
