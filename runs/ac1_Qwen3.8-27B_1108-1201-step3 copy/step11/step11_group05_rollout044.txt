2. Peak-directed subgradient with adaptive active set tracking
3. Multi-n exploration (1200, 1500, 1800) with interpolation
4. Lopsided initializations exploiting the asymmetry of convolution
5. Final fine-grained Polish with top-peak targeting
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def local_score(a):
    """Compute the objective value locally."""
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS, cp.GUROBI, cp.MOSEK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_response(a_fixed, T_mat, solver, n):
    """Given fixed a, find b minimizing max_k conv(a,b)[k] subject to b>=0, sum(b)=1."""
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    for s in [solver, cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                break
        except Exception:
            continue
    else:
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    b_val[np.isnan(b_val)] = 0
    b_val[np.isinf(b_val)] = 0
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(max(0.0, M_var.value))


def nesterov_lp_optimize(a0, time_limit, solver, n_max_iters=500000):
    """Nesterov-accelerated LP alternating optimization."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)
    T_cache = None
    a_cache = None

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Recompute T matrix if a has changed significantly
        if T_cache is None or it % 50 == 0:
            T_cache = build_conv_matrix(y)
            a_cache = y.copy()

        b_new, M_b = solve_lp_response(y, T_cache, solver, n)
        if b_new is None:
            continue

        # Nesterov dual averaging
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = local_score(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = local_score(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 30:
                # Perturb to escape
                x = np.linspace(0, 4 * np.pi, n)
                freq = rng.integers(2, 15)
                y = y + 0.02 * np.cos(freq * x / (n / 200) + rng.uniform(0, 2 * np.pi))
                y = np.maximum(y, 0)
                y = y / y.sum()
                T_cache = None
                stall_count = 0

            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def peak_directed_subgradient(a0, time_limit, n_steps=2000000):
    """Fast subgradient method focusing on top peaks with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()

    # Adam parameters
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    lr = 0.01
    step_count = 0
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        score = 2.0 * n * max_b

        step_count += 1
        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            lr *= 0.9
            no_improve = 0
            if lr < 1e-6:
                break

        # Adaptive active set: top fraction of peaks
        top_frac = max(0.05, 0.3 - 0.25 * min(1.0, step / 50000))
        top_k = max(3, int(top_frac * len(b)))
        idx = np.argpartition(b, -top_k)[-top_k:]
        
        # Soft weights on active peaks
        b_active = b[idx]
        max_active = b_active.max()
        # Use a soft max weighting
        scale = max(0.01, (max_active - b_active.min()) / max(1, len(idx)))
        w_active = np.exp((b_active - max_active) / scale)
        w_active = w_active / w_active.sum()
        
        w = np.zeros(len(b))
        w[idx] = w_active

        # Gradient: g_a[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i)
        # Use FFT for efficiency
        fft_size = 1 << (2 * n - 1).bit_length()
        W = rfft(w, fft_size)
        A_rev = rfft(a[::-1], fft_size)
        g_a = irfft(W * A_rev, fft_size)[:n]
        g_a *= 2.0  # factor of 2 from the objective

        # Project gradient to simplex tangent space
        mean_g = g_a.mean()
        g_proj = g_a - mean_g

        # Adam update
        m = beta1 * m + (1 - beta1) * g_proj
        v = beta2 * v + (1 - beta2) * g_proj ** 2
        m_hat = m / (1 - beta1 ** (step + 1))
        v_hat = v / (1 - beta2 ** (step + 1))
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        a = a_new

        if step % 10000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s step={step}: score={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def logsumexp_adam(a0, time_limit, T_schedule, lr_init=0.005):
    """Adam optimization in log-space with LogSumExp approximation of max."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-30)).copy()
    z -= z.mean()
    t0 = time.monotonic()

    best_a = a0.copy()
    best_score = local_score(a0)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step_count = 0

    total_stages = len(T_schedule)
    steps_per_stage = max(2000, int(time_limit * 8000 / total_stages))

    for stage_idx, T in enumerate(T_schedule):
        if time.monotonic() > t0 + time_limit:
            break
        if T < 0:
            break

        top_k = max(5, int(0.15 * (2 * n - 1)))

        for step in range(steps_per_stage):
            if time.monotonic() > t0 + time_limit:
                break

            a = np.exp(z - z.max())
            a = a / a.sum()
            b = fast_convolve(a)
            max_b = b.max()
            score = 2.0 * n * max_b
            step_count += 1

            if score < best_score:
                best_score = score
                best_a = a.copy()

            # Compute gradient using top-k LogSumExp
            if T < 0.01:
                idx_peak = np.argpartition(b, -top_k)[-top_k:]
                b_top = b[idx_peak]
                w_peak = np.ones(len(idx_peak)) / len(idx_peak)
                w = np.zeros(len(b))
                w[idx_peak] = w_peak
            else:
                idx_peak = np.argpartition(b, -top_k)[-top_k:]
                b_top = b[idx_peak]
                max_top = b_top.max()
                exp_top = np.exp((b_top - max_top) / T)
                w_top = exp_top / exp_top.sum()
                w = np.zeros(len(b))
                w[idx_peak] = w_top

            # Gradient w.r.t. a
            fft_size = 1 << (2 * n - 1).bit_length()
            W = rfft(w, fft_size)
            A_rev = rfft(a[::-1], fft_size)
            g_a = irfft(W * A_rev, fft_size)[:n]
            g_a *= 2.0

            # Convert to z-space gradient
            S = np.dot(g_a, a)
            grad_z = a * (g_a - S)

            # Adam update
            lr = lr_init * (0.85 ** stage_idx)
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** (step + 1))
            v_hat = v / (1 - beta2 ** (step + 1))
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            z -= z.mean()

            if (step + 1) % 5000 == 0:
                elapsed = time.monotonic() - t0
                if elapsed > time_limit:
                    break

    return best_a, best_score


def lopsided_init(n, rng, style=0):
    """Create lopsided initial conditions that avoid symmetric center peak."""
    x = np.linspace(0, 1, n)
    if style == 0:
        # Exponential decay
        a = np.exp(-3.0 * x)
    elif style == 1:
        # Power law decay
        a = (1 - x) ** 3 + 0.01
    elif style == 2:
        # Gaussian shifted to left
        a = np.exp(-8 * (x - 0.25) ** 2) + 0.05
    elif style == 3:
        # Cosine bump on left half
        a = np.cos(np.pi * x / 2) ** 4 * (x < 0.6) + 0.02 * (x >= 0.6)
    elif style == 4:
        # Two-sided with asymmetry
        a = 0.7 * np.exp(-5 * (x - 0.3) ** 2) + 0.3 * np.exp(-10 * (x - 0.8) ** 2)
    elif style == 5:
        # Sawtooth + decay
        a = (1 - x) * (1 + 0.3 * np.sin(6 * np.pi * x))
        a = np.maximum(a, 0.01)
    elif style == 6:
        # Broad left, narrow right
        a = np.where(x < 0.5, 1.5 * (1 - 2*x), 0.3 * np.exp(-10*(x-0.75)**2))
        a = np.maximum(a, 0.005)
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def multiplicative_weights(a0, time_limit, n_steps=500000):
    """Mirror descent / multiplicative weights on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    eta = 0.05
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Active set: peaks within 1% of max
        threshold = max_b * 0.995
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Compute gradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Multiplicative update
        log_update = -eta * g
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        score = local_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            eta *= 0.7
            no_improve = 0
            if eta < 1e-5:
                break

        if step % 50000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    MW {elapsed:.0f}s: score={best_score:.8f} eta={eta:.6f}")

    return best_a, best_score


def propose_candidate(seed=42, time_limit=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    
    Strategy:
    - Phase 1: Lopsided initialization + fast mirror descent warmup
    - Phase 2: Nesterov LP alternating (main workhorse)
    - Phase 3: LogSumExp Adam refinement
    - Phase 4: Second LP round + multiplicative weights polish
    - Phase 5: Try n=1500 and n=1800
    - Phase 6: Final targeted peak reduction
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(time_limit) - 15.0)
    
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Budget: {time_limit:.0f}s")

    # Load parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = local_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except (NameError, Exception) as e:
        print(f"No parent available: {e}")

    if parent is None:
        n_init = 1200
        # Try multiple lopsided inits and pick best
        best_init = None
        best_init_score = float('inf')
        for style in range(7):
            a_try = lopsided_init(n_init, rng, style=style)
            sc = local_score(a_try)
            if sc < best_init_score:
                best_init_score = sc
                best_init = a_try
        best_seq = best_init
        best_score = best_init_score
        print(f"Best init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = local_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}) ***")

    # === Phase 1: Fast mirror descent warmup (diverse starts) ===
    print(f"\n=== Phase 1: Mirror Descent Warmup n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        # From parent
        a1, _ = multiplicative_weights(best_seq, t_alloc * 0.6, n_steps=200000)
        try_update(a1, "mw_parent")
        
        rem = deadline - time.monotonic()
        if rem > 30:
            # From a lopsided init
            a_lops = lopsided_init(n_best, rng, style=1)
            a2, _ = multiplicative_weights(a_lops, t_alloc * 0.4, n_steps=100000)
            try_update(a2, "mw_lopsided")

    # === Phase 2: Main LP Alternating with Nesterov ===
    print(f"\n=== Phase 2: Nesterov LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_optimize(best_seq, t_alloc, solver, n_max_iters=500000)
        try_update(a_lp, "nesterov_lp_1")

    # === Phase 3: LogSumExp Adam refinement ===
    print(f"\n=== Phase 3: Adam LogSumExp (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        T_sched = [8.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005]
        a_adam, _ = logsumexp_adam(best_seq, t_alloc, T_sched, lr_init=0.004)
        try_update(a_adam, "adam_lse")

    # === Phase 4: Second LP + polish ===
    print(f"\n=== Phase 4: LP Round 2 + Polish (score={best_score:.8f}) ===")

        print(f"  Budget: {t_alloc:.0f}s")

        try_update(a_lp2, "nesterov_lp_2")
        
        rem = deadline - time.monotonic()
        if rem > 20:
            a_mw, _ = multiplicative_weights(best_seq, min(rem * 0.5, 60.0), n_steps=100000)
            try_update(a_mw, "mw_polish")

        print(f"  Trying n={n_try}, budget={t_n:.0f}s")
        init_1500 = interpolate_seq(best_seq, n_try)
        
        # Quick subgradient first
        t_sub = t_n * 0.3
        a_sg, _ = peak_directed_subgradient(init_1500, t_sub, n_steps=300000)
        try_update(a_sg, f"subgrad_n{n_try}")
        
        rem = deadline - time.monotonic()
        if rem > 60:
            t_lp_n = min(rem * 0.5, 100.0)
            a_lp_n, _ = nesterov_lp_optimize(a_sg if local_score(a_sg) < local_score(init_1500) else init_1500, t_lp_n, solver, n_max_iters=200000)

    if rem > 100:

        print(f"  Trying n={n_try}, budget={t_n:.0f}s")
        init_1800 = interpolate_seq(best_seq, n_try)
        
        t_sub = t_n * 0.4

        try_update(a_sg, f"subgrad_n{n_try}")
        
        rem = deadline - time.monotonic()
        if rem > 40:
            t_lp_n = min(rem * 0.5, 60.0)
            a_lp_n, _ = nesterov_lp_optimize(a_sg, t_lp_n, solver, n_max_iters=100000)
            try_update(a_lp_n, f"lp_n{n_try}")

    # Try smaller n=800 (sometimes smaller n has better score due to the 2n factor)

        n_try = 800
        t_n = min(rem * 0.15, 80.0)

        init_800 = interpolate_seq(best_seq, n_try)

        rem = deadline - time.monotonic()
        if rem > 20:
            a_lp_n, _ = nesterov_lp_optimize(a_sg, min(rem * 0.5, 30.0), solver, n_max_iters=50000)
            try_update(a_lp_n, f"lp_n{n_try}")

    # === Phase 6: Final targeted peak reduction ===
    print(f"\n=== Phase 6: Final Peak Targeting (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        n_cur = len(best_seq)
        t_fin = min(rem * 0.4, 60.0)
        print(f"  Budget: {t_fin:.0f}s, n={n_cur}")
        
        # Very fine Adam with low temperature
        T_final = [0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.001]
        a_fin, _ = logsumexp_adam(best_seq, t_fin, T_final, lr_init=0.002)
        try_update(a_fin, "final_adam")

        rem = deadline - time.monotonic()
        if rem > 20:
            a_fin2, _ = multiplicative_weights(best_seq, min(rem * 0.4, 30.0), n_steps=100000)
            try_update(a_fin2, "final_mw")

        rem = deadline - time.monotonic()
        if rem > 15:
            a_fin3, _ = nesterov_lp_optimize(best_seq, min(rem * 0.5, 20.0), solver, n_max_iters=50000)
            try_update(a_fin3, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = local_score(best_seq)
        if np.isfinite(fs):
            best_score = fs
        final_eval = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
        print(f"\nFINAL SCORE (local): {best_score:.8f} (n={len(best_seq)})")
        print(f"FINAL SCORE (evaluate_sequence): {final_eval:.8f}")
    else:
        best_seq = np.ones(1200) / 1200.0
        print("\nFALLBACK TO UNIFORM")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
