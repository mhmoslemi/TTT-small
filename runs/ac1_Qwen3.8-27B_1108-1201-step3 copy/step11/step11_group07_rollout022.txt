```python
    # Maximize sum(g) subject to:
    #   (f * g)[k] <= M for k = 0, ..., 2n-2
    #   g[j] >= 0 for j = 0, ..., n-1
```

```python
1. LP-based iterative refinement (main driver)

4. Multi-length exploration

import numpy as np

from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def score(a):
    """Compute the objective value."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(f):
    """Build the (2n-1) x n matrix A where conv(f,g) = A @ g."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi+1] = f[k-lo::-1][:hi-lo+1]
    return A


def lp_step(f, M, verbose=False):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = build_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 60})
    if res.success:
        return res.x
    return None


def lp_refine(a0, time_limit, max_iter=500, verbose=True):
    """Iteratively refine using LP steps."""
    n = len(a0)
    f = normalize(a0.copy())
    best_f = f.copy()
    best_score = score(f)
    t0 = time.monotonic()
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_f = fast_convolve(f)
        M = conv_f.max()
        
        g = lp_step(f, M)
        if g is None:
            break
        
        g = normalize(g)
        
        # Line search on t
        found = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.01, 0.005]:
            f_new = normalize((1 - t) * f + t * g)
            sc = score(f_new)
            if sc < best_score:
                best_score = sc
                best_f = f_new.copy()
                f = f_new.copy()
                found = True
                break
            # Also accept if M decreased even if score increased slightly (shouldn't happen)
        
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP iter {it}: score={best_score:.8f}, M={M:.6f}, elapsed={elapsed:.0f}s")
        
        if not found:
            # Try smaller steps or break
            if it > 50:
                break
            # Jitter
            g = normalize(g + 0.01 * np.random.default_rng(it).normal(0, 1, n))
    
    return best_f, best_score
```

```python
def lse_objective_grad(z, T):
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    conv_a = fast_convolve(a)
    c_max = conv_a.max()
    exp_c = np.exp((conv_a - c_max) / T)
    w = exp_c / exp_c.sum()
    lse = c_max + T * np.log(exp_c.sum())
    obj = 2.0 * n * lse
    
    fft_size = 1 << (3*n - 2 - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    cross = irfft(W * np.conj(A), fft_size)[:n]
    g_a = 4.0 * n * cross
    
    S = np.dot(g_a, a)
    grad_z = a * (g_a - S)
    
    return obj, grad_z
```
