    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    # For row k, column j: T[k,j] = a[k-j] if 0 <= k-j < n else 0
    # This is a Toeplitz matrix
    T = np.zeros((size, n))
    for j in range(n):
        # Row k, col j: a[k-j] for k-j in [0, n)
        # k ranges from j to j+n-1
        T[j:j+n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return None


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b>=0, sum(b)=1 minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers_to_try = [s for s in [solver, cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS] if s is not None]
    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                b_val[np.isnan(b_val)] = 0
                sv = b_val.sum()
                if sv < 1e-15:
                    continue
                b_val = b_val / sv
                return b_val
        except Exception:
            continue
    
    return None


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.05
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.25 * np.cos(2 * np.pi * 7 * x) + 0.15 * np.cos(2 * np.pi * 13 * x + 0.5)
        a = np.maximum(a, 0)
    elif choice == 7:
        # Hamming-like window
        a = 0.54 - 0.46 * np.cos(2 * np.pi * x)
    elif choice == 8:
        # Flat top
        a = 0.2156 + 0.4160 * np.cos(2 * np.pi * x) + 0.2742 * np.cos(4 * np.pi * x) + 0.0840 * np.cos(6 * np.pi * x) + 0.0088 * np.cos(8 * np.pi * x)
    else:
        a = np.ones(n) / n
        a += 0.05 * np.random.default_rng(12345).normal(size=n)
        a = np.maximum(a, 0)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def adam_logsumexp(a0, time_limit, T_stages, lr_init=0.003, lr_decay=0.85, top_k_frac=0.1):
    """Adam optimization in log-space with temperature annealing and top-k focusing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step_count = 0
    
    total_stages = len(T_stages)
    
    for stage_idx, T_stage in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 1:
            break
        
        top_k = max(10, int(top_k_frac * (2 * n - 1)))
        steps_in_stage = 0
        stage_lr = lr_init * (lr_decay ** stage_idx)
        
        while time.monotonic() < deadline:
            a = softmax_from_z(z)
            b = fast_convolve(a)
            max_b = b.max()
            score = 2.0 * n * max_b
            step_count += 1
            steps_in_stage += 1
            
            if score < best_score:
                best_score = score
                best_a = a.copy()
            
            # Compute weights
            if T_stage < 0.005:
                # Pure active set
                threshold = max_b * (1.0 - 1e-5)
                active = np.where(b >= threshold)[0]
                if len(active) == 0:
                    active = np.array([np.argmax(b)])
                w = np.zeros(len(b))
                w[active] = 1.0 / len(active)
            else:
                # Top-k LogSumExp
                if top_k < len(b):
                    idx = np.argpartition(b, -top_k)[-top_k:]
                    b_top = b[idx]
                    max_top = b_top.max()
                    exp_top = np.exp((b_top - max_top) / T_stage)
                    w_top = exp_top / exp_top.sum()
                    w = np.zeros(len(b))
                    w[idx] = w_top
                else:
                    max_b_all = b.max()
                    exp_all = np.exp((b - max_b_all) / T_stage)
                    w = exp_all / exp_all.sum()
            
            # Gradient in a-space
            g_a = compute_grad_a(w, a)
            
            # Gradient in z-space
            S = np.dot(g_a, a)
            grad_z = a * (g_a - S)
            
            # Adam update
            decay = 1.0 / (1.0 + 0.05 * steps_in_stage / 10000)
            lr = stage_lr * decay
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** (step_count + 1))
            v_hat = v / (1 - beta2 ** (step_count + 1))
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Centering
            z -= z.mean()
            
            if steps_in_stage % 50000 == 0:
                elapsed = time.monotonic() - t0
                cur_score = 2.0 * n * fast_convolve(softmax_from_z(z)).max()
                print(f"    Adam stage {stage_idx} (T={T_stage:.3f}) step {steps_in_stage} ({elapsed:.0f}s): best={best_score:.8f} cur={cur_score:.8f}")
                if cur_score > best_score * 1.05:
                    # Diverged, reset
                    z = z_from_a(best_a).copy()
                    m = np.zeros(n)
                    v = np.zeros(n)
                    break
    
    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver, verbose=True):
    """Nesterov dual averaging LP alternating."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(777)
    lp_times = []

    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new = solve_lp_best_response(y, solver)
        lp_times.append(time.monotonic() - lp_t0)

        if b_new is None:
            continue

        avg_lp_time = np.mean(lp_times[-10:]) if len(lp_times) >= 10 else 1.0
        
        # Nesterov dual averaging
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 20 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 40:
                # Perturb to escape
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 10)
                noise = 0.02 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                a = a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall_count = 0

            if verbose and (it + 1) % 100 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} avg_lp={avg_lp_time*1000:.1f}ms")

        if len(lp_times) > 30 and avg_lp_time > 3.0:
            if verbose:
                print(f"    LP too slow ({avg_lp_time:.2f}s/iter), stopping after {it+1} iters")
            break

    return best_a, best_score


def multiplicative_weights(a0, time_limit, eta_base=0.02):
    """Mirror descent (multiplicative weights) targeting active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    no_improve = 0
    scale = 1.0

    for step in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Adaptive active set
        threshold = max_b * (1.0 - max(1e-5, 5e-4 / (1 + step / 500)))
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([0])

        # Gradient: g[i] = mean of a[k-i] over active k
        g = compute_grad_a(np.zeros(len(b)), a)  # placeholder
        # Actually just compute directly
        w = np.zeros(len(b))
        w[active] = 1.0 / len(active)
        g = compute_grad_a(w, a)

        eta = eta_base * scale
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 30:
            scale *= 0.85
            no_improve = 0
            if scale < 0.003:
                break

    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based auto-convolution and gradient computation for speed
2. Adam optimization in log-space with adaptive top-k focusing and aggressive annealing
3. LP-based best response with Nesterov acceleration for global refinement
4. Multi-length exploration (n=2000, 2500) to exploit larger search space
5. Structured initializations, perturbation restarts, and multiplicative weights polish

Key improvements over previous:
- FFT-based gradient (80x faster than np.correlate for large n)
- Larger primary n (2000-2500 vs 1200) for more optimization freedom
- More aggressive temperature annealing schedule
- Better time allocation favoring LP (guaranteed improvement per iteration)
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_grad_a(w, a):
    """Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    w has length 2n-1 (weights on convolution positions), a has length n.
    Returns gradient of length n.
    """
    n = len(a)
    n_w = len(w)
    conv_len = n_w + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()

    W = rfft(w, fft_size)
    W_rev = rfft(w[::-1], fft_size)
    A = rfft(a, fft_size)

    # Second term: (w * a)[:n] gives sum_k w[k]*a[i-k]
    conv_wa = irfft(W * A, fft_size)
    term2 = conv_wa[:n]

    # First term: (w_rev * a)[n_w-1 : n_w-1+n] gives sum_k w[k]*a[k-i]
    conv_wra = irfft(W_rev * A, fft_size)
    term1 = conv_wra[n_w - 1:n_w - 1 + n]

    return term1 + term2


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return None


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b>=0, sum(b)=1 minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    seen = set()
    solvers_to_try = [s for s in [solver, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
                      if s is not None and s not in seen and not seen.add(s)]
    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                b_val[np.isnan(b_val)] = 0
                sv = b_val.sum()
                if sv < 1e-15:
                    continue
                b_val = b_val / sv
                return b_val
        except Exception:
            continue

    return None


def evaluate_local(a):
    """Evaluate sequence score locally (fast, assumes sum(a)=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.05
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.25 * np.cos(2 * np.pi * 7 * x) + 0.15 * np.cos(2 * np.pi * 13 * x + 0.5)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = 0.54 - 0.46 * np.cos(2 * np.pi * x)
    elif choice == 8:
        a = 0.2156 + 0.4160 * np.cos(2 * np.pi * x) + 0.2742 * np.cos(4 * np.pi * x) + 0.0840 * np.cos(6 * np.pi * x) + 0.0088 * np.cos(8 * np.pi * x)
    else:
        a = np.ones(n) / n
        a += 0.05 * rng.normal(size=n)
        a = np.maximum(a, 0)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def adam_optimize(a0, time_limit, T_stages, lr_init=0.003, lr_decay=0.85, top_k_frac=0.1):
    """Adam optimization in log-space with temperature annealing and top-k focusing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit

    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step_count = 0

    for stage_idx, T_stage in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 0.5:
            break

        top_k = max(10, int(top_k_frac * (2 * n - 1)))
        steps_in_stage = 0
        stage_lr = lr_init * (lr_decay ** stage_idx)

        while time.monotonic() < deadline:
            a = softmax_from_z(z)
            b = fast_convolve(a)
            max_b = b.max()
            score = 2.0 * n * max_b
            step_count += 1
            steps_in_stage += 1

            if score < best_score:
                best_score = score
                best_a = a.copy()

            # Compute weights on convolution positions
            if T_stage < 0.003:
                # Pure active set
                threshold = max_b * (1.0 - 1e-5)
                active = np.where(b >= threshold)[0]
                if len(active) == 0:
                    active = np.array([np.argmax(b)])
                w = np.zeros(len(b))
                w[active] = 1.0 / len(active)
            else:
                # Top-k LogSumExp weights
                if top_k < len(b):
                    idx = np.argpartition(b, -top_k)[-top_k:]
                    b_top = b[idx]
                    max_top = b_top.max()
                    exp_top = np.exp((b_top - max_top) / T_stage)
                    w_top = exp_top / exp_top.sum()
                    w = np.zeros(len(b))
                    w[idx] = w_top
                else:
                    exp_all = np.exp((b - max_b) / T_stage)
                    w = exp_all / exp_all.sum()

            # Gradient in a-space via FFT
            g_a = compute_grad_a(w, a)

            # Gradient in z-space
            S = np.dot(g_a, a)
            grad_z = a * (g_a - S)

            # Adam update with linear decay within stage
            lr = stage_lr / (1.0 + 0.1 * steps_in_stage / 8000)
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** (step_count + 1))
            v_hat = v / (1 - beta2 ** (step_count + 1))
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            # Centering for numerical stability
            z -= z.mean()

            if steps_in_stage % 30000 == 0:
                cur_score = 2.0 * n * fast_convolve(softmax_from_z(z)).max()
                if cur_score > best_score * 1.1:
                    # Diverged, reset to best
                    z = z_from_a(best_a).copy()
                    m = np.zeros(n)
                    v = np.zeros(n)
                    break
                if steps_in_stage % 100000 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"    Adam s{stage_idx} T={T_stage:.3f} st={steps_in_stage} ({elapsed:.0f}s): best={best_score:.8f} cur={cur_score:.8f}")

    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver, verbose=True):
    """Nesterov dual averaging LP alternating method."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(777)
    lp_times = []

    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new = solve_lp_best_response(y, solver)
        lp_times.append(time.monotonic() - lp_t0)

        if b_new is None:
            continue

        avg_lp_time = np.mean(lp_times[-10:]) if len(lp_times) >= 10 else 1.0

        # Nesterov dual averaging step
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 10 == 0:
            if best_score < prev_best - 1e-12:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 30:
                # Perturb to escape local minimum
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 12)
                noise = 0.015 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                a = a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall_count = 0

            if verbose and (it + 1) % 50 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovLP it={it+1} ({elapsed:.0f}s): score={best_score:.8f} lp={avg_lp_time*1000:.0f}ms")

        if len(lp_times) > 20 and avg_lp_time > 4.0:
            if verbose:
                print(f"    LP too slow ({avg_lp_time:.2f}s/iter), stopping at it={it+1}")
            break

    return best_a, best_score


def multiplicative_weights(a0, time_limit, eta_base=0.015):
    """Mirror descent (multiplicative weights) targeting active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    no_improve = 0
    scale = 1.0

    for step in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Adaptive active set (widen as we go)
        threshold = max_b * (1.0 - max(1e-5, 3e-4 / (1 + step / 300)))
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([0])

        w = np.zeros(len(b))
        w[active] = 1.0 / len(active)
        g = compute_grad_a(w, a)

        eta = eta_base * scale
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 25:
            scale *= 0.85
            no_improve = 0
            if scale < 0.002:
                break

    return best_a, best_score

            print(f"  LP n=2000 budget: {t_lp_up:.0f}s")
            a_up_lp, _ = nesterov_lp_refine(best_seq, t_lp_up, solver)
            local_update(a_up_lp, "lp_n2000")

    # === Phase 3: Main LP on current best ===
    print(f"\n=== Phase 3: Nesterov LP n={len(best_seq)} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.22, 300.0)
    if t_alloc > 30 and solver is not None:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver)
        local_update(a_lp, "nesterov_lp_main")

    # === Phase 4: Fine Adam with aggressive annealing ===
    print(f"\n=== Phase 4: Adam Fine (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        T_fine = [3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001]
        a_adam2, _ = adam_optimize(best_seq, t_alloc, T_fine,
                                    lr_init=0.003, lr_decay=0.87, top_k_frac=0.05)
        local_update(a_adam2, "adam_fine")

    # === Phase 5: Try n=2500 ===
    print(f"\n=== Phase 5: n=2500 exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 150 and len(best_seq) < 2500:
        n_try = 2500
        a_try = interpolate_sequence(best_seq, n_try)
        t_sub = min(rem * 0.25, 150.0)
        print(f"  Budget: {t_sub:.0f}s")
        T_try = [5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
        a_try_opt, _ = adam_optimize(a_try, t_sub, T_try,
                                      lr_init=0.004, lr_decay=0.85, top_k_frac=0.1)
        local_update(a_try_opt, "adam_n2500")

        rem = deadline - time.monotonic()
        if rem > 60 and solver is not None:
            t_lp_try = min(rem * 0.3, 100.0)
            a_try_lp, _ = nesterov_lp_refine(a_try_opt, t_lp_try, solver, verbose=False)
            local_update(a_try_lp, "lp_n2500")

    # === Phase 6: Second LP round ===
    print(f"\n=== Phase 6: LP Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 180.0)
    if t_alloc > 30 and solver is not None:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver)
        local_update(a_lp2, "lp_round2")

    # === Phase 7: Multiplicative weights ===
    print(f"\n=== Phase 7: MultiWeights (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 80.0)
    if t_alloc > 10:
        a_mw, _ = multiplicative_weights(best_seq, t_alloc)
        local_update(a_mw, "mult_weights")

    # === Phase 8: Perturbation restarts ===
    print(f"\n=== Phase 8: Perturbation (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        n_cur = len(best_seq)
        x = np.linspace(0, 2 * np.pi, n_cur)
        for trial in range(4):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 15:
                break
            freq = rng.integers(1, 15)
            phase = rng.uniform(0, 2 * np.pi)
            strength = 0.005 + 0.004 * trial
            a_pert = best_seq + strength * np.cos(freq * x + phase)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            t_sub = min(rem_sub * 0.2, 25.0)
            T_pert = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.003]
            a_opt, _ = adam_optimize(a_pert, t_sub, T_pert,
                                      lr_init=0.003, lr_decay=0.85, top_k_frac=0.1)
            local_update(a_opt, f"pert_{trial}")

    # === Phase 9: Final polish ===
    print(f"\n=== Phase 9: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        t_fin = min(rem * 0.4, 20.0)
        T_final = [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01, 0.005, 0.002, 0.001]
        a_fin, _ = adam_optimize(best_seq, t_fin, T_final,
                                  lr_init=0.001, lr_decay=0.9, top_k_frac=0.03)
        local_update(a_fin, "final_adam")

    rem = deadline - time.monotonic()
    if rem > 8 and solver is not None:
        a_fin_lp, _ = nesterov_lp_refine(best_seq, min(rem * 0.7, 15.0), solver, verbose=False)
        local_update(a_fin_lp, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        final_eval = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
        print(f"\nFINAL SCORE (local): {best_score:.8f} (n={len(best_seq)})")
        print(f"FINAL SCORE (evaluate_sequence): {final_eval:.8f}")
        print(f"Total time: {time.monotonic() - t_start:.1f}s")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200.0

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
