- reverse(a) has elements: a[n-1], a[n-2], ..., a[0]
- g_k[i] = 2*a[k-i]

Let j = k-i, so i = k-j. g_k[k-j] = 2*a[j].
As i goes from i_min to i_max, j goes from k-i_min to k-i_max.

For k < n: i from 0 to k, j from k to 0.

For k >= n: i from k-n+1 to n-1, j from n-1 to k-n+1.
g_k[k-n+1] = 2*a[n-1], ..., g_k[n-1] = 2*a[k-n+1].

    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.exp(-4 * (x - 0.5) ** 2)
    elif choice == 1:
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        a = 1.0 / (1.0 + 20.0 * (x - 0.5) ** 2)
    elif choice == 3:
        a = np.sin(np.pi * x) ** 0.5
    elif choice == 4:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.6 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 5:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.1 * np.cos(8 * np.pi * x)
    elif choice == 6:
        a = (1.0 - np.abs(2 * x - 1)) ** 0.7
    else:
        a = np.exp(-2 * (x - 0.5) ** 2) * (1.0 + 0.3 * np.cos(6 * np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def build_conv_matrix(a_fixed):
    """Build convolution matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lse_optimize(a0, time_limit, T_start=100.0, T_end=0.3, n_stages=20):
    """Multi-stage LSE optimization using L-BFGS with FFT gradients."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    z = np.log(np.maximum(a, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()

    global _gT, _gN
    _gN = n

    total_iters = max(50, min(500, int(20 * n_stages)))
    iters_per_stage = max(5, total_iters // n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        _gT = float(stage_T)
        max_iter = max(10, iters_per_stage)

        try:
            res = minimize(
                fun=_lse_obj, x0=z, jac=_lse_jac, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': max_iter * 300,
                         'disp': False}
            )
            z = res.x
            a = softmax_fn(z)
            score = 2.0 * n * fast_conv(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception:
            pass

    return best_a, best_score


def mirror_descent_optimize(a0, time_limit, top_k=5, eta_init=None, seed=42):
    """
    Mirror descent on simplex with subgradient of max_k R_k.
    Uses running average for convergence guarantee.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    if eta_init is None:
        eta_init = 5.0 / n

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()
    rng = np.random.default_rng(seed)

    avg_a = np.zeros(n)
    n_avg = 0
    eta = eta_init
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        # Compute subgradient of max
        R = fast_conv(a)
        top_idx = np.argsort(R)[::-1][:top_k]
        g = np.zeros(n)
        for k in top_idx:
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            for i in range(i_min, i_max + 1):
                g[i] += 2.0 * a[k - i]
        g /= top_k

        # Mirror descent step: a_new = a * exp(-eta * g) / Z
        log_ratio = -eta * g
        log_ratio -= log_ratio.max()
        weights = np.exp(log_ratio)
        a_new = a * weights
        s = a_new.sum()
        if s < 1e-30:
            a_new = a.copy()
        else:
            a_new = a_new / s

        # Check score
        score = 2.0 * n * fast_conv(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        # Running average (theoretical convergence guarantee)
        avg_a += a_new
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_conv(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        a = a_new
        iter_count += 1

        # Adaptive eta: decrease slowly
        if iter_count % 100 == 0:
            eta *= 0.97

        # Occasionally restart average to avoid being stuck
        if iter_count % 500 == 0:
            avg_a = a.copy()
            n_avg = 1

        if iter_count % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MD iter={iter_count} ({elapsed:.0f}s): best={best_score:.8f}, eta={eta:.6f}")

    # Also check the original a in case last step was bad
    score_orig = 2.0 * n * fast_conv(a).max()
    if score_orig < best_score:
        best_score = score_orig
        best_a = a.copy()

    return best_a, best_score
