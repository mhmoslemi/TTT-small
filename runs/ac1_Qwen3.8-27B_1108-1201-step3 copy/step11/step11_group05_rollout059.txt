def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

    n = len(a)
    size = 2 * n - 1

    # Use scipy's scipy.linalg.toeplitz or direct construction
    T = np.zeros((size, n))
    for j in range(n):

    return T

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (normalized), find b minimizing max conv(a,b)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    for s in [solver, cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                break
        except Exception:
            continue
    else:
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    sv = b_val.sum()
    if sv < 1e-15:
        return None
    return b_val / sv


def coordinate_descent(a_init, time_limit, rng, grid_points=150, fine_points=80,
                       random_order=False, renorm_interval=1):
    """
    Coordinate descent: for each coordinate, find optimal value via 1D grid+refinement.
    The 1D subproblem minimizes 2n*max_k(b_0[k]+c[k]*x+d[k]*x^2)/(S+x)^2 over x>=0.
    """
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    prev_best = best_score

    # Pre-allocate arrays for speed
    c = np.zeros(2 * n - 1)
    d = np.zeros(2 * n - 1)

    while time.monotonic() - t0 < time_limit:
        if random_order:
            indices = rng.permutation(n)
        else:
            indices = np.arange(n)

        for i in indices:
            if time.monotonic() - t0 > time_limit:
                break

            old_ai = a[i]
            a[i] = 0.0
            b_0 = fast_convolve(a)
            S_other = a.sum()

            # Build c and d vectors
            c.fill(0.0)
            d.fill(0.0)

            k_start = max(0, i)
            k_end = min(2 * n - 2, i + n - 1)
            if k_start <= k_end:
                c[k_start:k_end + 1] = 2.0 * a[k_start - i:k_end - i + 1]

            k_center = 2 * i
            if 0 <= k_center < 2 * n - 1:
                d[k_center] = 1.0

            # Grid search
            x_max = max(2.0 * old_ai + 0.001, 0.005)
            x_grid = np.linspace(0.0, x_max, grid_points)

            # Compute scores vectorized
            B = b_0[:, None] + c[:, None] * x_grid[None, :] + d[:, None] * (x_grid[None, :] ** 2)
            max_B = B.max(axis=0)
            scores = 2.0 * n * max_B / (S_other + x_grid) ** 2

            j_best = np.argmin(scores)
            x_best = x_grid[j_best]

            # Refine
            if 0 < j_best < grid_points - 1:
                x_lo = x_grid[max(0, j_best - 2)]
                x_hi = x_grid[min(grid_points - 1, j_best + 2)]
            else:
                x_lo = 0.0
                x_hi = x_grid[-1]

            if x_hi - x_lo > 1e-10:
                x_fine = np.linspace(x_lo, x_hi, fine_points)
                B_fine = b_0[:, None] + c[:, None] * x_fine[None, :] + d[:, None] * (x_fine[None, :] ** 2)
                max_Bf = B_fine.max(axis=0)
                scores_fine = 2.0 * n * max_Bf / (S_other + x_fine) ** 2
                x_best = x_fine[np.argmin(scores_fine)]

            a[i] = x_best

        # End of coordinate loop
        sweeps += 1

        # Renormalize
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa > 1e-15:
            a = a / sa

        score = evaluate_local(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if sweeps % 20 == 1:
            elapsed = time.monotonic() - t0
            print(f"    CoordDescent sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

        if no_improve >= 30:
            # Perturb to escape
            x_pert = np.linspace(0, 2 * np.pi, n)
            freq = rng.integers(1, 15)
            phase = rng.uniform(0, 2 * np.pi)
            strength = 0.005 + 0.003 * no_improve
            a = a + strength * np.cos(freq * x_pert + phase)
            a = np.maximum(a, 0)
            sa = a.sum()
            if sa > 1e-15:
                a = a / sa
            no_improve = 0

    return best_a, best_score


def lp_alternating(a_init, time_limit, solver, n_max_iters=100000):

    n = len(a_init)
    a = np.maximum(a_init.copy(), 0)

    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0

        if time.monotonic() - t0 > time_limit:
            break

        b_new = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa > 1e-15:
            a = a / sa

        score_a = evaluate_local(a)
        score_y = evaluate_local(y)

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 30 == 0:
            if best_score < prev_best - 1e-9:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

                y = np.maximum(y, 0)
                y = y / y.sum()
                stall = 0

            print(f"    LP iter={it+1} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

def adam_logsumexp(a_init, time_limit, T_stages, lr_init, lr_decay, top_k_frac=0.1):
    """Adam optimization in log-space with temperature annealing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-300))
    z -= z.mean()
    t0 = time.monotonic()
    deadline = t0 + time_limit

    best_a = a_init.copy()
    best_score = evaluate_local(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step_count = 0

    total_stages = len(T_stages)
    steps_per_stage = max(3000, int(time_limit * 40000 / total_stages))

    for stage_idx, T_stage in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 1:
            break

        top_k = max(5, int(top_k_frac * (2 * n - 1)))

        for step in range(steps_per_stage):
            if time.monotonic() > deadline:
                break

            a = np.exp(z - z.max())
            a = a / a.sum()
            b = fast_convolve(a)
            max_b = b.max()
            score = 2.0 * n * max_b
            step_count += 1

            if score < best_score:
                best_score = score
                best_a = a.copy()

            # Compute weights
            if T_stage < 0.01:
                threshold = max_b * (1.0 - 1e-4)
                active = np.where(b >= threshold)[0]
                if len(active) == 0:
                    active = np.array([np.argmax(b)])
                w = np.zeros(len(b))
                w[active] = 1.0 / len(active)
            else:
                if top_k < len(b):
                    idx = np.argpartition(b, -top_k)[-top_k:]
                    b_top = b[idx]
                    max_top = b_top.max()
                    exp_top = np.exp((b_top - max_top) / T_stage)
                    w_top = exp_top / exp_top.sum()
                    w = np.zeros(len(b))
                    w[idx] = w_top
                else:
                    max_b_all = b.max()
                    exp_all = np.exp((b - max_b_all) / T_stage)
                    w = exp_all / exp_all.sum()

            # Gradient
            g_a_raw = np.correlate(w, a, mode='full')
            g_a = g_a_raw[n - 1:2 * n - 1] * 2.0
            S = np.dot(g_a, a)
            grad_z = a * (g_a - S)

            # Adam update
            lr = lr_init * (lr_decay ** stage_idx) / (1.0 + 0.1 * step / steps_per_stage)
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** (step + 1))
            v_hat = v / (1 - beta2 ** (step + 1))
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            z -= z.mean()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def spectral_perturb(a, rng, n_modes=5):
    """Perturb in frequency domain to escape local minima."""
    n = len(a)
    A = np.fft.rfft(a)
    mask = np.ones(A.shape, dtype=bool)
    mask[:2] = False
    mask[-1:] = False
    # Perturb a random subset of frequencies
    n_pert = min(n_modes, mask.sum())
    idx = rng.choice(np.where(mask)[0], size=n_pert, replace=False)
    for i in idx:
        A[i] *= (1.0 + rng.uniform(-0.1, 0.1))
    a_new = np.fft.irfft(A, n=n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}) ***")

    # === Phase 1: Coordinate descent from parent (new method) ===
    print(f"\n=== Phase 1: Coordinate Descent n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 250.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd, _ = coordinate_descent(best_seq, t_alloc, rng, grid_points=200,
                                     fine_points=100, random_order=False)
        update_best(a_cd, "coord_descent_1")

    # === Phase 2: LP Alternating ===
    print(f"\n=== Phase 2: LP Alternating n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 40:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = lp_alternating(best_seq, t_alloc, solver, n_max_iters=50000)
        update_best(a_lp, "lp_1")

    # === Phase 3: Coordinate descent from LP result ===
    print(f"\n=== Phase 3: Coord Descent Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd2, _ = coordinate_descent(best_seq, t_alloc, rng, grid_points=200,
                                      fine_points=100, random_order=True)
        update_best(a_cd2, "coord_descent_2")

    # === Phase 4: Adam LogSumExp fine annealing ===
    print(f"\n=== Phase 4: Adam Fine Annealing (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages = [3.0, 1.5, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01]
        a_adam, _ = adam_logsumexp(best_seq, t_alloc, T_stages,
                                   lr_init=0.003, lr_decay=0.85, top_k_frac=0.1)
        update_best(a_adam, "adam_fine")

    # === Phase 5: LP from Adam result ===
    print(f"\n=== Phase 5: LP Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = lp_alternating(best_seq, t_alloc, solver, n_max_iters=30000)
        update_best(a_lp2, "lp_2")

    # === Phase 6: Multi-length exploration ===
    for n_try in [1400, 1600, 1000]:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_alloc = min(rem * 0.12, 120.0)
        if t_alloc < 20:
            break
        print(f"\n=== Phase 6: Try n={n_try} (score={best_score:.8f}) ===")
        print(f"  Budget: {t_alloc:.0f}s")
        init_try = interpolate_sequence(best_seq, n_try)

        # Quick coordinate descent
        t_sub = t_alloc * 0.6
        a_try, _ = coordinate_descent(init_try, t_sub, rng, grid_points=150,
                                      fine_points=80, random_order=False)
        update_best(a_try, f"coord_n{n_try}")

        # Then LP if time
        rem = deadline - time.monotonic()
        if rem > 30:
            t_sub_lp = min(rem * 0.3, 40.0)
            a_try_lp, _ = lp_alternating(a_try, t_sub_lp, solver, n_max_iters=10000)
            update_best(a_try_lp, f"lp_n{n_try}")

    # === Phase 7: Spectral perturbation escape + refinement ===
    print(f"\n=== Phase 7: Spectral Perturbation Escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_cur = len(best_seq)
        for trial in range(3):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 25:
                break
            t_sub = min(rem_sub * 0.25, 50.0)

            a_pert = spectral_perturb(best_seq, rng, n_modes=3 + trial * 2)
            a_opt, _ = coordinate_descent(a_pert, t_sub, rng, grid_points=150,
                                          fine_points=80, random_order=False)
            update_best(a_opt, f"spectral_pert_{trial}")

            # Quick LP polish
            rem = deadline - time.monotonic()
            if rem > 20:
                a_lp_p, _ = lp_alternating(a_opt, min(rem * 0.2, 20.0), solver, n_max_iters=5000)
                update_best(a_lp_p, f"lp_pert_{trial}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.4, 30.0)
        if t_fin > 5:
            a_fin, _ = coordinate_descent(best_seq, t_fin, rng, grid_points=300,
                                          fine_points=150, random_order=False)
            update_best(a_fin, "final_coord")

    rem = deadline - time.monotonic()
    if rem > 15:
        a_fin2, _ = lp_alternating(best_seq, min(rem * 0.5, 20.0), solver, n_max_iters=5000)
        update_best(a_fin2, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        final_eval = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
        print(f"\nFINAL SCORE (local): {best_score:.8f} (n={len(best_seq)})")
        print(f"FINAL SCORE (evaluate_sequence): {final_eval:.8f}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200.0

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
