def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring accurate ones."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.GUROBI, cp.MOSEK, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, verbose=False):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    solved = False
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s, **{
                'tol_gap_rel': 1e-9,
                'tol_ineas': 1e-9,
                'tol_feas': 1e-9,
            }) if s in (cp.CLARABEL, cp.PDLP) else prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            try:
                prob.solve(solver=s)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue

    if not solved:
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value) if M_var.value is not None else None


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lbgfs_obj(z, n, T_stage, deadline):
    """Objective for L-BFGS."""
    if time.monotonic() > deadline:
        return 1e15
    obj, _ = logsumexp_obj_grad(z, n, T_stage)
    return obj


def lbgfs_jac(z, n, T_stage, deadline):
    """Gradient for L-BFGS."""
    if time.monotonic() > deadline:
        return np.zeros_like(z)
    _, g = logsumexp_obj_grad(z, n, T_stage)
    return g


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage, deadline=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit
    n_stages = len(T_stages)
    iters_per_stage = max(30, maxiter_per_stage // n_stages)

    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break

        stages_left = n_stages - stage_i
        avail_time = remaining
        iters = max(20, min(iters_per_stage * 2, int(avail_time * 40 / max(1, stages_left))))

        def func(z_in):
            return lbgfs_obj(z_in, n, stage_T, deadline)

        def jac(z_in):
            return lbgfs_jac(z_in, n, stage_T, deadline)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-25,
                         'gtol': 1e-18, 'maxfun': iters * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def spectral_flatten(a, n_iters=50, blend=0.4, deadline=None):
    """Iterative spectral flattening."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)

        mags = np.abs(F)
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])

        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

    return a


def symmetrize(a):

def trust_region_polish(a0, time_limit, n_peaks=5, n_outer=500):
    """
    Trust-region QP polish using the exact Hessian structure of autocorrelation.
    The Hessian of g_k(a) = conv(a,a)[k] is 2*M_k where M_k[i,j] = 1 if i+j=k.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return None, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    delta = 0.1  # trust region radius
    no_improve = 0

    for outer in range(n_outer):
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - t0 > time_limit - 5:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Identify top peaks
        sort_idx = np.argsort(b)[::-1]
        active = sort_idx[:n_peaks]

        # Compute subgradient: weighted average of gradients of active peaks
        # grad g_k = 2 * a convolved with delta at position k (i.e., shifted a)
        weights = np.ones(len(active)) / len(active)
        grad = np.zeros(n)
        for i, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += weights[i] * 2.0 * a[k - idx]

        # Project gradient to simplex tangent space (remove mean component)
        grad_tan = grad - np.mean(grad)

        # Trust region step: d = -delta * grad_tan / ||grad_tan||
        gnorm = np.linalg.norm(grad_tan)
        if gnorm < 1e-15:
            break
        d = -delta * grad_tan / gnorm

        # Ensure a + d >= 0 and sum(a+d) = 0
        # Scale d down if needed for non-negativity
        min_ratio = np.inf
        for j in range(n):
            if d[j] < -1e-15:
                min_ratio = min(min_ratio, -a[j] / d[j])
        if min_ratio < 1.0:
            d = min_ratio * 0.99 * d

        a_new = a + d
        # Renormalize
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn

        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            delta = min(delta * 1.1, 0.5)
        else:
            no_improve += 1
            delta *= 0.7
            if delta < 1e-6:
                break

        if no_improve >= 30:
            break

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    rem = deadline - time.monotonic()
    print(f"Total budget: {rem:.0f}s, starting score: {best_score:.10f}")

    # === Phase 0: Symmetrize and quick spectral flatten (20s) ===
    print(f"\n=== Phase 0: Init ({best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 25:
        a_sym = symmetrize(best_seq)
        local_update(a_sym, "symmetrized")
        t_spec = min(rem * 0.012, 15.0)
        if t_spec > 5:
            a_sp = spectral_flatten(best_seq, n_iters=30, blend=0.3, deadline=time.monotonic() + t_spec)
            a_sp = symmetrize(a_sp)
            local_update(a_sp, "spec_flatten")

    # === Phase 1: LP Alternating - Main (500s) ===
    print(f"=== Phase 1: LP Alt n={n_best} ({best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_lp1 = min(rem * 0.28, 500.0)
    if t_lp1 > 30:
        print(f"  Budget: {t_lp1:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_lp1, solver, n_max_iters=2000000)
        local_update(a_lp, "lp_main")

    # === Phase 2: Trust Region QP Polish (150s) ===
    print(f"=== Phase 2: Trust Region ({best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_tr = min(rem * 0.08, 150.0)
    if t_tr > 10:
        print(f"  Budget: {t_tr:.0f}s")
        # Try with different numbers of peaks
        for npk in [3, 5, 8, 12]:
            rem_sub = deadline - time.monotonic()
            if rem_sub < 10:
                break
            a_tr, _ = trust_region_polish(best_seq, min(rem_sub * 0.25, t_tr / 4), n_peaks=npk, n_outer=2000)
            local_update(a_tr, f"tr_npk{npk}")

    # === Phase 3: Fine L-BFGS with LogSumExp (400s) ===
    print(f"=== Phase 3: L-BFGS fine ({best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_lbfgs = min(rem * 0.22, 400.0)
    if t_lbfgs > 30:
        T_stages = [5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.15,
                    0.1, 0.07, 0.05, 0.03, 0.02, 0.015, 0.01, 0.008, 0.005,
                    0.004, 0.003, 0.002, 0.0015, 0.001]
        print(f"  T_stages: {len(T_stages)}, budget={t_lbfgs:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_lbfgs, T_stages, maxiter_per_stage=10000,
                            deadline=time.monotonic() + t_lbfgs)
        a_lb = symmetrize(a_lb)
        local_update(a_lb, "lbgfs_fine")

    # === Phase 4: Peak Shaving (100s) ===
    print(f"=== Phase 4: Peak Shave ({best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_peak = min(rem * 0.06, 100.0)
    if t_peak > 10:
        a_pk, _ = peak_shave_refined(best_seq, t_peak, n_steps=200000)
        local_update(a_pk, "peak_shave")

    # === Phase 5: Second LP round with perturbation (200s) ===
    print(f"=== Phase 5: LP Round 2 ({best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        for trial in range(2):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 30:
                break
            strength = 0.01 + 0.008 * trial
            x = np.linspace(0, 2 * np.pi, len(best_seq))
            freq = rng.integers(1, 7)
            noise = strength * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
            a_pert = best_seq + noise
            a_pert = np.maximum(a_pert, 0)
            sp = a_pert.sum()
            if sp > 1e-15:
                a_pert = a_pert / sp
            t_sub = min(rem_sub * 0.4, 100.0)
            if t_sub > 15:
                a_lp2, _ = nesterov_lp_refine(a_pert, t_sub, solver, n_max_iters=500000)
                local_update(a_lp2, f"lp_pert{trial}")
                # Quick trust region on the result
                rem_sub2 = deadline - time.monotonic()
                if rem_sub2 > 10:
                    a_tr2, _ = trust_region_polish(a_pert, min(rem_sub2 * 0.2, 30.0), n_peaks=5, n_outer=1000)
                    local_update(a_tr2, f"tr_pert{trial}")

    # === Phase 6: Try different n (200s) ===
    print(f"=== Phase 6: Different n ({best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        for n_try in [1000, 1500]:
            rem_sub = deadline - time.monotonic()
            if rem_sub < 50:
                break
            print(f"  Trying n={n_try}, budget={rem_sub:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            init_try = symmetrize(init_try)
            a_sp_try = spectral_flatten(init_try, n_iters=20, blend=0.3, deadline=time.monotonic() + 10)
            t_try = min(rem_sub * 0.4, 100.0)
            if t_try > 20:
                a_try, _ = nesterov_lp_refine(a_sp_try, t_try, solver, n_max_iters=300000)
                local_update(a_try, f"lp_n{n_try}")

    # === Phase 7: Final multi-method polish (remaining time - 20s) ===
    print(f"=== Phase 7: Final Polish ({best_score:.10f}) ===")
    rem = deadline - time.monotonic() - 20.0
    if rem > 30:
        # Split remaining time among final polish methods
        t_chunk = rem / 5.0

        # 7a: Ultra-fine L-BFGS
        if t_chunk > 10:
            T_fin = [2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001]
            a_fin = lbgfs_refine(best_seq, t_chunk, T_fin, maxiter_per_stage=5000,
                                 deadline=time.monotonic() + t_chunk)
            local_update(a_fin, "fin_lbgfs")

        # 7b: Trust region with more peaks
        rem = deadline - time.monotonic() - 20.0
        if rem > 15:
            a_trf, _ = trust_region_polish(best_seq, min(rem * 0.2, 40.0), n_peaks=8, n_outer=3000)
            local_update(a_trf, "fin_tr")

        # 7c: Peak shaving
        rem = deadline - time.monotonic() - 20.0
        if rem > 10:
            a_psf, _ = peak_shave_refined(best_seq, min(rem * 0.2, 30.0), n_steps=50000)
            local_update(a_psf, "fin_peak")

        # 7d: One more LP round
        rem = deadline - time.monotonic() - 20.0
        if rem > 10:
            a_lpf, _ = nesterov_lp_refine(best_seq, min(rem * 0.3, 40.0), solver, n_max_iters=100000)
            local_update(a_lpf, "fin_lp")

        # 7e: Final symmetrization + evaluate
        rem = deadline - time.monotonic() - 20.0
        if rem > 5:
            a_final_sym = symmetrize(best_seq)
            local_update(a_final_sym, "fin_sym")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)}) at {elapsed:.1f}s")
        test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            eval_score = evaluate_sequence(test_seq)
            print(f"Verified with evaluate_sequence: {eval_score:.10f}")
        except Exception as e:
            print(f"Verification error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
