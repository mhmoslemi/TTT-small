    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    if T_init is None:
        c = fast_convolve(a)
        T = c.max() * 1.1
    
    # Bisection on T won't work directly (non-convex), so we'll just optimize the soft penalty
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        c_max = c.max()
        
        # Soft penalty: sum of (c[k] - T)^2 for c[k] > T
        exceeding = c[c > T]
        if len(exceeding) > 0:
            penalty = np.sum((exceeding - T) ** 2)
            # Gradient of penalty w.r.t. a
            g = np.zeros(n)
            for k in np.where(c > T)[0]:
                overshoot = c[k] - T
                # d(c[k])/d(a[m]) = 2*a[k-m] if 0 <= k-m < n
                js = max(0, k - n + 1)
                je = min(n - 1, k)

            g = np.zeros(n)

        lr = 0.01 / (np.max(np.abs(g)) + 1e-10)

        a_new = np.maximum(a_new, 0)
        s = a_new.sum()

            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
        
        # Adapt T
        T = c_max * 0.99
    
    return best_a, best_score

    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def lse_obj_and_grad(a, T):
    """Compute LSE-smoothed objective and gradient for probability vector a."""
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    lse_val = c_max + T * np.log(exp_c.sum())
    obj = 2.0 * n * lse_val
    w = exp_c / exp_c.sum()
    # Gradient: d(obj)/d(a[m]) = 4n * sum_k w[k]*a[k-m] (cross-correlation)
    N = 1 << (len(w) + n - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    cross = irfft(W * np.conj(A), N)[:n]
    grad = 4.0 * n * cross
    return obj, grad


def lse_z_obj_and_grad(z, n, T):
    """LSE objective and gradient in log-parameterization."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    obj, grad_a = lse_obj_and_grad(a, T)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def spectral_flatten(a, time_limit, alpha_init=0.3, n_iters=50000):
    """Spectral flattening via multiplicative magnitude update."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    t0 = time.monotonic()
    alpha = alpha_init
    no_improve = 0

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if it % 500 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            if elapsed > 0:
                remaining = time_limit - elapsed
                if remaining < 5:
                    break

        A = rfft(a, fft_size)
        mag = np.abs(A)
        P = mag ** 2
        P_mean = P.mean()

        # Multiplicative update: |A_new| = |A| * (1 - alpha*(P-P_mean)/P_mean)
        factor = 1.0 - alpha * (P - P_mean) / (P_mean + 1e-30)
        factor = np.maximum(factor, 0.01)  # Don't kill any frequency completely

        mag_new = mag * factor
        # Preserve phase
        phase = np.ones_like(A)
        nonzero = mag > 1e-15
        phase[nonzero] = A[nonzero] / mag[nonzero]
        A_new = mag_new * phase
        A_new[0] = np.abs(A_new[0])

        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            alpha *= 0.5
            continue
        a_new = a_new / s

        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
            if sc < best_score + 1e-10:
                alpha = min(alpha_init, alpha * 1.05)
        else:
            no_improve += 1
            if no_improve > 20:
                alpha *= 0.7
                no_improve = 0
                if alpha < 1e-4:
                    break

    return best_a, best_score


def lbgfs_optimize(a0, time_limit, T_stages, maxiter_per_stage=3000):
    """L-BFGS with LSE smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_in):
            obj, _ = lse_z_obj_and_grad(z_in, n, stage_T)
            return obj

        def jac(z_in):
            _, g = lse_z_obj_and_grad(z_in, n, stage_T)
            return g

        iters = min(maxiter_per_stage, max(5, int(remaining * 20)))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-16,
                         'gtol': 1e-14, 'maxfun': iters * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def soft_peak_descent(a0, time_limit, T_init=None, n_iters=50000):
    """Subgradient descent on soft peak penalty."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    if T_init is None:
        c = fast_convolve(a)
        T = c.max() * 0.95

    lr = 0.01
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if it % 1000 == 0 and it > 0:
            if time.monotonic() - t0 > time_limit - 10:
                break

        c = fast_convolve(a)
        c_max = c.max()

        # Gradient of sum((c[k]-T)^2) for c[k] > T
        exceed = np.maximum(c - T, 0)
        if exceed.sum() < 1e-15:
            # All below T, try to reduce T further
            T = c_max * 0.98
            exceed = np.maximum(c - T, 0)

        N = 1 << (len(exceed) + n - 1).bit_length()
        E = rfft(exceed, N)
        A = rfft(a, N)
        cross = irfft(E * np.conj(A), N)[:n]
        grad = 8.0 * cross  # d/d(a_m) sum_k (c[k]-T)^+ * ... 

        # Also add a small regularization to prevent collapse
        grad += 1e-6 * a

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Adaptive learning rate with backtracking
        for try_lr in [lr, lr * 0.5, lr * 0.1]:
            a_new = a - try_lr * grad / (grad_norm + 1e-10)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                break
            else:
                continue

        # Adapt T: track the current max to keep penalty active
        T = max(T * 0.999, c_max * 0.9)

        # Adapt learning rate
        if it % 500 == 0:
            lr *= 0.9

    return best_a, best_score


def fourier_basis_optimize(a0, time_limit, n_bands=20):
    """Optimize in a Fourier cosine basis (low-dimensional)."""
    n = len(a0)
    best_a = normalize(a0.copy())
    best_score = evaluate_local(best_a)
    t0 = time.monotonic()

    x = np.arange(n)
    # Build basis: [1, cos(2*pi*1*j/n), cos(2*pi*2*j/n), ..., cos(2*pi*K*j/n)]
    K = n_bands
    basis = np.ones((n, K + 1))
    for k in range(1, K + 1):
        basis[:, k] = np.cos(2 * np.pi * k * x / n)

    # Initial coefficients from current a
    # a ≈ basis @ coeffs
    coeffs0, _, _, _ = np.linalg.lstsq(basis, best_a, rcond=None)

    def fourier_score(coeffs):
        a = basis @ coeffs
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            return 1e10
        a = a / s
        return evaluate_local(a)

    def fourier_score_grad(coeffs):
        a = basis @ coeffs
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            return 1e10, np.zeros(K + 1)
        a = a / s
        sc = evaluate_local(a)
        # Numerical gradient
        eps = 1e-6
        grad = np.zeros(K + 1)
        for i in range(K + 1):
            c_plus = coeffs.copy()
            c_plus[i] += eps
            c_minus = coeffs.copy()
            c_minus[i] -= eps
            grad[i] = (fourier_score(c_plus) - fourier_score(c_minus)) / (2 * eps)
        return sc, grad

    # Use L-BFGS on the low-dimensional problem
    x0 = coeffs0.copy()
    best_coeffs = x0.copy()

    for restart in range(3):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        # Small perturbation for restart
        if restart > 0:
            x0 = best_coeffs + np.random.default_rng(restart).normal(0, 0.1, K + 1)

        res = minimize(
            fun=fourier_score, x0=x0, method='Nelder-Mead',
            options={'maxiter': 2000, 'fatol': 1e-10, 'xatol': 1e-8}
        )
        a_new = basis @ res.x
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new / s
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                best_coeffs = res.x.copy()
                print(f"    Fourier basis: score={sc:.8f}")

    return best_a, best_score


def population_search(a_init, time_limit, pop_size=12, rng=None):
    """Population-based search with spectral and time-domain mutations."""
    if rng is None:
        rng = np.random.default_rng(789)
    n = len(a_init)
    fft_size = 1 << (2 * n - 1).bit_length()

    population = [normalize(a_init.copy())]
    # Diverse initial mutations
    methods = ['spectral', 'gaussian', 'low_freq', 'block', 'edge']
    for i in range(pop_size - 1):
        population.append(_mutate(a_init, methods[i % len(methods)], rng, strength=0.03 + 0.01 * (i % 3)))

    scores = [evaluate_local(p) for p in population]
    best_idx = np.argmin(scores)
    best_a = population[best_idx].copy()
    best_score = scores[best_idx]
    t0 = time.monotonic()
    gen = 0
    adaptive_strength = 0.03

    while time.monotonic() - t0 < time_limit:
        gen += 1
        if time.monotonic() - t0 > time_limit - 10:
            break

        order = np.argsort(scores)
        population = [population[i] for i in order]
        scores = [scores[i] for i in order]

        if scores[0] < best_score:
            best_score = scores[0]
            best_a = population[0].copy()

        keep = max(2, pop_size // 2)
        elites = population[:keep]
        new_pop = list(elites)

        for i in range(pop_size - len(new_pop)):
            parent = elites[rng.integers(0, keep)]
            method = methods[rng.integers(0, len(methods))]
            strength = adaptive_strength * (1 + 0.3 * np.sin(gen * 0.05))
            child = _mutate(parent, method, rng, strength)
            new_pop.append(child)

        if pop_size > len(new_pop) and len(elites) >= 2:
            p1, p2 = elites[0], elites[1]
            t = rng.uniform(0.3, 0.7)
            child = normalize(t * p1 + (1 - t) * p2)
            new_pop.append(child)

        population = new_pop[:pop_size]
        scores = [evaluate_local(p) for p in population]

        if gen % 30 == 0:
            if scores[0] >= best_score - 1e-10:
                adaptive_strength *= 0.85
            else:
                adaptive_strength = min(0.08, adaptive_strength * 1.1)

        if adaptive_strength < 0.001:
            break

    return best_a, best_score


def _mutate(a, method, rng, strength=0.02):
    """Apply a mutation to sequence a."""
    n = len(a)
    a = normalize(a.copy())
    if method == 'gaussian':
        noise = rng.normal(0, strength, n)
        a_new = a + noise
    elif method == 'low_freq':
        freq = rng.integers(1, 10)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 2 * np.pi, n)
        noise = strength * np.cos(freq * x + phase)
        a_new = a + noise
    elif method == 'block':
        a_new = a.copy()
        bs = max(5, rng.integers(10, max(11, n // 20)))
        i = rng.integers(0, n - bs)
        j = rng.integers(0, n - bs)
        if abs(i - j) > bs:
            a_new[i:i + bs], a_new[j:j + bs] = a[j:j + bs].copy(), a[i:i + bs].copy()
        else:
            a_new = a + rng.normal(0, strength, n)
    elif method == 'edge':
        a_new = a.copy()
        edge = rng.integers(0, 2)
        w = rng.uniform(0.01, 0.05)
        width = max(5, n // 20)
        if edge == 0:
            a_new[:width] += w
            a_new[width:] -= w * width / max(1, n - width)
        else:
            a_new[-width:] += w
            a_new[:-width] -= w * width / max(1, n - width)
    elif method == 'spectral':
        fft_size = 1 << (2 * n - 1).bit_length()
        A = rfft(a, fft_size)
        mag = np.abs(A)
        # Random spectral perturbation
        mag_pert = mag * (1 + rng.normal(0, strength, len(mag)))
        mag_pert = np.maximum(mag_pert, 0)
        phase = np.ones_like(A)
        nz = mag > 1e-15
        phase[nz] = A[nz] / mag[nz]
        A_new = mag_pert * phase
        A_new[0] = np.abs(A_new[0])
        a_new = irfft(A_new, fft_size)[:n]
    else:
        a_new = a + rng.normal(0, strength, n)
    return normalize(a_new)


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    return normalize(f_interp)


def alternating_projection(a0, time_limit, n_iters=10000):
    """
    Alternate between:
    1. Flatten the power spectrum (frequency domain step)
    2. Project to non-negative and normalize (time domain step)
    With an additional step: blend with the uniform target to prevent collapse.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    t0 = time.monotonic()
    alpha = 0.2  # Step size for spectral update
    blend = 0.01  # Small blend toward uniform to prevent collapse

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if it % 2000 == 0 and it > 0:
            if time.monotonic() - t0 > time_limit - 5:
                break

        # Step 1: Spectral update
        A = rfft(a, fft_size)
        mag = np.abs(A)
        P = mag ** 2
        P_mean = P.mean()

        # Target: make P flat. Use a smoother update:
        # |A_new|^2 = |A|^2 * (P_mean / P)^gamma  (power-law flattening)
        # This is equivalent to |A_new| = |A| * (P_mean/P)^(gamma/2)
        gamma = alpha
        ratio = P_mean / (P + 1e-15)
        # Clip ratio to avoid extreme values
        ratio = np.clip(ratio, 0.1, 10.0)
        mag_new = mag * ratio ** (gamma / 2.0)

        phase = np.ones_like(A)
        nz = mag > 1e-15
        phase[nz] = A[nz] / mag[nz]
        A_new = mag_new * phase
        A_new[0] = np.abs(A_new[0])

        a_new = irfft(A_new, fft_size)[:n]

        # Step 2: Time domain projection
        a_new = np.maximum(a_new, 0)
        # Blend toward uniform to maintain "spreading"
        a_new = (1 - blend) * a_new + blend * np.ones(n) / n
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            alpha *= 0.95
            if alpha < 0.01:
                alpha = 0.01  # Don't go too small

        # Periodically increase blend to prevent collapse
        if it % 500 == 0:
            blend = max(0.001, blend * 0.8)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses spectral flattening, alternating projection, L-BFGS with LSE,
    soft-peak descent, Fourier basis optimization, and population search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        # Create diverse initializations
        x = np.linspace(0, 1, n_init)
        candidates = []
        candidates.append(normalize(np.exp(-8 * (x - 0.5) ** 2)))
        candidates.append(normalize(0.5 * (1 + np.cos(2 * np.pi * x))))
        candidates.append(normalize(np.sin(np.pi * x) ** 2))
        candidates.append(normalize(np.ones(n_init) / n_init))
        # Random smooth
        coeffs = rng.normal(0, 1, 15)
        a_rand = np.zeros(n_init)
        for i, c in enumerate(coeffs):
            a_rand += c * np.cos(2 * np.pi * (i + 1) * x / n_init)
        candidates.append(normalize(np.maximum(a_rand + 0.5, 0)))
        scores = [evaluate_local(c) for c in candidates]
        best_idx = np.argmin(scores)
        best_seq = candidates[best_idx].copy()
        best_score = scores[best_idx]
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.10f}")
    t_start = time.monotonic()

    # === Phase 1: Spectral Flattening (fast, high-impact) ===
    print(f"\n=== Phase 1: Spectral Flattening (t={time.monotonic()-t_start:.0f}s) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_out, _ = spectral_flatten(best_seq, t_alloc, alpha_init=0.3)
        local_update(a_out, "spectral")

    # === Phase 2: Alternating Projection (novel) ===
    print(f"\n=== Phase 2: Alternating Projection (t={time.monotonic()-t_start:.0f}s) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_out, _ = alternating_projection(best_seq, t_alloc)
        local_update(a_out, "alt_proj")

    # === Phase 3: L-BFGS with LSE ===
    print(f"\n=== Phase 3: L-BFGS LSE (t={time.monotonic()-t_start:.0f}s) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 350.0)
    if t_alloc > 30:
        T_stages = [30, 15, 8, 5, 3, 2, 1.5, 1, 0.7, 0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005]
        print(f"  T_stages={len(T_stages)}, budget={t_alloc:.0f}s")
        a_out = lbgfs_optimize(best_seq, t_alloc, T_stages, maxiter_per_stage=2000)
        local_update(a_out, "lbgfs_1")

    # === Phase 4: Soft-peak Descent ===
    print(f"\n=== Phase 4: Soft-Peak Descent (t={time.monotonic()-t_start:.0f}s) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_out, _ = soft_peak_descent(best_seq, t_alloc)
        local_update(a_out, "soft_peak")

    # === Phase 5: Population Search ===
    print(f"\n=== Phase 5: Population Search (t={time.monotonic()-t_start:.0f}s) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_out, _ = population_search(best_seq, t_alloc, pop_size=12, rng=rng)
        local_update(a_out, "population")

    # === Phase 6: Fourier Basis Optimization ===
    print(f"\n=== Phase 6: Fourier Basis (t={time.monotonic()-t_start:.0f}s) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 80.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_out, _ = fourier_basis_optimize(best_seq, t_alloc, n_bands=15)
        local_update(a_out, "fourier")

    # === Phase 7: L-BFGS Polish (fine temperatures) ===
    print(f"\n=== Phase 7: Fine L-BFGS Polish (t={time.monotonic()-t_start:.0f}s) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 15:
        T_fine = [1, 0.5, 0.3, 0.2, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001]
        print(f"  Budget={t_alloc:.0f}s")
        a_out = lbgfs_optimize(best_seq, t_alloc, T_fine, maxiter_per_stage=1500)
        local_update(a_out, "lbgfs_fine")

    # === Phase 8: Second Round of Spectral + L-BFGS ===
    print(f"\n=== Phase 8: Second Round (t={time.monotonic()-t_start:.0f}s) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        # Spectral again from the new best
        t_sub = min(rem * 0.3, 80.0)
        a_out, _ = spectral_flatten(best_seq, t_sub, alpha_init=0.15)
        local_update(a_out, "spectral_2")

        rem = deadline - time.monotonic()
        if rem > 30:
            t_sub = min(rem * 0.4, 100.0)
            T_fine2 = [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002]
            a_out = lbgfs_optimize(best_seq, t_sub, T_fine2, maxiter_per_stage=1000)
            local_update(a_out, "lbgfs_2")

    # === Phase 9: Try different lengths ===
    print(f"\n=== Phase 9: Length Exploration (t={time.monotonic()-t_start:.0f}s) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        for n_try in [n_best + 50, n_best - 50, n_best + 100, n_best + 200]:
            if n_try < 200 or n_try > 3000:
                continue
            rem_sub = deadline - time.monotonic()
            if rem_sub < 25:
                break
            init = interpolate_sequence(best_seq, n_try)
            # Quick spectral + L-BFGS
            t_sub = min(rem_sub * 0.3, 50.0)
            a_try, _ = spectral_flatten(init, min(t_sub * 0.4, 20.0), alpha_init=0.2)
            local_update(a_try, f"spec_n{n_try}")
            rem_sub = deadline - time.monotonic()
            if rem_sub > 15:
                T_short = [5, 2, 1, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
                a_try = lbgfs_optimize(a_try, min(rem_sub * 0.6, 40.0), T_short, maxiter_per_stage=800)
                local_update(a_try, f"lbgfs_n{n_try}")

    # === Phase 10: Final intensive polish ===
    print(f"\n=== Phase 10: Final Polish (t={time.monotonic()-t_start:.0f}s) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        # Very fine L-BFGS
        T_ultra = [0.1, 0.05, 0.03, 0.02, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001, 0.0005]
        t_sub = min(rem * 0.5, 100.0)
        a_out = lbgfs_optimize(best_seq, t_sub, T_ultra, maxiter_per_stage=2000)
        local_update(a_out, "final_lbgfs")

        rem = deadline - time.monotonic()
        if rem > 15:
            # One more spectral pass
            a_out, _ = spectral_flatten(best_seq, min(rem * 0.5, 30.0), alpha_init=0.05)
            local_update(a_out, "final_spec")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.10f}, n={len(best_seq)}, time={time.monotonic()-t_start:.0f}s")

    if best_seq is None:
        # Fallback: uniform
        best_seq = np.ones(n_best) / n_best

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
