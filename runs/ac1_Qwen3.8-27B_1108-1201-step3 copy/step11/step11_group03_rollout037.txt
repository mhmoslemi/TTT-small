    b_max = b.max()
    exp_b = np.exp((b - b_max) / T)
    w = exp_b / exp_b.sum()
    lse_val = b_max + T * np.log(exp_b.sum())
    
    # Gradient: grad_i = 4n * sum_k w_k * a_{k-i}
    # = 4n * (conv(a_reversed, w))[n-1 : n-1+n]
    a_rev = a[::-1]
    conv_aw = fast_cross_convolve(a_rev, w)
    c = conv_aw[n - 1: n - 1 + n]
    grad = 4.0 * n * c
    return grad, b, w, lse_val


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    cond = u * rho_candidates > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

            if f_z <= rhs + 1e-12:
                break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = a_new + ((t_mom - 1.0) / t_new) * (a_new - a)
        a = a_new

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), size)
    return irfft(A * A, size)[:2 * n - 1]


def fast_cross_convolve(a, b):
    """Linear cross-convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = 1 << (n_a + n_b - 2).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), size)
    B = rfft(np.asarray(b, dtype=np.float64), size)
    return irfft(A * B, size)[:n_a + n_b - 1]


def compute_lse_grad(a, T):
    """
    Compute gradient of 2n * LSE_T(conv(a,a)) w.r.t. a.

    conv_aw = fast_cross_convolve(a_rev, w)
    c = conv_aw[n - 1: n - 1 + n]

def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS, cp.GUROBI, cp.MOSEK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS, cp.GUROBI, cp.MOSEK]
    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    return None, None
                b_val = b_val / s_sum
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(123)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)
        lp_elapsed = time.monotonic() - lp_t0

        if b_new is None:
            continue

        alpha_step = 2.0 / (it + 2)
        y = (1.0 - alpha_step) * y + alpha_step * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha_step * (it + 2) * b_new) / (it + 1 + alpha_step * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-12:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 5:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 4)
                noise = 0.02 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                stall_count = 0

            if (it + 1) % 1000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f} lp_time={lp_elapsed*1000:.1f}ms")

        if lp_elapsed > 3.0:
            print(f"    LP too slow ({lp_elapsed:.2f}s), stopping at iter {it+1}")
            break

    return best_a, best_score


def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def solve_active_qcqp(a, solver, rel_tol=1e-5, max_active=15, max_rounds=10):
    """
    Solve active-set QCQP with outer loop for constraint identification.
    min t s.t. a^T Q_k a <= t for k in active, sum(a)=1, a>=0
    """
    n = len(a)
    a_cur = a.copy()
    best_a = a_cur.copy()
    best_score = evaluate_local(a_cur)

    for round_idx in range(max_rounds):
        b = fast_convolve(a_cur)
        max_b = b.max()
        threshold = max_b * (1.0 - rel_tol)
        active = np.where(b >= threshold)[0]

        if len(active) > max_active:
            active = active[np.argsort(b[active])[-max_active:]]

        if len(active) == 0:
            active = np.array([np.argmax(b)])

        a_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1]

        for k in active:
            Q = build_q_sparse(int(k), n)
            constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

        prob = cp.Problem(cp.Minimize(t_var), constraints)

        solved = False
        solvers_to_try = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.ECOS, cp.SCS]
        for s in solvers_to_try:
            try:
                prob.solve(solver=s)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    a_val = np.asarray(a_var.value, dtype=np.float64)
                    a_val = np.maximum(a_val, 0)
                    s_sum = a_val.sum()
                    if s_sum < 1e-15:
                        break
                    a_val = a_val / s_sum
                    solved = True
                    break
            except Exception:
                continue

        if not solved:
            break

        score = evaluate_local(a_val)
        if score < best_score:
            best_score = score
            best_a = a_val.copy()
            a_cur = a_val.copy()
        else:
            # Check if solution violates constraints not in active set
            b_new = fast_convolve(a_val)
            if b_new.max() <= max_b * (1.0 + rel_tol):
                # No significant violation, converged
                break
            # Add violated constraints to active set for next round
            a_cur = a_val.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def lse_smooth_lbgfs(a0, time_limit, T_stages, maxiter_per_stage=2000):
    """L-BFGS with LogSumExp smoothing and temperature annealing (final polish)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / max(s, 1e-15)

    # Work in log-domain: a = exp(z) / sum(exp(z))
    z = np.log(np.maximum(a, 1e-300)).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    best_a = a.copy()
    best_score = evaluate_local(a)

    for T in T_stages:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break

        def lse_obj(z_vec):
            z_s = z_vec - z_vec.max()
            a_vec = np.exp(z_s)
            a_vec = a_vec / a_vec.sum()
            b = fast_convolve(a_vec)
            b_max = b.max()
            exp_b = np.exp((b - b_max) / T)
            lse = b_max + T * np.log(exp_b.sum())
            return 2.0 * n * lse

        def lse_grad(z_vec):
            z_s = z_vec - z_vec.max()
            a_vec = np.exp(z_s)
            a_vec = a_vec / a_vec.sum()
            grad_a, _, _, _ = compute_lse_grad(a_vec, T)
            # Chain rule: d/dz = a * (grad_a - sum(grad_a * a))
            S_val = np.dot(grad_a, a_vec)
            return a_vec * (grad_a - S_val)

        try:
            res = minimize(
                fun=lse_obj, x0=z, jac=lse_grad,
                method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-18, 'maxfun': maxiter_per_stage * 50, 'disp': False}
            )
            z = res.x
            z_s = z - z.max()
            a_new = np.exp(z_s)
            a_new = a_new / a_new.sum()
            score = evaluate_local(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
            a = a_new
        except Exception:
            pass

    return best_a, best_score
