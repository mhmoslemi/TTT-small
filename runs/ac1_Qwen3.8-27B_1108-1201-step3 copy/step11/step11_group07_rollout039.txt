```python
import cvxpy as cp

def lp_step(a0, n_constraints=None):
    n = len(a0)
    a = cp.Variable(n)
    t = cp.Variable()
    
    # conv(a0, a) can be precomputed as a matrix-vector product
    # conv(a0, a)[k] = sum_j a0[j] * a[k-j] (with appropriate bounds)
    
    # Build the convolution matrix: C[k, m] = a0[k-m] if 0 <= k-m < n, else 0
    C = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for m in range(n):
            j = k - m
            if 0 <= j < n:
                C[k, m] = a0[j]
    
    # Linearization: conv(a,a) ≈ conv(a0,a0) + 2*conv(a0, a-a0)
    # = conv(a0,a0) + 2*(C@a - conv(a0,a0))
    # = 2*C@a - conv(a0,a0)
    
    c0 = np.convolve(a0, a0)  # linear convolution
    
    constraints = [
        sum(a) == 1,
        a >= 0
    ]
    
    # For efficiency, only use constraints near the max
    max_c0 = c0.max()
    if n_constraints is not None:
        # Only include constraints where c0[k] is within some fraction of max
        active = c0 >= max_c0 * 0.99
        for k in range(2*n-1):
            if active[k]:
                constraints.append(2*(C[k,:] @ a) - c0[k] <= t)
    else:
        for k in range(2*n-1):
            constraints.append(2*(C[k,:] @ a) - c0[k] <= t)
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS)  # or CLARABEL, SCS
    
    return a.value, t.value
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-guided descent,
subgradient methods, and LSE-LBFGS polishing.
Key innovation: successive linear programming relaxation for the
non-convex max-convolution minimization.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def lse_obj_grad_z(z, n, T):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    c = fast_convolve(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    lse = c_max + T * np.log(exp_c.sum())
    obj = 2.0 * n * lse
    
    w = exp_c / exp_c.sum()
    
    N = 1 << (len(w) + n - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    R = irfft(W * np.conj(A), N)
    b = R[:n]
    
    g = 2.0 * b
    idx = 2 * np.arange(n)
    valid = idx < (2 * n - 1)
    g[valid] -= a[valid] * w[idx[valid]]
    
    S = np.dot(g, a)
    grad_z = a * (g - S)
    
    return obj, grad_z

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def lp_guided_step(a_current, active_frac=1.0):
    """Solve LP relaxation to get a descent direction.
    Returns (a_lp, t_lp) or (None, None) on failure.
    """
    n = len(a_current)
    c0 = fast_convolve(a_current)  # linear auto-conv, length 2n-1
    
    # Build constraint matrix: 2*C@a - c0 <= t
    # C[k,m] = a_current[k-m] if 0 <= k-m < n, else 0
    # Instead of building the full matrix, use the fact that C @ a = conv(a_current, a)
    
    # For the LP, the constraint is: 2*conv(a_current, a)[k] - c0[k] <= t
    # conv(a_current, a)[k] = sum_m a_current[k-m] * a[m] (with bounds)
    # This is a matrix-vector product where the matrix is a Hankel matrix
    
    # For efficiency, let's use the sparse structure or just build the matrix
    # For n=1200, the matrix is 2399 x 1200, which is about 2.9M entries
    # That's fine for modern LP solvers
    
    # Actually, let me use a different approach: express conv(a0, a) as C @ a
    # where C is the convolution matrix
    
    # C[k, m] = a0[k-m] if 0 <= k-m <= n-1, else 0
    # This is a lower-Hankel matrix
    
    # For the LP:
    # minimize t
    # subject to: 2*(C@a)[k] - t <= c0[k] for k=0,...,2n-2
    #             sum(a) = 1
    #             a >= 0
    #             t >= 0 (or free)
    
    # In scipy linprog form:
    # c = [0,...,0, 1] (minimize t)
    # A_ub @ x <= b_ub where x = [a, t]
    # A_eq @ x = b_eq
    # bounds on x
    
    # A_ub[k, 0:n] = 2*C[k,:], A_ub[k, n] = -1
    # b_ub[k] = c0[k]
    
    # A_eq = [1,...,1, 0], b_eq = 1
    # bounds: a >= 0, t free
    
    # Build C (the convolution matrix)
    # C[k, m] = a0[k-m] if 0 <= k-m < n
    C = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        m_start = max(0, k - n + 1)
        m_end = min(n-1, k)
        if m_start <= m_end:
            C[k, m_start:m_end+1] = a0[k-m_end:k-m_start+1][::-1]
    
    # Actually, let me double-check: 
    # conv(a0, a)[k] = sum_j a0[j] * a[k-j] where 0<=j<n, 0<=k-j<n
    # = sum_j a0[j] * a[k-j] for j in [max(0,k-n+1), min(n-1,k)]
    # So C[k, m] (where m is the index of a) = a0[k-m] if 0 <= k-m < n
    # For a fixed k: m ranges from max(0, k-n+1) to min(n-1, k)
    # C[k, m] = a0[k-m]
    
    # Let me rebuild more carefully
    C = np.zeros((2*n-1, n))
    for m in range(n):
        # C[k, m] = a0[k-m] if 0 <= k-m < n, i.e., m <= k <= m+n-1
        k_start = m
        k_end = min(2*n-2, m+n-1)
        C[k_start:k_end+1, m] = a0[0:n]
    
    # Verify: C[k,m] should be a0[k-m]
    # C[k,m] = a0[k-m] for k in [m, m+n-1]
    # So C[k,m] = a0[0], a0[1], ..., a0[n-1] for k = m, m+1, ..., m+n-1
    # That looks right.
    
    # LP formulation
    c_vec = np.concatenate([np.zeros(n), [1.0]])  # minimize t
    
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = 2 * C
    A_ub[:, n] = -1.0
    b_ub = c0.copy()
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    A_eq[0, n] = 0.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(-np.inf, None)]
    
    try:
        result = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        if result.success:
            a_lp = result.x[:n]
            t_lp = result.x[n]
            return a_lp, t_lp
    except Exception as e:
        pass
    
    return None, None

def lp_guided_descent(a0, time_limit, max_iters=200):
    """Iterative LP-guided descent."""
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_lp, t_lp = lp_guided_step(a)
        if a_lp is None:
            break
        
        # Try to take a step toward the LP solution
        a_lp = normalize(a_lp)
        
        improved = False
        for alpha in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]:
            a_try = (1 - alpha) * a + alpha * a_lp
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-15:
                continue
            a_try = a_try / s
            sc = evaluate_local(a_try)
            if sc < best_score:
                best_score = sc
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break
        
        if not improved:
            # Try subgradient fallback
            c = fast_convolve(a)
            c_max = c.max()
            active = np.where(c >= c_max * 0.999)[0]
            g = np.zeros(len(a))
            for k in active:
                for m in range(len(a)):
                    j1 = k - m
                    j2 = m - k
                    if 0 <= j1 < len(a):
                        g[m] += a[j1]
                    if 0 <= j2 < len(a):
                        g[m] += a[j2]
            g /= len(active)
            g -= g.mean()  # center for simplicity
            
            for lr in [0.01, 0.005, 0.001, 0.0005]:
                a_try = a - lr * g
                a_try = np.maximum(a_try, 0)
                s = a_try.sum()
                if s < 1e-15:
                    continue
                a_try = a_try / s
                sc = evaluate_local(a_try)
                if sc < best_score:
                    best_score = sc
                    best_a = a_try.copy()
                    a = a_try.copy()
                    break
            else:
                if it > 20:
                    break
        
        if it % 20 == 0:
            print(f"    LP iter {it}: score={best_score:.8f}")
    
    return best_a, best_score

def subgradient_descent(a0, time_limit, max_iters=50000):
    """Direct subgradient descent with adaptive step size."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    # Polak-Ribiere parameters
    gamma = 0.5
    k = 1
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        c_max = c.max()
        
        # Active set
        active = np.where(c >= c_max - 1e-12)[0]
        
        # Subgradient
        g = np.zeros(n)
        for k_idx in active:
            m_start = max(0, k_idx - n + 1)
            m_end = min(n - 1, k_idx)
            for m in range(m_start, m_end + 1):
                g[m] += a[k_idx - m]
                # Note: for auto-conv, dc[k]/da[m] = a[k-m] + a[m-k]
                # But a[m-k] is the same as a[k-m] when we sum over all valid ranges
                # Actually for the LINEAR conv: c[k] = sum_j a[j]*a[k-j]
                # dc[k]/da[m] = a[k-m] + a[m-k] (if both indices valid)
                # If m >= k: only a[m-k] is valid (a[k-m] would be negative index)
                # If m < k: a[k-m] is valid, and a[m-k] might be valid too
                # Let me just compute it properly
        
        # Recompute more carefully
        g = np.zeros(n)
        for k_idx in active:
            for m in range(n):
                # dc[k_idx]/da[m] = a[k_idx - m] (if 0 <= k_idx-m < n) + a[m - k_idx] (if 0 <= m-k_idx < n)
                jm = k_idx - m
                mj = m - k_idx
                if 0 <= jm < n:
                    g[m] += a[jm]
                if 0 <= mj < n:
                    g[m] += a[mj]
        g /= len(active)
        
        # Step size (Polak-Ribiere)
        step = (c_max - c[active].min() + 1e-10) / (np.dot(g, g) + 1e-30)
        step = min(step, 0.1)
        
        a_try = a - step * g
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        sc = evaluate_local(a_try)
        
        if sc < best_score:
            best_score = sc
            best_a = a_try.copy()
            a = a_try.copy()
            k = k + 1
        else:
            # Backtrack
            for factor in [0.5, 0.25, 0.1, 0.05]:
                a_try = a - step * factor * g
                a_try = np.maximum(a_try, 0)
                s = a_try.sum()
                if s < 1e-15:
                    continue
                a_try = a_try / s
                sc = evaluate_local(a_try)
                if sc < best_score:
                    best_score = sc
                    best_a = a_try.copy()
                    a = a_try.copy()
                    break
            else:
                if it > 1000:
                    break
        
        if it % 5000 == 0:
            print(f"    SubGrad iter {it}: score={best_score:.8f}")
    
    return best_a, best_score

def lse_lbgfs(a0, time_limit, T_stages=None, maxiter_per_stage=2000):
    """L-BFGS with LSE smoothing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [10, 5, 2, 1, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01]
    
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func(z_in):
            obj, _ = lse_obj_grad_z(z_in, n, stage_T)
            return obj
        
        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, n, stage_T)
            return g
        
        iters = min(maxiter_per_stage, max(10, int(remaining * 20)))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-15,
                         'gtol': 1e-14, 'maxfun': iters * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    return normalize(f_interp)

def perturb_sequence(a, rng, method='gaussian', strength=0.02):
    n = len(a)
    if method == 'gaussian':
        noise = rng.normal(0, strength, n)
        a_new = a + noise
    elif method == 'low_freq':
        freq = rng.integers(1, 8)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 2 * np.pi, n)
        noise = strength * np.cos(freq * x + phase)
        a_new = a + noise
    elif method == 'block_swap':
        a_new = a.copy()
        block_size = min(rng.integers(5, 30), n // 4)
        i = rng.integers(0, n - block_size)
        j = rng.integers(0, n - block_size)
        if abs(i - j) > block_size:
            a_new[i:i+block_size], a_new[j:j+block_size] = a[j:j+block_size].copy(), a[i:i+block_size].copy()
    elif method == 'mirror':
        a_new = a.copy()
        k = rng.integers(1, n//2)
        a_new[:k] = a_new[n-k:]
    else:
        a_new = a + rng.normal(0, strength, n)
    return normalize(a_new)
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-guided descent (successive linear programming relaxation)
2. FFT-accelerated subgradient with adaptive step sizes
3. LSE-LBFGS polishing with temperature annealing
4. Multi-start exploration with diverse perturbations
Key innovation: iterative LP relaxation of the non-convex max-convolution
minimization, providing strong descent directions without smoothing.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_subgradient(a, active_indices):
    """Compute subgradient of max(conv(a,a)) at active set using FFT."""
    n = len(a)
    s = np.zeros(2 * n - 1)
    s[active_indices] = 1.0

    fft_size = 1 << (3 * n).bit_length()
    S = rfft(s, fft_size)
    A = rfft(a, fft_size)

    # T2[m] = sum_k s[k]*a[m-k] = (s * a)[m]
    T2 = irfft(S * A, fft_size)[:n]

    # T1[m] = sum_k s[k]*a[k-m] = cross-correlation = IDFT(conj(S)*A)
    T1 = irfft(np.conj(S) * A, fft_size)[:n]

    g = (T1 + T2) / len(active_indices)
    return g


def lse_obj_grad_z(z, n, T):
    """Compute LSE objective and gradient in log-parameterization."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()

    c = fast_convolve(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    lse = c_max + T * np.log(exp_c.sum())
    obj = 2.0 * n * lse

    w = exp_c / exp_c.sum()

    N = 1 << (len(w) + n - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    R = irfft(W * np.conj(A), N)
    b = R[:n]

    g = 2.0 * b
    idx = 2 * np.arange(n)
    valid = idx < (2 * n - 1)
    g[valid] -= a[valid] * w[idx[valid]]

    S_val = np.dot(g, a)
    grad_z = a * (g - S_val)

    return obj, grad_z


def softmax_from_z(z):

    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def lst_guided_step(a_current, use_all_constraints=True):
    """Solve LP relaxation to get a descent direction.
    Returns (a_lp, t_lp) or (None, None) on failure.
    """
    n = len(a_current)
    c0 = fast_convolve(a_current)
    n_constraints = 2 * n - 1

    if not use_all_constraints:
        c0_max = c0.max()
        active_mask = c0 >= c0_max * 0.95
        active_idx = np.where(active_mask)[0]
    else:
        active_idx = np.arange(n_constraints)

    # Build constraint matrix C: C[k,m] = a0[k-m] if 0 <= k-m < n
    # For efficiency, only build rows we need
    C_subset = np.zeros((len(active_idx), n))
    for i, k in enumerate(active_idx):
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        if m_start <= m_end:
            C_subset[i, m_start:m_end + 1] = a_current[k - m_end:k - m_start + 1][::-1]

    # LP: minimize t
    # subject to: 2*(C@a)[k] - t <= c0[k] for k in active_idx
    #             sum(a) = 1
    #             a >= 0
    #             t free

    c_vec = np.concatenate([np.zeros(n), [1.0]])

    A_ub = np.zeros((len(active_idx), n + 1))
    A_ub[:, :n] = 2.0 * C_subset
    A_ub[:, n] = -1.0
    b_ub = c0[active_idx].copy()

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(-np.inf, None)]

    try:
        result = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'time_limit': 10})
        if result.success and result.x is not None:
            a_lp = result.x[:n]
            t_lp = result.x[n]
            if a_lp.sum() > 1e-10:
                return a_lp, t_lp
    except Exception:
        pass

    return None, None


def lp_guided_descent(a0, time_limit, max_iters=500, use_all_constraints=True):
    """Iterative LP-guided descent with line search."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    no_improve_count = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        a_lp, t_lp = lst_guided_step(a, use_all_constraints)
        if a_lp is None:
            break

        a_lp = normalize(a_lp)

        # Check if LP solution is close to current (convergence)
        if np.linalg.norm(a_lp - a) < 1e-10:
            break

        improved = False
        for alpha in [1.0, 0.75, 0.5, 0.3, 0.15, 0.07, 0.03]:
            a_try = (1 - alpha) * a + alpha * a_lp
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-15:
                continue
            a_try = a_try / s
            sc = evaluate_local(a_try)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                no_improve_count = 0
                break

        if not improved:
            no_improve_count += 1
            if no_improve_count >= 15:
                # Try subgradient fallback
                c = fast_convolve(a)
                c_max = c.max()
                active = np.where(c >= c_max - 1e-12)[0]
                g = compute_subgradient(a, active)
                g -= a * g.mean()

                for lr in [0.02, 0.01, 0.005, 0.002, 0.001]:
                    a_try = a - lr * g
                    a_try = np.maximum(a_try, 0)
                    s = a_try.sum()
                    if s < 1e-15:
                        continue
                    a_try = a_try / s
                    sc = evaluate_local(a_try)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_try.copy()
                        a = a_try.copy()
                        no_improve_count = 0
                        break

                if no_improve_count > 30:
                    break

        if it % 50 == 0:
            print(f"    LP iter {it}: score={best_score:.8f}, norm_diff={np.linalg.norm(a_lp-a):.6f}")

    return best_a, best_score


def fft_subgradient_descent(a0, time_limit, max_iters=100000):

    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    # Adaptive step size parameters
    base_lr = 0.05
    decay = 0.999

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        c_max = c.max()

        # Active set: indices where c is close to max
        threshold = c_max * (1.0 - 1e-4)
        active = np.where(c >= threshold)[0]

        g = compute_subgradient(a, active)

        # Center the gradient (to stay on the simplex)
        g = g - a * (g @ a)

        # Step size with decay
        lr = base_lr * (decay ** (it // 100))
        lr = min(lr, 0.1)

        a_try = a - lr * g
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        sc = evaluate_local(a_try)

        if sc < best_score:
            best_score = sc
            best_a = a_try.copy()
            a = a_try.copy()
        else:
            # Backtrack
            for factor in [0.5, 0.25, 0.1]:
                a_try = a - lr * factor * g
                a_try = np.maximum(a_try, 0)
                s = a_try.sum()
                if s < 1e-15:
                    continue
                a_try = a_try / s
                sc = evaluate_local(a_try)
                if sc < best_score:
                    best_score = sc
                    best_a = a_try.copy()
                    a = a_try.copy()
                    break

        if it % 10000 == 0:
            print(f"    SubGrad iter {it}: score={best_score:.8f}")

    return best_a, best_score


def lse_lbgfs(a0, time_limit, T_stages=None, maxiter_per_stage=2000):
    """L-BFGS with LSE smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [10, 5, 2, 1, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01]

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_in):
            obj, _ = lse_obj_grad_z(z_in, n, stage_T)
            return obj

        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, n, stage_T)
            return g

        iters = min(maxiter_per_stage, max(10, int(remaining * 20)))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-15,
                         'gtol': 1e-14, 'maxfun': iters * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    candidates = []
    x = np.linspace(0, 1, n)

    a = np.exp(-8 * (x - 0.5) ** 2)
    candidates.append(normalize(a))

    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    candidates.append(normalize(a))

    a = np.sin(np.pi * x) ** 2
    candidates.append(normalize(a))

    a = (x * (1 - x)) ** 0.5
    candidates.append(normalize(a))

    a = np.exp(-10 * (x - 0.3) ** 2) + np.exp(-10 * (x - 0.7) ** 2)
    candidates.append(normalize(a))

    a = 0.5 + 0.4 * np.cos(2 * np.pi * 2 * x) + 0.1 * np.cos(2 * np.pi * 5 * x)
    a = np.maximum(a, 0)
    candidates.append(normalize(a))

    coeffs = rng.normal(0, 1, 12)
    a = np.zeros(n)
    for i, c in enumerate(coeffs):
        a += c * np.cos(2 * np.pi * (i + 1) * x + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a + 0.5, 0)
    candidates.append(normalize(a))

    candidates.append(np.ones(n) / n)

    a = np.exp(-5 * x) + np.exp(-5 * (1 - x))
    candidates.append(normalize(a))

    a = np.minimum(x, 1 - x)
    candidates.append(normalize(a))

    # Multi-bump
    a = np.zeros(n)
    for center in [0.2, 0.5, 0.8]:
        a += np.exp(-15 * (x - center) ** 2)
    candidates.append(normalize(a))

    return candidates


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    return normalize(f_interp)


def perturb_sequence(a, rng, method='gaussian', strength=0.02):
    """Perturb a sequence using various methods."""
    n = len(a)
    if method == 'gaussian':
        noise = rng.normal(0, strength, n)
        a_new = a + noise
    elif method == 'low_freq':
        freq = rng.integers(1, 8)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 2 * np.pi, n)
        noise = strength * np.cos(freq * x + phase)
        a_new = a + noise
    elif method == 'block_swap':
        a_new = a.copy()
        block_size = min(rng.integers(5, 30), n // 4)
        i = rng.integers(0, max(1, n - block_size))
        j = rng.integers(0, max(1, n - block_size))
        if abs(i - j) > block_size:
            tmp = a_new[i:i + block_size].copy()
            a_new[i:i + block_size] = a_new[j:j + block_size]
            a_new[j:j + block_size] = tmp
    elif method == 'mirror':
        a_new = a.copy()
        k = rng.integers(1, max(2, n // 3))
        a_new[:k] = a_new[n - k:]
    elif method == 'taper':
        a_new = a.copy()
        w = rng.uniform(0.02, 0.08)
        edge = rng.integers(1, max(2, n // 10) + 1)
        a_new[:edge] += w
        a_new[edge:] -= w * edge / (n - edge)
    else:
        a_new = a + rng.normal(0, strength, n)
    return normalize(a_new)


def try_update_best(best_seq, best_score, a, label):
    """Try to update best. Returns (best_seq, best_score, improved)."""
    if a is None:
        return best_seq, best_score, False
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score, False
    a = a / s
    sc = evaluate_local(a)
    if np.isfinite(sc) and sc < best_score:
        print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
        return a.copy(), sc, True
    return best_seq, best_score, False


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LP-guided descent, FFT subgradient, LSE-LBFGS, and multi-start.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_local(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 1200
        candidates = diverse_init(n_init, rng)
        scores = [evaluate_local(c) for c in candidates]
        best_idx = np.argmin(scores)
        best_seq = candidates[best_idx].copy()
        best_score = scores[best_idx]
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.8f}")

    # === Phase 1: LP-guided descent (main innovation) ===
    print(f"\n=== Phase 1: LP-Guided Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = lp_guided_descent(best_seq, t_alloc, max_iters=400, use_all_constraints=True)
        best_seq, best_score, _ = try_update_best(best_seq, best_score, a_lp, "lp_1")

    # === Phase 2: FFT subgradient ===
    print(f"\n=== Phase 2: FFT Subgradient (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_sg, _ = fft_subgradient_descent(best_seq, t_alloc, max_iters=100000)
        best_seq, best_score, _ = try_update_best(best_seq, best_score, a_sg, "subgrad")

    # === Phase 3: LSE-LBFGS ===
    print(f"\n=== Phase 3: LSE-LBFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        T_stages = [5, 2, 1, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01,
                    0.005, 0.003, 0.002, 0.001]
        print(f"  T_stages ({len(T_stages)}), budget={t_alloc:.0f}s")
        a_lb = lse_lbgfs(best_seq, t_alloc, T_stages, maxiter_per_stage=2000)
        best_seq, best_score, _ = try_update_best(best_seq, best_score, a_lb, "lbgfs_1")

    # === Phase 4: LP + Subgradient alternate ===
    print(f"\n=== Phase 4: LP + SubGrad Alternate (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_cur = best_seq.copy()
        for phase in range(6):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 15:
                break
            t_sub = min(t_alloc / 6, rem_sub * 0.4)
            if t_sub < 10:
                break
            if phase % 2 == 0:
                a_cur, _ = lp_guided_descent(a_cur, t_sub, max_iters=50, use_all_constraints=False)
                best_seq, best_score, _ = try_update_best(best_seq, best_score, a_cur, f"lp_alt_{phase}")
            else:
                a_cur, _ = fft_subgradient_descent(a_cur, t_sub, max_iters=20000)
                best_seq, best_score, _ = try_update_best(best_seq, best_score, a_cur, f"sg_alt_{phase}")

    # === Phase 5: Multi-start exploration ===
    print(f"\n=== Phase 5: Multi-Start (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_starts = 5
        methods = ['gaussian', 'low_freq', 'block_swap', 'mirror', 'taper']
        strengths = [0.04, 0.03, 0.05, 0.04, 0.03]
        for i in range(n_starts):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 25:
                break
            method = methods[i % len(methods)]
            strength = strengths[i % len(strengths)]
            a_pert = perturb_sequence(best_seq, rng, method, strength)

            # Quick LP refinement
            t_sub = min(rem_sub * 0.35, 50.0)
            a_pert, _ = lp_guided_descent(a_pert, t_sub, max_iters=40, use_all_constraints=False)
            best_seq, best_score, _ = try_update_best(best_seq, best_score, a_pert, f"multi_lp_{i}")

            # Quick LSE-LBFGS
            rem_sub = deadline - time.monotonic()
            if rem_sub > 15:
                t_sub2 = min(rem_sub * 0.4, 45.0)
                a_pert = lse_lbgfs(a_pert, t_sub2,
                                   T_stages=[1, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01],
                                   maxiter_per_stage=500)
                best_seq, best_score, _ = try_update_best(best_seq, best_score, a_pert, f"multi_lse_{i}")

    # === Phase 6: Length exploration ===
    print(f"\n=== Phase 6: Length Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_options = [n_best + 100, n_best - 100, n_best + 200, n_best + 400, n_best - 200]
        for n_try in n_options:
            if n_try < 100 or n_try > 5000:
                continue
            rem_sub = deadline - time.monotonic()
            if rem_sub < 25:
                break
            init = interpolate_sequence(best_seq, n_try)
            t_sub = min(rem_sub * 0.25, 50.0)
            a_try = lse_lbgfs(init, t_sub, T_stages=[5, 2, 1, 0.5, 0.2, 0.1, 0.05],
                              maxiter_per_stage=500)
            best_seq, best_score, _ = try_update_best(best_seq, best_score, a_try, f"len_{n_try}")

    # === Phase 7: Final refinement ===
    print(f"\n=== Phase 7: Final Refinement (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        a_fin, _ = lp_guided_descent(best_seq, min(rem * 0.4, 100), max_iters=150)
        best_seq, best_score, _ = try_update_best(best_seq, best_score, a_fin, "final_lp")

    rem = deadline - time.monotonic()
    if rem > 15:
        a_fin = lse_lbgfs(best_seq, min(rem * 0.5, 60),
                          T_stages=[0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001],
                          maxiter_per_stage=1000)
        best_seq, best_score, _ = try_update_best(best_seq, best_score, a_fin, "final_lse")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_fin, _ = fft_subgradient_descent(best_seq, min(rem * 0.7, 30), max_iters=50000)
        best_seq, best_score, _ = try_update_best(best_seq, best_score, a_fin, "final_sg")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
```
