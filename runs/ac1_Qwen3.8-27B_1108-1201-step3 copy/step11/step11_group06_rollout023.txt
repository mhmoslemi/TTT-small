def lse_obj_grad(z, n, T_temp):
    """LogSumExp objective and gradient in log-domain."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        a = np.ones(n) / n
    else:
        a = exp_z / s
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T_temp
    scaled_max = scaled.max()
    exp_terms = np.exp(scaled - scaled_max)
    lse = max_b + T_temp * (scaled_max + np.log(exp_terms.sum() + 1e-300))
    obj = 2.0 * n * lse
    w = exp_terms / (exp_terms.sum() + 1e-300)
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        grad_a[js:je+1] += w[k] * a[k - js - np.arange(je - js + 1)]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z

        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def lse_func(zx):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = lse_obj_grad(zx, n, stage_T)
            return obj

        def lse_jac(zx):
            if time.monotonic() > deadline:
                return np.zeros(len(zx))
            _, g = lse_obj_grad(zx, n, stage_T)
            return g

        try:
            res = minimize(
                fun=lse_func, x0=z, jac=lse_jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': maxiter_per_stage * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = np.exp(z - z.max())
    return normalize(a_out)


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 5 * x + 0.5)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = 1.0 / (1.0 + 3.0 * np.abs(x - 0.5))
    elif choice == 8:
        a = 0.5 + 0.2 * np.cos(2 * np.pi * 7 * x) + 0.1 * np.cos(2 * np.pi * 13 * x)
        a = np.maximum(a, 0)
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_palindromic(a):
    """Make a sequence palindromic."""
    n = len(a)
    a_pal = (a + a[::-1]) / 2.0
    a_pal = np.maximum(a_pal, 0)
    s = a_pal.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_pal / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def perturb_sequence(a, rng, strength=0.03):
    """Create a perturbed version with structured frequency components."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq1 = rng.integers(1, 20)
    freq2 = rng.integers(2, 25)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    pert = strength * (np.cos(freq1 * x + p1) + 0.5 * np.sin(freq2 * x + p2))
    a_pert = np.maximum(a + pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def subgradient_polish(a0, time_limit, n_steps=50000, eta0=0.05):
    """Direct subgradient method with multiplicative update."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        # Use all peaks within 0.5% of max
        threshold = max_b * 0.995
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        w_active = b[active]
        w_active = w_active / w_active.sum()
        g = np.zeros(n)
        for wi, k in zip(w_active, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            g[js:je+1] += wi * a[k - js - np.arange(je - js + 1)]
        # Multiplicative update with learned step size
        log_mult = -eta * g / max_b
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.002, 0.3)
        else:
            eta *= 0.9
            if eta < 1e-12:
                break
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses adaptive MW, hybrid LP alternating, fine LSE L-BFGS,
    multi-length exploration, and direct subgradient polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {budget_s:.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Fast MW on parent ===
    print(f"\n=== Phase 1: MW n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.04, 60.0)
    if t1 > 5:
        a_mw, _ = multiplicative_weights(best_seq, t1, n_steps=200000, eta0=0.1)
        try_update(a_mw, "mw_1")

    # === Phase 2: Hybrid LP alternating (main workhorse) ===
    print(f"\n=== Phase 2: LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.30, 500.0)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t2, solver, n_max_iters=500000)
        try_update(a_lp, "lp_1")

    # === Phase 3: Palindromic LP ===
    print(f"\n=== Phase 3: Palindromic LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t3 = min(rem * 0.10, 150.0)
        a_pal = make_palindromic(best_seq)
        a_pal_lp, _ = nesterov_lp_refine(a_pal, t3, solver, n_max_iters=100000)
        try_update(a_pal_lp, "pal_lp")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    lengths_to_try = [600, 800, 1000, 1500, 2000]
    for ln in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        if len(best_seq) == ln:
            continue
        t_ln = min(rem * 0.12, 150.0)
        if t_ln < 20:
            continue
        print(f"  Trying n={ln}, budget={t_ln:.0f}s")
        init_ln = interpolate_sequence(best_seq, ln)
        # Quick MW
        a_ln_mw, _ = multiplicative_weights(init_ln, t_ln * 0.25, n_steps=100000, eta0=0.1)
        try_update(a_ln_mw, f"mw_n{ln}")
        # Then LP
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.4, t_ln * 0.6)
        if t_lp > 20:
            a_ln_lp, _ = nesterov_lp_refine(a_ln_mw, t_lp, solver, n_max_iters=100000)
            try_update(a_ln_lp, f"lp_n{ln}")

    # === Phase 5: LSE L-BFGS on best ===
    print(f"\n=== Phase 5: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t5 = min(rem * 0.20, 300.0)
        T_stages = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1,
                    0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
        print(f"  {len(T_stages)} stages, budget={t5:.0f}s")
        a_lse = lbgfs_refine(best_seq, t5, T_stages, maxiter_per_stage=500)
        try_update(a_lse, "lse_fine")

    # === Phase 6: Perturbation escape ===
    print(f"\n=== Phase 6: Perturbation (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for trial in range(5):
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            strength = 0.01 + 0.008 * trial
            a_pert = perturb_sequence(best_seq, rng, strength=strength)
            t_sub = min(rem * 0.2, 50.0)
            a_opt, _ = nesterov_lp_refine(a_pert, t_sub, solver, n_max_iters=50000)
            try_update(a_opt, f"pert{trial}")

    # === Phase 7: Final LP polish ===
    print(f"\n=== Phase 7: Final LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t7 = min(rem * 0.25, 100.0)
        a_fin_lp, _ = nesterov_lp_refine(best_seq, t7, solver, n_max_iters=200000)
        try_update(a_fin_lp, "final_lp")

    # === Phase 8: Subgradient polish ===
    print(f"\n=== Phase 8: Subgradient (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        t8 = min(rem * 0.3, 30.0)
        a_sub, _ = subgradient_polish(best_seq, t8, n_steps=100000, eta0=0.03)
        try_update(a_sub, "subgrad")

    # === Phase 9: Ultra-fine LSE on final best ===
    print(f"\n=== Phase 9: Ultra-fine LSE (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t9 = min(rem * 0.4, 60.0)
        T_stages_fine = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
        a_ultra = lbgfs_refine(best_seq, t9, T_stages_fine, maxiter_per_stage=300)
        try_update(a_ultra, "ultra_lse")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
