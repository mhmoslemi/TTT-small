            if tol is not None:
                kwargs['tol'] = tol
            prob.solve(solver=s, **kwargs)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue

    if not solved:
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(max(0, M_var.value))


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lbgfs_obj_closure(z, n, T_stage, deadline):
    """Objective for L-BFGS - module level wrapper."""
    if time.monotonic() > deadline[0]:
        return 1e15
    obj, _ = logsumexp_obj_grad(z, n, T_stage)
    return obj


def lbgfs_jac_closure(z, n, T_stage, deadline):
    """Gradient for L-BFGS - module level wrapper."""
    if time.monotonic() > deadline[0]:
        return np.zeros_like(z)
    _, g = logsumexp_obj_grad(z, n, T_stage)
    return g


def spectral_flatten_multi(a, n_iters=50, blends=(0.3, 0.5, 0.7), deadline=None):
    """
    Iterative spectral flattening with multiple blend factors.
    Returns the best result across different blend factors.
    """
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    for blend in blends:
        a_curr = a.copy()
        for i in range(n_iters):
            if deadline is not None and time.monotonic() > deadline:
                break

            a_padded = np.zeros(L, dtype=np.complex128)
            a_padded[:n] = a_curr
            F = np.fft.fft(a_padded)

            mags = np.abs(F)
            non_dc = mags[1:]
            if non_dc.min() < 1e-15:
                target = np.mean(non_dc)
            else:
                target = np.exp(np.mean(np.log(non_dc + 1e-30)))

            phases = np.angle(F)
            F_new = F.copy()
            for k in range(1, L // 2 + 1):
                new_mag = (1 - blend) * mags[k] + blend * target
                F_new[k] = new_mag * np.exp(1j * phases[k])
                if k < L - k:
                    F_new[L - k] = (1 - blend) * mags[L - k] + blend * target
                    F_new[L - k] = (1 - blend) * mags[L - k] + blend * target
                    F_new[L - k] = new_mag * np.exp(1j * phases[L - k])

            a_new = np.fft.ifft(F_new).real[:n]
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_curr = a_new / s_new

        score = 2.0 * n * fast_convolve(a_curr).max()
        if score < best_score:
            best_score = score
            best_a = a_curr.copy()

        if deadline is not None and time.monotonic() > deadline:
            break

    return best_a


def spectral_flatten_sqrt_target(a, n_iters=50, deadline=None):
    """
    Spectral flattening targeting 1/sqrt(freq) magnitude.
    This can sometimes give better results than flat target.
    """
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    freqs = np.arange(L)
    # Target: 1/sqrt(1+freq) for non-DC
    target_mags = np.ones(L)
    target_mags[0] = a.sum()
    for k in range(1, L):
        target_mags[k] = 1.0 / np.sqrt(1.0 + k)
    # Normalize target to match total energy
    target_mags = target_mags * (np.sum(target_mags**2) / (L * (a.sum()**2 / L))) ** 0.5

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        a_padded = np.zeros(L, dtype=np.complex128)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)

        mags = np.abs(F)
        phases = np.angle(F)

        blend = 0.3
        F_new = F.copy()
        for k in range(1, L):
            new_mag = (1 - blend) * mags[k] + blend * target_mags[k]
            F_new[k] = new_mag * np.exp(1j * phases[k])

        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

    return a


def peak_shave_adaptive(a, time_limit, n_steps=200000):
    """
    Adaptive peak-targeting with backtracking line search.
    Uses exponential weighting on top peaks and adapts step size.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return None, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.03
    p_power = 10.0  # Higher = more focus on top peaks
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Weight peaks exponentially
        b_norm = b / max_b
        weights = b_norm ** p_power
        total_w = weights.sum()
        if total_w < 1e-30:
            weights = np.zeros(len(b))
            weights[np.argmax(b)] = 1.0
        else:
            weights = weights / total_w

        # Compute weighted gradient
        g = np.zeros(n)
        for k in range(len(b)):
            if weights[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += weights[k] * a[k - idx]

        # Try different step sizes (backtracking)
        a_new = a.copy()
        eta_try = eta
        improved = False
        for _bt in range(8):
            log_scale = -eta_try * g
            log_scale -= log_scale.max()
            a_candidate = a * np.exp(log_scale)
            a_candidate = np.maximum(a_candidate, 0)
            s_c = a_candidate.sum()
            if s_c < 1e-15:
                eta_try *= 0.5
                continue
            a_candidate = a_candidate / s_c
            score_c = 2.0 * n * fast_convolve(a_candidate).max()
            if score_c < best_score - 1e-15:
                best_score = score_c
                best_a = a_candidate.copy()
                a = a_candidate
                a_new = a_candidate
                improved = True
                no_improve = 0
                break
            else:
                eta_try *= 0.5

        if not improved:
            no_improve += 1
            # Adaptive step size reduction
            if no_improve % 20 == 0:
                eta *= 0.7
                p_power = max(2.0, p_power * 0.8)
            if eta < 0.001:
                break

        if step % 30000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShaveAdaptive {elapsed:.0f}s: score={best_score:.8f} eta={eta:.4f} p={p_power:.1f}")

    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=1000000):
    """
    Nesterov dual averaging LP alternating with adaptive parameters.
    Improved version with better step size schedule and stall handling.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(99999)
    lp_times = []

    # Adaptive Nesterov parameters
    base_alpha = 2.0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)
        lp_time = time.monotonic() - lp_t0
        lp_times.append(lp_time)

        if b_new is None:
            continue

        # Check if LP is too slow
        if len(lp_times) >= 30:
            avg_lp_time = np.mean(lp_times[-30:])
            if avg_lp_time > 8.0 and len(lp_times) > 50:
                print(f"    LP too slow ({avg_lp_time:.2f}s/iter), stopping at iter {it+1}")
                break

        # Nesterov step with adaptive alpha
        alpha = base_alpha / (it + 2.0)
        alpha = min(alpha, 0.5)  # Cap alpha for stability

        # Update y (dual variable)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        # Update a (primal variable) - Nesterov extrapolation
        a_prev = a.copy()
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        # Check improvement
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Also check blend of a and y
        for t_blend in [0.3, 0.5, 0.7]:
            a_blend = (1 - t_blend) * a + t_blend * y
            a_blend = np.maximum(a_blend, 0)
            sb = a_blend.sum()
            if sb > 1e-15:
                a_blend = a_blend / sb
                score_blend = 2.0 * n * fast_convolve(a_blend).max()
                if score_blend < best_score:
                    best_score = score_blend
                    best_a = a_blend.copy()

        # Stall detection and diversification
        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 5:
                # Diversify using structured perturbation
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 10)
                noise = 0.02 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                # Also diversify a
                a = a + 0.01 * noise
                a = np.maximum(a, 0)
                sa = a.sum()
                if sa > 1e-15:
                    a = a / sa
                stall_count = 0

            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage, deadline):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    n_stages = len(T_stages)
    iters_per_stage = max(20, maxiter_per_stage // n_stages)

    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break

        stages_left = n_stages - stage_i
        avail_time = remaining
        iters = max(15, min(iters_per_stage * 2, int(avail_time * 80 / max(1, stages_left))))

        dl_arr = [deadline]

        def func_lse(z_in):
            if time.monotonic() > dl_arr[0]:
                return 1e15
            obj, _ = logsumexp_obj_grad(z_in, n, stage_T)
            return obj

        def jac_lse(z_in):
            if time.monotonic() > dl_arr[0]:
                return np.zeros_like(z_in)
            _, g = logsumexp_obj_grad(z_in, n, stage_T)
            return g

        try:
            res = minimize(
                fun=func_lse, x0=z, jac=jac_lse, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-30,
                         'gtol': 1e-20, 'maxfun': iters * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        # Also record progress
        if stage_i % 5 == 0:
            a_curr = softmax_from_z(z)
            sc = 2.0 * n * fast_convolve(a_curr).max()
            print(f"      L-BFGS stage {stage_i}/{n_stages} T={stage_T}: score={sc:.8f}")

    return softmax_from_z(z)


def multiplicative_weights(a0, time_limit, n_steps=100000):
    """
    Mirror descent with multiplicative weights and adaptive focusing.
    Uses a smooth approximation of the max via power weighting.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta_base = 0.05
    power = 5.0
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Power-weighted subgradient
        b_shifted = b / (max_b + 1e-30)
        weights = b_shifted ** power
        tw = weights.sum()
        if tw < 1e-30:
            weights = np.zeros(len(b))
            weights[np.argmax(b)] = 1.0
        else:
            weights = weights / tw

        g = np.zeros(n)
        for k in range(len(b)):
            if weights[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += weights[k] * a[k - idx]

        # Multiplicative update with adaptive eta
        eta = eta_base * max(0.01, 1.0 / (1.0 + 0.01 * step))
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 30:
            power = max(1.0, power * 0.85)
            eta_base *= 0.85
            no_improve = 0
            if eta_base < 0.005:
                break

    return best_a, best_score


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        center = n // 2
        t = (np.arange(n) - center) / center
        a = np.abs(np.sinc(t))
    elif choice == 7:
        # Bump function
        a = np.exp(-10 * (x - 0.5) ** 4)
    elif choice == 8:
        # Two bumps
        a = np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2)
    else:
        # Flat with slight modulation
        a = 1.0 + 0.1 * np.cos(2 * np.pi * 5 * x)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturbed_sequence(a, rng, strength=0.03):
    """Create a perturbed version of sequence a using random Fourier features."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    # Use multiple random frequencies
    n_freqs = rng.integers(2, 5)
    perturbation = np.zeros(n)
    for _ in range(n_freqs):
        freq = rng.integers(1, 10)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.5, 1.0)
        perturbation += amp * np.cos(freq * x + phase)
    perturbation *= strength / n_freqs
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def coordinate_descent_polish(a0, time_limit, n_rounds=50):
    """
    Coordinate descent: for each coordinate, find the optimal value
    given all others fixed. This is exact for each coordinate.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return None
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            a_old = a[i]
            # Fix all other coordinates, optimize a[i]
            # conv(a,a)[k] is quadratic in a[i]:
            # c[k] = sum_j a[j]*a[k-j], terms involving a[i]:
            #   a[i]*a[k-i] (if k-i is valid and != i) + a[i]^2 (if k-i == i, i.e., k=2i)
            # Simpler: just try a few values of a[i]
            # Compute convolution without a[i]
            a_no_i = a.copy()
            a_no_i[i] = 0
            c_base = fast_convolve(a_no_i)

            # For each k, c[k] with a[i]=v: c_base[k] + v * (sum of a[j] where j+i=k, j!=i)
            # = c_base[k] + v * a_no_i[k-i] (if 0<=k-i<n and k-i != i) + v^2 (if k=2i and i<n)

            # The max of conv is piecewise quadratic in v.
            # Just try a grid of values.
            best_v = a_old
            best_max_c = float('inf')
            # Scale v to try
            v_range = np.linspace(0, 2 * a_old + 0.01 / n, 10) if a_old > 0 else np.linspace(0, 1.0 / n, 10)

            for v in v_range:
                a_test = a_no_i.copy()
                a_test[i] = v
                # Quick check: compute max conv
                # c[k] = c_base[k] + v * contrib[k] + v^2 * (1 if k==2i else 0)
                contrib = np.zeros(2 * n - 1)
                if 0 <= k_idx := (2 * i) < 2 * n - 1:
                    contrib[2 * i] = 1.0  # v^2 term
                # Linear term: v * a_no_i[k-i] for valid k-i
                for k in range(2 * n - 1):
                    j = k - i
                    if 0 <= j < n and j != i:
                        contrib[k] += a_no_i[j]
                    # Also the other way: a[i] * a_no_i[k-i] is already counted
                    # Wait, conv(a,a)[k] = sum_j a[j]*a[k-j]
                    # Terms with a[i]: j=i gives a[i]*a[k-i] = v*a_no_i[k-i]
                    #                    j=k-i gives a[k-i]*a[i] = a_no_i[k-i]*v (same term if k-i != i)
                    #                    if j=i and k-i=i (k=2i): v*v = v^2
                # Actually this is getting complicated. Let me just compute directly for few candidates.
                c_test = fast_convolve(a_test)
                max_c = c_test.max()
                if max_c < best_max_c:
                    best_max_c = max_c
                    best_v = v

            if best_v != a_old:
                a[i] = best_v
                improved = True

        # Renormalize
        s = a.sum()
        if s > 1e-15:
            a = a / s
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if not improved:
            break

    return best_a


def hessian_free_newton(a0, time_limit, n_iters=200):
    """
    Hessian-free optimization using the log-sum-exp smoothed objective.
    Uses conjugate gradient to approximate Newton direction.
    """
    n = len(a0)
    z = z_from_a(a0).copy()
    T_smooth = 0.01  # Temperature for smoothing
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        f0, g0 = logsumexp_obj_grad(z, n, T_smooth)

        # Approximate Newton direction via CG on (H + lambda*I) p = -g
        # Use finite differences for Hv
        lambda_reg = 1e-6

        def hessian_vec_product(v):
            eps = 1e-5
            g_plus, _ = logsumexp_obj_grad(z + eps * v, n, T_smooth)
            g_minus, _ = logsumexp_obj_grad(z - eps * v, n, T_smooth)
            return (g_plus - g_minus) / (2 * eps) + lambda_reg * v

        # CG to solve (H + lambda*I) p = -g
        p = -g0.copy()
        r = (hessian_vec_product(p) + g0)  # r = Hp + g = (H+lambda)p + g
        # Actually we want (H+lambda)p = -g, so residual r = (H+lambda)p + g
        p_old = p.copy()
        rr = np.dot(r, r)
        for cg_iter in range(10):
            if rr < 1e-20:
                break
            q = hessian_vec_product(p_old)
            alpha = rr / max(1e-20, np.dot(p_old, q))
            p = p + alpha * p_old
            r = r + alpha * q
            rr_new = np.dot(r, r)
            if rr_new > rr:
                p = p - alpha * p_old
                r = r - alpha * q
                break
            p_old = (rr_new / max(1e-20, rr)) * p_old
            r = r + p_old
            rr = rr_new

        # Line search
        f_new = f0
        for ls in range(20):
            z_try = z + 0.5 ** ls * p
            f_try, _ = logsumexp_obj_grad(z_try, n, T_smooth)
            if f_try < f_new:
                f_new = f_try
                z = z_try
                break

        a_curr = softmax_from_z(z)
        score = 2.0 * n * fast_convolve(a_curr).max()
        if score < best_score:
            best_score = score
            best_a = a_curr.copy()

        if it % 20 == 0:
            print(f"      HF-Newton iter {it}: score={score:.8f}")

    return best_a


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Key improvements over previous:
    - Multi-target spectral flattening
    - Adaptive peak-targeting with backtracking
    - Hessian-free Newton for final polish
    - Better Nesterov LP with blend checking
    - More aggressive temperature annealing
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    rem = deadline - time.monotonic()
    print(f"Total budget: {rem:.0f}s, starting score: {best_score:.8f}")

    # === Phase 1: Multi-target Spectral Flattening ===
    print(f"\n=== Phase 1: Multi-target Spectral (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_spec = min(rem * 0.04, 60.0)
    if t_spec > 15:
        print(f"  Budget: {t_spec:.0f}s")
        a_spec = spectral_flatten_multi(best_seq, n_iters=100, blends=(0.2, 0.4, 0.6, 0.8),
                                        deadline=time.monotonic() + t_spec * 0.6)
        local_update(a_spec, "spectral_multi")

        rem = deadline - time.monotonic()
        if rem > 20:
            a_spec2 = spectral_flatten_sqrt_target(best_seq, n_iters=50,
                                                   deadline=time.monotonic() + min(rem, 30))
            local_update(a_spec2, "spectral_sqrt")

    # === Phase 2: Nesterov LP (main workhorse) ===
    print(f"=== Phase 2: Nesterov LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lp = min(rem * 0.30, 450.0)
    if t_lp > 30:
        print(f"  Budget: {t_lp:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_lp, solver, n_max_iters=1000000)
        local_update(a_lp, "nesterov_lp_1")

    # === Phase 3: Adaptive Peak Shaving ===
    print(f"=== Phase 3: Adaptive Peak Shaving (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_peak = min(rem * 0.10, 120.0)
    if t_peak > 15:
        a_pk, _ = peak_shave_adaptive(best_seq, t_peak, n_steps=200000)
        local_update(a_pk, "peak_shave_adaptive")

    # === Phase 4: Multiplicative Weights ===
    print(f"=== Phase 4: Multiplicative Weights (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_mw = min(rem * 0.08, 80.0)
    if t_mw > 10:
        a_mw, _ = multiplicative_weights(best_seq, t_mw, n_steps=100000)
        local_update(a_mw, "mult_weights")

    # === Phase 5: Ultra-fine LogSumExp L-BFGS ===
    print(f"=== Phase 5: L-BFGS LogSumExp (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lbfgs = min(rem * 0.25, 400.0)
    if t_lbfgs > 30:
        T_stages = [5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1,
                    0.05, 0.03, 0.02, 0.015, 0.01, 0.007,
                    0.005, 0.003, 0.002, 0.0015, 0.001]
        print(f"  T_stages: {len(T_stages)} stages, budget={t_lbfgs:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_lbfgs, T_stages, maxiter_per_stage=15000,
                            deadline=time.monotonic() + t_lbfgs)
        local_update(a_lb, "lbgfs_fine")

    # === Phase 6: Second Nesterov LP round ===
    print(f"=== Phase 6: Nesterov LP Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lp2 = min(rem * 0.12, 180.0)
    if t_lp2 > 20:
        print(f"  Budget: {t_lp2:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_lp2, solver, n_max_iters=500000)
        local_update(a_lp2, "nesterov_lp_2")

    # === Phase 7: Hessian-Free Newton Polish ===
    print(f"=== Phase 7: HF-Newton Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_hf = min(rem * 0.06, 60.0)
    if t_hf > 15:
        a_hf = hessian_free_newton(best_seq, t_hf, n_iters=100)
        local_update(a_hf, "hf_newton")

    # === Phase 8: Try different n values ===
    print(f"=== Phase 8: Try different n (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    n_candidates = [1000, 1400, 1800, 2400]
    for n_try in n_candidates:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_n = min(rem * 0.12, 100.0)
        if t_n < 30:
            continue
        print(f"  Trying n={n_try}, budget={t_n:.0f}s")
        init_n = interpolate_sequence(best_seq, n_try)
        # Quick spectral
        a_sp = spectral_flatten_multi(init_n, n_iters=30, blends=(0.3, 0.5),
                                      deadline=time.monotonic() + t_n * 0.3)
        # LP
        a_n, _ = nesterov_lp_refine(a_sp, t_n * 0.6, solver, n_max_iters=50000)
        local_update(a_n, f"lp_n{n_try}")

    # === Phase 9: Final polish with very low temperature ===
    print(f"=== Phase 9: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        T_final = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
        t_fin = min(rem * 0.5, 120.0)
        a_fin = lbgfs_refine(best_seq, t_fin, T_final, maxiter_per_stage=10000,
                             deadline=time.monotonic() + t_fin)
        local_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 20:
        # Final peak shave
        t_ps = min(rem * 0.5, 40.0)
        a_ps, _ = peak_shave_adaptive(best_seq, t_ps, n_steps=50000)
        local_update(a_ps, "final_peak")

    rem = deadline - time.monotonic()
    if rem > 15:
        # Final quick LP
        t_flp = min(rem * 0.6, 30.0)
        a_flp, _ = nesterov_lp_refine(best_seq, t_flp, solver, n_max_iters=20000)
        local_update(a_flp, "final_lp")

    # Final verification and output
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}) at {elapsed:.1f}s")
        # Verify with evaluate_sequence
        test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            eval_score = evaluate_sequence(test_seq)
            print(f"Verified with evaluate_sequence: {eval_score:.8f}")
        except Exception as e:
            print(f"Verification error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
