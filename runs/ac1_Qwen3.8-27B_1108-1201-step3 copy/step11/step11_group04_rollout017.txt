import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_stable(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64)
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_lse_obj_grad(z, T, n=None):
    """Compute LSE-smoothed objective and gradient w.r.t. z in softmax param."""
    a = softmax_stable(z)
    if n is None:
        n = len(a)
    c = fast_convolve(a)
    max_c = c.max()
    shifted = (c - max_c) / T
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    exp_sum = exp_shifted.sum()
    if exp_sum < 1e-300:
        lse_val = max_c
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        lse_val = max_c + T * np.log(exp_sum)
        w = exp_shifted / exp_sum
    obj = 2.0 * n * lse_val
    # grad_a[m] = 2*n * sum_k w[k] * dc[k]/da[m] = 2*n * 2 * correlate(w, a, valid)[m]
    corr = np.correlate(w, a, mode='valid')
    grad_a = 4.0 * n * corr
    S = np.dot(a, grad_a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def compute_max_obj_grad(z, n=None):
    """Compute exact max objective and subgradient w.r.t. z."""
    a = softmax_stable(z)
    if n is None:

    obj = 2.0 * n * max_c
    # Subgradient: find active peaks (within numerical tolerance)

    active = np.where(c >= max_c - tol)[0]
    w = np.zeros(len(c))

    grad_a = 4.0 * n * corr
    S = np.dot(a, grad_a)

def lse_lbfgs_phase(z0, T_list, time_budget, deadline, n=None):
    """Run L-BFGS through a sequence of temperatures."""
    if n is None:
        n = len(z0)
    z = z0.copy()
    best_z = z.copy()
    best_obj = compute_lse_obj_grad(z, T_list[0], n)[0]
    t0 = time.monotonic()

    for i, T in enumerate(T_list):
        if time.monotonic() > deadline or (time.monotonic() - t0) > time_budget:
            break

        def func(z_in):
            if time.monotonic() > deadline:
                return 1e15
            return compute_lse_obj_grad(z_in, T, n)[0]

        def jac(z_in):
            if time.monotonic() > deadline:
                return np.zeros_like(z_in)
            return compute_lse_obj_grad(z_in, T, n)[1]

        remaining = max(1.0, deadline - time.monotonic())
        maxiter = max(10, min(2000, int(remaining * 20 / (len(T_list) - i))))

        try:

                fun=func, x0=z, jac=jac, method='L-BFGS-B',

                         'maxls': 50, 'disp': False}

            if obj < best_obj:

                best_z = z.copy()
        except Exception:
            pass

def peak_mirror_descent(z0, time_budget, deadline, n=None, lr_base=0.01, top_k=5):
    """Adaptive peak-targeting mirror descent in softmax space."""
    if n is None:
        n = len(z0)
    z = z0.copy()
    best_z = z.copy()
    a = softmax_stable(z)
    c = fast_convolve(a)
    best_obj = 2.0 * n * c.max()
    t0 = time.monotonic()
    lr = lr_base
    no_improve = 0

    while time.monotonic() < deadline and (time.monotonic() - t0) < time_budget:
        a = softmax_stable(z)
        c = fast_convolve(a)
        max_c = c.max()
        obj = 2.0 * n * max_c

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            lr *= 0.5
            no_improve = 0
            if lr < 1e-6:
                break

        # Find top-k peaks
        sort_idx = np.argsort(c)[::-1]
        top_peaks = sort_idx[:top_k]
        # Weight peaks by their value (normalized)
        peak_vals = c[top_peaks]
        weights = peak_vals / peak_vals.sum()

        # Compute weighted subgradient
        w = np.zeros(len(c))
        for idx, wp in zip(top_peaks, weights):
            w[idx] = wp

        corr = np.correlate(w, a, mode='valid')
        grad_a = 4.0 * n * corr
        S = np.dot(a, grad_a)
        grad_z = a * (grad_a - S)

        # Normalize gradient for stable updates
        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-15:
            break
        grad_z_norm = grad_z / grad_norm

        # Mirror descent step
        z_new = z - lr * grad_z_norm

        # Check improvement
        a_new = softmax_stable(z_new)
        c_new = fast_convolve(a_new)
        obj_new = 2.0 * n * c_new.max()

        if obj_new < obj:
            z = z_new
        else:
            # Try half step
            z_half = z - 0.5 * lr * grad_z_norm
            a_half = softmax_stable(z_half)
            c_half = fast_convolve(a_half)
            obj_half = 2.0 * n * c_half.max()
            if obj_half < obj:
                z = z_half
                lr *= 0.8
            else:
                lr *= 0.5

    return best_z, best_obj


def spectral_init(n, seed=0):
    """Generate a spectral initialization with flat magnitude spectrum."""
    rng = np.random.default_rng(seed)
    # Random phases with constrained magnitude
    L = 1 << (2 * n - 1).bit_length()
    # Start with flat spectrum + random phases
    F = np.ones(L, dtype=np.complex128)
    for k in range(1, L // 2 + 1):
        phase = rng.uniform(0, 2 * np.pi)
        mag = 1.0 + 0.1 * rng.standard_normal()
        mag = max(0.5, mag)
        F[k] = mag * np.cos(phase) + 1j * mag * np.sin(phase)
        if k < L - k:
            F[L - k] = np.conj(F[k])
    a = np.fft.ifft(F).real[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        center = n // 2
        t = (np.arange(n) - center) / center
        a = np.sinc(t)
        a = np.maximum(a, 0)
    elif choice == 7:
        # Flat-top window
        a = 0.54 + 0.46 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0)
    elif choice == 8:
        # Partially flat
        a = np.ones(n)
        a[:n//4] *= 0.7
        a[3*n//4:] *= 0.7
    else:
        a = 1.0 - np.abs(2 * x - 1) ** 0.5

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturb_sequence(a, rng, strength=0.03):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    n_freqs = rng.integers(1, 6)
    perturbation = np.zeros(n)
    for _ in range(n_freqs):
        freq = rng.integers(1, 12)
        phase = rng.uniform(0, 2 * np.pi)
        amp = strength * rng.uniform(0.5, 1.5)
        perturbation += amp * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def diversify_explore(a_best, n, rng, time_budget, deadline):
    """Try multiple perturbations and keep the best."""
    best_a = a_best.copy()
    best_score = evaluate_local(a_best)

    n_trials = max(1, int(time_budget / 30.0))
    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        strength = 0.01 + 0.02 * trial
        a_pert = perturb_sequence(a_best, rng, strength=strength)
        # Quick refinement
        z_pert = np.log(np.maximum(a_pert, 1e-300))
        T_quick = [2.0, 0.5, 0.1, 0.03]
        t_sub = min(25.0, (deadline - time.monotonic()) * 0.8 / max(1, n_trials - trial))
        if t_sub < 5:
            continue
        z_opt, _ = lse_lbfgs_phase(z_pert, T_quick, t_sub, deadline, n=n)
        a_opt = softmax_stable(z_opt)
        sc = evaluate_local(a_opt)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()

    return best_a, best_score


def multi_temperature_lbfgs(a_input, time_budget, deadline, n=None):
    """Full multi-temperature L-BFGS optimization from a given sequence."""
    if n is None:
        n = len(a_input)
    z = np.log(np.maximum(a_input, 1e-300))
    # High temperature stage
    T_high = [5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5]
    t_high = time_budget * 0.3
    z, _ = lse_lbfgs_phase(z, T_high, t_high, deadline, n=n)

    # Medium temperature stage
    T_med = [0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03]
    t_med = time_budget * 0.35
    z, _ = lse_lbfgs_phase(z, T_med, t_med, deadline, n=n)

    # Low temperature stage
    T_low = [0.02, 0.015, 0.01, 0.008, 0.005, 0.003, 0.002]
    t_low = time_budget * 0.25
    z, _ = lse_lbfgs_phase(z, T_low, t_low, deadline, n=n)

    # Peak mirror descent
    t_peak = max(5.0, time_budget * 0.1)
    z, _ = peak_mirror_descent(z, t_peak, deadline, n=n, lr_base=0.005)

    a_out = softmax_stable(z)
    return a_out


def adam_peak_descent(z0, time_budget, deadline, n=None, lr=0.001, beta1=0.9, beta2=0.999):
    """Adam optimizer targeting the max peak in softmax space."""
    if n is None:
        n = len(z0)
    z = z0.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0
    best_z = z.copy()
    a = softmax_stable(z)
    c = fast_convolve(a)
    best_obj = 2.0 * n * c.max()
    t0 = time.monotonic()

    while time.monotonic() < deadline and (time.monotonic() - t0) < time_budget:
        t += 1
        a = softmax_stable(z)
        c = fast_convolve(a)
        max_c = c.max()
        obj = 2.0 * n * max_c

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_z = z.copy()

        # Use LSE with very low T for gradient
        T = 0.01
        _, grad_z = compute_lse_obj_grad(z, T, n)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        if t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AdamPeak iter={t} ({elapsed:.0f}s): obj={best_obj:.8f}")

    return best_z, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LSE-smoothed L-BFGS with fine temperature annealing,
    peak-targeting mirror descent, and diversification.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    t_start = time.monotonic()

    # Load parent sequence
    best_a = None
    best_score = float('inf')
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_a = p / p.sum()
            best_score = evaluate_local(best_a)
            print(f"Parent loaded: n={len(best_a)}, score={best_score:.8f}")
    except Exception:
        pass

    if best_a is None:
        n_init = 1200
        best_a = structured_init(n_init, rng)
        best_score = evaluate_local(best_a)
        print(f"Generated init: n={len(best_a)}, score={best_score:.8f}")

    n_best = len(best_a)

    def update_best(a_new, label):
        nonlocal best_a, best_score
        if a_new is None:
            return
        a_new = np.maximum(np.asarray(a_new, dtype=np.float64), 0)
        s = a_new.sum()
        if s < 0.01:
            return
        a_new = a_new / s
        sc = evaluate_local(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a_new)})")

    rem = deadline - time.monotonic()
    print(f"Total budget: {rem:.0f}s, starting score: {best_score:.8f}")

    # === Phase 1: Coarse L-BFGS warmup (60s) ===
    print(f"\n=== Phase 1: Coarse L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.1, 60.0)
    if t1 > 10:
        print(f"  Budget: {t1:.0f}s")

        T1 = [10.0, 5.0, 3.0, 2.0, 1.5, 1.0]

    # === Phase 2: Fine multi-temperature L-BFGS (300s) ===
    print(f"=== Phase 2: Fine Multi-Temp L-BFGS (score={best_score:.8f}) ===")

        print(f"  Budget: {t2:.0f}s")
        a2 = multi_temperature_lbfgs(best_a, t2, deadline, n=n_best)
        update_best(a2, "multi_temp")

    # === Phase 3: Peak mirror descent (150s) ===
    print(f"=== Phase 3: Peak Mirror Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 150.0)
    if t3 > 15:
        print(f"  Budget: {t3:.0f}s")
        z3 = np.log(np.maximum(best_a, 1e-300))
        z3_opt, _ = peak_mirror_descent(z3, t3, deadline, n=n_best, lr_base=0.003, top_k=3)
        update_best(softmax_stable(z3_opt), "peak_md_1")

    # === Phase 4: Ultra-fine L-BFGS (300s) ===
    print(f"=== Phase 4: Ultra-fine L-BFGS (score={best_score:.8f}) ===")

        # Alternate: peak MD then L-BFGS

    rem = deadline - time.monotonic()

    return [float(max(0.0, min(1000.0, x))) for x in best_a]
