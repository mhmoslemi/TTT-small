def fast_convolve(a):

    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def convolve_with(a, b):

    n = len(a)
    size = 2 * n - 1

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
from scipy import sparse


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    """Linear cross-convolution via FFT."""
    n = max(len(a), len(b))
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_subgradient(w, a):
    """Compute subgradient from weights w (on autocorrelation) and sequence a.
    Returns s[i] = 2 * sum_k w[k] * a[k-i] for valid k-i.
    """
    n = len(a)
    # s[i] = 2 * sum_j w[i+j] * a[j]
    # = 2 * np.correlate(w, a, 'full')[n-1 : 2*n-1]
    # Use FFT for speed
    corr_size = len(w) + n - 1
    fft_size = 1 << (corr_size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a[::-1], fft_size)
    corr = irfft(W * A, fft_size)[:corr_size]
    # corr[k] = sum_j w[k+j] * a_rev[j] = sum_j w[k+j] * a[n-1-j]
    # We want: s[i] = 2*sum_j w[i+j]*a[j]
    # corr[i] = sum_j w[i+j] * a[n-1-j] = sum_j w[i+n-1-j] * a[j] (substituting j->n-1-j)
    # Hmm, let me just use np.correlate which is C-optimized
    corr_full = np.correlate(w, a, 'full')
    return 2.0 * corr_full[n - 1:2 * n - 1]


def solve_lp_best_response(a):
    """Given a (normalized, sum=1, non-negative), find b minimizing max_k conv(a,b)[k].
    Uses scipy linprog with HiGHS solver.
    """
    n = len(a)
    num_constraints = 2 * n - 1

    # Build sparse convolution matrix T where (T @ b)[k] = conv(a, b)[k]
    rows = []
    cols = []
    vals = []
    for j in range(n):
        r0 = j
        r1 = j + n
        rows.extend(range(r0, r1))
        cols.extend([j] * n)
        vals.extend(a)

    T = sparse.csr_matrix((vals, (rows, cols)), shape=(num_constraints, n))

    # Full A_ub: [T, -I] @ [b; M] <= 0
    # Shape: (2n-1) x (n+1)
    # T is (2n-1) x n, -I is (2n-1) x 1
    A_ub = sparse.hstack([T, -sparse.eye(num_constraints, format='csr')], format='csr')
    b_ub = np.zeros(num_constraints)

    # Equality: sum(b) = 1
    A_eq = sparse.csr_matrix(np.hstack([np.ones(n), [0.0]]).reshape(1, -1))
    b_eq = np.array([1.0])

    # Objective: minimize M (last variable)
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # Bounds
    bounds = [(0, None)] * n + [(0, None)]

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': 30})

    if result.success and result.x is not None:
        b_val = np.asarray(result.x[:n], dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        b_val[~np.isfinite(b_val)] = 0
        s = b_val.sum()
        if s > 1e-15:
            b_val /= s
        return b_val, max(0.0, float(result.x[-1]))
    return None, None


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-30))


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def adam_logsumexp_optimize(a0, time_limit, T_stages, lr_init, lr_decay, top_k_frac=0.1,
                            momentum=0.9, print_interval=20000):
    """Adam optimization in log-space with temperature annealing and top-k focusing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit

    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    step_count = 0

    total_stages = len(T_stages)
    steps_per_stage = max(3000, int(time_limit * 80000 / total_stages))

    for stage_idx, T_stage in enumerate(T_stages):
        if time.monotonic() > deadline:
            break

        top_k = max(10, int(top_k_frac * (2 * n - 1)))

        for step in range(steps_per_stage):
            if time.monotonic() > deadline:
                break

            a = softmax_from_z(z)
            b = fast_convolve(a)
            max_b = b.max()
            score = 2.0 * n * max_b
            step_count += 1

            if score < best_score:
                best_score = score
                best_a = a.copy()

            # Compute weights
            if T_stage < 0.005:
                # Active set only
                threshold = max_b * (1.0 - 1e-5)
                active = np.where(b >= threshold)[0]
                if len(active) == 0:
                    active = np.array([np.argmax(b)])
                w = np.zeros(len(b))
                w[active] = 1.0 / len(active)
            elif top_k < len(b):
                idx = np.argpartition(b, -top_k)[-top_k:]
                b_top = b[idx]
                max_top = b_top.max()
                exp_top = np.exp((b_top - max_top) / T_stage)
                w_top = exp_top / exp_top.sum()
                w = np.zeros(len(b))
                w[idx] = w_top
            else:
                max_b_all = b.max()
                exp_all = np.exp((b - max_b_all) / T_stage)
                w = exp_all / exp_all.sum()

            # Gradient in a-space
            g_a = compute_subgradient(w, a)
            S = np.dot(g_a, a)
            grad_z = a * (g_a - S)

            # Adam update
            lr = lr_init * (lr_decay ** stage_idx)
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** (step + 1))
            v_hat = v / (1 - beta2 ** (step + 1))
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            # Centering
            z -= z.mean()

            if (step + 1) % print_interval == 0:
                elapsed = time.monotonic() - t0
                print(f"    Adam stage {stage_idx} (T={T_stage:.3f}) step {step+1} "
                      f"({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lp_alternate_nesterov(a0, time_limit, max_iters=500000, print_interval=100):
    """LP alternating with Nesterov acceleration using scipy HiGHS."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)
    lp_times = []

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y)
        lp_times.append(time.monotonic() - lp_t0)

        if b_new is None:
            if len(lp_times) > 5 and np.mean(lp_times[-5:]) > 10.0:
                print(f"    LP too slow, stopping at iter {it}")
                break
            continue

        avg_lp_time = np.mean(lp_times[-10:]) if len(lp_times) >= 10 else 1.0

        # Nesterov dual averaging
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % print_interval == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            # Perturbation on stall
            if stall_count >= 15:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 10)
                noise = 0.02 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

            print(f"    LP-Nest iter={it+1} ({time.monotonic()-t0:.0f}s): "
                  f"score={best_score:.8f} avg_lp={avg_lp_time*1000:.0f}ms")

        # Stop if LP is too slow
        if len(lp_times) > 20 and avg_lp_time > 5.0:
            print(f"    LP too slow ({avg_lp_time:.2f}s/iter), stopping")
            break

    return best_a, best_score


def multiplicative_weights_optimize(a0, time_limit, eta_init=0.05, print_interval=5000):
    """Multiplicative weights (mirror descent) with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = eta_init
    no_improve = 0

    for step in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Adaptive active set
        threshold = max_b * (1.0 - max(5e-6, 1e-3 / (1 + step / 500)))
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Compute subgradient (averaged over active set)
        w = np.zeros(len(b))
        w[active] = 1.0 / len(active)
        g = compute_subgradient(w, a)

        # Multiplicative update
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 30:
            eta *= 0.7
            no_improve = 0
            if eta < 1e-5:
                break

        if step % print_interval == 0 and step > 0:
            print(f"    MultiWt {time.monotonic()-t0:.0f}s: score={best_score:.8f} eta={eta:.6f}")

    return best_a, best_score


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.25 * np.cos(2 * np.pi * 7 * x) + 0.15 * np.cos(2 * np.pi * 13 * x + 0.5)
        a = np.maximum(a, 0)
    elif choice == 7:
        # Two-peak structure
        a = 1.0 + 0.4 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    elif choice == 8:
        # Ramp with oscillation
        a = x * (1 - x) * 4 + 0.3 * np.cos(2 * np.pi * 5 * x)
        a = np.maximum(a, 0)
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturbed_sequence(a, rng, strength=0.02):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 15)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    # Also add a low-frequency perturbation
    freq2 = rng.integers(1, 5)
    phase2 = rng.uniform(0, 2 * np.pi)
    perturbation += 0.5 * strength * np.sin(freq2 * x + phase2)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def kkt_polish(a0, time_limit, n_steps=10000):
    """Polish using KKT conditions: at optimum, the subgradient at active peaks
    should be 'balanced' across the sequence."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        score = 2.0 * n * max_b

        # Find active peaks
        threshold = max_b * (1.0 - 1e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # For each active peak k, the "pressure" on a[i] is a[k-i]
        # We want to redistribute mass from high-pressure to low-pressure positions
        pressure = np.zeros(n)
        for k in active:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += a[j]
        pressure /= len(active)

        # Normalized pressure
        p_mean = pressure.mean()
        if p_mean < 1e-15:
            break

        # Update: reduce a where pressure is high, increase where low
        # Use a gentle multiplicative update
        lr = 0.01 / (1 + step / 1000)
        ratio = pressure / p_mean
        a_new = a * np.exp(-lr * (ratio - 1.0))
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new /= s_new

        new_score = 2.0 * n * fast_convolve(a_new).max()
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
        else:
            if step > 100:
                break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses Adam log-sum-exp, LP alternating with HiGHS, multiplicative weights,
    KKT polishing, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}) ***")

    n_best = len(best_seq)
    print(f"\nStarting optimization: n={n_best}, initial score={best_score:.8f}")
    print(f"Deadline in {deadline - time.monotonic():.0f}s")

    # === Phase 1: Quick Adam warm-up ===
    print(f"\n=== Phase 1: Adam Warm-up n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages = [30.0, 15.0, 8.0, 4.0, 2.0]
        a_adam, _ = adam_logsumexp_optimize(best_seq, t_alloc, T_stages,
                                             lr_init=0.008, lr_decay=0.8,
                                             top_k_frac=0.3, print_interval=5000)
        local_update(a_adam, "adam_warm")

    # === Phase 2: LP Alternating (main workhorse) ===
    print(f"\n=== Phase 2: LP Alternating n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = lp_alternate_nesterov(best_seq, t_alloc, max_iters=200000,
                                         print_interval=50)
        local_update(a_lp, "lp_alt_1")

    # === Phase 3: Fine Adam with low temperature ===
    print(f"\n=== Phase 3: Adam Fine (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 350.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages_fine = [5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01, 0.005]
        a_adam2, _ = adam_logsumexp_optimize(best_seq, t_alloc, T_stages_fine,
                                              lr_init=0.004, lr_decay=0.88,
                                              top_k_frac=0.08, print_interval=5000)
        local_update(a_adam2, "adam_fine")

    # === Phase 4: Second LP round ===
    print(f"\n=== Phase 4: LP Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 300.0)
    if t_alloc > 40:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = lp_alternate_nesterov(best_seq, t_alloc, max_iters=100000,
                                          print_interval=50)
        local_update(a_lp2, "lp_alt_2")

    # === Phase 5: Multiplicative weights polish ===
    print(f"\n=== Phase 5: MultiWeights Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 120.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ms, _ = multiplicative_weights_optimize(best_seq, t_alloc, eta_init=0.03)
        local_update(a_ms, "mult_weights")

    # === Phase 6: KKT polish ===
    print(f"\n=== Phase 6: KKT Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 60.0)
    if t_alloc > 10:
        a_kkt, _ = kkt_polish(best_seq, t_alloc, n_steps=5000)
        local_update(a_kkt, "kkt_polish")

    # === Phase 7: Try different lengths ===
    print(f"\n=== Phase 7: Multi-Length (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        n_cur = len(best_seq)
        for n_try in [n_cur + 300, n_cur - 200, n_cur + 600]:
            rem = deadline - time.monotonic()
            if rem < 50 or n_try < 200:
                continue
            t_sub = min(rem * 0.25, 120.0)
            if t_sub < 20:
                continue
            print(f"  Trying n={n_try} (budget {t_sub:.0f}s)...")
            init_try = interpolate_sequence(best_seq, n_try)
            # Quick Adam
            T_short = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05]
            a_try, _ = adam_logsumexp_optimize(init_try, t_sub * 0.6, T_short,
                                                lr_init=0.004, lr_decay=0.85,
                                                top_k_frac=0.15, print_interval=3000)
            local_update(a_try, f"adam_n{n_try}")

    # === Phase 8: Perturbation escape ===
    print(f"\n=== Phase 8: Perturbation Escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_cur = len(best_seq)
        for trial in range(3):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 25:
                break
            strength = 0.008 + 0.008 * trial
            a_pert = perturbed_sequence(best_seq, rng, strength=strength)
            t_sub = min(rem_sub * 0.3, 60.0)
            T_pert = [3.0, 1.5, 0.8, 0.4, 0.2, 0.1, 0.05, 0.02]
            a_opt, _ = adam_logsumexp_optimize(a_pert, t_sub, T_pert,
                                                lr_init=0.003, lr_decay=0.9,
                                                top_k_frac=0.1, print_interval=3000)
            local_update(a_opt, f"pert_t{trial}")

    # === Phase 9: Final LP polish ===
    print(f"\n=== Phase 9: Final LP Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.4, 80.0)
        if t_fin > 15:
            a_finlp, _ = lp_alternate_nesterov(best_seq, t_fin, max_iters=50000,
                                                print_interval=30)
            local_update(a_finlp, "final_lp")

    # === Phase 10: Final Adam micro-polish ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t_micro = min(rem * 0.4, 30.0)
        T_micro = [0.3, 0.15, 0.1, 0.05, 0.02, 0.01, 0.005, 0.003]
        a_micro, _ = adam_logsumexp_optimize(best_seq, t_micro, T_micro,
                                              lr_init=0.001, lr_decay=0.92,
                                              top_k_frac=0.05, print_interval=5000)
        local_update(a_micro, "final_adam")

    # Final verification
    if best_seq is not None:
        best_seq = np.maximum(best_seq, 0)
        s = best_seq.sum()
        if s > 0:
            best_seq = best_seq / s
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        final_eval = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
        print(f"\nFINAL SCORE (local): {best_score:.8f} (n={len(best_seq)})")
        print(f"FINAL SCORE (evaluate_sequence): {final_eval:.8f}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200.0

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
