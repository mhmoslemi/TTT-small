import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog

def fast_convolve(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) for sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def evaluate_safe(a):
    """Safely evaluate the objective."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        s = sum(a_list)
        if s < 0.01:
            return float('inf')
        b = np.convolve(a_list, a_list)
        return float(2 * len(a_list) * b.max() / (s**2))
    except Exception:
        return float('inf')

def build_conv_matrix(a):
    """Build the convolution matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        # T[k, j] = a[k-j] for valid k-j
        # k ranges from j to j+n-1
        T[j:j+n, j] = a
    return T

def solve_lp_best_response(a, method='highs'):
    """Solve the LP best response: min M s.t. T@b <= M, sum(b)=1, b>=0."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    
    # Variables: [b_0, ..., b_{n-1}, M]
    # Minimize: M -> c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: T @ b - M <= 0 -> [T, -1] @ [b; M] <= 0
    A_ub = np.hstack([T, -np.ones((size, 1))])
    b_ub = np.zeros(size)
    
    # Equality: sum(b) = 1 -> [1^T, 0] @ [b; M] = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: b_i >= 0, M >= 0
    bounds = [(0, None)] * (n + 1)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                    bounds=bounds, method=method)
    
    if result.status != 0:
        return None, None
    
    b_val = result.x[:n]
    M_val = result.x[n]
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, M_val

def nesterov_lp(a0, time_limit, max_iters=500000):
    global _deadline, _best_seq, _best_score
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:

    else:
        a = a / s

    local_best_a = a.copy()
    local_best_score = score_normalized(a)

    last_print = t0
    stall_count = 0
    prev_best = local_best_score + 1.0

    for it in range(max_iters):
        if time.monotonic() > _deadline or time.monotonic() - t0 > time_limit:
            break

        if b_new is None:

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) ** 2)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)

        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)

        sy = y_new.sum()
        if sy < 1e-15:

            y_new = y_new / sy

        a = a_new

            local_best_a = a.copy()
        sc_y = score_normalized(y)

            if sy > 1e-15:

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lse_objective_and_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 500:
        max_Tb = 500
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-16:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=50.0, T_end=0.01, n_stages=20):
    global _deadline, _best_seq, _best_score
    n = len(a0)

    t0 = time.monotonic()

        remaining = min(time_limit, _deadline) - time.monotonic()
        if remaining < 2:

            obj, _ = lse_objective_and_grad(z_val, stage_T, n)
            return obj

            _, grad = lse_objective_and_grad(z_val, stage_T, n)
            return grad

            res = minimize(
                fun=func_lse, x0=z, jac=jac_lse, method='L-BFGS-B',
                options={'maxiter': maxiter_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': maxiter_stage * 300,
                         'disp': False}

            z = res.x
            a_check = softmax_from_z(res.x)

                local_best_a = a_check.copy()

        except Exception as e:
            print(f"    LSE error at stage {si}: {e}")

    a_final = softmax_from_z(z)
    sc_final = score_normalized(a_final)
    if sc_final < local_best_score:
        local_best_score = sc_final
        local_best_a = a_final.copy()

    if local_best_score < _best_score:
        _best_score = local_best_score
        _best_seq = local_best_a.copy()
    return local_best_a, local_best_score


def subgradient_polish(a0, time_limit, n_steps=200000):
    global _deadline, _best_seq, _best_score
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    z = z_from_a(a).copy()
    local_best_a = a.copy()
    local_best_score = score_normalized(a)
    t0 = time.monotonic()
    eta_start = 1.0
    eta_end = 0.005

    for step in range(n_steps):
        if time.monotonic() > _deadline or time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.85
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        S = np.dot(g, a)
        grad_z = a * (g - S)

        progress = step / max(1, n_steps)
        eta = eta_start * (eta_end / eta_start) ** progress
        z = z - eta * grad_z
        a = softmax_from_z(z)
        sc = score_normalized(a)
        if sc < local_best_score:
            local_best_score = sc
            local_best_a = a.copy()

    if local_best_score < _best_score:
        _best_score = local_best_score
        _best_seq = local_best_a.copy()
    return local_best_a, local_best_score


def structured_init(n, rng, mode=None):
    if mode is None:
        mode = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = (1.0 - x) ** 1.0
    elif mode == 1:
        a = np.where(x < 0.5, 1.0, 0.5)
    elif mode == 2:
        a = 1.0 - 0.5 * x
    elif mode == 3:
        a = np.maximum(0.1, (1.0 - x) ** 2)
    elif mode == 4:
        a = np.cos(np.pi * x / 2) ** 2
    elif mode == 5:
        a = 1.0 / (1.0 + x * 5.0)
    elif mode == 6:
        a = np.maximum(0.05, 1.0 - 0.8 * x ** 0.5)
    else:
        a = np.where(x < 0.6, 1.0 + 0.2 * np.cos(3 * np.pi * x), 0.4)
    a = np.maximum(a, 0.01)
    s = a.sum()
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def nesterov_lp_budget(a0, time_limit):
    """Wrapper that checks global deadline."""
    global _deadline
    remaining = min(time_limit, _deadline - time.monotonic())
    if remaining < 5:
        return a0.copy(), score_normalized(a0)
    return nesterov_lp(a0, remaining)


def propose_candidate(budget_s=1800.0):
    global _deadline, _best_seq, _best_score, _lp_count
    rng = np.random.default_rng(12345)
    _deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    _best_seq = None
    _best_score = float('inf')
    _lp_count = 0

    print(f"Total budget: {budget_s}s, deadline in {budget_s - 30}s")

    # Get parent sequence as one initialization option
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_safe(parent):.8f}")
    except Exception:
        pass

    # Phase 1: Find best initialization from multiple structured starts
    print("\n=== Phase 1: Initialization search ===")
    t_init_end = time.monotonic() + 60.0
    n_base = 1200

    init_candidates = []
    if parent is not None:
        init_candidates.append(('parent', parent))

    for mode in range(8):
        a_try = structured_init(n_base, rng, mode=mode)
        init_candidates.append((f'struct_{mode}', a_try))

    # Quick evaluation of each
    for name, a_try in init_candidates:
        sc = score_normalized(a_try)
        print(f"  Init '{name}': score={sc:.8f}")
        if sc < _best_score:
            _best_score = sc
            _best_seq = a_try.copy()

    # Quick Nesterov on top-3 initializations
    init_candidates.sort(key=lambda x: score_normalized(x[1]))
    for name, a_try in init_candidates[:3]:
        if time.monotonic() > t_init_end:
            break
        remaining = t_init_end - time.monotonic()
        if remaining < 5:
            break
        print(f"  Quick Nesterov on '{name}' ({remaining:.0f}s)...")
        a_out, sc_out = nesterov_lp_budget(a_try, remaining)
        if sc_out < _best_score:
            _best_score = sc_out
            _best_seq = a_out.copy()

    print(f"  Phase 1 best: {_best_score:.8f} (n={len(_best_seq)})")

    # Phase 2: Main Nesterov LP optimization (60% of remaining budget)
    print(f"\n=== Phase 2: Main Nesterov LP (score={_best_score:.8f}) ===")
    remaining_total = _deadline - time.monotonic()
    t_main = 0.55 * remaining_total
    a_main, sc_main = nesterov_lp_budget(_best_seq, t_main)
    if sc_main < _best_score:
        _best_score = sc_main
        _best_seq = a_main.copy()
    print(f"  Phase 2 best: {_best_score:.8f}")

    # Phase 3: LSE polishing (20% of remaining)
    print(f"\n=== Phase 3: LSE L-BFGS polish ===")
    remaining = _deadline - time.monotonic()
    t_lse = 0.25 * remaining
    if t_lse > 20:
        a_lse, sc_lse = lbgfs_logsumexp(_best_seq, t_lse, T_start=80.0, T_end=0.01, n_stages=25)
        if sc_lse < _best_score:
            _best_score = sc_lse
            _best_seq = a_lse.copy()
    print(f"  Phase 3 best: {_best_score:.8f}")

    # Phase 4: Second Nesterov LP round (10%)
    print(f"\n=== Phase 4: Nesterov LP round 2 ===")
    remaining = _deadline - time.monotonic()
    t_r2 = 0.12 * remaining
    if t_r2 > 10:
        a_r2, sc_r2 = nesterov_lp_budget(_best_seq, t_r2)
        if sc_r2 < _best_score:
            _best_score = sc_r2
            _best_seq = a_r2.copy()
    print(f"  Phase 4 best: {_best_score:.8f}")

    # Phase 5: Subgradient polish (5%)
    print(f"\n=== Phase 5: Subgradient polish ===")
    remaining = _deadline - time.monotonic()
    t_sg = 0.08 * remaining
    if t_sg > 10:
        a_sg, sc_sg = subgradient_polish(_best_seq, t_sg, n_steps=300000)
        if sc_sg < _best_score:
            _best_score = sc_sg
            _best_seq = a_sg.copy()
    print(f"  Phase 5 best: {_best_score:.8f}")

    # Phase 6: Try different lengths (remaining time)
    print(f"\n=== Phase 6: Length exploration ===")
    n_current = len(_best_seq)
    candidates_n = []
    for delta in [50, 100, -50, -100, 150, 200, -150]:
        n_try = n_current + delta
        if 200 <= n_try <= 3000:
            candidates_n.append(n_try)

    remaining = _deadline - time.monotonic()
    if remaining > 30 and candidates_n:
        t_per = min(remaining / max(1, len(candidates_n)), 60.0)
        for n_try in candidates_n[:4]:
            remaining = _deadline - time.monotonic()
            if remaining < 15:
                break
            t_alloc = min(t_per, remaining - 5)
            if t_alloc < 10:
                break
            print(f"  Trying n={n_try} ({t_alloc:.0f}s)...")
            init_try = interpolate_sequence(_best_seq, n_try)
            a_try, sc_try = nesterov_lp_budget(init_try, t_alloc * 0.7)
            if sc_try < _best_score:
                _best_score = sc_try
                _best_seq = a_try.copy()
                print(f"  NEW BEST at n={n_try}: {_best_score:.8f}")

    # Phase 7: Final fine-tune LSE
    print(f"\n=== Phase 7: Final LSE fine-tune ===")
    remaining = _deadline - time.monotonic()
    if remaining > 15:
        t_final = min(remaining * 0.7, 90.0)
        if t_final > 10:
            a_fin, sc_fin = lbgfs_logsumexp(_best_seq, t_final, T_start=20.0, T_end=0.005, n_stages=15)
            if sc_fin < _best_score:
                _best_score = sc_fin
                _best_seq = a_fin.copy()

    # Final verification
    print(f"\n=== Final Result ===")
    final_score = evaluate_safe(_best_seq)
    print(f"  n={len(_best_seq)}, score={final_score:.8f}, LPs solved={_lp_count}")
    print(f"  Target: 1.5027, Previous: 1.504520")

    if not np.isfinite(final_score):
        # Fallback to uniform
        _best_seq = np.ones(1200) / 1200
        final_score = evaluate_safe(_best_seq)
        print(f"  Fallback to uniform: {final_score:.8f}")

    return [float(x) for x in _best_seq]
