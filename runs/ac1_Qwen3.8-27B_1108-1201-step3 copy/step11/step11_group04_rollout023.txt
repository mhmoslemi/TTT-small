def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix_topk(a, K=100):
    """Build matrix for top-K autocorrelation constraints."""
    n = len(a)
    b = fast_convolve(a)
    # Get top K lags
    top_idx = np.argsort(b)[::-1][:K]
    T = np.zeros((K, n))
    for i, k in enumerate(top_idx):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        T[i, idx] = a[k - idx]
    return T, top_idx


def solve_lp_best_response(a_fixed, solver, K=200):
    """LP best response with top-K constraints."""
    n = len(a_fixed)
    b = fast_convolve(a_fixed)
    # Get top K lags
    top_idx = np.argsort(b)[::-1][:K]
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    T = np.zeros((K, n))
    for i, k in enumerate(top_idx):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        T[i, idx] = a_fixed[k - idx]
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers_to_try = [solver, cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]
    solved = False
    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue
    
    if not solved:
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s_val = b_val.sum()
    if s_val < 1e-15:
        return None, None
    b_val = b_val / s_val
    return b_val, float(max(0, M_var.value))


def spectral_flatten(a, n_iters=50, blend=0.4, deadline=None):
    """GS spectral flattening."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)
        
        mags = np.abs(F)
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        
        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])
        
        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new
    
    return a


def lse_objective_and_grad(z, n, T):
    """LogSumExp objective and gradient in log-domain."""
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    s = a.sum()
    if s < 1e-300:
        a = np.ones(n) / n
    else:
        a = a / s
    
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T)
    lse_val = max_b + T * np.log(exp_terms.sum())
    obj = 2.0 * n * lse_val
    
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lbgfs_continuation(a0, time_limit, deadline=None):
    """L-BFGS with adaptive temperature continuation."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit
    
    # Adaptive temperature schedule
    T = 1.0  # Start temperature
    min_T = 0.0005
    T_decay = 0.7
    
    last_obj = lse_objective_and_grad(z, n, T)[0]
    no_improve = 0
    best_obj = last_obj
    best_z = z.copy()
    
    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        
        def func(z_in):
            if time.monotonic() > deadline:
                return 1e15
            return lse_objective_and_grad(z_in, n, T)[0]
        
        def jac(z_in):
            if time.monotonic() > deadline:
                return np.zeros_like(z_in)
            return lse_objective_and_grad(z_in, n, T)[1]
        
        # Determine max iterations based on remaining time
        maxiter = max(50, min(2000, int(remaining * 100)))
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-15, 'maxfun': maxiter * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            break
        
        obj = lse_objective_and_grad(z, n, T)[0]
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_z = z.copy()
        else:
            no_improve += 1
        
        # Adaptive temperature reduction
        if no_improve >= 3 or T <= min_T * 1.5:
            T *= T_decay
            no_improve = 0
            if T < min_T:
                T = min_T
        
        # Check convergence
        if T <= min_T and no_improve >= 5:
            break
    
    # Return softmax of best z
    z_best = best_z - best_z.max()
    a = np.exp(z_best)
    s = a.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return a / s

def subgradient_simplex(a0, time_limit, n_steps=500000):
    """Mirror descent (multiplicative weights) on the simplex with subgradients."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    eta = 0.01
    scale = 1.0
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Active set: top few peaks
        threshold = max_b * (1.0 - 1e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        # Multiplicative update
        eta_eff = eta * scale
        a_new = a * np.exp(-eta_eff * g)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-18:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 100:
            scale *= 0.5
            no_improve = 0
            if scale < 0.001:
                break
    
    return best_a, best_score


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP, cp.GUROBI, cp.MOSEK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def perturb_sequence(a, rng, strength=0.02):
    """Perturb sequence with structured noise."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 10)
    phase = rng.uniform(0, 2 * np.pi)
    a_pert = a + strength * np.cos(freq * x + phase)
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
