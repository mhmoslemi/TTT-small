def get_lp_solver():

            prob.solve(solver=solver_name)

def build_conv_matrix(a_fixed):
    """Build (2n-1) x n convolution matrix."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_fast(a_fixed, solver):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def peak_shave_intensive(a0, time_limit, seed=123):

    """

        je = min(n - 1, peak_idx)

        src_local = np.argmax(contrib)

        # Destination: positions that don't contribute to the peak (or contribute less)
        # and have low values
        
        # Find all positions NOT in the peak's support

        non_peak = [j for j in range(n) if j not in peak_support]
        
        if non_peak:
            # Sort by value (lowest first)
            non_peak_sorted = sorted(non_peak, key=lambda j: a[j])
            candidates = non_peak_sorted[:min(20, len(non_peak_sorted))]
        else:
            candidates = [j for j in range(n) if j != src]
            candidates.sort(key=lambda j: a[j])
            candidates = candidates[:10]
        
        improved = False
        for dst in candidates:
            for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.05, 0.1]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try random moves
            for _ in range(10):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.02)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt

        else:
            stall = 0
    
    return best_a, best_score


def gradient_step_optimize(a0, time_limit):
    """
    Subgradient method with projection onto simplex.
    Uses the fact that the subgradient of max_k (a*a)[k] is 2*T_k^T * a for k in argmax.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Precompute convolution matrix
    T = build_conv_matrix(a)  # This changes with a, so we can't precompute
    
    eta = 0.1  # Initial step size
    for t in range(1, 10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        peak_idx = int(np.argmax(b))
        
        # Subgradient: grad[i] = 2 * sum_{j: j + i = peak_idx} a[j] * delta
        # Actually, d/d a[i] of (a*a)[peak_idx] = 2 * a[peak_idx - i] (for valid i)
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        grad = np.zeros(n)
        for i in range(js, je + 1):
            grad[i] = 2.0 * a[peak_idx - i]
        
        # Also consider other peaks (top few)
        top_k = np.argsort(b)[::-1][:3]
        for k in top_k:
            js2 = max(0, k - n + 1)
            je2 = min(n - 1, k)
            for i in range(js2, je2 + 1):
                grad[i] += a[k - i]
        
        # Gradient step
        eta_t = eta / np.sqrt(t + 1)
        a_new = a - eta_t * grad
        
        # Project onto simplex {x >= 0, sum(x) = 1}
        a_new = project_simplex(a_new, 1.0)
        
        # Check score
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score


def project_simplex(v, s=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = s}."""
    n = len(v)
    # Scale to target sum
    v = v / max(v.sum(), 1e-300) * s
    
    b = np.sort(v)[::-1]
    cssv = np.cumsum(b)
    rho = np.nonzero(b * np.arange(1, n + 1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated LP best-response with running average (main workhorse)
2. Intensive multi-direction peak shaving coordinate descent
3. Multi-length exploration with interpolation transfer
4. Symmetric-structure exploitation for faster convergence
5. Adaptive time allocation based on observed convergence rate
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def get_lp_solver():
    """Find a working LP solver with speed test."""
    best_solver = cp.ECOS
    best_time = float('inf')
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(100, nonneg=True)
            c = np.random.default_rng(0).random(100)
            A_mat = np.random.default_rng(1).random((50, 100))
            b_vec = A_mat @ np.ones(100) / 100
            prob = cp.Problem(cp.Minimize(c @ x), [A_mat @ x <= b_vec, cp.sum(x) == 1])
            t0 = time.monotonic()
            prob.solve(solver=solver_name)
            elapsed = time.monotonic() - t0
            if prob.status in ('optimal', 'optimal_inaccurate') and elapsed < best_time:
                best_time = elapsed
                best_solver = solver_name
        except Exception:
            continue
    return best_solver


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n convolution matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_fast(a_fixed, solver):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, eps=1e-9, max_iter=100000)
    except Exception:
        try:
            prob.solve(solver=solver)
        except Exception:
            pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, eps=1e-9)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver, check_interval=100):
    """
    Nesterov-accelerated LP best-response with running average.
    Returns the best sequence found (either iterate or running average).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Running average
    sum_avg = np.zeros(n)
    count = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_fast(y, solver)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            sa = a_new.sum()
            if sa < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / sa

        y = y_new
        a = a_new
        sum_avg += a
        count += 1

        # Check scores at intervals
        if t % check_interval == 0:
            score_a = 2.0 * n * fast_autocorr(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()

            score_avg = 2.0 * n * fast_autocorr(sum_avg / count).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = (sum_avg / count).copy()

            score_y = 2.0 * n * fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()

            if t % (check_interval * 5) == 0:
                elapsed = time.monotonic() - t0
                print(f"        NLP t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    # Final check
    score_final = 2.0 * n * fast_autocorr(sum_avg / count).max()
    if score_final < best_score:
        best_score = score_final
        best_a = (sum_avg / count).copy()

    return best_a, best_score


def peak_shave(a0, time_limit, seed=123):
    """
    Intensive coordinate descent peak shaving.
    Tries many directions and step sizes to reduce the max autocorrelation.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:5]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            # Top 3 contributors
            top_contrib = np.argsort(contrib)[::-1][:3]

            for tc in top_contrib:
                src = idx[tc]
                if a[src] < 1e-12:
                    continue

                # Build candidate destination list
                # Prefer positions outside the peak's support
                peak_support = set(idx.tolist())
                non_peak = [j for j in range(n) if j not in peak_support]

                if non_peak:
                    # Sort by value, take lowest
                    non_peak.sort(key=lambda j: a[j])
                    candidates = non_peak[:min(15, len(non_peak))]
                else:
                    candidates = [j for j in range(n) if j != src]
                    candidates.sort(key=lambda j: a[j])
                    candidates = candidates[:10]

                # Also add some random candidates
                for _ in range(3):
                    rc = int(rng.integers(n))
                    if rc != src and rc not in candidates:
                        candidates.append(rc)

                for dst in candidates:
                    if dst == src:
                        continue
                    for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.06, 0.1]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < 0:
                            continue
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try random 2-swap moves
            found_rand = False
            for _ in range(20):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-10:
                    continue
                for frac in [0.002, 0.005, 0.01, 0.02]:
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        found_rand = True
                        break
                if found_rand:
                    break

            if not improved:
                stall += 1
                if stall > 300:
                    break
            else:
                stall = 0
        else:
            stall = 0

    return best_a, best_score


def logsumexp_refine(a0, time_limit):
    """
    LogSumExp-smoothed optimization using scipy L-BFGS-B.
    Works in the log-parameterization (softmax).
    """
    from scipy.optimize import minimize

    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a_init = a_init / a_init.sum()
    z = np.log(a_init)

    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    t0 = time.monotonic()

    def softmax_stable(zz):
        zz = zz - zz.max()
        e = np.exp(zz)
        s = e.sum()
        if s < 1e-300:
            return np.ones(len(zz)) / len(zz)
        return e / s

    def lse_and_grad(zz, T):
        aa = softmax_stable(zz)
        b = fast_autocorr(aa)
        Tb = T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        obj = (max_Tb + np.log(exp_Tb.sum())) / T
        w = exp_Tb / exp_Tb.sum()

        # grad_a = 2 * cross-correlation of a and w
        conv_len = n + (2 * n - 1) - 1
        fft_sz = 1 << (conv_len - 1).bit_length()
        if conv_len > fft_sz:
            fft_sz *= 2
        A_rev = rfft(aa[::-1].copy(), fft_sz)
        W = rfft(w, fft_sz)
        conv = irfft(A_rev * W, fft_sz)
        grad_a = 2.0 * conv[n - 1:n - 1 + n]

        S = np.dot(grad_a, aa)
        grad_z = aa * (grad_a - S)
        return obj, grad_z

    # Multi-stage annealing
    temps = [50.0, 20.0, 8.0, 3.0, 1.0, 0.3, 0.1, 0.05]
    for Ti, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        max_iter = max(10, min(100, int(remaining / len(temps) * 2)))

        def func(zz):
            return lse_and_grad(zz, T)[0]

        def jac(zz):
            return lse_and_grad(zz, T)[1]

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-12, 'maxfun': max_iter * 300, 'disp': False}
            )
            z = res.x
            a = softmax_stable(z)
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception:
            pass

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
