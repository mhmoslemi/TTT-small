import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import lil_matrix, csc_matrix, coo_matrix

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_convolve_cross(a, b):
    """Compute convolve(a, b) using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)

    for solver_name in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:

            if prob.status in ('optimal', 'optimal_inaccurate'):

        except Exception:

    return cp.ECOS

    if s < 0.01:

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def precompute_lp_indices(n):
    """Precompute row/col indices for the convolve matrix T where T@b = convolve(a,b)."""
    rows_list = []

        rows_list.extend(range(j, j + n))
        cols_list.extend([j] * n)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)

        violated = np.where(b_new > t_sol + tol)[0]

            print(f"    CP iter {cp_iter+1}: score={best_score:.10f}, active={len(active)}, "
                  f"violated={len(violated)}, t_sol={t_sol:.10f}")
        
        if len(violated) == 0:
            print(f"    Cutting-plane converged at iter {cp_iter+1}!")
            break
        
        # Add the most violated constraint
        k_most = int(violated[np.argmax(b_new[violated])])
        if k_most not in active:
            active.append(k_most)
        
        # Limit active set size
        if len(active) > 40:
            # Keep the ones with highest b_new values
            active_vals = [(k, b_new[k]) for k in active]
            active_vals.sort(key=lambda x: -x[1])
            active = [k for k, _ in active_vals[:40]]
            active.sort()
        
        # Convergence check: if improvement is tiny, stop
        if cp_iter > 10:
            prev_score = 2.0 * n * fast_convolve(best_a).max()  # This is wasteful
            # Actually, let's just check if the gap is small
            if best_score - 2.0 * n * t_sol < 1e-12:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))
