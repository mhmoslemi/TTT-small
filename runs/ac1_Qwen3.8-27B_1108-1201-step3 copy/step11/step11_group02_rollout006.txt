"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) FFT-based LogSumExp mirror descent, (2) Nesterov LP alternation,
(3) multi-n exploration, (4) coordinate descent polish.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_score(a):
    """Compute the score (lower is better)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def lse_obj_grad(a, T):
    """LogSumExp smoothed objective and gradient, computed via FFT."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp(b / T - max_b / T)
    w = exp_b / exp_b.sum()
    
    # Gradient: grad[i] = (4n/T) * sum_j w[i+j] * a[j]
    # Computed as cross-correlation of w and a
    fft_size = 1 << (2 * n - 2).bit_length()  # >= 2n-1
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    xc = irfft(W * np.conj(A), fft_size)
    grad = (4.0 * n / T) * xc[:n]
    
    obj = (2.0 * n / T) * (max_b / T + np.log(exp_b.sum()))
    return obj, grad


def mirror_descent_lse(a0, time_limit, T_start=50.0, T_end=0.05, n_iters=None):
    """Mirror descent with LogSumExp smoothing and adaptive step size."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
        s = 1.0
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    
    # T schedule: geometric from T_start to T_end
    if n_iters is None:
        n_iters = 200000
    
    T_positions = np.geomspace(T_start, T_end, n_iters)
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        T = T_positions[it]
        obj, g = lse_obj_grad(a, T)
        
        # Adaptive step size: based on gradient magnitude
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        eta = 0.1 / (g_norm / n + 1e-10)
        eta = min(eta, 1.0)
        
        # Mirror descent update (multiplicative weights)
        a_trial = a * np.exp(-eta * g)
        a_trial = np.maximum(a_trial, 1e-300)
        st = a_trial.sum()
        if st < 1e-300:
            break
        a_trial = a_trial / st
        
        score = eval_score(a_trial)
        if score < best_score:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
        else:
            # Try smaller step
            a_trial2 = a * np.exp(-eta * 0.1 * g)
            a_trial2 = np.maximum(a_trial2, 1e-300)
            st2 = a_trial2.sum()
            if st2 > 1e-300:
                a_trial2 = a_trial2 / st2
                score2 = eval_score(a_trial2)
                if score2 < best_score:
                    best_score = score2
                    best_a = a_trial2.copy()
                    a = a_trial2
                elif score2 < eval_score(a):
                    a = a_trial2
    
    return best_a, best_score


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b) (element-wise product)."""
    n = len(a)
    size = 2 * n - 1
    # T[k, j] = a[k-j] if 0 <= k-j < n
    row_ind = []
    col_ind = []
    data = []
    for k in range(size):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            row_ind.append(k)
            col_ind.append(j)
            data.append(a[k - j])
    T = csr_matrix((data, (np.array(row_ind), np.array(col_ind))), shape=(size, n))
    return T


def solve_lp_best_response(a, T_mat, solver):
    """Given fixed a (sum=1), find b minimizing max_k (a*b)[k]."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [T_mat @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    try:
        prob.solve(solver=solver)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            for alt in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
                if alt == solver:
                    continue
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
    except Exception:
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val = b_val / s
    return b_val


def nesterov_lp(a0, time_limit, solver, max_iters=100000):
    """Nesterov LP alternation with periodic mirror descent refinement."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(98765)
    stall = 0
    stall_threshold = 15
    
    # Precompute: we'll rebuild T each iteration since a changes
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Build T for current y
        T_mat = build_conv_matrix(y)
        b_new = solve_lp_best_response(y, T_mat, solver)
        if b_new is None:
            continue
        
        # Nesterov update
        alpha = 2.0 / (np.sqrt(it + 1) + 1.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        # FTRL-style weighted average for a
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
            sa = 1.0
        a = a / sa
        
        # Check score
        score_a = eval_score(a)
        score_y = eval_score(y)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        # Periodic mirror descent refinement
        if (it + 1) % 200 == 0:
            T_md = min(5.0, 50.0 / (np.sqrt(it + 1) + 1))
            time_left = time_limit - (time.monotonic() - t0)
            if time_left > 10:
                a_md, _ = mirror_descent_lse(a, min(15, time_left * 0.1),
                                              T_start=T_md, T_end=T_md * 0.1,
                                              n_iters=500)
                sc_md = eval_score(a_md)
                if sc_md < best_score:
                    best_score = sc_md
                    best_a = a_md.copy()
                    a = a_md.copy()
                    y = a.copy()
                    stall = 0
        
        # Stall handling: structured perturbation
        if stall >= stall_threshold:
            b_cur = fast_convolve(y)
            peak_idx = np.argmax(b_cur)
            # Reduce top contributors to peak
            js = max(0, peak_idx - n + 1)
            je = min(n - 1, peak_idx)
            idx = np.arange(js, je + 1)
            contribs = y[idx] * y[peak_idx - idx]
            n_reduce = max(5, len(idx) // 8)
            top_pos = idx[np.argsort(contribs)[::-1][:n_reduce]]
            for pos in top_pos:
                y[pos] *= 0.95
            # Add noise
            noise = rng.standard_normal(n) * 0.02
            y = y + noise
            y = np.maximum(y, 0)
            sy2 = y.sum()
            if sy2 > 1e-15:
                y = y / sy2
            a = y.copy()
            stall = 0
            stall_threshold = min(30, stall_threshold + 5)
        
        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def interpolate_sequence(a, new_n):
    """Interpolate to new length, preserving sum."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp = a_interp / ss
    return a_interp


def coord_descent(a0, time_limit, max_steps=500000):

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)

            break
        
        b = fast_convolve(a)
        max_b = b.max()
        # Find top peaks
        peaks = np.argsort(b)[::-1][:3]
        
        improved = False

            if improved:
                break
            peak_k = int(peak_k)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            
            # Try reducing top contributors
            top_m = min(10, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]
            
            for pos in top_positions:
                if a[pos] < 1e-14:
                    continue
                for factor in [0.9999, 0.9998, 0.9995, 0.999, 0.998, 0.995]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    # Redistribute the difference proportionally to others
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-18:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] += diff * a_trial[mask] / others_sum
                    a_trial = a_trial / a_trial.sum()
                    
                    score = eval_score(a_trial)
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()

                        no_improve = 0

                    break

                a_trial = a_trial / a_trial.sum()
                score = eval_score(a_trial)
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()

        else:
            no_improve = 0
        
        if (step + 1) % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: (1) FFT-accelerated LogSumExp mirror descent for fast local steps,
(2) Nesterov-accelerated LP alternation for exact max reduction,
(3) multi-resolution exploration, (4) fine coordinate descent polish.
Key improvements: adaptive LP cadence, periodic MD refinement within LP loop,
better stall recovery, and tighter final polish.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_score(a):
    """Compute the objective score (lower is better)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def lse_obj_grad(a, T):
    """LogSumExp smoothed objective and gradient via FFT cross-correlation."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp(b / T - max_b / T)
    w = exp_b / exp_b.sum()
    fft_size = 1 << (2 * n - 2).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    xc = irfft(W * np.conj(A), fft_size)
    grad = (4.0 * n / T) * xc[:n]
    obj = (2.0 * n / T) * (max_b / T + np.log(exp_b.sum()))
    return obj, grad


def mirror_descent_lse(a0, time_limit, T_start=10.0, T_end=0.05, max_iters=200000):
    """Mirror descent with LogSumExp smoothing and backtracking step size."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
        s = 1.0
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    T_positions = np.geomspace(T_start, T_end, max_iters)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        T = T_positions[it]
        obj, g = lse_obj_grad(a, T)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        eta = 0.5 / (g_norm / n + 1e-10)
        eta = min(eta, 2.0)
        # Try full step, then halve if no improvement
        for attempt in range(3):
            a_trial = a * np.exp(-eta * g)
            a_trial = np.maximum(a_trial, 1e-300)
            st = a_trial.sum()
            if st < 1e-300:
                break
            a_trial = a_trial / st
            score = eval_score(a_trial)
            if score < best_score:
                best_score = score
                best_a = a_trial.copy()
                a = a_trial
                break
            elif score < eval_score(a):
                a = a_trial
                break
            eta *= 0.5
    return best_a, best_score


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = element-wise product for convolve(a,b)."""
    n = len(a)
    size = 2 * n - 1
    row_idx = np.zeros(size * n, dtype=np.int32)
    col_idx = np.zeros(size * n, dtype=np.int32)
    data = np.zeros(size * n, dtype=np.float64)
    count = 0
    for k in range(size):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            row_idx[count] = k
            col_idx[count] = j
            data[count] = a[k - j]
            count += 1
    row_idx = row_idx[:count]
    col_idx = col_idx[:count]
    data = data[:count]
    return csr_matrix((data, (row_idx, col_idx)), shape=(size, n))


def get_lp_solver():
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp(a, T_mat, solver):
    """LP: given fixed a (sum=1), find b>=0, sum(b)=1 minimizing max_k (a*b)[k]."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [T_mat @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver, eps_abs=1e-10, eps_rel=1e-10)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            for alt in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
                if alt == solver:
                    continue
                prob.solve(solver=alt, eps_abs=1e-10, eps_rel=1e-10)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
    except Exception:
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0)
    sv = b_val.sum()
    if sv < 1e-15:
        return None
    return b_val / sv


def nesterov_lp(a0, time_limit, solver, max_iters=200000):
    """Nesterov LP alternation with periodic MD refinement and adaptive stall recovery."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    stall_thresh = 20
    prev_best = best_score
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        T_mat = build_conv_matrix(y)
        b_new = solve_lp(y, T_mat, solver)
        if b_new is None:
            continue
        
        alpha = 2.0 / (np.sqrt(it + 1) + 1.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        w_old = it + 1.0
        w_new = alpha * (it + 2.0)
        a = (w_old * a + w_new * b_new) / (w_old + w_new)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
            sa = 1.0
        a = a / sa
        
        score_a = eval_score(a)
        score_y = eval_score(y)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        # Periodic MD refinement every 150 iterations
        if (it + 1) % 150 == 0 and (time.monotonic() - t0) < time_limit - 20:
            T_md = max(0.5, 20.0 / (np.sqrt(it + 1) + 1.0))
            t_md = min(10.0, (time_limit - (time.monotonic() - t0)) * 0.05)
            if t_md > 3:
                a_md, _ = mirror_descent_lse(a, t_md, T_start=T_md, T_end=T_md * 0.05, max_iters=300)
                sc_md = eval_score(a_md)
                if sc_md < best_score:
                    best_score = sc_md
                    best_a = a_md.copy()
                    a = a_md.copy()
                    y = a.copy()
                    stall = 0
        
        # Stall recovery
        if stall >= stall_thresh:
            b_cur = fast_convolve(y)
            peak_idx = int(np.argmax(b_cur))
            js = max(0, peak_idx - n + 1)
            je = min(n - 1, peak_idx)
            idx = np.arange(js, je + 1)
            contribs = y[idx] * y[peak_idx - idx]
            n_red = max(5, len(idx) // 6)
            top_pos = idx[np.argsort(contribs)[::-1][:n_red]]
            for pos in top_pos:
                y[pos] *= 0.93
            # Add structured + random perturbation
            x_linspace = np.linspace(0, 1, n)
            osc = 0.03 * np.sin(4 * np.pi * x_linspace + rng.uniform(0, 2 * np.pi))
            noise = rng.standard_normal(n) * 0.015
            y = y + osc + noise
            y = np.maximum(y, 0)
            sy2 = y.sum()
            if sy2 > 1e-15:
                y = y / sy2
            a = y.copy()
            stall = 0
            stall_thresh = min(40, stall_thresh + 5)
        
        # Check for progress every 100 iters
        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-12:
                prev_best = best_score
        
        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_n = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_i = np.interp(new_x, old_x, a_n)
    a_i = np.maximum(a_i, 0)
    ss = a_i.sum()
    if ss > 1e-15:
        a_i = a_i / ss
    return a_i


def coord_descent(a0, time_limit, max_steps=1000000):
    """Fine coordinate descent targeting peak-contributing elements with redistribution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    no_improve = 0
    
    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_m = min(12, len(idx))
        top_pos = idx[np.argsort(contribs)[::-1][:top_m]]
        
        improved = False
        for pos in top_pos:
            if a[pos] < 1e-14:
                continue
            for factor in [0.99995, 0.9999, 0.9998, 0.9995, 0.999, 0.998, 0.995]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] - a_trial[pos]
                if diff < 1e-18:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                ot = a_trial[mask].sum()
                if ot < 1e-15:
                    break
                a_trial[mask] += diff * a_trial[mask] / ot
                a_trial = a_trial / a_trial.sum()
                score = eval_score(a_trial)
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                    break
            if improved:
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 3000:
                # Simultaneous small reduction of top contributors
                a_trial = a.copy()
                for pos in top_pos[:5]:
                    if a_trial[pos] > 1e-14:
                        a_trial[pos] *= 0.999
                a_trial = a_trial / a_trial.sum()
                sc = eval_score(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    no_improve = 0
                elif no_improve > 15000:
                    break
        else:
            no_improve = 0
        
        if (step + 1) % 100000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score
