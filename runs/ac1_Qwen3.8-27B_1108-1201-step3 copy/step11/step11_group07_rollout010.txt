Let me also think about the LP more carefully. The constraint matrix A has a special structure: it's a "convolution matrix" (or "Toeplitz" matrix). Specifically:

A[k, i] = a[i] if 0 <= k-i < n, else 0

This means A[k, i] = a[i] * 1_{0 <= k-i < n}

The (k,i) entry depends on k-i, so it's a Hankel matrix (not Toeplitz, since it's a[i] not a[k-i]).

Actually wait: A[k,i] = a[i] if max(0, k-n+1) <= i <= min(n-1, k).

This is a banded matrix with:
- For k < n: the non-zero part starts at column 0 and ends at column k (width k+1)
- For k >= n: the non-zero part starts at column k-n+1 and ends at column n-1 (width 2n-1-k)

The maximum band width is n (at k = n-1).

For a sparse LP solver, this should be efficient.

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iteration
(primal approach to reduce autocorrelation peak), combined with
LogSumExp gradient polishing and multi-start exploration.
The LP step solves: max sum(b) s.t. conv(a,b) <= M, b >= 0,
which finds the best direction to reduce the autocorrelation peak.

import numpy as np

from scipy.sparse import csr_matrix

def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    B = rfft(b.astype(np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def evaluate_local(a):
    """Evaluate score: 2*n*max(conv(a,a))/(sum(a))^2"""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_constraint_matrix(a):
    """Build the sparse constraint matrix A where A[k,i] = a[i] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            i_range = range(i_min, i_max + 1)
            rows.extend([k] * (i_max - i_min + 1))
            cols.extend(list(i_range))
            vals.extend(list(a[i_min:i_max + 1]))
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_step(a, timeout=5.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0.
    Returns the optimal b (normalized to sum=1) or None if failed.
    """
    n = len(a)
    # Scale a to sum=n for numerical stability
    s = a.sum()
    a_scaled = a * n / s
    
    M = fast_convolve(a_scaled).max()
    
    # Build constraint matrix
    A_mat = build_constraint_matrix(a_scaled)
    
    # LP: minimize -sum(b) s.t. A_mat @ b <= M, b >= 0
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    try:
        res = linprog(c, A_ub=A_mat, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')
        if res.status == 0:
            b = res.x
            b = np.maximum(b, 0)
            bs = b.sum()
            if bs < 1e-15:
                return None
            return b / bs  # return normalized
    except Exception:
        return None


def lp_iteration(a, n_tries=10):
    """One LP-based iteration with line search."""
    n = len(a)
    a = normalize(a)
    
    b = lp_step(a)
    if b is None:
        return a, evaluate_local(a)
    
    best_a = a.copy()
    best_score = evaluate_local(a)
    
    # Backtracking line search
    for t in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.01]:
        a_new = (1 - t) * a + t * b
        a_new = normalize(a_new)
        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)

    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    return normalize(f_interp)

    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative refinement.
The LP step finds the optimal direction (maximizing mass under the current
autocorrelation envelope), providing a principled descent direction.
Combined with subgradient polishing and multi-start exploration.
"""
import numpy as np

from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def fast_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    B = rfft(b.astype(np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def evaluate_local(a):
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    """Build sparse matrix A where (A @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        count = i_max - i_min + 1
        if count > 0:
            rows.extend([k] * count)
            cols.extend(range(i_min, i_max + 1))
            vals.extend(a[i_min:i_max + 1].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(a):
    """Solve: max sum(b) s.t. conv(a,b) <= M, b >= 0. Returns b normalized."""
    n = len(a)
    s = a.sum()
    a_sc = a * n / s  # scale to sum = n
    M = fast_convolve(a_sc).max()
    
    A_mat = build_conv_matrix(a_sc)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    try:
        res = linprog(c, A_ub=A_mat, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')
        if res.status == 0:
            b = np.maximum(res.x, 0)
            bs = b.sum()
            if bs > 1e-12:
                return b / bs
    except Exception:
        pass
    return None


def lp_iteration(a, rng):
    """One LP iteration with line search. Returns (new_a, score)."""
    a = normalize(a)
    b = solve_lp(a)
    if b is None:
        return a, evaluate_local(a)
    
    best_a = a.copy()
    best_sc = evaluate_local(a)
    
    for t in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
        a_new = normalize((1 - t) * a + t * b)
        sc = evaluate_local(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new
            break
    
    return best_a, best_sc


def subgradient_polish(a, time_limit, n_iters=5000):
    """Polish using subgradient descent on max(conv(a,a))."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = evaluate_local(a)
    t0 = time.monotonic()
    lr = 0.01
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        k_star = int(np.argmax(c))
        
        # Subgradient at k_star
        g = np.zeros(n)
        i_min = max(0, k_star - n + 1)
        i_max = min(n - 1, k_star)
        for i in range(i_min, i_max + 1):
            j = k_star - i
            g[i] += a[j]
            if i != j:
                g[j] += a[i]
        
        # Projected gradient step
        a_new = a - lr * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        sc = evaluate_local(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
        elif it % 100 == 0:
            lr *= 0.9
        
        if lr < 1e-8:
            break
    
    return best_a, best_sc


def lse_polish(a, time_limit):
    """Polish using LSE-smoothed gradient descent."""
    n = len(a)
    T_stages = [10, 5, 2, 1, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
    
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = evaluate_local(a)
    t0 = time.monotonic()
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        z = np.log(np.maximum(a, 1e-30))
        lr = 0.5 / n
        fft_size = 1 << (2 * n).bit_length()
        
        n_iters = min(200, max(20, int(remaining * 5)))
        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Compute a from z
            z_sh = z - z.max()
            ez = np.exp(z_sh)
            a_loc = ez / ez.sum()
            
            # Compute LSE gradient
            c = fast_convolve(a_loc)
            c_max = c.max()
            ec = np.exp((c - c_max) / T)
            w = ec / ec.sum()
            
            # Cross-correlation of w and a
            W = rfft(w, fft_size)
            Al = rfft(a_loc, fft_size)
            R = irfft(W * np.conj(Al), fft_size)
            b_arr = R[:n]
            
            g = 2.0 * b_arr
            idx = 2 * np.arange(n)
            valid = idx < (2 * n - 1)
            g[valid] -= a_loc[valid] * w[idx[valid]]
            
            # Project to tangent space (sum(a)=1, a>=0)
            S = np.dot(g, a_loc)
            grad_z = a_loc * (g - S)
            
            # Gradient step
            z_new = z - lr * grad_z
            z_sh2 = z_new - z_new.max()
            ez2 = np.exp(z_sh2)
            a_new = ez2 / ez2.sum()
            
            sc = evaluate_local(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
                a = a_new
                z = z_new
            else:
                lr *= 0.5
                if lr < 1e-10:
                    break
        
        a = best_a  # reset to best for next stage
    
    return best_a, best_sc

def interp_seq(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:

    fn = normalize(f)
    ox = np.linspace(0, 1, old_n)
    nx = np.linspace(0, 1, new_n)
    fi = np.interp(nx, ox, fn)
    fi = np.maximum(fi, 0)
    return normalize(fi)
