**Key Idea: Use the "mirror descent" / exponentially weighted average approach**

For the zero-sum game:
- Player 1 (minimizer): chooses a (probability vector)
- Player 2 (maximizer): chooses b (distribution on lags)

The mirror descent update for the minimizer is:
a_{t+1}(i) ∝ a_t(i) * exp(-η * g_t(i))

where g_t is the payoff gradient. This is the "multiplicative weights" update.

The convergence is O(sqrt(n/T)) for the game value, and the running average gives O(1/sqrt(T)).

But the LP-based Nesterov should be better (O(1/T) rate).

Use the LP to find the optimal b distribution, then use that to compute a better gradient for the LogSumExp optimization.

**Key Idea: Use the "active set" KKT conditions**

At the optimum:
- There exists a set K of active lags and weights λ_k such that:
  - For all i with a_i > 0: sum_{k in K} λ_k * (a_{k-i} + a_{i-k}) = constant (call it c)
  - For all i with a_i = 0: sum_{k in K} λ_k * (a_{k-i} + a_{i-k}) <= c
  - sum_{k in K} λ_k = 1, λ_k >= 0
  - b_k = M for all k in K (the active lags all achieve the same max)

This gives a system of equations. We could try to solve it iteratively:
1. Start with an initial a
2. Find K = argmax lags
3. Solve for λ such that the gradient condition holds (this is a small LP or QP)
4. Update a based on the KKT conditions
5. Repeat

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_from_a(a):
    """Compute the score 2n*max(b)/(sum(a))^2."""
    n = len(a)
    b = fast_autocorr(a)
    s = a.sum()
    return 2.0 * n * b.max() / (s * s)

def lse_obj_grad(z, T):
    """
    LogSumExp smoothing of max(b) with a = exp(z) parameterization.
    Returns (objective, gradient).
    """
    n = len(z)
    # For numerical stability, shift z
    z_shift = z - z.max()
    a = np.exp(z_shift)
    
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    sum_exp = exp_Tb.sum()
    obj = (max_Tb + np.log(sum_exp)) / T
    
    w = exp_Tb / sum_exp
    
    # grad_a[i] = 2 * sum_k w[k] * a[k-i] (a zero-extended)
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    A_rev = rfft(a[::-1], fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad_a = 2.0 * conv[n - 1:n - 1 + n]
    
    # grad_z[i] = a[i] * grad_a[i]
    grad_z = a * grad_a
    
    return obj, grad_z

def lse_optimize(a_init, time_limit, T_values):
    """
    Multi-stage LogSumExp optimization.
    a_init: initial sequence (will be converted to z = log(a))
    time_limit: total time in seconds
    T_values: list of temperatures to use (decreasing)
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).astype(np.float64)
    z = np.log(a / a.max())  # Shift for stability
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = score_from_a(a)
    
    for T in T_values:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # Determine max iterations based on remaining time
        # Each iteration of L-BFGS takes roughly 2-5 FFT operations
        # For n=1600, one FFT takes ~0.1ms, so one L-BFGS iter takes ~1-5ms
        # Plus the function/gradient evaluation overhead
        max_iter = min(2000, max(50, int(remaining * 200)))
        
        def func(z_in):
            val, _ = lse_obj_grad(z_in, T)
            return val
        
        def jac(z_in):
            _, g = lse_obj_grad(z_in, T)
            return g
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': max_iter * 50, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            score_new = score_from_a(a_new)
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
            z = np.log(np.maximum(a_new, 1e-30) / a_new.max())
        except Exception as e:
            print(f"    LSE stage T={T} failed: {e}")
    
    return best_a, best_score

def active_set_polish(a_init, time_limit, K=5):
    """
    Active-set subgradient method: identify top-K lags, average their gradients,
    take projected gradient steps.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    
    lr = 0.1  # Learning rate (will be adapted)
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        
        # Find top-K lags
        top_k = np.argsort(b)[::-1][:K]
        
        # Compute average gradient of top-K b_k
        # grad_a[i] = 2 * sum_{k in top_k} b_k/max_b * a[k-i] / K
        # Actually, the subgradient of max(b) is the average of gradients of active components
        w = np.zeros(2 * n - 1)
        for k in top_k:
            w[k] = 1.0 / K
        
        # grad_a[i] = 2 * sum_k w[k] * a[k-i]
        conv_len = n + (2 * n - 1) - 1
        fft_size = 1 << (conv_len - 1).bit_length()
        A_rev = rfft(a[::-1], fft_size)
        W = rfft(w, fft_size)
        conv = irfft(A_rev * W, fft_size)
        grad_a = 2.0 * conv[n - 1:n - 1 + n]
        
        # Projected gradient step (project onto a >= 0, sum(a) = 1)
        a_trial = a - lr * grad_a
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-15:
            break
        a_trial = a_trial / s
        
        score_trial = score_from_a(a_trial)
        if score_trial < best_score - 1e-15:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            stall = 0
            lr = min(lr * 1.05, 1.0)  # Increase learning rate slightly
        else:
            stall += 1
            lr *= 0.8
            if lr < 1e-10:
                break
            if stall > 200:
                break
    
    return best_a, best_score

def peak_shaving(a_init, time_limit, seed=42):
    """Coordinate descent peak shaving."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init, 0).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 500:
        b = fast_autocorr(a)
        max_k = b.argmax()
        
        # Find the position that most contributes to the peak
        js = max(0, max_k - n + 1)
        je = min(n - 1, max_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[max_k - idx]
        src = idx[np.argmax(contrib)]
        
        # Find a good target position
        # Prefer positions that don't contribute to any top peak
        top_peaks = np.argsort(b)[::-1][:10]
        peak_areas = set()
        for k in top_peaks:
            js2 = max(0, k - n + 1)
            je2 = min(n - 1, k)
            peak_areas.update(range(js2, je2 + 1))
        
        candidates = [j for j in range(n) if j not in peak_areas and j != src]
        if not candidates:
            candidates = [j for j in range(n) if j != src]
        
        # Pick the candidate with lowest a value
        dst = min(candidates, key=lambda j: a[j])
        
        improved = False
        for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.05]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = a_trial / a_trial.sum()
            score = score_from_a(a_trial)
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                stall = 0
                break
        
        if not improved:
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.001, 0.01)
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = score_from_a(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if not improved:
                stall += 1
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

Actually, here's a clever initialization idea: if we want the autocorrelation to be as flat as possible, we want the Fourier transform of a to have a specific shape. Since b = IFFT(FFT(a) * FFT(a)), we want |FFT(a)|^2 to be as flat as possible (for the valid lags). This is related to the "Golay complementary sequence" problem.

For a continuous function, if we want the autocorrelation to be flat, we need |A(f)|^2 = constant (where A is the Fourier transform). This means |A(f)| = constant, i.e., A(f) is a "flat" function in frequency. In the time domain, this corresponds to a Dirac delta (which is not a valid sequence). So we can't achieve perfectly flat autocorrelation with a bounded sequence.

The next best thing is to make |A(f)|^2 as flat as possible over the frequency range that matters. For a sequence of length n, the "effective" frequency range is [0, 1] (in normalized frequency). We want |A(f)|^2 to be approximately constant on this range.

One way to achieve this: use a window function whose Fourier transform is approximately flat. The "flat-top" window has this property! Its Fourier transform is approximately 1 over a certain frequency range and then drops off.

So a good initialization might be a "flat-top" window or a similar window with flat spectral magnitude.

Actually, for the autocorrelation to be flat, we need the squared magnitude of the Fourier transform to be flat. The autocorrelation is the inverse FFT of |FFT(a)|^2. So we want |FFT(a)|^2 to be flat, which means |FFT(a)| to be flat.

For a real, non-negative sequence, FFT(a) is the DTFT. We want |DTFT(a)| to be approximately constant.

A function with flat magnitude spectrum is a "constant amplitude-zero angular frequency" (CAZAF) sequence, but those have constant amplitude in time (not what we want).

What we want is: a non-negative sequence whose Fourier transform has approximately constant magnitude. This is related to the "minimum uncertainty" problem.

For a continuous function on [0,1], if we want |A(f)| = c for all f, then a(x) = c (constant). But that gives a triangular autocorrelation, not flat.

The issue is that for a function on [0,1] (finite support), the autocorrelation has support on [0,1] too (in the continuous case), and we can't make it perfectly flat unless it's zero (which contradicts the integral constraint).

In the discrete case, the autocorrelation has 2n-1 values, and their sum is (sum a)^2. So the "average" value is (sum a)^2 / (2n-1). We want all values to be close to this average.

For large n, the ratio (sum a)^2 / (2n-1) / (sum a)^2 / n = n/(2n-1) ≈ 1/2. So the center lag is about twice the average, and the edge lags are much smaller. We need to "spread out" the autocorrelation.

- b_k(exp(z)) = sum_{i,j} M_{k,ij} exp(z_i) exp(z_j) where M is the convolution matrix
- This is a sum of exponentials of linear functions of z, which is convex in z (since exp is convex and the composition of a convex non-decreasing function with a convex function is convex).

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-stage LogSumExp optimization with L-BFGS (convex in log-space)
2. Active-set subgradient polishing targeting top-K autocorrelation peaks
3. Coordinate-descent peak shaving for final micro-improvements
4. Multi-length exploration (n=1600 parent, n=2000 interpolated)
Key insight: The smoothed objective is convex in z=log(a) space, enabling
fast L-BFGS convergence. Active-set method handles the non-smooth max directly.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_autocorr(a):
    """Compute full autocorrelation a*a using FFT (O(n log n))."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_a(a):
    """Compute score = 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    b = fast_autocorr(a)
    s = np.sum(a)
    if s < 1e-15:
        return float('inf')
    return float(2.0 * n * b.max() / (s * s))


def safe_eval(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def lse_obj_grad(z, T):
    """
    LogSumExp smoothly approximating max_k b_k, with a = exp(z) parameterization.
    The objective is convex in z, enabling fast L-BFGS convergence.
    Returns (objective_value, gradient_vector).
    """
    n = len(z)
    # Numerical stability: shift z so max is 0
    z_max = z.max()
    a = np.exp(z - z_max)

    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    sum_exp = exp_Tb.sum()
    obj = (max_Tb + np.log(sum_exp)) / T

    # Softmax weights over the autocorrelation values
    w = exp_Tb / sum_exp

    # Gradient w.r.t. a: grad_a[i] = 2 * sum_k w[k] * a[k-i] (a zero-padded)
    # Computed via FFT cross-correlation: conv(a[::-1], w)[n-1+i]
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad_a = 2.0 * conv[n - 1:n - 1 + n]

    # Chain rule: grad_z[i] = a[i] * grad_a[i]
    grad_z = a * grad_a

    return obj, grad_z


def lse_multi_stage_optimize(a_init, time_limit, seed=0):
    """
    Multi-stage LogSumExp optimization with decreasing temperature.
    Uses L-BFGS which exploits the convexity of LSE in z-space.
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 1e-30)
    z = np.log(a) - np.log(a).max()  # Shift for stability

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = score_from_a(a)

    # Temperature schedule: start high (smooth), end low (sharp)
    temperatures = [100.0, 40.0, 20.0, 10.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2]

    for ti, T in enumerate(temperatures):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        # Scale iterations: more for early stages (smoother = easier)
        frac = remaining / time_limit
        max_iter = int(300 + 700 * frac)
        max_iter = max(100, min(max_iter, 2000))

        def func(z_in):
            val, _ = lse_obj_grad(z_in, T)
            return val

        def jac(z_in):
            _, g = lse_obj_grad(z_in, T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': max_iter * 100, 'disp': False}
            )
            z_new = res.x
            a_new = np.exp(z_new - z_new.max())
            score_new = score_from_a(a_new)
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
            # Update z (re-normalize for stability)
            z = z_new - z_new.max()
            elapsed = time.monotonic() - t0
            print(f"    LSE T={T:.1f} ({elapsed:.0f}s): score={best_score:.6f}, status={res.status}")
        except Exception as e:
            print(f"    LSE T={T:.1f} failed: {e}")
            # Try with fewer iterations
            try:
                res = minimize(
                    fun=func, x0=z, jac=jac, method='L-BFGS-B',
                    options={'maxiter': 50, 'ftol': 1e-10, 'gtol': 1e-8,
                             'maxfun': 5000, 'disp': False}
                )
                z = res.x - res.x.max()
                a_new = np.exp(res.x - res.x.max())
                score_new = score_from_a(a_new)
                if score_new < best_score:
                    best_score = score_new
                    best_a = a_new.copy()
            except Exception:
                pass

    return best_a, best_score


def active_set_optimize(a_init, time_limit, K=5, seed=0):
    """
    Active-set subgradient method: directly minimizes the max autocorrelation
    by averaging gradients of the top-K active lags and taking projected steps.
    """
    n = len(a_init)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a_init.copy(), score_from_a(a_init)
    a = a / s

    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    # Adaptive learning rate
    lr = 0.5 / n
    stall_count = 0
    best_lr = lr
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if stall_count > 500:
            break

        b = fast_autocorr(a)
        max_b = b.max()

        # Find top-K lags (with some randomization to avoid stagnation)
        if rng.random() < 0.1 and no_improve > 20:
            # Randomly perturb which lags we consider
            indices = np.arange(2 * n - 1)
            top_all = np.argsort(b)[::-1][:max(K * 3, 15)]
            top_k = np.random.choice(top_all, size=K, replace=False)
        else:
            top_k = np.argsort(b)[::-1][:K]

        # Uniform weights on active lags
        w = np.zeros(2 * n - 1)
        for k in top_k:
            w[int(k)] = 1.0 / K

        # Compute gradient: grad_a[i] = 2 * sum_k w[k] * a[k-i]
        conv_len = n + (2 * n - 1) - 1
        fft_size = 1 << (conv_len - 1).bit_length()
        A_rev = rfft(a[::-1].copy(), fft_size)
        W = rfft(w, fft_size)
        conv = irfft(A_rev * W, fft_size)
        grad_a = 2.0 * conv[n - 1:n - 1 + n]

        # Projected gradient step with backtracking
        improved = False
        for lr_try in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 2.0, lr * 4.0]:
            a_trial = a - lr_try * grad_a
            a_trial = np.maximum(a_trial, 0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            a_trial = a_trial / s_trial
            score_trial = score_from_a(a_trial)
            if score_trial < best_score - 1e-15:
                a = a_trial
                best_score = score_trial
                best_a = a_trial.copy()
                lr = lr_try
                best_lr = lr_try
                stall_count = 0
                no_improve = 0
                improved = True
                break
            elif score_trial < score_from_a(a):
                # Accept even if not best (helps with exploration)
                a = a_trial
                lr = lr_try
                stall_count = 0
                improved = True
                break

        if not improved:
            stall_count += 1
            no_improve += 1
            lr *= 0.7
            if lr < 1e-12:
                # Reset with random perturbation
                a_pert = a + rng.normal(0, 1e-4, n)
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
                score_pert = score_from_a(a_pert)
                if score_pert < best_score:
                    a = a_pert
                    best_score = score_pert
                    best_a = a.copy()
                lr = best_lr
                stall_count = 0

        if time.monotonic() - t0 - (t0) > 0 and (int(time.monotonic() - t0) % 60 == 0):
            elapsed = time.monotonic() - t0
            print(f"    AS t={elapsed:.0f}s: score={best_score:.6f}, lr={lr:.2e}")

    return best_a, best_score


def peak_shaving_optimize(a_init, time_limit, seed=42):
    """
    Coordinate-descent peak shaving: directly reduces the top autocorrelation peak
    by moving mass from high-contributing positions to low-contributing ones.
    """
    n = len(a_init)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a_init.copy(), score_from_a(a_init)
    a = a / s

    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 1000:
        b = fast_autocorr(a)

        # Process top few peaks
        top_peaks = np.argsort(b)[::-1][:6]

        improved = False
        for pk in top_peaks:
            js = max(0, int(pk) - n + 1)
            je = min(n - 1, int(pk))
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[int(pk) - idx]
            src = int(idx[np.argmax(contrib)])

            # Find target: position not in any top peak's support
            peak_areas = set()
            for k2 in top_peaks:
                js2 = max(0, int(k2) - n + 1)
                je2 = min(n - 1, int(k2))
                peak_areas.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in peak_areas and j != src]
            if not candidates:
                candidates = [j for j in range(n) if j != src]

            # Sort by current value (move to lowest)
            candidates.sort(key=lambda j: a[j])
            dst_pool = candidates[:max(1, len(candidates) // 4)]
            dst = int(dst_pool[rng.integers(len(dst_pool))])

            for frac in [0.001, 0.003, 0.007, 0.015, 0.025, 0.04, 0.06]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = score_from_a(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            # Random exploration
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-12:
                frac = float(rng.uniform(0.002, 0.02))
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = score_from_a(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            stall += 1

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations for exploration."""
    choice = int(rng.integers(0, 8))
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 1.5
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.6 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        # Flat-top window
        x = np.arange(n) / n
        a = (0.21557895 - 0.41663158 * np.cos(2 * np.pi * x)
             + 0.277263158 * np.cos(4 * np.pi * x)
             - 0.083578947 * np.cos(6 * np.pi * x)
             + 0.006947368 * np.cos(8 * np.pi * x))
    else:
        # Triangular
        x = np.arange(n)
        mid = n / 2
        a = 1 - np.abs(x - mid) / mid
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

    if t6 > 20:
        # Perturb best solution
        a_pert = best_seq + rng.normal(0, 0.01, len(best_seq)) * best_seq.max()
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        a_out, _ = lse_multi_stage_optimize(a_pert, t6, seed=seed + 5)
        try_update(a_out, f"lse_restart_n{n_base}")

    # === Phase 7: Final peak shaving on global best ===
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.20, 150.0)
    print(f"\n=== Phase 7: Final peak shaving (budget={t7:.0f}s) ===")
    if t7 > 5:
        a_out, _ = peak_shaving_optimize(best_seq, t7, seed=seed + 10)
        try_update(a_out, "final_peak")

    # === Phase 8: One more LSE on the best (if time remains) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t8 = min(remaining * 0.30, 100.0)
        print(f"\n=== Phase 8: Final LSE polish (budget={t8:.0f}s) ===")
        if t8 > 10:
            a_out, _ = lse_multi_stage_optimize(best_seq, t8, seed=seed + 20)
            try_update(a_out, "final_lse")

    # Final verification
    if best_seq is not None:
        fs = safe_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
