    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation (scale-invariant)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def compute_lasso_grad(a, T):
    """
    Compute gradient of (2n/T)*LogSumExp(conv(a,a)/T) for normalized a.
    Returns (gradient_array, max_conv_value).
    Uses FFT cross-correlation for O(n log n) gradient.
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    # Softmax weights
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum
    # cross_corr(w, a)[i] = sum_j w[i+j] * a[j]
    # scipy.signal.correlate(w, a, mode='valid') gives exactly this
    grad = _scipy_correlate(w, a, mode='valid', method='fft')
    # Full gradient: (4n/T^2) * grad, but we absorb (4n) into step size
    grad *= 1.0 / (T * T)
    return grad, max_b


def compute_topk_grad(a, K):
    """
    Compute gradient of (2n/K)*sum of top-K peaks of conv(a,a).
    Returns (gradient_array, max_conv_value, topK_values).
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    # Get top K peak indices
    K_eff = min(K, len(b))
    topk_idx = np.argsort(b)[::-1][:K_eff]
    w = np.zeros(len(b))
    w[topk_idx] = 1.0 / K_eff
    grad = _scipy_correlate(w, a, mode='valid', method='fft')
    return grad, max_b, b[topk_idx].max()


def nesterov_lse(a0, time_limit, T_start=None, T_end=0.02, eta0=0.5,
                  eval_interval=10, restart_threshold=80):
    """
    Nesterov-accelerated mirror descent on LogSumExp smoothed objective.
    Work in log-space for natural simplex constraint.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    if T_start is None:
        b = fast_convolve(a)
        T_start = max(0.05, 0.1 * b.max())

    # Work in log-space
    log_a = np.log(a + 1e-30)
    T_cur = T_start
    eta = eta0
    no_improve = 0
    it = 0
    prev_best = best_score
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        # Compute gradient
        grad, max_b = compute_lasso_grad(a, T_cur)

        # Step size: eta / (max|grad| + eps) for normalized step
        g_scale = np.max(np.abs(grad))
        if g_scale < 1e-15:
            g_scale = 1e-15
        step = eta / g_scale

        # Mirror descent in log-space
        log_a = log_a - step * grad
        log_a -= log_a.max()
        a = np.exp(log_a)
        a = a / a.sum()

        it += 1

        # Periodic evaluation
        if it % eval_interval == 0:
            sc = evaluate_local(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            # Temperature decay
            T_cur = max(T_end, T_cur * 0.97)

            # Adaptive eta
            if best_score < prev_best - 1e-9:
                eta = min(eta * 1.05, 3.0)
            elif no_improve > 20:
                eta = max(eta * 0.9, 0.05)
            prev_best = best_score

            # Restart if stuck
            if no_improve >= restart_threshold:
                log_a = np.log(best_a + 1e-30)
                # Perturb
                log_a += 0.05 * rng.standard_normal(n)
                log_a -= log_a.max()
                a = np.exp(log_a)
                a = a / a.sum()
                no_improve = 0
                eta = eta0
                T_cur = max(T_end, 0.05 * max_b)

        # Update eta schedule based on iteration
        if it % 500 == 0:
            eta = eta0 * max(0.3, 1.0 / np.sqrt(1 + it / 5000.0))

    return best_a, best_score


def topk_optimize(a0, time_limit, K=5, eta0=0.3):
    """
    Optimize using top-K peak penalty (sharper than LogSumExp).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()

    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    log_a = np.log(a + 1e-30)
    eta = eta0
    no_improve = 0
    it = 0
    rng = np.random.default_rng(7)

    while time.monotonic() - t0 < time_limit:
        grad, max_b, _ = compute_topk_grad(a, K)
        g_scale = np.max(np.abs(grad))
        if g_scale < 1e-15:
            g_scale = 1e-15
        step = eta / g_scale

        log_a = log_a - step * grad
        log_a -= log_a.max()
        a = np.exp(log_a)
        a = a / a.sum()

        it += 1

        if it % 10 == 0:
            sc = evaluate_local(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
                eta = min(eta * 1.02, 2.0)
            else:
                no_improve += 1
                if no_improve > 30:
                    eta = max(eta * 0.95, 0.05)

            if no_improve >= 100:
                log_a = np.log(best_a + 1e-30)
                log_a += 0.03 * rng.standard_normal(n)
                log_a -= log_a.max()
                a = np.exp(log_a)
                a = a / a.sum()
                no_improve = 0

    return best_a, best_score


def coordinate_descent_peak(a0, time_limit):
    """
    Coordinate descent targeting the peak of the convolution.
    For each peak, identify top contributing coordinates and try to reduce them.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()
        # Find top peaks
        peak_threshold = max_b * 0.999
        peaks = np.where(b >= peak_threshold)[0][:3]

        improved = False
        for pk in peaks:
            if improved:
                break
            # Find contributing coordinates
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            top_m = min(6, len(idx))
            top_pos = idx[np.argsort(contribs)[::-1][:top_m]]

            for pos in top_pos:
                if a[pos] < 1e-12:
                    continue
                # Try reducing this coordinate
                for frac in [0.99, 0.98, 0.95, 0.90]:
                    a_trial = a.copy()
                    reduction = a_trial[pos] * (1.0 - frac)
                    a_trial[pos] *= frac
                    # Redistribute to other coordinates proportionally
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    osum = others.sum()
                    if osum < 1e-15:
                        break
                    a_trial[mask] = others + reduction * others / osum
                    a_trial = a_trial / a_trial.sum()
                    sc = evaluate_local(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 200:
                # Simultaneous reduction of top contributors across all peaks
                peak_k = int(np.argmax(b))
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[peak_k - idx]
                top_positions = idx[np.argsort(contribs)[::-1][:4]]
                a_trial = a.copy()
                for pos in top_positions:
                    if a_trial[pos] > 1e-12:
                        a_trial[pos] *= 0.995
                a_trial = a_trial / a_trial.sum()
                sc = evaluate_local(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial.copy()
                    best_score = sc
                    best_a = a.copy()
                    stall = 0
                elif stall > 1000:
                    break
        else:
            stall = 0

        if (stall + 1) % 5000 == 0:
            print(f"      CD: {best_score:.8f} (stall={stall})")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-3.0 * (x - 0.5) ** 2 / 0.15)
    elif choice == 5:
        # Bimodal
        a = np.exp(-10 * (x - 0.3) ** 2) + np.exp(-10 * (x - 0.7) ** 2)
    elif choice == 6:
        # Triangular with ripple
        a = np.sin(np.pi * x) * (1 + 0.3 * np.cos(2 * np.pi * 5 * x))
        a = np.maximum(a, 0)
    else:
        # Flat with edge concentration
        a = 0.3 + 0.7 * (x ** 0.5 + (1 - x) ** 0.5)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def lp_best_direction(a_fixed, time_per_lp=2.0):
    """
    Solve LP: min t s.t. conv(a_fixed, b)_k <= t, sum(b)=1, b>=0.
    Returns b or None.
    """
    if not HAS_CVXPY:
        return None
    n = len(a_fixed)
    size = 2 * n - 1
    t_start = time.monotonic()

    # Build sparse convolution matrix
    row_ind = []
    col_ind = []
    data = []
    for i in range(n):
        row_ind.extend(range(i, i + n))
        col_ind.extend([i] * n)
        data.extend(a_fixed)

    T_mat = csr_matrix((data, (np.array(row_ind), np.array(col_ind))),
                       shape=(size, n))

    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [T_mat @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)

    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                bs = b_val.sum()
                if bs > 1e-15:
                    b_val = b_val / bs
                    return b_val
                break
        except Exception:
            continue
        if time.monotonic() - t_start > time_per_lp:
            break
    return None


def lp_nesterov_hybrid(a0, time_limit, solver=None):
    """
    Hybrid: alternate LP best-response with Nesterov update.
    Falls back to gradient if LP is too slow.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    y = a.copy()
    stall = 0
    rng = np.random.default_rng(99)
    use_lp = HAS_CVXPY

    it = 0
    while time.monotonic() - t0 < time_limit:
        # Try LP direction
        b_dir = None
        if use_lp and it % 3 == 0:
            b_dir = lp_best_direction(y, time_per_lp=1.5)

        if b_dir is None:
            # Fallback: gradient direction
            T = 0.05
            grad, _ = compute_lasso_grad(y, T)
            # Convert gradient to a direction on the simplex
            b_dir = y - 0.1 * grad / (np.max(np.abs(grad)) + 1e-10)
            b_dir = np.maximum(b_dir, 0)
            bs = b_dir.sum()
            if bs < 1e-15:
                b_dir = y.copy()
            else:
                b_dir = b_dir / bs

        # Nesterov update
        alpha = 2.0 / (np.sqrt(it + 1) + 1.0)
        y = (1.0 - alpha) * y + alpha * b_dir
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        # FTRL-style update for a
        a = ((it + 1) * a + (it + 2) * b_dir) / (2 * it + 3)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
            sa = 1.0
        a = a / sa

        it += 1

        if it % 5 == 0:
            sc_a = evaluate_local(a)
            sc_y = evaluate_local(y)
            if sc_a < best_score:
                best_score = sc_a
                best_a = a.copy()
                stall = 0
            if sc_y < best_score:
                best_score = sc_y
                best_a = y.copy()
                stall = 0

            if stall > 30:
                # Perturb
                y += 0.02 * rng.standard_normal(n)
                y = np.maximum(y, 0)
                y = y / y.sum()
                a = y.copy()
                stall = 0

        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LPHybrid iter={it} ({elapsed:.0f}s): {best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Multi-n LogSumExp warm-start ===
    print(f"\n=== Phase 1: Multi-n LSE Warm-start (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    n_candidates = [len(best_seq)]
    n_base = len(best_seq)
    for n_try in [n_base - 200, n_base + 200, n_base * 2, n_base // 2]:
        if 200 <= n_try <= 4000 and n_try not in n_candidates:
            n_candidates.append(n_try)

    t_per_n = max(20.0, min(rem * 0.20 / len(n_candidates), 120.0))
    print(f"  n candidates: {n_candidates}, ~{t_per_n:.0f}s each")

    candidates = {}
    for n_try in n_candidates:
        if time.monotonic() > deadline - 30:
            break
        a_interp = interpolate_sequence(best_seq, n_try)
        t_sub = min(t_per_n, deadline - time.monotonic() - 20)
        if t_sub < 10:
            continue
        print(f"  n={n_try}, budget={t_sub:.0f}s")
        a_opt, sc = nesterov_lse(a_interp, t_sub, T_start=0.15, T_end=0.02,
                                  eta0=0.5, eval_interval=10, restart_threshold=60)
        candidates[n_try] = (a_opt, sc)
        try_update(a_opt, f"LSE_n{n_try}")

    # Pick best candidate as main working sequence
    if candidates:
        best_n = min(candidates.keys(), key=lambda k: candidates[k][1])
        main_a = candidates[best_n][0].copy()
    else:
        main_a = best_seq.copy()

    # === Phase 2: Intensive Nesterov LSE on best n ===
    print(f"\n=== Phase 2: Intensive Nesterov LSE (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, n={len(main_a)}")
        a_opt, sc = nesterov_lse(main_a, t_alloc, T_start=0.08, T_end=0.01,
                                  eta0=0.8, eval_interval=5, restart_threshold=50)
        try_update(a_opt, "LSE_main")
        main_a = best_seq.copy()

    # === Phase 3: Top-K optimization ===
    print(f"\n=== Phase 3: Top-K Gradient (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        for K in [3, 5, 10]:
            sub_t = t_alloc / 3
            if time.monotonic() > deadline - 10:
                break
            print(f"  K={K}, budget={sub_t:.0f}s")
            a_opt, sc = topk_optimize(main_a, sub_t, K=K, eta0=0.4)
            try_update(a_opt, f"TopK_{K}")
            main_a = best_seq.copy()

    # === Phase 4: LP Hybrid (if available) ===
    if HAS_CVXPY:
        print(f"\n=== Phase 4: LP Hybrid (score={best_score:.8f}) ===")
        rem = deadline - time.monotonic()
        t_alloc = min(rem * 0.20, 250.0)
        if t_alloc > 30:
            print(f"  Budget: {t_alloc:.0f}s")
            a_opt, sc = lp_nesterov_hybrid(main_a, t_alloc)
            try_update(a_opt, "LP_hybrid")
            main_a = best_seq.copy()
    else:
        # Use more LSE time instead
        print(f"\n=== Phase 4: Extended LSE (no cvxpy) (score={best_score:.8f}) ===")
        rem = deadline - time.monotonic()
        t_alloc = min(rem * 0.20, 250.0)
        if t_alloc > 30:
            a_opt, sc = nesterov_lse(main_a, t_alloc, T_start=0.05, T_end=0.005,
                                      eta0=1.0, eval_interval=5, restart_threshold=40)
            try_update(a_opt, "LSE_ext")
            main_a = best_seq.copy()

    # === Phase 5: Coordinate Descent Polish ===
    print(f"\n=== Phase 5: Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_opt, sc = coordinate_descent_peak(best_seq, t_alloc)
        try_update(a_opt, "coord_desc")

    # === Phase 6: Final fine LSE + CD ===
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 10:
        print(f"\n=== Phase 6: Final refinement (score={best_score:.8f}) ===")
        t_lse = t_alloc * 0.5
        t_cd = t_alloc * 0.5
        a_opt, sc = nesterov_lse(best_seq, t_lse, T_start=0.03, T_end=0.005,
                                  eta0=1.5, eval_interval=5, restart_threshold=30)
        try_update(a_opt, "final_LSE")
        if t_cd > 5:
            a_opt, sc = coordinate_descent_peak(best_seq, t_cd)
            try_update(a_opt, "final_CD")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
