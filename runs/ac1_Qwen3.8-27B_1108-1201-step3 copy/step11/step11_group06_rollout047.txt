from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix_sparse(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    rows = np.zeros(n * n, dtype=np.int32)
    cols = np.zeros(n * n, dtype=np.int32)
    vals = np.zeros(n * n, dtype=np.float64)
    count = 0
    for j in range(n):
        for k in range(j, min(j + n, size)):
            rows[count] = k
            cols[count] = j
            vals[count] = a[k - j]
            count += 1
    return csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(size, n))


def build_conv_matrix_active(a, active_indices):
    """Build sparse matrix using only active row indices."""
    n = len(a)
    size = 2 * n - 1
    m = len(active_indices)
    rows = np.zeros(n * m, dtype=np.int32)
    cols = np.zeros(n * m, dtype=np.int32)
    vals = np.zeros(n * m, dtype=np.float64)
    count = 0
    for j in range(n):
        for ii, k in enumerate(active_indices):
            if k >= j and k < j + n:
                rows[count] = ii
                cols[count] = j
                vals[count] = a[k - j]
                count += 1
    return csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(m, n))


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_warm_start(a_fixed, solver, T_mat, warm_b=None, warm_M=None):
    """Solve LP: min M s.t. T@b <= M, sum(b)=1, b>=0. With warm start."""
    n = len(a_fixed)
    m = T_mat.shape[0]
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    # Warm start
    if warm_b is not None:
        b_var.value = warm_b
    if warm_M is not None:
        M_var.value = warm_M
    
    solvers_to_try = [solver, cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]
    solved = False
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue
    if not solved:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(max(0, M_var.value))


def solve_lp_active_set(a_fixed, solver, T_full, K=200):
    """Solve LP using only top-K active constraints for speed."""
    n = len(a_fixed)
    b = fast_convolve(a_fixed)
    # Find active set: positions where b is near max
    max_b = b.max()
    threshold = max_b * (1.0 - 1e-3)
    active = np.where(b >= threshold)[0]
    if len(active) > K:
        # Keep only top K
        top_idx = np.argsort(b[active])[::-1][:K]
        active = active[top_idx]
    if len(active) == 0:
        active = np.array([np.argmax(b)])
    
    T_active = build_conv_matrix_active(a_fixed, active)
    b_val, M_val = solve_lp_warm_start(a_fixed, solver, T_active, warm_b=a_fixed, warm_M=max_b)
    
    if b_val is None:
        return None, None
    
    # Verify solution satisfies all constraints
    b_check = fast_convolve(a_fixed * b_val)
    if b_check.max() > M_val * (1 + 1e-4):
        # Solution violates inactive constraints, fall back to full LP
        b_val, M_val = solve_lp_warm_start(a_fixed, solver, T_full, warm_b=b_val, warm_M=b_check.max())
    
    return b_val, M_val


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


LSE_deadline = [0.0]  # Use list to avoid closure issues


def lse_func(z):
    """LogSumExp objective for L-BFGS."""
    n = len(z)
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    T = 0.01
    scaled = (b - max_b) / T
    scaled_shifted = scaled - scaled.max()
    exp_terms = np.exp(scaled_shifted)
    lse = max_b + T * (scaled.max() + np.log(exp_terms.sum() + 1e-300))
    return 2.0 * n * lse


def lse_grad(z):
    """Gradient of LogSumExp objective."""
    n = len(z)
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    T = 0.01
    scaled = (b - max_b) / T
    scaled_shifted = scaled - scaled.max()
    exp_terms = np.exp(scaled_shifted)
    w = exp_terms / exp_terms.sum()
    
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return 2.0 * n * grad_z


def lse_lbgfs(a0, T_val, maxiter=1000):
    """L-BFGS with LogSumExp at fixed temperature."""
    global LSE_deadline
    n = len(a0)
    z = z_from_a(a0).copy()
    
    def func(z):
        if time.monotonic() > LSE_deadline[0]:
            return 1e15
        a = softmax_from_z(z)
        b = fast_convolve(a)
        max_b = b.max()
        scaled = (b - max_b) / T_val
        scaled_shifted = scaled - scaled.max()
        exp_terms = np.exp(scaled_shifted)
        lse = max_b + T_val * (scaled.max() + np.log(exp_terms.sum() + 1e-300))
        return 2.0 * n * lse
    
    def jac(z):
        if time.monotonic() > LSE_deadline[0]:
            return np.zeros(len(z))
        a = softmax_from_z(z)
        b = fast_convolve(a)
        max_b = b.max()
        scaled = (b - max_b) / T_val
        scaled_shifted = scaled - scaled.max()
        exp_terms = np.exp(scaled_shifted)
        w = exp_terms / exp_terms.sum()
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-18:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        return 2.0 * n * grad_z
    
    try:
        res = minimize(
            fun=func, x0=z, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-25, 'gtol': 1e-18,
                     'maxfun': maxiter * 200, 'disp': False}
        )
        z = res.x
    except Exception:
        pass
    return softmax_from_z(z)


def multiplicative_weights(a0, time_limit, n_steps=500000, eta0=0.05):
    """Fast multiplicative weights with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 5e-5)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        w_active = b[active] / b[active].sum()
        g = np.zeros(n)
        for w, k in zip(w_active, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w * a[k - idx]
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.005, 0.25)
        else:
            eta *= 0.9
            if eta < 1e-9:
                break
    return best_a, best_score


def peak_subgradient(a0, time_limit, n_steps=100000, lr0=0.001):
    """Peak-targeted subgradient: directly reduce the maximum convolution value."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = lr0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        # Use top few peaks for smoother gradient
        top_k = min(10, len(b))
        sorted_idx = np.argsort(b)[::-1][:top_k]
        w = b[sorted_idx]
        w = w / w.sum()
        
        g = np.zeros(n)
        for w_i, k in zip(w, sorted_idx):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w_i * a[k - idx]
        
        # Project gradient onto simplex (maintain sum=1, a>=0)
        g_proj = g - np.mean(g)
        a_new = a - lr * g_proj
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.01, 0.01)
        else:
            lr *= 0.85
            if lr < 1e-12:
                break
    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=500000, use_active_set=True):
    """LP alternating with Nesterov acceleration, warm starts, and active-set LP."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)
    T_mat = build_conv_matrix_sparse(y)
    warm_b = a.copy()
    warm_M = fast_convolve(a).max()
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        if use_active_set:
            b_new, M_b = solve_lp_active_set(y, solver, T_mat, K=300)
        else:
            b_new, M_b = solve_lp_warm_start(y, solver, T_mat, warm_b=warm_b, warm_M=warm_M)
        
        if b_new is None:
            # Try full LP as fallback
            b_new, M_b = solve_lp_warm_start(y, solver, T_mat, warm_b=warm_b, warm_M=fast_convolve(y).max())
            if b_new is None:
                continue
        warm_b = b_new
        warm_M = M_b
        
        alpha = 2.0 / (it + 2)
        y_old = y.copy()
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = y_old
            sy = y_old.sum()
        y = y / sy
        
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        T_mat = build_conv_matrix_sparse(y)
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 4:
                # Restart with perturbation
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 15)
                phase = rng.uniform(0, 2 * np.pi)
                strength = 0.01 + 0.02 * min(stall_count, 5)
                noise = strength * (np.cos(freq * x + phase) + 0.5 * np.sin(2 * freq * x))
                y = np.maximum(y + noise, 0)
                y = y / y.sum()
                T_mat = build_conv_matrix_sparse(y)
                stall_count = 0
            if (it + 1) % 1000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 5 * x + 0.5)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = 1.0 / (1.0 + 3.0 * np.abs(x - 0.5))
    elif choice == 8:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 7 * x) + 0.2 * np.cos(2 * np.pi * 11 * x)
        a = np.maximum(a, 0)
    else:
        a = np.abs(np.sin(np.pi * x * 3)) ** 0.5
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Fast local evaluation (assumes a >= 0, sum > 0)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturb_sequence(a, rng, strength=0.03):
    """Create a perturbed version with structured frequency components."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq1 = rng.integers(1, 10)
    freq2 = rng.integers(2, 15)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    pert = strength * (np.cos(freq1 * x + p1) + 0.5 * np.sin(freq2 * x + p2) + 0.3 * np.cos(3 * freq1 * x))
    a_pert = np.maximum(a + pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses warm-started LP with active set, peak subgradient, multi-length
    exploration, and LSE L-BFGS polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {budget_s:.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
        return sc

    # === Phase 1: Quick MW + Peak Subgradient ===
    print(f"\n=== Phase 1: MW + PeakSG n={len(best_seq)} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.04, 60.0)
    if t1 > 15:
        a_mw, _ = multiplicative_weights(best_seq, t1 * 0.5, n_steps=200000, eta0=0.04)
        try_update(a_mw, "mw_1")
        rem = deadline - time.monotonic()
        if rem > 10:
            a_pg, _ = peak_subgradient(best_seq, min(rem * 0.5, 30.0), n_steps=50000, lr0=0.002)
            try_update(a_pg, "peaksg_1")

    # === Phase 2: Main LP alternating (warm-started, active set) ===
    print(f"\n=== Phase 2: LP n={len(best_seq)} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.4, 700.0)
    if t2 > 60:
        print(f"  Budget: {t2:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t2, solver, n_max_iters=2000000, use_active_set=True)
        try_update(a_lp, "lp_main")

    # === Phase 3: Peak subgradient polish ===
    print(f"\n=== Phase 3: PeakSG polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t3 = min(rem * 0.08, 120.0)
        a_pg, _ = peak_subgradient(best_seq, t3, n_steps=200000, lr0=0.003)
        try_update(a_pg, "peaksg_2")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    n_current = len(best_seq)
    lengths_to_try = [1500, 2000, 2500, 3000]
    for ln in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 120:
            break
        if ln == n_current:
            continue
        if ln > 3000 and rem < 200:
            continue
        t_ln = min(rem * 0.12, 180.0)
        if t_ln < 40:
            continue
        print(f"  Trying n={ln}, budget={t_ln:.0f}s")
        init_ln = interpolate_sequence(best_seq, ln)
        # MW first
        t_mw = t_ln * 0.2
        a_ln_mw, _ = multiplicative_weights(init_ln, t_mw, n_steps=100000, eta0=0.04)
        try_update(a_ln_mw, f"mw_n{ln}")
        # LP
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.4, t_ln * 0.7)
        if t_lp > 30:
            a_ln_lp, _ = nesterov_lp_refine(a_ln_mw, t_lp, solver, n_max_iters=500000, use_active_set=True)
            try_update(a_ln_lp, f"lp_n{ln}")
        # Peak subgradient
        rem = deadline - time.monotonic()
        if rem > 20 and best_seq is not None:
            a_ln_pg, _ = peak_subgradient(best_seq, min(rem * 0.3, 30.0), n_steps=100000, lr0=0.002)
            try_update(a_ln_pg, f"pg_n{ln}")

    # === Phase 5: LSE L-BFGS polish ===
    print(f"\n=== Phase 5: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t5 = min(rem * 0.15, 200.0)
        global LSE_deadline
        LSE_deadline[0] = time.monotonic() + t5
        T_stages = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
        a_cur = best_seq.copy()
        for stage_T in T_stages:
            if time.monotonic() > LSE_deadline[0]:
                break
            remaining = LSE_deadline[0] - time.monotonic()
            if remaining < 5:
                break
            a_cur = lse_lbgfs(a_cur, stage_T, maxiter=300)
            try_update(a_cur, f"lse_T{stage_T}")

    # === Phase 6: Perturbation escape + re-optimize ===
    print(f"\n=== Phase 6: Perturbation escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for trial in range(5):
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            strength = 0.01 + 0.015 * trial
            a_pert = perturb_sequence(best_seq, rng, strength=strength)
            t_sub = min(rem * 0.3, 50.0)
            a_opt, _ = peak_subgradient(a_pert, t_sub * 0.5, n_steps=50000, lr0=0.002)
            try_update(a_opt, f"pert_pg{trial}")
            rem = deadline - time.monotonic()
            if rem > 20:
                a_opt2, _ = nesterov_lp_refine(a_opt, min(rem * 0.4, 30.0), solver, n_max_iters=100000, use_active_set=True)
                try_update(a_opt2, f"pert_lp{trial}")

    # === Phase 7: Final combined polish ===
    print(f"\n=== Phase 7: Final polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        # Peak subgradient
        t7a = min(rem * 0.3, 40.0)
        a_pg, _ = peak_subgradient(best_seq, t7a, n_steps=200000, lr0=0.001)
        try_update(a_pg, "final_pg")
        rem = deadline - time.monotonic()
        # LP
        if rem > 20:
            t7b = min(rem * 0.5, 60.0)
            a_fin_lp, _ = nesterov_lp_refine(best_seq, t7b, solver, n_max_iters=200000, use_active_set=True)
            try_update(a_fin_lp, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
