def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix_sparse(a):
    n = len(a)
    size = 2 * n - 1
    col_lengths = np.minimum(n, size - np.arange(n))
    total = int(col_lengths.sum())
    rows = np.zeros(total, dtype=np.int32)
    cols = np.zeros(total, dtype=np.int32)
    vals = np.zeros(total, dtype=np.float64)
    pos = 0
    for j in range(n):
        length = int(col_lengths[j])
        i = pos
        rows[i:i+length] = np.arange(j, j + length, dtype=np.int32)
        cols[i:i+length] = j
        vals[i:i+length] = a[:length]
        pos += length
    return csr_matrix((vals, (rows, cols)), shape=(size, n))


def get_lp_solver():
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:

            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_sota_lp(a_fixed, M, T_mat, solver):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    solvers_to_try = [solver, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        try:
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                sb = b_val.sum()
                if sb < 1e-10:
                    return None
                return b_val
        except Exception:
            continue
    return None

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def _lse_obj_grad(z, n, T_temp):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T_temp
    scaled_shifted = scaled - scaled.max()
    exp_terms = np.exp(scaled_shifted)
    lse = max_b + T_temp * (scaled.max() + np.log(exp_terms.sum() + 1e-300))
    obj = 2.0 * n * lse
    w = exp_terms / (exp_terms.sum() + 1e-300)
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z

            if time.monotonic() > deadline:
                return np.zeros(len(z_in))
            _, g = _lse_obj_grad(z_in, n, stage_T)
            return g

        try:
            maxiter = min(800, int(remaining * 50))
            res = minimize(
                fun=_func, x0=z, jac=_jac, method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-30,
                         'gtol': 1e-20, 'maxfun': maxiter * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def multiplicative_weights(a0, time_limit, eta0=0.05):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = eta0

    for step in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 3e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        w_active = b[active] / b[active].sum()
        g = np.zeros(n)
        for w, k in zip(w_active, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w * a[k - idx]
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        score_new = evaluate_local(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.02, 0.4)
        else:
            eta *= 0.9
            if eta < 1e-9:
                break
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturb_sequence(a, rng, strength=0.02):
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq1 = rng.integers(1, 10)
    freq2 = rng.integers(2, 15)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    pert = strength * (np.cos(freq1 * x + p1) + 0.5 * np.sin(freq2 * x + p2))
    a_pert = np.maximum(a + pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {budget_s:.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = np.ones(n_init) / n_init
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Fast MW warmup ===
    print(f"\n=== Phase 1: MW warmup (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.04, 50.0)
    if t1 > 10:
        a_mw, _ = multiplicative_weights(best_seq, t1, eta0=0.08)
        try_update(a_mw, "mw_1")

    # === Phase 2: SOTA LP alternating ===
    print(f"\n=== Phase 2: SOTA LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.25, 400.0)
    if t2 > 60:
        print(f"  Budget: {t2:.0f}s")
        a_sota, _ = sota_lp_alternate(best_seq, t2, solver)
        try_update(a_sota, "sota_lp")

    # === Phase 3: Coordinate descent ===
    print(f"\n=== Phase 3: Coordinate Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 200.0)
    if t3 > 20:
        print(f"  Budget: {t3:.0f}s")
        a_cd, _ = coordinate_descent(best_seq, t3)
        try_update(a_cd, "cd_1")

    # === Phase 4: Nesterov LP min-peak ===
    print(f"\n=== Phase 4: Nesterov LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.22, 350.0)
    if t4 > 60:
        print(f"  Budget: {t4:.0f}s")
        a_nlp, _ = nesterov_lp_minpeak(best_seq, t4, solver)
        try_update(a_nlp, "nlp")
        # Follow up with CD
        rem = deadline - time.monotonic()
        if rem > 30:
            t4b = min(rem * 0.1, 60.0)
            a_cd2, _ = coordinate_descent(a_nlp, t4b)
            try_update(a_cd2, "cd_2")

    # === Phase 5: LSE L-BFGS fine polish ===
    print(f"\n=== Phase 5: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t5 = min(rem * 0.18, 250.0)
        T_stages = [5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        print(f"  {len(T_stages)} stages, budget={t5:.0f}s")
        a_lse = lse_lbgfs(best_seq, t5, T_stages)
        try_update(a_lse, "lse")
        # CD after LSE
        rem = deadline - time.monotonic()
        if rem > 20:
            t5b = min(rem * 0.15, 60.0)
            a_cd3, _ = coordinate_descent(a_lse, t5b)
            try_update(a_cd3, "cd_3")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length (score={best_score:.8f}) ===")
    n_current = len(best_seq)
    lengths_to_try = [int(n_current * 0.8), int(n_current * 1.2), int(n_current * 1.5), 2000, 3000]
    lengths_to_try = sorted(set(l for l in lengths_to_try if l != n_current and 500 <= l <= 5000))
    for ln in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_ln = min(rem * 0.18, 150.0)
        if t_ln < 40:
            continue
        print(f"  Trying n={ln}, budget={t_ln:.0f}s")
        init_ln = interpolate_sequence(best_seq, ln)
        # MW first
        t_mw = t_ln * 0.15
        a_ln_mw, _ = multiplicative_weights(init_ln, t_mw, eta0=0.05)
        try_update(a_ln_mw, f"mw_n{ln}")
        # LP
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.3, t_ln * 0.5)
        if t_lp > 30:
            a_ln_lp, _ = sota_lp_alternate(a_ln_mw, t_lp, solver)
            try_update(a_ln_lp, f"lp_n{ln}")
            # CD
            rem = deadline - time.monotonic()
            if rem > 20:
                t_cd = min(rem * 0.2, 40.0)
                a_ln_cd, _ = coordinate_descent(a_ln_lp, t_cd)
                try_update(a_ln_cd, f"cd_n{ln}")

    # === Phase 7: Perturbation + re-optimize ===
    print(f"\n=== Phase 7: Perturbation escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        for trial in range(3):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            a_pert = perturb_sequence(best_seq, rng, strength=0.01 + 0.01 * trial)
            t_sub = min(rem * 0.2, 80.0)
            a_opt, _ = sota_lp_alternate(a_pert, t_sub, solver)
            try_update(a_opt, f"pert{trial}")
            rem = deadline - time.monotonic()
            if rem > 15:
                a_cd_p, _ = coordinate_descent(a_opt, min(rem * 0.3, 30.0))
                try_update(a_cd_p, f"cd_pert{trial}")

    # === Phase 8: Final CD polish ===
    print(f"\n=== Phase 8: Final CD polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t8 = min(rem * 0.4, 120.0)
        a_fin_cd, _ = coordinate_descent(best_seq, t8)
        try_update(a_fin_cd, "final_cd")

    # === Phase 9: Final LP polish ===
    print(f"\n=== Phase 9: Final LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t9 = min(rem * 0.5, 80.0)
        a_fin_lp, _ = sota_lp_alternate(best_seq, t9, solver)
        try_update(a_fin_lp, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
