1. Fast LP alternating with scipy HiGHS solver (main workhorse)
2. Multi-n exploration to find optimal dimension

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix, hstack


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_local(a):
    """Evaluate score using local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_T_data(a):
    """Build arrays for sparse convolution matrix construction."""
    n = len(a)
    size = 2 * n - 1
    row_ind = np.repeat(np.arange(n, dtype=np.int32), n) + np.tile(np.arange(n, dtype=np.int32), n)
    col_ind = np.tile(np.arange(n, dtype=np.int32), n)
    data = np.repeat(a.astype(np.float64), n)
    return row_ind, col_ind, data, size


def solve_lp_higs(y):
    """Solve LP best response using scipy HiGHS. Returns normalized b or None."""
    n = len(y)
    size = 2 * n - 1
    row_ind, col_ind, data, _ = build_T_data(y)
    T = csr_matrix((data, (row_ind, col_ind)), shape=(size, n))
    # A_ub @ [b; t] <= 0  =>  T @ b - t <= 0
    neg_I = csr_matrix(np.eye(size, 1) * (-1.0))
    A_ub = hstack([T, neg_I])
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    bounds = [(0, None)] * n + [(-np.inf, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=np.zeros(size),
                      A_eq=A_eq, b_eq=np.array([1.0]),
                      bounds=bounds, method='highs')
        if res.success:
            b_val = np.maximum(res.x[:n], 0)
            s = b_val.sum()
            if s > 1e-15:
                return b_val / s
    except Exception:
        pass
    return None


_lp_solver = None


def get_cvxpy_solver():

        return _lp_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            p = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            p.solve(solver=s)
            if p.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = s
                return s
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_cvxpy(y, solver):
    """Solve LP best response using cvxpy."""
    n = len(y)
    size = 2 * n - 1
    row_ind, col_ind, data, _ = build_T_data(y)
    T = csr_matrix((data, (row_ind, col_ind)), shape=(size, n))
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [T @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            for alt in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
                if alt == solver:
                    continue
                try:
                    prob.solve(solver=alt)
                    if prob.status in ('optimal', 'optimal_inaccurate'):
                        break
                except Exception:
                    continue
    except Exception:
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

    best_a = a.copy()
    best_score = score_local(a)

    stall = 0
    prev_best = best_score
    rng = np.random.default_rng(99)

            a = y.copy()
            sa = 1.0
        a = a / sa

        sc_a = score_local(a)
        sc_y = score_local(y)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            stall = 0
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stall = 0

        if (it + 1) % 50 == 0:
            if best_score > prev_best - 1e-12:
                stall += 1
            else:
                stall = 0
            prev_best = best_score

            if stall >= 10:
                b_cur = fast_convolve(y)
                peak_idx = int(np.argmax(b_cur))
                js = max(0, peak_idx - n + 1)
                je = min(n - 1, peak_idx)
                idx = np.arange(js, je + 1)
                if len(idx) > 0:
                    contribs = y[idx] * y[peak_idx - idx]
                    n_red = max(2, len(idx) // 15)
                    top_pos = idx[np.argsort(contribs)[::-1][:n_red]]
                    for pos in top_pos:
                        if y[pos] > 1e-10:
                            y[pos] *= 0.95
                noise = rng.standard_normal(n) * 0.005
                y = np.maximum(y + noise, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                    a = y.copy()
                stall = 0

            if verbose and (it + 1) % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter={it+1} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def logsumexp_obj_grad(a, b, T):
    """Compute LogSumExp smoothed objective and gradient."""
    n = len(a)
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T)
    w = exp_b / exp_b.sum()
    # c[i] = sum_j w[i+j] * a[j] for i=0,...,n-1
    c = np.correlate(w, a, mode='valid')
    c = c[:n]
    grad = (4.0 * n / T) * c
    obj = (2.0 * n / T) * (max_b / T + np.log(exp_b.sum()))
    return obj, grad


def logsumexp_lbgfs(a0, time_limit, verbose=True):
    """L-BFGS on LogSumExp smoothed objective with multi-temperature schedule."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = score_local(a)

    T_stages = [30.0, 15.0, 8.0, 4.0, 2.0, 1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015]
    n_stages = len(T_stages)
    iters_per = max(50, 3000 // n_stages)

    for si, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit - 3:
            break

        state = {'count': 0}

        def func(z):
            state['count'] += 1
            if time.monotonic() - t0 > time_limit - 1:
                return 1e18
            at = np.maximum(z, 0)
            st_val = at.sum()
            if st_val < 1e-15:
                return 1e18
            at = at / st_val
            b = fast_convolve(at)
            obj, _ = logsumexp_obj_grad(at, b, T)
            sc = score_local(at)
            if sc < best_score:
                best_score = sc
                best_a = at.copy()
            return obj

        def jac(z):
            at = np.maximum(z, 0)
            st_val = at.sum()
            if st_val < 1e-15:
                return np.zeros_like(z)
            at = at / st_val
            b = fast_convolve(at)
            _, grad = logsumexp_obj_grad(at, b, T)
            return grad

        try:
            res = minimize(fun=func, x0=a, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters_per, 'ftol': 1e-16,
                                    'gtol': 1e-14, 'maxfun': iters_per * 100})
            a = np.maximum(res.x, 0)
            sa = a.sum()
            if sa > 1e-15:
                a = a / sa
        except Exception:
            pass

        sc = score_local(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE T={T:.4f} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=True):
    """Aggressive coordinate descent with line search on peak contributors."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_local(a)
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(7)

    for step in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 1e-5)
        peaks = np.where(b >= threshold)[0]
        if len(peaks) == 0:
            peaks = np.array([int(np.argmax(b))])

        improved = False
        for pk in peaks[:2]:
            if improved:
                break
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contribs = a[idx] * a[pk - idx]
            top_m = min(6, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

            for pos in top_positions:
                if a[pos] < 1e-14:
                    continue
                for factor in [0.9995, 0.999, 0.998, 0.995, 0.99, 0.98, 0.96, 0.93]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-16:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    sc = score_local(a_trial)
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 300:
                b = fast_convolve(a)
                pk = int(np.argmax(b))
                js = max(0, pk - n + 1)
                je = min(n - 1, pk)
                idx = np.arange(js, je + 1)
                if len(idx) > 0:
                    contribs = a[idx] * a[pk - idx]
                    top_positions = idx[np.argsort(contribs)[::-1][:3]]
                    a_trial = a.copy()
                    for pos in top_positions:
                        if a_trial[pos] > 1e-14:
                            a_trial[pos] *= 0.997
                    a_trial = np.maximum(a_trial, 0)
                    st_val = a_trial.sum()
                    if st_val > 1e-15:
                        a_trial = a_trial / st_val
                        sc = score_local(a_trial)
                        if sc < best_score - 1e-15:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            stall = 0
                        elif stall > 1500:
                            noise = rng.standard_normal(n) * 0.002
                            a = np.maximum(a + noise, 0)
                            a = a / a.sum()
                            stall = 0
        else:
            stall = 0

        if verbose and (step + 1) % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc {elapsed:.0f}s: score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Linear interpolation of sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses HiGHS LP solver for fast best-response computation, with multi-n
    exploration, deep LP refinement, LogSumExp L-BFGS, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = score_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = score_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = score_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Quick LP warm start (80s) ===
    print(f"\n=== Phase 1: LP Warm Start (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 80.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = lp_alternate(best_seq, t_alloc, use_higs=True)
        try_update(a_lp, "LP_p1")

    # === Phase 2: Multi-n exploration (200s) ===
    print(f"\n=== Phase 2: Multi-n Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 220.0)
    if t_alloc > 30:
        n_best = len(best_seq)
        candidate_ns = sorted(set([
            max(400, n_best - 300), max(500, n_best - 150),
            max(600, n_best + 150), max(800, n_best + 300),
            max(400, n_best // 2), min(4000, n_best * 2)
        ]))
        per_n = t_alloc / max(1, len(candidate_ns))
        print(f"  n values: {candidate_ns}, ~{per_n:.0f}s each")
        for n_try in candidate_ns:
            if time.monotonic() > deadline - 20:
                break
            if n_try == n_best:
                continue
            t_sub = min(per_n, deadline - time.monotonic() - 10)
            if t_sub < 8:
                continue
            a_interp = interpolate_sequence(best_seq, n_try)
            print(f"  n={n_try}, budget={t_sub:.0f}s")
            a_sub, _ = lp_alternate(a_interp, t_sub, use_higs=True, verbose=False)
            try_update(a_sub, f"multi_n_{n_try}")

    # === Phase 3: Deep LP refinement (700s) ===
    print(f"\n=== Phase 3: Deep LP (score={best_score:.8f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.40, 700.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_deep, _ = lp_alternate(best_seq, t_alloc, use_higs=True)
        try_update(a_deep, "deep_lp")

    # === Phase 4: LogSumExp L-BFGS (400s) ===
    print(f"\n=== Phase 4: LogSumExp L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lb, _ = logsumexp_lbgfs(best_seq, t_alloc)
        try_update(a_lb, "lbgfs")

    # === Phase 5: Coordinate descent (250s) ===
    print(f"\n=== Phase 5: Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 10:
        a_cd, _ = coordinate_descent(best_seq, t_alloc)
        try_update(a_cd, "coord")

    # === Phase 6: Final LP polish (remaining time) ===
    print(f"\n=== Phase 6: Final LP Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 5:
        a_fin, _ = lp_alternate(best_seq, t_alloc, use_higs=True, verbose=False)
        try_update(a_fin, "final_lp")

    # === Phase 7: Final coord descent polish ===
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 40.0)
    if t_alloc > 5:
        a_cd2, _ = coordinate_descent(best_seq, t_alloc, verbose=False)
        try_update(a_cd2, "final_coord")

    # Final verification
    if best_seq is not None:
        fs = score_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
