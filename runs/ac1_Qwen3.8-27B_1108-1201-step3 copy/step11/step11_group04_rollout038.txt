import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring fastest ones."""
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]
    solved = False
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue

    if not solved:
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = M_var.value
    if M_val is None:
        M_val = float(fast_convolve(a_fixed).max())
    return b_val, float(max(0, M_val))


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lbgfs_obj(z, n, T_stage, deadline):
    """Objective for L-BFGS."""
    if time.monotonic() > deadline:
        return 1e15
    obj, _ = logsumexp_obj_grad(z, n, T_stage)
    return obj


def lbgfs_jac(z, n, T_stage, deadline):
    """Gradient for L-BFGS."""
    if time.monotonic() > deadline:
        return np.zeros_like(z)
    _, g = logsumexp_obj_grad(z, n, T_stage)
    return g


def lbgfs_refine(a0, time_limit, T_stages, deadline=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit
    n_stages = len(T_stages)

    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break

        stages_left = n_stages - stage_i
        avail_time = remaining
        iters = max(30, int(avail_time * 80 / max(1, stages_left)))
        iters = min(iters, 200000)

        def func(z_in):
            return lbgfs_obj(z_in, n, stage_T, deadline)

        def jac(z_in):
            return lbgfs_jac(z_in, n, stage_T, deadline)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-30,
                         'gtol': 1e-20, 'maxfun': iters * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def spectral_flatten(a, n_iters=50, blend=0.4, deadline=None):
    """Iterative spectral flattening (Gerchberg-Saxton style)."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)

        mags = np.abs(F)
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])

        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

    return a

            break

                break

        # Nesterov dual averaging
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new

        else:
            y = y / sy

        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

            best_a = y.copy()

        # Stall detection with diverse restarts

                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

                    y = y + shift
                elif choice == 2:

                else:

                    y = y + 0.02 * rng.standard_normal(n)

            if (it + 1) % 2000 == 0:

def subgradient_polyak(a0, time_limit, n_steps=500000):
    """
    Direct subgradient method on the true max objective with Polyak averaging.
    This avoids the smoothing bias of LogSumExp.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    # Polyak averaging
    a_avg = np.zeros(n)
    eta = 0.1
    no_improve = 0
    scale = 1.0
    step = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        k_star = np.argmax(b)

        # Subgradient of max(conv(a,a)) at a: 2 * T_k* a
        # T_k* a has (T_k* a)[i] = a[k*-i] if 0<=k*-i<n else 0
        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for i in range(js, je + 1):
            sg[i] = 2.0 * a[k_star - i]

        # Projected subgradient step
        eta_eff = eta * scale / (1.0 + 0.01 * step)
        a_new = a - eta_eff * sg
        # Project onto simplex {a >= 0, sum a = 1}
        a_new = _project_simplex(a_new)

        a_avg = (step * a_avg + a_new) / (step + 1)

        score = 2.0 * n * fast_convolve(a_new).max()
        score_avg = 2.0 * n * fast_convolve(a_avg).max()

        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            a = a_new
            no_improve += 1

        if score_avg < best_score - 1e-15:
            best_score = score_avg
            best_a = a_avg.copy()
            no_improve = 0

        if no_improve >= 50:
            scale *= 0.8
            no_improve = 0
            if scale < 0.01:
                break

        if step % 50000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad+Polyak step={step} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def _project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def peak_shave(a, time_limit, n_steps=200000):
    """Directly target the maximum of the autocorrelation using multiplicative updates."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return None, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.03
    scale = 1.0
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Find top peaks
        threshold = max_b * (1.0 - 5e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        eta_eff = eta * scale
        log_scale = -eta_eff * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            a = a_new
            no_improve += 1

        if no_improve >= 40:
            scale *= 0.75
            no_improve = 0
            if scale < 0.01:
                break

        if step % 50000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave {elapsed:.0f}s: score={best_score:.8f} scale={scale:.4f}")

    return best_a, best_score


def asymmetric_init(n, rng):
    """Create deliberately asymmetric initial sequences."""
    x = np.linspace(0, 1, n)
    choice = rng.integers(0, 10)

    if choice == 0:
        # Exponential ramp (mass concentrated at one end)
        a = np.exp(-3.0 * x)
    elif choice == 1:
        # Power law ramp
        a = (1 - x) ** 2 + 0.05
    elif choice == 2:
        # Two-hump asymmetric
        a = np.exp(-20 * (x - 0.3) ** 2) + 0.5 * np.exp(-20 * (x - 0.8) ** 2)
    elif choice == 3:
        # Cosine window (natural asymmetry)
        a = np.cos(np.pi * x / 2) ** 1.5 + 0.1
    elif choice == 4:
        # Sinc-like (centered but with sidelobes)
        center = n // 2
        t = (np.arange(n) - center) / (n * 0.3)
        a = np.sinc(t) ** 2
        a = np.maximum(a, 0)
    elif choice == 5:
        # Triangle shifted left
        a = np.maximum(0, 1 - np.abs(x - 0.3))
    elif choice == 6:
        # Quadratic with peak at 0.25
        a = np.exp(-8 * (x - 0.25) ** 2) + 0.3 * np.exp(-8 * (x - 0.75) ** 2)
    elif choice == 7:
        # Smooth step (S-curve)
        a = 1.0 / (1.0 + np.exp(-10 * (x - 0.5)))
    elif choice == 8:
        # Bump at edge
        a = np.exp(-5 * x) + 0.2 * np.exp(-5 * (x - 1))
    else:
        # Uniform (baseline)
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturb_sequence(a, rng, strength=0.03):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 8)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Key strategies:
    - Asymmetric initialization exploiting conv != autocorr for non-palindromic sequences
    - LP alternating with Nesterov (primary optimizer)
    - Subgradient with Polyak averaging (bias-free)
    - Ultra-fine L-BFGS LogSumExp (final polish)
    - Multi-n exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 30.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = asymmetric_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    rem = deadline - time.monotonic()
    print(f"Total budget: {rem:.0f}s, starting score: {best_score:.8f}")

    # === Phase 1: Quick diversification ===
    print(f"\n=== Phase 1: Diversification ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_div = min(rem * 0.015, 20.0)
        # Spectral flatten
        a_sp = spectral_flatten(best_seq, n_iters=30, blend=0.3,
                                deadline=time.monotonic() + t_div)
        local_update(a_sp, "spectral_init")
        # Perturbed
        for strength in [0.02, 0.05, 0.1]:
            a_p = perturb_sequence(best_seq, rng, strength)
            local_update(a_p, f"pert_{strength}")

    # === Phase 2: Main LP Alternating ===
    print(f"\n=== Phase 2: Nesterov LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lp = min(rem * 0.35, 600.0)
    if t_lp > 30:
        print(f"  Budget: {t_lp:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_lp, solver, n_max_iters=500000)
        local_update(a_lp, "nesterov_lp_1")

    # === Phase 3: Subgradient + Polyak Averaging ===
    print(f"\n=== Phase 3: Subgradient+Polyak (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_sg = min(rem * 0.15, 200.0)
    if t_sg > 20:
        print(f"  Budget: {t_sg:.0f}s")
        a_sg, _ = subgradient_polyak(best_seq, t_sg, n_steps=200000)
        local_update(a_sg, "subgrad_polyak")

    # === Phase 4: Ultra-fine LogSumExp L-BFGS ===
    print(f"\n=== Phase 4: L-BFGS LogSumExp (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lbfgs = min(rem * 0.25, 400.0)
    if t_lbfgs > 30:
        T_stages = [5.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1,
                    0.05, 0.03, 0.02, 0.01, 0.008, 0.005,
                    0.003, 0.002, 0.001, 0.0008, 0.0005]
        print(f"  T_stages: {len(T_stages)} stages, budget={t_lbfgs:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_lbfgs, T_stages,
                            deadline=time.monotonic() + t_lbfgs)
        local_update(a_lb, "lbgfs_fine")

    # === Phase 5: Peak Shaving ===
    print(f"\n=== Phase 5: Peak Shaving (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_peak = min(rem * 0.08, 100.0)
    if t_peak > 10:
        a_pk, _ = peak_shave(best_seq, t_peak, n_steps=300000)
        local_update(a_pk, "peak_shave")

    # === Phase 6: Second LP round on improved sequence ===
    print(f"\n=== Phase 6: Nesterov LP Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lp2 = min(rem * 0.15, 250.0)
    if t_lp2 > 20:
        print(f"  Budget: {t_lp2:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_lp2, solver, n_max_iters=300000)
        local_update(a_lp2, "nesterov_lp_2")

    # === Phase 7: Try different n values ===
    print(f"\n=== Phase 7: Different n (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    n_candidates = [800, 1000, 1500, 2000]
    n_candidates = [n for n in n_candidates if n != n_best]

    for n_try in n_candidates:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_n = min(rem * 0.15, 120.0)
        if t_n < 30:
            break
        print(f"  Trying n={n_try}, budget={t_n:.0f}s")
        init_n = interpolate_sequence(best_seq, n_try)
        # Quick spectral
        a_sp_n = spectral_flatten(init_n, n_iters=20, blend=0.3)
        # LP
        a_lp_n, _ = nesterov_lp_refine(a_sp_n, t_n * 0.7, solver, n_max_iters=50000)
        local_update(a_lp_n, f"lp_n{n_try}")
        # Quick L-BFGS if improved
        rem = deadline - time.monotonic()
        if best_seq is not None and len(best_seq) == n_try and rem > 30:
            T_st2 = [3.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
            a_lb_n = lbgfs_refine(best_seq, min(rem * 0.3, 60.0), T_st2,
                                  deadline=time.monotonic() + min(rem * 0.3, 60.0))
            local_update(a_lb_n, f"lbgfs_n{n_try}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.35, 60.0)
        if t_fin > 10:
            T_stages_fin = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001, 0.0005]
            a_fin = lbgfs_refine(best_seq, t_fin, T_stages_fin,
                                 deadline=time.monotonic() + t_fin)
            local_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_ps = min(rem * 0.3, 30.0)
        if t_ps > 5:
            a_ps, _ = peak_shave(best_seq, t_ps, n_steps=100000)
            local_update(a_ps, "final_peak")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_flp = min(rem * 0.5, 30.0)
        if t_flp > 5:
            a_flp, _ = nesterov_lp_refine(best_seq, t_flp, solver, n_max_iters=50000)
            local_update(a_flp, "final_lp")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_fsg = min(rem * 0.5, 20.0)
        if t_fsg > 5:
            a_fsg, _ = subgradient_polyak(best_seq, t_fsg, n_steps=50000)
            local_update(a_fsg, "final_subgrad")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}) at {elapsed:.1f}s")
        test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            eval_score = evaluate_sequence(test_seq)
            print(f"Verified with evaluate_sequence: {eval_score:.8f}")
        except Exception as e:
            print(f"Verification error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
