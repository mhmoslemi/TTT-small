    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = inf_score(a)
    t0 = time.monotonic()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Soft subgradient using softmax of peaks
        max_b = b.max()
        exp_b = np.exp((b - max_b) / T_md)
        w = exp_b / exp_b.sum()

        # grad_i = sum_k w_k * a[k-i] via FFT
        size_b = 2 * n - 1
        fft_size = 1 << (size_b + n - 2).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        corr_full = irfft(W * np.conj(A), fft_size)
        grad = corr_full[:n]

        # Multiplicative weights update with adaptive step
        lr = eta0 / (1.0 + 0.01 * np.sqrt(it + 1.0))
        grad_shifted = grad - grad.min()
        a_new = a * np.exp(-lr * grad_shifted)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-300:
            a_new = a.copy()
            a_new_sum = 1.0
        a_new = a_new / a_new_sum
        a = a_new

        if (it + 1) % 1000 == 0:
            sc = inf_score(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 20:
                # Reset temperature and adapt
                T_md = max(0.1, T_md * 0.5)
                eta0 = max(0.01, eta0 * 0.7)
                no_improve = 0
            if (it + 1) % 10000 == 0:
                print(f"    MD iter={it+1} ({time.monotonic()-t0:.1f}s): best={best_score:.8f}")

    # Final check
    sc = inf_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a


def lp_best_response(a_fixed, solver):
    """Given fixed a (sum=1, nonneg), find b>=0, sum(b)=1 minimizing max conv(a,b)."""
    n = len(a_fixed)
    size = 2 * n - 1

    # Build sparse matrix: row r, col c has entry a_fixed[c] if 0 <= r-c < n
    # T @ b = conv(a_fixed, b)
    row_ind = []
    col_ind = []
    data = []
    for i in range(n):
        rows = np.arange(i, i + n)
        row_ind.extend(rows.tolist())
        col_ind.extend([i] * n)
        data.extend(a_fixed.tolist())

    T_sparse = csr_matrix((data, (np.array(row_ind, dtype=np.int32),
                                   np.array(col_ind, dtype=np.int32))),
                          shape=(size, n))

    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [T_sparse @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)

    try:
        prob.solve(solver=solver)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            for alt in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
                if alt == solver:
                    continue
                try:
                    prob.solve(solver=alt)
                    if prob.status in ('optimal', 'optimal_inaccurate'):
                        break
                except Exception:
                    continue
    except Exception:
        return None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def nesterov_lp_refine(a0, time_limit, solver, max_iters=50000):
    """Nesterov-accelerated LP alternating method."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()
    y = a.copy()
    best_a = a.copy()
    best_score = inf_score(a)
    t0 = time.monotonic()
    stall = 0
    stall_threshold = 15
    rng = np.random.default_rng(777)
    count_skipped = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new = lp_best_response(y, solver)
        if b_new is None:
            count_skipped += 1
            if count_skipped > 10:
                break
            continue
        count_skipped = 0

        # Nesterov step
        alpha = 2.0 / (np.sqrt(float(it + 1)) + 2.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        # Weighted average (FTRL-style)
        weight_old = it + 1
        weight_new = alpha * (it + 2)
        a = (weight_old * a + weight_new * b_new) / (weight_old + weight_new)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
            sa = 1.0
        a = a / sa

        sc_a = inf_score(a)
        sc_y = inf_score(y)
        if sc_a < best_score - 1e-14:
            best_score = sc_a
            best_a = a.copy()
            stall = 0
        elif sc_y < best_score - 1e-14:
            best_score = sc_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        # Structured exploration on stall
        if stall >= stall_threshold:
            b_cur = fast_convolve(y)
            peak_idx = int(np.argmax(b_cur))
            js = max(0, peak_idx - n + 1)
            je = min(n - 1, peak_idx)
            idx = np.arange(js, je + 1)
            contribs = y[idx] * y[peak_idx - idx]

            # Reduce top contributors
            n_reduce = max(5, len(idx) // 15)
            top_pos = idx[np.argsort(contribs)[::-1][:n_reduce]]
            for pos in top_pos:
                if y[pos] > 1e-10:
                    y[pos] *= 0.95

            # Add random perturbation
            noise = rng.standard_normal(n) * 0.015
            y = y + noise
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy
            a = y.copy()
            stall = 0
            stall_threshold = min(30, stall_threshold + 2)

        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NL iter={it+1} ({elapsed:.1f}s): best={best_score:.8f}")

    return best_a


def coordinate_descent(a0, time_limit, max_steps=200000):
    """Targeted coordinate descent on peak-contributing elements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = inf_score(a)
    t0 = time.monotonic()
    stall = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.9995
        peaks = np.where(b >= threshold)[0]
        if len(peaks) == 0:
            peaks = np.array([int(np.argmax(b))])

        improved = False
        for peak_k in peaks[:3]:
            if improved:
                break
            js = max(0, int(peak_k) - n + 1)
            je = min(n - 1, int(peak_k))
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[int(peak_k) - idx]
            top_m = min(10, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

            for pos in top_positions:
                if a[pos] < 1e-12:
                    continue
                for factor in [0.9995, 0.999, 0.998, 0.995, 0.99, 0.98]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-16:
                        break
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = a_trial[mask] + diff * a_trial[mask] / others_sum
                    a_trial = a_trial / a_trial.sum()
                    sc = inf_score(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 200:
                # Simultaneous reduction
                peak_k = int(np.argmax(b))
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[peak_k - idx]
                top_pos = idx[np.argsort(contribs)[::-1][:5]]
                a_trial = a.copy()
                for pos in top_pos:
                    if a_trial[pos] > 1e-12:
                        a_trial[pos] *= 0.9985
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = inf_score(a_trial)
                if sc < best_score - 1e-13:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    stall = 0
                elif stall > 1000:
                    break
        else:
            stall = 0

        if (step + 1) % 20000 == 0:
            print(f"    CD {time.monotonic()-t0:.1f}s: score={best_score:.8f}")

    return best_a


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-3 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-2.0 * (x - 0.5) ** 2 / 0.15)
    elif choice == 5:
        a = np.exp(-6 * (x - 0.25) ** 2) + np.exp(-6 * (x - 0.75) ** 2)
    elif choice == 6:
        # Triangle-ish
        a = 1.0 - np.abs(2 * x - 1)
        a = a ** 0.7
    else:
        a = np.ones(n) * (0.8 + 0.2 * np.sin(4 * np.pi * x))
        a = np.maximum(a, 0.01)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses FFT-accelerated LogSumExp, mirror descent, LP Nesterov, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    # Load parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = inf_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = inf_score(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = inf_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    n_cur = len(best_seq)

    # === Phase 1: LogSumExp L-BFGS (fast, FFT-based) ===
    print(f"\n=== Phase 1: LSE L-BFGS (score={best_score:.8f}, n={n_cur}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages = [200.0, 100.0, 50.0, 30.0, 18.0, 10.0, 6.0, 4.0, 2.5, 1.5, 1.0, 0.6, 0.35, 0.2, 0.1]
        a_ref = lbgfs_softmax(best_seq, t_alloc, T_stages=T_stages, maxiter_per_stage=300)
        try_update(a_ref, "LSE_LB")

    # === Phase 2: Mirror Descent (fast divergence) ===
    print(f"\n=== Phase 2: Mirror Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md = mirror_descent_soft(best_seq, t_alloc, eta0=0.8, T_md=2.0, max_iters=300000)
        try_update(a_md, "MirrorDesc")

    # === Phase 3: LP Nesterov (precise) ===
    print(f"\n=== Phase 3: LP Nesterov (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 550.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp = nesterov_lp_refine(best_seq, t_alloc, solver=solver, max_iters=50000)
        try_update(a_lp, "LP_Nesterov")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 300.0)
    if t_alloc > 40:
        n_best = len(best_seq)
        candidate_ns = [n_best - 200, n_best + 200, n_best + 500, max(400, n_best // 2)]
        candidate_ns = sorted(set([max(300, x) for x in candidate_ns if x != n_best]))
        per_n = t_alloc / max(1, len(candidate_ns))
        print(f"  Trying n values: {candidate_ns}, {per_n:.0f}s each")
        for n_try in candidate_ns:
            if time.monotonic() > deadline - 30:
                break
            a_interp = interpolate_sequence(best_seq, n_try)
            t_sub = min(per_n, deadline - time.monotonic() - 15)
            if t_sub < 20:
                continue
            print(f"  n={n_try}, budget={t_sub:.0f}s")
            # Quick LSE + LP
            t_lse = t_sub * 0.4
            t_lp = t_sub * 0.6
            a_sub = lbgfs_softmax(a_interp, t_lse,
                                  T_stages=[50, 25, 15, 8, 5, 3, 2, 1, 0.5, 0.2],
                                  maxiter_per_stage=200)
            sc_sub = inf_score(a_sub)
            a_sub2 = nesterov_lp_refine(a_sub, t_lp, solver=solver, max_iters=20000)
            try_update(a_sub2, f"multi_n_{n_try}")
            try_update(a_sub, f"multi_n_{n_try}_lse")

    # === Phase 5: Another LSE pass on best ===
    print(f"\n=== Phase 5: LSE Refine (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages2 = [20.0, 12.0, 7.0, 4.0, 2.5, 1.5, 1.0, 0.6, 0.35, 0.2, 0.12, 0.08]
        a_ref2 = lbgfs_softmax(best_seq, t_alloc, T_stages=T_stages2, maxiter_per_stage=250)
        try_update(a_ref2, "LSE_refine")

    # === Phase 6: Final LP polish ===
    print(f"\n=== Phase 6: Final LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_fin = nesterov_lp_refine(best_seq, t_alloc, solver=solver, max_iters=30000)
        try_update(a_fin, "final_LP")

    # === Phase 7: Coordinate Descent Polish ===
    print(f"\n=== Phase 7: Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 120.0)
    if t_alloc > 10:
        a_cd = coordinate_descent(best_seq, t_alloc, max_steps=100000)
        try_update(a_cd, "coord_desc")

    # Final verification
    if best_seq is not None:
        fs = inf_score(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
        print(f"Total time: {time.monotonic() - t_start:.1f}s")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
