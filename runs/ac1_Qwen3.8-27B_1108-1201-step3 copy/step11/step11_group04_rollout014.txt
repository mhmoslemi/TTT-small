import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def get_lp_solver():
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def solve_lp_best_response(a_fixed, solver):
    """Find b >= 0, sum(b)=1, minimizing max_k (a*b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers_to_try = [solver, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS, cp.GUROBI, cp.MOSEK]
    solved = False
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue
    
    if not solved:
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(max(0, M_var.value))

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def logsumexp_obj_grad(z, n, T_temp):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse
    
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z

def evaluate_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def multiplicative_subgradient(a0, time_limit, n_steps=200000):
    """Mirror descent targeting top peaks with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_score(a)
    t0 = time.monotonic()
    
    eta_base = 0.03
    no_improve = 0
    scale = 1.0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Use weighted subgradient: weight proportional to (b[k] - mean_b)
        mean_b = b.mean()
        excess = np.maximum(b - mean_b, 0)
        total_excess = excess.sum()
        if total_excess < 1e-15:
            break
        w = excess / total_excess
        
        g = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-10:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w[k] * a[k - idx]
        
        eta = eta_base * scale
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        score = evaluate_score(a_new)
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 40:
            scale *= 0.75
            no_improve = 0
            if scale < 0.003:
                break
    
    return best_a, best_score

def spectral_flatten_autocorr(a, n_iters=100, blend=0.3, deadline=None):
    """
    Spectral method: iterate between time domain (enforce non-negativity)
    and frequency domain (flatten autocorrelation spectrum).
    
    Key insight: We want the autocorrelation to be flat, which means
    the power spectrum |F(a)|^2 should be a delta at DC (plus constraints).
    Actually, flat autocorrelation means its DFT is concentrated, which means
    |F(a)|^2 is concentrated. But we also need a >= 0.
    
    Alternative: iterate between time domain (a >= 0, normalize) and
    frequency domain (modify |F(a)| to make autocorrelation flatter).
    """
    n = len(a)
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Compute FFT of zero-padded a
        a_padded = np.zeros(L, dtype=np.complex128)
        a_padded[:n] = a
        F = fft(a_padded)
        
        # Compute current autocorrelation
        b = irfft(fft(a_padded) * fft(a_padded).conj(), L)[:size].real
        # Actually b = ifft(|F|^2)
        b = ifft(F * F.conj()).real[:size]  # This gives circular autocorrelation
        # For linear, we need to be more careful
        b = fast_convolve(a)
        
        # Target: flat autocorrelation (all values = mean)
        target_val = b.sum() / len(b)  # = (sum a)^2 / (2n-1) = 1/(2n-1)
        
        # Compute how much each frequency contributes to peaks
        # Modify the spectrum to reduce peak autocorrelation
        mags = np.abs(F)
        phases = np.angle(F)
        
        # Strategy: boost frequencies that correspond to "spreading out" the autocorrelation
        # The autocorrelation at lag k is sum_l |F_l|^2 * exp(2*pi*i*l*k/L) / L
        # To flatten, we want to reduce the "concentration" of |F|^2
        # This is equivalent to making |F| more uniform
        
        # Use geometric mean as target (preserves total energy)
        non_dc = mags[1:L//2+1]
        if non_dc.min() < 1e-15:
            target_mag = np.mean(non_dc)
        else:
            target_mag = np.exp(np.mean(np.log(non_dc + 1e-30)))
        
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target_mag
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])
        
        a_new = ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new
    
    return a

def frank_wolfe_refine(a0, time_limit, n_steps=50000):
    """
    Frank-Wolfe (conditional gradient) on the simplex.
    Minimize max_k (a*a)[k] subject to a >= 0, sum(a) = 1.
    Subgradient at a: 2 * T_{k*} * a where k* is argmax.
    FW step: find vertex e_i minimizing (subgrad)^T e_i, then line search.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find active set (positions near max)
        threshold = max_b * (1.0 - 1e-4)
        active = np.where(b >= threshold)[0]
        
        # Compute subgradient (average of active subgradients)
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)
        
        # Frank-Wolfe: find i that minimizes g[i] (since (g)^T e_i = g[i])
        i_min = np.argmin(g)
        
        # Step size: gamma = 2/(step+2) for FW
        gamma = 2.0 / (step + 2)
        a_new = (1 - gamma) * a + gamma * 0  # (1-gamma)*a + gamma*e_{i_min}
        e = np.zeros(n)
        e[i_min] = 1.0
        a_new = (1 - gamma) * a + gamma * e
        
        # Line search (backtracking)
        a_try = a_new.copy()
        s_try = a_try.sum()
        if s_try < 1e-15:
            continue
        a_try = a_try / s_try
        score_try = 2.0 * n * fast_convolve(a_try).max()
        
        if score_try < best_score:
            best_score = score_try
            best_a = a_try.copy()
            a = a_try
        else:
            # Try smaller step
            for frac in [0.5, 0.25, 0.1]:
                a_try = (1 - gamma*frac) * a + gamma*frac * e
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                score_try = 2.0 * n * fast_convolve(a_try).max()
                if score_try < best_score:
                    best_score = score_try
                    best_a = a_try.copy()
                    a = a_try
                    break
    
    return best_a, best_score

def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage, deadline=None):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit
    n_stages = len(T_stages)
    
    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break
        
        stages_left = n_stages - stage_i
        avail_time = remaining
        iters = max(20, min(maxiter_per_stage, int(avail_time * 30 / stages_left)))
        
        def func(z_in):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = logsumexp_obj_grad(z_in, n, stage_T)
            return obj
        
        def jac(z_in):
            if time.monotonic() > deadline:
                return np.zeros_like(z_in)
            _, g = logsumexp_obj_grad(z_in, n, stage_T)
            return g
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-25,
                         'gtol': 1e-18, 'maxfun': iters * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)

def structured_perturb(a, rng, strength=0.03):
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 10)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
