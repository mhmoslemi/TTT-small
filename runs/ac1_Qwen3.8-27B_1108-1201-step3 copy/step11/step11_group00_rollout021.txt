```python
def mirror_descent_step(a, T, eta):
    n = len(a)
    b = fast_autocorr(a)  # O(n log n)
    # Compute LogSumExp weights
    max_b = b.max()
    exp_b = np.exp(T * (b - max_b))
    w = exp_b / exp_b.sum()
    
    # Compute smooth gradient
    N = 1 << (n + 2*n-1 - 1).bit_length()  # = 1 << (3n-2).bit_length()
    A = rfft(a, N)
    W_rev = rfft(w[::-1], N)
    conv = irfft(A * W_rev, N)[:3*n-2]
    g = 2.0 * conv[2*n-2:n-2:-1]
    
    # Mirror descent update
    log_a = np.log(np.maximum(a, 1e-30))
    log_a_new = log_a - eta * g
    log_a_new -= log_a_new.max()  # for numerical stability
    a_new = np.exp(log_a_new)
    a_new /= a_new.sum()
    
    return a_new, g
```
