The objective is `2*n*max(conv(a,a))/(sum(a))^2`. 

If we let `p = a / sum(a)` (normalized to sum 1), the objective becomes `2*n*max(conv(p,p))`.

So we're minimizing `max(p*p)` where `p ≥ 0, sum(p) = 1`.

The LP alternation finds: given p, find q that minimizes max(conv(p,q)). At a fixed point p* = q*, we have that p* minimizes max(conv(p*,p)) in a "best response" sense.

But this is NOT the same as p* minimizing max(conv(p,p))! The fixed point of the best response map is a Nash equilibrium of the game, not necessarily the minimum of the original problem.

However, for this specific problem, the Nash equilibrium should be close to the minimum because:
- If p minimizes max(conv(p,p)), then max(conv(p,q)) ≤ max(conv(p,p)) for all q (since the peak of conv(p,p) is achieved at some k, and for any q, conv(p,q)[k] ≤ max_j q[j] * conv(p,p)[k]... no that's not right)

Actually, I think the Nash equilibrium IS the minimum for this problem. Here's why:
- At a Nash equilibrium p*, for all q: max_k conv(p*,q)[k] ≥ max_k conv(p*,p*)[k]
- In particular, taking q = p*: max_k conv(p*,p*)[k] ≥ max_k conv(p*,p*)[k] (trivially)
- And for the other direction: max_k conv(q,p*)[k] ≥ max_k conv(p*,p*)[k]

But we want to show that p* minimizes max_k conv(p,p)[k] over all p. This doesn't follow directly from the Nash equilibrium condition.

However, by symmetry of the game (the payoff is symmetric), the Nash equilibrium should be close to the minimum. In fact, for zero-sum games, the value is well-defined, and the minmax = maxmin. Here:
- min_p max_q max_k conv(p,q)[k] ≤ max_k conv(p*,p*)[k] ≤ max_p min_q max_k conv(p,q)[k]

            best_a = a_uniform.copy()
            improved = True
        
        if improved:
            stall = 0
        else:
            stall += 1
        
        # Restart
        if stall >= stall_threshold:
            # Structured perturbation
            b_cur = fast_convolve(a)
            k_peak = int(np.argmax(b_cur))
            noise = rng.standard_normal(n) * (0.03 + 0.02 * min(stall / 50.0, 1.0))
            a = a + noise
            a = np.maximum(a, 0)
            sa = a.sum()
            if sa > 1e-15:
                a /= sa
            a_ema = a.copy()
            a_uniform = a.copy()
            stall = 0
            stall_threshold = min(50, stall_threshold + 5)
        
        if iter_count % 100 == 0:
            print(f"    LP iter={iter_count} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def logsumexp_smoothed_lbgfs(a0, time_limit, T_stages=None):
    """L-BFGS on LogSumExp smoothed objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    t0 = time.monotonic()
    if T_stages is None:
        T_stages = [10.0, 5.0, 3.0, 2.0, 1.2, 0.7, 0.4, 0.2, 0.1]
    
    best_a = a.copy()
    best_score = evaluate_local(a)
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit - 3:
            break
        
        # Compute LogSumExp objective and gradient
        b = fast_convolve(a)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        w = exp_b / exp_b.sum()
        logsumexp_val = max_b / T + np.log(exp_b.sum())
        obj_base = 2.0 * n * logsumexp_val
        
        def func(z):
            if time.monotonic() - t0 > time_limit - 1:
                return 1e15
            a_t = np.maximum(z, 0)
            st = a_t.sum()
            if st < 1e-15:
                return 1e15
            a_t = a_t / st
            b_t = fast_convolve(a_t)
            mb = b_t.max()
            eb = np.exp(b_t / T - mb / T)
            le = mb / T + np.log(eb.sum())
            return 2.0 * n * le
        
        def jac(z):
            if time.monotonic() - t0 > time_limit - 1:
                return np.zeros(len(z))
            a_t = np.maximum(z, 0)
            st = a_t.sum()
            if st < 1e-15:
                return np.zeros(len(z))
            a_t = a_t / st
            b_t = fast_convolve(a_t)
            mb = b_t.max()
            eb = np.exp(b_t / T - mb / T)
            w_t = eb / eb.sum()
            # grad[i] = (2n/T) * sum_k w[k] * (a_t[k-i] + a_t[i-k]) (valid terms)
            grad = np.zeros(n)
            for k in range(len(w_t)):
                if w_t[k] < 1e-15:
                    continue
                # a_t[k-i] valid for i in [max(0,k-n+1), min(n-1,k)]
                i_s = max(0, k - n + 1)
                i_e = min(n - 1, k)
                if i_e >= i_s:
                    grad[i_s:i_e+1] += w_t[k] * a_t[k - i_e:k - i_s - 1:-1]
                # a_t[i-k] valid for i in [k, min(n-1, k+n-1)]
                i_s2 = max(0, k)
                i_e2 = min(n - 1, k + n - 1)
                if i_e2 >= i_s2:
                    grad[i_s2:i_e2+1] += w_t[k] * a_t[i_s2 - k:i_e2 - k + 1]
            grad *= 2.0 * n / T
            return grad
        
        try:
            res = minimize(fun=func, x0=a, jac=jac, method='L-BFGS-B',
                           options={'maxiter': 500, 'ftol': 1e-16,
                                    'gtol': 1e-12, 'maxfun': 5000,
                                    'disp': False})
            a = np.maximum(res.x, 0)
            sa = a.sum()
            if sa > 1e-15:
                a /= sa
            sc = evaluate_local(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception:
            pass
    
    return best_a, best_score


def coordinate_descent_refine(a0, time_limit):
    """Fine-grained coordinate descent."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    no_improve = 0
    
    for step in range(500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        k_star = int(np.argmax(b))
        
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_star - idx]
        
        improved = False
        # Try reducing top 3 contributors
        top_m = min(3, len(idx))
        top_pos = idx[np.argsort(contribs)[::-1][:top_m]]
        
        for pos in top_pos:
            if a[pos] < 1e-12:
                continue
            for factor in [0.999, 0.998, 0.995, 0.99, 0.98]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] - a_trial[pos]
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] = others + diff * others / others_sum
                a_trial /= a_trial.sum()
                sc = evaluate_local(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 2000:
                break
        
        if (step + 1) % 10000 == 0:
            print(f"    CD {time.monotonic()-t0:.0f}s: score={best_score:.8f}")
    
    return best_a, best_score


def structured_init(n, rng):
    """Create a good structured initial sequence."""
    x = np.linspace(0, 1, n)
    choice = rng.integers(0, 5)
    if choice == 0:
        a = np.cos(np.pi * x / 2)**2
    elif choice == 1:
        a = np.sin(np.pi * x)**2
    elif choice == 2:
        a = np.exp(-4 * x * (1-x))
    elif choice == 3:
        a = 0.5 + 0.4 * np.cos(2*np.pi*3*x)
        a = np.maximum(a, 0)
    else:
        a = np.exp(-10*(x-0.3)**2) + np.exp(-10*(x-0.7)**2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
