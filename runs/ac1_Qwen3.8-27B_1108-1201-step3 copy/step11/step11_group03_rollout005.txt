```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using iterative active-set QCQP
with palindromic symmetry exploitation and multi-n exploration.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_reduced_Q(k, n):
    """
    Build m x m matrix Q'_k for palindromic sequence of length n.
    m = (n+1)//2 free variables.
    Q'_k satisfies: x^T Q'_k x = conv(a,a)[k] where a is the expanded palindromic sequence.
    """
    m = (n + 1) // 2
    Q = np.zeros((m, m))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        p = min(j, n - 1 - j)
        kj = k - j
        q = min(kj, n - 1 - kj)
        if p == q:
            Q[p, p] += 1.0
        else:
            Q[p, q] += 0.5
            Q[q, p] += 0.5
    return Q


def expand_palindromic(x, n):
    """Expand palindromic free variables to full sequence of length n."""
    a = np.zeros(n)
    for j in range(n):
        a[j] = x[min(j, n - 1 - j)]
    return a


def palindromic_from_full(a, n):
    """Extract palindromic free variables from full sequence."""
    m = (n + 1) // 2
    x = np.zeros(m)
    for j in range(m):
        x[j] = (a[j] + a[n - 1 - j]) / 2.0
    x = np.maximum(x, 0)
    s = x.sum()
    if s > 1e-15:
        x = x / s
    else:
        x = np.ones(m) / m
    return x


def get_qcqp_solver():
    """Find a working QCQP solver."""
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.XPRESS, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            t = cp.Variable()
            prob = cp.Problem(cp.Minimize(t), [cp.quad_form(x, np.eye(3), assume_PSD=True) <= t, cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return None


def solve_active_qcqp(active_positions, n, x_init, solver):
    """
    Solve min t s.t. x^T Q'_k x <= t for k in active, sum(x)=1, x>=0
    where x are the palindromic free variables.
    """
    m = (n + 1) // 2
    x_var = cp.Variable(m, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(x_var) == 1]
    
    for k in active_positions:
        Qp = build_reduced_Q(int(k), n)
        constraints.append(cp.quad_form(x_var, Qp, assume_PSD=True) <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    solvers_to_try = []
    if solver is not None:
        solvers_to_try.append(solver)
    solvers_to_try.extend([cp.MOSEK, cp.GUROBI, cp.XPRESS, cp.CLARABEL, cp.ECOS, cp.SCS])
    
    seen = set()
    for s in solvers_to_try:
        if s is None or s in seen:
            continue
        seen.add(s)
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                x_val = np.asarray(x_var.value, dtype=np.float64)
                x_val = np.maximum(x_val, 0)
                s_sum = x_val.sum()
                if s_sum < 1e-15:
                    return None, None
                x_val = x_val / s_sum
                return x_val, float(max(0.0, t_var.value))
        except Exception:
            continue
    
    return None, None


def iterative_active_set(a_init, n, solver, time_limit, max_active=100, tol=1e-10):
    """
    Iterative active-set QCQP with palindromic constraint.
    Returns (best_x, best_score, best_a_full).
    """
    t0 = time.monotonic()
    m = (n + 1) // 2
    
    # Initialize
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        x = np.ones(m) / m
    else:
        x = palindromic_from_full(a, n)
    
    best_x = x.copy()
    best_score = float('inf')
    
    for iteration in range(500):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute convolution for current x
        a_full = expand_palindromic(x, n)
        b = fast_convolve(a_full)
        max_b = b.max()
        score = 2.0 * n * max_b
        
        if score < best_score:
            best_score = score
            best_x = x.copy()
        
        # Initialize or update active set
        if iteration == 0:
            # Start with top 3 positions
            top_k = np.argsort(b)[::-1][:3]
            active = list(top_k)
        else:
            # Active set is maintained from previous iteration
        
        # Solve QP
        x_new, t_val = solve_active_qcqp(active, n, x, solver)
        if x_new is None:
            break
        
        # Check for convergence and find violated constraints
        a_new = expand_palindromic(x_new, n)
        b_new = fast_convolve(a_new)
        max_new = b_new.max()
        score_new = 2.0 * n * max_new
        
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_x = x_new.copy()
        
        # Find most violated constraint
        ref_t = t_val if t_val is not None else max_new
        violations = b_new - ref_t
        max_viol = violations.max()
        
        if max_viol < tol * max(1.0, ref_t) and len(active) >= 2:
            # Converged
            if score_new < best_score:
                best_score = score_new
                best_x = x_new.copy()
            break
        
        # Add violated constraint
        if max_viol > 0:
            k_viol = int(np.argmax(violations))
            if k_viol not in active:
                active.append(k_viol)
                # Limit active set
                if len(active) > max_active:
                    # Keep the most violated ones
                    viol_scores = [b_new[k] for k in active]
                    active = [active[i] for i in np.argsort(viol_scores)[::-1][:max_active]]
        
        x = x_new
        
        if (iteration + 1) % 5 == 0:
            print(f"    iter={iteration+1}, active={len(active)}, score={score_new:.10f}, max_viol={max_viol:.2e}")
    
    # Final evaluation
    a_final = expand_palindromic(best_x, n)
    b_final = fast_convolve(a_final)
    final_score = 2.0 * n * b_final.max()
    
    return best_x, final_score, a_final


def multiplicative_weights_refine(a0, n, time_limit, n_steps=10000):
    """Multiplicative weights refinement (no palindromic constraint)."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    eta = 0.05
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Active positions
        threshold = max_b * (1.0 - 1e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        # Multiplicative update
        a_new = a * np.exp(-eta * g)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 50:
            eta *= 0.7
            no_improve = 0
            if eta < 1e-6:
                break
    
    return best_a, best_score


def lbgfs_polish(a0, n, time_limit):
    """L-BFGS polish using LogSumExp smoothing."""
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
        s = 1.0
    a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-300))
    
    t0 = time.monotonic()
    deadline = t0 + time_limit
    
    def lse_obj(z_arr):
        if time.monotonic() > deadline:
            return 1e15
        z_sh = z_arr - z_arr.max()
        exp_z = np.exp(z_sh)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        # Low temperature LogSumExp
        T = 0.01
        exp_terms = np.exp((b - max_b) / T)
        lse = max_b + T * np.log(exp_terms.sum())
        return 2.0 * n * lse
    
    def lse_grad(z_arr):
        if time.monotonic() > deadline:
            return np.zeros_like(z_arr)
        z_sh = z_arr - z_arr.max()
        exp_z = np.exp(z_sh)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        T = 0.01
        exp_terms = np.exp((b - max_b) / T)
        w = exp_terms / exp_terms.sum()
        
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        
        S_val = np.dot(grad_a, a)
        grad_z = a * (grad_a - S_val)
        return 2.0 * n * grad_z
    
    T_stages = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=lse_obj, x0=z, jac=lse_grad,
                method='L-BFGS-B',
                options={'maxiter': remaining * 100, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': remaining * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    z_sh = z - z.max()
    exp_z = np.exp(z_sh)
    a_final = exp_z / exp_z.sum()
    a_final = np.maximum(a_final, 0)
    s_f = a_final.sum()
    if s_f > 1e-15:
        a_final = a_final / s_f
    return a_final


def evaluate_local(a):
    """Evaluate score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
