import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv)/(sum)^2 efficiently."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    c = fast_conv(a)
    return float(2.0 * n * c.max() / (S * S))


def lse_and_grad(a, T):
    """
    Compute (1/T)*log(sum exp(T*c_k)) and gradient w.r.t. a.
    Uses FFT for O(n log n) computation.
    Returns (lse_value, gradient_array).
    """
    n = len(a)
    c = fast_conv(a)
    Tc = T * c
    max_Tc = Tc.max()
    e = np.exp(Tc - max_Tc)
    lse_val = (max_Tc + np.log(e.sum())) / T
    w = e / e.sum()

    # gradient[i] = 2 * sum_k w[k] * a[k-i] (with zero padding)
    # = 2 * (convolve(a, w))[i] where convolve is full linear convolution
    conv_len = n + (2 * n - 1) - 1  # = 3n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    conv_full = irfft(A * W, fft_size)
    grad = 2.0 * conv_full[:n]
    return lse_val, grad


def lse_softmax_obj_grad(z, T):
    """Objective and gradient for softmax parameterization a = softmax(z)."""
    n = len(z)
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a = a / a.sum()

    val, grad_a = lse_and_grad(a, T)

    # Transform gradient to z-space
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return val, grad_z


def optimize_lse(a0, time_limit, T_start=200.0, T_end=0.1, n_stages=20):
    """
    Multi-stage LogSumExp optimization using L-BFGS in softmax space.
    Returns best (a, score) found.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()

    best_a = a.copy()
    best_score = compute_score(a)

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        iters = max(20, min(100, int(50 + 30 * si / max(1, n_stages - 1))))

        try:
            res = minimize(
                fun=lambda z_in: lse_softmax_obj_grad(z_in, T)[0],
                x0=z,
                jac=lambda z_in: lse_softmax_obj_grad(z_in, T)[1],
                method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxfun': iters * 300, 'disp': False}
            )
            z = res.x
            z_shifted = z - z.max()
            a = np.exp(z_shifted)
            a = a / a.sum()
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception:
            pass

    return best_a, best_score


def coord_descent_sym(a0, time_limit, n_sweeps=None):
    """
    Exact coordinate descent for symmetric sequences (a_j = a_{n-1-j}).
    Only the first half is free. Each coordinate update solves a 1D convex minimization.
    Returns (a, score).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
        return a, compute_score(a)
    a = a / S

    # Enforce symmetry
    half = (n + 1) // 2
    a_sym = np.zeros(n)
    for i in range(half):
        a_sym[i] = a_sym[n - 1 - i] = 0.5 * (a[i] + a[n - 1 - i])
    a = a_sym / a_sym.sum()

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    if n_sweeps is None:
        n_sweeps = 10000

    sweep = 0
    while sweep < n_sweeps and time.monotonic() - t0 < time_limit:
        improved = False
        # Random order for diversification
        order = np.random.permutation(half)

        for i in order:
            x_old = a[i]
            if x_old < 1e-15 and i != n - 1 - i:
                # Mirror element also gets set to 0
                continue

            # Compute w_k = derivative of conv_k w.r.t. a_i (for single element, not symmetric pair)
            # For symmetric update, we change a[i] and a[n-1-i] simultaneously
            j = n - 1 - i
            if i == j:
                # Center element (only when n is odd)
                # d conv_k / d a_i = 2 * a[k-i]
                w = np.zeros(2 * n - 1)
                for k in range(2 * n - 1):
                    m = k - i
                    if 0 <= m < n:
                        w[k] = 2.0 * a[m]
                # conv_2i has extra: d/dx(x) in c_{2i} includes x term already in w
            else:
                # Changing both a[i] and a[j]
                # d conv_k / d(a_i) = 2*a[k-i], d conv_k / d(a_j) = 2*a[k-j]
                w = np.zeros(2 * n - 1)
                for k in range(2 * n - 1):
                    m1 = k - i
                    m2 = k - j
                    val = 0.0
                    if 0 <= m1 < n:
                        val += 2.0 * a[m1]
                    if 0 <= m2 < n:
                        val += 2.0 * a[m2]
                    w[k] = val

            c = fast_conv(a)

            # Find optimal delta > 0 (increase) or < 0 (decrease)
            # We parameterize: new_x = x_old + delta, new_y = (S - x_old - x_old) * something
            # Actually for symmetric: new a[i] = new a[j] = x_old + delta
            # New sum = S + 2*delta
            # We search over delta in [-x_old, max_delta]

            # Evaluate score as function of delta
            # score(delta) = 2n * max_k (c_k + delta*w_k) / (S + 2*delta)^2
            # Note: for k=2i or k=2j, there's a quadratic term in delta if i==j
            # For i != j: conv_k is linear in delta (since both a_i and a_j change linearly)
            #   Actually: new_conv_k = c_k + delta * 2*a[k-i] + delta * 2*a[k-j]
            #   But a[k-i] already includes the old a[i] or a[j]! We need to be careful.
            #
            # Let me recompute: new_conv_k = sum_p a_p' * a_{k-p}'
            # where a_i' = a_i + delta, a_j' = a_j + delta, others unchanged.
            # = conv_k + delta * (a[k-i] + a[k-j]) + delta * (a[k-i] + a[k-j])
            #   + delta^2 * (delta_{i==k-i} * delta_{j==k-j} + ...)
            #
            # This is getting complicated. Let me just use the simple approach:
            # For small delta, the change is approximately linear.
            # Let's do a local search: try several delta values and pick the best.

            # Try a set of delta values (log-spaced around 0)
            best_delta = 0.0
            best_local_score = best_score

            # Quick evaluation using the current conv
            S_new_base = S + 2 * 0.0  # S unchanged at delta=0
            max_c_base = c.max()
            base_score = 2.0 * n * max_c_base / (S * S)

            deltas = [-0.9, -0.7, -0.5, -0.3, -0.1, -0.03, -0.01,
                      0.01, 0.03, 0.1, 0.3, 0.5, 0.7, 0.9]
            deltas = [d * x_old for d in deltas if x_old > 1e-15]
            if not deltas:
                continue

            for delta in deltas:
                if x_old + delta < 0:
                    continue
                new_conv = c + delta * w
                # For the symmetric pair, there might be quadratic corrections
                # but for a first-order approximation this is fine
                max_new = new_conv.max()
                S_new = S + 2 * delta
                if S_new < 0.01:
                    continue
                new_score = 2.0 * n * max_new / (S_new * S_new)
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_delta = delta

            if best_delta != 0:
                a[i] += best_delta
                a[j] += best_delta
                a = a / a.sum()
                S += 0  # We renormalized, so S goes back to 1
                sc = compute_score(a)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a.copy()
                    improved = True

        sweep += 1
        if not improved and sweep % 50 == 0:
            # No improvement for 50 sweeps, try random perturbations
            for _ in range(20):
                i = np.random.randint(half)
                j = n - 1 - i
                frac = np.random.uniform(-0.1, 0.1)
                a_trial = a.copy()
                a_trial[i] = max(0, a[i] * (1 + frac))
                a_trial[j] = a_trial[i]
                s_t = a_trial.sum()
                if s_t > 0.01:
                    a_trial = a_trial / s_t
                    sc = compute_score(a_trial)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        improved = True
            if not improved:
                break

    return best_a, best_score


def coord_descent_general(a0, time_limit, max_sweeps=500):
    """
    General (non-symmetric) coordinate descent with 1D subproblem solver.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
        return a, compute_score(a)
    a = a / S

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    c = fast_conv(a)

    no_improve = 0
    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        order = np.random.permutation(n)
        improved = False

        for i in order:
            x_old = a[i]
            if x_old < 1e-14:
                continue

            # w_k = derivative of conv_k w.r.t. a_i = 2 * a[k-i] (zero-padded)
            w = np.zeros(2 * n - 1)
            # k-i ranges from -i to 2n-2-i
            # a[k-i] is nonzero when 0 <= k-i < n, i.e., i <= k <= i+n-1
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            m = np.arange(lo, hi + 1) - i
            w[lo:hi + 1] = 2.0 * a[m]

            # Minimize max_k (c_k + delta * w_k) / (S + delta)^2
            # Over delta in [-x_old, delta_max]
            # The function is not quite convex due to the denominator, but is unimodal in practice

            # Sample-based 1D minimization
            # Try geometric spacing of delta values
            delta_range = x_old  # maximum possible change
            n_samples = 30
            deltas = np.concatenate([
                np.linspace(-delta_range, -1e-6, 10),
                np.linspace(1e-6, delta_range, 20)
            ])

            best_delta = 0.0
            best_val = 2.0 * n * c.max() / (S * S)

            new_convs = c[None, :] + deltas[:, None] * w[None, :]
            max_convs = new_convs.max(axis=1)
            S_new = S + deltas
            valid = (S_new > 0.01) & ~np.isnan(max_convs)
            if not valid.any():
                continue
            scores = np.where(valid, 2.0 * n * max_convs / S_new**2, np.inf)
            idx = np.argmin(scores)
            if scores[idx] < best_val - 1e-15:
                best_delta = deltas[idx]
                best_val = scores[idx]

            if abs(best_delta) > 1e-12:
                a[i] = x_old + best_delta
                a = np.maximum(a, 0)
                S = a.sum()
                a = a / S
                # Update conv incrementally
                c = c + best_delta * w
                # Fix the quadratic term for k = 2i (if valid)
                k_self = 2 * i
                if 0 <= k_self < 2 * n - 1:
                    c[k_self] += best_delta * best_delta  # extra x^2 term

                sc = 2.0 * n * c.max() / (S * S)
                if sc < best_score - 1e-15:
                    best_score = sc
                    best_a = a.copy()
                    improved = True

        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0

    return best_a, best_score


def peak_shave_random(a0, time_limit, seed=123):
    """
    Random perturbation + peak shaving for final polish.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    n_iter = 0

    while time.monotonic() - t0 < time_limit:
        n_iter += 1
        # Get current peaks
        c = fast_conv(a)
        top_k = np.argsort(c)[::-1][:5]

        # Strategy 1: Move mass from high contributors to low areas
        k_peak = top_k[0]
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        if len(idx) > 0:
            contrib = a[idx] * a[k_peak - idx]
            src = idx[np.argmax(contrib)]

            # Find a good destination (low value, away from peaks)
            peak_zones = set()
            for kp in top_k:
                jl = max(0, kp - n + 1)
                jr = min(n - 1, kp)
                peak_zones.update(range(max(0, jl - 5), min(n, jr + 6)))

            candidates = [j for j in range(n) if j not in peak_zones and j != src]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            # Pick from lowest-value candidates
            candidates.sort(key=lambda j: a[j])
            dst = candidates[rng.integers(min(len(candidates), 50))]

            for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                sc = compute_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    break

        # Strategy 2: Random small perturbation
        if n_iter % 3 == 0:
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.1)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    sc = compute_score(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def symmetric_init(n, rng):
    """Generate a symmetric initial sequence."""
    half = (n + 1) // 2
    choice = rng.integers(0, 5)
    if choice == 0:
        x = np.linspace(0, 1, half)
        h = 0.5 * (1 + np.cos(np.pi * x))
    elif choice == 1:
        x = np.linspace(0, 1, half)
        h = np.exp(-2 * x * x)
    elif choice == 2:
        x = np.linspace(0, 1, half)
        h = np.sin(np.pi * x / 2)
    elif choice == 3:
        x = np.linspace(0, 1, half)
        h = 1.0 - 0.5 * x
    else:
        x = np.linspace(0, 1, half)
        h = np.exp(-1.5 * (x - 0.3)**2) + 0.5 * np.exp(-1.5 * (x - 0.7)**2)

    h = np.maximum(h, 0)
    # Make symmetric
    a = np.zeros(n)
    for i in range(half):
        a[i] = h[i]
        a[n - 1 - i] = h[i]
    a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization combining:
    - Symmetric and general coordinate descent
    - LogSumExp gradient optimization with adaptive annealing
    - Multi-length exploration
    - Random peak shaving for final polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = compute_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except (NameError, Exception):
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = compute_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # If no parent, create initial
    if parent is None:
        n0 = 1600
        best_seq = symmetric_init(n0, rng)
        best_score = compute_score(best_seq)
        print(f"Default init: n={n0}, score={best_score:.6f}")

    remaining = time_budget - (time.monotonic() - t_start)
    n_cur = len(best_seq)

    # === Phase 1: Symmetric coordinate descent on parent length ===
    print(f"\n=== Phase 1: Symmetric CD at n={n_cur} ===")
    t1 = min(remaining * 0.20, 300.0)
    if t1 > 20:
        a_out, _ = coord_descent_sym(best_seq, t1)
        try_update(a_out, f"symCD_n{n_cur}")

    # === Phase 2: LogSumExp on parent length ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 2: LSE at n={n_cur} ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 15:
        a_out, _ = optimize_lse(best_seq, t2, T_start=100.0, T_end=0.1, n_stages=15)
        try_update(a_out, f"lse_n{n_cur}")

    # === Phase 3: General coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 3: General CD at n={n_cur} ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 20:
        a_out, _ = coord_descent_general(best_seq, t3, max_sweeps=200)
        try_update(a_out, f"genCD_n{n_cur}")

    # === Phase 4: Try longer lengths ===
    for n_try in [int(n_cur * 1.25), int(n_cur * 1.5), int(n_cur * 1.75)]:
        remaining = time_budget - (time.monotonic() - t_start)
        if remaining < 100:
            break
        print(f"\n=== Phase 4: Explore n={n_try} ===")
        init = interpolate_seq(best_seq, n_try)
        # Symmetric CD
        t4 = min(remaining * 0.15, 150.0)
        if t4 > 15:
            a_out, _ = coord_descent_sym(init, t4)
            try_update(a_out, f"symCD_n{n_try}")
        # LSE
        remaining = time_budget - (time.monotonic() - t_start)
        if remaining < 50:
            break
        t4b = min(remaining * 0.10, 100.0)
        if t4b > 10:
            a_out, _ = optimize_lse(best_seq, t4b, T_start=50.0, T_end=0.05, n_stages=12)
            # Interpolate result back if needed
            if a_out is not None and len(a_out) != n_try:
                a_out = interpolate_seq(a_out, n_try)
            try_update(a_out, f"lse_n{n_try}")

    # === Phase 5: Try shorter lengths ===
    for n_try in [int(n_cur * 0.8), int(n_cur * 0.6)]:
        remaining = time_budget - (time.monotonic() - t_start)
        if remaining < 80:
            break
        print(f"\n=== Phase 5: Explore n={n_try} ===")
        init = interpolate_seq(best_seq, n_try)
        t5 = min(remaining * 0.10, 100.0)
        if t5 > 10:
            a_out, _ = coord_descent_sym(init, t5)
            try_update(a_out, f"symCD_n{n_try}")

    # === Phase 6: Final LSE polish on best ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 6: Final LSE polish (n={len(best_seq)}) ===")
    t6 = min(remaining * 0.20, 200.0)
    if t6 > 15:
        a_out, _ = optimize_lse(best_seq, t6, T_start=20.0, T_end=0.01, n_stages=15)
        try_update(a_out, "final_lse")

    # === Phase 7: Final coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 7: Final CD polish ===")
    t7 = min(remaining * 0.20, 150.0)
    if t7 > 10:
        a_out, _ = coord_descent_sym(best_seq, t7)
        try_update(a_out, "final_symCD")

    # === Phase 8: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 8: Peak shaving ===")
    t8 = min(remaining * 0.25, 200.0)
    if t8 > 10:
        a_out, _ = peak_shave_random(best_seq, t8, seed=999)
        try_update(a_out, "peak_shave")

    # === Phase 9: One more LSE with very low T ===
    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 20:
        print(f"\n=== Phase 9: Ultra-fine LSE ===")
        t9 = min(remaining * 0.5, 100.0)
        if t9 > 10:
            a_out, _ = optimize_lse(best_seq, t9, T_start=5.0, T_end=0.005, n_stages=10)
            try_update(a_out, "ultra_lse")

    # Final verification
    if best_seq is not None:
        fs = compute_score(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
