def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence (scale-invariant version)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build dense matrix T such that T @ b = convolve(a, b). Only for small n."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        T[i:i+n, i] = a
    return T


def build_sparse_conv_matrix(a):
    """Build sparse matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    row_ind = []
    col_ind = []
    data = []
    for i in range(n):
        row_ind.extend(range(i, i + n))
        col_ind.extend([i] * n)
        data.extend(a)
    return csr_matrix((data, (np.array(row_ind), np.array(col_ind))), shape=(size, n))


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return _lp_solver
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max conv(a,b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = build_sparse_conv_matrix(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [T @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    prob.solve(solver=solver)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val = b_val / s
    return b_val

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def logsumexp_gradient_step(a, T, lr):
    """One gradient step on LogSumExp smoothed objective."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T)
    w = exp_b / exp_b.sum()
    
    # Gradient: g[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i in [0, n-1])
    # = 2 * cross_correlation(w, a)[i]
    # cross_correlation(w, a)[i] = sum_j w[i+j] * a[j]
    # Use FFT for efficiency
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # conv(w, a) = irfft(W * A)
    # But we need cross-correlation: sum_j w[i+j] * a[j]
    # This is conv(flip(w), a)[i + (size-1)]... let me think more carefully.
    # Actually w has length 2n-1 and a has length n.
    # sum_j w[i+j] * a[j] for j=0..n-1, valid when i+j < 2n-1
    # This is the cross-correlation of w and a, which equals conv(flip(a), w)
    # Let me just use the direct formula via FFT
    # cross_corr[i] = sum_j w[i+j] * a[j]
    # = conv(flip(a), w)[i + n - 1] ... hmm this is getting complicated.
    # Let me use a simpler FFT approach:
    # Define u[j] = a[j] for j=0..n-1, 0 otherwise
    # Define v[k] = w[k] for k=0..2n-2, 0 otherwise
    # Then cross_corr[i] = sum_j v[i+j] * u[j] = (cross_correlation)(v, u)[i]
    # cross_correlation(v, u) = conv(flip(u), v)
    # flip(u) has u[n-1-j] at position j
    # conv(flip(u), v)[m] = sum_j flip(u)[j] * v[m-j] = sum_j u[n-1-j] * v[m-j]
    # Let l = n-1-j: = sum_l u[l] * v[m-n+1+l]
    # So cross_correlation(v,u)[i] = conv(flip(u),v)[i+n-1] = sum_l u[l] * v[i+l]
    # Yes! So cross_corr[i] = conv(flip(a), w)[i+n-1]
    
    flip_a = a[::-1]
    fa = rfft(flip_a, fft_size)
    fw = rfft(w, fft_size)
    conv_result = irfft(fa * fw, fft_size)
    # cross_corr[i] = conv_result[i + n - 1] for i=0..n-1
    cross = conv_result[n-1:2*n-1]
    
    g = 2.0 * cross
    # Normalize gradient
    g_norm = np.linalg.norm(g)
    if g_norm > 1e-15:
        g = g / g_norm * lr
    a_new = a - g
    a_new = np.maximum(a_new, 0)
    s_new = a_new.sum()
    if s_new < 1e-15:
        return a
    return a_new / s_new

def coordinate_refine(a0, time_limit, max_steps=50000):

    t0 = time.monotonic()

        contribs = a[idx] * a[peak_k - idx]
        top_m = min(5, len(idx))

            if a[pos] < 1e-12:
                continue

                mask[pos] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 100:
                # Try random perturbation
                pos = rng.integers(0, n)
                for factor in [0.99, 0.98, 0.97, 1.01, 1.02, 1.05]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    a_trial = np.maximum(a_trial, 0)

                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if not improved and stall > 500:
                    break
            if improved:
                stall = 0
        else:
            stall = 0

        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc {elapsed:.0f}s: score={best_score:.8f}")

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence (scale-invariant)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()

        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_sparse_conv_matrix(a):
    """Build sparse matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    row_ind = []
    col_ind = []
    data = []
    for i in range(n):
        row_ind.extend(range(i, i + n))
        col_ind.extend([i] * n)
        data.extend(a)
    return csr_matrix((data, (np.array(row_ind), np.array(col_ind))), shape=(size, n))

def solve_lp_single(a_fixed, solver):

    n = len(a_fixed)

    prob = cp.Problem(cp.Minimize(t_var), constraints)
    prob.solve(solver=solver)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:

                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def logsumexp_grad(a, T):
    """Compute LogSumExp smoothed objective gradient via FFT cross-correlation."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T)
    w = exp_b / exp_b.sum()

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    flip_a = a[::-1]
    fa = rfft(flip_a, fft_size)
    fw = rfft(w, fft_size)
    conv_result = irfft(fa * fw, fft_size)
    cross = conv_result[n - 1:2 * n - 1]
    return 2.0 * cross


def logsumexp_refine(a0, time_limit):

                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 80:
                pos = rng.integers(0, n)
                for factor in [0.99, 0.97, 0.95, 1.02, 1.05]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if not improved and stall > 400:
                    break
        else:
            stall = 0

        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {elapsed:.0f}s: {best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Wide exploration across n values and initializations (quick LP)
    2. Tournament selection
    3. Deep LP on top candidates
    4. LogSumExp + coordinate descent polishing
    5. Final LP refinement
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        best_seq = structured_init(1200, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Quick LP on parent ===
    print(f"\n=== Phase 1: Quick LP on parent (score={best_score:.8f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a1, _ = nesterov_lp(best_seq, t_alloc, solver, max_iters=50000)
        try_update(a1, "LP_p1")

    # === Phase 2: Wide multi-n exploration ===
    print(f"\n=== Phase 2: Multi-n exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 400.0)
    if t_alloc > 30:
        n_best = len(best_seq)
        # Generate diverse (n, init_type) combinations
        candidates = []
        # Interpolated parent at various n
        for n_try in [n_best - 200, n_best - 100, n_best + 100, n_best + 200, n_best + 400]:
            if 500 <= n_try <= 4000:
                candidates.append(('interp', n_try))
        # Structured inits at various n
        for n_try in [600, 800, 1000, 1200, 1500, 1800, 2000, 2500, 3000]:
            for init_seed in [0, 1, 2]:
                candidates.append(('struct', n_try))
        # Deduplicate
        seen = set()
        unique_candidates = []
        for c in candidates:
            if c[1] not in [x[1] for x in unique_candidates if x[0] == c[0]]:
                unique_candidates.append(c)
        # Limit to feasible number
        per_candidate = max(10.0, t_alloc / max(1, len(unique_candidates)))
        unique_candidates = unique_candidates[:int(t_alloc / max(10.0, per_candidate))]
        print(f"  {len(unique_candidates)} candidates, ~{per_candidate:.0f}s each")

        local_best = []
        for ci, (init_type, n_try) in enumerate(unique_candidates):
            if time.monotonic() > deadline - 20:
                break
            t_sub = min(per_candidate, deadline - time.monotonic() - 10)
            if t_sub < 8:
                continue
            if init_type == 'interp':
                a_start = interpolate_sequence(best_seq, n_try)
            else:
                a_start = structured_init(n_try, rng)
            sc_start = evaluate_local(a_start)
            a_out, sc_out = nesterov_lp(a_start, t_sub, solver, max_iters=15000)
            sc_final = evaluate_local(a_out)
            local_best.append((sc_final, n_try, a_out))
            print(f"    [{ci+1}/{len(unique_candidates)}] n={n_try}, {init_type}: {sc_final:.8f}")
            try_update(a_out, f"explore_{n_try}")

        # Sort by score, keep top 3 for deep refinement
        local_best.sort(key=lambda x: x[0])
        print(f"  Top candidates: {[f'{x[0]:.6f}(n={x[1]})' for x in local_best[:3]]}")

    # === Phase 3: Deep LP on top candidates ===
    print(f"\n=== Phase 3: Deep LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 350.0)
    if t_alloc > 30:
        # Deep refine the overall best
        t_sub = t_alloc * 0.6
        print(f"  Deep LP on best (n={len(best_seq)}), budget={t_sub:.0f}s")
        a_deep, _ = nesterov_lp(best_seq, t_sub, solver, max_iters=200000)
        try_update(a_deep, "deep_lp")

        # Also try deep on top local candidate if different
        rem = deadline - time.monotonic()
        t_sub2 = min(rem * 0.4, 100.0)
        if t_sub2 > 15 and len(local_best) > 0:
            top_sc, top_n, top_a = local_best[0]
            if top_n != len(best_seq):
                print(f"  Deep LP on n={top_n}, budget={t_sub2:.0f}s")
                a_deep2, _ = nesterov_lp(top_a, t_sub2, solver, max_iters=100000)
                try_update(a_deep2, f"deep_lp_{top_n}")

    # === Phase 4: LogSumExp refinement ===
    print(f"\n=== Phase 4: LogSumExp (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 10:
        a_lse = logsumexp_refine(best_seq, t_alloc)
        try_update(a_lse, "lse")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 5:
        a_cd, _ = coord_descent(best_seq, t_alloc)
        try_update(a_cd, "coord")

    # === Phase 6: Final LP polish ===
    print(f"\n=== Phase 6: Final LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 5:
        a_fin, _ = nesterov_lp(best_seq, t_alloc, solver, max_iters=50000)
        try_update(a_fin, "final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
