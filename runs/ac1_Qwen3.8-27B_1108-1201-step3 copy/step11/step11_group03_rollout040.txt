```python
def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_cross_convolve(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = len(a)

def sym_extend(x, n):
    nf = (n + 1) // 2
    a = np.zeros(n)
    if n % 2 == 0:
        m = n // 2
        a[:m] = x[:m]
        a[m:] = x[:m][::-1]
    else:
        m = n // 2
        a[:m+1] = x[:m+1]
        a[m+1:] = x[1:m+1][::-1]

def build_T_symmetric(a, n):

    nf = get_n_free(n)
    T = np.zeros((2*n-1, nf))

        p2 = n - 1 - i

        for k in range(max(0, p1), min(2*n-1, p1+n)):

        if p2 != p1:

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)

        max_b = b.max()

        active = np.where(b >= threshold)[0]
        if len(active) == 0:

        a_new = a - eta * g
        a_new = np.maximum(a_new, 0)

        if s_new < 1e-15:

            no_improve = 0
        elif no_improve > 50:

            no_improve = 0
            if eta < 1e-6:
                break
    
    return best_a, best_score

def interpolate_seq(f, new_n):

    result = np.maximum(result, 0)
    ss = result.sum()
    if ss > 1e-15:
        result = result / ss
    return result
```

d/dx_i [LogSumExp(b/T)] = (1/T) * sum_k exp(b[k]/T) / sum_k exp(b[k]/T) * db[k]/dx_i

And db[k]/dx_i = sum_j (delta_{k, i+j} * a[j] + delta_{k, j+i} * a[j]) = a[k-i] + (if k-i != i) a[i-k]... 

Hmm, this is getting complicated. Let me just use the FFT-based approach:
1. Compute b = conv(a,a) via FFT: O(n log n)
2. Compute weights w_k = exp((b[k]-max(b))/T) / sum(exp(...)): O(n)
3. Compute gradient: grad[i] = sum_k w_k * (a[k-i] + a[i-k] if k-i != i)
   This is like a weighted convolution: grad = conv(w, a) + conv(rev(w), a) (approximately)
   
Actually, db[k]/da[i] = a[k-i] (if 0 <= k-i < n) + a[i-k] (if 0 <= i-k < n and i-k != k-i)
Wait, more carefully:
b[k] = sum_j a[j] * a[k-j]
db[k]/da[i] = a[k-i] (from the j=i term) + (if k-i != i) a[i-k] (from the j=k-i term)
            = a[k-i] + a[i-k] if k-i != i
            = 2*a[k-i] if k-i == i (i.e., k = 2i)
            (with out-of-range a[.] = 0)

So grad[i] = sum_{k: 0<=k<2n-1} w[k] * (a[k-i] + a[i-k] if k≠2i, or 2*a[k-i] if k=2i)

This can be computed as:
grad = conv(w, a) + conv(rev(w), a) - diagonal_correction

where conv here is a cross-convolution. This is O(n log n) via FFT!

So the LogSumExp gradient can be computed in O(n log n).

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Symmetric formulation (a[j]=a[n-1-j]) - WLOG by symmetry, halves variables
2. Nesterov-accelerated LP alternating via HiGHS (main workhorse, O(1/t) rate)
3. LogSumExp L-BFGS polishing with FFT-based O(n log n) gradients
4. Subgradient refinement on active constraints
5. Multi-n exploration (n=1198..1202) to find globally best length
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = max(len(a), len(b))
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def sym_extend(x, n):
    """Extend symmetric free variables to full length-n vector."""
    nf = (n + 1) // 2
    a = np.zeros(n)
    if n % 2 == 0:
        m = n // 2
        a[:m] = x[:m]
        a[m:] = x[:m][::-1]
    else:
        m = n // 2
        a[:m + 1] = x[:m + 1]
        a[m + 1:] = x[1:m + 1][::-1]
    return a


def get_n_free(n):
    """Number of free variables in symmetric formulation."""
    return (n + 1) // 2


def build_T_symmetric(a, n):
    """Build T (2n-1 x nf) such that T @ x = conv(a, sym_extend(x, n))."""
    nf = get_n_free(n)
    T = np.zeros((2 * n - 1, nf))
    for i in range(nf):
        p1 = i
        p2 = n - 1 - i
        lo1 = max(0, p1)
        hi1 = min(2 * n - 1, p1 + n)
        for k in range(lo1, hi1):
            T[k, i] += a[k - p1]
        if p2 != p1:
            lo2 = max(0, p2)
            hi2 = min(2 * n - 1, p2 + n)
            for k in range(lo2, hi2):
                T[k, i] += a[k - p2]
    return T


def solve_lp_response(a, n):
    """Solve LP: min m s.t. conv(a,b)[k]<=m, sum(b)=1, b>=0, b symmetric."""
    nf = get_n_free(n)
    T = build_T_symmetric(a, n)

    c = np.zeros(nf + 1)
    c[0] = 1.0

    A_ub = np.empty((2 * n - 1, nf + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = T
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, nf + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (nf + 1)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.success:
        b_x = np.maximum(res.x[1:], 0)
        s = b_x.sum()
        if s < 1e-15:
            return None, None
        b_x = b_x / s
        b_full = sym_extend(b_x, n)
        return b_full, float(res.x[0])
    return None, None


def eval_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def nesterov_lp_refine_sym(a0, n, time_limit, seed=42):
    """Nesterov-accelerated LP alternating in symmetric space."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(seed)
    lp_count = 0
    lp_time_sum = 0.0

    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        lp_start = time.monotonic()
        b_new, M_b = solve_lp_response(y, n)
        lp_time_sum += time.monotonic() - lp_start
        lp_count += 1

        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = eval_score(a)
        score_y = eval_score(y)

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 300 == 0:
            if best_score < prev_best - 1e-11:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= 12:
                noise = 0.01 * rng.standard_normal(n)
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                stall = 0

            elapsed = time.monotonic() - t0
            avg_lp = lp_time_sum / max(1, lp_count) * 1000
            print(f"    iter={it+1} ({elapsed:.0f}s): score={best_score:.10f} avg_lp={avg_lp:.1f}ms")

        if lp_count > 50 and lp_time_sum / lp_count > 5.0:
            print(f"    LP too slow, stopping at iter {it+1}")
            break

    return best_a, best_score


def lse_objective_and_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z (log-domain)."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    b = fast_convolve(a)
    bm = b.max()
    shifted = (b - bm) / T_temp
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    lse_val = bm + T_temp * np.log(exp_shifted.sum())

    # grad_a[i] = 2 * sum_k w[k] * db[k]/da[i]
    # db[k]/da[i] = a[k-i] + a[i-k] (with out-of-range = 0), doubled if k=2i
    # grad_a = 2 * (cross_conv(w, a) + cross_conv(flip(w), a)) - diag_correction
    w_rev = w[::-1]
    # cross_conv(w, a)[i] = sum_k w[k] * a[k-i] (shifted)
    # We need: sum_k w[k] * a[k-i] for k in [i, i+2n-2], a valid for k-i in [0,n-1]
    # This is essentially a correlation, computable via cross-convolution
    grad_a = 2.0 * (fast_cross_convolve(w, a) + fast_cross_convolve(w_rev, a))
    # The above gives 2*sum_k w[k]*a[k-i] + 2*sum_k w[k]*a[i-k]
    # But we need to subtract the diagonal term (where k-i = i-k, i.e., k=2i)
    # which gets double-counted. The formula db[k]/da[i] = a[k-i] + a[i-k] when k!=2i,
    # and = 2*a[i] when k=2i. Our formula gives a[k-i]+a[i-k] = 2*a[i] when k=2i, which is correct.
    # So actually no correction needed!

    # Project gradient to simplex: grad_z = a * (grad_a - <grad_a, a>)
    S_val = np.dot(grad_a, a)
    grad_z = a * (grad_a - S_val)
    obj = 2.0 * n * lse_val
    return obj, 2.0 * n * grad_z


def lse_func_wrapper(z, n, T_temp, deadline):
    """Wrapper for L-BFGS objective."""
    if time.monotonic() > deadline:
        return 1e15
    obj, _ = lse_objective_and_grad(z, n, T_temp)
    return obj


def lse_jac_wrapper(z, n, T_temp, deadline):
    """Wrapper for L-BFGS gradient."""
    if time.monotonic() > deadline:
        return np.zeros_like(z)
    _, g = lse_objective_and_grad(z, n, T_temp)
    return g


def lse_polish(a0, n, time_limit):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    z = np.log(np.maximum(a0, 1e-30))
    z = z - z.mean()  # center for numerical stability
    t0 = time.monotonic()
    deadline = t0 + time_limit

    T_stages = [10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
    for T_s in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=lse_func_wrapper, x0=z, jac=lse_jac_wrapper,
                args=(n, T_s, deadline),
                method='L-BFGS-B',
                options={'maxiter': 200, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': 10000, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out


def subgradient_polish(a0, n, time_limit, n_iters=100000):
    """Subgradient descent on active constraints."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()

    eta = 0.05
    best_eta = eta
    no_improve = 0

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        threshold = max_b * 0.99
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                partner = k - j
                g[j] += a[partner]
        g /= len(active)

        a_new = a - eta * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = eval_score(a_new)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            best_eta = eta
        else:
            no_improve += 1

        if no_improve == 30:
            eta *= 0.7
            no_improve = 0
        elif no_improve > 100:
            eta = best_eta
            no_improve = 0
            if eta < 1e-7:
                break

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, f_norm)
    result = np.maximum(result, 0)
    ss = result.sum()
    if ss > 1e-15:
        result = result / ss
    return result


def cosine_init(n):
    """Generate a cosine-based initial symmetric sequence."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x / 2) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def run_lp_refine_with_update(a_input, n, t_budget, seed, best_data):

    a_out, sc = nesterov_lp_refine_sym(a_input, n, t_budget, seed=seed)
    if a_out is not None:
        a_out = np.maximum(a_out, 0)

                print(f"  LP improved: {sc:.10f} (n={n})")
    return best_data
```
