def fast_convolve(a):

    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

def evaluate_local(a):

def compute_subgradient_fft(a, n_active=20, rel_tol=1e-4):
    """Compute active-set subgradient of max(conv(a,a)) using FFT.
    
    Returns (c_max, subgradient) where subgradient has same length as a.
    The subgradient is (1/|S|) * sum_{k in S} dc_k/da, where S is the active set.
    dc_k/da_j = 2*a_{k-j} if 0 <= k-j < n.
    """
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    
    # Active set
    threshold = c_max * (1 - rel_tol)
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])

    if len(active) > n_active:

        active = active[order]
    
    # Build indicator vector
    e = np.zeros(2 * n - 1)

    # Cross-correlation: subgrad_j = (2/|S|) * sum_k e_k * a_{k-j}

    A = rfft(a, fft_size)
    R = irfft(E * np.conj(A), fft_size)
    subgrad = (2.0 / len(active)) * R[:n]
    
    return c_max, subgrad

            rows.append(row)
            rhs.append(c_max)
        
        if len(rows) == 0:
            break
        
        A_ineq = np.array(rows)
        b_ineq = np.array(rhs)
        
        # Equality: sum a_j = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        # Bounds: a_j >= 0, M >= 0
        bounds = [(0, None)] * (n + 1)
        
        try:
            from scipy.optimize import linprog
            res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq,
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
            if res.success:
                a_new = res.x[:n]
                a_new = np.maximum(a_new, 0)
                s = a_new.sum()
                if s < 1e-15:
                    continue
                a_new = a_new / s
                sc = evaluate_local(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new.copy()
                else:
                    # Blend with old
                    t = 0.5
                    a_blend = (1 - t) * a + t * a_new
                    a_blend = normalize(a_blend)
                    sc_blend = evaluate_local(a_blend)
                    if sc_blend < best_score:
                        best_score = sc_blend
                        best_a = a_blend.copy()
                        a = a_blend.copy()
        except Exception as e:
            # LP failed, fall back to subgradient step
            c_max_val, sg = compute_subgradient_fft(a, n_active=10, rel_tol=1e-4)
            for scale in [0.01, 0.005, 0.001]:
                a_new = normalize(a - scale * sg)
                sc = evaluate_local(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new.copy()
                break
    
    return best_a, best_score


def spectral_peak_shave(a0, time_limit, n_iters=5000):
    """Directly reduce the peak of the auto-convolution by modifying the spectrum.
    
    Idea: The peak of ifft(|A|^2) is what we want to reduce. We can do this
    by redistributing energy in the spectrum away from frequencies that
    contribute to the peak.
    """
    a = normalize(a0.copy())
    n = len(a)
    best_a = a.copy()
    best_score = evaluate_local(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    t0 = time.monotonic()
    
    lr = 0.003
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = rfft(a, fft_size)
        P = np.abs(A) ** 2
        c = irfft(A * A, fft_size)[:2*n-1]
        c_max = c.max()
        c_mean = c.mean()
        
        # Find which frequency bins contribute most to the peak
        # The peak at lag 0 gets contribution from all frequencies equally
        # The peak at other lags gets contribution weighted by cos(omega*k)
        
        # Simpler approach: reduce variance of P
        P_std = P.std()
        if P_std < 1e-10:
            break
        
        # Target: make P more uniform
        # Gradient of (P - P_mean)^2 w.r.t. a:
        # dP_k = 2*Re(A_k * conj(dA_k))
        # dA_k/da_j = conj(exp(-2*pi*i*k*j/fft_size)) = exp(2*pi*i*k*j/fft_size)
        # So dP_k/da_j = 2*Re(A_k * exp(2*pi*i*k*j/fft_size))
        # = 2*|A_k|*cos(angle(A_k) + 2*pi*k*j/fft_size)
        
        # We want to minimize sum_k (P_k - P_mean)^2
        # Gradient w.r.t. a_j: sum_k 2*(P_k - P_mean) * dP_k/da_j
        # = sum_k 4*(P_k - P_k_mean)*|A_k|*cos(angle(A_k) + 2*pi*k*j/fft_size)
        
        # This can be computed via FFT:
        # Let r_k = (P_k - P_mean) * A_k (for k = 0..fft_size//2)
        # Then gradient_j = 4 * Re(sum_k r_k * exp(2*pi*i*k*j/fft_size))
        # = 4 * Re(ifft(r)[j])  (approximately, with the right padding)
        
        r = (P - c_mean * (2*n-1) / fft_size) * A  # Note: P_mean in conv domain vs freq domain
        # Actually, let's just use P - P.mean()
        r = (P - P.mean()) * A
        # Extend to full spectrum for IFFT
        r_full = np.zeros(fft_size, dtype=np.complex128)
        r_full[:fft_size//2+1] = r
        r_full[fft_size//2+1:] = np.conj(r[1:fft_size//2][::-1])
        grad = np.real(ifft(r_full, fft_size)[:n])  # Hmm, ifft of real array...
        
        # Actually, let me use a simpler approach:
        # We want to move a in the direction that reduces the peak c_max.
        # The peak is at some lag k*. The gradient of c_{k*} w.r.t. a is 2*a shifted.
        # So we want to move a in the direction -2*a shifted by k*.
        # But we also want to maintain non-negativity.
        
        c_peak_idx = int(np.argmax(c))
        # Direction: reduce c_{c_peak_idx}
        # dc_{k*}/da_j = 2*a_{k*-j} (if 0 <= k*-j < n)
        direction = np.zeros(n)
        j_start = max(0, c_peak_idx - n + 1)
        j_end = min(n - 1, c_peak_idx)
        if j_start <= j_end:
            idx = c_peak_idx - np.arange(j_start, j_end + 1)
            direction[j_start:j_end+1] = 2.0 * a[idx]
        
        # Normalize direction
        d_norm = np.linalg.norm(direction)
        if d_norm < 1e-15:
            break
        direction = direction / d_norm
        
        # Line search
        improved = False
        for scale in [lr, lr * 0.5, lr * 0.2, lr * 0.1]:
            a_new = a - scale * direction
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                improved = True
            break
        
        if not improved:
            lr *= 0.7
            if lr < 1e-10:
                break
        else:
            lr = min(lr * 1.1, 0.01)
    
    return best_a, best_score


def diverse_perturb(a, rng, strength=0.03):
    """Generate a diverse perturbation of a."""
    n = len(a)
    method = rng.integers(0, 5)
    
    if method == 0:
        # Gaussian noise
        a_new = a + rng.normal(0, strength, n)
    elif method == 1:
        # Low-frequency modulation
        freq = rng.uniform(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 1, n)
        a_new = a * (1 + strength * 2 * np.cos(2 * np.pi * freq * x + phase))
    elif method == 2:
        # Edge modification
        edge_width = max(10, n // 10)
        a_new = a.copy()
        w = rng.uniform(0.01, strength)
        if rng.random() < 0.5:
            a_new[:edge_width] += w
            a_new[edge_width:] -= w * edge_width / (n - edge_width)
        else:
            a_new[-edge_width:] += w
            a_new[:-edge_width] -= w * edge_width / (n - edge_width)
    elif method == 3:
        # Block swap
        a_new = a.copy()
        bs = max(5, n // 20)
        i = rng.integers(0, n - bs)
        j = rng.integers(0, n - bs)
        if abs(i - j) > bs:
            a_new[i:i+bs], a_new[j:j+bs] = a_new[j:j+bs].copy(), a_new[i:i+bs].copy()
    else:
        # Sharpen or smooth
        if rng.random() < 0.5:
            # Sharpen
            a_max = a.max()
            a_new = a / (a_max * (1 - strength) + 1e-10)
        else:
            # Smooth (moving average)
            k = max(3, n // 50)
            kernel = np.ones(k) / k
            a_new = np.convolve(a, kernel, mode='same')
    
    return normalize(a_new)
