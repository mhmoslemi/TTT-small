def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix_sparse(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    rows = np.zeros(n * n, dtype=np.int32)
    cols = np.zeros(n * n, dtype=np.int32)
    vals = np.zeros(n * n, dtype=np.float64)
    count = 0
    for j in range(n):
        for k in range(j, min(j + n, size)):
            rows[count] = k
            cols[count] = j
            vals[count] = a[k - j]
            count += 1
    return csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(size, n))


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""

            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:

    return cp.SCS


def solve_lp_best_response(a_fixed, solver, T_mat, prev_b=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]
    solved = False
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            kwargs = {}
            if s == cp.PDLP:
                kwargs = {'verbose': False}
            if s == cp.CLARABEL:
                kwargs = {}
            if s == cp.SCS:
                kwargs = {'verbose': False, 'max_iters': 20000, 'eps': 1e-7}
            if s == cp.ECOS:
                kwargs = {'verbose': False}
            prob.solve(solver=s, **kwargs)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue
    if not solved:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(max(0, M_var.value))


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z for softmax parameterization."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T_temp
    scaled_shifted = scaled - scaled.max()
    exp_terms = np.exp(scaled_shifted)
    lse = max_b + T_temp * np.log(exp_terms.sum() + 1e-300)
    obj = 2.0 * n * lse
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(2 * n - 1)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


class LSEFunc:
    def __init__(self, n, T, deadline):

        return g


def multiplicative_weights_adaptive(a0, time_limit, n_steps, eta0=0.05):
    """Fast multiplicative weights with adaptive backtracking step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()

    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        # Use top-fraction active set for better gradient
        threshold = max_b * (1.0 - 5e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        w_active = b[active]
        w_active = w_active / w_active.sum()
        g = np.zeros(n)
        for w, k in zip(w_active, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w * a[k - idx]
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.02, 0.3)
        else:
            eta *= 0.85
            if eta < 1e-9:
                break
    return best_a, best_score

def spectral_shaping(a0, time_limit, n_iter=100000):
    """FFT-based spectral shaping to flatten the power spectrum."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1 - 1).bit_length()

    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        # Compute power spectrum
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        # Target: flatten the power spectrum
        # Use a gentle compression: new_amplitude = power^alpha
        alpha = 0.7  # compression factor
        new_amp = power ** 0.5 * (power ** (alpha - 0.5))
        new_amp = np.maximum(new_amp, 1e-10)
        # Keep phase, modify amplitude
        phase = np.angle(A)
        A_new = new_amp * np.exp(1j * phase)
        # Inverse FFT
        a_new = np.real(np.fft.irfft(A_new, fft_size))[:n]
        # Enforce non-negativity
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        # Blend with current to avoid oscillation
        blend = 0.3
        a_blend = (1 - blend) * a + blend * a_new
        a_blend = np.maximum(a_blend, 0)
        sb = a_blend.sum()
        if sb < 1e-15:
            a_blend = a.copy()
        else:
            a_blend = a_blend / sb
        score = 2.0 * n * fast_convolve(a_blend).max()
        if score < best_score:
            best_score = score
            best_a = a_blend.copy()
        a = a_blend
    return best_a, best_score


def coordinate_descent_polish(a0, time_limit):
    """Polish by directly reducing the peak convolution values."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        peak_k = np.argmax(b)
        # Get the indices contributing to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contributors = a[idx] * a[peak_k - idx]
        # Find the element with largest contribution
        max_contrib_idx = idx[np.argmax(contributors)]
        # Try reducing this element
        for frac in [0.5, 0.3, 0.1, 0.05, 0.02]:
            a_trial = a.copy()
            a_trial[max_contrib_idx] *= (1 - frac)
            # Redistribute to a random other element
            other = rng.integers(0, n)
            a_trial[other] += a[max_contrib_idx] * frac
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            sc = 2.0 * n * fast_convolve(a_trial).max()
            if sc < best_score:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial
                break
    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage=500):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        func = LSEFunc(n, stage_T, deadline)
        jac = LSEJac(n, stage_T, deadline)
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-25,
                         'gtol': 1e-20, 'maxfun': maxiter_per_stage * 500,
                         'disp': False, 'maxls': 50}
            )
            z = res.x
        except Exception:
            pass
    return softmax_from_z(z)

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Fast local evaluation (assumes a >= 0, sum > 0)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturb_sequence(a, rng, strength=0.03):

            try_update(a_ln_lse, f"lse_n{ln}")

    # === Phase 5: Perturbation + re-optimize ===
    print(f"\n=== Phase 5: Perturbation escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    for trial in range(5):
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        a_pert = perturb_sequence(best_seq, rng, strength=0.01 + 0.005 * trial)
        t_sub = min(rem * 0.2, 60.0)
        if t_sub < 10:
            break
        a_opt, _ = nesterov_lp_refine(a_pert, t_sub, solver, n_max_iters=100000)
        try_update(a_opt, f"pert{trial}")

    # === Phase 6: Coordinate descent polish ===
    print(f"\n=== Phase 6: Coord descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.10, 100.0)
    if t6 > 10:
        a_cd, _ = coordinate_descent_polish(best_seq, t6)
        try_update(a_cd, "coord")

    # === Phase 7: Final LP polish ===
    print(f"\n=== Phase 7: Final LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.25, 200.0)
    if t7 > 30:
        a_fin, _ = nesterov_lp_refine(best_seq, t7, solver, n_max_iters=500000)
        try_update(a_fin, "final_lp")

    # === Phase 8: Final LSE ultra-fine ===
    print(f"\n=== Phase 8: Final LSE (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t8 = min(rem * 0.30, 150.0)
        T_fine = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
        a_fin_lse = lbgfs_refine(best_seq, t8, T_fine, maxiter_per_stage=400)
        try_update(a_fin_lse, "fin_lse")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
