def fast_autocorr(a):
    """Full convolution a*a using FFT. O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def softmax_fn(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64)
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

    n = len(a)

def compute_subgradient_peaks(a, top_k=5):
    """
    Compute subgradient using top-k peaks of autocorrelation.
    Returns (gradient, weights_over_peaks).
    """
    n = len(a)
    b = fast_autocorr(a)
    # Get top peaks
    indices = np.argpartition(b, -top_k)[-top_k:]
    values = b[indices]
    # Softmax weights over top peaks
    max_v = values.max()
    w_peaks = np.exp(values - max_v)
    w_peaks = w_peaks / w_peaks.sum()
    
    # Compute gradient: g[j] = sum_i w_i * (a[k_i - j] + a[j - k_i]) (with bounds)
    grad = np.zeros(n)
    for i in range(top_k):
        k = int(indices[i])
        wi = w_peaks[i]
        # a[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k)
        if hi1 >= lo1:
            # j in [lo1, hi1], a[k-j] ranges from a[k-lo1] down to a[k-hi1]
            # a[k-j] for j=lo1,...,hi1 is a[k-lo1], a[k-lo1-1], ..., a[k-hi1]
            idx_a = np.arange(k - hi1, k - lo1 + 1)  # = k-hi1, ..., k-lo1
            grad[lo1:hi1+1] += wi * a[idx_a][::-1]
        
        # a[j-k] is valid when 0 <= j-k < n, i.e., k <= j <= k+n-1
        lo2 = max(0, k)
        hi2 = min(n - 1, k + n - 1)
        if hi2 >= lo2:
            # j in [lo2, hi2], a[j-k] ranges from a[lo2-k] to a[hi2-k]
            idx_a = np.arange(lo2 - k, hi2 - k + 1)
            grad[lo2:hi2+1] += wi * a[idx_a]
    
    return 2.0 * grad

def lse_objective_and_grad(z, T):
    """
    Log-sum-exp smoothed objective and gradient in z-space.
    a = softmax(z), objective = (1/T) * log(sum exp(T * conv(a,a)))
    """
    n = len(z)
    a = softmax_fn_safe(z)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T
    
    # weights
    w = exp_Tb / exp_Tb.sum()
    
    # Gradient w.r.t. a: grad_a[j] = 2 * sum_k w[k] * (a[k-j] + a[j-k])
    # Use the same method as compute_subgradient_peaks but with full w
    grad_a = np.zeros(n)
    
    # For efficiency, use FFT-based cross-correlation
    # termB[j] = sum_k w[k] * a[j-k] * 1{0<=j-k<n} = (w*a)[j] for j=0..n-1 (truncated conv)
    # termA[j] = sum_k w[k] * a[k-j] * 1{0<=k-j<n}
    #   = conv(a, w_rev)[2n-2-j] where w_rev = w[::-1]
    
    # Actually for the full convolution of length 2n-1:
    # (w*a)[m] = sum_i w[i]*a[m-i] for valid i,m-i
    # We want termB[j] = sum_k w[k]*a[j-k] for 0<=j-k<n, i.e., k<=j
    #   = sum_{k=0}^{j} w[k]*a[j-k] = (w*a)[j] (standard conv, first n elements)
    
    # termA[j] = sum_k w[k]*a[k-j] for 0<=k-j<n, i.e., k>=j
    #   = sum_{k=j}^{2n-2} w[k]*a[k-j]
    #   Let i=k-j: sum_{i=0}^{2n-2-j} w[i+j]*a[i]
    #   This is the cross-correlation of w and a, or equivalently:
    #   = (a * w_rev_full)[2n-2-j] where w_rev_full[i] = w[2n-2-i]
    
    w_rev = w[::-1].copy()  # length 2n-1
    # conv(a, w) has length n + (2n-1) - 1 = 3n-2
    # But we only need specific elements.
    # Let's use FFT.
    
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    
    # conv(a, w) = ifft(A * W), length 3n-2
    conv_aw = irfft(A * W, fft_size)[:3*n-2]
    # termB[j] = conv_aw[j] for j=0..n-1
    termB = conv_aw[:n]
    
    # conv(a, w_rev) = ifft(A * W_rev)
    W_rev = rfft(w_rev, fft_size)
    conv_awrev = irfft(A * W_rev, fft_size)[:3*n-2]
    # termA[j] = conv_awrev[2n-2-j] for j=0..n-1
    # j=0: conv_awrev[2n-2]
    # j=n-1: conv_awrev[2n-2-(n-1)] = conv_awrev[n-1]
    termA = conv_awrev[n-1:2*n-1][::-1]
    
    grad_a = 2.0 * (termA + termB)
    
    # Transform to z-space: grad_z[i] = a[i] * (grad_a[i] - sum(grad_a * a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj_val, grad_z

    z = np.log(np.maximum(a_init, 1e-30))

            z = res.x

            score = score_normalized(a)

                best_a = a.copy()
        except Exception:
            pass
    
    return best_a, best_score

def get_lp_solver():
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):

def solve_lp(a_fixed, solver):
    """LP: min M s.t. conv(a_fixed, b)[k] <= M for all k, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build constraint: conv(a_fixed, b)[k] <= M for k=0,...,2n-2
    # conv(a_fixed, b)[k] = sum_j a_fixed[j] * b[k-j] (valid indices)
    # This is a_toeplitz(a_fixed) @ b_var
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val /= s
    return b_val, float(M_var.value)

def subgrad_topk(a, k=5):
    n = len(a)

    grad = np.zeros(n)
    for i in range(k):

        wi = w[i]
        lo1 = max(0, ki - n + 1)
        hi1 = min(n - 1, ki)
        if hi1 >= lo1:

        lo2 = max(0, ki)
        hi2 = min(n - 1, ki + n - 1)
