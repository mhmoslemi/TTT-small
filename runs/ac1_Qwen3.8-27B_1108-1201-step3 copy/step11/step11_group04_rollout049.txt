import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.GUROBI, cp.MOSEK]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                sv = b_val.sum()
                if sv < 1e-15:
                    continue
                b_val = b_val / sv
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        wk = w[k]
        if wk < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += wk * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lbgfs_obj_wrapper(z, n, T_stage, deadline):
    """Objective for L-BFGS."""
    if time.monotonic() > deadline:
        return 1e15
    obj, _ = logsumexp_obj_grad(z, n, T_stage)
    return obj


def lbgfs_jac_wrapper(z, n, T_stage, deadline):
    """Gradient for L-BFGS."""
    if time.monotonic() > deadline:
        return np.zeros_like(z)
    _, g = logsumexp_obj_grad(z, n, T_stage)
    return g


def spectral_flatten(a, n_iters=100, blend=0.4, deadline=None):
    """Iterative spectral flattening (Gerchberg-Saxton style)."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)

        mags = np.abs(F)
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])

        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

    return a


def soft_peak_shave(a, time_limit, n_steps=500000):
    """
    Soft-weighted peak shaving: weights each convolution bin by a smooth
    function of its distance to the maximum, rather than hard top-k selection.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return None, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.015
    scale = 1.0
    no_improve = 0
    width = 0.005  # Width of the soft weighting window

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        min_b = b.min()
        rng_b = max_b - min_b
        if rng_b < 1e-30:
            break

        # Soft weighting: exponential of negative distance to max
        # w[k] = exp(-(max_b - b[k]) / (width * max_b))
        # This gives all bins some weight but heavily favors near-max bins
        w = np.exp(-(max_b - b) / (width * max_b + 1e-30))
        w = w / w.sum()

        g = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-20:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w[k] * a[k - idx]

        eta_eff = eta * scale
        log_scale = -eta_eff * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0

            # Adaptively narrow the window as we converge
            if best_score < 2.0:
                width = max(0.001, width * 0.999)
        else:
            no_improve += 1

        if no_improve >= 40:
            scale *= 0.7
            no_improve = 0
            width = max(0.001, width * 0.5)
            if scale < 0.003:
                break

        if step % 30000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    SoftPeakShave {elapsed:.0f}s: score={best_score:.8f} scale={scale:.4f} width={width:.6f}")

    return best_a, best_score

            continue

def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage, deadline=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit
    n_stages = len(T_stages)

    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break

        stages_left = n_stages - stage_i
        # Allocate time proportionally: give more time to lower temperature stages
        time_frac = 1.0 / stages_left
        avail_time = remaining * time_frac
        iters = max(20, min(maxiter_per_stage, int(avail_time * 80)))

        def func(z_in):
            return lbgfs_obj_wrapper(z_in, n, stage_T, deadline)

        def jac(z_in):
            return lbgfs_jac_wrapper(z_in, n, stage_T, deadline)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-30,
                         'gtol': 1e-20, 'maxfun': iters * 200, 'disp': False}
            )
            z = res.x
            if time.monotonic() > deadline:
                break
        except Exception:
            pass

    return softmax_from_z(z)


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturbed_sequence(a, rng, strength=0.03):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 10)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s

        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    rem = deadline - time.monotonic()
    print(f"Total budget: {rem:.0f}s, starting score: {best_score:.8f}")

    # === Phase 0: Multi-n exploration ===
    print(f"\n=== Phase 0: Multi-n Exploration ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_explore = min(rem * 0.02, 40.0)
        n_candidates = [n_best]
        # Add nearby n values
        for dn in [-200, -100, -50, 50, 100, 200]:
            nc = n_best + dn
            if 500 <= nc <= 2000:
                n_candidates.append(nc)

        best_n = n_best
        best_n_score = best_score

        for nc in n_candidates:
            if time.monotonic() > time.monotonic() + t_explore or time.monotonic() > deadline - 150:
                break
            if nc == n_best:
                continue
            a_interp = interpolate_sequence(best_seq, nc)
            a_spec = spectral_flatten(a_interp, n_iters=30, blend=0.35)
            sc = evaluate_local(a_spec)
            print(f"  n={nc}: score={sc:.8f}")
            if sc < best_n_score:
                best_n_score = sc
                best_n = nc
                best_seq = a_spec.copy()
                best_score = sc

        if best_n != n_best:
            print(f"  Switched to n={best_n}, score={best_score:.8f}")
            n_best = best_n

    # === Phase 1: Spectral Flattening ===
    print(f"=== Phase 1: Spectral Flattening (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_spec = min(rem * 0.06, 90.0)
    if t_spec > 15:
        print(f"  Budget: {t_spec:.0f}s")
        a_spec = spectral_flatten(best_seq, n_iters=300, blend=0.35,
                                  deadline=time.monotonic() + t_spec)
        local_update(a_spec, "spectral")
        a_spec2 = spectral_flatten(best_seq, n_iters=100, blend=0.55,
                                   deadline=time.monotonic() + min(rem * 0.02, 20))
        local_update(a_spec2, "spectral_strong")

    # === Phase 2: Nesterov LP (main workhorse) ===
    print(f"=== Phase 2: Nesterov LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lp = min(rem * 0.30, 540.0)
    if t_lp > 40:
        print(f"  Budget: {t_lp:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_lp, solver, n_max_iters=2000000)
        local_update(a_lp, "nesterov_lp_1")

    # === Phase 3: LogSumExp L-BFGS (ultra-fine) ===
    print(f"=== Phase 3: L-BFGS LogSumExp (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lbfgs = min(rem * 0.25, 400.0)
    if t_lbfgs > 40:
        # Very fine temperature schedule with extended low-temperature tail
        T_stages = [5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25,
                    0.18, 0.13, 0.1, 0.07, 0.05, 0.04, 0.03,
                    0.02, 0.015, 0.012, 0.008, 0.006, 0.004, 0.003,
                    0.002, 0.0015, 0.001, 0.0008, 0.0005]
        print(f"  T_stages: {len(T_stages)} stages (min T={T_stages[-1]}), budget={t_lbfgs:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_lbfgs, T_stages, maxiter_per_stage=30000,
                            deadline=time.monotonic() + t_lbfgs)
        local_update(a_lb, "lbgfs_fine")

    # === Phase 4: Soft Peak Shaving ===
    print(f"=== Phase 4: Soft Peak Shaving (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_peak = min(rem * 0.10, 150.0)
    if t_peak > 15:
        a_pk, _ = soft_peak_shave(best_seq, t_peak, n_steps=500000)
        local_update(a_pk, "soft_peak")

    # === Phase 5: Second Nesterov LP round ===
    print(f"=== Phase 5: Nesterov LP Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lp2 = min(rem * 0.15, 200.0)
    if t_lp2 > 25:
        print(f"  Budget: {t_lp2:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_lp2, solver, n_max_iters=1000000)
        local_update(a_lp2, "nesterov_lp_2")

    # === Phase 6: Perturbation escape + re-optimize ===
    print(f"=== Phase 6: Perturbation Escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        for trial in range(4):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 40:
                break
            strength = 0.01 + 0.008 * trial
            a_pert = perturbed_sequence(best_seq, rng, strength=strength)
            a_sp = spectral_flatten(a_pert, n_iters=20, blend=0.3)
            t_sub = min(rem_sub * 0.2, 60.0)
            if t_sub > 15:
                a_opt, _ = nesterov_lp_refine(a_sp, t_sub, solver, n_max_iters=300000)
                local_update(a_opt, f"pert_trial{trial}")

    # === Phase 7: Try a different n ===
    print(f"=== Phase 7: Try different n (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        # Try n + 200
        n_alt = n_best + 200
        if n_alt <= 2500:
            t_alt = min(rem * 0.12, 100.0)
            if t_alt > 30:
                print(f"  Trying n={n_alt}, budget={t_alt:.0f}s")
                init_alt = interpolate_sequence(best_seq, n_alt)
                a_sp_alt = spectral_flatten(init_alt, n_iters=30, blend=0.3)
                a_alt, _ = nesterov_lp_refine(a_sp_alt, t_alt * 0.7, solver, n_max_iters=200000)
                local_update(a_alt, f"lp_n{n_alt}")

    rem = deadline - time.monotonic()
    if rem > 60:
        n_alt2 = n_best - 150
        if n_alt2 >= 500:
            t_alt2 = min(rem * 0.12, 80.0)
            if t_alt2 > 25:
                print(f"  Trying n={n_alt2}, budget={t_alt2:.0f}s")
                init_alt2 = interpolate_sequence(best_seq, n_alt2)
                a_sp_alt2 = spectral_flatten(init_alt2, n_iters=30, blend=0.3)
                a_alt2, _ = nesterov_lp_refine(a_sp_alt2, t_alt2 * 0.7, solver, n_max_iters=200000)
                local_update(a_alt2, f"lp_n{n_alt2}")

    # === Phase 8: Final L-BFGS polish (extremely fine) ===
    print(f"=== Phase 8: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.5, 120.0)
        if t_fin > 20:
            # Ultra-fine temperature schedule for final polish
            T_stages_fin = [1.0, 0.5, 0.3, 0.2, 0.1, 0.07, 0.05, 0.03,
                            0.02, 0.015, 0.01, 0.007, 0.005, 0.003,
                            0.002, 0.001, 0.0007, 0.0004]
            a_fin = lbgfs_refine(best_seq, t_fin, T_stages_fin, maxiter_per_stage=20000,
                                 deadline=time.monotonic() + t_fin)
            local_update(a_fin, "final_lbgfs")

    # === Phase 9: Final peak shave ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t_ps = min(rem * 0.5, 40.0)
        if t_ps > 8:
            a_ps, _ = soft_peak_shave(best_seq, t_ps, n_steps=100000)
            local_update(a_ps, "final_peak")

    # === Phase 10: Last LP polish ===
    rem = deadline - time.monotonic()
    if rem > 12:
        t_flp = min(rem * 0.6, 25.0)
        if t_flp > 8:
            a_flp, _ = nesterov_lp_refine(best_seq, t_flp, solver, n_max_iters=50000)
            local_update(a_flp, "final_lp")

    # Final verification and output
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}) at {elapsed:.1f}s")
        test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            eval_score = evaluate_sequence(test_seq)
            print(f"Verified with evaluate_sequence: {eval_score:.8f}")
        except Exception as e:
            print(f"Verification error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
