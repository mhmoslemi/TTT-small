def lse_gradient_fft(a, w):
    """Compute gradient of LogSumExp-smoothed max(C) using FFT.
    grad[i] = 2 * sum_k w[k] * a[k-i] (with zero padding)
    This is a cross-correlation, computable via FFT.

    size = 2 * n - 1

    # Or use the fact that it's a "sliding dot product"
    
    # Actually, let's just do it directly for the active window
    # For efficiency, use the fact that a has limited support
    grad = np.zeros(n)
    for i in range(n):
        # sum_k w[k] * a[k-i] where 0 <= k-i < n, i.e., i <= k < i+n
        ks = np.arange(max(0, i), min(size, i + n))
        js = ks - i
        grad[i] = 2.0 * np.dot(w[ks], a[js])
    return grad


def lse_objective_and_grad(a, T):
    n = len(a)
    C = fast_convolve(a)
    max_C = C.max()
    # Numerically stable softmax
    shifted = C - max_C
    exp_C = np.exp(shifted / T)
    w = exp_C / exp_C.sum()
    
    # Objective: (1/T) * log(sum exp(C/T)) = max_C/T + log(sum exp((C-max_C)/T))/T
    logsumexp_val = max_C / T + np.log(exp_C.sum()) / T
    obj = 2.0 * n * logsumexp_val
    
    # Gradient
    grad = np.zeros(n)
    for i in range(n):
        ks = np.arange(max(0, i), min(2*n-1, i+n))
        js = ks - i
        grad[i] = 2.0 * (2.0 * n) / T * np.dot(w[ks], a[js])
    
    return obj, grad
