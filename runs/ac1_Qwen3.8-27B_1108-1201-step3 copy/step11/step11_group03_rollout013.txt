"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using active-set convex QCQP.
The problem is convex (min of max of PSD quadratics over simplex), so
we use an active-set method: iteratively solve the QCQP with the most
active constraints, update the active set, and repeat until convergence.
Also tries multiple sequence lengths and initializations.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_q_matrix(k, n):
    """Dense Q matrix for conv(a,a)[k] = a^T Q a."""
    Q = np.zeros((n, n))
    for j in range(max(0, k - n + 1), min(n - 1, k) + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return Q

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def solve_active_qcqp(a_init, active_set, solver):
    """Solve min t s.t. a^T Q_k a <= t for k in active_set, sum(a)=1, a>=0."""
    n = len(a_init)
    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]
    
    for k in active_set:
        Q = build_q_matrix(int(k), n)
        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    solvers_to_try = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, solver]
    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum < 1e-15:
                    return None, None
                return a_val / s_sum, float(max(0, t_var.value))
        except Exception:
            continue
    return None, None

def active_set_method(a_init, max_iters, max_active, solver, time_limit):
    """Iterative active-set method for the convex QCQP."""
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = evaluate_local(a)
    
    active_set = set()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute current convolution
        b = fast_convolve(a)
        max_b = b.max()
        
        # Update active set: keep positions where b is close to max
        threshold = max_b * 0.999
        candidates = np.where(b >= threshold)[0]
        
        # Also keep previously active positions
        for k in candidates:
            active_set.add(int(k))
        
        # Limit active set size
        if len(active_set) > max_active:
            # Keep the ones with highest b values
            active_scores = {k: b[k] for k in active_set}
            sorted_keys = sorted(active_scores, key=active_scores.get, reverse=True)
            active_set = set(sorted_keys[:max_active])
        
        # Solve QCQP
        a_new, t_val = solve_active_qcqp(a, list(active_set), solver)
        
        if a_new is None:
            break
        
        # Check improvement
        new_score = evaluate_local(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
        
        # Check convergence
        if abs(max_b - t_val) < 1e-8:
            break
    
    return best_a, best_score

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_q_dense(k, n):
    """Dense Q matrix for conv(a,a)[k] = a^T Q a."""
    Q = np.zeros((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return Q


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def solve_qcqp_with_active(a_init, active_set, solver, max_active=40):

        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

def lp_best_response(a_fixed, solver):
    """Given fixed a (sum=1), find b minimizing max conv(a,b)."""

    # Build convolution constraint: (T @ b)[k] <= M for all k
    # conv(a, b)[k] = sum_j a[j] * b[k-j]
    # = sum_j a[max(0, min(n-1, k-j))] * b[j] for appropriate j
    # More precisely: for fixed k, conv(a,b)[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[k-j]*b[j]
    
    # Build sparse matrix
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        for i in range(n):
            k = i + j  # position in convolution where a[i]*b[j] contributes
            T[k, j] += a_fixed[i]
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers_to_try = [solver, cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]
    for s in solvers_to_try:
        try:
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                if b_var.value is not None:
                    b_val = np.asarray(b_var.value, dtype=np.float64)
                    b_val = np.maximum(b_val, 0)
                    s_sum = b_val.sum()
                    if s_sum < 1e-15:
                        return None, None
                    return b_val / s_sum, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            update_best(parent, "parent")
    except Exception:
        pass
    
    n_base = len(parent) if parent is not None else 1200
    
    # Always try uniform (theoretical optimum)
    uniform = np.ones(n_base) / n_base
    update_best(uniform, f"uniform_n{n_base}")
    
    # Phase 1: Active-set QCQP on best
    print(f"\n=== Phase 1: Active-set QCQP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_res, _ = active_set_method(best_seq, max_iters=80, max_active=25, solver=solver, time_limit=t_alloc)
        update_best(a_res, "active_set")
    
    # Phase 2: Nesterov LP alternating
    print(f"\n=== Phase 2: Nesterov LP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 350.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_res, _ = nesterov_lp(best_seq, t_alloc, solver, max_iters=200000)
        update_best(a_res, "nesterov_lp")
    
    # Phase 3: Active-set again (polish)
    print(f"\n=== Phase 3: Active-set Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_alloc = min(rem * 0.15, 150.0)
        a_res, _ = active_set_method(best_seq, max_iters=50, max_active=15, solver=solver, time_limit=t_alloc)
        update_best(a_res, "active_polish")
    
    # Phase 4: Projected gradient for fine-tuning
    print(f"\n=== Phase 4: Projected Gradient (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_alloc = min(rem * 0.10, 100.0)
        a_res, _ = projected_gradient(best_seq, t_alloc, n_steps=500000)
        update_best(a_res, "proj_grad")
    
    # Phase 5: Try other n values
    print(f"\n=== Phase 5: Multi-n exploration (score={best_score:.10f}) ===")
    for n_try in [n_base - 1, n_base + 1, n_base + 50, n_base + 100]:
        if time.monotonic() > deadline:
            break
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        
        init = interpolate_sequence(best_seq, n_try)
        # Also try uniform for this n
        update_best(np.ones(n_try) / n_try, f"uniform_n{n_try}")
        
        t_alloc = min(rem * 0.15, 120.0)
        if t_alloc < 20:
            continue
        
        print(f"  Trying n={n_try}, budget={t_alloc:.0f}s")
        a_res, _ = active_set_method(init, max_iters=30, max_active=15, solver=solver, time_limit=t_alloc)
        update_best(a_res, f"active_n{n_try}")
    
    # Final active-set polish with very tight active set
    print(f"\n=== Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        a_res, _ = active_set_method(best_seq, max_iters=20, max_active=5, solver=solver, time_limit=min(rem, 60))
        update_best(a_res, "final_polish")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        return [float(max(0.0, min(1000.0, x))) for x in best_seq]
    
    # Fallback: uniform
    print("Using fallback uniform sequence")
    return [1.0] * 1200
