    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:

    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed

    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS]:

"""

1. LP-based Nesterov accelerated best-response with adaptive restart
2. FFT-accelerated LogSumExp L-BFGS multi-stage annealing

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver=None, max_iters=1000000, restart_threshold=100):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0
    no_improve_count = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_prev)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a_prev = a_new.copy()
        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score - 1e-12:
            best_score = sc_a
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            no_improve_count = 0

        # Adaptive restart: if no improvement for many iterations, reset momentum
        if no_improve_count > restart_threshold:
            y = a.copy()
            a_prev = a.copy()
            no_improve_count = 0

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 5:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} M_lp={M_lp:.6f}")
                last_print = elapsed

    return best_a, best_score


def lse_objective_grad_fft(z_in, T_val, n):
    """Compute LSE objective and gradient w.r.t. z using FFT-accelerated cross-correlation."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        Tb = Tb - max_Tb + 700
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    # Compute gradient using FFT cross-correlation
    # grad_a_j = 2n * sum_{m=0}^{n-1} w[j+m] * a[m]
    # = 2n * (w cross-correlate a)[j]
    L = 1
    while L < 3 * n:
        L *= 2
    W = rfft(w, L)
    A = rfft(a, L)
    # Cross-correlation: (w ⋆ a)[j] = IFFT(conj(W) * A)[j]
    corr = irfft(np.conj(W) * A, L)
    grad_a = 2.0 * n * corr[:n]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=200.0, T_end=0.05, n_stages=20):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_z = z.copy()
    best_score = score_normalized(a0)

    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = 40

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad_fft(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad_fft(z_val, stage_T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-22,
                         'gtol': 1e-18, 'maxfun': iters_per_stage * 300,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)
            if sc < best_score:
                best_score = sc
                best_z = z.copy()
            if (si + 1) % 5 == 0:
                print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.3f}: score={sc:.8f}")
        except Exception:
            pass

    final_a = softmax_from_z(best_z)
    final_sc = score_normalized(final_a)
    return final_a


def active_set_polish(a0, time_limit, max_rounds=50):
    """Directly reduce the maximum convolution peak using active-set style updates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Find all peaks above 98% of max
        threshold = max_b * 0.98
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # For each active peak k, we want to reduce (a*a)_k
        # (a*a)_k = sum_j a_j * a_{k-j}
        # We can reduce it by decreasing a_j where a_{k-j} is large
        # Use a simple gradient step on the max
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Project gradient to tangent space (sum = 0)
        S = np.dot(g, a)
        grad_tangent = g - S * a  # in simplex tangent space

        # Step in log-parameterization
        z = z_from_a(a)
        # Adaptive step size
        step = 0.1 / (1.0 + rnd * 0.1)
        z_new = z - step * a * grad_tangent
        a_new = softmax_from_z(z_new)
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa > 1e-15:
            a_new = a_new / sa

        sc_new = score_normalized(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            z_new = z - step * 0.1 * a * grad_tangent
            a_new = softmax_from_z(z_new)
            a_new = np.maximum(a_new, 0)
            sa = a_new.sum()
            if sa > 1e-15:
                a_new = a_new / sa
            sc_new = score_normalized(a_new)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                a = a_new
            else:
                # Try increasing some middle elements (where convolution is low)
                mid = b < np.median(b)
                a_temp = a.copy()
                a_temp[mid] += 0.001
                a_temp = np.maximum(a_temp, 0)
                a_temp = a_temp / a_temp.sum()
                sc_temp = score_normalized(a_temp)
                if sc_temp < best_score:
                    best_score = sc_temp
                    best_a = a_temp.copy()
                    a = a_temp

        if (rnd + 1) % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ActiveSet round={rnd+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x ** 2)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 4:
        a = np.ones(n) * (1.0 + 0.2 * np.cos(2 * np.pi * 3 * np.arange(n) / n))
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.15 * np.cos(7 * x)
    elif choice == 6:
        # Smooth random
        k = rng.integers(3, 10)
        freqs = rng.integers(1, 15, size=k)
        phases = rng.uniform(0, 2 * np.pi, size=k)
        amps = rng.uniform(0.1, 1.0, size=k)
        x = np.linspace(0, 2 * np.pi, n)
        a = np.ones(n) * 0.5
        for fi, ph, am in zip(freqs, phases, amps):
            a += am * np.cos(fi * x + ph)
    else:
        # Bimodal
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.3) ** 2) + np.exp(-20 * (x - 0.7) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

        a_lse = lbgfs_logsumexp(best_seq, t2, T_start=150.0, T_end=0.03, n_stages=20)
        try_update(a_lse, "lse_polish")
    print(f"  Phase 2 done: best_score={best_score:.8f}")

    # === Phase 3: Active-set peak shaving (8% of budget) ===
    print(f"\n=== Phase 3: Active-set polish (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t3 = min(0.08 * (budget_s - 30), remaining * 0.4)
    if t3 > 10:
        a_as, _ = active_set_polish(best_seq, t3, max_rounds=200)
        try_update(a_as, "active_set")
    print(f"  Phase 3 done: best_score={best_score:.8f}")

    # === Phase 4: More Nesterov LP (12% of budget) ===
    print(f"\n=== Phase 4: Nesterov LP round 2 (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t4 = min(0.12 * (budget_s - 30), remaining * 0.5)
    if t4 > 10:
        a_out2, _ = nesterov_lp(best_seq, t4, solver=solver)
        try_update(a_out2, "nesterov_lp_r2")
    print(f"  Phase 4 done: best_score={best_score:.8f}")

    # === Phase 5: LSE fine polish (5% of budget) ===
    print(f"\n=== Phase 5: Final LSE fine polish (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t5 = min(0.05 * (budget_s - 30), remaining * 0.5)
    if t5 > 10:
        a_final5 = lbgfs_logsumexp(best_seq, t5, T_start=10.0, T_end=0.01, n_stages=12)
        try_update(a_final5, "final_lse")
    print(f"  Phase 5 done: best_score={best_score:.8f}")

    # === Phase 6: Length exploration (2% of budget) ===
    print(f"\n=== Phase 6: Length exploration (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t6_total = min(0.02 * (budget_s - 30), remaining * 0.5)
    if t6_total > 15:
        for n_try in [n_base + 100, n_base - 100, n_base + 200, n_base - 200]:
            remaining = _global_deadline - time.monotonic()
            if remaining < 15:
                break
            if n_try < 100:
                continue
            t6 = min(t6_total / 4.0, remaining * 0.35)
            if t6 < 5:
                break
            print(f"  Trying n={n_try}, budget={t6:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp(init_try, t6, solver=solver)
            try_update(a_try, f"len_{n_try}")
    print(f"  Phase 6 done: best_score={best_score:.8f}")

    # === Phase 7: Final combined polish ===
    print(f"\n=== Phase 7: Final combined polish (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 20:
        # Quick LSE + active set combo
        t7a = min(remaining * 0.5, 40.0)
        if t7a > 5:
            a_p = lbgfs_logsumexp(best_seq, t7a, T_start=50.0, T_end=0.02, n_stages=10)
            try_update(a_p, "final_combined_1")
        remaining = _global_deadline - time.monotonic()
        if remaining > 10:
            a_p2, _ = active_set_polish(best_seq, remaining * 0.5, max_rounds=100)
            try_update(a_p2, "final_combined_2")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
