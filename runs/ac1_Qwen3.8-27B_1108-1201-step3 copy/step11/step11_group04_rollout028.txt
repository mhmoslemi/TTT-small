- Maintain a "lookahead" point y: z_y = (1-alpha) * z_y + alpha * z_a
- Compute gradient at y: g = grad_z f(a_y)
- Update: z_a = z_a - eta * g
- Update y: z_y = z_y + (1 - alpha) * (z_a - z_{a,prev})

where alpha = 2/(t+2).

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LP alternating with active set reduction, Nesterov acceleration, and HiGHS solver
2. KKT-based support identification and active set refinement
3. Mirror descent (log-domain) with Nesterov momentum for fine polishing
4. Multi-n exploration with interpolation and re-optimization
5. Adaptive time allocation based on convergence progress
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_T_rows(a, active_indices):
    """Build rows of T matrix for given active indices. T @ b = convolve(a, b)."""
    n = len(a)
    m = len(active_indices)
    T_act = np.zeros((m, n))
    k_arr = np.asarray(active_indices)
    # For row k: T[k, j] = a[k-j] if 0 <= k-j < n
    j_arr = np.arange(n)
    idx = k_arr[:, None] - j_arr[None, :]
    mask = (idx >= 0) & (idx < n)
    T_act[mask] = a[idx[mask]]
    return T_act


def solve_best_response(a_fixed, active_indices, tol=1e-12):
    """
    Solve: min t s.t. (T_a @ b)[k] <= t for k in active, sum(b)=1, b>=0.
    Returns (b_opt, t_opt) or (None, None) on failure.
    """
    n = len(a_fixed)
    m = len(active_indices)

    T_act = build_T_rows(a_fixed, active_indices)

    # Variables: [b_0, ..., b_{n-1}, t], minimize t
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.hstack([T_act, -np.ones((m, 1))])
    b_ub = np.zeros(m)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True,
                               'dual_feasibility_tolerance': tol,
                               'primal_feasibility_tolerance': tol,
                               'ipm_gap_tolerance': tol})
        if res.status != 0:
            return None, None
        b_val = np.maximum(res.x[:n], 0)
        t_val = float(res.x[-1])
    except Exception:
        return None, None

    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, max(0.0, t_val)


def get_active_set(b, n_active=200):
    """Get indices of top n_active peaks."""
    size = len(b)
    if n_active >= size:
        return np.arange(size)
    idx = np.argpartition(b, -n_active)[-n_active:]
    return np.sort(idx)


def nesterov_lp_optimize(a0, time_limit, n_active=200, check_interval=100):
    """
    Nesterov-accelerated LP alternating method.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    it = 0

    # Adaptive active set size
    current_n_active = n_active

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Widen active set periodically to escape local structures
        if it % 500 == 0:
            b_check = fast_convolve(y)
            max_b = b_check.max()
            # Use peaks within 1% of max
            threshold = max_b * 0.999
            active_full = np.where(b_check >= threshold)[0]
            if len(active_full) < current_n_active:
                current_n_active = max(50, len(active_full))
            else:
                current_n_active = min(n_active, len(active_full) + 20)

        active = get_active_set(fast_convolve(y), current_n_active)
        b_new, M_b = solve_best_response(y, active, tol=1e-12)

        if b_new is None:
            # Fallback: use subgradient step
            b_y = fast_convolve(y)
            max_b = b_y.max()
            active_idx = get_active_set(b_y, 50)
            g = np.zeros(n)
            for k in active_idx:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                g[js:je+1] += y[je:js-1:-1] if js <= je else 0
            # Simpler: just add noise
            b_new = y + 0.01 * np.random.randn(n)
            b_new = np.maximum(b_new, 0)
            b_new = b_new / b_new.sum()

        # Nesterov dual averaging
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        # Nesterov primal update
        a_momentum = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a_momentum = np.maximum(a_momentum, 0)
        sa = a_momentum.sum()
        if sa < 1e-15:
            a_momentum = y.copy()
        else:
            a_momentum = a_momentum / sa

        a_prev = a.copy()
        a = a_momentum

        # Track best
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Stall detection
        if it % check_interval == 0:
            if best_score < prev_best - 1e-12:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 15:
                # Perturb to escape
                rng_local = np.random.default_rng(it)
                noise = 0.005 * rng_local.standard_normal(n)
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                stall_count = 0
                # Reset momentum
                a = y.copy()
                a_prev = a.copy()

            if it % (check_interval * 10) == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter={it} ({elapsed:.0f}s): score={best_score:.8f} active={current_n_active}")

    return best_a, best_score


def mirror_descent_nesterov(a0, time_limit, eta0=0.1, decay=0.999):
    """
    Mirror descent in log-domain with Nesterov acceleration.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    z = np.log(np.maximum(a, 1e-30))
    z_y = z.copy()  # Nesterov lookahead
    z_prev = z.copy()
    v = np.zeros(n)  # momentum

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0
    it = 0
    stall_count = 0
    prev_best = best_score

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute a from z_y (lookahead)
        z_y_shifted = z_y - z_y.max()
        exp_zy = np.exp(z_y_shifted)
        a_y = exp_zy / exp_zy.sum()

        # Compute subgradient from active peaks
        b_y = fast_convolve(a_y)
        max_b = b_y.max()
        threshold = max_b * (1.0 - 1e-6)
        active = np.where(b_y >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b_y)])

        # Subgradient in a-space
        g_a = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g_a[idx] += a_y[k - idx]
        g_a /= len(active)

        # Convert to z-space gradient (mirror descent)
        S = np.dot(g_a, a_y)
        g_z = a_y * (g_a - S)

        # Nesterov step
        alpha = 2.0 / (it + 2)
        z = z - eta * g_z
        z_y = z + (1.0 - alpha) * (z - z_prev)
        z_prev = z.copy()

        # Clip z for numerical stability
        z = np.clip(z, -50, 50)
        z_y = np.clip(z_y, -50, 50)

        # Compute a from z
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a_new = exp_z / exp_z.sum()

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        # Decrease step size
        eta *= decay
        if eta < 1e-8:
            break

        # Stall check
        if it % 200 == 0:
            if best_score < prev_best - 1e-14:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 10:
                break
            if it % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    MirrorDescent iter={it} ({elapsed:.0f}s): score={best_score:.8f} eta={eta:.2e}")

    return best_a, best_score


def fine_subgradient(a0, time_limit, n_active=50):
    """
    Very fine subgradient method with decreasing step size.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.005
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        b = fast_convolve(a)
        max_b = b.max()

        # Top active peaks
        active = get_active_set(b, n_active)

        # Weighted subgradient (weight by proximity to max)
        weights = b[active]
        weights = np.maximum(weights - max_b + 1e-10, 0)
        if weights.sum() < 1e-15:
            weights = np.ones(len(active))
        weights = weights / weights.sum()

        g = np.zeros(n)
        for i, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += weights[i] * a[k - idx]

        # Multiplicative (mirror) update
        log_scale = -eta * g / max(g.max(), 1e-15)
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new

        # Decrease step size
        if it % 1000 == 0:
            eta *= 0.8
            if eta < 1e-6:
                break

    return best_a, best_score


def kkt_polish(a0, time_limit, n_active_peaks=50, n_support=100):
    """
    KKT-based polishing: identify active sets and solve reduced problems.
    Alternates between:
    - Given support B of a, find optimal active set A
    - Given active set A, find optimal support B
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * n
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for round_i in range(50):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Identify active peaks
        active = get_active_set(b, n_active_peaks)

        # Identify support of a
        support_thresh = max(a.min(), 1e-10)
        support = np.where(a > support_thresh * 0.01)[0]
        if len(support) > n_support:
            # Keep top n_support elements
            top_idx = np.argsort(a)[-n_support:]
            support = np.sort(top_idx)

        # Solve: given support B, find b (dual) that minimizes max over active
        # This is the LP best response restricted to active peaks
        b_sol, M_sol = solve_best_response(a, active, tol=1e-12)
        if b_sol is not None:
            # Update a towards b_sol
            alpha = 0.3 / (1.0 + round_i * 0.1)
            a_new = (1 - alpha) * a + alpha * b_sol
            a_new = np.maximum(a_new, 0)
            # Zero out elements outside support
            a_new[~np.isin(np.arange(n), support)] = 0
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
                score = 2.0 * n * fast_convolve(a_new).max()
                if score < best_score:
                    best_score = score
                    best_a = a_new.copy()
                    a = a_new

        # Also try the reverse: fix b (=a), find optimal a on the support
        # This is: min max_k in active (a_new * a)[k] s.t. sum(a_new)=1, a_new>=0, support(B)
        # Which is the same LP but with roles swapped
        a_sol, M_a = solve_best_response(a, active, tol=1e-12)
        if a_sol is not None:
            a_new = np.zeros(n)
            for j in support:
                a_new[j] = a_sol[j]
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
                score = 2.0 * n * fast_convolve(a_new).max()
                if score < best_score:
                    best_score = score
                    best_a = a_new.copy()
                    a = a_new

        if round_i < 10 and (round_i + 1) % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    KKT iter={round_i} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def spectral_flatten_fast(a, n_iters=30, blend=0.3):
    """Quick spectral flattening."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    for i in range(n_iters):
        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.rfft(a_padded)

        mags = np.abs(F)
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc[non_dc > 1e-15]) if np.any(non_dc > 1e-15) else 1.0/n
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, len(F)):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * phases[k] / mags[k] if mags[k] > 1e-15 else 0

        a_new = np.fft.irfft(F_new, L).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

    return a


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses fast LP alternating, mirror descent, KKT polishing, and multi-n exploration.
    """
    np.random.seed(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = np.ones(n_init) / n_init
        # Add some structure
        x = np.linspace(0, 1, n_init)
        best_seq = 0.5 + 0.3 * np.cos(2 * np.pi * x)
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    rem = deadline - time.monotonic()
    print(f"Total budget: {rem:.0f}s, starting score: {best_score:.8f}")

    # === Phase 1: Quick spectral initialization ===
    print(f"\n=== Phase 1: Spectral Init ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        a_spec = spectral_flatten_fast(best_seq, n_iters=20, blend=0.3)
        local_update(a_spec, "spectral_init")

    # === Phase 2: Main LP Alternation ===
    print(f"=== Phase 2: LP Alternation n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lp1 = min(rem * 0.40, 700.0)
    if t_lp1 > 30:
        print(f"  Budget: {t_lp1:.0f}s")
        a_lp1, _ = nesterov_lp_optimize(best_seq, t_lp1, n_active=200)
        local_update(a_lp1, "lp_main_1")

    # === Phase 3: LP with different active set size ===
    print(f"=== Phase 3: LP Alt (wider) (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lp2 = min(rem * 0.20, 350.0)
    if t_lp2 > 20:
        print(f"  Budget: {t_lp2:.0f}s")
        a_lp2, _ = nesterov_lp_optimize(best_seq, t_lp2, n_active=400)
        local_update(a_lp2, "lp_wider")

    # === Phase 4: Mirror Descent Polish ===
    print(f"=== Phase 4: Mirror Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_md = min(rem * 0.15, 250.0)
    if t_md > 15:
        print(f"  Budget: {t_md:.0f}s")
        a_md, _ = mirror_descent_nesterov(best_seq, t_md, eta0=0.05, decay=0.9995)
        local_update(a_md, "mirror")

    # === Phase 5: Fine Subgradient ===
    print(f"=== Phase 5: Fine Subgradient (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_fs = min(rem * 0.10, 150.0)
    if t_fs > 10:
        print(f"  Budget: {t_fs:.0f}s")
        a_fs, _ = fine_subgradient(best_seq, t_fs, n_active=30)
        local_update(a_fs, "fine_subgrad")

    # === Phase 6: KKT Polish ===
    print(f"=== Phase 6: KKT Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_kkt = min(rem * 0.10, 150.0)
    if t_kkt > 10:
        print(f"  Budget: {t_kkt:.0f}s")
        a_kkt, _ = kkt_polish(best_seq, t_kkt)
        local_update(a_kkt, "kkt_polish")

    # === Phase 7: Multi-n exploration ===
    print(f"=== Phase 7: Multi-n (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    for test_n in [1000, 1440, 1500, 2000]:
        if rem < 60:
            break
        if test_n == n_best:
            continue
        t_sub = min(rem * 0.15, 120.0)
        if t_sub < 30:
            break
        print(f"  Trying n={test_n}, budget={t_sub:.0f}s")
        rem = deadline - time.monotonic()
        init_t = interpolate_sequence(best_seq, test_n)
        # Quick LP
        a_t, _ = nesterov_lp_optimize(init_t, t_sub * 0.7, n_active=150)
        local_update(a_t, f"lp_n{test_n}")
        rem = deadline - time.monotonic()
        if rem > 20:
            a_t2, _ = fine_subgradient(best_seq if len(best_seq) == test_n else a_t,
                                       min(rem * 0.3, 30.0), n_active=30)
            local_update(a_t2, f"fine_n{test_n}")

    # === Phase 8: Final combined polish ===
    print(f"=== Phase 8: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # One more LP round
        t_final_lp = min(rem * 0.4, 60.0)
        if t_final_lp > 10:
            a_fin, _ = nesterov_lp_optimize(best_seq, t_final_lp, n_active=100)
            local_update(a_fin, "final_lp")

        rem = deadline - time.monotonic()
        if rem > 15:
            a_fin2, _ = fine_subgradient(best_seq, min(rem * 0.4, 20.0), n_active=20)
            local_update(a_fin2, "final_fine")

        rem = deadline - time.monotonic()
        if rem > 10:
            a_fin3, _ = mirror_descent_nesterov(best_seq, min(rem * 0.5, 15.0),
                                                 eta0=0.01, decay=0.9999)
            local_update(a_fin3, "final_mirror")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}) at {elapsed:.1f}s")
        test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            eval_score = evaluate_sequence(test_seq)
            print(f"Verified with evaluate_sequence: {eval_score:.8f}")
        except Exception as e:
            print(f"Verification error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
