import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def convolve_ab(a, b):
    """Compute convolve(a, b) efficiently."""
    return np.convolve(a, b)


def get_lp_solver():
    """Find a working LP solver, preferring fast accurate ones."""
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def subgradient_of_max_conv(a):
    """
    Compute a subgradient of f(a) = max_k conv(a,a)[k] at point a.
    Returns (subgradient_vector, max_value, argmax_indices).
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    
    # Use all indices within 1% of max for a smoother subgradient
    threshold = max_b * 0.999
    active = np.where(b >= threshold)[0]
    
    # Weight by proximity to max
    weights = (b[active] - threshold) / max(1e-30, max_b - threshold)
    if weights.sum() < 1e-30:
        weights = np.ones(len(active))
    weights = weights / weights.sum()
    
    g = np.zeros(n)
    for i, k in enumerate(active):
        w = weights[i]
        for j in range(n):

                else:
                    g[j] += w * 2.0 * a[idx]  # off-diagonal: 2 * a_j * a_idx
    
    return g, max_b, active

        # Gradient step
        a_new = y - eta * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        # Nesterov update
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * a_new
        
        # Track best
        b_new = fast_convolve(a_new)
        score = 2.0 * n * b_new.max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        if (it + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def fista_lp_alternate(a0, time_limit, solver, n_max_iters=1000000):
    """
    FISTA-accelerated LP best-response alternating method.
    At each step: solve LP for best response, then take FISTA step.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    lp_times = []
    no_improve = 0
    restart_count = 0
    prev_best = best_score
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP for best response to y
        lp_t0 = time.monotonic()
        b_new, M_b = solve_best_response_lp(y, solver)
        lp_time = time.monotonic() - lp_t0
        lp_times.append(lp_time)
        
        if b_new is None:
            continue
        
        # FISTA step: a_new = P(y - eta * (a - b_new))
        # Actually for this problem, the "gradient" of the LP objective at a is related to b_new
        # The best response b_new minimizes max_k conv(a_fixed, b), so the "descent direction" is b_new - a
        # Use a simple mixing with decreasing step
        
        # Adaptive step: start aggressive, decrease if no improvement
        alpha = 2.0 / (np.sqrt(it + 1) + 1)
        
        y_new = y + alpha * (b_new - y)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        # Also try the direct update
        a_new = a + alpha * (b_new - a)
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / sa
        
        # Evaluate both
        score_y = 2.0 * n * fast_convolve(y_new).max()
        score_a = 2.0 * n * fast_convolve(a_new).max()
        
        if score_y < best_score:
            best_score = score_y
            best_a = y_new.copy()
            no_improve = 0
        if score_a < best_score:
            best_score = score_a
            best_a = a_new.copy()
            no_improve = 0
        
        a = a_new
        y = y_new
        
        # Check for stall
        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-12:
                prev_best = best_score
            else:
                no_improve += 1
                if no_improve >= 10:
                    # Restart
                    y = best_a.copy()
                    a = best_a.copy()
                    no_improve = 0
                    restart_count += 1
                    print(f"    FISTA restart #{restart_count} at iter {it+1}, score={best_score:.8f}")
        
        # Monitor LP speed
        if len(lp_times) >= 50 and np.mean(lp_times[-20:]) > 10.0:
            print(f"    LP too slow ({np.mean(lp_times[-20:]):.2f}s/iter), stopping at iter {it+1}")
            break
        
        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            avg_lp = np.mean(lp_times[-10:]) if len(lp_times) >= 10 else 0
            print(f"    FISTA-LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} lp_avg={avg_lp:.3f}s")
    
    return best_a, best_score


def spectral_flatten(a, n_iters=100, blend=0.4, deadline=None):
    """Gerchberg-Saxton style spectral flattening."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)
        
        mags = np.abs(F)
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        
        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])
        
        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new
        
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a


def peak_equalize(a, time_limit, n_steps=100000):
    """
    Iteratively equalize the autocorrelation by reducing peaks and 
    boosting valleys. Uses a water-filling approach.
    """
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    eta = 0.01
    scale = 1.0
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        mean_b = b.sum() / len(b)
        
        # Target: reduce peaks above mean, boost valleys below mean
        # Weight = how much above the target level
        target_level = max_b * 0.95  # work on the top 5%
        excess = np.maximum(b - target_level, 0)
        total_excess = excess.sum()
        
        if total_excess < 1e-15:
            target_level = max_b * 0.99
            excess = np.maximum(b - target_level, 0)
            total_excess = excess.sum()
        
        if total_excess < 1e-15:
            break
        
        # Subgradient direction: for each peak k with excess[k], we want to 
        # reduce a to lower conv(a,a)[k]
        # ∂conv(a,a)[k]/∂a_j = a[k-j] (times 2 if j != k-j)
        g = np.zeros(n)
        active = np.where(excess > 1e-15)[0]
        for k in active:
            w = excess[k]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                idx = k - j
                if j == idx:
                    g[j] += w * a[j]
                else:
                    g[j] += w * 2.0 * a[idx]
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        
        # Update
        eta_eff = eta * scale
        a_new = a - eta_eff * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 100:
            scale *= 0.7
            no_improve = 0
            if scale < 0.001:
                break
    
    return best_a


def lbgfs_logsumexp(a0, time_limit, T_stages, maxiter_per_stage, deadline=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    
    def softmax_from_z(z):
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        s2 = exp_z.sum()
        if s2 < 1e-300:
            return np.ones(len(z)) / len(z)
        return exp_z / s2
    
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit
    
    n_stages = len(T_stages)
    
    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break
        
        stages_left = n_stages - stage_i
        iters = max(10, min(maxiter_per_stage, int(remaining * 30 / max(1, stages_left))))
        
        def func(z_in):
            if time.monotonic() > deadline:
                return 1e15
            a_in = softmax_from_z(z_in)
            b_in = fast_convolve(a_in)
            max_b = b_in.max()
            exp_terms = np.exp((b_in - max_b) / stage_T)
            lse = max_b + stage_T * np.log(exp_terms.sum())
            return 2.0 * n * lse
        
        def jac(z_in):
            if time.monotonic() > deadline:
                return np.zeros_like(z_in)
            a_in = softmax_from_z(z_in)
            b_in = fast_convolve(a_in)
            max_b = b_in.max()
            exp_terms = np.exp((b_in - max_b) / stage_T)
            w = exp_terms / exp_terms.sum()
            
            grad_a = np.zeros(n)
            for k in range(len(b_in)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += w[k] * a_in[k - idx]
            
            S = np.dot(grad_a, a_in)
            grad_z = a_in * (grad_a - S)
            return 2.0 * n * grad_z
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': iters * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring fast accurate ones."""
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:

def solve_best_response_lp(a_fixed, solver):
    """
    Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k].
    Returns (b_val, t_val, dual_vars) where dual_vars are the optimal dual
    multipliers for the constraint T(a)b <= t*1.
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()

    # T(a) @ b <= t * ones
    constraints = [T_mat @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    solvers_to_try = [solver, cp.PDLP, cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.ECOS, cp.SCS]
    solved = False
    dual_y = None
    
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                # Try to extract dual variables
                dual_constraints = constraints
                if len(dual_constraints) >= 1:
                    try:
                        dv = dual_constraints[0].dual_value
                        if dv is not None:

                    except Exception:
                        pass
                break

            continue

        return None, None, None
    b_val = b_val / s

def subgradient_from_conv(a):
    """
    Compute subgradient of f(a) = max_k conv(a,a)[k] at point a.
    Uses all peaks within 0.1% of max for smoother gradient.
    Returns (gradient, max_value).
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    
    threshold = max_b * 0.9995
    active = np.where(b >= threshold)[0]
    
    if len(active) == 0:
        active = np.array([np.argmax(b)])
    
    # Soft weights
    weights = b[active] - threshold
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(len(active))
        w_sum = len(active)
    weights = weights / w_sum
    
    g = np.zeros(n)
    for i, k in enumerate(active):
        w = weights[i]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            idx = k - j
            if j == idx:
                g[j] += w * a[idx]
            else:
                g[j] += w * 2.0 * a[idx]
    
    return g, max_b


def nesterov_subgradient(a0, time_limit, n_max_iters=500000):
    """
    Nesterov-accelerated subgradient descent on f(a) = max_k conv(a,a)[k]
    projected onto {a >= 0, sum(a) = 1}.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    C_init = 0.5 * n * best_score / max(n, 1)
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        g, max_b = subgradient_from_conv(y)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-30:
            break
        g = g / g_norm
        
        eta = C_init / np.sqrt(it + 1)
        
        a_new = y - eta * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * a_new
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        if (it + 1) % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score

def mirror_descent_lp(a0, time_limit, solver, n_max_iters=200000):

    no_improve = 0

        # Get best response

        if b_new is None:
            continue
        
        # Mirror descent step: a_new = a * (b_new/a)^(eta)
        # This is the KL proximal step
        ratio = b_new / np.maximum(a, 1e-30)

def spectral_flatten(a, n_iters=100, blend=0.4, deadline=None):
    """Gerchberg-Saxton style spectral flattening."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)
        
        mags = np.abs(F)
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        
        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])
        
        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new
        
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a

def lbgfs_logsumexp(a0, time_limit, T_stages, maxiter_per_stage, deadline=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    
    def softmax_from_z(z):
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        s2 = exp_z.sum()
        if s2 < 1e-300:
            return np.ones(len(z)) / len(z)
        return exp_z / s2
    
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit
    
    n_stages = len(T_stages)
    
    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break
        
        stages_left = n_stages - stage_i
        iters = max(15, min(maxiter_per_stage, int(remaining * 40 / max(1, stages_left))))
        
        def func(z_in):
            if time.monotonic() > deadline:
                return 1e15
            a_in = softmax_from_z(z_in)
            b_in = fast_convolve(a_in)
            max_b = b_in.max()
            exp_terms = np.exp((b_in - max_b) / stage_T)
            lse = max_b + stage_T * np.log(exp_terms.sum())
            return 2.0 * n * lse
        
        def jac(z_in):
            if time.monotonic() > deadline:
                return np.zeros_like(z_in)
            a_in = softmax_from_z(z_in)
            b_in = fast_convolve(a_in)
            max_b = b_in.max()
            exp_terms = np.exp((b_in - max_b) / stage_T)
            w = exp_terms / exp_terms.sum()
            
            grad_a = np.zeros(n)
            for k in range(len(b_in)):
                if w[k] < 1e-12:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx_arr = np.arange(js, je + 1)
                grad_a[idx_arr] += w[k] * a_in[k - idx_arr]
            
            S = np.dot(grad_a, a_in)
            grad_z = a_in * (grad_a - S)
            return 2.0 * n * grad_z
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-25,
                         'gtol': 1e-20, 'maxfun': iters * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
