def fast_convolve(a):
    """Compute linear convolution of a with itself using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1 and a>=0: returns 2*n*max(conv(a,a))."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):

def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Log of a (for log-parametrization)."""
    return np.log(np.maximum(a, 1e-300))


def build_conv_matrix(a_fixed):
    """Build the matrix T such that (T @ b)_k = (a_fixed * b)_k (linear convolution)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n), dtype=np.float64)
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


_solver = None

                return _solver

def solve_lp_best_response(a_fixed, T, solver=None):

    if solver is None:

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:

                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break

                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, prob
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:

    b_val = b_val / s
    return b_val, float(M_var.value), prob


def nesterov_lp_optimize(a0, T, time_limit, solver=None, max_iters=500000,

def lse_objective_grad(z_in, T_val, n):
    """LogSumExp approximation of max(conv(a,a)) with gradient in z-space."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    # Gradient with respect to a
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    # Convert to z-space gradient (softmax Jacobian)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=100.0, T_end=0.05, n_stages=20):
    """Multi-stage LogSumExp L-BFGS optimization."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = np.asarray(a0, dtype=np.float64)
    best_score = score_normalized(best_a)

    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = 50

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 500,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(res.x)
            sc = score_normalized(a_check)
            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()
            if (si + 1) % 5 == 0:
                print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.2f}: score={sc:.8f}")
        except Exception as e:
            print(f"    LSE stage={si+1} error: {e}")

    return best_a, best_score


def peak_shaving(a0, time_limit, n_steps=100000):
    """
    Targeted peak reduction: at each step, identify the peak in conv(a,a)
    and take a step to reduce it.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step % 5000 == 0 and step > 0:
            remaining = time_limit - (time.monotonic() - t0)
            if remaining < 1:
                break

        b = fast_convolve(a)
        max_b = b.max()
        # Find positions where conv is near the max
        threshold = max_b * 0.9
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Compute subgradient of max at the active set
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Project gradient onto the simplex tangent space
        S = np.dot(g, a)
        grad_z = a * (g - S)

        # Adaptive step size
        eta = 0.1 / (1.0 + step * 0.001)
        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)
        sc_new = score_normalized(a_new)
        
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            z = z_new
            a = a_new
        else:
            # Backtrack
            eta = eta * 0.5
            if eta < 1e-10:
                break
            z_new = z - eta * grad_z
            a_new = softmax_from_z(z_new)
            sc_new = score_normalized(a_new)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                z = z_new
                a = a_new

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a different length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        a = np.exp(-4 * (x - 0.5) ** 2)
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 4:
        a = np.ones(n) * (1.0 + 0.3 * np.cos(2 * np.pi * 2 * x))
    elif choice == 5:
        # U-shaped (high at ends)
        a = (2 * x - 1) ** 2
    elif choice == 6:
        # Multi-peak
        a = np.exp(-8*(x-0.25)**2) + np.exp(-8*(x-0.75)**2)
    else:
        # Random
        a = np.abs(rng.standard_normal(n)) + 0.1
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def nesterov_with_rebuild(a0, time_limit, solver=None, max_iters=500000, verbose=True):
    """
    Nesterov LP optimization that rebuilds the convolution matrix each iteration.
    This is the correct approach since T depends on the current y.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0
    lp_count = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Build T from current y
        T_y = build_conv_matrix(y)
        
        # Solve LP
        b_new, M_lp, _ = solve_lp_best_response(y, T_y, solver)
        if b_new is None:
            continue

        lp_count += 1
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if verbose and lp_count % 500 == 1:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 5:
                print(f"    NLP it={it+1} LPs={lp_count} ({elapsed:.0f}s): "
                      f"score={best_score:.8f} M={M_lp:.6f} alpha={alpha:.4f}")
                last_print = elapsed

    return best_a, best_score, lp_count

def fast_convolve(a):
    """Compute linear autocorrelation (a*a) using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1 and a>=0: returns 2*n*max(conv(a,a))."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

    # Bounds: b_i >= 0, M >= 0
    bounds = [(0.0, None)] * (n + 1)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True})

    if res.success:
        b_val = res.x[:n].copy()
        M_val = float(res.x[n])
        b_val = np.maximum(b_val, 0)
        s = b_val.sum()
        if s > 1e-15:
            b_val /= s
            return b_val, M_val
    return None, None

    """
    Nesterov accelerated LP-based optimization.
    Each iteration: solve LP best response given y, then update via Nesterov scheme.
    """

    s = a.sum()
    if s < 1e-15:

    best_a = a.copy()
    best_score = score_normalized(a)

    last_print = t0
    lp_count = 0
    failed_count = 0

        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        if b_new is None:
            failed_count += 1
            if failed_count > 100:
                break
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new /= sa

        y_new = np.maximum(y_new, 0)

        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new /= sy

        y = y_new

            best_score = sc_a
