import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def conv_matrix_vec(a, v):
    """Compute convolve(a, v) efficiently using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    V = rfft(v, fft_size)
    return irfft(A * V, fft_size)[:size]

def build_conv_matrix(a):
    """Build matrix T such that T @ v = convolve(a, v)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def get_lp_solver():
    for name in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return name
        except Exception:
            continue
    return cp.SCS

def solve_best_response_lp(a_fixed, solver, prev_b=None):
    """Solve: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    T = build_conv_matrix(a_fixed)
    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [T @ b <= M, cp.sum(b) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [solver, cp.PDLP, cp.CLARABEL, cp.SCS]
    seen = set()
    for s in solvers:
        if s is None or s in seen:
            continue
        seen.add(s)
        try:
            if prev_b is not None:
                b.value = prev_b
                prob.solve(solver=s, warm_start=True)
            else:
                prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate') and b.value is not None:
                b_val = np.asarray(b.value, dtype=np.float64).flatten()
                b_val = np.maximum(b_val, 0)
                b_val[np.isnan(b_val)] = 0
                s_val = b_val.sum()
                if s_val < 1e-15:
                    continue
                b_val = b_val / s_val
                return b_val
        except Exception:
            continue
    return None

def lp_alternating(a0, time_limit, solver, verbose=True):
    """Run LP alternating until time limit."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    b = a.copy()  # Initialize b same as a
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    stall_count = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Solve for best response b given a
        b_new = solve_best_response_lp(a, solver, b)
        if b_new is None:
            stall_count += 1
            if stall_count > 10:
                break
            continue
        
        # Update a as best response to b (solve another LP)
        a_new = solve_best_response_lp(b_new, solver, a)
        if a_new is None:
            a_new = b_new  # Fallback
        
        # Evaluate
        score_a = 2.0 * n * fast_convolve(a_new).max()
        score_b = 2.0 * n * fast_convolve(b_new).max()
        
        if score_a <= score_b:
            a = a_new
            b = b_new
        else:
            a = b_new
            b = a_new
        
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP[{it}] ({elapsed:.0f}s): score={best_score:.8f} stall={stall_count}")
        
        if stall_count > 200:
            if verbose:
                print(f"    LP stalled at iter {it}, score={best_score:.8f}")
            break
    
    return best_a, best_score

        steps_estimate = int(target_time * 3000)  # ~3000 steps/sec estimate
        steps = max(100, min(50000, steps_estimate))

            if time.monotonic() - t0 > time_limit:

            a = a / a.sum()
            b = fast_convolve(a)
            score = 2.0 * n * b.max()

                best_score = score
                best_a = a.copy()

                max_b = b.max()
                thresh = max_b * (1 - 1e-8)
                active = np.where(b >= thresh)[0]
                if len(active) == 0:
                    active = np.array([np.argmax(b)])
                w = np.zeros(len(b))
                w[active] = 1.0 / len(active)
            else:

                b_top = b[idx]
                max_top = b_top.max()
                w_top = np.exp((b_top - max_top) / T)
                w = np.zeros(len(b))
                w[idx] = w_top / w_top.sum()

- For all b with sum=1, b>=0: max_k (a* * b)_k >= max_k (a* * b*)_k = M*

And M* = max_k (a* * b*)_k <= max_k (b* * b*)_k (by the first inequality with a = b*).

"""
Hybrid LP-Adam optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses intensive LP alternating (Nash equilibrium approach) combined with
Adam logsumexp refinement, multi-n exploration, and symmetrization.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ v = convolve(a, v)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for name in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return name
        except Exception:
            continue
    return cp.SCS


def solve_best_response(a_fixed, solver, prev_b=None):
    """Solve: min M s.t. conv(a_fixed, b)_k <= M for all k, sum(b)=1, b>=0."""
    n = len(a_fixed)
    a_fixed = np.asarray(a_fixed, dtype=np.float64)
    T = build_conv_matrix(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.PDLP, cp.CLARABEL, cp.SCS]
    seen = set()
    for s in solvers_to_try:
        if s is None or s in seen:
            continue
        seen.add(s)
        try:
            if prev_b is not None and len(prev_b) == n:
                b_var.value = prev_b
                prob.solve(solver=s, verbose=False, warm_start=True)
            else:
                prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                b_val = np.asarray(b_var.value, dtype=np.float64).flatten()
                b_val = np.maximum(b_val, 0)
                b_val[~np.isfinite(b_val)] = 0
                s_val = b_val.sum()
                if s_val < 1e-15:
                    continue
                b_val = b_val / s_val
                return b_val
        except Exception:
            continue
    return None


def lp_alternating(a0, time_limit, solver, verbose=True):
    """Run LP alternating (Nash equilibrium) until time limit."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    b = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    stall_count = 0
    it = 0
    mix_factor = 1.0  # Full update initially

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Solve for best response b given current a
        b_new = solve_best_response(a, solver, b)
        if b_new is None:
            stall_count += 1
            if stall_count > 20:
                break
            continue

        # Solve for best response a given new b
        a_new = solve_best_response(b_new, solver, a)
        if a_new is None:
            a_new = b_new

        # Adaptive mixing: reduce update size if stalled
        if stall_count > 50:
            mix_factor = max(0.3, mix_factor * 0.9)
        elif best_score < 2.0 * n * fast_convolve(a_new).max():
            mix_factor = min(1.0, mix_factor * 1.05)

        # Update using relaxed step
        a_candidate = (1.0 - mix_factor) * a + mix_factor * a_new
        a_candidate = np.maximum(a_candidate, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            a_candidate = a_new
        else:
            a_candidate = a_candidate / sc

        b_candidate = (1.0 - mix_factor) * b + mix_factor * b_new
        b_candidate = np.maximum(b_candidate, 0)
        sb = b_candidate.sum()
        if sb < 1e-15:
            b_candidate = b_new
        else:
            b_candidate = b_candidate / sb

        score_a = 2.0 * n * fast_convolve(a_candidate).max()
        score_b = 2.0 * n * fast_convolve(b_candidate).max()

        if score_a <= score_b:
            a = a_candidate
            b = b_candidate
            cur_score = score_a
        else:
            a = b_candidate
            b = a_candidate
            cur_score = score_b

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        if verbose and it % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it} ({elapsed:.0f}s): best={best_score:.8f} cur={cur_score:.8f} stall={stall_count}")

        if stall_count > 500:
            if verbose:
                print(f"    LP converged/stalled at it={it}, best={best_score:.8f}")
            break

    return best_a, best_score


def adam_logsumexp(a0, time_limit, T_list, lr_init=0.01, lr_decay=0.95, top_k_frac=0.3):
    """Adam optimization in log-space with logsumexp smoothing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z -= z.mean()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    step = 0

    best_a = np.asarray(a0, dtype=np.float64).copy()
    best_score = 2.0 * n * fast_convolve(best_a).max()
    t0 = time.monotonic()

    total_stages = len(T_list)

    for stage, T in enumerate(T_list):
        if time.monotonic() - t0 > time_limit:
            break

        remaining_stages = total_stages - stage
        time_per_stage = (time_limit - (time.monotonic() - t0)) / max(1, remaining_stages)
        steps = max(500, int(time_per_stage * 2000))
        steps = min(steps, 100000)

        for inner in range(steps):
            if time.monotonic() - t0 > time_limit:
                break
            step += 1

            a = np.exp(z)
            a = a / a.sum()
            b = fast_convolve(a)
            score = 2.0 * n * b.max()

            if score < best_score:
                best_score = score
                best_a = a.copy()

            # Compute soft-max weights on b
            if T < 0.005:
                max_b = b.max()
                thresh = max_b - 1e-10
                active = np.where(b >= thresh)[0]
                if len(active) == 0:
                    active = np.array([np.argmax(b)])
                w = np.zeros(len(b))
                w[active] = 1.0 / len(active)
            else:
                top_k = max(20, int(top_k_frac * len(b)))
                if top_k < len(b):
                    idx = np.argpartition(b, -top_k)[-top_k:]
                    b_top = b[idx]
                    max_top = b_top.max()
                    w_top = np.exp((b_top - max_top) / T)
                    w = np.zeros(len(b))
                    w[idx] = w_top / w_top.sum()
                else:
                    max_b = b.max()
                    w = np.exp((b - max_b) / T)
                    w = w / w.sum()

            # Gradient: d/da_i [LSE(b;T)] = (1/T) * sum_k w_k * 2*a_{k-i}
            # So grad of (2n * LSE) w.r.t. a_i = (4n/T) * sum_k w_k * a_{k-i}
            # Compute R[i] = sum_k w_k * a[k-i] via FFT cross-correlation
            a_flip = a[::-1]
            fft_size = 1 << (3 * n).bit_length()
            W_f = rfft(w, fft_size)
            Af_f = rfft(a_flip, fft_size)
            R_full = irfft(W_f * Af_f, fft_size)
            # R_full[n-1+i] = sum_k w_k * a[k-i]
            R = R_full[n - 1:n - 1 + n]

            grad_a = (4.0 * n / T) * R

            # Convert to z-space: grad_z = a * (grad_a - a^T * grad_a)
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            # Adam step
            lr = lr_init * (lr_decay ** stage) / (1.0 + 0.02 * inner / max(1, steps))
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * grad_z ** 2
            m_hat = m / (1.0 - beta1 ** (step + 1))
            v_hat = v / (1.0 - beta2 ** (step + 1))
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            z -= z.mean()

            if step % 10000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Adam st={stage} T={T:.3f} step={step} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def score_of(a):
    """Compute the objective score."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    return float(2.0 * n * fast_convolve(a).max() / (s * s))


def symmetrize(a):
    """Average a with its reversal (symmetrization)."""
    return 0.5 * (a + a[::-1])


def interpolate_seq(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a = a / s
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s_new = a_new.sum()
    if s_new > 1e-15:
        a_new = a_new / s_new
    return a_new


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x + rng.uniform(0, 2))

        a = 0.5 + 0.2 * np.cos(2 * np.pi * 7 * x) + 0.1 * np.cos(2 * np.pi * 13 * x + 0.5)
    elif choice == 4:
        a = (1 - (2 * x - 1) ** 4) ** 0.5
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

    except Exception:
        pass

    return result
