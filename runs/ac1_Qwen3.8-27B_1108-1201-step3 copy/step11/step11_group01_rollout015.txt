        active = np.array([int(np.argmax(b))])

    g = np.zeros(n)
    for k in active:

        je = min(n - 1, k)

        valid = (ik >= 0) & (ik < n)
        if np.any(valid):

    g /= len(active)
    return g

def build_conv_matrix(a_fixed):

            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver=None, frac=1.0, max_iters=1000000):
    """Nesterov accelerated LP best-response (Euclidean geometry)."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0
    
    y = a.copy()
    a_prev = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_lp = solve_lp_best_response(y, frac=frac, solver=solver)
        if b_new is None:
            continue
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) ** 2)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k
        
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = normalize(a_new)
        
        y_new = a_new + momentum * (a_new - a_prev)
        y_new = normalize(y_new)
        
        a_prev = a.copy()
        a = a_new
        y = y_new
        
        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 15:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
                last_print = elapsed
    
    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    """LogSumExp smoothed objective and gradient."""
    a = softmax_from_log(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp
    
    # Gradient w.r.t. a
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    # Also need the a[i-k] part
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k)
        je = min(n - 1, k + n - 1)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[idx - k]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=100.0, T_end=0.01, n_stages=20):
    """L-BFGS on LogSumExp smoothed objective with multi-stage annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = 50
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj
        
        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad
        
        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-25,
                         'gtol': 1e-18, 'maxfun': iters_per_stage * 300,
                         'disp': False}
            )
            z = res.x
            if (si + 1) % 5 == 0:
                a_check = softmax_from_log(res.x)
                sc = score_normalized(a_check)
                print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.4f}: score={sc:.8f}")
        except Exception:
            pass
    
    return softmax_from_log(z)


def coordinate_descent_polish(a0, time_limit, n_trials_per_coord=3):
    """
    Coordinate descent: for each coordinate, try moving mass to other coordinates.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    # Precompute which pairs to try
    # For each i, try moving a small fraction to j
    fractions = [0.01, 0.05, 0.1, 0.2]
    
    n_outer = 0
    while time.monotonic() - t0 < time_limit:
        n_outer += 1
        improved = False
        
        # Random subset of coordinates for speed
        if n > 200:
            coords = rng.choice(n, size=min(n, 200), replace=False)
        else:
            coords = np.arange(n)
        
        for i in coords:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-8:
                continue
            
            # Try moving mass from i to a random set of targets
            targets = rng.choice(n, size=min(n, 20), replace=False)
            
            for frac in fractions[:2]:  # Use smaller fractions for speed
                for j in targets:
                    if j == i:
                        continue
                    delta = frac * a[i]
                    a_trial = a.copy()
                    a_trial[i] -= delta
                    a_trial[j] += delta
                    if a_trial[i] < 0:
                        continue
                    a_trial = normalize(a_trial)
                    sc = score_normalized(a_trial)
                    if sc < best_score - 1e-10:
                        best_score = sc
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved and n_outer > 50:
            break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)


def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x ** 2)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 4:
        a = np.ones(n) * (1.0 + 0.2 * np.cos(2 * np.pi * 3 * np.arange(n) / n))
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.15 * np.cos(7 * x)
    elif choice == 6:
        # Asymmetric: more weight on one side
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    else:
        # Two-toned
        x = np.linspace(0, 1, n)
        a = 0.4 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.2 * np.cos(2 * np.pi * 5 * x)
    a = np.maximum(a, 0)
    return normalize(a)


def perturbed_sequence(a, rng, scale=0.05):
    """Create a perturbed version of a sequence."""
    n = len(a)
    noise = rng.normal(0, scale, size=n)
    a_pert = a + noise
    return normalize(a_pert)
