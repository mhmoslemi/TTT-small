def fast_convolve(a):
    """Compute linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def evaluate_safe(a):

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)

def z_from_a(a):

def build_conv_matrix(a_fixed):
    """Build matrix T such that T @ b = conv(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def solve_lp_best_response(a_fixed, solver=None):
    """Solve LP: min M s.t. T_a @ b <= M, sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)

    b_val = np.asarray(b_var.value, dtype=np.float64)

    s = b_val.sum()

def nesterov_lp(a0, time_limit, solver=None, max_iters=1000000):
    """Nesterov accelerated LP best-response method."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:

    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0

        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 15:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
                last_print = elapsed

    return best_a, best_score

def lse_objective_grad(z_in, T_val, n):
    """Compute log-sum-exp objective and gradient w.r.t. z."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def lbgfs_logsumexp(a0, time_limit, T_start=100.0, T_end=0.01, n_stages=25):
    """Multi-stage L-BFGS on log-sum-exp approximation."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = 50

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-18, 'maxfun': iters_per_stage * 500,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

        a_check = softmax_from_z(z)
        sc = score_normalized(a_check)
        if (si + 1) % 5 == 0 or si == n_stages - 1:

def subgradient_nesterov(a0, time_limit, n_steps=200000, eta_start=0.3, eta_end=0.001):
    """Subgradient descent with Nesterov momentum on the max-conv objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    v = np.zeros(n)  # momentum
    beta = 0.9

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Use weighted active set (positions within 99% of max)
        threshold = max_b * 0.99
        active = b >= threshold
        if not active.any():
            active[np.argmax(b)] = True
        
        # Compute subgradient
        g = np.zeros(n)
        for k in np.where(active)[0]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= active.sum()

        S = np.dot(g, a)
        grad_z = a * (g - S)

        eta = eta_start * (eta_end / eta_start) ** (step / max(1, n_steps))
        
        # Nesterov-style momentum
        v = beta * v - eta * grad_z
        z_new = z + v
        a_new = softmax_from_z(z_new)

        z = z_new
        a = a_new
        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad iter={step+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def fast_convolve(a):
    """Compute linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)

def softmax_from_z(z):
    """Compute softmax of z, handling underflow."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Compute log of a, handling zeros."""
    return np.log(np.maximum(a, 1e-300))

    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver=None, max_iters=1000000):
    """Nesterov accelerated LP best-response method."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 15:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
                last_print = elapsed

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    """Compute log-sum-exp objective and gradient w.r.t. z."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_func_wrapper(z_val, T_val, n):
    """Wrapper for scipy minimize fun."""
    obj, _ = lse_objective_grad(z_val, T_val, n)
    return obj


def lse_jac_wrapper(z_val, T_val, n):
    """Wrapper for scipy minimize jac."""
    _, grad = lse_objective_grad(z_val, T_val, n)
    return grad


def lbgfs_logsumexp(a0, time_limit, T_start=50.0, T_end=0.005, n_stages=40):
    """Multi-stage L-BFGS on log-sum-exp approximation with very low final temps."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = 80

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        try:
            res = minimize(
                fun=lse_func_wrapper, x0=z, jac=lse_jac_wrapper,
                args=(stage_T, n), method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-18, 'maxfun': iters_per_stage * 500,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

        a_check = softmax_from_z(z)
        sc = score_normalized(a_check)
        if (si + 1) % 8 == 0 or si == n_stages - 1:
            print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.5f}: score={sc:.8f}")

    return softmax_from_z(z)


def subgradient_nesterov(a0, time_limit, n_steps=300000, eta_start=0.2, eta_end=0.0005):
    """Subgradient descent with momentum on exact max-conv objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    v = np.zeros(n)
    beta = 0.9

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Tight active set: positions within 99.5% of max
        threshold = max_b * 0.995
        active = b >= threshold
        if not active.any():
            active[np.argmax(b)] = True
        active_idx = np.where(active)[0]

        # Compute subgradient as average over active set
        g = np.zeros(n)
        for k in active_idx:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active_idx)

        S = np.dot(g, a)
        grad_z = a * (g - S)

        # Adaptive step size with exponential decay
        progress = step / max(1, n_steps)
        eta = eta_start * (eta_end / eta_start) ** progress

        # Momentum update
        v = beta * v - eta * grad_z
        z_new = z + v
        a_new = softmax_from_z(z_new)

        z = z_new
        a = a_new
        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if (step + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad iter={step+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturbation_reoptimize(a0, rng, n_perturb, time_per_perturb, solver):
    """Try several perturbations and re-optimize each briefly."""
    best_a = a0.copy()
    best_score = score_normalized(a0)
    n = len(a0)

    for i in range(n_perturb):
        noise_scale = 0.02 / (i + 1)
        noise = rng.normal(0, noise_scale, n)
        a_pert = np.maximum(a0 + noise, 0)
        s = a_pert.sum()
        if s < 1e-15:
            continue
        a_pert = a_pert / s

        a_out, sc = nesterov_lp(a_pert, time_per_perturb, solver=solver, max_iters=50000)
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
            print(f"    Perturb {i+1}: improved to {sc:.8f}")

    return best_a, best_score


def exact_subgradient_step(a, z, eta, beta, v):
    """Single exact subgradient step with momentum. Returns (new_z, new_v, new_a, score)."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    threshold = max_b * 0.995
    active = b >= threshold
    if not active.any():
        active[np.argmax(b)] = True
    active_idx = np.where(active)[0]

    g = np.zeros(n)
    for k in active_idx:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active_idx)

    S = np.dot(g, a)
    grad_z = a * (g - S)
    v_new = beta * v - eta * grad_z
    z_new = z + v_new
    a_new = softmax_from_z(z_new)
    sc = score_normalized(a_new)
    return z_new, v_new, a_new, sc


def targeted_refinement(a0, time_limit, n_iter=50000):
    """Very fine subgradient refinement with decreasing step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    v = np.zeros(n)
    beta = 0.95  # Higher momentum for fine refinement

    eta_start = 0.05
    eta_end = 0.0001

    for step in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break

        progress = step / max(1, n_iter)
        eta = eta_start * (eta_end / eta_start) ** progress

        z, v, a_new, sc = exact_subgradient_step(a, z, eta, beta, v)
        a = a_new

        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if (step + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Refine iter={step+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function to find the best sequence."""
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Total budget: {budget_s:.0f}s, effective deadline in {float(budget_s)-30:.0f}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    n_base = 1200

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        n_base = 1200
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    # Track best
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            remaining = total_deadline - time.monotonic()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, remaining={remaining:.0f}s)")

    # === Phase 1: Nesterov LP main (35%) ===
    phase1_time = 0.35 * (budget_s - 30)
    print(f"\n=== Phase 1: Nesterov LP n={n_base} score={best_score:.8f} time={phase1_time:.0f}s ===")
    a_out, _ = nesterov_lp(best_seq, phase1_time, solver=solver)
    try_update(a_out, "nesterov_lp_p1")
    print(f"  Phase 1 done: {best_score:.8f}")

    # === Phase 2: Ultra LSE (20%) ===
    remaining = total_deadline - time.monotonic()
    phase2_time = min(0.20 * (budget_s - 30), remaining * 0.6)
    if phase2_time > 20:
        print(f"\n=== Phase 2: Ultra LSE score={best_score:.8f} time={phase2_time:.0f}s ===")
        a_lse = lbgfs_logsumexp(best_seq, phase2_time, T_start=30.0, T_end=0.003, n_stages=50)
        try_update(a_lse, "ultra_lse")
    print(f"  After Phase 2: {best_score:.8f}")

    # === Phase 3: Targeted subgradient refinement (15%) ===
    remaining = total_deadline - time.monotonic()
    phase3_time = min(0.15 * (budget_s - 30), remaining * 0.5)
    if phase3_time > 15:
        print(f"\n=== Phase 3: Targeted refinement score={best_score:.8f} time={phase3_time:.0f}s ===")
        a_ref, _ = targeted_refinement(best_seq, phase3_time, n_iter=80000)
        try_update(a_ref, "targeted_refine")
    print(f"  After Phase 3: {best_score:.8f}")

    # === Phase 4: Second LSE pass from improved point (12%) ===
    remaining = total_deadline - time.monotonic()
    phase4_time = min(0.12 * (budget_s - 30), remaining * 0.5)
    if phase4_time > 15:
        print(f"\n=== Phase 4: LSE pass 2 score={best_score:.8f} time={phase4_time:.0f}s ===")
        a_lse2 = lbgfs_logsumexp(best_seq, phase4_time, T_start=10.0, T_end=0.002, n_stages=40)
        try_update(a_lse2, "lse_pass2")
    print(f"  After Phase 4: {best_score:.8f}")

    # === Phase 5: Perturbation + re-optimize (10%) ===
    remaining = total_deadline - time.monotonic()
    phase5_time = min(0.10 * (budget_s - 30), remaining * 0.5)
    if phase5_time > 30:
        print(f"\n=== Phase 5: Perturbation re-opt score={best_score:.8f} time={phase5_time:.0f}s ===")
        n_pert = 6
        t_per = phase5_time / n_pert - 1
        a_pert, _ = perturbation_reoptimize(best_seq, rng, n_pert, t_per, solver)
        try_update(a_pert, "perturb_reopt")
    print(f"  After Phase 5: {best_score:.8f}")

    # === Phase 6: Length exploration (5%) ===
    remaining = total_deadline - time.monotonic()
    phase6_time = min(0.05 * (budget_s - 30), remaining * 0.4)
    if phase6_time > 20:
        print(f"\n=== Phase 6: Length exploration score={best_score:.8f} time={phase6_time:.0f}s ===")
        for n_try in [n_base + 100, n_base - 100, n_base + 200]:
            remaining = total_deadline - time.monotonic()
            if remaining < 10:
                break
            t_try = min(phase6_time / 3.0, remaining * 0.4)
            if t_try < 5:
                break
            print(f"  Trying n={n_try} budget={t_try:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp(init_try, t_try, solver=solver, max_iters=50000)
            try_update(a_try, f"len_{n_try}")
    print(f"  After Phase 6: {best_score:.8f}")

    # === Phase 7: Final ultra-fine refinement (3%) ===
    remaining = total_deadline - time.monotonic()
    phase7_time = min(0.03 * (budget_s - 30), remaining * 0.6)
    if phase7_time > 10:
        print(f"\n=== Phase 7: Final refinement score={best_score:.8f} time={phase7_time:.0f}s ===")
        a_fin, _ = targeted_refinement(best_seq, phase7_time, n_iter=30000)
        try_update(a_fin, "final_refine")
        # One more LSE
        remaining = total_deadline - time.monotonic()
        if remaining > 10:
            a_fin2 = lbgfs_logsumexp(best_seq, min(remaining * 0.5, 60), T_start=3.0, T_end=0.001, n_stages=15)
            try_update(a_fin2, "final_lse")

    # === Phase 8: Final Nesterov if time remains ===
    remaining = total_deadline - time.monotonic()
    if remaining > 15:
        t8 = min(remaining * 0.5, 90)
        print(f"\n=== Phase 8: Final Nesterov score={best_score:.8f} time={t8:.0f}s ===")
        a_last, _ = nesterov_lp(best_seq, t8, solver=solver)
        try_update(a_last, "final_nesterov")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        best_seq = np.ones(1200) / 1200
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in best_seq]
