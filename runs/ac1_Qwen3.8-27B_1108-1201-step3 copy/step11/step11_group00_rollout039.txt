import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_conv(a, b=None):
    """Full linear convolution a*b (or a*a if b is None) using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def normalize_a(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def mw_best_response(a_fixed, n_steps=200, eta=None):
    """
    Multiplicative Weights approximation of:
      min_b max_k (a_fixed * b)_k  s.t. sum b = 1, b >= 0
    Each step is O(n). Returns (b_approx, M_approx).
    """
    n = len(a_fixed)
    b = np.ones(n) / n
    if eta is None:
        eta = 0.5

    # Precompute: for a given k*, the update vector is a_fixed[k*-j] for j=0..n-1
    # i.e., a_fixed shifted and reversed appropriately
    for step in range(n_steps):
        # Compute convolution a_fixed * b
        conv = fast_conv(a_fixed, b)  # length 2n-1
        k_star = int(np.argmax(conv))
        # Gradient for constraint k_star: grad[j] = a_fixed[k_star - j] (0 if out of range)
        js = np.arange(n)
        idx = k_star - js
        valid = (idx >= 0) & (idx < n)
        grad = np.zeros(n)
        grad[valid] = a_fixed[idx[valid]]
        
        # MW update
        b = b * np.exp(-eta * grad)
        b = np.maximum(b, 1e-300)
        s = b.sum()
        if s < 1e-300:
            b = np.ones(n) / n
        else:
            b = b / s

    conv_final = fast_conv(a_fixed, b)
    return b, float(np.max(conv_final))

            a_new = a_new / s

        a = a_new

def gs_spectral_shaping(a0, time_limit, n_iterations=None):
    """
    Gerchberg-Saxton algorithm for spectral shaping.
    Goal: make |FFT(a_padded)| as flat as possible, which flattens the convolution.
    Returns (best_a, best_score).
    """
    n = len(a0)
    a = normalize_a(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()

    # Use FFT size >= 2n-1 for linear convolution
    N = 1 << (2 * n - 1).bit_length()

    t0 = time.monotonic()
    it = 0
    if n_iterations is None:
        n_iterations = 10000000  # Let time limit control

    while time.monotonic() - t0 < time_limit and it < n_iterations:
        it += 1

        # Zero-pad and FFT
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)

        # Current convolution
        B = A * A
        b = np.fft.ifft(B).real[:2 * n - 1]

        # Target: flatten the magnitude of A
        # Use geometric mean of current |A| and a flat target
        mag_A = np.abs(A)
        mag_A = np.maximum(mag_A, 1e-20)
        flat_target = np.median(mag_A)

        # Blend: new magnitude is between flat and current
        # alpha controls how aggressive the flattening is
        alpha = 0.3
        new_mag = flat_target ** alpha * mag_A ** (1.0 - alpha)

        # Preserve phase
        phase = A / np.maximum(mag_A, 1e-20)
        A_new = new_mag * phase

        # IFFT and extract
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        # Check score
        score = 2.0 * n * fast_conv(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      GS it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_obj_grad(z, T):
    """
    LogSumExp objective and gradient in z-space (a = softmax(z)).
    Objective: (1/T) * log(sum_k exp(T * (a*a)_k))
    """
    n = len(z)
    a = softmax_fn(z)
    b = fast_conv(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T
    w = exp_Tb / exp_Tb.sum()

    # Gradient w.r.t. a: grad_a[i] = 2 * sum_k w[k] * a[k-i] (with valid indices)
    # Use FFT cross-correlation
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad_a = 2.0 * conv[n - 1:n - 1 + n]

    # Transform to z-space
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z


def lse_optimize(a0, time_limit, T_start=50.0, T_end=0.5, n_stages=10):
    """
    LogSumExp-smoothed optimization with L-BFGS and multi-stage annealing.
    """
    n = len(a0)
    a_init = normalize_a(a0)
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_conv(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        iters = max(20, min(100, int(30 + 70 * (si / n_stages))))

        def func(z_in):
            val, _ = lse_obj_grad(z_in, stage_T)
            return val

        def jac(z_in):
            _, g = lse_obj_grad(z_in, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters * 500,
                         'disp': False}
            )
            z = res.x
            a = softmax_fn(z)
            score = 2.0 * n * fast_conv(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception:
            pass

    return best_a, best_score

                in_peak.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in in_peak and j != src]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            candidates.sort(key=lambda j: a[j])
            dst_pool = candidates[:max(1, len(candidates) // 4)]

            for dst in dst_pool[:3]:
                for ss in step_sizes:
                    amt = a[src] * ss
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_conv(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random exploration
            stall += 1
            for _ in range(10):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-12:
                    continue
                frac = float(rng.uniform(0.002, 0.02))
                amt = a[i] * frac
                direction = 1 if rng.random() < 0.5 else -1
                a_trial = a.copy()
                a_trial[i] -= direction * amt
                a_trial[j] += direction * amt
                if (a_trial[i] < 0) or (a_trial[j] < 0):
                    continue
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_conv(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

            if stall > max_stall:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations with asymmetry."""
    choice = int(rng.integers(0, 8))
    if choice == 0:
        # Skewed Gaussian
        x = np.linspace(0, 1, n)
        mu = 0.45 + 0.1 * rng.random()
        sig = 0.15 + 0.05 * rng.random()
        a = np.exp(-((x - mu) ** 2) / (2 * sig ** 2))
    elif choice == 1:
        # Power law (asymmetric)
        x = np.linspace(0, 1, n)
        p = 0.5 + rng.random()
        a = (1 - x) ** p
    elif choice == 2:
        # Raised cosine with skew
        x = np.linspace(0, 1, n)
        gamma = 0.1 + 0.2 * rng.random()
        a = (0.5 + 0.5 * np.cos(2 * np.pi * x)) * (1 + gamma * (2 * x - 1))
    elif choice == 3:
        # Bimodal asymmetric
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * (x - 0.3) ** 2) + 0.6 * np.exp(-8 * (x - 0.7) ** 2)
    elif choice == 4:
        # Polynomial with skew
        x = np.linspace(0, 1, n)
        a = (1 - x) * (1 + 0.5 * x) ** 2
    elif choice == 5:
        # Near-uniform with small skew
        x = np.linspace(0, 1, n)
        eps = 0.05 + 0.1 * rng.random()
        a = np.ones(n) * (1 + eps * (2 * x - 1))
    elif choice == 6:
        # Sine-squared with phase shift
        x = np.linspace(0, 1, n)
        phase = 0.1 * rng.random()
        a = np.sin(np.pi * (x + phase)) ** 2
    else:
        # Beta distribution-like
        x = np.linspace(0.001, 0.999, n)
        alpha_b = 1.5 + rng.random()
        beta_b = 1.5 + rng.random()
        a = x ** (alpha_b - 1) * (1 - x) ** (beta_b - 1)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def multi_start_nesterov(a0, time_limit, n_starts=3, mw_steps=200):
    """
    Run Nesterov-MW from multiple perturbations of a0, return the best.
    """
    n = len(a0)
    rng = np.random.default_rng(12345)
    t0 = time.monotonic()
    time_per_start = time_limit / n_starts

    best_a = a0.copy()
    best_score = 2.0 * n * fast_conv(a0).max()

    for s in range(n_starts):
        if time.monotonic() - t0 > time_limit - 10:
            break

        # Create perturbation
        if s == 0:
            a_init = a0.copy()
        else:
            noise_scale = 0.05 * np.std(a0)
            noise = rng.normal(0, noise_scale, n)
            a_init = a0 + noise
            a_init = normalize_a(a_init)

        a_out, sc = nesterov_mw_optimize(a_init, time_per_start, mw_steps=mw_steps)
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
            elapsed = time.monotonic() - t0
            print(f"      Multi-start {s}: {sc:.8f} ({elapsed:.0f}s)")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization:
    1. MW-Nesterov at parent length (main workhorse)
    2. GS spectral shaping (diverse exploration)
    3. Multi-length MW-Nesterov
    4. LogSumExp refinement
    5. Coordinate descent polish
    6. Final micro-adjustments
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    # Get parent sequence
    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    n_base = len(best_seq)
    print(f"\nStarting length: n={n_base}")

    # === Phase 1: MW-Nesterov at parent length (500s) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: MW-Nesterov at n={n_base} (budget={min(remaining*0.30, 500.0):.0f}s) ===")
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 30:
        a_out, _ = nesterov_mw_optimize(best_seq, t1, mw_steps=300, eta=0.5)
        try_update(a_out, "mw_nesterov_main")

    # === Phase 2: GS Spectral Shaping (200s) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: GS Spectral Shaping at n={len(best_seq)} (budget={min(remaining*0.12, 200.0):.0f}s) ===")
    t2 = min(remaining * 0.12, 200.0)
    if t2 > 20:
        a_gs, _ = gs_spectral_shaping(best_seq, t2)
        try_update(a_gs, "gs_shaping")

    # === Phase 3: Multi-length exploration (400s) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Multi-length exploration (budget={min(remaining*0.25, 400.0):.0f}s) ===")
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 60:
        # Try several lengths
        n_candidates = [1400, 1500, 1700, 1800, 2000]
        # Remove lengths too similar to current best
        n_candidates = [nn for nn in n_candidates if abs(nn - len(best_seq)) > 50]
        n_candidates = sorted(n_candidates, key=lambda x: abs(x - len(best_seq)))
        n_to_try = n_candidates[:3]  # Try 3 closest different lengths

        time_per_len = t3 / max(1, len(n_to_try))
        for nn in n_to_try:
            if time.monotonic() - t_global_start > time_budget - 60:
                break
            print(f"  Trying n={nn} (time={time_per_len:.0f}s)")
            init = interpolate_sequence(best_seq, nn)
            a_out, sc = nesterov_mw_optimize(init, time_per_len, mw_steps=200, eta=0.5)
            try_update(a_out, f"mw_n{nn}")

    # === Phase 4: LogSumExp refinement (200s) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: LogSumExp at n={len(best_seq)} (budget={min(remaining*0.12, 200.0):.0f}s) ===")
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 15:
        a_out, _ = lse_optimize(best_seq, t4, T_start=30.0, T_end=0.5, n_stages=8)
        try_update(a_out, "lse_refine")

    # === Phase 5: Coordinate descent polish (200s) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Coordinate Descent at n={len(best_seq)} (budget={min(remaining*0.12, 200.0):.0f}s) ===")
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 10:
        a_out, _ = coordinate_descent_polish(best_seq, t5, seed=777)
        try_update(a_out, "coord_desc_1")

    # === Phase 6: Final micro-adjustments and restarts (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Final refinement (budget={remaining:.0f}s) ===")
    if remaining > 30:
        # Another round of MW-Nesterov from the current best
        t6a = min(remaining * 0.4, 200.0)
        if t6a > 20:
            a_out, _ = nesterov_mw_optimize(best_seq, t6a, mw_steps=200, eta=0.3)
            try_update(a_out, "mw_final")

        # Coordinate descent again
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining > 20:
            t6b = min(remaining * 0.3, 100.0)
            if t6b > 10:
                a_out, _ = coordinate_descent_polish(best_seq, t6b, seed=999)
                try_update(a_out, "coord_desc_2")

        # Try multi-start from best
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining > 60:
            t6c = min(remaining * 0.5, 150.0)
            if t6c > 30:
                print(f"  Multi-start Nesterov (budget={t6c:.0f}s)")
                a_out, _ = multi_start_nesterov(best_seq, t6c, n_starts=2, mw_steps=150)
                try_update(a_out, "multi_start")

        # Final GS attempt
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining > 30:
            t6d = min(remaining * 0.4, 60.0)
            if t6d > 10:
                a_out, _ = gs_spectral_shaping(best_seq, t6d)
                try_update(a_out, "gs_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
