"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam with exact LogSumExp gradient (vectorized via sliding window)
2. L-BFGS-B refinement in log-space (scipy)
3. Exact active-set subgradient with adaptive step size
4. Multi-length exploration (n=1000 to 2000)
5. Structured initialization and perturbation-based restarts
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.stats import special  # not needed, just in case

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def local_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a = a / s
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())

def lse_value(a, T):
    """LogSumExp approximation of max, scaled by 2n."""
    b = fast_convolve(a)
    max_b = b.max()
    shifted = (b - max_b) / T
    # Use log-sum-exp trick
    max_shifted = shifted.max()
    log_sum = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
    lse = max_b + T * log_sum
    return 2.0 * len(a) * lse

def lse_gradient(a, T):
    """
    Gradient of 2n * LogSumExp(conv(a,a)/T) w.r.t. a.
    Uses vectorized sliding window dot product.
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    shifted = (b - max_b) / T
    max_shifted = shifted.max()
    exp_shifted = np.exp(shifted - max_shifted)
    w = exp_shifted / exp_shifted.sum()
    # grad[i] = (4n/T) * dot(w[i:i+n], a)
    W = np.lib.stride_tricks.sliding_window_view(w, n)
    grad = (4.0 * n / T) * (W @ a)
    return grad

def exact_active_gradient(a, tolerance=1e-6):

    max_b = b.max()

    w = np.zeros_like(b)
    w[active] = 1.0 / len(active)
    W = np.lib.stride_tricks.sliding_window_view(w, n)
    grad = 2.0 * n * (W @ a) * 2.0  # factor of 4n from the auto-conv derivative

    # = (4n/|A|) * sum_{k in A} a[k-i]

def logspace_objective(z, T):
    """Objective in log-space for scipy optimizer."""
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    s = a.sum()
    if s < 1e-300:
        return 1e10
    a = a / s
    return lse_value(a, T)

def logspace_gradient(z, T):
    """Gradient of log-space objective."""
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    s = a.sum()
    if s < 1e-300:
        return np.zeros(len(z))
    a = a / s
    g_a = lse_gradient(a, T)
    S = np.dot(g_a, a)
    g_z = a * (g_a - S)
    return g_z

            g_z = a * (g_a - S)
            
            # Adam update
            lr = lr_init * (lr_decay ** (T_schedule.index(T)))
            m = beta1 * m + (1 - beta1) * g_z
            v = beta2 * v + (1 - beta2) * g_z ** 2
            m_hat = m / (1 - beta1 ** (step + 1))
            v_hat = v / (1 - beta2 ** (step + 1))
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            z -= z.mean()
            step += 1
            
            # Check time
            if time.monotonic() > deadline:
                break
    
    return best_a, best_score

def softmax(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def subgradient_polish(a0, time_limit, n_max=100000):
    """Exact active-set subgradient with adaptive step size."""
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    
    eta = 0.01
    no_improve = 0
    
    for step in range(n_max):
        if time.monotonic() - t0 > time_limit:
            break
        
        grad, max_b, active = exact_active_gradient(a, tolerance=1e-8)
        
        # Projected subgradient step
        # a_new = proj_simplex(a - eta * grad)
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        score = local_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 20:
            eta *= 0.7
            no_improve = 0
            if eta < 1e-8:
                break
        
        if step % 5000 == 0 and step > 0:
            print(f"      SubGrad step {step}: score={best_score:.8f} eta={eta:.6f} |A|={len(active)}")
    
    return best_a, best_score

def lbfgs_polish(a0, T, maxiter=200):
    """L-BFGS-B refinement in log-space."""
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-300))
    z0 -= z0.mean()
    
    res = minimize(
        logspace_objective, z0, args=(T,),
        jac=logspace_gradient, method='L-BFGS-B',
        options={'maxiter': maxiter, 'gtol': 1e-8, 'ftol': 1e-12}
    )
    
    a = softmax(res.x)
    return a, local_score(a)

def structured_init(n, rng):
    """Create structured initial sequence."""
    x = np.linspace(0, 1, n)
    choice = rng.integers(0, 6)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-2*(x-0.3)**2) + 0.5*np.exp(-2*(x-0.7)**2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s if s > 1e-15 else np.ones(n)/n

def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    f_norm = f / s if s > 1e-15 else np.ones(old_n)/old_n
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    return f_interp / ss if ss > 1e-15 else np.ones(new_n)/new_n
