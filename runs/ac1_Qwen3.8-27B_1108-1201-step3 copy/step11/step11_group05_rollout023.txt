    3. Use Nesterov momentum

    n = len(a)
    a = np.maximum(a.copy(), 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Nesterov state
    y = a.copy()  # Nesterov lookahead
    a_avg = a.copy()  # Running average
    eta = eta_init
    stall = 0
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP using y (Nesterov lookahead) as the fixed sequence
        y_norm = y / y.sum()
        b_new, M_val, dual = solve_lp_response(y_norm, T_mat, solver)
        
        if b_new is None:
            continue
        
        # Get the gradient direction from dual
        if dual is not None:
            # g[i] = (T^T @ dual)[i] = sum_k dual[k] * T[k,i] = sum_k dual[k] * a[k-i]
            # This is the correlation of dual with a
            g = T_mat.T @ dual
            g = np.asarray(g, dtype=np.float64).flatten()
        else:
            # Fallback: use subgradient at the peak
            b_conv = fast_convolve(y_norm)
            peak = np.argmax(b_conv)
            g = np.zeros(n)
            js = max(0, peak - n + 1)
            je = min(n - 1, peak)
            for i in range(js, je + 1):
                g[i] = y_norm[peak - i]
        
        # Normalize gradient
        g = np.maximum(g, 0)
        g_mean = g.mean()
        if g_mean > 1e-15:
            g = g / g_mean
        
        # Multiplicative update with Nesterov
        alpha = 2.0 / (it + 2)
        
        # Exponentiated gradient on y
        log_update = -eta * g
        log_update -= log_update.max()
        y_new = y * np.exp(log_update)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
            sy = y.sum()
        y_new = y_new / sy
        
        # Nesterov update
        y = (1 - alpha) * y_new + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = y_new
            sy = y_new.sum()
        y = y / sy
        
        # Average iterate
        a_avg = ((it * a_avg + (it + 1) * y) / (it + 1)) if it > 0 else y.copy()
        
        # Check best
        for candidate in [y, a_avg, b_new]:
            c = np.maximum(candidate, 0)
            sc = 2.0 * n * fast_convolve(c / c.sum()).max()
            if sc < best_score:
                best_score = sc
                best_a = c / c.sum()
                stall = 0
            else:
                stall += 1

        if stall >= 30:
            eta *= 0.8
            stall = 0
            if eta < 0.01:
                eta = eta_init  # Reset
                # Try perturbation
                noise = 0.01 * np.random.default_rng(it).standard_normal(n)
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()

            print(f"    DG-MW iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} eta={eta:.4f} M={M_val:.8f}")
    
    return best_a, best_score

def adam_refine(a, time_limit, T_list=None, lr=0.002):
    """Adam optimization in log-space with log-sum-exp smoothing."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-300)).copy()
    z -= z.mean()
    t0 = time.monotonic()
    
    if T_list is None:
        T_list = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    
    steps_per_stage = max(2000, int(time_limit * 30000 / len(T_list)))
    
    for T in T_list:
        if time.monotonic() - t0 > time_limit:
            break
        
        for _ in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            step += 1
            
            a_cur = np.exp(z - z.max())
            a_cur = a_cur / a_cur.sum()
            
            b = fast_convolve(a_cur)
            max_b = b.max()
            score = 2.0 * n * max_b
            
            if score < best_score:
                best_score = score
                best_a = a_cur.copy()
            
            # Log-sum-exp gradient
            if T < 0.01:
                # Hard: just use the peak
                peak = np.argmax(b)
                w = np.zeros(2 * n - 1)
                w[peak] = 1.0
            else:
                # Soft: log-sum-exp
                shifted = (b - max_b) / T
                exp_shifted = np.exp(shifted)
                w = exp_shifted / exp_shifted.sum()
            
            # Gradient: g_a[i] = 2 * sum_k w[k] * a[k-i]
            # = 2 * correlate(w, a) at appropriate lag
            g = 2.0 * np.correlate(w, a_cur, mode='full')[n - 1:2 * n - 1]
            
            # Convert to z-space
            S = np.dot(g, a_cur)
            grad_z = a_cur * (g - S)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** (step + 1))
            v_hat = v / (1 - beta2 ** (step + 1))
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            z -= z.mean()
    
    return best_a, best_score

def coordinate_descent_polish(a, time_limit, n_sweeps=100):
    """
    Coordinate descent: optimize each element a[i] one at a time.
    For each i, try to reduce the max convolution value by adjusting a[i].
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Determine direction (alternate left-to-right and right-to-left)
        if sweep % 2 == 0:
            order = range(n)
        else:
            order = range(n - 1, -1, -1)
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Current convolution
            b = fast_convolve(a)
            peak = b.max()
            
            # Try adjusting a[i]
            # The effect of a[i] on b[k] is a[i] * a[k-i] for valid k-i
            # So increasing a[i] increases b[k] for k where a[k-i] > 0
            # To reduce the peak, we want to decrease a[i] if it contributes to the peak
            
            # Subgradient approach: if a[i] contributes to the peak, decrease it
            peak_idx = np.argmax(b)
            # a[i] contributes to b[peak_idx] if 0 <= peak_idx - i < n
            j = peak_idx - i
            if 0 <= j < n and a[j] > 0:
                # Decrease a[i]
                factor = 0.99
                a[i] *= factor
            else:
                # a[i] doesn't contribute to the peak, maybe increase slightly
                # (to maintain sum = 1)
                pass
        
        # Renormalize
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            break
        a = a / a.sum()
        
        cur_score = 2.0 * n * fast_convolve(a).max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for i in range(n):
        for k in range(i, i + n):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_response(a_fixed, T_mat, solver):
    """Given fixed a (sum=1), find b minimizing max_k conv(a,b)[k] with sum(b)=1, b>=0."""
    n = a_fixed.shape[0]
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    c1 = T_mat @ b_var <= M_var
    c2 = cp.sum(b_var) == 1
    prob = cp.Problem(cp.Minimize(M_var), [c1, c2])

    solvers_to_try = [solver, cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]
    seen = set()
    for s in solvers_to_try:
        if s is None or s in seen:
            continue
        seen.add(s)
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                b_val[np.isnan(b_val)] = 0
                s_val = b_val.sum()
                if s_val < 1e-15:
                    continue
                b_val = b_val / s_val
                dual = None
                try:
                    dual_raw = np.asarray(c1.dual_value, dtype=np.float64)
                    if dual_raw is not None and np.all(np.isfinite(dual_raw)):
                        dual = np.maximum(dual_raw, 0)
                        ds = dual.sum()
                        if ds > 1e-15:
                            dual = dual / ds
                        else:
                            dual = None
                except Exception:
                    dual = None
                return b_val, float(max(0.0, M_var.value)), dual
        except Exception:
            continue
    return None, None, None

def dual_guided_mw(a0, T_mat, solver, time_limit, eta_init=0.3, n_max_iters=500000):
    """
    Dual-guided multiplicative weights with Nesterov acceleration.
    Uses LP dual variables to guide exponentiated gradient updates.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    y = a.copy()
    eta = eta_init
    stall = 0
    prev_best = best_score
    rng = np.random.default_rng(12345)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        y_norm = y / y.sum()
        b_new, M_val, dual = solve_lp_response(y_norm, T_mat, solver)
        if b_new is None:
            continue

        # Compute gradient direction from dual
        if dual is not None:
            g = np.asarray(T_mat.T @ dual, dtype=np.float64).flatten()
            g = np.maximum(g, 0)
        else:
            b_conv = fast_convolve(y_norm)
            peak_val = b_conv.max()
            threshold = peak_val * 0.999
            active = np.where(b_conv >= threshold)[0]
            if len(active) == 0:
                active = np.array([np.argmax(b_conv)])
            g = np.zeros(n)
            for k in active:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                for i in range(js, je + 1):
                    g[i] += y_norm[k - i]
            g /= len(active)

        g_mean = g.mean()
        if g_mean > 1e-15:
            g = g / g_mean
        else:
            g = np.ones(n) / n

        # Nesterov parameter
        alpha = 2.0 / (it + 2)

        # Exponentiated gradient update on y
        log_update = -eta * g
        log_update -= log_update.max()
        y_expected = y * np.exp(log_update)
        y_expected = np.maximum(y_expected, 0)
        se = y_expected.sum()
        if se < 1e-15:
            y_expected = y.copy()
            se = y.sum()
        y_expected = y_expected / se

        # Nesterov step
        y = (1.0 - alpha) * y_expected + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = y_expected
            sy = y_expected.sum()
        y = y / sy

        # Check candidates
        for cand in [y, y_expected, b_new]:
            c = np.maximum(cand, 0)
            cs = c.sum()
            if cs < 1e-15:
                continue
            c = c / cs
            sc = 2.0 * n * fast_convolve(c).max()
            if sc < best_score:
                best_score = sc
                best_a = c.copy()
                stall = 0
            else:
                stall += 1

        # Adaptive step size
        if stall >= 40:
            eta *= 0.7
            stall = 0
            if eta < 0.02:
                eta = eta_init
                # Perturb to escape
                noise = 0.005 * rng.standard_normal(n)
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()

        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DG-MW iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} eta={eta:.4f} M_lp={M_val:.8f}")

    return best_a, best_score


def adam_refine(a0, time_limit, T_list=None, lr=0.003, top_k_frac=0.15):
    """Adam optimization in log-space with log-sum-exp smoothing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    z -= z.mean()
    t0 = time.monotonic()

    if T_list is None:
        T_list = [3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005]

    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0

    steps_per_stage = max(1000, int(time_limit * 40000 / max(1, len(T_list))))

    for T in T_list:
        if time.monotonic() - t0 > time_limit:
            break

        top_k = max(5, int(top_k_frac * (2 * n - 1)))

        for _ in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            step += 1

            a_cur = np.exp(z - z.max())
            a_cur = a_cur / a_cur.sum()

            b = fast_convolve(a_cur)
            max_b = b.max()
            score = 2.0 * n * max_b

            if score < best_score:
                best_score = score
                best_a = a_cur.copy()

            # Compute weights
            if T < 0.01:
                peak = np.argmax(b)
                w = np.zeros(2 * n - 1)
                w[peak] = 1.0
            elif top_k < len(b):
                idx = np.argpartition(b, -top_k)[-top_k:]
                b_top = b[idx]
                max_top = b_top.max()
                exp_top = np.exp((b_top - max_top) / T)
                w_top = exp_top / exp_top.sum()
                w = np.zeros(len(b))
                w[idx] = w_top
            else:
                shifted = (b - max_b) / T
                exp_s = np.exp(shifted)
                w = exp_s / exp_s.sum()

            # Gradient in a-space: g[i] = 2 * sum_k w[k] * a[k-i]
            g_a = 2.0 * np.correlate(w, a_cur, mode='full')[n - 1:2 * n - 1]

            # Convert to z-space
            S = np.dot(g_a, a_cur)
            grad_z = a_cur * (g_a - S)

            # Adam
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** (step + 1))
            v_hat = v / (1 - beta2 ** (step + 1))
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            z -= z.mean()

    return best_a, best_score


def directional_sweep(a0, time_limit, n_sweeps=200):
    """
    Directional coordinate descent (MRF-inspired).
    Alternates left-to-right and right-to-left sweeps.
    At each position, adjusts the element to reduce the peak convolution.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    decay = 0.995
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        order = range(n) if sweep % 2 == 0 else range(n - 1, -1, -1)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            b = fast_convolve(a)
            peak_val = b.max()
            peak_idx = np.argmax(b)

            # Check if a[i] contributes to the peak
            j = peak_idx - i
            if 0 <= j < n and a[j] > 1e-10:
                # Reduce a[i]
                a[i] *= decay
            else:
                # Slightly increase (to compensate for reductions elsewhere)
                a[i] *= (2.0 - decay)

        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            break
        a = a / s

        cur_score = 2.0 * n * fast_convolve(a).max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        elif sweep > 50:
            # Gradually increase aggressiveness
            decay -= 0.001
            decay = max(decay, 0.95)

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        t_p5 = min(rem * 0.12, 200.0)
        print(f"  Time budget: {t_p5:.0f}s")
        init_1500 = interpolate_sequence(best_seq, 1500)
        # Quick Adam
        t_a = t_p5 * 0.4
        a_1500a, _ = adam_refine(init_1500, t_a,
                                  T_list=[2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01],
                                  lr=0.003)
        local_update(a_1500a, "adam_n1500")
        rem = deadline - time.monotonic()
        if rem > 50:
            T_1500 = build_conv_matrix(best_seq)
            t_lp = min(rem * 0.5, 100.0)
            a_1500l, _ = dual_guided_mw(best_seq, T_1500, solver, t_lp, eta_init=0.15)
            local_update(a_1500l, "dg_mw_n1500")

    # === Phase 6: Perturbation restarts (10%) ===
    print(f"\n=== Phase 6: Perturbation Restarts score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t_p6 = min(rem * 0.10, 150.0)
        n_cur = len(best_seq)
        n_trials = min(5, max(2, int(t_p6 / 30)))
        print(f"  Trials: {n_trials}, budget: {t_p6:.0f}s")
        for trial in range(n_trials):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 30:
                break
            strength = 0.003 + 0.003 * trial
            x = np.linspace(0, 2 * np.pi, n_cur)
            freq1 = rng.integers(1, 6)
            freq2 = rng.integers(6, 15)
            phase1 = rng.uniform(0, 2 * np.pi)
            phase2 = rng.uniform(0, 2 * np.pi)
            pert = strength * np.cos(freq1 * x + phase1) + 0.5 * strength * np.cos(freq2 * x + phase2)
            a_pert = best_seq + pert
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()

            t_sub = min(rem_sub * 0.25, 40.0)
            T_pert = build_conv_matrix(a_pert)
            a_opt, _ = dual_guided_mw(a_pert, T_pert, solver, t_sub, eta_init=0.15)
            local_update(a_opt, f"pert_{trial}")

    # === Phase 7: Final Adam polish ===
    print(f"\n=== Phase 7: Final Polish score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t_fin = min(rem * 0.6, 40.0)
        a_fin, _ = adam_refine(best_seq, t_fin,
                                T_list=[0.5, 0.2, 0.1, 0.05, 0.03, 0.01, 0.005, 0.002],
                                lr=0.001, top_k_frac=0.05)
        local_update(a_fin, "final_adam")

    # Final verification
    if best_seq is not None:
        fs = 2.0 * len(best_seq) * fast_convolve(best_seq).max()
        if fs < best_score:
            best_score = fs
        final_eval = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
        print(f"\nFINAL SCORE (local): {best_score:.8f} (n={len(best_seq)})")
        print(f"FINAL SCORE (evaluate_sequence): {final_eval:.8f}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200.0

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
