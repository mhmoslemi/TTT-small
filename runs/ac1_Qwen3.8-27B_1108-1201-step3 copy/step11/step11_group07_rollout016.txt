import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def project_simplex(v, cap=1.0):
    """Project v onto the simplex {x: sum(x) = 1, 0 <= x <= cap}."""
    # Standard simplex projection: {x: sum(x) = 1, x >= 0}
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w  # This gives sum = 1

def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def subgradient_topk(a, k_peaks=20):
    """Compute subgradient of max(conv(a,a)) using top-k peaks."""
    n = len(a)
    c = fast_convolve(a)  # length 2n-1
    # Get top k peaks
    if k_peaks >= len(c):
        idx = np.argsort(c)[::-1]
    else:
        idx = np.argpartition(c, -k_peaks)[-k_peaks:]
        idx = idx[np.argsort(c[idx])[::-1]]
    
    # Gradient of conv(a,a)[k] at position m is 2*a[k-m] if 0 <= k-m < n
    g = np.zeros(n)
    for k in idx:
        # g[m] += 2*a[k-m] for m such that 0 <= k-m < n
        # i.e., m is in [max(0, k-n+1), min(n-1, k)]
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        if m_start <= m_end:
            # g[m_start:m_end+1] += 2 * a[k-m] for m in m_start..m_end
            # k-m ranges from k-m_start to k-m_end
            a_slice = a[k - m_end : k - m_start + 1 : -1]  # reversed
            g[m_start:m_end+1] += 2.0 * a_slice
    
    # Normalize by number of peaks
    g /= len(idx)
    return g, c.max(), c.sum()

    G = 2 * np.conj(A) * W
    g = irfft(G, fft_size)[:n]

1. Momentum-based subgradient descent on the exact max function
2. Adaptive active set (top-K peaks) with smooth transition
3. Spectral-domain random kicks for escaping local minima
4. Polyak-type adaptive step size with momentum restart
5. Periodic 'direction refinement' using pairwise peak cancellation
6. Multi-strategy exploration across lengths and initializations

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    """Evaluate score locally (matches evaluate_sequence for valid inputs)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def compute_subgradient(a, c, top_k_indices):
    """Compute subgradient of sum_{k in S} c[k] / |S| where c = conv(a,a).
    grad(c[k])[m] = 2*a[k-m] if 0 <= k-m < n, else 0."""
    n = len(a)
    g = np.zeros(n)
    for k in top_k_indices:
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        if m_start <= m_end:
            # g[m] += 2*a[k-m] for m in [m_start, m_end]
            # k-m ranges from k-m_start down to k-m_end
            indices = k - np.arange(m_start, m_end + 1)
            g[m_start:m_end+1] += 2.0 * a[indices]
    g /= len(top_k_indices)
    return g

def spectral_kick(a, rng, strength, fft_size):
    """Add random spectral perturbation."""
    n = len(a)
    A = rfft(a, fft_size)
    # Generate random smooth perturbation (low-pass filtered noise)
    noise = rng.normal(0, 1, fft_size // 2 + 1).astype(np.complex128)
    noise += 1j * rng.normal(0, 1, fft_size // 2 + 1)
    # Low-pass filter: keep only low frequencies
    cutoff = max(3, fft_size // 20)
    if cutoff < fft_size // 2 + 1:
        noise[cutoff:] = 0
    noise[0] = 0  # Don't change DC
    A_new = A + strength * noise * np.abs(A) / (np.abs(A).max() + 1e-15)
    a_new = irfft(A_new, fft_size)[:n]
    return normalize(a_new)

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    return normalize(f_interp)

def momentum_subgradient(a_init, time_limit, rng, K_init=10, K_max=80,
                         beta=0.7, eta_init=0.01, kick_interval=40,
                         kick_strength=0.02, restart_patience=100):
    """
    Momentum-based subgradient descent with spectral kicks.
    
    a_init: initial sequence (will be normalized)
    time_limit: seconds to run
    rng: random number generator
    K_init: initial number of peaks to use
    K_max: maximum number of peaks
    beta: momentum coefficient
    eta_init: initial step size
    kick_interval: iterations between spectral kicks
    kick_strength: strength of spectral kicks
    restart_patience: iterations without improvement before restart
    """
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    fft_size = 1 << (2 * n - 1).bit_length()
    momentum = np.zeros(n)
    eta = eta_init
    K = K_init
    no_improve = 0
    kick_str = kick_strength
    
    iter_count = 0
    while time.monotonic() - t0 < time_limit:
        # Get current convolution
        c = fast_convolve(a)
        c_max = c.max()
        
        # Select top-K peaks (adaptive K)
        if K >= len(c):
            top_k = np.argsort(c)[::-1]
        else:
            # Use argpartition for efficiency
            part_idx = np.argpartition(c, -K)[-K:]
            top_k = part_idx[np.argsort(c[part_idx])[::-1]]
        
        # Compute subgradient
        g = compute_subgradient(a, c, top_k)
        
        # Momentum update
        momentum = beta * momentum - eta * g
        
        # Take step
        a_new = a + momentum
        a_new = normalize(a_new)
        
        # Evaluate
        score = evaluate_local(a_new)
        
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
            # Slightly increase K as we improve
            K = min(K_max, K + 1)
        else:
            a = a_new.copy()  # Still take the step (for momentum)
            no_improve += 1
        
        iter_count += 1
        
        # Adaptive step size
        if no_improve > 30:
            eta *= 0.7
            momentum *= 0.5
            no_improve = 0
            if eta < 1e-6:
                eta = eta_init * 0.1
                momentum = np.zeros(n)
        
        # Spectral kick
        if iter_count % kick_interval == 0:
            a = spectral_kick(a, rng, kick_str, fft_size)
            score_after_kick = evaluate_local(a)
            if score_after_kick < best_score:
                best_score = score_after_kick
                best_a = a.copy()
            # Decay kick strength
            kick_str *= 0.98
        
        # Restart if stuck
        if no_improve >= restart_patience:
            # Perturb from best
            noise = rng.normal(0, 0.01, n)
            a = normalize(best_a + noise)
            momentum = np.zeros(n)
            eta = eta_init * 0.5
            no_improve = 0
            kick_str = min(kick_str, 0.03)
        
        # Log progress
        if iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    iter={iter_count}, t={elapsed:.0f}s, best={best_score:.8f}, K={K}, eta={eta:.6f}")
    
    return best_a, best_score

def pairwise_cancellation(a, c, top_k, n_pairs=5):
    """Try to find directions that reduce multiple peaks simultaneously.
    For each pair of top peaks, compute the "cancellation" direction."""
    n = len(a)
    best_direction = None
    best_improvement = 0
    
    peaks = top_k[:min(10, len(top_k))]
    
    for i in range(min(n_pairs, len(peaks))):
        for j in range(i+1, min(n_pairs, len(peaks))):
            k1, k2 = peaks[i], peaks[j]
            g1 = np.zeros(n)
            g2 = np.zeros(n)
            
            m1_start = max(0, k1 - n + 1)
            m1_end = min(n - 1, k1)
            if m1_start <= m1_end:
                g1[m1_start:m1_end+1] = 2.0 * a[k1 - np.arange(m1_start, m1_end+1)]
            
            m2_start = max(0, k2 - n + 1)
            m2_end = min(n - 1, k2)
            if m2_start <= m2_end:
                g2[m2_start:m2_end+1] = 2.0 * a[k2 - np.arange(m2_start, m2_end+1)]
            
            # Try d = -(g1 + g2) / 2 (reduce both peaks)
            d = -(g1 + g2) / 2
            # Project to zero-mean (to preserve normalization approximately)
            d -= d.mean()
            d_norm = np.linalg.norm(d)
            if d_norm < 1e-15:
                continue
            d /= d_norm
            
            # Check improvement
            for lr in [0.005, 0.002, 0.001]:
                a_new = normalize(a - lr * d)
                c_new = fast_convolve(a_new)
                # Check if the specific peaks decreased
                if c_new[k1] < c[k1] and c_new[k2] < c[k2]:
                    improvement = (c[k1] + c[k2]) - (c_new[k1] + c_new[k2])
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_direction = a_new.copy()
                    break
    
    return best_direction
