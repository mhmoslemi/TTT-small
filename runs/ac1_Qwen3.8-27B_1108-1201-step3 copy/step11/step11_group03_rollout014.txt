def proj_simplex(v):

def logsumexp_projected_gradient(a0, time_limit, T_schedule, max_iters, verbose=True):
    """
    Projected gradient descent with LogSumExp smoothing.
    Uses adaptive temperature annealing and Barzilai-Borwein step sizes.
    """
    n = len(a0)
    a = proj_simplex(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    # BB state
    prev_a = a.copy()
    prev_grad = None
    eta = 0.05

    n_stages = len(T_schedule)
    iters_per_stage = max(5, max_iters // n_stages)
    no_improve_count = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        stage = min(it // iters_per_stage, n_stages - 1)
        T = T_schedule[stage]

        # Compute LogSumExp weights
        b = fast_convolve(a)
        max_b = b.max()
        shifted = (b - max_b) / T
        # Clip for numerical stability
        shifted = np.clip(shifted, -50, 50)
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()

        # Gradient via cross-correlation
        grad = 2.0 * np.correlate(w, a, mode='full')[:n]

        # Barzilai-Borwein step size
        if prev_grad is not None and it >= 2:
            s_vec = a - prev_a
            y_vec = grad - prev_grad
            ss = np.dot(s_vec, s_vec)
            sy = np.dot(s_vec, y_vec)
            if ss > 1e-30 and abs(sy) > 1e-30:
                if sy > 0:
                    if it % 2 == 0:
                        eta_new = ss / sy
                    else:
                        eta_new = sy / ss
                    eta = 0.3 * eta + 0.7 * min(eta_new, 1.0)
                    eta = max(eta, 1e-6)

        # Line search (backtracking)
        alpha = eta
        obj_current = 2.0 * n * (max_b + T * np.log(exp_shifted.sum()))
        for ls in range(20):
            a_try = a - alpha * grad
            a_try = proj_simplex(a_try)
            b_try = fast_convolve(a_try)
            max_b_try = b_try.max()
            shifted_try = (b_try - max_b_try) / T
            shifted_try = np.clip(shifted_try, -50, 50)
            exp_try = np.exp(shifted_try)
            obj_try = 2.0 * n * (max_b_try + T * np.log(exp_try.sum()))
            if obj_try < obj_current - 1e-14 * abs(obj_current):
                break
            alpha *= 0.5

        a_new = proj_simplex(a - alpha * grad)

        # Evaluate actual (unsmoothed) score
        score_new = evaluate_local(a_new)
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        # Update BB state
        prev_a = a.copy()
        prev_grad = grad.copy()
        a = a_new

        if verbose and (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-GD iter={it+1} ({elapsed:.0f}s): score={best_score:.10f} T={T:.4f} eta={eta:.6f}")

        if no_improve_count > 100:
            eta *= 0.5
            no_improve_count = 0
            if eta < 1e-8:
                break

    return best_a, best_score

def solve_active_qcqp_cvxpy(a_init, active_indices, solver, max_iters_check=3):

    """

                return a_val, float(max(0, t_var.value))

def solve_active_slsqp(a_init, active_indices, time_limit):
    """
    Solve the active-set problem using SLSQP.
    Minimize max(a^T Q_k a for k in active) subject to sum(a)=1, a>=0.
    Uses the smooth max approximation.
    """
    n = len(a_init)
    Q_list = [build_q_dense(int(k), n) for k in active_indices]
    t0 = time.monotonic()

    def obj(x):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        vals = [x @ Q @ x for Q in Q_list]
        return max(vals)

    def obj_grad(x):
        if time.monotonic() - t0 > time_limit:
            return np.zeros(n)
        vals = [x @ Q @ x for Q in Q_list]
        max_val = max(vals)
        # Subgradient: average of active Q*x
        active_in_subgrad = [i for i, v in enumerate(vals) if v >= max_val * (1 - 1e-6)]
        g = np.zeros(n)
        for i in active_in_subgrad:
            g += 2.0 * Q_list[i] @ x
        g /= len(active_in_subgrad)
        return g

    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - 1.0, 'jac': lambda x: np.ones(n)}]
    bounds = [(0, None)] * n

    x0 = a_init.copy()
    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 500, 'ftol': 1e-16, 'disp': False})
        if res.success or res.fun < 1e10:
            a_opt = np.maximum(res.x, 0)
            s = a_opt.sum()
            if s > 1e-15:
                a_opt /= s
                return a_opt
    except Exception:
        pass
    return None


def identify_active_set(a, rel_tol=1e-5):

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return np.asarray(f, dtype=np.float64).copy()
    s = np.asarray(f).sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = np.asarray(f, dtype=np.float64) / s
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def multiplicative_weights_refine(a0, time_limit, n_steps=20000):
    """
    Multiplicative weights (mirror descent) targeting active peaks.
    Fast and effective for final polishing.
    """
    n = len(a0)
    a = proj_simplex(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    eta_base = 0.02
    scale = 1.0
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Focus on top peaks
        threshold = max_b * (1.0 - 5e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        # Use top 5 active
        if len(active) > 5:
            active = active[np.argsort(b[active])[-5:]]

        # Gradient: weighted average of contributions from active peaks
        g = np.zeros(n)
        for k in active:
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            g[idx] += a[k_int - idx]
        g /= len(active)

        # Multiplicative update
        eta = eta_base * scale
        a_new = a * np.exp(-eta * g)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = evaluate_local(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 50:
            scale *= 0.8
            no_improve = 0
            if scale < 0.003:
                break

    return best_a, best_score

    # === Phase 1: LogSumExp projected gradient (main workhorse) ===
    print(f"\n=== Phase 1: LSE Projected Gradient n={n_best} (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        T_schedule = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        a_lp, _ = logsumexp_projected_gradient(best_seq, t_alloc, T_schedule, max_iters=500000)
        update_best(a_lp, "lse_gd_phase1")

    # === Phase 2: Active-set identification + QCQP polish ===
    print(f"\n=== Phase 2: Active QCQP Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        # Identify active set with tight tolerance
        active_tight = identify_active_set(best_seq, rel_tol=1e-6)
        active_medium = identify_active_set(best_seq, rel_tol=1e-4)
        active_wide = identify_active_set(best_seq, rel_tol=1e-3)
        print(f"  Active sets: tight={len(active_tight)}, medium={len(active_medium)}, wide={len(active_wide)}")
        print(f"  Tight active indices: {active_tight[:20]}...")

        # Try QCQP with tight active set
        if len(active_tight) <= 30:
            a_q1, t_val = solve_active_qcqp_cvxpy(best_seq, active_tight, solver)
            update_best(a_q1, "qcqp_tight")

        # Try with medium active set
        if len(active_medium) <= 50:
            a_q2, t_val2 = solve_active_qcqp_cvxpy(best_seq, active_medium, solver)
            update_best(a_q2, "qcqp_medium")

        # Try SLSQP on tight active set (for larger n)
        if len(active_tight) <= 20:
            t_slsqp = min(rem * 0.05, 30.0)
            a_s1 = solve_active_slsqp(best_seq, active_tight, t_slsqp)
            update_best(a_s1, "slsqp_tight")

    # === Phase 3: Multiplicative weights polish ===
    print(f"\n=== Phase 3: Multiplicative Weights (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_ms = min(rem * 0.10, 100.0)
    if t_ms > 10:
        a_ms, _ = multiplicative_weights_refine(best_seq, t_ms, n_steps=30000)
        update_best(a_ms, "mult_weights")

    # === Phase 4: Another LSE pass with lower temperatures ===
    print(f"\n=== Phase 4: LSE Low-Temp Pass (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 150.0)
    if t_alloc > 15:
        T_schedule2 = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
        a_lp2, _ = logsumexp_projected_gradient(best_seq, t_alloc, T_schedule2, max_iters=200000)
        update_best(a_lp2, "lse_lowtemp")

    # === Phase 5: Final QCQP polish ===
    print(f"\n=== Phase 5: Final QCQP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        active_final = identify_active_set(best_seq, rel_tol=1e-7)
        if len(active_final) <= 20:
            a_qf, _ = solve_active_qcqp_cvxpy(best_seq, active_final, solver)
            update_best(a_qf, "qcqp_final")
        # Also try SLSQP
        if len(active_final) <= 15:
            a_sf = solve_active_slsqp(best_seq, active_final, min(rem * 0.05, 20.0))
            update_best(a_sf, "slsqp_final")

    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration (score={best_score:.10f}) ===")
    n_current = len(best_seq)
    n_candidates = [1197, 1198, 1199, 1201, 1202, 1203, 1204]
    n_candidates = [n for n in n_candidates if n != n_current]

    for n_try in n_candidates:
        rem = deadline - time.monotonic()
        if rem < 40:
            print(f"  Running out of time, stopping n exploration")
            break

        t_n = min(rem * 0.10, 60.0)
        if t_n < 15:
            continue

        print(f"  Trying n={n_try}, budget={t_n:.0f}s")
        init_n = interpolate_sequence(best_seq, n_try)

        # Quick LSE refinement
        T_sched_n = [0.5, 0.1, 0.02, 0.005, 0.001]
        a_n, sc_n = logsumexp_projected_gradient(init_n, t_n * 0.7, T_sched_n, max_iters=100000, verbose=False)
        update_best(a_n, f"lse_n{n_try}")

        # Quick multiplicative polish
        rem = deadline - time.monotonic()
        if rem > 10 and best_seq is not None and len(best_seq) == n_try:
            a_n2, _ = multiplicative_weights_refine(best_seq, min(rem * 0.03, 10.0), n_steps=5000)
            update_best(a_n2, f"mult_n{n_try}")

    # === Phase 7: Perturbation exploration ===
    print(f"\n=== Phase 7: Perturbation Exploration (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        n_pert = len(best_seq)
        for attempt in range(5):
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            t_pert = min(rem * 0.05, 20.0)
            a_p, sc_p = retry_with_different_start(best_seq, n_pert, t_pert, rng)
            update_best(a_p, f"perturb_{attempt}")

    # === Phase 8: Final ultimate polish ===
    print(f"\n=== Phase 8: Ultimate Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        # Very tight LSE
        T_ult = [0.001, 0.0005, 0.0002, 0.0001]
        a_ult, _ = logsumexp_projected_gradient(best_seq, min(rem * 0.4, 30.0), T_ult, max_iters=50000, verbose=False)
        update_best(a_ult, "ult_lse")

        rem = deadline - time.monotonic()
        if rem > 10:
            # Final QCQP with very tight active set
            active_ult = identify_active_set(best_seq, rel_tol=5e-8)
            if len(active_ult) <= 10:
                a_qult, _ = solve_active_qcqp_cvxpy(best_seq, active_ult, solver)
                update_best(a_qult, "ult_qcqp")

            rem = deadline - time.monotonic()
            if rem > 5:
                a_mf, _ = multiplicative_weights_refine(best_seq, min(rem * 0.5, 10.0), n_steps=10000)
                update_best(a_mf, "ult_mult")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
