def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

    """Cross-correlation of w (length 2n-1) and a (length n), result length n.
    returns[m] = sum_j w[m+j] * a[j] for j=0..n-1 (where valid)

    """Solve LP: max sum(b) s.t. f*b <= M, b >= 0.
    Returns optimal b (normalized) and the optimal value.
    """
    n = len(f)
    
    # Build constraint matrix F (2n-1 x n)
    # F[k, j] = f[k-j] if 0 <= k-j <= n-1
    # Use sparse matrix for efficiency
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    F = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # LP: max 1^T b s.t. F b <= M, b >= 0

        b = np.maximum(b, 0)
        s = b.sum()
        if s > 1e-15:
            b = b / s
        return b, -res.fun
    return None, None

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_corr_w_a(w, a):
    """Compute R[m] = sum_j w[m+j] * a[j] for m=0..n-1 via FFT."""
    n = len(a)
    L = 1 << (2 * n - 1 + n).bit_length()
    W = np.fft.fft(w, L)
    A = np.fft.fft(a, L)
    R = np.fft.ifft(W * np.conj(A)).real[:n]
    return R


def evaluate_local(a):
    """Evaluate score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize_a(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def lse_objective_and_grad_z(z, T):
    """LSE-smoothed objective and gradient in log-parameterization.
    a = softmax(z), returns (obj, grad_z).
    """
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    n = len(a)

    c = fast_convolve(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    lse_val = c_max + T * np.log(exp_c.sum())
    obj = 2.0 * n * lse_val

    w = exp_c / exp_c.sum()
    grad_a = 2.0 * n * fast_cross_corr_w_a(w, a)

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def _func_wrapper(z, T):
    obj, _ = lse_objective_and_grad_z(z, T)
    return obj


def _jac_wrapper(z, T):
    _, g = lse_objective_and_grad_z(z, T)
    return g


def lbgfs_optimize(a0, time_limit, T_stages, maxiter_per_stage=2000):

def lp_refine(a0, time_limit, t_blend=0.1, max_iters=200):
    """Iterative LP-based refinement. Blend f with LP solution each step."""
    a = normalize_a(a0.copy())
    n = len(a)
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    it = 0
    t = t_blend

    while it < max_iters and time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        M = c.max()

        b_lp = solve_lp_direction(a, M)
        if b_lp is None:
            break

        # Blend
        a_new = (1 - t) * a + t * b_lp
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        sc = evaluate_local(a_new)

        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            it += 1
            if it % 10 == 0:
                print(f"    LP it={it}: score={best_score:.8f} t={t:.4f}")
        else:
            t *= 0.5
            if t < 0.001:
                break
        it += 1

    return best_a, best_score

    lr = 0.005

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        A = rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        r = (P - P_mean) / (P_mean + 1e-15)

        H = np.zeros(fft_size // 2 + 1, dtype=np.complex128)
        valid = P > 1e-12
        H[valid] = -r[valid] * P_mean * np.conj(A[valid]) / (P[valid] + 1e-10)
        H[0] = 0.0
        h = irfft(H, fft_size)[:n]
        h -= h.mean()
        h_norm = np.linalg.norm(h)
        if h_norm < 1e-15:
            break
        h = h / h_norm

        improved = False
        for scale in [lr, lr * 0.3, lr * 0.1]:
            a_new = a + scale * h
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                improved = True
            break
        if not improved:
            lr *= 0.7
        if it % 200 == 0 and it > 0:
            lr *= 0.9

    return best_a, best_score


def diverse_init(n, rng):
    """Generate diverse initial sequences with harmonic/analytical structure."""
    candidates = []
    x = np.linspace(0, 1, n)

    # Half-sine
    a = np.sin(np.pi * x)
    candidates.append(normalize_a(a))

    # Asymmetric: half-sine with exponential decay
    a = np.sin(np.pi * x) * np.exp(-1.5 * x)
    candidates.append(normalize_a(a))

    # Asymmetric: half-sine with exponential growth
    a = np.sin(np.pi * x) * np.exp(1.5 * x)
    candidates.append(normalize_a(a))

    # Power-skewed
    a = x ** 0.5 * np.sin(np.pi * x)
    candidates.append(normalize_a(a))

    a = (1 - x) ** 0.5 * np.sin(np.pi * x)
    candidates.append(normalize_a(a))

    # Dual frequency
    a = np.sin(np.pi * x) ** 2 + 0.3 * np.sin(2 * np.pi * x) ** 2
    candidates.append(normalize_a(a))

    # Cosine modulation
    a = 0.5 + 0.4 * np.cos(2 * np.pi * 1.5 * x) + 0.1 * np.cos(2 * np.pi * 4 * x)
    a = np.maximum(a, 0)
    candidates.append(normalize_a(a))

    # Edges weighted
    a = np.exp(-3 * x) + np.exp(-3 * (1 - x)) + 0.5
    candidates.append(normalize_a(a))

    # Random smooth (few harmonics)
    coeffs = rng.normal(0, 1, 8)
    a = np.zeros(n)
    for i, c_val in enumerate(coeffs):
        a += c_val * np.cos(2 * np.pi * (i + 1) * x + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a + 0.5, 0)
    candidates.append(normalize_a(a))

    # Uniform
    candidates.append(np.ones(n) / n)

    # Strongly asymmetric (checkerboard-like but smooth)
    a = np.exp(-5 * (x - 0.3) ** 2) + 0.3 * np.exp(-5 * (x - 0.7) ** 2)
    candidates.append(normalize_a(a))

    # High-frequency modulation
    a = np.sin(np.pi * x) * (1 + 0.3 * np.cos(2 * np.pi * 10 * x))
    a = np.maximum(a, 0)
    candidates.append(normalize_a(a))

    return candidates


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize_a(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return normalize_a(f_interp)


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LP-based direction finding, LSE gradient descent, spectral shaping,
    and diverse initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # Phase 1: Diverse initialization
    print(f"\n=== Phase 1: Diverse Initialization ===")
    n_init = len(best_seq) if best_seq is not None else 1200

    if best_seq is None:
        candidates = diverse_init(1200, rng)
        for i, c in enumerate(candidates):
            sc = evaluate_local(c)
            print(f"  Init {i}: score={sc:.6f}")
            if sc < best_score:
                best_score = sc
                best_seq = c.copy()
        print(f"  Best init: {best_score:.8f}")
    else:
        # Also try some diverse inits at the same length
        candidates = diverse_init(n_init, rng)
        for i, c in enumerate(candidates[:5]):
            sc = evaluate_local(c)
            if sc < best_score:
                best_score = sc
                best_seq = c.copy()
                print(f"  New best from init {i}: {sc:.8f}")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.8f}")

    # Phase 2: L-BFGS with LSE (main length)
    print(f"\n=== Phase 2: L-BFGS LSE (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages = [20, 10, 5, 3, 2, 1, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002]
        a_res = lbgfs_optimize(best_seq, t_alloc, T_stages, maxiter_per_stage=2000)
        try_update(a_res, "lbgfs_main")

    # Phase 3: LP-based refinement (most important)
    print(f"\n=== Phase 3: LP Refinement (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = lp_refine(best_seq, t_alloc, t_blend=0.15, max_iters=300)
        try_update(a_lp, "lp_refine_1")

    # Phase 3b: Second LP pass with smaller blend
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        print(f"  LP Pass 2 budget: {t_alloc:.0f}s")
        a_lp2, _ = lp_refine(best_seq, t_alloc, t_blend=0.05, max_iters=200)
        try_update(a_lp2, "lp_refine_2")

    # Phase 4: Spectral shaping
    print(f"\n=== Phase 4: Spectral Shaping (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_spec, _ = spectral_refine(best_seq, t_alloc, n_iters=5000)
        try_update(a_spec, "spectral_1")

    # Phase 5: L-BFGS polish after all heuristics
    print(f"\n=== Phase 5: L-BFGS Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        T_stages2 = [2, 1, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]
        a_pol = lbgfs_optimize(best_seq, t_alloc, T_stages2, maxiter_per_stage=1500)
        try_update(a_pol, "lbgfs_polish")

    # Phase 6: Try different lengths
    print(f"\n=== Phase 6: Length Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for n_try in [n_best + 100, n_best - 100, n_best + 200, n_best + 300]:
            if n_try < 200 or n_try > 3000:
                continue
            rem_sub = deadline - time.monotonic()
            if rem_sub < 30:
                break
            init = interpolate_sequence(best_seq, n_try)
            t_sub = min(rem_sub * 0.25, 80.0)
            T_short = [5, 2, 1, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
            a_try = lbgfs_optimize(init, t_sub, T_short, maxiter_per_stage=1000)
            try_update(a_try, f"length_{n_try}")

    # Phase 7: Final LP + LSE combo
    print(f"\n=== Phase 7: Final Combo (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        # Short LP pass
        t_lp = min(rem * 0.3, 60.0)
        a_lp3, _ = lp_refine(best_seq, t_lp, t_blend=0.03, max_iters=100)
        try_update(a_lp3, "final_lp")

        # Then LSE polish
        rem = deadline - time.monotonic()
        if rem > 20:
            T_final = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
            a_fin = lbgfs_optimize(best_seq, min(rem * 0.6, 45.0), T_final, maxiter_per_stage=1000)
            try_update(a_fin, "final_lse")

    # Final LP spray
    rem = deadline - time.monotonic()
    if rem > 15:
        a_final_lp, _ = lp_refine(best_seq, min(rem * 0.5, 20.0), t_blend=0.01, max_iters=50)
        try_update(a_final_lp, "final_lp_spray")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
