from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

# LSE objective with active set
def lse_active_set_obj_grad(a, T, top_k=50):
    """LSE over top-k peaks for focused gradient."""
    n = len(a)
    c = fast_convolve(a)
    # Find top-k peaks
    k = min(top_k, len(c))
    top_indices = np.argpartition(c, -k)[-k:]
    c_top = c[top_indices]
    c_max = c_top.max()
    exp_c = np.exp((c_top - c_max) / T)
    w = exp_c / exp_c.sum()
    lse_val = c_max + T * np.log(exp_c.sum())
    obj = 2.0 * n * lse_val
    
    # Full LSE weight for gradient (but focused on active set)
    c_full_max = c.max()
    exp_full = np.exp((c - c_full_max) / T)
    W = exp_full / exp_full.sum()
    
    # Gradient computation
    N = 1 << (len(W) + n - 1).bit_length()
    Wf = rfft(W, N)
    Af = rfft(a, N)
    R = irfft(Wf * np.conj(Af), N)
    b = R[:n]
    g = 2.0 * b
    idx = 2 * np.arange(n)
    valid = idx < (2 * n - 1)
    g[valid] -= a[valid] * W[idx[valid]]
    
    return obj, g

# Full LSE
def lse_full_obj_grad(a, T):
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    w = exp_c / exp_c.sum()
    lse_val = c_max + T * np.log(exp_c.sum())
    obj = 2.0 * n * lse_val
    
    N = 1 << (len(w) + n - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    R = irfft(W * np.conj(A), N)
    b = R[:n]
    g = 2.0 * b
    idx = 2 * np.arange(n)
    valid = idx < (2 * n - 1)
    g[valid] -= a[valid] * w[idx[valid]]
    
    return obj, g

# Subgradient of the max (exact, not smoothed)
def max_subgrad(a):
    """Return subgradient of 2n * max(conv(a,a)) (assuming sum(a)=1)."""
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    # All ties
    tol = 1e-10
    active = np.where(c >= c_max - tol)[0]
    
    g = np.zeros(n)
    for k in active:
        # d(conv(a,a)[k])/da[m] = a[k-m] + a[k-m] (if k-m valid) ... 
        # Actually conv(a,a)[k] = sum_i a[i]*a[k-i]
        # d/d a[m] = a[k-m] (from i=m term) + a[m] (from k-i=m, i=k-m term)
        # But if m == k-m, i.e., k=2m, then it's 2*a[m]
        # So d/d a[m] = a[k-m] + a[m] if m != k-m, else 2*a[m]
        # Wait, that's not right. Let me be more careful.
        # conv(a,a)[k] = sum_{i: valid} a[i] * a[k-i]
        # d/d a[m]: appears when i=m (giving a[k-m]) or when k-i=m, i.e., i=k-m (giving a[m])
        # If m == k-m (k=2m), both terms are the same: 2*a[m]
        # Otherwise: a[k-m] + a[m]
        # But we need to check bounds: 0 <= i < n and 0 <= k-i < n
        # i=m requires 0<=m<n and 0<=k-m<n, i.e., k-n+1 <= m <= k
        # i=k-m requires 0<=k-m<n and 0<=m<n, same condition
        # So for valid m (k-n+1 <= m <= k and 0 <= m < n):
        #   d/d a[m] = a[k-m] + a[m] if m != k-m
        #   d/d a[m] = 2*a[m] if m == k-m
        #   But actually, if m == k-m, the term a[m]*a[m] appears once in the sum,
        #   so d/d a[m] = 2*a[m]. If m != k-m, both a[m]*a[k-m] and a[k-m]*a[m] appear,
        #   so d/d a[m] = a[k-m] + a[m].
        #   Wait no. In the sum sum_i a[i]*a[k-i], the term a[m]*a[k-m] appears when i=m,
        #   and the term a[k-m]*a[m] appears when i=k-m. These are the SAME term if k=2m.
        #   So: if k=2m, d/d a[m] = 2*a[m]. Otherwise, d/d a[m] = a[k-m] + a[m].
        
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for m in range(js, je + 1):
            if k == 2 * m:
                g[m] += 2.0 * a[m]
            else:
                g[m] += a[k - m] + a[m]
    
    g /= len(active)
    return 2.0 * n * g  # Scale by 2n

# Peak equalization: minimize the max by pushing down the highest peaks
def peak_equalize(a, n_iters=1000, lr=0.001):
    """Iteratively reduce the peak autocorrelation."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > 60:  # Time limit per call
            break
        
        c = fast_convolve(a)
        c_max = c.max()
        
        # Find the top few peaks
        top_k = min(20, 2*n-1)
        top_indices = np.argpartition(c, -top_k)[-top_k:]
        
        # Compute subgradient from top peaks only
        g = np.zeros(n)
        for k in top_indices:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for m in range(js, je + 1):
                if k == 2 * m:
                    g[m] += 2.0 * a[m]
                else:
                    g[m] += a[k - m] + a[m]
        g /= len(top_indices)
        
        # Project gradient to tangent space of simplex
        g_tan = g - np.mean(g)
        
        # Step
        a_new = a - lr * g_tan
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            # Decrease lr slowly
            if it % 100 == 0:
                lr *= 0.95
        else:
            lr *= 0.8
    
    return best_a, best_score

# L-BFGS optimization
def lbgfs_optimize(a0, time_limit, T_stages=None, maxiter_per_stage=2000):
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [20, 10, 5, 2, 1, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func(z_in):
            a = _softmax(z_in)
            obj, _ = lse_full_obj_grad(a, stage_T)
            return obj
        
        def jac(z_in):
            a = _softmax(z_in)
            _, grad_a = lse_full_obj_grad(a, stage_T)
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        
        iters = min(maxiter_per_stage, max(10, int(remaining * 20)))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-16,
                         'maxfun': iters * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return _softmax(z)

def _softmax(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

Actually, let me think about this more carefully. For a "random" sequence (each element independent, Chebyshev random variables centered at 0 with variance 1/n^2), the autocorrelation at lag k (for |k| >> 1) is approximately 0 with standard deviation ~ 1/n. So max over 2n lags would be ~ sqrt(2 log(2n)) / n. Score ~ 2n * sqrt(2 log(2n)) / n = 2*sqrt(2 log(2n)). For n=1200: 2*sqrt(2*ln(2400)) ≈ 2*sqrt(2*7.78) ≈ 2*3.95 ≈ 7.9. That's much worse than 1.5.

For a "structured" sequence like a sine wave: a[i] = (1+cos(2πi/n))/n. The autocorrelation will be peaked. Score ≈ 2.

The optimal seems to be around 1.5, which suggests a very specific structure.

Let me focus on making the optimization work well. The key issues with the previous code:
1. L-BFGS convergence: need enough iterations at each temperature
2. Temperature schedule: too many stages waste time, too few don't converge
3. The subgradient method is slow for n=1200 (O(n) per peak, and we have up to 20 peaks)

Let me optimize the subgradient computation using FFT:

For a given lag k, the gradient contribution to a[m] is:
- a[k-m] for m in [max(0,k-n+1), min(n-1,k)]
- + a[m] for the same range (if k != 2m)
- + a[m] extra (if k == 2m, making it 2*a[m])

The first part (a[k-m]) is a cross-correlation of a with a delta at k.
The second part (a[m]) is just a itself, restricted to the valid range.

So for a set of active lags S:
g[m] = sum_{k in S} a[k-m] * I[valid] + sum_{k in S} a[m] * I[valid, k!=2m] + sum_{k in S: k=2m} a[m]

The first term is the cross-correlation of a with the indicator of S (shifted).
The second term is a[m] * |{k in S: k-n+1 <= m <= k, k != 2m}|.

This can be computed efficiently with FFT for the first term.

Actually, for the LSE approach, the gradient is already computed via FFT in the previous code. Let me just make sure the temperature schedule is optimal.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def softmax_z(z):
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def lse_obj_grad_z(z, T):
    a = softmax_z(z)
    n = len(a)
    c = fast_convolve(a)
    cmax = c.max()
    exp_c = np.exp((c - cmax) / T)
    w = exp_c / exp_c.sum()
    obj = 2.0 * n * (cmax + T * np.log(exp_c.sum()))
    
    # Gradient w.r.t. a
    N = 1 << (len(w) + n - 1).bit_length()
    Wf = rfft(w, N)
    Af = rfft(a, N)
    R = irfft(Wf * np.conj(Af), N)
    grad_a = 2.0 * R[:n]
    idx = 2 * np.arange(n)
    valid = idx < (2 * n - 1)
    grad_a[valid] -= a[valid] * w[idx[valid]]
    
    # Gradient w.r.t. z
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated LSE gradient with adaptive temperature annealing via L-BFGS-B
2. Vectorized subgradient descent on top-k autocorrelation peaks (FFT cross-correlation)
3. Peak equalization with adaptive learning rate
4. Multi-length exploration with interpolation
Key insight: gradient of max at active lag k is 2*a[k-m] (vectorizable via FFT),
and optimal sequences are asymmetric to shift peak autocorrelation off-center.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_from_z(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_obj_grad_z(z, T):
    """LSE-smoothed objective and gradient in log-parameterization."""
    a = softmax_from_z(z)
    n = len(a)
    c = fast_convolve(a)
    cmax = c.max()
    exp_c = np.exp((c - cmax) / T)
    w = exp_c / exp_c.sum()
    obj = 2.0 * n * (cmax + T * np.log(exp_c.sum()))

    # Gradient w.r.t. a via FFT cross-correlation
    N = 1 << (len(w) + n - 1).bit_length()
    Wf = rfft(w, N)
    Af = rfft(a, N)
    R = irfft(Wf * np.conj(Af), N)
    grad_a = 2.0 * R[:n]
    idx = 2 * np.arange(n)
    valid = idx < (2 * n - 1)
    grad_a[valid] -= a[valid] * w[idx[valid]]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbfgs_lse(a0, time_limit, T_list, verbose=False):
    """L-BFGS-B optimization with LSE smoothing and temperature annealing."""
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()

    for i, T in enumerate(T_list):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        maxiter = max(20, int(remaining * 12))

        def func(x):
            return lse_obj_grad_z(x, T)[0]

        def jac(x):
            return lse_obj_grad_z(x, T)[1]

        try:
            res = minimize(func, z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': maxiter, 'ftol': 1e-20,
                                    'gtol': 1e-15, 'maxfun': maxiter * 100,
                                    'disp': False})
            z = res.x
            if verbose and (i % 3 == 0 or i == len(T_list) - 1):
                sc = evaluate_local(softmax_from_z(z))
                print(f"    LSE T={T:.4f}: score={sc:.8f}, iters={res.nit}")
        except Exception:
            pass

    return softmax_from_z(z)


def subgrad_fft(a, top_k=10):
    """Compute subgradient of max(conv(a,a)) using top-k peaks (FFT-accelerated)."""
    n = len(a)
    c = fast_convolve(a)
    k_eff = min(top_k, len(c))
    top = np.argpartition(c, -k_eff)[-k_eff:]

    # Build indicator of active peaks
    u = np.zeros(2 * n - 1)
    u[top] = 1.0

    # Cross-correlation: g[m] = sum_k u[k] * a[k-m] for valid indices
    # Use FFT: cross_corr = ifft(fft(u) * conj(fft(a)))
    fft_size = 1 << (3 * n - 2).bit_length()
    Uf = rfft(u, fft_size)
    Af = rfft(a, fft_size)
    cross = irfft(Uf * np.conj(Af), fft_size)
    # cross[m] = sum_k u[k] * a[(k-m) mod fft_size]
    # For valid m in [0, n-1] and k in [0, 2n-2], k-m is in [-(n-1), 2n-2]
    # The linear cross-correlation at lag m is cross[m] (with proper indexing)
    g = 2.0 * cross[:n] / k_eff

    # Project to tangent space of simplex
    g -= g.mean()
    return g, c


def subgrad_descent_fft(a0, time_limit, top_k=10, lr0=0.002, verbose=False):
    """Subgradient descent with FFT-vectorized gradient, adaptive step size."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    lr = lr0
    stagnation = 0
    step = 0

    while time.monotonic() - t0 < time_limit:
        g, c = subgrad_fft(a, top_k)

        # Backtracking line search
        accepted = False
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            a_new = a - lr * factor * g
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                stagnation = 0
                accepted = True
                break
        if not accepted:
            lr *= 0.8
            stagnation += 1
            if stagnation > 80:
                lr = lr0 * 0.5
                stagnation = 0
                # Adaptive: increase top_k as we stagnate
                top_k = min(top_k + 2, 50)

        step += 1
        if verbose and step % 500 == 0:
            rem = time_limit - (time.monotonic() - t0)
            print(f"    SubGrad step={step}: score={best_score:.8f}, lr={lr:.6f}, top_k={top_k}, rem={rem:.0f}s")

    return best_a, best_score


def lse_with_adaptive_topk(a0, time_limit, init_T=20.0):
    """LSE optimization where effective temperature adapts based on peak spread."""
    z = np.log(np.maximum(a0, 1e-300)).copy()
    n = len(a0)
    t0 = time.monotonic()
    T = init_T

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # Adaptive: reduce T based on how "sharp" the current solution is
        a = softmax_from_z(z)
        c = fast_convolve(a)
        cmax = c.max()
        # If most of the mass in LSE is on one peak, T can be very small
        # Measure: ratio of top peak to second peak
        sorted_c = np.sort(c)[::-1]
        ratio = sorted_c[0] / max(sorted_c[1], 1e-15) if len(sorted_c) > 1 else 1e10
        # If ratio > 10, the problem is nearly solved, use very small T
        if ratio > 50:
            T = max(T * 0.3, 0.0001)
        elif ratio > 10:
            T = max(T * 0.5, 0.001)
        else:
            T = max(T * 0.7, 0.01)

        maxiter = max(30, int(remaining * 10))

        def func(x):
            return lse_obj_grad_z(x, T)[0]

        def jac(x):
            return lse_obj_grad_z(x, T)[1]

        try:
            res = minimize(func, z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': maxiter, 'ftol': 1e-20,
                                    'gtol': 1e-15, 'maxfun': maxiter * 80,
                                    'disp': False})
            z = res.x
        except Exception:
            T = max(T * 0.5, 0.0001)

    return softmax_from_z(z)


def diverse_perturb(a, rng, strength=0.01):
    """Generate diverse perturbed versions of a."""
    n = len(a)
    results = []

    # Gaussian noise
    results.append(normalize(a + rng.normal(0, strength, n)))

    # Low-frequency perturbation
    freq = rng.integers(1, 5)
    phase = rng.uniform(0, 2 * np.pi)
    x = np.linspace(0, 2 * np.pi, n)
    results.append(normalize(a + strength * np.cos(freq * x + phase)))

    # Asymmetric shift (push mass to one side)
    shift = rng.uniform(-0.02, 0.02)
    x = np.linspace(0, 1, n)
    results.append(normalize(a * (1 + shift * (2 * x - 1))))

    # Edge emphasis
    edge = rng.integers(0, 2)
    w = rng.uniform(0.005, 0.02)
    a_new = a.copy()
    if edge == 0:
        a_new[:n // 4] += w
        a_new[n // 4:] -= w * (n // 4) / (3 * n // 4)
    else:
        a_new[-n // 4:] += w
        a_new[:-n // 4] -= w * (n // 4) / (3 * n // 4)
    results.append(normalize(a_new))

    return results


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f)
    return normalize(np.maximum(f_new, 0))


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
    
    Phases:
    1. LSE + L-BFGS with adaptive temperature (primary refinement)
    2. FFT-vectorized subgradient on top peaks (global coordination)
    3. Fine LSE polish (convergence to near-exact max)
    4. Multi-length exploration
    5. Final subgradient + LSE alternation
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a.astype(np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        # Create asymmetric initial sequences
        n_init = 1200
        inits = []
        x = np.linspace(0, 1, n_init)
        # Asymmetric shapes (key: optimal is asymmetric)
        inits.append(normalize(np.exp(-2.0 * (x - 0.25) ** 2) + 0.1))
        inits.append(normalize(np.exp(-2.0 * (x - 0.75) ** 2) + 0.1))
        inits.append(normalize(0.5 + 0.4 * np.cos(2 * np.pi * (x - 0.3))))
        inits.append(normalize(np.sin(np.pi * x) ** 0.5))
        inits.append(normalize(np.exp(-4 * x) + 0.3))
        inits.append(normalize(1.0 + 0.3 * np.cos(4 * np.pi * x)))
        inits.append(normalize(np.ones(n_init)))

        scores = [evaluate_local(c) for c in inits]
        best_idx = np.argmin(scores)
        best_seq = inits[best_idx].copy()
        best_score = scores[best_idx]
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n = len(best_seq)
    print(f"Starting: n={n}, score={best_score:.8f}")

    # === Phase 1: LSE + L-BFGS (primary) ===
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.30, 400.0)
    if t1 > 30:
        print(f"\n=== Phase 1: LSE L-BFGS (budget={t1:.0f}s) ===")
        # Fewer, longer stages for better convergence
        T_stages = [20, 10, 5, 3, 2, 1, 0.7, 0.5, 0.3, 0.2,
                    0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001]
        a1 = lbfgs_lse(best_seq, t1, T_stages, verbose=True)
        update(a1, "lse_1")

    # === Phase 2: FFT subgradient on top peaks ===
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 250.0)
    if t2 > 20:
        print(f"\n=== Phase 2: SubGrad FFT (budget={t2:.0f}s) ===")
        a2, _ = subgrad_descent_fft(best_seq, t2, top_k=8, lr0=0.003, verbose=True)
        update(a2, "subgrad_1")

    # === Phase 3: Fine LSE polish ===
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 200.0)
    if t3 > 20:
        print(f"\n=== Phase 3: Fine LSE (budget={t3:.0f}s) ===")
        T_fine = [1, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005,
                  0.003, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
        a3 = lbfgs_lse(best_seq, t3, T_fine, verbose=True)
        update(a3, "lse_fine")

    # === Phase 4: SubGrad with more peaks ===
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.12, 120.0)
    if t4 > 15:
        print(f"\n=== Phase 4: SubGrad expanded (budget={t4:.0f}s) ===")
        a4, _ = subgrad_descent_fft(best_seq, t4, top_k=20, lr0=0.001, verbose=True)
        update(a4, "subgrad_2")

    # === Phase 5: Length exploration ===
    rem = deadline - time.monotonic()
    if rem > 60:
        print(f"\n=== Phase 5: Length Exploration ===")
        lengths_to_try = sorted(set([n - 100, n + 100, n - 200, n + 200, n + 300]))
        for n_try in lengths_to_try:
            rem_sub = deadline - time.monotonic()
            if rem_sub < 30:
                break
            if n_try < 400 or n_try > 4000:
                continue
            print(f"  Trying n={n_try} (remaining={rem_sub:.0f}s)")
            init = interpolate_sequence(best_seq, n_try)
            t_sub = min(rem_sub * 0.3, 80.0)
            T_short = [5, 3, 2, 1, 0.5, 0.3, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
            a_try = lbfgs_lse(init, t_sub, T_short)
            update(a_try, f"len_{n_try}")
            # Quick subgrad on the best at this length
            rem_sub2 = deadline - time.monotonic()
            if rem_sub2 > 15:
                a_sg, _ = subgrad_descent_fft(a_try, min(rem_sub2 * 0.3, 20.0), top_k=10, lr0=0.002)
                update(a_sg, f"len_{n_try}_sg")

    # === Phase 6: Adaptive LSE (temperature based on peak sharpness) ===
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.15, 150.0)
    if t6 > 15:
        print(f"\n=== Phase 6: Adaptive LSE (budget={t6:.0f}s) ===")
        a6 = lse_with_adaptive_topk(best_seq, t6, init_T=1.0)
        update(a6, "adaptive_lse")

    # === Phase 7: Final alternation (subgrad + LSE) ===
    rem = deadline - time.monotonic()
    if rem > 20:
        print(f"\n=== Phase 7: Final Alternation ===")
        for round_i in range(4):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 8:
                break
            t_sg = min(rem_sub * 0.4, 20.0)
            t_lse = min(rem_sub * 0.4, 20.0)
            a_sg, _ = subgrad_descent_fft(best_seq, t_sg, top_k=5, lr0=0.001)
            update(a_sg, f"alt_sg_{round_i}")
            T_final = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
            a_l = lbfgs_lse(best_seq, t_lse, T_final)
            update(a_l, f"alt_lse_{round_i}")

    # === Final verification ===
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
