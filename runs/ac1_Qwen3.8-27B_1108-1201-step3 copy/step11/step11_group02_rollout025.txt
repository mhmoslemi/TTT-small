import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a = a / s
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())

def multiplicative_weights_descent(a0, eta, max_iters, time_limit):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find contributing positions (within 99.9% of max)
        threshold = max_b * 0.999
        active = np.where(b >= threshold)[0]
        
        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            # g[i] += a[k-i] for valid i
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        # Multiplicative update: a_new[i] ∝ a[i] * exp(-eta * g[i])
        # = exp(log(a[i]) - eta * g[i])
        log_a = np.log(np.maximum(a, 1e-300))
        g_centered = g - g.mean()  # Center for numerical stability
        log_a_new = log_a - eta * g_centered
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new /= a_new.sum()
        
        # Optional: blend with previous for stability
        a = 0.9 * a + 0.1 * a_new  # or just a = a_new
        
        sc = score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score

    A = rfft(a, fft_size)
