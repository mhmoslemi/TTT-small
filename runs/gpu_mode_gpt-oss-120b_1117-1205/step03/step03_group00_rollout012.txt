"""
TriMul (outgoing) forward pass.

The implementation follows the reference PyTorch module but moves the
heavy O(N³) contraction into a custom Triton kernel.  Steps:

1. Input LayerNorm (FP32) → cast to FP16.
2. Linear projections to hidden dimension (FP16).
3. Compute sigmoid gates (FP16) and optionally fuse a binary mask with the
   left/right tensors using a tiny Triton kernel.
4. Core contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   realized as a batched GEMM (matrix‑multiply) for each (batch, head)
   pair with a Triton matmul kernel (block‑wise accumulation in FP32).
5. Output LayerNorm, out‑gate scaling and final linear projection (FP16).
6. Cast result back to FP32.

Only the mask‑gate fusion and the O(N³) contraction are executed on
Triton; the remaining ops use highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1. Mask + gate fusion kernel (optional)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # [total] fp16
    right_ptr,              # [total] fp16
    left_gate_ptr,          # [total] fp16
    right_gate_ptr,         # [total] fp16
    mask_ptr,               # [mask_total] fp16 (broadcasted)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,    # 1 → mask present, 0 → no mask
    H: tl.constexpr,           # hidden dimension (size of innermost channel)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each hidden channel shares the same mask entry
        mask_offs = offs // H
        mask_val = tl.load(mask_ptr + mask_offs, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fuse:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    `mask` may be None (no masking). All tensors must be contiguous
    FP16 tensors.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when unused

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # operation is performed in‑place; nothing is returned


# ----------------------------------------------------------------------
# 2. Core O(N³) contraction kernel (batched GEMM)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,            # fp16  [B, N, N, H]
    right_ptr,           # fp16  [B, N, N, H]
    out_ptr,             # fp16  [B, N, N, H]
    N: tl.int32,         # sequence length
    H: tl.int32,         # hidden dimension
    # strides for left/right (batch, i, k)
    stride_left_batch: tl.int64,
    stride_left_i: tl.int64,
    stride_left_k: tl.int64,
    # strides for right (batch, j, k) – same layout as left
    stride_right_batch: tl.int64,
    stride_right_i: tl.int64,
    stride_right_k: tl.int64,
    # strides for out (batch, i, j)
    stride_out_batch: tl.int64,
    stride_out_i: tl.int64,
    stride_out_j: tl.int64,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # output rows (i)
    pid_n = tl.program_id(1)   # output cols (j)
    pid_bh = tl.program_id(2)  # flattened batch * head index

    # ------------------------------------------------------------------
    # decode batch and head
    # ------------------------------------------------------------------
    bh = pid_bh
    b = bh // H
    d = bh % H
    # base offset for this (batch, head) pair; d is innermost dim (stride = 1)
    base = b * stride_left_batch + d

    # ------------------------------------------------------------------
    # compute tile start offsets
    # ------------------------------------------------------------------
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 for better numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # tiles over the reduction dimension k
    # ------------------------------------------------------------------
    for k in range(0, N, BLOCK_K):
        k0 = k
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: A[i, k]  (shape M x K)
        a_ptrs = left_ptr + base + \
                 offs_i[:, None] * stride_left_i + \
                 offs_k[None, :] * stride_left_k
        mask_a = mask_i[:, None] & mask_k[None, :]
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)

        # right block: B[k, j]  (shape K x N)   note: right is indexed as (k, j)
        b_ptrs = right_ptr + base + \
                 offs_k[:, None] * stride_right_k + \
                 offs_j[None, :] * stride_right_i
        mask_b = mask_k[:, None] & mask_j[None, :]
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # dot product over the K dimension
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # write back tile
    # ------------------------------------------------------------------
    mask_ij = mask_i[:, None] & mask_j[None, :]
    out_ptrs = out_ptr + base + \
               offs_i[:, None] * stride_out_i + \
               offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_ij)


# ----------------------------------------------------------------------
# 3. Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored if
                    config['nomask'] is True
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → FP16
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (FP16)
    # ------------------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gating (FP16)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) with gates via a tiny Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → FP16 flat
        mask_h = mask.to(torch.float16).contiguous()
        mask_f = mask_h.view(-1)

        # flatten tensors for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            hidden_dim,
        )
        # reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # simple element‑wise multiplication when mask is omitted
        left = left * left_gate
        right = right * right_gate

    # free unused temporaries
    del left_gate, right_gate, x_h

    # ------------------------------------------------------------------
    # 5. Core O(N³) contraction: batched GEMM via Triton
    # ------------------------------------------------------------------
    B, N_seq, _, H = left.shape
    assert N_seq == N, "sequence length mismatch"
    out = torch.empty_like(left)  # FP16, same layout as left/right

    # compute strides (all tensors share the same memory layout)
    stride_left_batch = left.stride(0)   # B‑major stride
    stride_left_i = left.stride(1)       # I‑major stride (N*H)
    stride_left_k = left.stride(2)       # K‑major stride (H)

    stride_right_batch = right.stride(0)
    stride_right_i = right.stride(1)     # J‑major stride (N*H) – same as left_i
    stride_right_k = right.stride(2)

    stride_out_batch = out.stride(0)
    stride_out_i = out.stride(1)         # output I‑major stride (N*H)
    stride_out_j = out.stride(2)         # output J‑major stride (H)

    # tile sizes – these work well for the tested shapes on B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows (i)
        triton.cdiv(N, BLOCK_N),   # cols (j)
        B * hidden_dim,            # batch * head
    )
    _trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        N,
        hidden_dim,
        stride_left_batch,
        stride_left_i,
        stride_left_k,
        stride_right_batch,
        stride_right_i,
        stride_right_k,
        stride_out_batch,
        stride_out_i,
        stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise scaling (FP16)

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )  # -> (B, N, N, dim) FP16

    # return in original float32 dtype
    return out.to(torch.float32)
