import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel for batched matrix multiplication (FP16 inputs, FP32 accumulation)
@triton.jit
def batched_matmul_kernel(
    A, B, C,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets inside block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    A_base = A + pid_b * stride_Ab
    B_base = B + pid_b * stride_Bb
    C_base = C + pid_b * stride_Cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        # load A tile
        a_ptrs = A_base + offs_m[:, None] * stride_Am + (k_base + tl.arange(0, BLOCK_K))[None, :] * stride_Ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & ((k_base + tl.arange(0, BLOCK_K)) < K), other=tl.float16(0.0))

        # load B tile
        b_ptrs = B_base + (k_base + tl.arange(0, BLOCK_K))[:, None] * stride_Bk + offs_n[None, :] * stride_Bn
        b = tl.load(b_ptrs, mask=((k_base + tl.arange(0, BLOCK_K))[:, None] < K) & mask_n[None, :], other=tl.float16(0.0))

        # accumulate
        acc += tl.dot(a, b)

    # store result
    c_ptrs = C_base + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    x_h = x.to(torch.float16)

    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # Gating tensors
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    if not nomask:

        mask_h = mask.unsqueeze(-1).to(torch.float16)  # shape (B,N,N,1)

    )

    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass – Triton‑fused O(N³) contraction.

Algorithm
---------
1. Layer‑norm the input (float32 → half).
2. Linear projections to hidden_dim (half precision).
3. Compute three gating tensors via linear + sigmoid (half).
4. Apply optional mask and element‑wise gates (pure PyTorch – cheap).
5. Core contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication (B*H) × (N×N) @ (N×N) performed
   by a custom Triton kernel that uses `tl.dot` (FP16 inputs, FP32 accumulation)
   and respects the 1024‑thread per block limit (32×32 tiles).
6. Output layer‑norm, output‑gate, final linear projection.
7. Return result in float32.

The Triton kernel replaces the expensive torch.bmm and gives a large speedup.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Batched matrix multiplication kernel (FP16 inputs, FP32 accumulation)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A, B, C,                     # pointers
    M, N, K,                     # matrix dimensions (all = seq_len)
    stride_Ab, stride_Am, stride_Ak,   # strides for A (batch, row, col)
    stride_Bb, stride_Bk, stride_Bn,   # strides for B (batch, row, col)
    stride_Cb, stride_Cm, stride_Cn,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)            # batch index (B * hidden_dim)
    pid_m = tl.program_id(1)            # tile row index
    pid_n = tl.program_id(2)            # tile col index

    # offsets within this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_base = A + pid_b * stride_Ab
    b_base = B + pid_b * stride_Bb
    c_base = C + pid_b * stride_Cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        k_idx = k_base + tl.arange(0, BLOCK_K)

        # A[offs_m, k_idx]   -> shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_m[:, None] * stride_Am + k_idx[None, :] * stride_Ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (k_idx[None, :] < K),
                    other=tl.float16(0.0))

        # B[k_idx, offs_n]   -> shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_base + k_idx[:, None] * stride_Bk + offs_n[None, :] * stride_Bn
        b = tl.load(b_ptrs,
                    mask=(k_idx[:, None] < K) & mask_n[None, :],
                    other=tl.float16(0.0))

        acc += tl.dot(a, b)   # accumulate in fp32

    # write out
    c_ptrs = c_base + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/float) or None
            - weights      : dict of model parameters (float32 tensors)
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half precision)
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Gating tensors (half precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 4. Apply mask (if any) and gates – simple elementwise ops
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # broadcast mask over hidden dimension
        mask_h = mask.unsqueeze(-1).to(torch.float16)  # (B, N, N, 1)
        left = left * left_gate * mask_h
        right = right * right_gate * mask_h
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via batched GEMM (Triton)
    #    Shape manipulation:
    #       left_perm  : (B, H, N, N) – rows i, cols k
    #       right_perm : (B, H, N, N) – rows k, cols j
    #    Treat batch = B*H and compute (N×N) @ (N×N) for each.
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    batch = B * H

    # reshape to (batch, N, N)
    left_mat = left_perm.reshape(batch, N, N)
    right_mat = right_perm.reshape(batch, N, N)

    # output buffer
    out_mat = torch.empty_like(left_mat, dtype=torch.float16, device=device)

    # strides (must be integers)
    stride_Ab = left_mat.stride(0)
    stride_Am = left_mat.stride(1)
    stride_Ak = left_mat.stride(2)

    stride_Bb = right_mat.stride(0)
    stride_Bk = right_mat.stride(1)
    stride_Bn = right_mat.stride(2)

    stride_Cb = out_mat.stride(0)
    stride_Cm = out_mat.stride(1)
    stride_Cn = out_mat.stride(2)

    # tiling parameters – product BLOCK_M*BLOCK_N ≤ 1024 (GPU thread limit)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,                      # M
        N,                      # N
        N,                      # K
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 7. Return in float32
    # ------------------------------------------------------------------
    return out.to(torch.float32)
