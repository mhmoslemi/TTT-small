@triton.jit
def fuse_proj_gate_mask(left_proj_ptr, left_gate_ptr, mask_ptr, out_left_ptr, total_elems, H, BLOCK, HAS_MASK):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elems

    # Load proj and gate
    proj = tl.load(left_proj_ptr + offs, mask=valid, other=0.0)
    gate_raw = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)

    # sigmoid
    gate = 1.0 / (1.0 + tl.exp(-gate_raw))

    if HAS_MASK:
        # mask offset: each mask value corresponds to H consecutive entries (since mask broadcast to hidden_dim)
        mask_off = offs // H
        mask_val = tl.load(mask_ptr + mask_off, mask=valid, other=0.0)
        proj = proj * gate * mask_val
    else:
        proj = proj * gate

import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def fuse_proj_gate_mask_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
                               total_elems: tl.int32,
                               hidden_dim: tl.int32,
                               HAS_MASK: tl.constexpr,
                               BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems
    proj = tl.load(proj_ptr + offs, mask=mask, other=tl.float16(0.0))
    gate_raw = tl.load(gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    gate = 1.0 / (1.0 + tl.exp(-gate_raw))
    if HAS_MASK:
        # Each mask entry corresponds to hidden_dim entries
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        proj = proj * gate * mask_val
    else:
        proj = proj * gate
    tl.store(out_ptr + offs, proj, mask=mask)

"""
TriMul outgoing kernel (AlphaFold3).

This implementation follows the reference PyTorch module but fuses many
operations to reduce memory traffic:

1. Input LayerNorm (half precision).
2. Two fused linear layers per side (left/right) that compute both the
   projection and its gating weights in a single GEMM.
3. A Triton kernel fuses the sigmoid gate, optional mask, and the
   element‑wise product: out = proj * sigmoid(gate) * mask.
4. The core O(N³) contraction is expressed as a batch of matrix
   multiplications: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d].
   This is implemented with a single torch.bmm over (B·H) matrices,
   which fully exploits Tensor Cores.
5. The result is normalized, multiplied by an output gate, and projected
   back to the original dimensionality.

All heavy arithmetic is performed in float16; only the final output is
returned in float32. A Triton kernel is required for the gate‑mask fusion,
satisfying the “custom kernel” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse projection, sigmoid gate, and optional mask
# ----------------------------------------------------------------------
@triton.jit
def _fuse_proj_gate_mask_kernel(
    proj_ptr,          # half *
    gate_ptr,          # half *
    mask_ptr,          # half *
    out_ptr,           # half *
    total_elems: tl.int32,
    hidden_dim: tl.int32,
    HAS_MASK: tl.constexpr,   # 1 if mask present, else 0
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elems

    # Load projection and gate values
    proj = tl.load(proj_ptr + offs, mask=valid, other=tl.float16(0.0))
    gate_raw = tl.load(gate_ptr + offs, mask=valid, other=tl.float16(0.0))

    # Sigmoid gate
    gate = 1.0 / (1.0 + tl.exp(-gate_raw))

    if HAS_MASK:
        # One mask value corresponds to `hidden_dim` consecutive entries
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=tl.float16(0.0))
        proj = proj * gate * mask_val
    else:
        proj = proj * gate

    tl.store(out_ptr + offs, proj, mask=valid)


def _apply_fuse_proj_gate_mask(proj, gate, mask, out, hidden_dim, has_mask):
    """
    In‑place: out = proj * sigmoid(gate) * (mask if has_mask else 1)

    All tensors must be contiguous torch.float16.
    """
    total = proj.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    mask_ptr = mask if has_mask else proj  # dummy pointer when mask not used
    _fuse_proj_gate_mask_kernel[grid](
        proj,
        gate,
        mask_ptr,
        out,
        total,
        hidden_dim,
        HAS_MASK=1 if has_mask else 0,
        BLOCK=BLOCK,
    )

        bias=weights["norm.bias"].to(torch.float16),
    )

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 2. Flatten for linear layers
    # ------------------------------------------------------------------
    x_flat = x.reshape(-1, dim)                     # (B*N*N, dim)

    # ------------------------------------------------------------------
    # 3. Fuse left/right projection & gate weights
    # ------------------------------------------------------------------
    left_comb_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                             # (2*H, dim)

    right_comb_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                             # (2*H, dim)

    # left side: proj | gate
    left_comb = F.linear(x_flat, left_comb_weight)  # (B*N*N, 2*H)
    left_proj_raw = left_comb[:, :hidden_dim]       # (B*N*N, H)
    left_gate_raw = left_comb[:, hidden_dim:]       # (B*N*N, H)
    del left_comb  # free

    # right side: proj | gate
    right_comb = F.linear(x_flat, right_comb_weight)  # (B*N*N, 2*H)
    right_proj_raw = right_comb[:, :hidden_dim]       # (B*N*N, H)
    right_gate_raw = right_comb[:, hidden_dim:]       # (B*N*N, H)
    del right_comb  # free

    # output gate (raw, will be sigmoid‑ed later)
    out_gate_raw = F.linear(
        x_flat, weights["out_gate.weight"].to(torch.float16)
    )                                                # (B*N*N, H)

    # ------------------------------------------------------------------
    # 4. Apply mask + sigmoid gates via Triton (or plain ops if no mask)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] -> broadcast over hidden_dim
        mask_h = mask.to(torch.float16).unsqueeze(-1).contiguous()          # (B,N,N,1)
        mask_flat = mask_h.view(-1)                                        # (B*N*N)

        left = torch.empty_like(left_proj_raw)
        _apply_fuse_proj_gate_mask(
            left_proj_raw, left_gate_raw, mask_flat, left, hidden_dim, has_mask=True
        )

        right = torch.empty_like(right_proj_raw)
        _apply_fuse_proj_gate_mask(
            right_proj_raw, right_gate_raw, mask_flat, right, hidden_dim, has_mask=True
        )
    else:
        # No mask: simple element‑wise multiplication with sigmoid
        left = left_proj_raw * left_gate_raw.sigmoid()
        right = right_proj_raw * right_gate_raw.sigmoid()

    # Reshape back to 4‑D tensors
    left = left.view(B, N, N, hidden_dim)      # (B,N,N,H)
    right = right.view(B, N, N, hidden_dim)    # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    # (B, H, N, N) where the last two dims are (i,k) for left and (k,j) for right
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) == rightᵀ

    B_, H_, N_, _ = left_perm.shape
    left_mat = left_perm.view(B_ * H_, N_, N_)          # (B·H, N, N)
    right_mat = right_perm.view(B_ * H_, N_, N_)        # (B·H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)           # (B·H, N, N)

    out = out_mat.view(B, H_, N_, N_).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # ------------------------------------------------------------------
    # 7. Apply output gate and final projection
    # ------------------------------------------------------------------
    out_gate = out_gate_raw.sigmoid().view(B, N, N, hidden_dim)    # (B,N,N,H)
    out_gated = out_norm * out_gate                               # (B,N,N,H)

    out_flat = out_gated.view(-1, hidden_dim)                     # (B*N*N, H)
    out_proj = F.linear(out_flat, weights["to_out.weight"].to(torch.float16))  # (B*N*N, dim)

    out_final = out_proj.view(B, N, N, dim)  # (B,N,N,dim)

    return out_final.to(torch.float32)
```
