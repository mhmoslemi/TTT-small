def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device
    # Convert necessary weights to half
    # maybe convert all to half now
    weight_half = {}
    for k,w in weights.items():
        weight_half[k] = w.to(torch.float16)
    # 1. LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x = x.to(torch.float16)  # [B,N,N,dim] half
    
    # 2. Projections
    left = F.linear(x, weight_half["left_proj.weight"])
    right = F.linear(x, weight_half["right_proj.weight"])
    
    # 3. Gates
    left_gate = F.linear(x, weight_half["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weight_half["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weight_half["out_gate.weight"]).sigmoid()
    
    # 4. Fuse mask and gates
    if not nomask and mask is not None:
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        # flatten everything
        total_elements = left.numel()
        block = 1024
        grid = (triton.cdiv(total_elements, block),)
        # Use kernel
        _gate_mask_fuse_kernel[grid](
            left, right,
            left_gate, right_gate,
            mask_h,
            total_elements,
            BLOCK=block,
            HAS_MASK=1,
            H=hidden_dim,
        )
    else:
        # Simple elementwise multiplication (in-place)
        left = left * left_gate
        right = right * right_gate
    
    # free gates left_gate,right_gate for memory
    # Not needed further
    # 5. Core contraction
    # Use einsum
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)  # half
    
    # 6. Output LN, out_gate, final linear
    # Out LN over hidden dim
    out = F.layer_norm(out, (hidden_dim,), weight=weight_half["to_out_norm.weight"], bias=weight_half["to_out_norm.bias"])

    out = F.linear(out, weight_half["to_out.weight"])
    
    # return cast to float32

@triton.jit
def _gate_mask_fuse_kernel(left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr, total_elements,
                           BLOCK: tl.constexpr, HAS_MASK: tl.constexpr, H: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask_range = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask_range, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask_range, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask_range, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask_range, other=tl.float16(0.0))

    left = left * left_gate
    right = right * right_gate

    if HAS_MASK:
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask_range, other=tl.float16(0.0))
        left = left * mask_val
        right = right * mask_val

    tl.store(left_ptr + offs, left, mask=mask_range)
    tl.store(right_ptr + offs, right, mask=mask_range)
```

"""
TriMul (outgoing) forward pass optimized with Triton.

Algorithm:
- LayerNorm on input.
- Linear projections to hidden dimension.
- Compute three gates (left, right, out) via linear+sigmoid.
- Fuse mask (if present) and left/right gates with an in‑place Triton kernel.
- Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h],
  performed as a batched matrix multiplication (B*H, N, N) @ (B*H, N, N).
- Output LayerNorm, elementwise out‑gate, final linear projection.
- Return float32 tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half*
    right_ptr,         # half*
    left_gate_ptr,     # half*
    right_gate_ptr,    # half*
    mask_ptr,          # half* (optional)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    # apply gates
    left = left * left_gate
    right = right * right_gate

    if HAS_MASK:
        # each H entries correspond to the same mask element
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * mask_val
        right = right * mask_val

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or numeric) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Cast model weights to half‑precision once
    w = {}
    for name, tensor in weights.items():
        w[name] = tensor.to(torch.float16, non_blocking=True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) -> half
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (half)
    # ------------------------------------------------------------------
    left = F.linear(x, w["left_proj.weight"])   # -> (B,N,N,hidden_dim)
    right = F.linear(x, w["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gating tensors (half)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, w["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, w["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, w["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) and gates via Triton (in‑place)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape [B,N,N] -> broadcast over hidden_dim
        mask_h = mask.unsqueeze(-1).to(torch.float16, non_blocking=True).contiguous()
        # Flatten tensors for kernel launch
        total_elements = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)

        # Flatten pointers
        left_flat = left.view(-1)
        right_flat = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)
        mask_flat = mask_h.view(-1)   # shape B*N*N

        _gate_mask_fuse_kernel[grid](
            left_flat, right_flat,
            left_gate_flat, right_gate_flat,
            mask_flat,
            total_elements,
            BLOCK=BLOCK,
            HAS_MASK=1,
            H=hidden_dim,
        )
    else:
        # Simple in‑place gate multiplication
        left.mul_(left_gate)
        right.mul_(right_gate)

    # Gates no longer needed
    del left_gate, right_gate, x

    # ------------------------------------------------------------------
    # 5. Core O(N³) contraction via batched GEMM (half precision)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) where H=hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)
    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N), half
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, w["to_out.weight"])

    # Return in float32 as required
    return out.to(torch.float32)
```

"""
TriMul (outgoing) forward pass optimized with Triton.

Algorithm:
1. LayerNorm on the input (float32) -> half.
2. Linear projections to hidden dimension (half precision).
3. Compute three gating vectors (left, right, out) via Linear + sigmoid.
4. Fuse mask (if present) and left/right gates in‑place with a Triton kernel.
5. Core contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM: reshape both tensors to (B*H, N, N) and call
   torch.bmm, which uses cuBLAS for high performance.
6. Output LayerNorm over the hidden dimension, element‑wise out‑gate, final
   linear projection back to the original channel dimension.
7. Return the result as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half*
    right_ptr,         # half*
    left_gate_ptr,     # half*
    right_gate_ptr,    # half*
    mask_ptr,          # half* (optional)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    # apply per‑element gates
    left = left * left_gate
    right = right * right_gate

    if HAS_MASK:
        # each H entries share the same mask value
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=0.0)
        left = left * mask_val
        right = right * mask_val

    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or numeric) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # Cast model weights to half‑precision once
    # --------------------------------------------------------------
    w = {k: v.to(torch.float16, non_blocking=True) for k, v in weights.items()}

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections to hidden dimension (half)
    # --------------------------------------------------------------
    left = F.linear(x, w["left_proj.weight"])   # (B, N, N, hidden_dim)
    right = F.linear(x, w["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Gating tensors (half)
    # --------------------------------------------------------------
    left_gate = F.linear(x, w["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, w["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, w["out_gate.weight"]).sigmoid()

    # --------------------------------------------------------------
    # 4. Fuse mask (if any) and gates via Triton (in‑place)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → broadcast over hidden_dim
        mask_h = mask.unsqueeze(-1).to(torch.float16, non_blocking=True).contiguous()
        total = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)

        # flatten tensors for kernel launch
        left_flat = left.view(-1)
        right_flat = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)
        mask_flat = mask_h.view(-1)

        _gate_mask_fuse_kernel[grid](
            left_flat, right_flat,
            left_gate_flat, right_gate_flat,
            mask_flat,
            total,
            BLOCK=BLOCK,
            HAS_MASK=1,
            H=hidden_dim,
        )
    else:
        # simple in‑place gate multiplication
        left.mul_(left_gate)
        right.mul_(right_gate)

    # Gates are no longer needed
    del left_gate, right_gate, x

    # --------------------------------------------------------------
    # 5. Core contraction via batched GEMM (half precision)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # reshape to (B, H, N, N) where H = hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)           # (B*H, N, N), half
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # --------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, w["to_out.weight"])

    # Return in float32 as required
    return out.to(torch.float32)
