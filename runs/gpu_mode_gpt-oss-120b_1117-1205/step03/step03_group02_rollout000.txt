"""
TriMul (outgoing) forward pass with a Triton‑fused mask‑gate kernel.

Algorithm
---------
1. Layer‑norm the input (float32) → half‑precision.
2. Linear projections to hidden_dim (half) for the left and right tensors.
3. Compute left/right/out gates (half, sigmoid).
4. Fuse optional mask, left‑gate and right‑gate into a single Triton kernel
   that updates `left` and `right` in‑place.
5. Core O(N³) contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   realized as a batched GEMM over the hidden dimension (half precision,
   float32 accumulation provided by cuBLAS via torch.bmm).
6. Output LayerNorm, multiplicative out‑gate, and final linear projection back
   to the original dimension.
7. Return a float32 tensor.

Only the mask‑gate fusion is performed in Triton; the remaining operations
use highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask (optional) with left/right gates
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # half*, [total]
    right_ptr,              # half*, [total]
    left_gate_ptr,          # half*, [total]
    right_gate_ptr,         # half*, [total]
    mask_ptr,               # half*, [mask_len] (broadcasted)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present, else 0
    H: tl.constexpr,          # hidden_dim
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elements

    # Load data
    left = tl.load(left_ptr + offs, mask=m, other=0.0)
    right = tl.load(right_ptr + offs, mask=m, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=m, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=m, other=0.0)

    if HAS_MASK:
        # One mask entry corresponds to H consecutive hidden values
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=m, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Write back in‑place
    tl.store(left_ptr + offs, left, mask=m)
    tl.store(right_ptr + offs, right, mask=m)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of ``left = left * left_gate * mask`` and
    ``right = right * right_gate * mask`` (if mask is provided).

    All tensors must be ``torch.float16`` and contiguous.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when no mask

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # No return – operation performed in‑place

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/int) – may be ignored
                    when ``config["nomask"]`` is True.
            - weights: dict of model parameter tensors.
            - config: dict containing ``dim`` (int), ``hidden_dim`` (int),
                      and optional ``nomask`` (bool, default True).

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> half)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half precision)
    # ------------------------------------------------------------------
    left = F.linear(
        x, weights["left_proj.weight"].to(torch.float16)
    )  # [B,N,N,hidden_dim]
    right = F.linear(
        x, weights["right_proj.weight"].to(torch.float16)
    )

    # ------------------------------------------------------------------
    # 3. Gates (half precision + sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = F.linear(
        x, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = F.linear(
        x, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) with left/right gates via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] -> broadcast over hidden_dim
        mask_h = mask.to(torch.float16).unsqueeze(-1).contiguous()
        # flatten for kernel
        left_view = left.view(-1)
        right_view = right.view(-1)
        left_gate_view = left_gate.view(-1)
        right_gate_view = right_gate.view(-1)
        mask_view = mask_h.view(-1)  # length B*N*N

        _apply_gate_and_mask(
            left_view,
            right_view,
            left_gate_view,
            right_gate_view,
            mask_view,
            hidden_dim,
        )
        # reshape back (views share storage, so left/right are updated)
        left = left_view.view_as(left)
        right = right_view.view_as(right)
    else:
        # No mask: simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as a batched GEMM over the hidden dimension.
    # ------------------------------------------------------------------
    # Transpose / permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.reshape(B * H, N, N)
    right_mat = right_perm.reshape(B * H, N, N)

    # Batched half‑precision GEMM (cuBLAS handles float32 accumulation internally)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N), half

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise gating (half)
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 7. Cast back to float32
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
