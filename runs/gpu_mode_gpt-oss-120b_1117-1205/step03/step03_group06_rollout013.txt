def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_heads: tl.int32,
    M: tl.int32, N: tl.int32, K: tl.int32,
    stride_ah: tl.int32, stride_am: tl.int32, stride_ak: tl.int32,
    stride_bh: tl.int32, stride_bk: tl.int32, stride_bn: tl.int32,
    stride_ch: tl.int32, stride_cm: tl.int32, stride_cn: tl.int32,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets for rows and columns
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute mask for valid rows/cols
    m_mask = offs_m < M
    n_mask = offs_n < N

    # Pointers start for this batch
    a_base = a_ptr + batch_id * stride_ah
    b_base = b_ptr + batch_id * stride_bh
    c_base = c_ptr + batch_id * stride_ch

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):

        acc += tl.dot(a, b)

    # Write back

    right = torch.nn.functional.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # Gates
    left_gate = torch.nn.functional.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Apply mask if present
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # shape B,N,N,1
        left = left * mask_h
        right = right * mask_h

    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)

"""
TriMul (outgoing) forward pass – Triton‑accelerated batched GEMM.

Algorithm
---------
1. Layer‑norm the input tensor (float32 → float16).
2. Linear projections (dim → hidden_dim) produce `left` and `right`.
3. Compute three sigmoid gates and apply them element‑wise:
   left = left * left_gate, right = right * right_gate.
   If a mask is supplied, multiply both tensors by mask[...,None].
4. Reshape `left` to (B, H, N, N) and `right` to (B, H, N, N) with
   the second spatial dimension transposed, i.e. right = right.permute(0,3,2,1).
5. Fuse the batch and head dimensions (B*H) and run a custom Triton
   kernel that computes a batched GEMM:
       out[b*h, i, j] = Σ_k left[b*h, i, k] * right[b*h, k, j]
   The kernel uses FP16 data with FP32 accumulation and supports
   arbitrary sequence lengths via block‑wise tiling (BLOCK_M, BLOCK_N,
   BLOCK_K).  This replaces the expensive torch.bmm call.
6. Reshape the result back to (B, N, N, H) (half precision).
7. Apply a second LayerNorm over the hidden dimension, multiply by the
   output gate, and a final linear projection back to `dim`.
8. Return the result in float32.

At least one custom Triton kernel is used (the batched GEMM).  All
other operations stay in PyTorch for simplicity and numerical
stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ B) where A, B are fp16 and C is fp16
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_heads: tl.int32,
    M: tl.int32, N: tl.int32, K: tl.int32,
    stride_ah: tl.int32, stride_am: tl.int32, stride_ak: tl.int32,
    stride_bh: tl.int32, stride_bk: tl.int32, stride_bn: tl.int32,
    stride_ch: tl.int32, stride_cm: tl.int32, stride_cn: tl.int32,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)          # which (B*H) matrix
    pid_m   = tl.program_id(1)          # block row index
    pid_n   = tl.program_id(2)          # block col index

    # --------------------------------------------------------------
    # offsets for the current block
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    m_valid = offs_m < M
    n_valid = offs_n < N

    # base pointers for this batch/head
    a_base = a_ptr + batch_id * stride_ah
    b_base = b_ptr + batch_id * stride_bh
    c_base = c_ptr + batch_id * stride_ch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # main loop over K dimension
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k   = k * BLOCK_K
        offs_k  = cur_k + tl.arange(0, BLOCK_K)
        k_valid = offs_k < K

        # A block (M x K)
        a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=m_valid[:, None] & k_valid[None, :], other=0.0)

        # B block (K x N)
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=k_valid[:, None] & n_valid[None, :], other=0.0)

        # block‑wise dot product
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # write result (cast back to fp16)
    # --------------------------------------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=m_valid[:, None] & n_valid[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or numeric) or None
            - weights: dict of model parameters (torch.Tensors)
            - config: dict with keys ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> half)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections (half‑precision)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 3. Gates (half‑precision)
    # --------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # --------------------------------------------------------------
    # 4. Apply gates (and optional mask)
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)            # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # --------------------------------------------------------------
    # 5. Core contraction via Triton‑batched GEMM
    #    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    # --------------------------------------------------------------
    # reshape to (B, H, N, N) and make B transposed on the last two axes
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    batch_heads = B * H

    # flatten batch & head dimension
    left_flat = left_perm.view(batch_heads, N, N)
    right_flat = right_perm.view(batch_heads, N, N)

    out_flat = torch.empty_like(left_flat)

    # stride information (in elements)
    a_stride_batch = left_flat.stride(0)
    a_stride_m = left_flat.stride(1)   # row stride (N)
    a_stride_k = left_flat.stride(2)   # column stride (1)

    b_stride_batch = right_flat.stride(0)
    b_stride_k = right_flat.stride(1)  # row stride (N)
    b_stride_n = right_flat.stride(2)  # column stride (1)

    c_stride_batch = out_flat.stride(0)
    c_stride_m = out_flat.stride(1)    # row stride (N)
    c_stride_n = out_flat.stride(2)    # column stride (1)

    # tiling sizes – these work well on B200 (Ampere)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (batch_heads,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    _batched_gemm_kernel[grid](
        left_flat, right_flat, out_flat,
        batch_heads,
        N, N, N,
        a_stride_batch, a_stride_m, a_stride_k,
        b_stride_batch, b_stride_k, b_stride_n,
        c_stride_batch, c_stride_m, c_stride_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 7. Return in required precision
    # --------------------------------------------------------------
    return out.to(torch.float32)
