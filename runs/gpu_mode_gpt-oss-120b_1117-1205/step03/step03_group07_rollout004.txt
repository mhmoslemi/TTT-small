def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. LayerNorm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # float32

    # 2. Cast to fp16
    x_h = x.to(torch.float16)

    # 3. Linear projections
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))  # shape (B,N,N,hidden_dim)
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # 4. Gates
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # Ensure contiguity
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    # 5. Fuse gate and mask via Triton
    HAS_MASK = 0
    mask_h = None
    if not nomask and mask is not None:
        HAS_MASK = 1
        mask_h = mask.to(torch.float16).contiguous()
    else:
        # dummy mask
        mask_h = left  # same dtype and device

    total_elements = left.numel()
    BLOCK = 8192  # 8k, adjust if needed - ensure divisible size but not required
    grid = (triton.cdiv(total_elements, BLOCK),)

    gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_h,
        total_elements,
        H=hidden_dim,
        HAS_MASK=HAS_MASK,
        BLOCK=BLOCK,
    )
    # after kernel left and right multiplied by gates and mask.

    # 6. Contraction using batched GEMM
    # Permute and reshape
    B, N, _, _ = left.shape  # left shape (B,N,N,hidden_dim)
    # Actually left.shape is (B,N,N,hidden_dim)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden_dim, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, hidden_dim, N, N)

    # Combine batch and hidden_dim
    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B*H, N, N)
    right_mat = right_perm.view(B*H, N, N)

    # Batch matmul
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N), dtype half

    # Reshape back
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # 7. Output LayerNorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    # 8. Apply out_gate and final linear
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # 9. Return float32
    return out.to(torch.float32)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def gate_mask_fuse_kernel(
    left_ptr,      # float16 *
    right_ptr,     # float16 *
    left_gate_ptr, # float16 *
    right_gate_ptr,# float16 *
    mask_ptr,      # float16 *
    total_elements: tl.int32,
    H: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # mask_idx maps each (b,i,k) element to its mask entry; each mask entry
        # corresponds to H consecutive channel values.
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def custom_kernel(data):
    """
    Outgoing TriMul forward pass.
    Steps:
    1) LayerNorm on input.
    2) Half‐precision linear projections (left/right) and sigmoidal gates.
    3) Fuse gate multiplication and optional mask with a Triton kernel.
    4) Batched GEMM implements the O(N³) contraction:
           out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    5) Output LayerNorm, out‑gate, final linear projection.
    Returns a float32 tensor of shape [B, N, N, dim].
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ---------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # shape [B, N, N, dim], dtype float32

    # ---------------------------------------------------------------
    # 2. Cast to fp16 for the heavy work
    # ---------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ---------------------------------------------------------------
    # 3. Linear projections (half)
    # ---------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ---------------------------------------------------------------
    # 4. Gates (half) + optional mask fusion via Triton kernel
    # ---------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # Ensure tensors are contiguous for the kernel
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    # Prepare mask (fp16) if required
    has_mask = 0
    if not nomask and mask is not None:
        has_mask = 1
        mask_h = mask.to(torch.float16).contiguous()
    else:
        # dummy tensor – will never be read when HAS_MASK == 0
        mask_h = left

    total_elements = left.numel()
    # a relatively large block size reduces launch overhead; 8192 works well on B200
    BLOCK = 8192
    grid = (triton.cdiv(total_elements, BLOCK),)

    gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_h,
        total_elements,
        H=hidden_dim,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )
    # After the kernel, left/right already contain the gated (and masked) values.

    # ---------------------------------------------------------------
    # 5. Core contraction via batched GEMM (fp16)
    # ---------------------------------------------------------------
    # left: [B, N, N, H]   right: [B, N, N, H]
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # BMM uses Tensor Cores for fp16 → fast O(N³) contraction
    out_mat = torch.bmm(left_mat, right_mat)   # (B*H, N, N), dtype half

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ---------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection (half)
    # ---------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ---------------------------------------------------------------
    # 7. Cast back to float32 for API compatibility
    # ---------------------------------------------------------------
    return out.to(torch.float32)
