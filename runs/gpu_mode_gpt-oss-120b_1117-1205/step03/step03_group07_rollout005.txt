Design Kernel:

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements
    # load values
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    if HAS_MASK:
        # each mask index is floor division by H
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

"""
Highly‑optimized outgoing TriMul implementation.

Algorithm
---------
1. Layer‑norm the input (float32).
2. Cast to float16 and compute all linear projections/gates with a single
   `torch.nn.functional.linear` call per tensor (efficient cuBLAS GEMMs).
3. Fuse the element‑wise gating and optional mask multiplication in a
   Triton kernel (`_gate_mask_fuse_kernel`).  The kernel works in‑place on the
   flattened tensors and avoids extra Python loops.
4. Compute the O(N³) contraction with a single `torch.einsum` call:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This leverages cuBLAS without any costly transposes or copies.
5. Apply the output LayerNorm, multiply by the output gate and a final linear
   projection back to `dim`.  The result is returned as float32.

The kernel fulfills the “must‑use‑a‑kernel” requirement while the rest of
the work stays in fast PyTorch operations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # float16 *
    right_ptr,         # float16 *
    left_gate_ptr,     # float16 *
    right_gate_ptr,    # float16 *
    mask_ptr,          # float16 *
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present, 0 otherwise
    H: tl.constexpr,          # hidden dimension (innermost size)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # load values (float16)
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # mask is stored without the hidden dimension; each mask element
        # corresponds to H consecutive hidden entries.
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fuse `left = left * left_gate * mask` and `right = right * right_gate * mask`
    using the Triton kernel above.
    All tensors must be contiguous ``torch.float16``.
    """
    total = left.numel()
    BLOCK = 32768  # number of threads per program; chosen for good occupancy
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when mask absent

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # No return – operation is in‑place


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict: mapping of parameter names to torch.Tensor
            - config_dict: must contain "dim" and "hidden_dim",
                           may contain "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to half‑precision for the heavy work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (half precision)
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 4. Gates (half precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 5. Fuse mask and gating via Triton kernel
    # ------------------------------------------------------------------
    # Ensure tensors are contiguous before flattening
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    if not nomask and mask_tensor is not None:
        mask_h = mask_tensor.to(torch.float16).contiguous()
        mask_f = mask_h.view(-1)                              # [B*N*N]
    else:
        mask_f = None

    # Flatten tensors for the kernel (they share storage with the original tensors)
    left_f = left.view(-1)
    right_f = right.view(-1)
    left_gate_f = left_gate.view(-1)
    right_gate_f = right_gate.view(-1)

    _apply_gate_and_mask(
        left_f,
        right_f,
        left_gate_f,
        right_gate_f,
        mask_f,
        hidden_dim,
    )
    # After the kernel, `left` and `right` already contain the gated (and masked) values

    # ------------------------------------------------------------------
    # 6. Core O(N³) contraction using einsum (no extra copies)
    # ------------------------------------------------------------------
    # Shape: (B, N, N, hidden_dim) -> einsum over the third axis (k)
    out = torch.einsum("bikd,bjkd->bijd", left, right)   # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return result in float32 as required by the API
    return out.to(torch.float32)
