"""
Optimized outgoing TriMul forward pass.

The operator computes:
    out[b,i,j,d] = Σ_k left[b,i,k,h] * right[b,j,k,h]
where left/right are linear projections of the input, gated by
sigmoid gates and (optionally) a pairwise mask.  The heavy elementwise
gate+mask fusion is implemented with a Triton kernel; the cubic
contraction is performed as a batched half‑precision GEMM using
torch.bmm (tensor cores).  All other ops (LayerNorm, final projection)
run in half precision and the final result is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half *
    right_ptr,         # half *
    left_gate_ptr,     # half *
    right_gate_ptr,    # half *
    mask_ptr,          # half *
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present, 0 otherwise
    H: tl.constexpr,          # hidden dimension (size of innermost channel)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # Load inputs
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each mask value is repeated H times in the flattened tensors
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * lgate * mask_val
        right = right * rgate * mask_val
    else:
        left = left * lgate
        right = right * rgate

    # Write back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    Fuse ``left = left * left_gate * mask`` and ``right = right * right_gate * mask``.
    All tensors must be contiguous torch.float16.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when mask absent

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # left and right are modified in‑place


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict: mapping from parameter names to tensors (float32)
            - config_dict: contains keys ``dim`` and ``hidden_dim`` and
                           optional ``nomask`` (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (half)
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Compute gates (half) → sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse elementwise mask & gates using Triton (if mask present)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape [B, N, N] → broadcast over hidden_dim
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()

        # Flatten tensors for Triton kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask_h.view(-1)  # length B*N*N

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            hidden_dim,
        )

        # Reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask: simple elementwise multiplication with gates
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    #    Implemented as batched GEMM (half precision)
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) where the two inner dims are (i,k) and (j,k)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, i->row, k->col)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N (j), k)

    B, H, N, _ = left_perm.shape

    # Flatten batch & head dimensions for torch.bmm
    left_mat = left_perm.view(B * H, N, N)            # (i,k)
    right_mat = right_perm.view(B * H, N, N)          # (j,k)

    # Batched matrix multiply with transpose on the second operand
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (i,j)

    # Restore shape (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # elementwise (half)
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return in the requested precision (float32)
    return out.to(torch.float32)
