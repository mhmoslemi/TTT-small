@triton.jit
def _gate_fuse_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr,
    total_elements,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # compute mask index: one mask per H entries
        mask_idx = offs // H
        # mask_ptr length = total_elements / H = B*N*N
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. Input LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # Cast to half
    x_h = x.to(torch.float16)

    # 2. Linear projections
    left = torch.nn.functional.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = torch.nn.functional.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # 3. Gates
    left_gate = torch.nn.functional.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # 4. Fuse gating (and mask) via Triton
    total_elements = left.numel()
    # ensure contiguous
    left_f = left.contiguous().view(-1)
    right_f = right.contiguous().view(-1)
    left_gate_f = left_gate.contiguous().view(-1)
    right_gate_f = right_gate.contiguous().view(-1)

    # prepare mask
    has_mask = (not nomask) and (mask is not None)
    if has_mask:
        mask_f = mask.to(torch.float16).contiguous().view(-1) # shape B*N*N
    else:
        mask_f = left_f  # dummy

    BLOCK = 1024  # tune
    grid = (triton.cdiv(total_elements, BLOCK),)

    _gate_fuse_kernel[grid](
        left_f,
        right_f,
        left_gate_f,
        right_gate_f,
        mask_f,
        total_elements,
        BLOCK=BLOCK,
        HAS_MASK=1 if has_mask else 0,
        H=hidden_dim,
    )

    left = left_f.view_as(left)
    right = right_f.view_as(right)

    # 5. Core contraction using batched GEMM
    # left: [B, N, N, H]; right similar
    # Permute to [B, H, N, N] and compute A * B^T
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N) but we need to transpose last two dims later

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N).transpose(1, 2)  # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # 6. Output LayerNorm, out_gate, final linear
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16))

    # Cast to float32
    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass using Triton.

Algorithm:
1. Layer‑norm the input.
2. Linear projections (left/right) and gate linear layers in FP16.
3. Fuse gating (and optional mask) with a Triton kernel:
   left  = left  * left_gate  * mask
   right = right * right_gate * mask
   (mask is broadcast over the hidden dimension).
4. Compute the O(N³) contraction as a batched matrix multiply:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   i.e. out = left @ rightᵀ for each (b,d) block.
5. Output LayerNorm, out‑gate multiplication and final linear projection.
6. Return FP32 tensor.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_fuse_kernel(
    left_ptr,           # half *
    right_ptr,          # half *
    left_gate_ptr,      # half *
    right_gate_ptr,     # half *
    mask_ptr,           # half *
    total_elements,     # int32
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present, 0 otherwise
    H: tl.constexpr,          # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # one mask value per H consecutive hidden entries
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def custom_kernel(data):
    """
    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict: model parameters
            - config_dict: must contain "dim" and "hidden_dim",
                           may contain "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for heavy work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (FP16)
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 4. Gates (FP16)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 5. Fuse gating (and optional mask) via Triton
    # ------------------------------------------------------------------
    total = left.numel()               # B * N * N * hidden_dim
    left_f = left.contiguous().view(-1)
    right_f = right.contiguous().view(-1)
    left_gate_f = left_gate.contiguous().view(-1)
    right_gate_f = right_gate.contiguous().view(-1)

    has_mask = (not nomask) and (mask is not None)
    if has_mask:
        mask_f = mask.to(torch.float16).contiguous().view(-1)   # shape B*N*N
    else:
        # dummy pointer; will never be dereferenced when HAS_MASK=0
        mask_f = left_f

    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    _gate_fuse_kernel[grid](
        left_f,
        right_f,
        left_gate_f,
        right_gate_f,
        mask_f,
        total,
        BLOCK=BLOCK,
        HAS_MASK=1 if has_mask else 0,
        H=hidden_dim,
    )

    left = left_f.view_as(left)
    right = right_f.view_as(right)

    # ------------------------------------------------------------------
    # 6. Core O(N³) contraction via batched GEMM
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # shape -> (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)                       # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N).transpose(1, 2)    # (B*H, N, N) -> transpose for B^T

    out_mat = torch.bmm(left_mat, right_mat)                     # (B*H, N, N)

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final linear projection (FP16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 8. Cast back to FP32
    # ------------------------------------------------------------------
    return out.to(torch.float32)
