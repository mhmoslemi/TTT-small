def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # LayerNorm input
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    x = x.to(torch.float16)

    # Linear projections
    left = torch.nn.functional.linear(
        x, weights["left_proj.weight"].to(torch.float16)
    )
    right = torch.nn.functional.linear(
        x, weights["right_proj.weight"].to(torch.float16)
    )

    # Gating
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    if not nomask:
        # mask shape [B,N,N] -> broadcast to hidden_dim
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        # flatten for kernel:
        total_elements = left.numel()
        left_flat = left.view(-1)
        right_flat = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)
        mask_flat = mask_h.view(-1)  # size B*N*N

        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)
        has_mask = 1
        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total_elements,
            hidden_dim,
            HAS_MASK=has_mask,
            BLOCK=BLOCK,
        )
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # free gates:
    del left_gate, right_gate

    # Core contraction:
    # left: [B,N,N,hidden_dim] -> permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)  # shape (B*H,N,N)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output layernorm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    # Apply out_gate
    out = out * out_gate

    # Final linear projection to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16))

    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass.

Algorithm
---------
1. LayerNorm over the last dimension (dim) of the input.
2. Linear projections `left` and `right` from dim → hidden_dim (half‑precision).
3. Compute three gating tensors (sigmoid of a linear projection).
4. Fuse optional mask with the left/right gating multiplications in a single
   Triton kernel (in‑place).
5. Core contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix‑multiply per head (B*hidden_dim, N, N) using
   torch.bmm (cuBLAS) – the fastest primitive for this operation.
6. LayerNorm on the hidden dimension, element‑wise output‑gate, and final
   linear projection back to `dim`.
7. Return result in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # half*  (flattened left)
    right_ptr,           # half*  (flattened right)
    left_gate_ptr,       # half*  (flattened left_gate)
    right_gate_ptr,      # half*  (flattened right_gate)
    mask_ptr,            # half*  (flattened mask, broadcasted)
    total_elements: tl.int32,
    hidden_dim: tl.int32,
    HAS_MASK: tl.constexpr,   # 1 if mask present, 0 otherwise
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # Index of the mask entry (shared across hidden_dim values)
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion:
        left = left * left_gate * mask   (mask optional)
        right = right * right_gate * mask
    All tensors are contiguous half‑precision.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy ptr when no mask

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        hidden_dim,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )

            - config: dict with keys:
                "dim" (int) – input channel size C

                "nomask" (bool, optional) – True → ignore mask

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """

    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data

    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------

        input_tensor,

    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # 4. Fuse mask (if any) with the gating multiplications via Triton
    # ------------------------------------------------------------------

        right_flat = right.contiguous().view(-1)

        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_h.view(-1)  # [B * N * N]

        _apply_gate_and_mask(
            left_flat, right_flat,
            left_gate_flat, right_gate_flat,
            mask_flat,
            hidden_dim,
        )
        # Reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask: simple elementwise multiplication
        left = left * left_gate
        right = right * right_gate

    # Drop gate tensors we no longer need
    del left_gate, right_gate, x

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as batched GEMM (cuBLAS) on half‑precision tensors.
    # ------------------------------------------------------------------
    # Permute to (B, H, N, N) and prepare transposed right for bmm
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) -> already transposed

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Batched matrix multiplication (half precision, float32 accumulation)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate  # element‑wise gating
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 7. Return in float32 as required
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
