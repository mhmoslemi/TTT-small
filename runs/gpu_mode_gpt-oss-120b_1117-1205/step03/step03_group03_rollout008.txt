@triton.jit
def gate_mask_kernel(left_ptr, right_ptr, left_gate_ptr, right_gate_ptr,
                     mask_ptr, total_elements, hidden_dim,
                     HAS_MASK: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    # Cast weights to half
    norm_weight_h = weights["norm.weight"].to(device).to(torch.float16)
    norm_bias_h = weights["norm.bias"].to(device).to(torch.float16)
    left_proj_weight_h = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight_h = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight_h = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight_h = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight_h = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_weight_h = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias_h = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_weight_h = weights["to_out.weight"].to(device).to(torch.float16)

    # 1. First layer norm
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])
    x_h = x.to(torch.float16)   # [B,N,N,dim] half

    # 2. Linear projections
    left = F.linear(x_h, left_proj_weight_h)   # [B,N,N,hidden]
    right = F.linear(x_h, right_proj_weight_h)

    # 3. Gating
    left_gate = F.linear(x_h, left_gate_weight_h).sigmoid()
    right_gate = F.linear(x_h, right_gate_weight_h).sigmoid()
    out_gate = F.linear(x_h, out_gate_weight_h).sigmoid()

    # 4. Mask+gate fusion
    if not nomask:
        # mask shape [B,N,N]; unsqueeze to -1 dim for broadcast
        mask_h = mask.unsqueeze(-1).to(torch.float16)
        # flatten
        left_flat = left.view(-1)
        right_flat = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)
        mask_flat = mask_h.view(-1)   # size B*N*N
        # kernel launch
        total = left_flat.shape[0]
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)

        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=1,
            BLOCK=BLOCK,
        )
        # reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # without mask
        left = left * left_gate
        right = right * right_gate

    # If mask not used, left_gate and right_gate no longer needed.
    # We can delete them.
    del left_gate, right_gate, x_h

    # 5. Core contraction using batched matrix multiplication
    B, N, _, _ = left.shape  # left shape (B,N,N,hidden)
    # reshape for bmm
    left_mat = left.permute(0,3,1,2).reshape(-1, N, N)  # (B*hidden,N,N)
    right_mat = right.permute(0,3,2,1).reshape(-1, N, N) # (B*hidden,N,N)
    out_mat = torch.bmm(left_mat, right_mat)   # (B*hidden,N,N)
    # Restore shape
    out = out_mat.reshape(B, hidden_dim, N, N).permute(0,2,3,1)  # (B,N,N,hidden)

    # 6. Output layernorm (half)
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_weight_h,
                       bias=to_out_norm_bias_h)

    # 7. Apply out_gate
    out = out * out_gate

    # 8. Final projection
    out = F.linear(out, to_out_weight_h)   # [B,N,N,dim] half

    # 9. Cast to float32
    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass – optimized Triton‑fused mask & gating.

Algorithm
---------
1. Layer‑norm the input (float32) → half.
2. Linear projections `left` / `right` to `hidden_dim` (half‑precision).
3. Compute three gating tensors (`left_gate`, `right_gate`, `out_gate`) via
   linear + sigmoid (half‑precision).
4. Fuse optional mask with the two gating multiplications in a single Triton
   kernel that works in‑place on the flattened tensors.
5. Core O(N³) contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM (torch.bmm) on the
   reshaped tensors (B·hidden, N, N).  This uses cuBLAS and is
   close to the hardware peak.
6. Second Layer‑norm, element‑wise out‑gate and final linear projection.
7. Cast to float32 and return.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse optional mask with left/right gating
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # half*  (flattened left)
    right_ptr,           # half*  (flattened right)
    left_gate_ptr,       # half*  (flattened left_gate)
    right_gate_ptr,      # half*  (flattened right_gate)
    mask_ptr,            # half*  (flattened mask, broadcasted)
    total_elements: tl.int32,   # B·N·N·H
    hidden_dim: tl.int32,       # H
    HAS_MASK: tl.constexpr,     # 1 if mask present, 0 otherwise
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    left = tl.load(left_ptr + offs, mask=active, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=active, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=tl.float16(0.0))

    if HAS_MASK:
        # each mask entry corresponds to `hidden_dim` consecutive channel values
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) – may be ignored
                    when config["nomask"] is True.
            - weights: dict of model parameters (float32 tensors)
            - config: dict containing
                * "dim" (int) – input channel size C
                * "hidden_dim" (int) – projection size H
                * "nomask" (bool, optional) – True if mask should be ignored

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Cast model weights to half‑precision once
    # ------------------------------------------------------------------
    # LayerNorm (first) stays in float32 – we keep its weight/bias as float32
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)                     # [B, N, N, dim] half

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dim (half)
    # ------------------------------------------------------------------
    left = F.linear(x_h, left_proj_w)            # [B, N, N, hidden]
    right = F.linear(x_h, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Gating tensors (half)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, left_gate_w).sigmoid()
    right_gate = F.linear(x_h, right_gate_w).sigmoid()
    out_gate = F.linear(x_h, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) and gating with a Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → broadcast over hidden dim
        mask_h = mask.unsqueeze(-1).to(torch.float16)          # [B, N, N, 1]
        # flatten everything (contiguous views)
        left_flat = left.view(-1)
        right_flat = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)
        mask_flat = mask_h.view(-1)                           # [B·N·N]

        total = left_flat.shape[0]                            # B·N·N·H
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)

        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=1,
            BLOCK=BLOCK,
        )
        # reshape back to original layout
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # simple element‑wise gating when mask is absent
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched GEMM (cuBLAS) on reshaped tensors.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape                        # (B, N, N, hidden)
    # (B, hidden, N, N) → reshape to (B·hidden, N, N) without extra copy
    left_mat = left.permute(0, 3, 1, 2).reshape(-1, N, N)   # (B·H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(-1, N, N) # (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat)                # (B·H, N, N)

    # restore (B, N, N, hidden)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (half), out‑gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate                       # element‑wise gating (half)
    out = F.linear(out, to_out_w)              # back to [B, N, N, dim] (half)

    # ------------------------------------------------------------------
    # 7. Cast back to float32 as required by the specification
    # ------------------------------------------------------------------
    return out.to(torch.float32)
