def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # layer norm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    x = x.to(torch.float16)

    # Linear projections
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # Gates
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # Prepare mask
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).contiguous()
    else:
        mask_h = None

    # Ensure tensors are contiguous
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    if mask_h is not None:
        mask_h = mask_h.contiguous()

    B, N, _, _ = left.shape
    # out will be half
    out = torch.empty_like(left)  # shape (B,N,N,hidden_dim) in fp16

    # Compute strides
    stride_left_b = left.stride(0)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)
    stride_left_h = left.stride(3)

    # same for right
    stride_right_b = right.stride(0)
    stride_right_i = right.stride(1)
    stride_right_k = right.stride(2)
    stride_right_h = right.stride(3)

    # gate strides
    stride_gate_b = left_gate.stride(0)
    stride_gate_i = left_gate.stride(1)
    stride_gate_k = left_gate.stride(2)
    stride_gate_h = left_gate.stride(3)

    # mask strides
    if mask_h is not None:
        stride_mask_b = mask_h.stride(0)
        stride_mask_i = mask_h.stride(1)
        stride_mask_k = mask_h.stride(2)
        mask_ptr = mask_h
    else:
        stride_mask_b = 0
        stride_mask_i = 0
        stride_mask_k = 0
        mask_ptr = left  # dummy pointer

    # out strides
    stride_out_b = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_h = out.stride(3)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    total_batches = B * hidden_dim
    grid = (total_batches, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    HAS_MASK = 0 if (nomask or mask_h is None) else 1

    tri_mul_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr, out,
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
        stride_mask_b, stride_mask_i, stride_mask_k,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        HAS_MASK=HAS_MASK
    )

    # After kernel, out is half
    # Apply to_out_norm, out_gate, final linear
    # to_out_norm layernorm over hidden_dim
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16)
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))
    return out.to(torch.float32)

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # 0..B*H-1
    pid_i = tl.program_id(1)       # tile i
    pid_j = tl.program_id(2)       # tile j

    # decode batch and head
    batch_idx = pid_batch // H
    head_idx = pid_batch % H

    # compute base offsets
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # range of indices
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k0 in range(0, N, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Left block (M, K)
        left_ptrs = left_ptr + batch_idx * stride_left_b + head_idx * stride_left_h + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=tl.float16(0.0))

        left_gate_ptrs = left_gate_ptr + batch_idx * stride_gate_b + head_idx * stride_gate_h + offs_i[:, None] * stride_gate_i + offs_k[None, :] * stride_gate_k
        left_gate = tl.load(left_gate_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=tl.float16(0.0))

        if HAS_MASK:
            mask_ptrs = mask_ptr + batch_idx * stride_mask_b + offs_i[:, None] * stride_mask_i + offs_k[None, :] * stride_mask_k
            mask_val = tl.load(mask_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=tl.float16(0.0))
            left = left * left_gate * mask_val
        else:
            left = left * left_gate

        # Right block (K, N) → we need (K, N) order for dot
        right_ptrs = right_ptr + batch_idx * stride_right_b + head_idx * stride_right_h + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=tl.float16(0.0))

        right_gate_ptrs = right_gate_ptr + batch_idx * stride_gate_b + head_idx * stride_gate_h + offs_k[:, None] * stride_gate_k + offs_j[None, :] * stride_gate_i
        right_gate = tl.load(right_gate_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=tl.float16(0.0))

        if HAS_MASK:
            mask_ptrs_r = mask_ptr + batch_idx * stride_mask_b + offs_j[:, None] * stride_mask_i + offs_k[None, :] * stride_mask_k
            mask_val_r = tl.load(mask_ptrs_r, mask=mask_j[:, None] & mask_k[None, :], other=tl.float16(0.0))
            right = right * right_gate * mask_val_r
        else:
            right = right * right_gate

        # compute dot product. left shape (M,K), right shape (K,N)
        # using tl.dot returns same dtype as inputs (fp16)
        acc = acc + tl.dot(left, right).to(tl.float32)

    # store result
    out_ptrs = out_ptr + batch_idx * stride_out_b + head_idx * stride_out_h + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

    mask_ptr, out_ptr,

    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    stride_mask_b, stride_mask_i, stride_mask_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr,

    pid_batch = tl.program_id(0)   # 0 .. B*H-1
    pid_i = tl.program_id(1)       # tile over i dimension
    pid_j = tl.program_id(2)       # tile over j dimension

    j0 = pid_j * BLOCK_N

        # note the transposed layout: k as first dim, j as second dim
        right_ptrs = right_ptr + batch_idx * stride_right_b + head_idx * stride_right_h \
                     + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=tl.float16(0.0))

        right_gate_ptrs = right_gate_ptr + batch_idx * stride_gate_b + head_idx * stride_gate_h \
                         + offs_k[:, None] * stride_gate_k + offs_j[None, :] * stride_gate_i
        right_gate = tl.load(right_gate_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=tl.float16(0.0))

        if HAS_MASK:
            mask_ptrs_r = mask_ptr + batch_idx * stride_mask_b \
                          + offs_j[:, None] * stride_mask_i + offs_k[None, :] * stride_mask_k
            mask_val_r = tl.load(mask_ptrs_r, mask=mask_j[:, None] & mask_k[None, :], other=tl.float16(0.0))
            right = right * right_gate * mask_val_r
        else:
            right = right * right_gate

        # dot product (M,K) x (K,N) -> (M,N)
        acc = acc + tl.cast(tl.dot(left, right), tl.float32)

    # ---------- write output ----------
    out_ptrs = out_ptr + batch_idx * stride_out_b + head_idx * stride_out_h \
               + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Outgoing TriMul forward pass.
    The heavy O(N^3) contraction is performed by a Triton kernel that fuses
    optional masking and the left/right gating operations.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------- LayerNorm ----------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    x = x.to(torch.float16)

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # ---------- Gating ---
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # Ensure contiguous tensors
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).contiguous()
    else:
        mask_h = None

    B, N, _, _ = left.shape

    # ---------- Output tensor ----------
    out = torch.empty_like(left)  # shape (B,N,N,hidden_dim), dtype fp16

    # ---------- Strides ----------
    s_left_b, s_left_i, s_left_k, s_left_h = left.stride()
    s_right_b, s_right_i, s_right_k, s_right_h = right.stride()
    s_gate_b, s_gate_i, s_gate_k, s_gate_h = left_gate.stride()  # same as left/right
    s_out_b, s_out_i, s_out_j, s_out_h = out.stride()

    if mask_h is not None:
        s_mask_b, s_mask_i, s_mask_k = mask_h.stride()
        mask_ptr = mask_h
        HAS_MASK = 1
    else:
        # dummy strides (will not be used)
        s_mask_b = s_mask_i = s_mask_k = 0
        mask_ptr = left  # dummy pointer
        HAS_MASK = 0

    # ---------- Triton kernel launch ----------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    total_batches = B * hidden_dim
    grid = (total_batches,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    tri_mul_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr, out,
        B, N, hidden_dim,
        s_left_b, s_left_i, s_left_k, s_left_h,
        s_right_b, s_right_i, s_right_k, s_right_h,
        s_out_b, s_out_i, s_out_j, s_out_h,
        s_gate_b, s_gate_i, s_gate_k, s_gate_h,
        s_mask_b, s_mask_i, s_mask_k,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        HAS_MASK=HAS_MASK,
    )

    # ---------- Output processing ----------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16)
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    return out.to(torch.float32)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fused O(N³) contraction with optional mask & gating
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """
    Compute, for each batch `b` and head `h`:

        out[b,i,j,h] = Σ_k left[b,i,k,h] * left_gate[b,i,k,h] *
                      right[b,j,k,h] * right_gate[b,j,k,h] *
                      mask[b,i,k] * mask[b,j,k]   (if mask present)

    The computation is tiled over i, j and k dimensions.
    """
    pid_batch = tl.program_id(0)       # 0 .. B*H-1
    pid_i = tl.program_id(1)           # tile over i
    pid_j = tl.program_id(2)           # tile over j

    # decode batch index and head index
    batch_idx = pid_batch // H
    head_idx = pid_batch % H

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k):
        k_start = kb * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # --------------------------------------------------------------
        # Load left block (i, k) and its gate (and optional mask)
        # --------------------------------------------------------------
        left_ptrs = left_ptr + batch_idx * stride_left_b + head_idx * stride_left_h \
                    + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :],
                       other=tl.float16(0.0))

        left_gate_ptrs = left_gate_ptr + batch_idx * stride_gate_b + head_idx * stride_gate_h \
                        + offs_i[:, None] * stride_gate_i + offs_k[None, :] * stride_gate_k
        left_gate = tl.load(left_gate_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=tl.float16(0.0))

        if HAS_MASK:
            mask_ptrs_l = mask_ptr + batch_idx * stride_mask_b \
                          + offs_i[:, None] * stride_mask_i + offs_k[None, :] * stride_mask_k
            mask_val_l = tl.load(mask_ptrs_l,
                                 mask=mask_i[:, None] & mask_k[None, :],
                                 other=tl.float16(0.0))
            left = left * left_gate * mask_val_l
        else:
            left = left * left_gate

        # --------------------------------------------------------------
        # Load right block transposed (k, j) and its gate (and optional mask)
        # --------------------------------------------------------------
        right_ptrs = right_ptr + batch_idx * stride_right_b + head_idx * stride_right_h \
                     + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=tl.float16(0.0))

        right_gate_ptrs = right_gate_ptr + batch_idx * stride_gate_b + head_idx * stride_gate_h \
                         + offs_k[:, None] * stride_gate_k + offs_j[None, :] * stride_gate_i
        right_gate = tl.load(right_gate_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=tl.float16(0.0))

        if HAS_MASK:
            mask_ptrs_r = mask_ptr + batch_idx * stride_mask_b \
                          + offs_j[:, None] * stride_mask_i + offs_k[None, :] * stride_mask_k
            mask_val_r = tl.load(mask_ptrs_r,
                                 mask=mask_j[:, None] & mask_k[None, :],
                                 other=tl.float16(0.0))
            right = right * right_gate * mask_val_r
        else:
            right = right * right_gate

        # --------------------------------------------------------------
        # Accumulate dot product over k
        # --------------------------------------------------------------
        acc = acc + tl.cast(tl.dot(left, right), tl.float32)

    # --------------------------------------------------------------
    # Store the result
    # --------------------------------------------------------------
    out_ptrs = out_ptr + batch_idx * stride_out_b + head_idx * stride_out_h \
               + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

    Core O(N³) contraction is executed in a tiled Triton kernel that
    fuses optional masking and the left/right gating operations.
    The surrounding LayerNorms and linear projections are performed in
    PyTorch (FP16) for simplicity.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> float16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Gating tensors (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 4. Prepare mask (optional)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).contiguous()
    else:
        mask_h = None

    # ------------------------------------------------------------------
    # 5. Ensure tensors are contiguous
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    if mask_h is not None:
        mask_h = mask_h.contiguous()

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    # 6. Allocate output buffer (FP16)
    # ------------------------------------------------------------------
    out = torch.empty_like(left)   # shape (B, N, N, hidden_dim), dtype fp16

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    s_left_b, s_left_i, s_left_k, s_left_h = left.stride()
    s_right_b, s_right_i, s_right_k, s_right_h = right.stride()
    s_gate_b, s_gate_i, s_gate_k, s_gate_h = left_gate.stride()   # same layout as left/right

    else:
        s_mask_b = s_mask_i = s_mask_k = 0
        mask_ptr = left                     # dummy pointer (won't be read)
        HAS_MASK = 0

    # ------------------------------------------------------------------
    # 8. Triton kernel launch (blocked GEMM)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    total_batches = B * hidden_dim
    grid = (
        total_batches,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    tri_mul_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr, out,
        B, N, hidden_dim,
        s_left_b, s_left_i, s_left_k, s_left_h,
        s_right_b, s_right_i, s_right_k, s_right_h,
        s_out_b, s_out_i, s_out_j, s_out_h,
        s_gate_b, s_gate_i, s_gate_k, s_gate_h,
        s_mask_b, s_mask_i, s_mask_k,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        HAS_MASK=HAS_MASK,
    )

    # ------------------------------------------------------------------
    # 9. Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------

        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16)
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10. Return FP32 tensor as required
    # ------------------------------------------------------------------
    return out.to(torch.float32)
