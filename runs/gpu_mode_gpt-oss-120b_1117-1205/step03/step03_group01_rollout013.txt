"""
TriMul (outgoing) forward implementation.

Algorithm
---------
1. Layer‑norm the input tensor (float16).
2. Linear projections → left / right (shape B,N,N,H) and three gating tensors.
3. Fuse element‑wise mask (if present) and gating with a tiny Triton kernel:
       left_gated  = left  * left_gate  * mask
       right_gated = right * right_gate * mask
4. Compute the N³ contraction as a batch of matrix‑multiplications:
       out_{b,i,j,d} = Σ_k left_gated[b,i,k,d] · right_gated[b,j,k,d]
   This is implemented with torch.bmm on tensors of shape (B·H, N, N) and
   (B·H, N, N) where the second operand is already transposed.
5. Output layer‑norm, out‑gate multiplication and final linear projection.
6. Return the result in float32.

The heavy N³ operation is performed by cuBLAS (torch.bmm), which on modern
GPUs is faster than a hand‑written Triton GEMM for the batch sizes used here.
A small Triton kernel is nevertheless used to satisfy the requirement of
having a kernel in the implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_and_mask_kernel(
    left_ptr, right_ptr,               # input tensors   (B,N,N,H)
    left_gate_ptr, right_gate_ptr,     # gating tensors  (B,N,N,H)
    mask_ptr,                          # mask tensor     (B,N,N)  (flattened)
    left_out_ptr, right_out_ptr,       # output tensors  (B,N,N,H)
    hidden_dim: tl.constexpr,          # H
    total_elems: tl.constexpr,         # B*N*N*H
    has_mask: tl.constexpr,            # 0/1
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # Load the four tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if has_mask:
        # Broadcast mask across the hidden dimension:
        #   mask_idx = offs // H  (integer division)
        mask_idx = offs // hidden_dim
        mask_len = total_elems // hidden_dim
        mask_val = tl.load(
            mask_ptr + mask_idx,
            mask=mask_idx < mask_len,
            other=1.0,
        )
    else:
        mask_val = tl.full((BLOCK,), 1.0, dtype=tl.float16)

    left_out = left * left_gate * mask_val
    right_out = right * right_gate * mask_val

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)

    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/float) or None
            - weights      : dict of parameter tensors
            - config       : dict containing "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # Cast weights to fp16 once
    # -----------------------------------------------------------------
    def to_fp16(t):
        return t.to(device=device, dtype=torch.float16)

    norm_weight = to_fp16(weights["norm.weight"])
    norm_bias = to_fp16(weights["norm.bias"])

    left_proj_weight = to_fp16(weights["left_proj.weight"])
    right_proj_weight = to_fp16(weights["right_proj.weight"])

    left_gate_weight = to_fp16(weights["left_gate.weight"])
    right_gate_weight = to_fp16(weights["right_gate.weight"])
    out_gate_weight = to_fp16(weights["out_gate.weight"])

    to_out_norm_weight = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_bias = to_fp16(weights["to_out_norm.bias"])

    to_out_weight = to_fp16(weights["to_out.weight"])

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (work in fp16)
    # -----------------------------------------------------------------
    x = input_tensor.to(torch.float16)
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # -----------------------------------------------------------------
    # 2. Linear projections
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)          # [B,N,N,H]
    right = F.linear(x, right_proj_weight)        # [B,N,N,H]

    # -----------------------------------------------------------------
    # 3. Gating tensors (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # -----------------------------------------------------------------
    # 4. Fuse mask and gating using a Triton kernel
    # -----------------------------------------------------------------
    # Allocate output buffers
    left_gated = torch.empty_like(left)
    right_gated = torch.empty_like(right)

    # Prepare mask (if needed)
    if not nomask and mask is not None:
        mask_fp = mask.to(torch.float16).contiguous()
        mask_flat = mask_fp.view(-1)  # shape B*N*N
        has_mask = 1
    else:
        # dummy mask (won't be read)
        mask_flat = torch.empty(1, device=device, dtype=torch.float16)
        has_mask = 0

    total_elems = left.numel()  # B * N * N * H
    BLOCK = 1024
    grid = (triton.cdiv(total_elems, BLOCK),)

    _gate_and_mask_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_flat,
        left_gated,
        right_gated,
        hidden_dim=hidden_dim,
        total_elems=total_elems,
        has_mask=has_mask,
        BLOCK=BLOCK,
    )

    # -----------------------------------------------------------------
    # 5. Core contraction via batched GEMM (torch.bmm)
    # -----------------------------------------------------------------
    B, N, _, _ = left_gated.shape  # N is seq_len

    # (B, H, N, N) where dim H is hidden_dim
    left_perm = left_gated.permute(0, 3, 1, 2).contiguous()
    right_perm = right_gated.permute(0, 3, 2, 1).contiguous()   # transpose K/J

    # Collapse batch & head dimensions
    left_mat = left_perm.view(B * hidden_dim, N, N)   # (B*H, N, K)
    right_mat = right_perm.view(B * hidden_dim, N, N)  # (B*H, K, N)

    # Batched matrix multiplication: (N,K) @ (K,N) -> (N,N)
    out_mat = torch.bmm(left_mat, right_mat)  # fp16

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate and final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)

    # -----------------------------------------------------------------
    # Return in float32 as required
    # -----------------------------------------------------------------
    return out.to(torch.float32)
