@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr,
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

Note: mask_ptr must be contiguous.

"""
TriMul (outgoing) forward pass – highly‑optimized Triton version.

Algorithm
---------
1. LayerNorm on the input (float32 for numeric stability).
2. Cast to FP16 for all heavy work.
3. Linear projections (left/right) and gate projections (left_gate,
   right_gate, out_gate) are computed with torch.nn.functional.linear.
4. A small Triton kernel fuses the element‑wise mask (if present) and the
   gate multiplications:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
   The kernel works on the flattened tensors and broadcasts the 3‑D mask
   across the hidden dimension.
5. The core contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is performed with a single torch.einsum, which under the hood is
   translated into a batched GEMM without any extra transposes or copies.
6. Output LayerNorm, out‑gate multiplication and the final linear projection
   are applied in FP16 and the result is cast back to float32.

Only the gate‑mask fusion is required to be a Triton kernel; the rest of the
operations stay in PyTorch for simplicity and to leverage cuBLAS/Torch
optimizations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask (if any) and gate multiplications in‑place
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # float16 *
    right_ptr,           # float16 *
    left_gate_ptr,       # float16 *
    right_gate_ptr,      # float16 *
    mask_ptr,            # float16 * (broadcasted mask) – dummy if HAS_MASK==0
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 → mask present, 0 → no mask
    H: tl.constexpr,          # hidden dimension (size of innermost channel)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    # load operands
    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    if HAS_MASK:
        # each consecutive H elements share the same mask entry
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # store results back in‑place
    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place gate & mask fusion via Triton.
    All tensors must be contiguous FP16.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when not used

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # No return – operation is in‑place


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (float32/bool) – may be ignored
            - weights_dict: mapping of parameter names to torch.Tensor (float32)
            - config_dict: must contain "dim" and "hidden_dim", may contain "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for projection / gating
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (FP16)
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 4. Gate projections (FP16)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 5. Fuse mask (if any) and gate multiplications via Triton
    # ------------------------------------------------------------------
    # Ensure all tensors are contiguous for the kernel
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).contiguous()
        # Flatten tensors for the kernel
        left_f = left.view(-1)
        right_f = right.view(-1)
        left_gate_f = left_gate.view(-1)
        right_gate_f = right_gate.view(-1)
        mask_f = mask_h.view(-1)  # [B*N*N]

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            hidden_dim,
        )
        # Reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask – still fuse via the same kernel with HAS_MASK=0 for consistency
        _apply_gate_and_mask(
            left.view(-1),
            right.view(-1),
            left_gate.view(-1),
            right_gate.view(-1),
            None,
            hidden_dim,
        )

    # Gates are no longer needed
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6. Core N³ contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented with torch.einsum which maps to a batched GEMM without extra copies.
    # ------------------------------------------------------------------
    out = torch.einsum("bikd,bjkd->bijd", left, right)  # [B, N, N, H] FP16
    del left, right  # free memory

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final projection (all FP16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 8. Cast back to float32 for the API
    # ------------------------------------------------------------------
    return out.to(torch.float32)
