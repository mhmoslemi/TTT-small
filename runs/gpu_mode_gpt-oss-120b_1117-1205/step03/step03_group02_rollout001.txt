"""
TriMul (outgoing) forward pass – fused Triton kernel for mask & gate.
---------------------------------------------------------------
Algorithm
1. Layer‑norm the input (float32 → half).
2. Linear projections (half) → left/right tensors of shape [B, N, N, H].
3. Compute gate tensors (half, sigmoid).  If a mask is present,
   fuse mask, left‑gate and right‑gate into a single Triton kernel
   that updates left/right in‑place.
4. Core O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply (left * rightᵀ) over
   (B·H) matrices of size N×N, performed with torch.bmm
   (cuBLAS strided‑batched GEMM).
5. Output LayerNorm (half), element‑wise multiplication by out‑gate,
   final linear projection back to dim.
6. Return float32 tensor.

At least one operation (mask‑gate fusion) runs on Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask (optional) with left/right gates
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # [total] half
    right_ptr,         # [total] half
    left_gate_ptr,     # [total] half
    right_gate_ptr,    # [total] half
    mask_ptr,          # [mask_total] half (broadcast)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present else 0
    H: tl.constexpr,          # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask_oob = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask_oob, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask_oob, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask_oob, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask_oob, other=0.0)

    if HAS_MASK:
        # One mask entry corresponds to H consecutive hidden values
        mask_offs = offs // H
        mask_val = tl.load(mask_ptr + mask_offs, mask=mask_oob, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store back in‑place
    tl.store(left_ptr + offs, left, mask=mask_oob)
    tl.store(right_ptr + offs, right, mask=mask_oob)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion: left = left * left_gate * mask,
                    right = right * right_gate * mask   (if mask provided)
    All tensors must be contiguous torch.float16.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when HAS_MASK==0

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # In‑place – nothing to return


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool, optional)

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32 → half)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections to hidden dimension
    # --------------------------------------------------------------
    left = F.linear(
        x, weights["left_proj.weight"].to(torch.float16)
    )  # [B, N, N, H]
    right = F.linear(
        x, weights["right_proj.weight"].to(torch.float16)
    )  # [B, N, N, H]

    # --------------------------------------------------------------
    # 3. Compute gates (sigmoid)
    # --------------------------------------------------------------
    left_gate = F.linear(
        x, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()  # [B, N, N, H]
    right_gate = F.linear(
        x, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = F.linear(
        x, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    # --------------------------------------------------------------
    # 4. Fuse mask (if any) with gates via Triton kernel
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → broadcast over hidden dimension
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        # Flatten tensors for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        lg_f = left_gate.contiguous().view(-1)
        rg_f = right_gate.contiguous().view(-1)
        mask_f = mask_h.view(-1)  # length B*N*N
        _apply_gate_and_mask(left_f, right_f, lg_f, rg_f, mask_f, hidden_dim)
        # Reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask – simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    = left * rightᵀ for each (b, h)
    # --------------------------------------------------------------
    # Rearrange to (B, H, N, N) where the last two dims are the matrices.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) = rightᵀ
    B, H, N, _ = left_perm.shape

    # Merge batch and head dimensions for batched GEMM
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Batched GEMM (half precision, float32 accumulation)
    out_mat = torch.bmm(left_mat, right_mat)          # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise gating (half)
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # Return float32 as required
    # --------------------------------------------------------------
    return out.to(torch.float32)
