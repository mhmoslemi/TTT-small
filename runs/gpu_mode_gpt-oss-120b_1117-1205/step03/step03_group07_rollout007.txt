import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask)
    right = tl.load(right_ptr + offs, mask=mask)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask)

    if HAS_MASK:
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

"""
TriMul (outgoing) forward pass.

- LayerNorm on input (float32)
- Linear projections + gates in float16
- A tiny Triton kernel fuses the optional mask with the gate multiplications
  (left = left * left_gate * mask, right = right * right_gate * mask)
- The N³ contraction is performed by a single einsum, which internally uses
  cuBLAS batched GEMM on half‑precision tensors (fast, highly optimized).
- Output LayerNorm, out‑gate and final linear projection (all half‑precision);
  result is cast back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # float16 *
    right_ptr,         # float16 *
    left_gate_ptr,     # float16 *
    right_gate_ptr,    # float16 *
    mask_ptr,          # float16 *
    total_elements,    # int32
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present, 0 otherwise
    H: tl.constexpr,          # hidden dimension (size of innermost channel)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    left = tl.load(left_ptr + offs, mask=valid)
    right = tl.load(right_ptr + offs, mask=valid)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid)

    if HAS_MASK:
        # each mask element corresponds to H consecutive channel entries
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict: mapping of parameter names to torch.Tensor
            - config_dict: must contain "dim" and "hidden_dim",
                           may contain "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to half‑precision for the heavy work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (half)
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 4. Gates (half)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 5. Fuse mask and gating via Triton (if a mask is present)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).contiguous()
        total = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)
        _gate_mask_fuse_kernel[grid](
            left,
            right,
            left_gate,
            right_gate,
            mask_h,
            total,
            BLOCK=BLOCK,
            HAS_MASK=1,
            H=hidden_dim,
        )
    else:
        left = left * left_gate
        right = right * right_gate

    # gates are no longer needed
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6. Core N³ contraction (batched GEMM) via einsum
    # ------------------------------------------------------------------
    # out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final linear projection (half)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 8. Cast back to float32 as required by the API
    # ------------------------------------------------------------------
    return out.to(torch.float32)
