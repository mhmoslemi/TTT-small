    b = tl.load(b_ptr + offs, mask=mask)

"""
TriMul (outgoing) forward pass – Triton‑accelerated elementwise gate.
The heavy O(N³) contraction is performed with a batched GEMM (torch.bmm);
layer‑norms and linear layers are executed in FP16.
A tiny Triton kernel fuses the final elementwise multiplication
(out_norm * out_gate) to satisfy the “must‑use‑a‑kernel” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _elem_mul_kernel(
    a_ptr,          # float16 *
    b_ptr,          # float16 *
    out_ptr,        # float16 *
    n_elements: tl.int32,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < n_elements

    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict: mapping of parameter names to torch.Tensor
            - config_dict: must contain "dim" and "hidden_dim",
                           may contain "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to half precision for the heavy work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Fused linear projections + gates (single matmul)
    # ------------------------------------------------------------------
    w_left_proj = weights["left_proj.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]

    # Combine into (5*hidden_dim, dim) and move to half
    w_combined = torch.cat(
        [w_left_proj, w_right_proj, w_left_gate, w_right_gate, w_out_gate],
        dim=0,
    ).to(device=device, dtype=torch.float16)

    # (B, N, N, dim) -> (B, N, N, 5*hidden_dim)
    combined = F.linear(x_h, w_combined)  # FP16 output

    # Split back into the five components
    left_proj, right_proj, left_gate_pre, right_gate_pre, out_gate_pre = combined.split(
        hidden_dim, dim=-1
    )

    # Gates (sigmoid)
    left_gate = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()
    out_gate = out_gate_pre.sigmoid()

    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Optional mask (broadcast over hidden dim)
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4. Core N³ contraction via batched GEMM:
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) and transpose right for the GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # Batched half‑precision GEMM (float32 accumulation)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)

    # Restore (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Output LayerNorm (half)
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # ------------------------------------------------------------------
    # 6. Elementwise multiply with out_gate via Triton
    # ------------------------------------------------------------------
    numel = out_norm.numel()
    out_scaled = torch.empty_like(out_norm)  # FP16 output buffer

    BLOCK = 1024
    grid = (triton.cdiv(numel, BLOCK),)
    _elem_mul_kernel[grid](
        out_norm.data_ptr(),
        out_gate.data_ptr(),
        out_scaled.data_ptr(),
        numel,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 7. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out_final = F.linear(
        out_scaled,
        weights["to_out.weight"].to(torch.float16),
    )

    # Return in float32 as required by the API
    return out_final.to(torch.float32)
