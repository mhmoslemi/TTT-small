def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', True)

    # Cast to fp16
    # LayerNorm
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=weights['norm.weight'].to(torch.float16),
                     bias=weights['norm.bias'].to(torch.float16))

    x = x.to(torch.float16)

    # Linear projections
    left = F.linear(x, weights['left_proj.weight'].to(torch.float16))
    right = F.linear(x, weights['right_proj.weight'].to(torch.float16))

    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights['left_gate.weight'].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x, weights['right_gate.weight'].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x, weights['out_gate.weight'].to(torch.float16)))

    # Apply gates and mask
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * left_gate * mask_h
        right = right * right_gate * mask_h
    else:
        left = left * left_gate
        right = right * right_gate

    # Ensure contiguous tensors
    left = left.contiguous()
    right = right.contiguous()

    # Allocate output
    out = torch.empty_like(left)  # same shape, fp16

    # Call Triton kernel for contraction
    _tri_mul_kernel_launcher(left, right, out, hidden_dim)

    # Output layernorm
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=weights['to_out_norm.weight'].to(torch.float16),
                       bias=weights['to_out_norm.bias'].to(torch.float16))

    # Multiply by out_gate (same shape as out) (out_gate computed earlier with same shape as left)
    out = out * out_gate

    # Final linear
    out = F.linear(out, weights['to_out.weight'].to(torch.float16))

    # Return float32
    return out.to(torch.float32)

"""
Highly‑optimized outgoing TriMul forward pass.

The core N³ contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed by a custom Triton batched‑GEMM kernel that works directly on the
original 4‑D layout (no costly permutes or copies).  All other operations are
executed in fp16 with Torch fused kernels; the final result is returned in fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton kernel for the triple‑multiplication (outgoing) step
# -------------------------------------------------------------
@triton.autotune(
    configs=[
        triton.Config(dict(BLOCK_M=64,  BLOCK_N=64,  BLOCK_K=32), num_warps=4),
        triton.Config(dict(BLOCK_M=128, BLOCK_N=128, BLOCK_K=32), num_warps=8),
        triton.Config(dict(BLOCK_M=128, BLOCK_N=64,  BLOCK_K=32), num_warps=4),
        triton.Config(dict(BLOCK_M=64,  BLOCK_N=128, BLOCK_K=32), num_warps=4),
    ],
    key=["M", "N", "K"],
)
@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, M, N, K, D,                     # B=batch, M=N_i, N=N_j, K=reduction, D=hidden_dim
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_i, stride_r_k, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]"""
    pid_m = tl.program_id(0)          # i‑tile
    pid_n = tl.program_id(1)          # j‑tile
    pid_h = tl.program_id(2)          # combined batch‑hidden index

    # split pid_h into batch and hidden‑dim indices
    hidden = D
    b = pid_h // hidden
    d = pid_h % hidden

    # offsets for the i and j dimensions
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over k
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = kk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile: (BLOCK_M, BLOCK_K)
        left_ptrs = (
            left_ptr
            + b * stride_l_b
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k
            + d * stride_l_d
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right tile: (BLOCK_K, BLOCK_N)   (note j‑dimension uses stride_r_i)
        right_ptrs = (
            right_ptr
            + b * stride_r_b
            + offs_j[:, None] * stride_r_i
            + offs_k[None, :] * stride_r_k
            + d * stride_r_d
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        acc += tl.dot(left, right)

    # write the result tile
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
        + d * stride_o_d
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def _tri_mul_launcher(left: torch.Tensor, right: torch.Tensor, out: torch.Tensor, hidden_dim: int):
    """
    Launches the Triton kernel.  Tensors must be fp16 and contiguous.
    left/right/out shape: [B, N, N, hidden_dim]
    """
    B, N, N2, D = left.shape
    assert N == N2 and D == hidden_dim

    # strides (in elements) for each tensor
    slb, sli, slk, sld = left.stride()
    srb, sri, srk, srd = right.stride()
    sob, soi, soj, sod = out.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )
    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        N,
        N,
        hidden_dim,
        slb,
        sli,
        slk,
        sld,
        srb,
        sri,
        srk,
        srd,
        sob,
        soi,
        soj,
        sod,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )


# -------------------------------------------------------------
# Entrypoint
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of parameter tensors
            - config: dict with "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 → fp16)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(torch.float16),
        bias=weights["norm.bias"].to(torch.float16),
    )
    x = x.to(torch.float16)                     # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2. Linear projections (fp16)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # -----------------------------------------------------------------
    # 3. Gating (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(torch.float16)))

    # -----------------------------------------------------------------
    # 4. Apply gates and optional mask (still fp16)
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → broadcast over hidden_dim
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * left_gate * mask_h
        right = right * right_gate * mask_h
    else:
        left = left * left_gate
        right = right * right_gate

    # Ensure contiguity for the kernel
    left = left.contiguous()
    right = right.contiguous()

    # -----------------------------------------------------------------
    # 5. Core contraction via Triton (out = Σ_k left * right)
    # -----------------------------------------------------------------
    out = torch.empty_like(left)   # shape [B, N, N, hidden_dim], fp16
    _tri_mul_launcher(left, right, out, hidden_dim)

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate and final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # -----------------------------------------------------------------
    # Return in float32 as required by the spec
    # -----------------------------------------------------------------
    return out.to(torch.float32)
