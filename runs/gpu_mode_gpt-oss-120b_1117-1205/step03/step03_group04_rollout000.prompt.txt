import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

## Triton launch convention

Every failure mode below has appeared in this search. Read this before writing.

A `@triton.jit` function is NOT a Python function. It is launched with a grid
subscript, and its `tl.constexpr` parameters exist only inside it:

```python
@triton.jit
def my_kernel(x_ptr, y_ptr, n, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < n
    tl.store(y_ptr + offs, tl.load(x_ptr + offs, mask=m, other=0.0), mask=m)

def custom_kernel(data):
    ...
    BLOCK = 128                                   # a host variable YOU define
    grid = (triton.cdiv(n, BLOCK),)               # host-side, use triton.cdiv
    my_kernel[grid](x, y, n, BLOCK=BLOCK)         # note the [grid] subscript
```

- `my_kernel(...)` without `[grid]` raises "Cannot call @triton.jit'd outside
  the scope of a kernel".
- `BLOCK`, `BLOCK_M`, `BLOCK_N` are not defined on the host unless you define
  them there. Referencing one you only declared as a `tl.constexpr` parameter
  raises NameError.
- `tl.cdiv` is device-side. On the host use `triton.cdiv`.
- Define the kernel at module top level, never nested inside `custom_kernel`.

You do not have to move everything into Triton. A correct submission that keeps
most of the computation in PyTorch and moves one hot region into a kernel scores;
one that fails to launch scores zero.

        right_flat = right.view(-1)

Approach and rationale:
We will implement the TriMul operator using a single Triton kernel for the core contraction operation to reduce host-device overhead. The operator requires applying layer normalization, three linear projections (left_proj, right_proj, and gates), optional masking of the activations, contraction (summing over k), and final output projection. The main bottleneck is the N^3 contraction, which must be avoided by tiling. We use a dimlet-based approach to split the operation into distinct stages in PyTorch, then use a persistent kernel for the expensive contraction part.

1. **Input Normalization**: Use LayerNorm with float16 precision (if input is float32).
2. **Linear Projection**: Project input to hidden_dim using two linear layers.
3. **Gating and Masking**: Compute gate activations (sigmoid) and apply element-wise masking.
4. **Contraction**: Use a kernel with persistent format to compute the contraction efficiently.
5. **Final Projection**: Apply the output gate and project back to output dim.

Mathematical specification:
- Input: `x` of shape `[B, N, N, C]`, optional mask `m` of shape `[B, N, N]`
- Normalization: `x_norm = F.layer_norm(x, [dim], γ, β)`
- Projection: `left = linear(x_norm, left_proj_weight)`
- Gate: `left_gate = linear(x_norm, left_gate_weight).sigmoid()`
- Scaled activation: `left_scaled = left * left_gate` (if mask present: `left_scaled = left_scaled * mask.half()`)
- Contraction: `out_{b,i,j,d} = sum_{k} left_scaled_{b,i,k,d} * right_scaled_{b,i,k,d}`
- Output: `out_proj = linear(out, to_out_weight)`
- Final: `out_final = out_proj * output_gate`

We use the following in Triton kernel:
- `BlockSize` to divide the N dimension.
- Transposed layout for the right tensor to enable persistent operation.
- Shared memory for intermediate results to amortize the cost.

   - Convert input to float16.

   - Compute linear projections.
   - Compute gate activations.

def custom_kernel(data):
    input, mask, weights, config = data
    bs, seqlen, _, dim = input.shape
    hidden_dim = config.get("hidden_dim", 128)

    # Convert to float16 for efficiency
    input_float16 = input.to(dtype=torch.float16, device="cuda")
    
    # Apply LayerNorm
    normalized = F.layer_norm(input_float16, [dim], 
                              weight=weights["norm.weight"].to(torch.float16),
                              bias=weights["norm.bias"].to(torch.float16))
    
    # Get dimensions
    seqlen = normalized.shape[1]

    # Project using two linear layers
    def project(tensor, weight):
        return F.linear(tensor, weight, None).atanh()

    left = project(normalized, weights["left_proj.weight"])
    right = project(normalized, weights["right_proj.weight"])
    
    # Compute gates in PyTorch
    if not nomask:
        mask = mask.to(torch.float16)
        left *= mask.view(bs, 1, seqlen, seqlen)

    left_gate = F.sigmoid(project(normalized, weights["left_gate.weight"]))
    right_gate = F.sigmoid(project(normalized, weights["right_gate.weight"]))
    
    # Scaled activations
    left_scaled = left * left_gate
    right_scaled = right * right_gate

    # Transpose for the persistent kernel
    # left_scaled: [bs, seqlen, hidden_dim] -> transposed [hidden_dim, seqlen, bs]
    # right_scaled: [bs, seqlen, hidden_dim] -> transposed [hidden_dim, 1, seqlen] but wait:

    # We need to change the layout for contraction: 
    #   left_scaled need to be [hidden_dim * seqlen, seqlen]
    #   right_scaled need to be [seqlen * hidden_dim, hidden_dim] but then transposed for persistent operation.

    #   left_scaled = left_scaled.view(-1, seqlen)  # [hidden_dim * seqlen, seqlen]
    #   for persistent kernel, we set the block dimensions differently.

    # However, note the contraction requires the matrix multiplication for each (b,d)
    #   out(b,i,j,d) = einsum over k: left_scaled(b,i,k)d * right_scaled(b,j,k)d

    # Using the new approach from the TP example:
    left_scaled_perm = left_scaled.permute(0, 2, 1, 3).contiguous().view(-1, 1, seqlen)

    right_scaled_perm = right_scaled.permute(0, 2, 1, 3).contiguous().view(-1, seqlen, hidden_dim)

    output *= out_gate[:, None, None, :]
