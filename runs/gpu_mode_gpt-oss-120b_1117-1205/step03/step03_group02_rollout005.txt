"""
TriMul (outgoing) forward pass with Triton fusion.

Algorithm
---------
1. Layer‑norm the input (float32 → float16).
2. Linear projection to a hidden dimension (half‑precision).
3. Compute three sigmoid gates (half‑precision).
4. If a mask is present, fuse the mask, the left/right gates,
   and element‑wise multiplication into a single Triton kernel.
   Otherwise perform the element‑wise multiplication in PyTorch.
5. Core contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication of shape (B·H, N, N) and
   is performed with PyTorch’s highly‑optimized cuBLAS kernel.
6. Apply a second LayerNorm, the output gate and a final linear
   projection back to the original dimension.
7. Return the result as float32.

Only the mask‑gate fusion is implemented in Triton; the rest
leverages PyTorch’s fast kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask (optional) with left/right gates
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # [total] half
    right_ptr,         # [total] half
    left_gate_ptr,     # [total] half
    right_gate_ptr,    # [total] half
    mask_ptr,          # [mask_total] half (broadcast)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask is present, else 0
    H: tl.constexpr,          # hidden dimension
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # Each mask entry corresponds to H consecutive hidden values
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store results back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of left/right multiplicative gates and optional mask.
    All tensors must be torch.float16 and contiguous.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when HAS_MASK==0

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # No return – operation performed in‑place


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or float), may be ignored if config["nomask"] is True
            - weights: dict mapping parameter names to torch.Tensor (float32)
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool, optional)

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension
    # ------------------------------------------------------------------
    left = F.linear(
        x, weights["left_proj.weight"].to(torch.float16)
    )  # [B, N, N, hidden]
    right = F.linear(
        x, weights["right_proj.weight"].to(torch.float16)
    )  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 3. Compute gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = F.linear(
        x, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = F.linear(
        x, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask with gates (optional) via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → broadcast over hidden dim
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        # Flatten tensors for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        lg_f = left_gate.contiguous().view(-1)
        rg_f = right_gate.contiguous().view(-1)
        mask_f = mask_h.view(-1)  # length B*N*N
        _apply_gate_and_mask(
            left_f, right_f, lg_f, rg_f, mask_f, hidden_dim
        )
        # Reshape back to original layout
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask: simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched matrix multiplication using torch.bmm
    # ------------------------------------------------------------------
    # Permute to (B, H, N, N) where the last two dims are the matrices.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) = (B,H,k,j)

    B, H, N, _ = left_perm.shape
    # Collapse batch & heads for batched GEMM
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Batched GEMM (half‑precision, cuBLAS kernels)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)

    # Restore (B, N, N, H) layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return in float32 as required
    return out.to(torch.float32)
