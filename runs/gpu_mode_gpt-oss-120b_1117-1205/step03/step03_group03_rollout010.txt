    stride_i: tl.int32,      # stride for i/j dimension (N*H)
    stride_k: tl.int32,      # stride for k dimension (H)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,

    pid_n = tl.program_id(1)

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

"""
TriMul (outgoing) forward pass – fully fused batched GEMM kernel.

Algorithm
---------
1. LayerNorm on the input tensor (float16 precision).
2. Linear projections `left` and `right` to `hidden_dim` (half‑precision).
3. Compute three gating tensors via linear + sigmoid (half‑precision).
4. Optional mask multiplication (element‑wise, kept in PyTorch).
5. Core contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched half‑precision GEMM in Triton (no data copies).
6. Output LayerNorm, gating, final linear projection (half‑precision).
7. Return float32 tensor.

The heavy O(N³) work is carried out by the Triton kernel, which
processes each (batch, head) pair as an independent GEMM using
BLOCK_M×BLOCK_N tiles and a BLOCK_K reduction loop.
All accumulations are performed in float32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_batched_kernel(
    left_ptr,          # half*   (B,N,N,H) input left tensor
    right_ptr,         # half*   (B,N,N,H) input right tensor
    out_ptr,           # half*   (B,N,N,H) output tensor
    N: tl.int32,               # sequence length
    stride_batch: tl.int32,    # stride between batches (B dimension) = N*N*H
    stride_i: tl.int32,        # stride for the `i` / `j` dimension = N*H
    stride_k: tl.int32,        # stride for the `k` dimension = H
    hidden_dim: tl.int32,      # number of heads / hidden dimension (H)
    BLOCK_M: tl.constexpr,     # tile size in M (i)
    BLOCK_N: tl.constexpr,     # tile size in N (j)
    BLOCK_K: tl.constexpr,     # tile size in K (k)
):
    pid_m = tl.program_id(0)        # tile row
    pid_n = tl.program_id(1)        # tile column
    pid_bh = tl.program_id(2)       # flattened batch * head index

    # Decode batch and head from flattened index.
    b = pid_bh // hidden_dim
    h = pid_bh % hidden_dim

    # Base offset for this (b, h) pair.
    base = b * stride_batch + h      # offset in elements

    # Offsets for the current tile.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers to the start of the tile for A (left) and B (rightᵀ).
    # A[i,k] : offset = base + i*stride_i + k*stride_k
    # B[k,j] : offset = base + k*stride_k + j*stride_i   (right transposed)
    a_ptrs = left_ptr + base + offs_m[:, None] * stride_i + offs_k[None, :] * stride_k
    b_ptrs = right_ptr + base + offs_k[:, None] * stride_k + offs_n[None, :] * stride_i

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension.
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_iter * BLOCK_K
        k_valid = (k_offset + offs_k) < N

        # Masks for the current K‑block.
        mask_a = mask_m[:, None] & k_valid          # (BLOCK_M, BLOCK_K)
        mask_b = k_valid[:, None] & mask_n          # (BLOCK_K, BLOCK_N)

        a = tl.load(a_ptrs, mask=mask_a, other=tl.float16(0.0))
        b = tl.load(b_ptrs, mask=mask_b, other=tl.float16(0.0))

        acc += tl.dot(a, b)   # ∈ float32

        # Advance pointers to the next K‑block.
        a_ptrs += BLOCK_K * stride_k
        b_ptrs += BLOCK_K * stride_k

    # Write the result back (output layout: (B,N,N,H) → stride_j = H)
    mask_out = mask_m[:, None] & mask_n
    out = acc.to(tl.float16)
    tl.store(
        out_ptr + base + offs_m[:, None] * stride_i + offs_n[None, :] * stride_k,
        out,
        mask=mask_out,
    )


# ----------------------------------------------------------------------
# Entry point exposed to the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict containing the model parameters (float32 tensors)
            - config: dict containing "dim", "hidden_dim", and optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Cast weights to half precision once (they are on the correct device)
    # ------------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    norm_weight = to_half(weights["norm.weight"])
    norm_bias = to_half(weights["norm.bias"])
    left_proj_weight = to_half(weights["left_proj.weight"])
    right_proj_weight = to_half(weights["right_proj.weight"])
    left_gate_weight = to_half(weights["left_gate.weight"])
    right_gate_weight = to_half(weights["right_gate.weight"])
    out_gate_weight = to_half(weights["out_gate.weight"])
    to_out_norm_weight = to_half(weights["to_out_norm.weight"])
    to_out_norm_bias = to_half(weights["to_out_norm.bias"])
    to_out_weight = to_half(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)      # [B, N, N, H]
    right = F.linear(x, right_proj_weight)    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Gating tensors (half)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    # ------------------------------------------------------------------
    # 4. Apply gates (and optional mask) – simple elementwise ops
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    if not nomask and mask is not None:
        # Broadcast mask over the hidden dimension
        mask_h = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_h
        right = right * mask_h

    # Ensure tensors are contiguous before launching Triton
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton batched GEMM
    #    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    # ------------------------------------------------------------------
    B, N_seq, _, H = left.shape
    out = torch.empty_like(left)  # half‑precision output buffer

    # Strides for the (B, N, N, H) layout
    stride_batch = left.stride(0)   # N * N * H
    stride_i = left.stride(1)       # N * H
    stride_k = left.stride(2)       # H

    # Tile sizes – tuned for typical N (up to 1024) on an A200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N_seq, BLOCK_M),
        triton.cdiv(N_seq, BLOCK_N),
        B * H,
    )

    tri_mul_batched_kernel[grid](
        left,
        right,
        out,
        N_seq,
        stride_batch,
        stride_i,
        stride_k,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear (all half‑precision)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # -> shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 7. Return result in float32 as required by the spec
    # ------------------------------------------------------------------
    return out.to(torch.float32)
