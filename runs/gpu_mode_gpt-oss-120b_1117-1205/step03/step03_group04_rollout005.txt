import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def triton_gemm_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    batch, N,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_n,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (M x K)
        left_ptrs = left_ptr + pid_batch * stride_left_batch + offs_m[:, None] * stride_left_m + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        left = left.to(tl.float32)

        # Load right tile (K x N)
        # Right layout: (K, N)
        right_ptrs = right_ptr + pid_batch * stride_right_batch + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_n
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        right = right.to(tl.float32)

        # Accumulate
        acc += tl.dot(left, right)

    # Write output (M x N)
    out_ptrs = out_ptr + pid_batch * stride_out_batch + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. LayerNorm on dim
    # Use FP32 for LN then cast to FP16
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # 2. Projections
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # 3. Gates
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # 4. Apply gates (and mask if present)
    if not nomask and mask is not None:
        # mask shape [B, N, N], need broadcast to hidden_dim
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * left_gate * mask_h
        right = right * right_gate * mask_h
    else:
        left = left * left_gate
        right = right * right_gate

    # 5. Prepare for contraction
    # Permute to bring hidden dim before sequence dims
    # left: (B, N, N, H) -> (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: (B, N, N, H) -> (B, H, N, N) where we swap seq dims to get (K,J) order
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) but with k as row, j as col

    B, H, N, _ = left_perm.shape
    batch = B * H
    # view as (batch, N, N)
    left_mat = left_perm.view(batch, N, N)
    right_mat = right_perm.view(batch, N, N)

    # Allocate output
    out_mat = torch.empty_like(left_mat)  # same dtype FP16

    # Get strides (int64)
    # left_strides
    stride_left_batch = left_mat.stride(0)
    stride_left_m = left_mat.stride(1)
    stride_left_k = left_mat.stride(2)
    # right_strides
    stride_right_batch = right_mat.stride(0)
    stride_right_k = right_mat.stride(1)  # K dimension
    stride_right_n = right_mat.stride(2)  # N dimension (columns)
    # out strides
    stride_out_batch = out_mat.stride(0)
    stride_out_m = out_mat.stride(1)
    stride_out_n = out_mat.stride(2)

    # Choose block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (batch, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    triton_gemm_outgoing_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch, N,
        stride_left_batch, stride_left_m, stride_left_k,
        stride_right_batch, stride_right_k, stride_right_n,
        stride_out_batch, stride_out_m, stride_out_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 6. Output layernorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return float32
    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass with a Triton kernel for the
pairwise contraction.

Algorithm
---------
1. Layer‑norm over the input channel dimension (dim).
2. Linear projections to hidden_dim for left/right tensors.
3. Compute per‑position sigmoid gates (left_gate, right_gate, out_gate).
4. Apply gates and an optional pairwise mask (broadcast over hidden_dim).
5. Contraction:
   out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   This is a batched matrix multiplication (B*hidden_dim) of
   (N × N) × (N × N) → (N × N). Implemented with a custom Triton GEMM
   that tiles the N dimension (BLOCK_M/N/K = 64).
6. Output LayerNorm, out‑gate, final linear projection back to dim.
   The final result is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the outgoing TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def _triton_gemm_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    batch, N,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_n,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # which (batch * hidden) we are processing
    pid_m = tl.program_id(1)       # row tile
    pid_n = tl.program_id(2)       # column tile

    # --------------------------------------------------------------
    # Offsets for the output tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # Accumulator (FP32 for precision)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the K dimension (the shared sequence axis)
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a tile of the left matrix: (M, K)
        left_ptrs = (
            left_ptr
            + pid_batch * stride_left_batch
            + offs_m[:, None] * stride_left_m
            + offs_k[None, :] * stride_left_k
        )
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        left = left.to(tl.float32)

        # Load a tile of the right matrix: (K, N)
        # right layout is (K, N) after the permute in host code
        right_ptrs = (
            right_ptr
            + pid_batch * stride_right_batch
            + offs_k[:, None] * stride_right_k
            + offs_n[None, :] * stride_right_n
        )
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        right = right.to(tl.float32)

        # Accumulate the dot product over the block‑K dimension
        acc += tl.dot(left, right)

    # --------------------------------------------------------------
    # Write the result tile
    # --------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + pid_batch * stride_out_batch
        + offs_m[:, None] * stride_out_m
        + offs_n[None, :] * stride_out_n
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Prepare tensors for the batched GEMM
    #    left:  (B, N, N, H) -> (B, H, N, N)
    #    right: (B, N, N, H) -> (B, H, N, N) with (K,J) layout
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) where dim2 is K, dim3 is J

    B, H, N, _ = left_perm.shape
    batch = B * H

    # View as (batch, N, N)
    left_mat = left_perm.view(batch, N, N)
    right_mat = right_perm.view(batch, N, N)

    # Output buffer
    out_mat = torch.empty_like(left_mat)

    # ------------------------------------------------------------------
    # 5. Launch the Triton GEMM kernel
    # ------------------------------------------------------------------
    # Strides (in elements, not bytes)
    stride_left_batch = left_mat.stride(0)
    stride_left_m = left_mat.stride(1)   # rows (i)
    stride_left_k = left_mat.stride(2)   # columns (k)

    stride_right_batch = right_mat.stride(0)
    stride_right_k = right_mat.stride(1)  # rows (k)
    stride_right_n = right_mat.stride(2)  # columns (j)

    stride_out_batch = out_mat.stride(0)
    stride_out_m = out_mat.stride(1)
    stride_out_n = out_mat.stride(2)

    # Tile sizes – 64 works well for the target dimensions
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _triton_gemm_outgoing_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch,
        N,
        stride_left_batch,
        stride_left_m,
        stride_left_k,
        stride_right_batch,
        stride_right_k,
        stride_right_n,
        stride_out_batch,
        stride_out_m,
        stride_out_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Reshape back to (B, N, N, hidden_dim)
    # ------------------------------------------------------------------
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return in float32 as required
    return out.to(torch.float32)
