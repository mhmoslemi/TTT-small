def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for M and N
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointer base for this batch
    a_ptr_batch = a_ptr + batch_idx * stride_a_batch
    b_ptr_batch = b_ptr + batch_idx * stride_b_batch
    c_ptr_batch = c_ptr + batch_idx * stride_c_batch

    # Loop over K dimension
    for k in range(0, K, BLOCK_K):

        mask_k = offs_k < K

        # Load a tile

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

    # Store result
    c_ptrs = c_ptr_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    c = acc.to(tl.float16)

    a_ptr, b_ptr, c_ptr,

    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr

"""
TriMul (outgoing) forward pass with a custom Triton kernel for the heavy
pairwise contraction. The kernel computes a batched matrix multiplication
for each hidden‑dimension slice:
    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
All other linear layers, gating and layer‑norms remain in PyTorch (FP16)
to keep the implementation simple yet fast.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch: tl.int32, M: tl.int32, N: tl.int32, K: tl.int32,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: C[batch,M,N] = A[batch,M,K] @ B[batch,K,N] (FP16 inputs,
    FP32 accumulation)."""
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch element
    a_batch_ptr = a_ptr + pid_batch * (M * K)
    b_batch_ptr = b_ptr + pid_batch * (K * N)
    c_batch_ptr = c_ptr + pid_batch * (M * N)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (BLOCK_M x BLOCK_K)
        a_ptrs = a_batch_ptr + offs_m[:, None] * K + offs_k[None, :]
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile (BLOCK_K x BLOCK_N)
        b_ptrs = b_batch_ptr + offs_k[:, None] * N + offs_n[None, :]
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Write back the result
    c_ptrs = c_batch_ptr + offs_m[:, None] * N + offs_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------
    # 1. Input LayerNorm (float32) -> FP16 for the rest
    # ------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------
    # 2. Linear projections (FP16)
    # ------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------
    # 3. Gating (FP16)
    # ------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------
    # 4. Apply gates and optional mask (elementwise, cheap)
    # ------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------
    # 5. Core contraction: batched GEMM via Triton
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------
    # Permute to (B, H, N, N) where H = hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) with k and j swapped

    B, H, N, _ = left_perm.shape
    batch = B * H

    # Flatten the leading dimensions -> (batch, N, N)
    A = left_perm.view(batch, N, N)
    B_mat = right_perm.view(batch, N, N)
    C = torch.empty_like(A)  # FP16 output

    # Tile sizes (tuned for typical B200 memory hierarchy)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_matmul_kernel[grid](
        A, B_mat, C,
        batch, N, N, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # Reshape back to (B, N, N, H)
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return in float32 as required
    return out.to(torch.float32)
