@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=0.)
    right = tl.load(right_ptr + offs, mask=mask, other=0.)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.)

    left = left * left_gate
    right = right * right_gate

    if HAS_MASK:
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.)
        left = left * mask_val
        right = right * mask_val

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,),
                      weight=weights["norm.weight"], bias=weights["norm.bias"])
    x = x.to(torch.float16)

    # Linear projections
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # Gate tensors
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # Fuse gates and mask
    if not nomask:
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        # flatten to 1D
        left_flat = left.view(-1)
        right_flat = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)
        mask_flat = mask_h.view(-1)  # shape B*N*N
        _apply_gate_and_mask(left_flat, right_flat,
                             left_gate_flat, right_gate_flat,
                             mask_flat, hidden_dim)
        # reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
        # can delete mask if not needed
    else:
        left = left * left_gate
        right = right * right_gate

    # core contraction
    # left_perm: (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()
    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B*H, N, N)
    right_mat = right_perm.view(B*H, N, N)

    # Use torch.bmm for half precision
    out_mat = torch.bmm(left_mat, right_mat)  # result half

    # reshape back
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    # Output LayerNorm + gating + final linear
    out = F.layer_norm(out, (hidden_dim,),
                       weight=weights["to_out_norm.weight"].to(torch.float16),
                       bias=weights["to_out_norm.bias"].to(torch.float16))
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return float32
    return out.to(torch.float32)

    BLOCK: tl.constexpr,
