```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements: tl.int32,
    H: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # apply gates
    left = left * left_gate
    right = right * right_gate

    if HAS_MASK:
        # mask index per H entries
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * mask_val
        right = right * mask_val

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def _apply_gate_and_mask(left, right, left_gate, right_gate, mask):
    # left,right, left_gate,right_gate: half tensors contiguous
    # mask: half tensor of shape [B,N,N] or None
    total = left.numel()
    BLOCK = 1024  # can tune
    grid = (triton.cdiv(total, BLOCK),)
    # mask pointer: if mask is None, can pass left tensor pointer as dummy
    mask_ptr = mask if mask is not None else left
    HAS_MASK = 1 if mask is not None else 0
    H = left.shape[-1]  # hidden dimension
    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        H=H,
        HAS_MASK=HAS_MASK,
        BLOCK=BLOCK,
    )

def custom_kernel(data):
    """
    Outgoing TriMul forward pass.
    Input: tuple (x, mask, weights, config)
    Returns: tensor [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --- LayerNorm on input (float32) ---
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to half for subsequent ops
    x_h = x_norm.to(torch.float16)

    # --- Linear projections to hidden dim (half) ---
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # --- Gating (half) ---
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # --- Fuse mask (if any) and gates in-place via Triton kernel ---
    if not nomask:
        # mask: [B, N, N] -> broadcast to hidden_dim
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        # Flatten for kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask_h.contiguous().view(-1)  # shape B*N*N
        _apply_gate_and_mask(left_f, right_f, left_gate_f, right_gate_f, mask_f)
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # simple elementwise multiplication, could be fused later
        left = left * left_gate
        right = right * right_gate

    # Release gates not needed
    del left_gate, right_gate, x_h

    # --- Core contraction: out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h] ---
    # Transform to (B*H, N, N) batched matmul
    # left: [B, N, N, H] -> (B, H, N, N) via permute
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right need to be transposed on the N axes for matmul: right[b, j, k, h] -> (B,H,N,N) where axes (i->j) mapping
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape
    # Reshape to (B*H, N, N)
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Half precision batched GEMM
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)
    # Reshape back to [B, H, N, N] then permute to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --- Output LayerNorm, out_gate scaling, final linear ---
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # elementwise, broadcasting over hidden dim
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return float32
    return out.to(torch.float32)

"""
Outgoing TriMul forward pass.

Algorithm:
1. LayerNorm the input tensor (float32) using provided weights.
2. Cast to float16 and apply two linear projections (left/right) to the hidden dimension.
3. Compute three gating tensors (left_gate, right_gate, out_gate) via linear+sigmoid.
4. Fuse mask (if present) and the left/right gates in‑place with a Triton kernel.
5. Core contraction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched half‑precision GEMM (torch.bmm) after permuting to (B*H,N,N).
6. Apply output LayerNorm, multiply by out_gate, and a final linear projection back to dim.
7. Return the result as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # half*  (total elements)
    right_ptr,           # half*  (total elements)
    left_gate_ptr,       # half*  (total elements)
    right_gate_ptr,      # half*  (total elements)
    mask_ptr,            # half*  (mask elements) – used only if HAS_MASK==1
    total_elements: tl.int32,
    H: tl.int32,         # hidden dimension size
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Apply per‑element gates
    left = left * left_gate
    right = right * right_gate

    if HAS_MASK:
        # One mask value corresponds to H consecutive hidden entries
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * mask_val
        right = right * mask_val

    # Write back fused tensors
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask):
    """
    In‑place fusion of ``left``/``right`` with their gates and optional mask
    using a Triton kernel.

    All tensors must be ``torch.float16`` and contiguous.
    ``mask`` is ``None`` when no mask is present; otherwise it has shape
    ``[B, N, N]`` (no trailing hidden dimension).
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when no mask
    H = left.shape[-1]

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        H,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )


def custom_kernel(data):
    """
    Outgoing TriMul forward pass.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32/bool) or None
            - weights: dict containing all model parameters (float32 tensors)
            - config: dict with keys ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. LayerNorm on the input (float32)
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x_norm.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half precision)
    # --------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 3. Gating tensors (half precision)
    # --------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # --------------------------------------------------------------
    # 4. Fuse mask (if any) and gates in‑place via Triton
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> broadcast over hidden_dim
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        # Flatten for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask_h.contiguous().view(-1)          # shape B*N*N
        _apply_gate_and_mask(left_f, right_f, left_gate_f, right_gate_f, mask_f)
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # Simple element‑wise gate multiplication when no mask
        left = left * left_gate
        right = right * right_gate

    # Gates no longer needed
    del left_gate, right_gate, x_h

    # --------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # Rearrange to (B, H, N, N) layout for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (transpose on N axes)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)  # (B*H, N, N)

    # Batched half‑precision matrix multiplication
    out_mat = torch.bmm(left_mat, right_mat)   # (B*H, N, N)

    # Restore original layout [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm, out_gate scaling, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate                         # element‑wise gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 7. Return float32 tensor
    # --------------------------------------------------------------
    return out.to(torch.float32)
