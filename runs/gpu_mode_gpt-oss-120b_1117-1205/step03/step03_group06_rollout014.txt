def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, M, N, K,
    stride_left_bh, stride_left_m, stride_left_k,
    stride_right_bh, stride_right_k, stride_right_n,
    stride_out_bh, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    total_batches = B * H
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch_idx = pid // (num_blocks_m * num_blocks_n)
    block_idx = pid % (num_blocks_m * num_blocks_n)
    pid_m = block_idx // num_blocks_n
    pid_n = block_idx % num_blocks_n

    # offsets within block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out of bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load left tile
        left_offset = left_ptr + batch_idx * stride_left_bh + offs_m[:, None] * stride_left_m + offs_k[None, :] * stride_left_k
        left = tl.load(left_offset, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load right tile
        right_offset = right_ptr + batch_idx * stride_right_bh + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_n
        right = tl.load(right_offset, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(left, right)

    # write back
    out_offset = out_ptr + batch_idx * stride_out_bh + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    tl.store(out_offset, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
Implementation of the outgoing Triangle Multiplicative Update (TriMul) operator.
The forward pass consists of:
1. LayerNorm over the input channel dimension.
2. Fused linear projections for left/right tensors together with their gates.
3. Optional mask/gate fusion using a small Triton kernel.
4. Batched matrix multiplication per hidden channel (i.e. Σ_k left[i,k,d] *
   right[j,k,d]) using a custom Triton GEMM kernel.
5. Output LayerNorm, gating and final linear projection back to the original
   dimension.
All heavy linear work is done in half precision (FP16) and the O(N³) contraction
is performed by a tiled Triton kernel that merges the batch and hidden dims.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Kernel 1: fuse mask and gates (optional)
# --------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,      # half *
    right_ptr,     # half *
    left_gate_ptr, # half *
    right_gate_ptr,# half *
    mask_ptr,      # half *
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,  # 1 if mask is present, else 0
    H: tl.constexpr,         # hidden dimension
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elems

    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    lgate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    if HAS_MASK:
        mask_idx = offs // H
        mval = tl.load(mask_ptr + mask_idx, mask=valid, other=0.0)
        left = left * lgate * mval
        right = right * rgate * mval
    else:
        left = left * lgate
        right = right * rgate

    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """In‑place mask & gate fusion for tensors of shape [B,N,N,H]."""
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when no mask
    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )


# --------------------------------------------------------------
# Kernel 2: batched GEMM for the TriMul contraction
# Computes out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,k,j]
# --------------------------------------------------------------
@triton.jit
def _tri_mul_matmul_kernel(
    left_ptr,   # half *
    right_ptr,  # half *
    out_ptr,    # half *
    M, N, K,    # dimensions (all ints)
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_n,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # number of blocks in each dimension
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # flatten (batch*head) dimension together with the block grid
    total_batches = tl.cdiv(pid, num_blocks_m * num_blocks_n)  # not used directly
    batch = pid // (num_blocks_m * num_blocks_n)
    tid = pid % (num_blocks_m * num_blocks_n)
    pid_m = tid // num_blocks_n
    pid_n = tid % num_blocks_n

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile: (M, K)
        a_ptrs = left_ptr + batch * stride_left_batch \
                 + offs_m[:, None] * stride_left_m \
                 + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right tile: (K, N)
        b_ptrs = right_ptr + batch * stride_right_batch \
                 + offs_k[:, None] * stride_right_k \
                 + offs_n[None, :] * stride_right_n
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # write back
    c_ptrs = out_ptr + batch * stride_out_batch \
             + offs_m[:, None] * stride_out_m \
             + offs_n[None, :] * stride_out_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------
# Entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.
    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)  # work in half precision from here

    # ------------------------------------------------------------------
    # 2. Fused linear projections for left (+left_gate) and right (+right_gate)
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    x_flat = x.view(-1, dim)                     # (B*N*N, dim)

    # left projection + left gate
    left_fused_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                          # (2*H, dim)
    left_fused = F.linear(x_flat, left_fused_w)  # (B*N*N, 2*H)
    left_fused = left_fused.view(B, N, N, 2 * hidden_dim)
    left = left_fused[..., :hidden_dim]          # (B,N,N,H)
    left_gate = left_fused[..., hidden_dim:].sigmoid()  # (B,N,N,H)

    # right projection + right gate
    right_fused_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                          # (2*H, dim)
    right_fused = F.linear(x_flat, right_fused_w)  # (B*N*N, 2*H)
    right_fused = right_fused.view(B, N, N, 2 * hidden_dim)
    right = right_fused[..., :hidden_dim]            # (B,N,N,H)
    right_gate = right_fused[..., hidden_dim:].sigmoid()  # (B,N,N,H)

    # out gate (no fusion needed)
    out_gate = torch.sigmoid(
        F.linear(x_flat, weights["out_gate.weight"].to(torch.float16))
    ).view(B, N, N, hidden_dim)                 # (B,N,N,H)

    # ------------------------------------------------------------------
    # 3. Apply mask (if any) together with gates using Triton
    # ------------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        # flatten tensors
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask_h.view(-1)  # (B*N*N)
        _apply_gate_and_mask(
            left_f, right_f,
            left_gate_f, right_gate_f,
            mask_f,
            hidden_dim,
        )
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # simple elementwise gating
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched GEMM per hidden channel via Triton.
    # ------------------------------------------------------------------
    # Bring hidden dimension to leading axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N) -> (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) -> (k,j)

    B_, H_, M, _ = left_perm.shape   # M == N == K
    out_perm = torch.empty_like(left_perm)

    # strides (number of elements)
    stride_left_batch = left_perm.stride(1)   # N*N
    stride_left_m = left_perm.stride(2)       # N
    stride_left_k = left_perm.stride(3)       # 1

    stride_right_batch = right_perm.stride(1)  # N*N
    stride_right_k = right_perm.stride(2)      # N
    stride_right_n = right_perm.stride(3)      # 1

    stride_out_batch = out_perm.stride(1)      # N*N
    stride_out_m = out_perm.stride(2)          # N
    stride_out_n = out_perm.stride(3)          # 1

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    total_batches = B_ * H_
    num_blocks_m = triton.cdiv(M, BLOCK_M)
    num_blocks_n = triton.cdiv(N, BLOCK_N)
    grid = (total_batches * num_blocks_m * num_blocks_n,)

    _tri_mul_matmul_kernel[grid](
        left_perm, right_perm, out_perm,
        M, N, M,  # K == M == N
        stride_left_batch, stride_left_m, stride_left_k,
        stride_right_batch, stride_right_k, stride_right_n,
        stride_out_batch, stride_out_m, stride_out_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, out‑gate and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return in the requested precision
    return out.to(torch.float32)
