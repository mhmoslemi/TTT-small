"""
TriMul (outgoing) forward pass – highly fused Triton + PyTorch implementation.

Algorithm
---------
1. Layer‑norm the [B,N,N,dim] input (float32 → float16).
2. Linear projections to hidden_dim for left/right tensors (half‑precision).
3. Compute left/right/out gates (sigmoid of linear).  If a mask is present,
   a Triton kernel fuses:   left = left * left_gate * mask,
   right = right * right_gate * mask   (broadcast mask over hidden_dim).
   Otherwise left/right are simply multiplied by their gates.
4. Core O(N³) contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix‑multiply per hidden dimension:
        out_d = left_d @ right_dᵀ
   Implemented with Torch’s highly‑optimized batched GEMM (FP16, FP32 accumulation).
5. Output LayerNorm, element‑wise out‑gate, and final linear projection back to dim.
6. Return the result in float32.

Only the mask‑gate fusion is implemented in Triton; everything else uses
high‑performance PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: in‑place fusion of mask (optional) and left/right gates
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # half*   flattened left tensor
    right_ptr,           # half*   flattened right tensor
    left_gate_ptr,       # half*   flattened left_gate tensor
    right_gate_ptr,      # half*   flattened right_gate tensor
    mask_ptr,            # half*   flattened mask (length = total_elements / H)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 → mask present, 0 → no mask
    H: tl.constexpr,          # hidden dimension (gate size)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load data
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # Each mask element corresponds to H consecutive hidden entries
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store results back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


# ----------------------------------------------------------------------
# Helper to fuse mask + gates (in‑place)
# ----------------------------------------------------------------------
def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    Fuse `left = left * left_gate * mask` and `right = right * right_gate * mask`
    (if mask provided) using a Triton kernel. All tensors must be half‑precision
    and contiguous.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when no mask

        left.view(-1), right.view(-1),
        left_gate.view(-1), right_gate.view(-1),
        mask_ptr.view(-1),
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # Result is written in‑place; no return value.

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) – may be ignored
            - weights: mapping name → torch.Tensor (float32)
            - config: dict with keys
                * "dim" (int) – input feature dimension
                * "hidden_dim" (int) – projection dimension
                * "nomask" (bool, optional) – True if mask should be ignored

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32 → float16)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections to hidden_dim
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 3. Gate values (sigmoid)
    # --------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # --------------------------------------------------------------
    # 4. Fuse mask (if any) with gates via Triton
    # --------------------------------------------------------------
    if not nomask:
        # Broadcast mask over hidden_dim
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        _apply_gate_and_mask(left, right, left_gate, right_gate, mask_h, hidden_dim)
    else:
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as batched GEMM: (B*H) matrices of size N×N.
    # --------------------------------------------------------------
    # Rearrange to (B, H, N, N) where last two dims are the matrix axes.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) = transpose k↔j
    B, H, N, _ = left_perm.shape

    # Collapse batch and head dimensions for torch.bmm
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Batched GEMM (FP16 inputs, FP32 accumulation)
    out_mat = torch.bmm(left_mat, right_mat)

    # Restore shape (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return float32 as required by the spec
    return out.to(torch.float32)
