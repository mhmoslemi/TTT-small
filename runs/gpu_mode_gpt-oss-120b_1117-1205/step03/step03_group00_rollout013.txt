"""
TriMul "outgoing" forward pass.

The implementation follows the reference PyTorch code but reduces the number of
large linear layers and fuses mask handling with gating in a tiny Triton
kernel.  The main O(N³) contraction is performed as a batched half‑precision
matrix multiplication (torch.bmm), which maps to cuBLAS and is highly
optimized on modern GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # half*  (flattened left tensor)
    right_ptr,              # half*  (flattened right tensor)
    left_gate_ptr,          # half*  (flattened left gate tensor, sigmoid already applied)
    right_gate_ptr,         # half*  (flattened right gate tensor, sigmoid already applied)
    mask_ptr,               # half*  (flattened mask tensor, broadcasted over hidden dim)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present else 0
    H: tl.constexpr,          # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    # Load tensors (masked load returns 0.0 for out‑of‑bounds)
    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    if HAS_MASK:
        # each hidden channel shares the same mask entry
        mask_offs = offs // H
        mask_val = tl.load(mask_ptr + mask_offs, mask=valid, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of left/right gating and optional mask.
    All tensors must be contiguous, half‑precision and have matching shapes:
        left / right          : [B, N, N, H]
        left_gate / right_gate: [B, N, N, H] (already sigmoid‑ed)
        mask                  : [B, N, N] (or None)
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when mask not used

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # operation is performed in‑place; nothing to return


def custom_kernel(data):
    """
    TriMul forward (outgoing) using half‑precision kernels.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask        : torch.Tensor [B, N, N] (bool or numeric) – may be None
            - weights     : dict of model parameters (float32)
            - config      : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float16

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (half precision)
    # ------------------------------------------------------------------
    x = input_tensor.to(dtype)
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # ------------------------------------------------------------------
    # 2. Combined linear projections for left/right & their gates
    # ------------------------------------------------------------------
    # combine proj and gate weights so we read x only once per side
    left_combined_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(dtype)
    right_combined_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(dtype)

    left_combined = F.linear(x, left_combined_w)   # [B, N, N, 2*H]
    right_combined = F.linear(x, right_combined_w)  # [B, N, N, 2*H]

    # split projection and gate, apply sigmoid to gates
    left = left_combined[..., :hidden_dim]
    left_gate = left_combined[..., hidden_dim:].sigmoid()
    right = right_combined[..., :hidden_dim]
    right_gate = right_combined[..., hidden_dim:].sigmoid()

    # ------------------------------------------------------------------
    # 3. Mask & gate fusion (if needed)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] -> half, flattened
        mask_h = mask.to(dtype).contiguous().view(-1)
        # flatten tensors
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(
            left_f, right_f, left_gate_f, right_gate_f, mask_h, hidden_dim
        )
        # reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # simple element‑wise gating (mask already applied as identity)
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Out‑gate (independent of mask)
    # ------------------------------------------------------------------
    out_gate = F.linear(x, weights["out_gate.weight"].to(dtype)).sigmoid()

    # ------------------------------------------------------------------
    # 5. Core O(N³) contraction: batched GEMM on half precision
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) for torch.bmm
    B, N, _, H = left.shape
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left_mat = left_perm.view(B * H, N, N)               # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)             # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(dtype),
        bias=None,
    )

    # Return in original (float32) dtype
    return out.to(torch.float32)
