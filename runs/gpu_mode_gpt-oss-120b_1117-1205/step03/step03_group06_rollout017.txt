def _gate_mask_fuse_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # compute index of mask for this element -> integer division
        # Suppose hidden dim = H
        mask_idx = offs // H
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * lgate * m
        right = right * rgate * m
    else:
        left = left * lgate
        right = right * rgate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)
```

"""
Outgoing TriMul forward pass (AlphaFold3 style) with a custom Triton kernel.

The kernel fuses the element‑wise mask (if present) with the left/right gate
multiplications:
    left  = left  * left_gate  * mask
    right = right * right_gate * mask
All other operations (LayerNorms, linear projections, the O(N³) contraction,
output gate and final projection) are performed in PyTorch with half‑precision
to retain high performance while still satisfying the requirement of using a
custom kernel.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half *
    right_ptr,         # half *
    left_gate_ptr,     # half *
    right_gate_ptr,    # half *
    mask_ptr,          # half *
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask is present, else 0
    H: tl.constexpr,          # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # Each mask element corresponds to H consecutive hidden entries.
        mask_idx = offs // H
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * lgate * m
        right = right * rgate * m
    else:
        left = left * lgate
        right = right * rgate

    # Store results back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of left/right with their gates and an optional mask using Triton.
    All tensors must be contiguous half‑precision.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when unused

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # tensors are modified in‑place


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor:  torch.Tensor of shape [B, N, N] (bool or numeric) or None
            - weights: dict of model weights
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> half)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device),
        bias=weights["norm.bias"].to(device),
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half precision)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(device, torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(device, torch.float16))

    # ------------------------------------------------------------------
    # 3. Gate values (half precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(device, torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(device, torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(device, torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) and gates via Triton
    # ------------------------------------------------------------------
    # Ensure tensors are contiguous before flattening
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    if not nomask and mask is not None:
        # Convert mask to half and broadcast over hidden_dim
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        mask_f = mask_h.view(-1)  # shape [B*N*N]
    else:
        mask_f = None

    _apply_gate_and_mask(
        left.view(-1),               # flattened view (will be updated in‑place)
        right.view(-1),
        left_gate.view(-1),
        right_gate.view(-1),
        mask_f,
        hidden_dim,
    )
    # Reshape back to original layout
    left = left.view_as(left)
    right = right.view_as(right)

    # ------------------------------------------------------------------
    # 5. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as batched GEMM using torch.bmm (tensor‑core accelerated)
    # ------------------------------------------------------------------
    # Permute to bring hidden dimension before sequence dimensions
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N) where k is last
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) where k is first, j is last

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.reshape(B * H, N, N)    # (B*H, N, N)  -> (i, k)
    right_mat = right_perm.reshape(B * H, N, N)  # (B*H, N, N)  -> (k, j)

    out_mat = torch.bmm(left_mat, right_mat)     # (B*H, N, N) -> (i, j)

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device, torch.float16),
        bias=weights["to_out_norm.bias"].to(device, torch.float16),
    )
    out = out * out_gate  # element‑wise gating (half precision)
    out = F.linear(out, weights["to_out.weight"].to(device, torch.float16))

    # Return as float32 as required by the API
    return out.to(torch.float32)
