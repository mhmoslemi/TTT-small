@triton.jit
def _gate_mask_fuse_kernel(left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr,
                          total_elements: tl.int32,
                          BLOCK: tl.constexpr,
                          HAS_MASK: tl.constexpr,
                          H: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # mask index for each element: each hidden block of size H

"""
Optimized Triton implementation of the outgoing TriMul operator.

The kernel fuses the optional pairwise mask with the left/right gating
multiplications in a single in‑place pass.  The costly pairwise contraction
is performed with a batched matrix multiplication (implemented via
torch.einsum, which dispatches to cuBLAS with Tensor‑cores).  LayerNorms,
linear projections and the final output projection remain in PyTorch.
The function returns a float‑32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask (optional) with left/right gate multiplications
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half*   [total]
    right_ptr,         # half*   [total]
    left_gate_ptr,     # half*   [total]
    right_gate_ptr,    # half*   [total]
    mask_ptr,          # half*   [mask_total] (broadcast over hidden dim)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask is present else 0
    H: tl.constexpr,          # hidden dimension
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each mask element corresponds to H consecutive hidden entries
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """In‑place: left = left * left_gate * mask; right = right * right_gate * mask."""
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when HAS_MASK==0

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # No return; tensors are modified in‑place


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/int) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> half)
    # ------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------
    # 2. Linear projections (FP16)
    # ------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------
    # 3. Gate values (FP16)
    # ------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------
    # 4. Fuse mask (if any) with left/right gate multiplication via Triton
    # ------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] -> broadcast over hidden_dim
        mask_h = mask.to(torch.float16).unsqueeze(-1).contiguous()
        # Flatten tensors for the kernel
        left_flat = left.view(-1)
        right_flat = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)
        mask_flat = mask_h.view(-1)  # size = B * N * N

        _apply_gate_and_mask(
            left_flat, right_flat,
            left_gate_flat, right_gate_flat,
            mask_flat,
            hidden_dim,
        )

        # Reshape back to original layout
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask – apply gates in‑place
        left = left.mul_(left_gate)
        right = right.mul_(right_gate)

    # ------------------------------------------------------------
    # 5. Core contraction: out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    #    Implemented via a batched matrix multiplication (einsum maps to cuBLAS)
    # ------------------------------------------------------------
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)

    # ------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Cast back to float32 as required by the specification
    return out.to(torch.float32)
