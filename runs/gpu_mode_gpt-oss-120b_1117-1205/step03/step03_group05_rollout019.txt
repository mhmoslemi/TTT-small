"""
TriMul (outgoing) forward pass with Triton accelerated batched GEMM.
The core O(N^3) contraction out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is implemented as a custom Triton kernel that treats (batch, hidden) as a
combined batch dimension and performs tiled matrix multiplication for each
(hidden) slice. All other operations (LayerNorm, Linear projections, gating,
masking) are performed with PyTorch (half precision) for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tril_mul_kernel(
    left_ptr,    # *fp16
    right_ptr,   # *fp16
    out_ptr,     # *fp16
    seq_len,     # i32
    hidden_dim,  # i32
    stride_batch,  # i32  stride for batch dimension (N*N*H)
    stride_i,      # i32  stride for i/j dimension (N*H)
    stride_k,      # i32  stride for k dimension (H)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_i = tl.program_id(0)  # row tile index
    pid_j = tl.program_id(1)  # column tile index
    pid_hb = tl.program_id(2) # combined batch*hidden index

    # decompose combined index into batch and hidden index
    hidden = hidden_dim
    batch_id = pid_hb // hidden
    h = pid_hb % hidden

    # base offset for this (batch, hidden) slice
    base = batch_id * stride_batch + h

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < seq_len
    mask_j = offs_j < seq_len

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(seq_len, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < seq_len

        # left block: (M, K)
        a_ptr = left_ptr + base + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k
        a = tl.load(a_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # right block: (K, N)  (k as rows, j as columns)
        b_ptr = right_ptr + base + offs_k[:, None] * stride_k + offs_j[None, :] * stride_i
        b = tl.load(b_ptr,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # matrix multiplication for this tile
        acc += tl.dot(a, b)

    # write result out[b,i,j,h]
    c_ptr = out_ptr + base + offs_i[:, None] * stride_i + offs_j[None, :] * hidden_dim
    tl.store(c_ptr, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

"""

import triton.language as tl

    out_ptr,     # *fp16  (output)

    stride_i,      # i32  stride for i/j dim (N*H)
    stride_k,      # i32  stride for k dim (H)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):

    pid_j = tl.program_id(1)          # tile over j
    pid_hb = tl.program_id(2)         # combined batch*hidden index

    h = pid_hb % hidden_dim

    base = batch_id * stride_batch + h

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_j = offs_j < seq_len

        stride_batch,
        stride_i,
        stride_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return in float32 as required by the specification
    return out.to(torch.float32)
```
