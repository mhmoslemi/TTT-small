"""
TriMul (outgoing) forward pass with Triton fusion.

The heavy O(N³·C) contraction
    out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
is performed by a custom Triton kernel that also fuses the per‑element
gating (sigmoid) and optional mask multiplication.  The rest of the
TriMul pipeline (LayerNorms, linear projections and final output
projection) is carried out with PyTorch (FP16 where possible).  The
kernel works on tiles of size BLOCK_M×BLOCK_N and iterates over the
reduction dimension K in BLOCK_K‑wide chunks, accumulating in FP32 for
numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_fused_kernel(
    # Tensor pointers
    left_ptr,            # left projection,    shape [B, N, N, C] (fp16)
    right_ptr,           # right projection,   shape [B, N, N, C] (fp16)
    left_gate_ptr,       # left gate,          shape [B, N, N, C] (fp16)
    right_gate_ptr,      # right gate,         shape [B, N, N, C] (fp16)
    mask_ptr,            # optional mask,      shape [B, N, N]    (fp16)
    out_ptr,             # output,             shape [B, N, N, C] (fp16)
    # Sizes
    B: tl.int32,
    N: tl.int32,
    C: tl.int32,
    # Strides (in elements)
    stride_left_b, stride_left_i, stride_left_k, stride_left_c,
    stride_right_b, stride_right_i, stride_right_k, stride_right_c,
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_c,
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_c,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_out_b, stride_out_i, stride_out_j, stride_out_c,
    # Tile sizes (compile‑time constants)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr,               # 1 if mask present, else 0
):
    # -----------------------------------------------------------------
    # Program IDs
    # -----------------------------------------------------------------
    pid = tl.program_id(0)         # combined batch * channel
    pid_i = tl.program_id(1)       # tile over i (first N)
    pid_j = tl.program_id(2)       # tile over j (second N)

    # Decode batch and channel from pid
    b = pid // C
    c = pid % C

    # -----------------------------------------------------------------
    # Tile offsets
    # -----------------------------------------------------------------
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    i_idx = i_start + tl.arange(0, BLOCK_M)   # [M]
    j_idx = j_start + tl.arange(0, BLOCK_N)   # [N]

    i_mask = i_idx < N
    j_mask = j_idx < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Main K‑loop
    # -----------------------------------------------------------------
    for k in range(0, N, BLOCK_K):
        k_idx = k + tl.arange(0, BLOCK_K)          # [K]
        k_mask = k_idx < N

        # Masks for the loads
        left_load_mask  = i_mask[:, None] & k_mask[None, :]
        right_load_mask = j_mask[:, None] & k_mask[None, :]

        # -----------------------------------------------------------------
        # Load left tile + gate (+ mask)
        # -----------------------------------------------------------------
        off_left = (
            b * stride_left_b
            + i_idx[:, None] * stride_left_i
            + k_idx[None, :] * stride_left_k
            + c * stride_left_c
        )
        left = tl.load(left_ptr + off_left, mask=left_load_mask, other=0.0)

        off_lgate = (
            b * stride_lgate_b
            + i_idx[:, None] * stride_lgate_i
            + k_idx[None, :] * stride_lgate_k
            + c * stride_lgate_c
        )
        lgate = tl.load(left_gate_ptr + off_lgate, mask=left_load_mask, other=0.0)

        left = left * lgate

        if HAS_MASK:
            off_mask_l = (
                b * stride_mask_b
                + i_idx[:, None] * stride_mask_i
                + k_idx[None, :] * stride_mask_k
            )
            mask_l = tl.load(mask_ptr + off_mask_l, mask=left_load_mask, other=0.0)
            left = left * mask_l

        # -----------------------------------------------------------------
        # Load right tile + gate (+ mask)
        # -----------------------------------------------------------------
        off_right = (
            b * stride_right_b
            + j_idx[:, None] * stride_right_i
            + k_idx[None, :] * stride_right_k
            + c * stride_right_c
        )
        right = tl.load(right_ptr + off_right, mask=right_load_mask, other=0.0)

        off_rgate = (
            b * stride_rgate_b
            + j_idx[:, None] * stride_rgate_i
            + k_idx[None, :] * stride_rgate_k
            + c * stride_rgate_c
        )
        rgate = tl.load(right_gate_ptr + off_rgate, mask=right_load_mask, other=0.0)

        right = right * rgate

        if HAS_MASK:
            off_mask_r = (
                b * stride_mask_b
                + j_idx[:, None] * stride_mask_i
                + k_idx[None, :] * stride_mask_k
            )
            mask_r = tl.load(mask_ptr + off_mask_r, mask=right_load_mask, other=0.0)
            right = right * mask_r

        # -----------------------------------------------------------------
        # Accumulate dot product over K
        # -----------------------------------------------------------------
        # left: [M, K]  right: [N, K] -> dot -> [M, N]
        acc += tl.dot(left, right)

    # -----------------------------------------------------------------
    # Write back result (cast to fp16)
    # -----------------------------------------------------------------
    out_val = acc.to(tl.float16)

    off_out = (
        b * stride_out_b
        + i_idx[:, None] * stride_out_i
        + j_idx[None, :] * stride_out_j
        + c * stride_out_c
    )
    store_mask = i_mask[:, None] & j_mask[None, :]
    tl.store(out_ptr + off_out, out_val, mask=store_mask)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict: dictionary of weight tensors
            - config_dict: must contain "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ---------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] float32

    # ---------------------------------------------------------------
    # 2. Cast to fp16 for the heavy work
    # ---------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ---------------------------------------------------------------
    # 3. Linear projections (fp16)
    # ---------------------------------------------------------------
    left_proj = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right_proj = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ---------------------------------------------------------------
    # 4. Gates (fp16)
    # ---------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ---------------------------------------------------------------
    # 5. Optional mask (fp16)
    # ---------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).contiguous()
        has_mask = True
    else:
        mask_h = torch.empty(0, device=device, dtype=torch.float16)  # dummy
        has_mask = False

    # ---------------------------------------------------------------
    # 6. Core contraction (Triton kernel)
    # ---------------------------------------------------------------
    B, N, _, _ = input_tensor.shape  # B, N, N, dim
    out_raw = torch.empty_like(left_proj)  # shape [B, N, N, hidden_dim] fp16

    # Strides (in elements)
    slb, sli, slk, slc = left_proj.stride()
    srb, sri, srk, src = right_proj.stride()
    slgb, slgi, slgk, slgc = left_gate.stride()
    srgb, srgi, srgk, srgc = right_gate.stride()
    sob, soi, soj, soc = out_raw.stride()
    if has_mask:
        smb, smi, smk = mask_h.stride()
    else:
        # dummy strides (won’t be used because HAS_MASK = 0)
        smb = smi = smk = 0

    # Tile sizes – chosen to fit in shared memory
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                     # batch * channel dimension
        triton.cdiv(N, BLOCK_M),            # tiles over i
        triton.cdiv(N, BLOCK_N),            # tiles over j
    )

    HAS_MASK = 1 if has_mask else 0

    tri_mul_fused_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask_h,
        out_raw,
        B,
        N,
        hidden_dim,
        slb, sli, slk, slc,
        srb, sri, srk, src,
        slgb, slgi, slgk, slgc,
        srgb, srgi, srgk, srgc,
        smb, smi, smk,
        sob, soi, soj, soc,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HAS_MASK=HAS_MASK,
    )

    # ---------------------------------------------------------------
    # 7. Post‑contraction processing
    # ---------------------------------------------------------------
    # Apply LayerNorm on the hidden dimension
    out = F.layer_norm(
        out_raw,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    # Multiply by out gate (as in the reference implementation)
    out = out * out_gate

    # Final linear projection back to ‘dim’
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return float32 as required by the API
    return out.to(torch.float32)
