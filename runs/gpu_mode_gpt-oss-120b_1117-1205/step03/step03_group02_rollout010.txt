"""
TriMul (outgoing) forward pass with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor (float32 → half).
2. Linear projections to the hidden dimension for the left/right tensors.
3. Compute left/right/out gates (sigmoid of a linear) and optionally apply a mask.
4. Core contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix‑multiply over the hidden dimension.
   We implement it with a custom Triton kernel that tiles the matrices in
   blocks (M×N) and iterates over the K‑dimension, accumulating in FP32.
5. Output LayerNorm, multiply by the output gate and a final linear projection.
6. Return float32 output.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the O(N³) contraction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,       # [batch, N, N]   half
    right_ptr,      # [batch, N, N]   half
    out_ptr,        # [batch, N, N]   half (output)
    N: tl.int32,                # sequence length
    stride_batch: tl.int32,     # N*N
    BLOCK_M: tl.constexpr,      # tile rows
    BLOCK_N: tl.constexpr,      # tile cols
    BLOCK_K: tl.constexpr,      # tile reduction
):
    pid_z = tl.program_id(2)          # batch index (B * hidden_dim)
    pid_x = tl.program_id(0)          # row block
    pid_y = tl.program_id(1)          # col block

    # Offsets for the output tile
    offs_i = pid_x * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_y * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + pid_z * stride_batch \
                    + offs_i[:, None] * N \
                    + offs_k[None, :]
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # Right tile (transposed): shape (BLOCK_K, BLOCK_N)
        # right[j, k] is stored as (row=j, col=k); we need (k, j)
        right_ptrs = right_ptr + pid_z * stride_batch \
                     + offs_k[:, None] * 1 \
                     + offs_j[None, :] * N
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # GEMM on the tile
        acc += tl.dot(left, right, out_dtype=tl.float32)

    # Write the result back (FP16)
    out_ptrs = out_ptr + pid_z * stride_batch \
               + offs_i[:, None] * N \
               + offs_j[None, :] * 1
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------

            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) – may be ignored
            - weights: dict of model parameters (torch.Tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32 → half)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections (half‑precision)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 3. Gate values (half‑precision, sigmoid)
    # --------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # --------------------------------------------------------------
    # 4. Apply mask (if any) and gates
    # --------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → broadcast over hidden_dim
        mask_f = mask.to(torch.float16).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_f * left_gate
        right = right * mask_f * right_gate
    else:
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction via Triton (batched GEMM)
    # --------------------------------------------------------------
    # reshape to (batch, N, N) where batch = B * hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N_, _ = left_perm.shape
    assert N_ == input_tensor.shape[1]  # sanity

    batch = B * H
    left_flat = left_perm.view(batch, N, N).contiguous()
    right_flat = right_perm.view(batch, N, N).contiguous()
    out_flat = torch.empty_like(left_flat)  # (batch, N, N), half

    # Block sizes (tuned for typical sequence lengths)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        batch,                     # batch dimension (B * hidden_dim)
    )
    stride_batch = N * N  # stride between successive (H) matrices

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return float32 as required
    return out.to(torch.float32)
