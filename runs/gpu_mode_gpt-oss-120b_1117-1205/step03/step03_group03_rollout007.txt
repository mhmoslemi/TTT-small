    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    if not nomask and mask_tensor is not None:
        # mask has shape [B, N, N]; broadcast over hidden_dim
        mask_h = mask_tensor.to(torch.float16).unsqueeze(-1).contiguous()

        left_flat = left.view(-1)
        right_flat = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)
        mask_flat = mask_h.view(-1)
        total = left_flat.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)
        HAS_MASK = 1
        gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=HAS_MASK,
            BLOCK=BLOCK,
        )
        # Reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate
    # Release gate tensors
    # Contraction: out = left @ right^T across the k dimension
    # left: (B,N,N,H), right: (B,N,N,H)
    # Permute to (B,H,N,N) for batched matmul
    left_perm = left.permute(0, 3, 1, 2)  # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1)  # (B, H, N, N)
    # Use torch.matmul which performs batched matrix multiplication
    # left_perm (B,H,N,N) @ right_perm (B,H,N,N) -> (B,H,N,N)
    # Since right_perm already transposed across last two dims, we just matmul
    # But torch.matmul expects (..., M, K) @ (..., K, N)
    # left_perm: (..., N, N) ; right_perm: (..., N, N) ; we want K = N, N = N, M = N
    # The product yields shape (B,H,N,N) as desired.
    out = torch.matmul(left_perm, right_perm)  # shape (B,H,N,N)
    # Permute back to (B,N,N,H)
    out = out.permute(0, 2, 3, 1).contiguous()
    # Output LayerNorm
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=weights["to_out_norm.weight"].to(torch.float16),
                       bias=weights["to_out_norm.bias"].to(torch.float16))
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))
    # Return float32
    return out.to(torch.float32)
```

"""
TriMul (outgoing) forward pass.

Algorithm
---------
1. Layer‑norm the input (float32) → half.
2. Linear projections `left` and `right` to `hidden_dim` (half‑precision).
3. Compute gating tensors (`left_gate`, `right_gate`, `out_gate`) and apply
   a sigmoid (half‑precision).
4. Fuse optional mask, `left_gate` and `right_gate` in a tiny Triton kernel
   that works in‑place on the flattened tensors.
5. Core contraction:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication for each head `d`.
   We permute to (B, H, N, N) and use `torch.bmm` (cuBLAS) which runs on
   Tensor‑cores in FP16.
6. Apply second LayerNorm, `out_gate`, and final linear projection back to
   `dim`.
7. Return the result in float32.

The Triton kernel only fuses mask & gating (the only part not already
accelerated by cuBLAS), satisfying the requirement to have a custom kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # half*  (flattened left)
    right_ptr,           # half*  (flattened right)
    left_gate_ptr,       # half*  (flattened left_gate)
    right_gate_ptr,      # half*  (flattened right_gate)
    mask_ptr,            # half*  (flattened mask, broadcasted)
    total_elements: tl.int32,
    hidden_dim: tl.int32,
    HAS_MASK: tl.constexpr,   # 1 if mask present, 0 otherwise
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # one mask value per (B,N,N) entry, broadcast over hidden_dim
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple of (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32, on CUDA)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping string → torch.Tensor (float32)
            - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool, ...}

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half)
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Gating tensors (half, sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse optional mask with gating via Triton
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask: [B,N,N] → broadcast over hidden_dim
        mask_h = mask_tensor.to(torch.float16).unsqueeze(-1).contiguous()

        # flatten everything for the kernel
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_h.view(-1)                         # [B*N*N]

        total = left_flat.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)

        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=1,
            BLOCK=BLOCK,
        )
        # reshape back to original layout
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # simple element‑wise multiplication when mask is absent
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # bring hidden_dim to the batch dimension for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) == rightᵀ per head

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # batched matrix multiply on FP16 (TensorCores)
    out_mat = torch.bmm(left_mat, right_mat)          # (B*H, N, N)

    # restore shape (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 7. Return float32 tensor
    # ------------------------------------------------------------------
    return out.to(torch.float32)
