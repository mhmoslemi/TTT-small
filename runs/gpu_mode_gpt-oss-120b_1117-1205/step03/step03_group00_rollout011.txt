def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(2)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(0)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute mask for m,n
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointer to start of A and B for this batch
    a_batch_ptr = a_ptr + pid_b * batch_stride_a
    b_batch_ptr = b_ptr + pid_b * batch_stride_b
    c_batch_ptr = c_ptr + pid_b * batch_stride_c

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute k offsets
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # mask for k dimension
        mask_k = offs_k < K

        # load A tile
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

The heavy O(N³·C) contraction
    out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
is implemented as a batched GEMM (left @ rightᵀ) using a custom
Triton kernel that processes many (batch, head) matrices in parallel.
All other ops (LayerNorm, Linear, gating, mask) are performed in PyTorch
in float16 for speed; the kernel guarantees at least one Triton‑based
fusion.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (A @ Bᵀ) for many heads.
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    batch_stride_a, stride_am, stride_ak,    # A strides (batch, row, col)
    batch_stride_b, stride_bk, stride_bn,    # B strides (batch, row, col) – B is already transposed
    batch_stride_c, stride_cm, stride_cn,    # C strides (batch, row, col)
    M, N, K,                                 # matrix dimensions (all = seq_len)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)                # batch * head dimension
    pid_m = tl.program_id(1)                # row block
    pid_n = tl.program_id(0)                # col block

    # --------------------------- offsets ------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # batch pointers
    a_batch = a_ptr + pid_b * batch_stride_a
    b_batch = b_ptr + pid_b * batch_stride_b
    c_batch = c_ptr + pid_b * batch_stride_c

    # accumulator in fp32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------- K-loop --------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B tile: (BLOCK_K, BLOCK_N) – B is already transposed
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product, accumulate in fp32
        acc += tl.dot(a, b).to(tl.float32)

    # -------------------------- store ---------------------------------
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

    BLOCK: tl.constexpr,
    H: tl.constexpr,             # hidden dimension
    HAS_MASK: tl.constexpr,      # 1 if mask provided else 0
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each hidden channel group shares the same mask entry
        mask_off = offs // H
        m = tl.load(mask_ptr + mask_off, mask=mask, other=0.0)
        left = left * left_gate * m
        right = right * right_gate * m
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """In‑place fusion of left/right with their gates and optional mask."""
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when not used

    gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        H=hidden_dim,
        HAS_MASK=has_mask,
    )
    # No return; tensors are modified in‑place.


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (optional) – may be None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections (half precision, no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16), bias=None)

    # --------------------------------------------------------------
    # 3. Gates (half precision, sigmoid)
    # --------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # --------------------------------------------------------------
    # 4. Apply mask and gating (fused Triton kernel if mask present)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).contiguous()
        # flatten tensors for kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_g_f = left_gate.contiguous().view(-1)
        right_g_f = right_gate.contiguous().view(-1)
        _apply_gate_and_mask(left_f, right_f, left_g_f, right_g_f, mask_h, hidden_dim)
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # simple element‑wise multiplication when mask is absent
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction: batched GEMM (left @ rightᵀ) per head
    #    left/right: [B, N, N, C] → (B, C, N, N)
    # --------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    # move channel to second dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, C, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, C, N, N)
    right_t = right_perm.transpose(-2, -1)              # logical transpose, no copy

    # flatten batch & head dimensions → (B*C, N, N)
    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_t.view(B * hidden_dim, N, N)
    out_mat = torch.empty_like(left_mat)  # half precision

    # strides (in elements) for the three tensors
    a_batch_stride = left_mat.stride(0)
    a_row_stride = left_mat.stride(1)   # = N
    a_col_stride = left_mat.stride(2)   # = 1

    b_batch_stride = right_mat.stride(0)
    b_row_stride = right_mat.stride(1)  # after transpose => 1
    b_col_stride = right_mat.stride(2)  # after transpose => N

    c_batch_stride = out_mat.stride(0)
    c_row_stride = out_mat.stride(1)    # = N
    c_col_stride = out_mat.stride(2)    # = 1

    # block sizes – tuned for B200 (half precision)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_N),   # program_id(0) – column blocks
        triton.cdiv(N, BLOCK_M),   # program_id(1) – row    blocks
        B * hidden_dim,            # program_id(2) – batch×head
    )

    batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        a_batch_stride,
        a_row_stride,
        a_col_stride,
        b_batch_stride,
        b_row_stride,
        b_col_stride,
        c_batch_stride,
        c_row_stride,
        c_col_stride,
        M=N,
        N=N,
        K=N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, C)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output gating and LayerNorm
    # --------------------------------------------------------------
    out = out * out_gate  # element‑wise scaling

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # --------------------------------------------------------------
    # 7. Final linear projection back to dim
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    return out.to(torch.float32)
