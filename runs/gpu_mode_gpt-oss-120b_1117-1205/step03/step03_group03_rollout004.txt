@triton.jit
def layernorm_kernel(x_ptr, out_ptr, gamma_ptr, beta_ptr, eps, rows, dim, stride_x_row, stride_x_feature, stride_out_row, stride_out_feature, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row = pid
    mask = row < rows

    # offsets for features
    offs = tl.arange(0, BLOCK_D)
    feature_mask = offs < dim

    # load x
    x = tl.load(x_ptr + row * stride_x_row + offs * stride_x_feature, mask=mask & feature_mask, other=0.0)

    # compute mean
    mean = tl.sum(x, axis=0) / dim

    # compute var
    diff = x - mean
    var = tl.sum(diff * diff, axis=0) / dim
    inv_std = 1.0 / tl.sqrt(var + eps)

    # normalize
    y = (x - mean) * inv_std

    # apply gamma,beta
    gamma = tl.load(gamma_ptr + offs * 1, mask=feature_mask, other=1.0)
    beta = tl.load(beta_ptr + offs * 1, mask=feature_mask, other=0.0)
    y = y * gamma + beta

    # store
    tl.store(out_ptr + row * stride_out_row + offs * stride_out_feature, y, mask=mask & feature_mask)

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements: tl.int32,
    hidden_dim: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # load
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # Compute mask index: each mask entry corresponds to hidden_dim consecutive channels
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

"""Trion (TriMul) outgoing kernel.

Implements the forward pass of the TriMul module (AlphaFold3 outgoing variant) using
torch operations for the bulk of the computation and two Triton kernels:
1) a fused kernel that multiplies the projected tensors with their gates and
   optionally with a mask;
2) (optional) a kernel for layer‑norm could be added, but the current implementation
   leverages torch.nn.functional.layer_norm for simplicity.

The kernel operates on a 4‑D input tensor of shape [B, N, N, C] and produces
an output tensor of the same shape. Mixed‑precision (float16) is used for the
majority of the work, and the final result is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse optional mask with left/right gating
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # half*  (flattened left)
    right_ptr,           # half*  (flattened right)
    left_gate_ptr,       # half*  (flattened left_gate)
    right_gate_ptr,      # half*  (flattened right_gate)
    mask_ptr,            # half*  (flattened mask, broadcasted)
    total_elements: tl.int32,
    hidden_dim: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    in_bounds = offs < total_elements

    # Load data
    left = tl.load(left_ptr + offs, mask=in_bounds, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=in_bounds, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=in_bounds, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=in_bounds, other=tl.float16(0.0))

    if HAS_MASK:
        # Each mask element expands over `hidden_dim` consecutive channel entries.
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=in_bounds, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Write the results back in‑place
    tl.store(left_ptr + offs, left, mask=in_bounds)
    tl.store(right_ptr + offs, right, mask=in_bounds)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """In‑place fusion of left/right tensors with their gates and optional mask.

    All tensors must be contiguous `torch.float16`. `mask` can be ``None`` when
    ``HAS_MASK`` is false. The function overwrites ``left`` and ``right``.
    """
    total = left.numel()
    BLOCK = 4096  # reasonable block size for 16‑bit data
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    # When there is no mask we still need to pass a pointer; we reuse `left`
    mask_ptr = mask if mask is not None else left

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        hidden_dim,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )
    # No return – tensors are modified in place.


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """Forward pass of the outgoing TriMul operator.

    Expected ``data`` format:
        (input_tensor, mask, weights_dict, config_dict)

    - ``input_tensor``: torch.Tensor of shape [B, N, N, C] (float32)
    - ``mask``: torch.Tensor of shape [B, N, N] (bool/float) or ``None``
    - ``weights_dict``: mapping from parameter names to tensors (float32)
    - ``config_dict``: contains at least ``dim`` and ``hidden_dim``; optional
      ``nomask`` boolean (True → ignore mask).

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Cast all model parameters to half precision once
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias = weights["norm.bias"].to(torch.float16)

    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)

    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)

    to_out_weight = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> float16)
    # ------------------------------------------------------------------
    # Using torch's layer_norm for numerical stability; result cast to half.
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Gating tensors (half‑precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    # ------------------------------------------------------------------
    # 4. Optional mask – fuse with gating via Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # Convert mask to half and flatten for kernel usage
        mask_h = mask.to(torch.float16).contiguous()
        # Ensure mask has shape [B, N, N]
        mask_flat = mask_h.view(-1)                     # (B*N*N,)

        # Flatten the tensors that the kernel will work on
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)

        # In‑place fusion kernel
        _apply_gate_and_mask(
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            hidden_dim,
        )

        # Restore original shapes
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask – simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched matrix multiplication (cuBLAS).
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden_dim, N, N) layout for GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) – transposed on seq dims

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Batched matrix multiplication (float16 inputs, float32 accumulation)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)

    # Reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)

    # ------------------------------------------------------------------
    # 7. Cast back to float32 for the final output
    # ------------------------------------------------------------------
    return out.to(torch.float32)
