```python
@triton.jit
def fuse_gate_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements: tl.int32,
    H: tl.constexpr,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
   pid = tl.program_id(0)
   offs = pid * BLOCK + tl.arange(0, BLOCK)
   mask = offs < total_elements

   left = tl.load(left_ptr + offs, mask=mask, other=0.0)
   right = tl.load(right_ptr + offs, mask=mask, other=0.0)
   left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
   right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
   left = left * left_gate
   right = right * right_gate
   if HAS_MASK:
       mask_idx = offs // H
       mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
       left = left * mask_val
       right = right * mask_val
   tl.store(left_ptr + offs, left, mask=mask)
   tl.store(right_ptr + offs, right, mask=mask)

"""
TriMul (outgoing) forward pass with Triton acceleration.

The kernel fuses element‑wise gating (and optional masking) into the projected
tensors and performs the core O(N³) contraction via a custom batched GEMM
implemented in Triton. All heavy arithmetic is done in float16 with fp32
accumulation; the final result is returned in float32.

Steps
-----
1. Layer‑norm on the input (float32 → float16).
2. Linear projections to hidden dimension (left/right).
3. Linear gates (left_gate, right_gate, out_gate) with sigmoid.
4. Fuse gates (and mask, if present) into left/right in‑place via a Triton
   kernel.
5. Contract left and right:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   realized as a batched matrix multiplication using a Triton GEMM kernel.
6. Output layer‑norm, element‑wise out‑gate, final linear projection.
7. Cast to float32 and return.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1. Gate + optional mask fusion
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,                # half*
    right_ptr,               # half*
    left_gate_ptr,           # half*
    right_gate_ptr,          # half*
    mask_ptr,                # half* (only used if HAS_MASK==1)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,  # 1 if mask present, 0 otherwise
    H: tl.constexpr,         # hidden dimension (known at compile time)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # apply gates
    left = left * left_gate
    right = right * right_gate

    if HAS_MASK:
        # each mask element corresponds to H consecutive hidden entries
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * mask_val
        right = right * mask_val

    # write back
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of left/right with their gates and (optionally) a mask.
    All tensors must be torch.float16, contiguous and on the same device.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 0 if mask is None else 1
    # dummy pointer when no mask is used – Triton will not read it
    mask_ptr = mask if mask is not None else left

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )


# ----------------------------------------------------------------------
# 2. Batched GEMM (A @ B) across the (batch * hidden) dimension
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch: tl.int32, M: tl.int32, N: tl.int32, K: tl.int32,
    stride_abh: tl.int32, stride_abl: tl.int32, stride_abk: tl.int32,
    stride_bbh: tl.int32, stride_bbk: tl.int32, stride_bbn: tl.int32,
    stride_cbh: tl.int32, stride_cbm: tl.int32, stride_cbn: tl.int32,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_b = tl.program_id(2)    # batch * hidden index
    pid_m = tl.program_id(0)    # row block
    pid_n = tl.program_id(1)    # col block

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds
    m_mask = offs_m < M
    n_mask = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_off < K

        # A block: [M, K]
        a_ptrs = (
            a_ptr
            + pid_b * stride_abh
            + offs_m[:, None] * stride_abl
            + k_off[None, :] * stride_abk
        )
        a = tl.load(a_ptrs, mask=m_mask[:, None] & k_mask[None, :], other=0.0)

        # B block: [K, N]
        b_ptrs = (
            b_ptr
            + pid_b * stride_bbh
            + k_off[:, None] * stride_bbk
            + offs_n[None, :] * stride_bbn
        )
        b = tl.load(b_ptrs, mask=k_mask[:, None] & n_mask[None, :], other=0.0)

        # accumulate
        acc += tl.dot(a, b)    # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # write back result
    c_ptrs = (
        c_ptr
        + pid_b * stride_cbh
        + offs_m[:, None] * stride_cbm
        + offs_n[None, :] * stride_cbn
    )
    tl.store(c_ptrs, acc.to(tl.float16), mask=m_mask[:, None] & n_mask[None, :])


def triton_bmm(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication using Triton.
    A: (batch, M, K)   fp16
    B: (batch, K, N)   fp16
    Returns C: (batch, M, N) fp16
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty(batch, M, N, dtype=A.dtype, device=A.device)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch,
    )

    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        batch,
        M,
        N,
        K,
        A.stride(0),
        A.stride(1),
        A.stride(2),
        B.stride(0),
        B.stride(1),
        B.stride(2),
        C.stride(0),
        C.stride(1),
        C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Gating tensors (half)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse gates and (optional) mask via Triton
    # ------------------------------------------------------------------
    # flatten to 1‑D for the kernel
    left_flat = left.view(-1)
    right_flat = right.view(-1)
    left_gate_flat = left_gate.view(-1)
    right_gate_flat = right_gate.view(-1)

    if not nomask:
        # mask: [B, N, N] → expand over hidden dim
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        mask_flat = mask_h.view(-1)  # size B*N*N
    else:
        mask_flat = None

    _apply_gate_and_mask(
        left_flat,
        right_flat,
        left_gate_flat,
        right_gate_flat,
        mask_flat,
        hidden_dim,
    )

    # reshape back
    left = left_flat.view_as(left)
    right = right_flat.view_as(right)

    # ------------------------------------------------------------------
    # 5. Core O(N³) contraction via Triton‑accelerated batched GEMM
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # permute to (B, H, N, N) where H = hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) with j/k swapped

    B, H, N, _ = left_perm.shape
    # reshape to (B*H, N, N)
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    out_mat = triton_bmm(left_mat, right_mat)   # (B*H, N, N)

    # restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 7. Return float32
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
