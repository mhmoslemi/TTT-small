    total_batch, N,

"""
TriMul (outgoing) forward pass with Triton‑fused mask/gate and
batched matrix multiplication.

The implementation follows the reference PyTorch code but:
  1. Applies mask and gating (left/right) in a single Triton kernel.
  2. Performs the core contraction out[b,i,j,d] = Σ_k left[b,i,k,d] *
     right[b,j,k,d] as a batched half‑precision GEMM using a custom Triton
     kernel (L @ Rᵀ) with float32 accumulation.
All heavy operations run in fp16; the final result is returned in fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton kernel: fuse optional mask with left/right gating
# -------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # half* (flattened left)
    right_ptr,           # half* (flattened right)
    left_gate_ptr,       # half* (flattened left_gate)
    right_gate_ptr,      # half* (flattened right_gate)
    mask_ptr,            # half* (flattened mask, broadcasted)
    total_elems: tl.int32,
    hidden_dim: tl.int32,
    HAS_MASK: tl.constexpr,   # 1 if mask present, 0 otherwise
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # each mask entry corresponds to `hidden_dim` consecutive channel values
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # store back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

# Triton kernel: batched matmul for out = left @ rightᵀ
# -------------------------------------------------------------
@triton.jit
def _batched_trmul_kernel(
    left_ptr,   # half*  (B*H, N, N)
    right_ptr,  # half*  (B*H, N, N)
    out_ptr,    # half*  (B*H, N, N)
    total_batch: tl.int32,      # B*H
    N: tl.int32,
    stride_batch: tl.int32,     # N*N
    stride_i: tl.int32,         # N   (row stride)
    stride_k: tl.int32,         # 1   (col stride)
    out_stride_batch: tl.int32,
    out_stride_i: tl.int32,
    out_stride_j: tl.int32,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # number of blocks per dimension
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # decode batch, m, n indices
    batch_idx = pid // (num_blocks_m * num_blocks_n)
    pid_in_batch = pid % (num_blocks_m * num_blocks_n)
    pid_m = pid_in_batch // num_blocks_n
    pid_n = pid_in_batch % num_blocks_n

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # offset to the first element of this matrix (batch)
    batch_offset = batch_idx * stride_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_tiles_k = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_tiles_k):
        k_offset = k_tile * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + batch_offset + \
                 offs_m[:, None] * stride_i + \
                 offs_k[None, :] * stride_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=tl.float16(0.0))

        # right tile: (BLOCK_N, BLOCK_K)  (note: we need Rᵀ, i.e. rows=j)
        b_ptrs = right_ptr + batch_offset + \
                 offs_n[:, None] * stride_i + \
                 offs_k[None, :] * stride_k
        b = tl.load(b_ptrs,
                    mask=mask_n[:, None] & mask_k[None, :],
                    other=tl.float16(0.0))

        acc += tl.dot(a, b)   # (M,K)·(N,K) -> (M,N)

    # write back
    c = acc.to(tl.float16)
    out_ptrs = out_ptr + batch_offset + \
               offs_m[:, None] * out_stride_i + \
               offs_n[None, :] * out_stride_j
    tl.store(out_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_trmul(left, right, hidden_dim):
    """
    Compute out = left @ rightᵀ for tensors of shape (B, N, N, hidden_dim)
    using a custom Triton kernel. Returns tensor of shape (B, N, N, hidden_dim)
    in half precision.
    """
    B, N, _, H = left.shape  # left shape [B, N, N, H]
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)
    out_mat = torch.empty_like(left_mat)

    total_batch = B * H
    stride_batch = N * N
    stride_i = N
    stride_k = 1
    out_stride_batch = N * N
    out_stride_i = N
    out_stride_j = 1

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (total_batch * triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N),)
    _batched_trmul_kernel[grid](
        left_mat, right_mat, out_mat,
        total_batch, N,
        stride_batch, stride_i, stride_k,
        out_stride_batch, out_stride_i, out_stride_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

    hidden_dim = config["hidden_dim"]

        input_tensor,

    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Gating tensors (half precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) and gate multiplications via Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        # flatten tensors for the kernel
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_h.view(-1)  # [B*N*N]

        _apply_gate_and_mask(
            left_flat, right_flat,
            left_gate_flat, right_gate_flat,
            mask_flat,
            hidden_dim,
        )
        # reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask: simple elementwise multiplication
        left = left * left_gate
        right = right * right_gate

    # -------------------------------------------------------------
    # 5. Core contraction: out = left @ rightᵀ  (batched GEMM)
    # -------------------------------------------------------------
    out = _batched_trmul(left, right, hidden_dim)  # shape [B, N, N, hidden_dim]

    # -------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # -------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise gating (half)
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # -------------------------------------------------------------
    # 7. Return in float32 as required
    # -------------------------------------------------------------
    return out.to(torch.float32)
```
