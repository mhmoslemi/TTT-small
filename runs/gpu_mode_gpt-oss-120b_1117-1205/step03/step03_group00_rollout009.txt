```python
@triton.jit
def gate_mask_fuse_kernel(
    left_ptr,
    right_ptr,
    left_gate_ptr,
    right_gate_ptr,
    mask_ptr,
    total_elements,
    hidden_dim: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)  # shape (BLOCK,)
    mask = offs < total_elements
    # Load
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    if HAS_MASK:
        mask_offs = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_offs, mask=mask, other=tl.float16(1.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)
```

"""
TriMul outgoing operator forward pass.
- LayerNorm on input.
- Linear projections → left, right.
- Sigmoid gates.
- Fuse mask (if present) and gates using a Triton kernel (in-place).
- Batched GEMM across hidden dimension (N x N matmul per hidden channel) using torch.bmm on half.
- Output LayerNorm, out_gate scaling, final linear.
All heavy elementwise ops are fused in the Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def gate_mask_fuse_kernel(
    left_ptr,            # * half
    right_ptr,           # * half
    left_gate_ptr,       # * half
    right_gate_ptr,      # * half
    mask_ptr,            # * half (broadcasted)
    total_elements,      # int32
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    left = tl.load(left_ptr + offs, mask=valid, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=valid, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=tl.float16(0.0))

    if HAS_MASK:
        mask_idx = offs // HIDDEN_DIM
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=tl.float16(1.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    `mask` may be None (no masking). All tensors must be contiguous
    float16.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy

    gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        HIDDEN_DIM=hidden_dim,
    )
    # No explicit return; operation performed in‑place


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
        - mask: torch.Tensor of shape [B, N, N] (bool / float) or None
        - weights: dict of torch.Tensors (float32)
        - config: dict containing at least "dim", "hidden_dim", optionally "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input layer‑norm (float32) → half precision
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half)
    # ------------------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gates (half)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) with gates using Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # Convert mask to half and flatten
        mask_h = mask.to(torch.float16).contiguous().view(-1)  # [B*N*N]
        # Ensure tensors are contiguous before flattening
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_h,
            hidden_dim,
        )
        # Reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # Simple element‑wise gating when no mask
        left = left * left_gate
        right = right * right_gate

    # Free gates that are no longer needed
    del left_gate, right_gate, x_h

    # ------------------------------------------------------------------
    # 5. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as batched GEMM (torch.bmm) on half‑precision tensors.
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden_dim, N, N) for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    # Collapse batch & hidden dimensions
    left_mat = left_perm.view(B * H, N, N)               # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)             # (B*H, N, N)

    # Batched GEMM with transposition on the second operand
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise scaling (half precision)
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in original dtype (float32)
    return out.to(torch.float32)

"""
Outgoing TriMul operator (AlphaFold3) forward pass.

Algorithm
---------
1. Layer‑norm the input tensor (float32 → float16).
2. Linear projections → left/right (half‑precision).
3. Compute three sigmoid gates (half‑precision).
4. Fuse mask (if present) and the left/right gates in a Triton kernel
   (in‑place element‑wise multiplication, broadcast mask over hidden dim).
5. Core O(N³) contraction:
   out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   Implemented as a batched GEMM (torch.bmm) on half tensors:
   for each hidden channel d we perform a matmul left_d @ right_dᵀ.
6. Output layer‑norm, multiply by out‑gate and final linear projection.
   The result is returned as float32.

Only the mask‑gate fusion is performed in Triton; the heavy matrix
multiplication uses highly tuned cuBLAS via torch.bmm.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def gate_mask_fuse_kernel(
    left_ptr,          # * half
    right_ptr,         # * half
    left_gate_ptr,     # * half
    right_gate_ptr,    # * half
    mask_ptr,          # * half (broadcasted)
    total_elements,    # int32
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):

    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=valid, other=tl.float16(0.0))

    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=tl.float16(0.0))

        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=tl.float16(1.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val

        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)

    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Compute gates (half)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) with gates via Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # Prepare mask
        mask_h = mask.to(torch.float16).contiguous().view(-1)   # [B*N*N]

        # Ensure contiguous tensors before flattening
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_h,
            hidden_dim,
        )
        # Reshape back to original layout
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask – simple element‑wise gating
        left = left * left_gate
        right = right * right_gate

    # Release temporary tensors
    del left_gate, right_gate, x_h

    # ------------------------------------------------------------------
    # 5. Core contraction via batched GEMM (half precision)
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden_dim, N, N) for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)               # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)             # (B*H, N, N)

    # Batched GEMM: left @ rightᵀ  (each hidden channel independent)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore (B, N, N, H) layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate                     # element‑wise scaling (half)
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in original dtype (float32)
    return out.to(torch.float32)
```
