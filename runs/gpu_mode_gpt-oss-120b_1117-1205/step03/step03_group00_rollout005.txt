    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # Cast to half
    x_h = x.to(torch.float16)

    # Linear projections
    left = torch.nn.functional.linear(x_h, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = torch.nn.functional.linear(x_h, weights["right_proj.weight"].to(torch.float16), bias=None)

    # Gates
    left_gate = torch.nn.functional.linear(x_h, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = torch.nn.functional.linear(x_h, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = torch.nn.functional.linear(x_h, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)

        right_gate_f = right_gate.contiguous().view(-1)
        total = left_f.numel()

        grid = (triton.cdiv(total, BLOCK), )
        HAS_MASK = 1

            left_f,
            right_f,
            left_gate_f,
            right_gate_f,

            HAS_MASK=HAS_MASK,
            HIDDEN_DIM=hidden_dim,
        )

        left = left_f.view_as(left)
        right = right_f.view_as(right)

        right = right * right_gate

    # Clean some intermediate tensors to free memory (optional)
    del left_gate, right_gate, x_h

    # Contraction via batched GEMM
    # Permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

        bias=weights["to_out_norm.bias"].to(torch.float16)

    # Final linear projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    # Convert to float32

"""
Optimized TriMul forward pass (outgoing) using half‑precision arithmetic.
- LayerNorm, linear projections and gates are performed in PyTorch (float16).
- A custom Triton kernel fuses element‑wise mask and sigmoid gates for `left`
  and `right` tensors, operating in‑place.
- The O(N³) contraction is realized as a batched GEMM (`torch.bmm`) after
  reshaping, which maps to cuBLAS.
- Output LayerNorm, out‑gate scaling and final linear projection complete the
  forward pass. The result is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # * half
    right_ptr,         # * half
    left_gate_ptr,     # * half
    right_gate_ptr,    # * half
    mask_ptr,          # * half (broadcast)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,   # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    # Load values
    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        mask_idx = offs // HIDDEN_DIM
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)

def _apply_gate_mask_fuse(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fuse of mask and gates using Triton.
    `mask` may be None (no masking). All tensors must be contiguous half‑precision.
    """
    total = left.numel()
    BLOCK = 32768  # 32K threads per block (multiple of 32)
    grid = (triton.cdiv(total, BLOCK),)
    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when no mask

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        HIDDEN_DIM=hidden_dim,
    )
    # No return, tensors are modified in‑place

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor of shape [B, N, N] (bool or float)
                             May be ignored when config["nomask"] is True.
            - weights: dict with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict containing at least "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> half)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half)
    # ------------------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gating values (half, sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) with gates using Triton
    # ------------------------------------------------------------------
    if not nomask:
        # Flatten mask to 1‑D half tensor
        mask_h = mask.to(torch.float16).contiguous().view(-1)  # shape [B*N*N]
        # Ensure tensors are contiguous before flattening
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)

Optimized TriMul forward pass (outgoing) using half‑precision arithmetic.

- LayerNorm, linear projections, and sigmoid gates are performed in PyTorch
  (float16) for speed.
- A custom Triton kernel fuses element‑wise mask and gating for the `left` and
  `right` tensors, operating in‑place.
- The O(N³) contraction is expressed as a batched GEMM via `torch.bmm`
  after reshaping, which maps to highly tuned cuBLAS kernels.
- Output LayerNorm, out‑gate scaling, and the final linear projection complete
  the forward pass. The result is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # * half
    right_ptr,         # * half
    left_gate_ptr,     # * half
    right_gate_ptr,    # * half
    mask_ptr,          # * half (broadcast)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,   # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    # Load values
    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        mask_idx = offs // HIDDEN_DIM
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)

    In‑place fuse of mask and gates using Triton.

        right,
        left_gate,
        right_gate,
        mask_ptr,

        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        HIDDEN_DIM=hidden_dim,
