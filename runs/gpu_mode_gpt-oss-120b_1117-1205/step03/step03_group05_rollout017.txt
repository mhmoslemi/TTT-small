```
def custom_kernel(data):
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    
    # cast weights to half once using .to(torch.float16, non_blocking=True)
    # maybe store them in a dict.
    
    # LayerNorm
    x = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x_h = x.to(torch.float16)

    # Flatten for linear
    B,N,N,_ = x_h.shape
    x_flat = x_h.view(-1, dim)   # shape (B*N*N, dim)
    
    # Linear projections
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)
    left = F.linear(x_flat, left_proj_weight).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, right_proj_weight).view(B, N, N, hidden_dim)

    # Gating
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    left_gate = torch.sigmoid(F.linear(x_flat, left_gate_weight)).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(x_flat, right_gate_weight)).view(B, N, N, hidden_dim)
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_weight)).view(B, N, N, hidden_dim)

    # Mask & gate fuse
    if not nomask:
        mask_h = mask.to(torch.float16).unsqueeze(-1)
        # flatten all
        left_flat = left.view(-1)
        right_flat = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)
        mask_flat = mask_h.view(-1)  # [B,N,N]
        total = left_flat.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)
        _gate_mask_fuse_kernel[grid](
            left_flat, right_flat,
            left_gate_flat, right_gate_flat,
            mask_flat,
            total,
            BLOCK=BLOCK, HAS_MASK=1, H=hidden_dim,
        )
        # reshape
        left = left_flat.view(B, N, N, hidden_dim)
        right = right_flat.view(B, N, N, hidden_dim)
    else:
        # No mask: use kernel with HAS_MASK=0 for memory efficiency?
        # We'll just do elementwise multiplication in PyTorch: left = left * left_gate; right = right * right_gate
        left = left * left_gate
        right = right * right_gate
    # done gating
    del left_gate, right_gate

    # Core contraction
    # left_perm: B, H, N, N
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()
    B_h, H, N, _ = left_perm.shape # Actually left_perm is shape (B, H, N, N)
    # flatten batch and hidden into one dimension
    left_mat = left_perm.view(B*H, N, N)
    right_mat = right_perm.view(B*H, N, N)
    # bmm in half precision
    out_mat = torch.bmm(left_mat, right_mat)  # shape (B*H, N, N)
    # reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # shape (B, N, N, H)

    # Output LayerNorm
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    # apply out_gate
    out = out * out_gate
    # final linear projection
    to_out_weight = weights["to_out.weight"].to(torch.float16)  # shape (dim, hidden_dim)
    out_flat = out.view(-1, hidden_dim)
    out = F.linear(out_flat, to_out_weight).view(B,N,N,dim)
    return out.to(torch.float32)
```

import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(left_ptr, right_ptr,
                          left_gate_ptr, right_gate_ptr,
                          mask_ptr,
                          total_elements: tl.int32,
                          BLOCK: tl.constexpr,
                          HAS_MASK: tl.constexpr,
                          H: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    left = left * left_gate
    right = right * right_gate
    if HAS_MASK:
        # compute masked index (each mask broadcast across H)
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * mask_val
        right = right * mask_val
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)
```

"""
Optimized outgoing TriMul forward pass.

Algorithm:
1. LayerNorm on input (float32) → half.
2. Linear projections (half) to hidden dimension `hidden_dim`.
3. Compute three gating vectors (half) via linear + sigmoid.
4. Fuse left/right gates and optional mask in a single Triton kernel
   (in‑place multiplication).
5. Core contraction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched matrix multiplication using torch.matmul
   on tensors (B, hidden_dim, N, N).
6. Output LayerNorm (half), multiply by out_gate, final linear projection
   back to `dim`.
7. Return result in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half*  (flattened left tensor)
    right_ptr,         # half*  (flattened right tensor)
    left_gate_ptr,     # half*  (flattened left_gate)
    right_gate_ptr,    # half*  (flattened right_gate)
    mask_ptr,          # half*  (flattened mask, length = B*N*N) – read only if HAS_MASK=1
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # compile‑time flag
    H: tl.constexpr,          # hidden dimension
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    # load tensors
    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    # apply gates
    left = left * left_gate
    right = right * right_gate

    if HAS_MASK:
        # each mask value is broadcast over H hidden channels
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=0.0)
        left = left * mask_val
        right = right * mask_val

    # write results back
    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (bool or numeric) – may be ignored
            - weights: dict with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict with entries "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # unpack
    inp, mask, weights, config = data
    device = inp.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) -> half
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    B, N, _, _ = x_h.shape
    # flatten for linear layers
    x_flat = x_h.reshape(-1, dim)  # (B*N*N, dim)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_flat, left_proj_w).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, right_proj_w).view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3. Gating tensors (half)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_flat, left_gate_w)).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(x_flat, right_gate_w)).view(B, N, N, hidden_dim)
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_w)).view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) and gates via Triton (in‑place)
    # ------------------------------------------------------------------
    if not nomask:
        # mask broadcast over hidden dimension
        mask_h = mask.to(torch.float16).unsqueeze(-1).contiguous()  # (B,N,N,1)

        # flatten all tensors
        left_flat = left.reshape(-1)
        right_flat = right.reshape(-1)
        left_gate_flat = left_gate.reshape(-1)
        right_gate_flat = right_gate.reshape(-1)
        mask_flat = mask_h.reshape(-1)  # length B*N*N

        total_elems = left_flat.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total_elems, BLOCK),)

        _gate_mask_fuse_kernel[grid](
            left_flat, right_flat,
            left_gate_flat, right_gate_flat,
            mask_flat,
            total_elems,
            BLOCK=BLOCK,
            HAS_MASK=1,
            H=hidden_dim,
        )

        # reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask – still fuse gates via a tiny kernel for consistency
        left_flat = left.reshape(-1)
        right_flat = right.reshape(-1)
        left_gate_flat = left_gate.reshape(-1)
        right_gate_flat = right_gate.reshape(-1)

        total_elems = left_flat.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total_elems, BLOCK),)

        _gate_mask_fuse_kernel[grid](
            left_flat, right_flat,
            left_gate_flat, right_gate_flat,
            left_flat,  # dummy mask pointer
            total_elems,
            BLOCK=BLOCK,
            HAS_MASK=0,
            H=hidden_dim,
        )

        left = left_flat.view_as(left)
        right = right_flat.view_as(right)

    # Release gates that are no longer needed
    del left_gate, right_gate, x_h, x_flat

    # ------------------------------------------------------------------
    # 5. Core O(N³) contraction via batched matrix multiplication
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # bring hidden dimension to batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    # batched matmul over (N x N) matrices for each (b,h)
    out = torch.matmul(left_perm, right_perm)  # (B, H, N, N)
    out = out.permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate  # element‑wise gating (half)

    to_out_w = weights["to_out.weight"].to(torch.float16)  # (dim, hidden_dim)
    out = F.linear(out.reshape(-1, hidden_dim), to_out_w).view(B, N, N, dim)

    # ------------------------------------------------------------------
    # 7. Return float32 tensor
    # ------------------------------------------------------------------
    return out.to(torch.float32)
