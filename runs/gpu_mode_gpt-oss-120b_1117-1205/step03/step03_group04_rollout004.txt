    # A, B shape (batch, M, K), (batch, K, N)
    # return C shape (batch, M, N)

@triton.jit
def trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_Abatch, stride_Am, stride_Ak,
    stride_Bbatch, stride_Bk, stride_Bn,
    stride_Cbatch, stride_Cm, stride_Cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Load A and B tiles
        a_ptrs = A_ptr + pid_batch * stride_Abatch + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
        b_ptrs = B_ptr + pid_batch * stride_Bbatch + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

        # Apply masks for out-of-bounds
        a_mask = (offs_m < M)[:, None] & (offs_k < K)[None, :]
        b_mask = (offs_k < K)[:, None] & (offs_n < N)[None, :]

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        acc += tl.dot(a, b)

    # Write back to C
    c_ptrs = C_ptr + pid_batch * stride_Cbatch + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    c_mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Normalization
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # cast to half
    x_h = x.to(torch.float16)

    # Projections
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # Gates
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    if not nomask:
        # mask shape (B,N,N)
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_h
        right = right * mask_h

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Prepare for batched matmul
    # left: (B,N,N,H) -> (B,H,N,N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # transpose k & j

    B, H, N, _ = left_perm.shape
    batch = B * H
    left_flat = left_perm.view(batch, N, N)
    right_flat = right_perm.view(batch, N, N)

    # Batched matmul with Triton
    out_flat = triton_batched_matmul(left_flat, right_flat, BLOCK_M=128, BLOCK_N=128, BLOCK_K=32)

    # Reshape back to [B,N,N,H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output LayerNorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    # Multiply out gate
    out = out * out_gate

    # Final projection
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return float32
    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass implemented with a custom Triton batched
matrix‑multiply kernel.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Linear projections → left/right tensors of shape [B, N, N, H].
3. Compute three gating tensors (left_gate, right_gate, out_gate) via
   linear + sigmoid.
4. Apply optional pairwise mask and gates (element‑wise).
5. Compute the core contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication of left with the transpose of
   right across the hidden dimension H.  This is performed by a Triton
   kernel that processes a batch of (N×N) × (N×N) multiplies in a single
   launch, accumulating in fp32 and storing as fp16/fp32.
6. Apply the output layer‑norm, multiply with out_gate, and final linear
   projection back to the original dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel for batched matrix multiplication (A @ B)
# --------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_batch = tl.program_id(0)   # batch index (B*H)
    pid_m = tl.program_id(1)       # block row
    pid_n = tl.program_id(2)       # block column

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # pointers for the current tile
        a_ptrs = (
            A_ptr
            + pid_batch * stride_Ab
            + offs_m[:, None] * stride_Am
            + offs_k[None, :] * stride_Ak
        )
        b_ptrs = (
            B_ptr
            + pid_batch * stride_Bb
            + offs_k[:, None] * stride_Bk
            + offs_n[None, :] * stride_Bn
        )

        # masks for out‑of‑bounds loads
        a_mask = (offs_m < M)[:, None] & (offs_k < K)[None, :]
        b_mask = (offs_k < K)[:, None] & (offs_n < N)[None, :]

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # dot‑product accumulation
        acc += tl.dot(a, b)

    # write back
    c_ptrs = (
        C_ptr
        + pid_batch * stride_Cb
        + offs_m[:, None] * stride_Cm
        + offs_n[None, :] * stride_Cn
    )
    c_mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def _batched_matmul(A: torch.Tensor, B: torch.Tensor,
                    BLOCK_M=128, BLOCK_N=128, BLOCK_K=32) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B with a single Triton launch.
    A: (batch, M, K)  fp16/fp32
    B: (batch, K, N)
    Returns C: (batch, M, N) with same dtype as A.
    """
    assert A.shape[0] == B.shape[0], "batch dimension must match"
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, "inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # strides
    stride_Ab, stride_Am, stride_Ak = A.stride()
    stride_Bb, stride_Bk, stride_Bn = B.stride()
    stride_Cb, stride_Cm, stride_Cn = C.stride()

    grid = (batch,
            triton.cdiv(M, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    _batched_matmul_kernel[grid](
        A, B, C,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        M, N, K,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# --------------------------------------------------------------
# Entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/int) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (float32)
    # ------------------------------------------------------------------
    left = F.linear(
        x, weights["left_proj.weight"]
    )  # [B, N, N, hidden_dim]
    right = F.linear(
        x, weights["right_proj.weight"]
    )  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Gate values (float32)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 4. Optional mask + gating (element‑wise, float32)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] -> broadcast over hidden dimension
        mask_f = mask.to(torch.float32).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: batched matmul via Triton
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) and flatten batch*H
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N), transposed k↔j

    B, H, N, _ = left_perm.shape
    batch = B * H
    left_flat = left_perm.view(batch, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(batch, N, N)  # (B*H, N, N)

    out_flat = _batched_matmul(left_flat, right_flat)  # (B*H, N, N)

    # Restore shape [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    # Return in float32 as required
    return out.to(torch.float32)
