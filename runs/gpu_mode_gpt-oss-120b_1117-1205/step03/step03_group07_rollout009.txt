@triton.jit
def gate_mask_kernel(left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr, total_elems, H, BLOCK: tl.constexpr, HAS_MASK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask)
    right = tl.load(right_ptr + offs, mask=mask)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask)
    if HAS_MASK:
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)
```

"""
TriMul (outgoing) forward pass – highly‑optimized implementation.

The kernel fuses the left/right linear projections with their gating and an
optional mask in a single elementwise Triton kernel.  The O(N³) contraction is
performed as a batched GEMM (torch.bmm) on the half‑precision tensors, which
leverages Tensor‑Core kernels.  All heavy linear algebra stays in FP16 while
LayerNorms run in FP32 for numerical stability.  The final output is cast back
to FP32 as required by the API.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse left/right gating and optional mask
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half *
    right_ptr,         # half *
    left_gate_ptr,     # half *
    right_gate_ptr,    # half *
    mask_ptr,          # half *
    total_elems: tl.int32,
    hidden_dim: tl.int32,          # runtime scalar (size of innermost channel)
    BLOCK: tl.constexpr,           # threads per program
    HAS_MASK: tl.constexpr,        # 1 if mask is present, 0 otherwise
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # Load operands
    left      = tl.load(left_ptr + offs,      mask=mask, other=tl.float16(0.0))
    right     = tl.load(right_ptr + offs,     mask=mask, other=tl.float16(0.0))
    left_g    = tl.load(left_gate_ptr + offs,mask=mask, other=tl.float16(0.0))
    right_g   = tl.load(right_gate_ptr + offs,mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # each mask element repeats `hidden_dim` times in the flattened view
        mask_idx = offs // hidden_dim
        mval = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left  = left  * left_g  * mval
        right = right * right_g * mval
    else:
        left  = left  * left_g
        right = right * right_g

    # Store results back in‑place
    tl.store(left_ptr + offs,  left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim, has_mask):
    """
    In‑place fusion of gating (and optional mask) using a Triton kernel.
    All tensors must be contiguous, half‑precision and flattened (1‑D).
    """
    total = left.numel()
    BLOCK = 8192                       # large 1‑D block (multiple of 32)
    grid = (triton.cdiv(total, BLOCK),)

    # The mask pointer can be any tensor when has_mask==0; it will never be dereferenced.
    mask_ptr = mask if has_mask else left

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        hidden_dim,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
    )
    # no return – operation is in‑place

        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Cast to half‑precision for the heavy work
    # --------------------------------------------------------------
    x_h = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Linear projections (half)
    # --------------------------------------------------------------
    left  = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 4. Gates (half)
    # --------------------------------------------------------------
    left_gate  = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate   = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # --------------------------------------------------------------
    # 5. Fuse gating (and optional mask) in a Triton kernel
    # --------------------------------------------------------------
    # Ensure contiguity and flatten for the kernel
    left_f      = left.contiguous().view(-1)
    right_f     = right.contiguous().view(-1)
    left_gate_f = left_gate.contiguous().view(-1)
    right_gate_f = right_gate.contiguous().view(-1)

    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).contiguous()
        has_mask = 1
        mask_f = mask_h.view(-1)   # shape [B*N*N]
    else:
        has_mask = 0
        # dummy mask (will not be read)
        mask_f = left_f

    _apply_gate_and_mask(
        left_f,
        right_f,
        left_gate_f,
        right_gate_f,
        mask_f,
        hidden_dim,
        has_mask,
    )

    # Reshape back to original layout
    left  = left_f.view_as(left)
    right = right_f.view_as(right)

    # --------------------------------------------------------------
    # 6. Core N³ contraction via batched GEMM:
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N) and right transposed on the last two axes
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat  = left_perm.reshape(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.reshape(B * H, N, N)  # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)    # (B*H, N, N)

    # Restore (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return float32 as required by the API
    return out.to(torch.float32)
```
