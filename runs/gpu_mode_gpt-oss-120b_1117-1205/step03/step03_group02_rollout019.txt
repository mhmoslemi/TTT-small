def tri_mul_outgoing_kernel(left_ptr, right_ptr, out_ptr,
                            B, H, N,
                            stride_left_b, stride_left_i, stride_left_k, stride_left_h,
                            stride_right_b, stride_right_i, stride_right_k, stride_right_h,
                            stride_out_b, stride_out_i, stride_out_j, stride_out_h,
                            BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_bh = tl.program_id(0)   # batchhead index (B*H)
    pid_i = tl.program_id(1)    # block row
    pid_j = tl.program_id(2)    # block col

    # compute batchhead indices
    total_heads = B * H
    # total heads dimension is known only as integer; we compute offsets
    # compute b and h
    b = pid_bh // H
    h = pid_bh % H

    # compute base offsets
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # compute start indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # offsets vectors
    i_offsets = i_start + tl.arange(0, BLOCK_M)
    j_offsets = j_start + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds in i, j
    mask_i = i_offsets < N
    mask_j = j_offsets < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, N, BLOCK_K):
        k_start = k
        k_offsets = k_start + tl.arange(0, BLOCK_K)
        # mask for k
        mask_k = k_offsets < N

        # A block: left[i,k,h]
        # A_ptrs shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + i_offsets[:, None] * stride_left_i + k_offsets[None, :] * stride_left_k
        # B block: right_transpose (k,j,h)
        b_ptrs = right_base + k_offsets[:, None] * stride_right_k + j_offsets[None, :] * stride_right_i

        # load with masking
        a = tl.load(a_ptrs, mask=(mask_i[:, None] & mask_k[None, :]), other=0.0)
        b = tl.load(b_ptrs, mask=(mask_k[:, None] & mask_j[None, :]), other=0.0)

        # accumulate
        acc += tl.dot(a, b)

    # write back
    out_ptrs = out_base + i_offsets[:, None] * stride_out_i + j_offsets[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=(mask_i[:, None] & mask_j[None, :]))

"""
TriMul (outgoing) forward pass with Triton fusion.

Algorithm
---------
1. Layer‑norm the input and cast to float16.
2. Linear projections (left/right) from dim → hidden_dim (half‑precision).
3. Compute left/right/out gates (sigmoid of a linear).
4. Fuse mask (if present) and gate multiplications into left/right in‑place
   with a tiny Triton kernel.
5. Core O(N³) contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched GEMM (left @ rightᵀ) in Triton, avoiding any
   permutes or extra copies.
6. Output LayerNorm over the hidden dimension, multiply by the out‑gate,
   and a final linear projection back to dim.
7. Return float32 output.

All heavy compute (mask‑gate fusion + the N³ contraction) runs on Triton.
The remaining ops (layer‑norms, linear layers, sigmoid) use PyTorch which
already maps to highly‑optimized CUDA kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1. Mask & gate fusion kernel (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # [total] half
    right_ptr,              # [total] half
    left_gate_ptr,          # [total] half
    right_gate_ptr,         # [total] half
    mask_ptr,               # [mask_total] half (broadcasted)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,    # 1 if mask present, else 0
    H: tl.constexpr,           # hidden_dim
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elements

    # load values
    left = tl.load(left_ptr + offs, mask=m, other=0.0)
    right = tl.load(right_ptr + offs, mask=m, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=m, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=m, other=0.0)

    if HAS_MASK:
        # broadcast mask over the hidden dimension
        mask_index = offs // H
        mask_val = tl.load(mask_ptr + mask_index, mask=m, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=m)
    tl.store(right_ptr + offs, right, mask=m)

# ----------------------------------------------------------------------
# 2. Core contraction kernel: out = left @ rightᵀ  (shape [B,N,N,H])
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr,      # [B,N,N,H] half
    right_ptr,     # [B,N,N,H] half
    out_ptr,       # [B,N,N,H] half
    B: tl.int32,
    H: tl.int32,
    N: tl.int32,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_i = tl.program_id(0)   # block row (i)
    pid_j = tl.program_id(1)   # block col (j)
    pid_bh = tl.program_id(2)  # batch * head index

    # recover batch and head indices
    b = pid_bh // H
    h = pid_bh % H

    # base pointers for this (b,h)
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # start offsets for the tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # offsets vectors
    i_offsets = i_start + tl.arange(0, BLOCK_M)
    j_offsets = j_start + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds rows / columns
    mask_i = i_offsets < N
    mask_j = j_offsets < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, N, BLOCK_K):
        k_start = k
        k_offsets = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < N

        # left block   : [i, k]
        a_ptrs = left_base + i_offsets[:, None] * stride_left_i + k_offsets[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=(mask_i[:, None] & mask_k[None, :]),
                    other=0.0)

        # right block  : [k, j]  (rightᵀ)
        b_ptrs = right_base + k_offsets[:, None] * stride_right_k + j_offsets[None, :] * stride_right_i
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_j[None, :]),
                    other=0.0)

        # accumulate
        acc += tl.dot(a, b)

    # write back result
    out_ptrs = out_base + i_offsets[:, None] * stride_out_i + j_offsets[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=(mask_i[:, None] & mask_j[None, :]))

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or numeric) or None
            - weights: dict mapping parameter names to torch.Tensor
            - config: dict with keys "dim", "hidden_dim", and optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 → half)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half‑precision)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Gate values (half‑precision, sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask and gates (if needed) using Triton
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → broadcast over hidden_dim
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()   # [B,N,N,1]

        # ensure contiguous tensors
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_h = mask_h.contiguous()

        total = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)

        _gate_mask_fuse_kernel[grid](
            left, right,
            left_gate, right_gate,
            mask_h,                 # broadcast mask
            total,
            BLOCK=BLOCK,
            HAS_MASK=1,
            H=hidden_dim,
        )
    else:
        # No mask: simple element‑wise gating
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out = left @ rightᵀ  (batched GEMM)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # ensure tensors are contiguous for stride calculations
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)   # [B,N,N,H] half

    # strides (int64)
    stride_left_b = left.stride(0)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)
    stride_left_h = left.stride(3)

    stride_right_b = right.stride(0)
    stride_right_i = right.stride(1)
    stride_right_k = right.stride(2)
    stride_right_h = right.stride(3)

    stride_out_b = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_h = out.stride(3)

    # tile sizes – tuned for typical N in the test suite
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_i = triton.cdiv(N, BLOCK_M)
    num_j = triton.cdiv(N, BLOCK_N)
    total_heads = B * H

    grid = (num_i, num_j, total_heads)

    _tri_mul_outgoing_kernel[grid](
        left, right, out,
        B, H, N,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate          # element‑wise gating (half)
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return float32 as required
    return out.to(torch.float32)
