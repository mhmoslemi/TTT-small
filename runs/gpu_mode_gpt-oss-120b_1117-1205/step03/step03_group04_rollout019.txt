@triton.jit
def gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elts: tl.int32,
    hidden_dim: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elts

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Sigmoid
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate))

    if HAS_MASK:
        # mask broadcast across hidden_dim
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def custom_kernel(data):
    # parse arguments
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # cast weights
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias = weights["norm.bias"].to(torch.float16)
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    to_out_weight = weights["to_out.weight"].to(torch.float16)

    # Step1: LayerNorm
    # Input maybe float32; convert to float16 for faster compute
    x = input_tensor.to(torch.float16)
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # Step2: linear combos for left and right + gates
    left_combined_weight = torch.cat([left_proj_weight, left_gate_weight], dim=0)  # (2H, C)
    right_combined_weight = torch.cat([right_proj_weight, right_gate_weight], dim=0)  # (2H, C)

    # Linear transforms
    left_combined = F.linear(x, left_combined_weight)   # shape (B,N,N,2H)
    right_combined = F.linear(x, right_combined_weight)

    # split
    left_proj = left_combined[..., :hidden_dim]
    left_gate_raw = left_combined[..., hidden_dim:]

    right_proj = right_combined[..., :hidden_dim]
    right_gate_raw = right_combined[..., hidden_dim:]

    # compute out_gate
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    # optional mask to fp16
    if not nomask:
        mask_fp = mask.to(torch.float16).contiguous()
    else:

"""
TriMul (outgoing) forward – fast Triton‑fused gating & optional mask.

Algorithm
---------
1. Layer‑norm the input (fp16).
2. Fuse each projection+gate linear layer:
   * left_combined = [left_proj ; left_gate]·x  → shape (...,2·H)
   * right_combined = [right_proj ; right_gate]·x → shape (...,2·H)
3. Split the combined tensors, then invoke a Triton kernel that
   * applies σ (sigmoid) to the gate halves,
   * multiplies the projection by its gate,
   * optionally multiplies by a broadcast mask.
   The result overwrites the projection tensors (in‑place).
4. Batched matrix‑multiply over the sequence dimension:
   out_{b,i,j,h} = Σ_k left_{b,i,k,h} · right_{b,j,k,h}
   performed as a strided‑batched GEMM with torch.bmm.
5. Layer‑norm on the hidden dimension, multiply by out_gate,
   and a final linear projection back to `dim`.
All heavy element‑wise work (step 3) is done in Triton;
the large GEMM (step 4) uses cuBLAS’s highly‑optimized batched matmul.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # fp16* [total]
    right_ptr,              # fp16* [total]
    left_gate_ptr,          # fp16* [total]
    right_gate_ptr,         # fp16* [total]
    mask_ptr,               # fp16* [mask_total] (broadcast)
    total_elts: tl.int32,   # total number of elements = B·N·N·H
    hidden_dim: tl.constexpr,
    HAS_MASK: tl.constexpr,  # 1 -> mask present, 0 -> no mask
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elts

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # sigmoid = 1/(1+exp(-x))
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate))

    if HAS_MASK:
        # each mask element corresponds to `hidden_dim` consecutive hidden entries
        mask_idx = offs // hidden_dim
        mval = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mval
        right = right * right_gate * mval
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_mask_fuse(left_proj, right_proj, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place: left_proj = left_proj * sigmoid(left_gate) * mask
              right_proj = right_proj * sigmoid(right_gate) * mask
    All arguments must be contiguous fp16 on the same CUDA device.
    """
    total = left_proj.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left_proj  # dummy pointer when HAS_MASK==0

    _gate_mask_fuse_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        hidden_dim,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )
    # tensors are modified in‑place; no return value


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or numeric) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict containing at least ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Cast weights to fp16 once
    # ------------------------------------------------------------------
    w_norm = weights["norm.weight"].to(torch.float16)
    b_norm = weights["norm.bias"].to(torch.float16)

    w_left_proj = weights["left_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)

    w_right_proj = weights["right_proj.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)

    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)

    w_to_out = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (fp16)
    # ------------------------------------------------------------------
    x = input_tensor.to(torch.float16)
    x = F.layer_norm(x, (dim,), weight=w_norm, bias=b_norm)

    # ------------------------------------------------------------------
    # 2. Fuse projection + gate linear layers (2 matmuls instead of 4)
    # ------------------------------------------------------------------
    left_comb_weight = torch.cat([w_left_proj, w_left_gate], dim=0)   # (2*H, dim)
    right_comb_weight = torch.cat([w_right_proj, w_right_gate], dim=0)  # (2*H, dim)

    left_combined = F.linear(x, left_comb_weight)    # (B, N, N, 2*H)
    right_combined = F.linear(x, right_comb_weight)  # (B, N, N, 2*H)

    # Split projection & gate pre‑sigmoid tensors
    left_proj = left_combined[..., :hidden_dim]
    left_gate_raw = left_combined[..., hidden_dim:]

    right_proj = right_combined[..., :hidden_dim]
    right_gate_raw = right_combined[..., hidden_dim:]

    # ------------------------------------------------------------------
    # 3. Out gate (separate linear)
    # ------------------------------------------------------------------
    out_gate = F.linear(x, w_out_gate).sigmoid()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4. Fuse gates and optional mask via Triton
    # ------------------------------------------------------------------
    mask_fp = None
    if not nomask:
        mask_fp = mask.to(torch.float16).contiguous()
    _apply_gate_mask_fuse(left_proj, right_proj,
                          left_gate_raw, right_gate_raw,
                          mask_fp, hidden_dim)

    # ------------------------------------------------------------------
    # 5. Core contraction – batched GEMM over the sequence dimension
    # ------------------------------------------------------------------
    B, N, _, H = left_proj.shape  # left_proj now holds the gated & masked values
    # reshape to (B*H, N, N) with left as (i,k) and right as (k,j)
    left_mat = left_proj.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right_proj.permute(0, 3, 2, 1).reshape(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)                     # (B*H, N, N)

    # Restore shape (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,),
                       weight=w_to_out_norm,
                       bias=b_to_out_norm)
    out = out * out_gate
    out = F.linear(out, w_to_out)

    return out.float()
