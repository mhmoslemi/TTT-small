import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(left_ptr, right_ptr, left_gate_ptr, right_gate_ptr,
                           mask_ptr, total_elements, BLOCK: tl.constexpr,
                           HAS_MASK: tl.constexpr, H: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask)
    right = tl.load(right_ptr + offs, mask=mask)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask)

    if HAS_MASK:
        # mask index per hidden channel group
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    # left, right, left_gate, right_gate are half tensors contiguous.
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    HAS_MASK = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when not used
    _gate_mask_fuse_kernel[grid](
        left, right, left_gate, right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=HAS_MASK,
        H=hidden_dim,
    )

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    # 1. LayerNorm
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])
    # cast to half
    x_h = x.to(torch.float16)
    # 2. Projections
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16), bias=None)
    # 3. Gates
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()
    # 4. Apply mask and gates
    if not nomask:
        # Ensure tensors contiguous and flatten
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask.to(torch.float16).contiguous().view(-1)  # [B,N,N]
        _apply_gate_and_mask(left_f, right_f, left_gate_f, right_gate_f, mask_f, hidden_dim)
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate
    # No longer need gates
    del left_gate, right_gate
    # 5. Contraction via batched GEMM
    # Permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    # 6. Output layernorm
    out = F.layer_norm(out, (hidden_dim,),
                       weight=weights["to_out_norm.weight"].to(torch.float16),
                       bias=weights["to_out_norm.bias"].to(torch.float16))
    # 7. Apply out gate
    out = out * out_gate
    # 8. Final projection
    out = F.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)
    # Return in float32
    return out.to(torch.float32)

TriMul forward pass (outgoing version) with a fused Triton kernel for
gate–mask multiplication. The expensive N³ contraction is performed as a
batched GEMM (torch.bmm) on half‑precision tensors, while the mask and
gate fusion is done in a custom Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half*  (flattened left tensor)
    right_ptr,         # half*  (flattened right tensor)
    left_gate_ptr,     # half*  (flattened left gate)
    right_gate_ptr,    # half*  (flattened right gate)
    mask_ptr,          # half*  (flattened mask, broadcasted over hidden dim)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 -> mask present, 0 -> no mask
    H: tl.constexpr,          # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load data
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # Each hidden channel shares a mask entry: mask index = offset // H
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store results back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    `mask` may be ``None`` (no mask). All tensors must be contiguous
    ``torch.float16`` tensors.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    # When mask is not used we pass a dummy pointer (left) to keep the
    # launch signature homogeneous.
    mask_ptr = mask if mask is not None else left

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # No return – operation is performed in‑place


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32) -> half precision
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections to hidden dimension (half)
    # --------------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # --------------------------------------------------------------
    # 3. Gates (half)
    # --------------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # --------------------------------------------------------------
    # 4. Fuse mask with gates via Triton (if mask present)
    # --------------------------------------------------------------
    if not nomask:
        # Flatten tensors for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask.to(torch.float16).contiguous().view(-1)  # [B*N*N]

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            hidden_dim,
        )
        # Reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # Simple element‑wise multiplication (no mask)
        left = left * left_gate
        right = right * right_gate

    # Gates no longer needed
    del left_gate, right_gate

    # --------------------------------------------------------------
    # 5. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as batched GEMM on half‑precision tensors.
    # --------------------------------------------------------------
    # Permute to (B, H, N, N) for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    # Collapse batch & head dimensions
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)  # (B*H, N, N)

    # Batched GEMM with second operand transposed
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise scaling (half precision)

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in original dtype (float32)
    return out.to(torch.float32)
