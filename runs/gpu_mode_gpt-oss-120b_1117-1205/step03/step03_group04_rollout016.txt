@triton.jit
def gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements: tl.int32,
    H: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offset = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offset < total_elements
    # Load
    left = tl.load(left_ptr + offset, mask=mask, other=0.0)
    right = tl.load(right_ptr + offset, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offset, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offset, mask=mask, other=0.0)
    # Compute
    left = left * left_gate
    right = right * right_gate
    if HAS_MASK:
        # mask index: each element corresponds to position in N,N dims, repeated across H
        mask_idx = offset // H   # integer division
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * mask_val
        right = right * mask_val
    # Store
    tl.store(left_ptr + offset, left, mask=mask)
    tl.store(right_ptr + offset, right, mask=mask)
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    # Ensure input is float32 or float16? It may be float32.
    # Convert to half for efficiency.
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x = x.to(torch.float16)
    # Linear projections in half.
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))
    # Gates
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()
    # apply gates and mask via triton
    if not nomask:
        mask_h = mask.unsqueeze(-1).to(torch.float16)  # shape [B,N,N,1]
        # flatten
        left_flat = left.view(-1)
        right_flat = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)
        mask_flat = mask_h.view(-1)  # size B*N*N
        total = left_flat.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)
        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            H=hidden_dim,
            HAS_MASK=1,
            BLOCK=BLOCK,
        )
        # reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate
    # Contraction
    # Use torch.einsum for simplicity.
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)
    # Output layernorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))
    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass.

The implementation follows the reference PyTorch module but fuses the
element‑wise gating and optional pairwise mask in a small Triton kernel.
All heavy linear operations and the N³ contraction are kept in FP16.
The contraction is performed as a batched GEMM (B·H independent
matrix multiplications) using torch.bmm, which maps to cuBLAS and
delivers high throughput on modern GPUs.

Steps
-----
1. Layer‑norm over the channel dimension (dim) → FP16.
2. Linear projections: left, right and gating vectors (hidden_dim).
3. Triton kernel fuses left/right gates with the optional mask:
       left = left * left_gate * mask
       right = right * right_gate * mask
   (mask is broadcast over the hidden dimension).
4. Batched matrix multiplication implements the einsum
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   by reshaping to (B·H, N, N) and using torch.bmm with a transpose.
5. Output Layer‑norm, out‑gate, final linear projection → FP32 output.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: fuse mask (optional) with left/right gates
# --------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half*   [total]
    right_ptr,         # half*   [total]
    left_gate_ptr,     # half*   [total]
    right_gate_ptr,    # half*   [total]
    mask_ptr,          # half*   [mask_total] (broadcast)
    total_elements: tl.int32,
    H: tl.constexpr,               # hidden dimension
    HAS_MASK: tl.constexpr,        # 1 if mask is present else 0
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Optional mask (broadcast over hidden dim)
    if HAS_MASK:
        # Each mask entry corresponds to H consecutive hidden‑dim entries
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * mask_val
        right = right * mask_val

    # Store results back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of gates and (optional) mask.
    All tensors must be contiguous torch.float16 on the same CUDA device.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    # dummy pointer when mask is absent (won't be dereferenced)
    mask_ptr = mask if mask is not None else left

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        H=hidden_dim,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )
    # No return; operation is performed in‑place


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> half)
    # ------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)  # FP16 for the rest of the pipeline

    # ------------------------------------------------------------
    # 2. Linear projections (FP16)
    # ------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------
    # 3. Gate values (FP16)
    # ------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------
    # 4. Fuse mask (if any) with left/right gate multiplication via Triton
    # ------------------------------------------------------------
    if not nomask:
        # mask shape [B,N,N] -> broadcast over hidden_dim
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        # Flatten everything for the kernel
        left_flat = left.view(-1)
        right_flat = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)
        mask_flat = mask_h.view(-1)  # size = B*N*N
        _apply_gate_and_mask(
            left_flat, right_flat,
            left_gate_flat, right_gate_flat,
            mask_flat,
            hidden_dim,
        )
        # Reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask → simple elementwise multiplication
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched GEMM (B·H independent N×N matmuls)
    # ------------------------------------------------------------
    B, N, _, _ = left.shape
    # Bring hidden dim to the batch dimension
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matrix multiplication: (i,k) @ (j,k)^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(-2, -1))  # (B*H, N, N), dtype=half

    # Restore original layout
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return in float32 as specified by the interface
    return out.to(torch.float32)
