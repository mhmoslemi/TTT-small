def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Step1: LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x = x.to(torch.float16)

    # Step2: projections
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # Step3: gates
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # Step4: apply gates and mask
    mask_tensor = None
    if not nomask and mask is not None:
        # mask is bool or numeric, broadcast to hidden_dim
        mask_tensor = mask.to(torch.float16).unsqueeze(-1)  # shape [B,N,N,1]
    # Flatten left/right/gates for kernel

    # Step5: core contraction via batched GEMM
    # left shape [B,N,N,H]; want (B,H,N,N)
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,2,1).contiguous()  # swap N axes

"""
TriMul (outgoing) forward pass with a Triton kernel.

Algorithm
---------
1. LayerNorm on the input tensor (float32) → half precision.
2. Linear projections (left/right) from dim → hidden_dim (half).
3. Compute gating tensors (left_gate, right_gate, out_gate) via Linear+Sigmoid (half).
4. Fuse gating (and optional mask) into left/right in‑place with a Triton kernel.
5. Core O(N³) contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM on half‑precision tensors (cuBLAS).
6. Output LayerNorm on hidden dimension, element‑wise out‑gate, final Linear
   projection back to dim (half → float32).

Only the gating‑mask fusion is performed in Triton; the heavy matrix
multiplication is delegated to cuBLAS (torch.bmm) which is already optimal.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half*  (total elements)
    right_ptr,         # half*  (total elements)
    left_gate_ptr,     # half*  (total elements)
    right_gate_ptr,    # half*  (total elements)
    mask_ptr,          # half*  (mask_total = total / H)  only read if HAS_MASK=1
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present, 0 otherwise
    H: tl.constexpr,          # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # load tensors
    left      = tl.load(left_ptr + offs,       mask=mask, other=0.0)
    right     = tl.load(right_ptr + offs,      mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs,  mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # apply gates
    left  = left  * left_gate
    right = right * right_gate

    if HAS_MASK == 1:
        # each mask element corresponds to H consecutive hidden entries
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left  = left  * mask_val
        right = right * mask_val

    # write back
    tl.store(left_ptr + offs,  left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of gates (and optional mask) into `left` and `right`.
    All tensors must be torch.float16 and contiguous.
    """
    total = left.numel()
    BLOCK = 1024  # good trade‑off for typical sizes
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when no mask

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or numeric) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (half)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Gating tensors (half)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse gates (and optional mask) via Triton (in‑place)
    # ------------------------------------------------------------------
    mask_tensor = None
    if not nomask and mask is not None:
        # Convert mask to half and broadcast over hidden_dim
        mask_tensor = mask.to(torch.float16).unsqueeze(-1).contiguous()
        mask_tensor = mask_tensor.view(-1)   # flatten B*N*N
    # Ensure tensors are contiguous before kernel launch
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    _apply_gate_and_mask(left, right, left_gate, right_gate, mask_tensor, hidden_dim)

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM (cuBLAS) on half tensors.
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) where H = hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()      # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()    # (B, H, N, N) with swapped N axes

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N), half precision
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise, half precision
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 7. Return in float32 as required
    # ------------------------------------------------------------------
    return out.to(torch.float32)
