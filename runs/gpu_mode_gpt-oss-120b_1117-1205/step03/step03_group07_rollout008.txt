"""tri_mul_outgoing_triton
Implementation of the outgoing TriMul operator (AlphaFold3) with a fused Triton
kernel for gating+masking and layout transformation. The heavy O(N³) contraction
is performed with a batched GEMM (torch.bmm) on half‑precision data, while the
kernel eliminates the extra transposes required by the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse left/right gating, optional mask, and produce the
# transposed layout needed for the batched matrix multiply.
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_transpose_kernel(
    left_ptr,               # half *
    right_ptr,              # half *
    left_gate_ptr,          # half *
    right_gate_ptr,         # half *
    mask_ptr,               # half *
    out_left_ptr,           # half *
    out_right_ptr,          # half *
    B, N, H,                # dimensions (int)
    HAS_MASK: tl.constexpr, # 1 if mask present else 0
    BLOCK: tl.constexpr,    # thread block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # Decode linear index -> (b, i, k, d)
    idx = offs
    d = idx % H
    idx = idx // H
    k = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    # Offsets for the original tensors: (b, i, k, d)
    src_offset = ((b * N + i) * N + k) * H + d

    left_val  = tl.load(left_ptr + src_offset,       mask=mask, other=tl.float16(0.0))
    right_val = tl.load(right_ptr + src_offset,      mask=mask, other=tl.float16(0.0))
    left_g    = tl.load(left_gate_ptr + src_offset,  mask=mask, other=tl.float16(0.0))
    right_g   = tl.load(right_gate_ptr + src_offset, mask=mask, other=tl.float16(0.0))

    left_val  = left_val  * left_g
    right_val = right_val * right_g

    if HAS_MASK:
        mask_idx = ((b * N + i) * N + k)                # (b,i,k) flat index
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left_val  = left_val  * m
        right_val = right_val * m

    # Store left in layout (B, H, N, N)  -> offset = ((b*H+d)*N + i)*N + k
    out_left_offset = ((b * H + d) * N + i) * N + k
    tl.store(out_left_ptr + out_left_offset, left_val, mask=mask)

    # Store right transposed: layout (B, H, N, N) with swapped i/k -> offset = ((b*H+d)*N + k)*N + i
    out_right_offset = ((b * H + d) * N + k) * N + i
    tl.store(out_right_ptr + out_right_offset, right_val, mask=mask)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for the heavy work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (FP16)
    # ------------------------------------------------------------------
    left_raw = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right_raw = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 4. Gating projections (FP16) + sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    B, N, _, _ = left_raw.shape  # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5. Fuse gating + optional mask and produce transposed layout
    # ------------------------------------------------------------------
    # Allocate output buffers in the layout required for the batched GEMM
    left_gated  = torch.empty((B, hidden_dim, N, N), dtype=torch.float16, device=device)
    right_gated = torch.empty((B, hidden_dim, N, N), dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).contiguous()
        has_mask = 1
        mask_ptr = mask_h
    else:
        # dummy pointer – not dereferenced when HAS_MASK == 0
        has_mask = 0
        mask_ptr = left_raw  # any tensor with correct dtype and device

    total_elements = B * N * N * hidden_dim
    BLOCK = 1024
    grid = (triton.cdiv(total_elements, BLOCK),)

    _gate_mask_transpose_kernel[grid](
        left_raw,
        right_raw,
        left_gate,
        right_gate,
        mask_ptr,
        left_gated,
        right_gated,
        B,
        N,
        hidden_dim,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 6. Core O(N³) contraction via batched GEMM
    #    left_gated: (B, H, N, N)  rows = i, cols = k
    #    right_gated: (B, H, N, N) rows = k, cols = j  (already transposed)
    # ------------------------------------------------------------------
    left_mat = left_gated.reshape(B * hidden_dim, N, N)
    right_mat = right_gated.reshape(B * hidden_dim, N, N)
    out_mat = torch.bmm(left_mat, right_mat)                     # (B*H, N, N)

    # Restore tensor shape (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final linear projection (still FP16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 8. Cast back to FP32 as required by the API
    # ------------------------------------------------------------------
    return out.to(torch.float32)
