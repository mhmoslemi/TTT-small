```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)
```

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor in float32 (as the parameters are float32),
   then cast to float16 for the rest of the computation.
2. Linear projections (half‑precision) produce left/right tensors of shape
   [B, N, N, H].
3. Compute left/right gating values (half‑precision, sigmoid) and an output
   gate.  Fuse the optional mask, left‑gate and right‑gate into a single
   Triton kernel that updates the left and right tensors in‑place.
4. Core O(N³) contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched GEMM over the hidden dimension and is performed with a
   single torch.einsum call (no extra memory copies).
5. Apply a second LayerNorm on the hidden dimension, multiply by the output
   gate, and a final linear projection back to the original feature dimension.
6. Return the result in float32.

The only Triton kernel is the mask‑gate fusion (step 3); the heavy matrix
multiplication (step 4) leverages cuBLAS via torch.einsum.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half*   left tensor (flattened)
    right_ptr,         # half*   right tensor (flattened)
    left_gate_ptr,     # half*   left gate (flattened)
    right_gate_ptr,    # half*   right gate (flattened)
    mask_ptr,          # half*   mask tensor (flattened, length = total/H) – unused if HAS_MASK=0
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present, else 0
    H: tl.constexpr,          # hidden dimension
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # one mask element corresponds to H consecutive hidden values
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    If ``mask`` is None, the mask term is omitted.
    All tensors must be contiguous ``torch.float16``.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when HAS_MASK==0

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # No return – the tensors are modified in‑place


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor[B, N, N, dim] (float32)
            - mask: torch.Tensor[B, N, N] (bool or numeric) – may be ignored if config["nomask"] is True
            - weights: dict of model parameters (float32 tensors)
            - config: dict containing at least "dim" and "hidden_dim"
    Returns:
        torch.Tensor[B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 0) Prepare weights (cast to half where appropriate)
    # ------------------------------------------------------------------
    # Input LayerNorm weights stay in float32 (LayerNorm expects same dtype as input)
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]

    # Projections and gates – half precision
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    # Output LayerNorm and final projection – half precision
    out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w   = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32) then cast to half
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden dimension
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # [B, N, N, H]
    right = F.linear(x, right_proj_w)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3) Compute gates (half) and fuse mask + gates
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    if not nomask:
        # mask -> half, broadcast over hidden dim
        mask_h = mask.to(torch.float16).unsqueeze(-1)          # [B, N, N, 1]

        # Flatten everything for the Triton kernel
        left_f      = left.contiguous().view(-1)
        right_f     = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f= right_gate.contiguous().view(-1)
        mask_f      = mask_h.contiguous().view(-1)            # length B*N*N

        _apply_gate_and_mask(left_f, right_f,
                             left_gate_f, right_gate_f,
                             mask_f, hidden_dim)

        # Reshape back to original layout
        left  = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask – simple element‑wise multiplication
        left  = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 4) Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented via a single einsum (batched GEMM) with half precision.
    # ------------------------------------------------------------------
    out = torch.einsum('bikd,bjkd->bijd', left, right)
    # Ensure dtype is half for subsequent layers
    out = out.to(torch.float16)

    # ------------------------------------------------------------------
    # 5) Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=out_norm_w, bias=out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)

    # ------------------------------------------------------------------
    # 6) Cast back to float32 for the final output
    # ------------------------------------------------------------------
    return out.to(torch.float32)
