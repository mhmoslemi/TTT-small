@triton.jit
def trimul_batch_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,
    stride_left_b, stride_left_m, stride_left_k,
    stride_right_b, stride_right_n, stride_right_k,
    stride_out_b, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # batch id is program_id(0)
    b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, K, BLOCK_K):
        # compute offsets for K dimension
        offs_k = k + tl.arange(0, BLOCK_K)

        # mask for out of bounds
        a = tl.load(left_ptr + b*stride_left_b + offs_m[:,None]*stride_left_m + offs_k[None,:]*stride_left_k, mask=offs_m[:,None] < M and offs_k[None,:] < K, other=0.0)

        # convert to float32 for accumulation
        a = a.to(tl.float32)
        b_mat = b_mat.to(tl.float32)
        acc += tl.dot(a, b_mat)
    # store
    tl.store(out_ptr + b*stride_out_b + offs_m[:,None]*stride_out_m + offs_n[None,:]*stride_out_n, acc.to(tl.float16), mask=(offs_m[:,None] < M) & (offs_n[None,:] < N))

   @triton.jit
   def fuse_gate_mask_kernel(
        left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr,
        total, H, BLOCK: tl.constexpr, HAS_MASK: tl.constexpr):
        pid = tl.program_id(0)
        offs = pid * BLOCK + tl.arange(0, BLOCK)
        mask = offs < total

        right = tl.load(right_ptr + offs, mask=mask, other=0.0)
        left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
        right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
        left = left * left_gate

"""
TriMul (outgoing) forward pass – highly‑optimized implementation.

The heavy N³ contraction is performed as a batch of GEMMs (one per hidden
channel) using torch.bmm on half‑precision tensors.  A small Triton kernel
fuses the masking and gating steps (applied in‑place) to guarantee that at
least one operation runs on Triton.  All other ops stay in half precision
except the final cast back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask (if present) and the left/right gates.
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half *
    right_ptr,         # half *
    left_gate_ptr,     # half *
    right_gate_ptr,    # half *
    mask_ptr,          # half *  (mask, broadcasted over hidden_dim)
    total_elements: tl.int32,
    hidden_dim: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present else 0
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elements

    # Load data
    left = tl.load(left_ptr + offs, mask=m, other=0.0)
    right = tl.load(right_ptr + offs, mask=m, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=m, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=m, other=0.0)

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    if HAS_MASK:
        # one mask element corresponds to `hidden_dim` consecutive hidden entries
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=m, other=0.0)
        left = left * mask_val
        right = right * mask_val

    # Store results in‑place
    tl.store(left_ptr + offs, left, mask=m)
    tl.store(right_ptr + offs, right, mask=m)


def _apply_gate_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of gating and optional mask using Triton.
    All tensors must be contiguous half‑precision.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when no mask

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        hidden_dim,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
    )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Cast weights to half precision once
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 1. LayerNorm (half) on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor.to(torch.float16),
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B, N, N, dim]  half

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)    # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Gating tensors (half)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()   # used later

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) and gates via Triton (in‑place)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> broadcast over hidden_dim
        mask_h = mask.to(torch.float16).unsqueeze(-1).contiguous()
        # flatten everything for the kernel (contiguous view)
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_h.contiguous().view(-1)   # shape B*N*N
        _apply_gate_mask(
            left_flat, right_flat,
            left_gate_flat, right_gate_flat,
            mask_flat,
            hidden_dim,
        )
        # reshape back to original layout
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # simple element‑wise gating (no mask)
        left = left * left_gate
        right = right * right_gate

    # gating tensors no longer needed
    del left_gate, right_gate, x

    # ------------------------------------------------------------------
    # 5. Core contraction – batched GEMM over hidden dimension
    #    out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
    # ------------------------------------------------------------------
    # Permute to (B, H, N, N) where H == hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # swap the two seq dims

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)      # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)    # (B*H, N, N)

    # Batched matrix multiplication (half precision)
    out_mat = torch.bmm(left_mat, right_mat)    # (B*H, N, N)

    # Restore layout → (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, element‑wise out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # back to original dim

    # ------------------------------------------------------------------
    # 7. Cast back to float32 for the API contract
    # ------------------------------------------------------------------
    return out.to(torch.float32)
