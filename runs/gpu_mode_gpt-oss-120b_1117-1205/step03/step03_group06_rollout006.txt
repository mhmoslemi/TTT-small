"""
Outgoing TriMul forward pass (AlphaFold3) with a small Triton kernel.

Algorithm
---------
1. Layer‑normalize the input over the channel dimension `dim`.
2. Project the normalized tensor to the hidden dimension `H` with two linear
   layers (`left_proj`, `right_proj`).  The result has shape [B,N,N,H].
3. Compute three sigmoid gates (`left_gate`, `right_gate`, `out_gate`) by a
   linear layer on the same normalized tensor.
4. Fuse mask (if present) and the two gates for `left` and `right` in a
   Triton kernel – this avoids an extra Python pass over the huge tensor.
5. Compute the core contraction
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   as a batched GEMM: for each head `d` we multiply the matrix
   L_{i,k} with the transpose of R_{j,k}.  This is performed with
   `torch.bmm` on half‑precision tensors (tensor cores).
6. Apply a second LayerNorm over the hidden dimension, multiply by the
   `out_gate`, and finally linearly project back to `dim`.
7. Return the result in float32.

Only the mask‑and‑gate fusion is written in Triton; the heavy matrix
multiplications use cuBLAS via `torch.bmm`, which is already highly
optimized on the B200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse optional mask with left/right gates
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half *
    right_ptr,         # half *
    left_gate_ptr,     # half *
    right_gate_ptr,    # half *
    mask_ptr,          # half *
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask is present else 0
    H: tl.constexpr,          # hidden dimension
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elems

    # Load tensors
    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    lgate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    if HAS_MASK:
        # One mask element corresponds to H consecutive hidden entries
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=0.0)
        left = left * lgate * mask_val
        right = right * rgate * mask_val
    else:
        left = left * lgate
        right = right * rgate

    # Write results back in place
    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place elementwise fusion:
        left  = left  * left_gate  * mask (if present)
        right = right * right_gate * mask (if present)

    All tensors must be contiguous torch.float16.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when no mask

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # tensors are modified in‑place


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (float32/bool) or None
            - weights: dict with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> half)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (half precision)
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(device, dtype=torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(device, dtype=torch.float16))

    # ------------------------------------------------------------------
    # 3. Gate values (half precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(device, dtype=torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(device, dtype=torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(device, dtype=torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask and gates via Triton (if a mask is present)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] -> broadcast over hidden_dim
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        # Flatten tensors for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask_h.contiguous().view(-1)  # shape [B*N*N]

        _apply_gate_and_mask(
            left_f, right_f,
            left_gate_f, right_gate_f,
            mask_f,
            hidden_dim,
        )
        # Restore shapes
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask – just apply the gates
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as batched GEMM over the hidden dimension.
    # ------------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) = right transposed on the last two dims
    B, H, N, _ = left_perm.shape

    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N) # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device, dtype=torch.float16),
        bias=weights["to_out_norm.bias"].to(device, dtype=torch.float16),
        eps=1e-5,
    )
    out = out * out_gate  # element‑wise, half precision
    out = F.linear(out, weights["to_out.weight"].to(device, dtype=torch.float16))

    # ------------------------------------------------------------------
    # 7. Cast back to float32 and return
    # ------------------------------------------------------------------
    return out.to(torch.float32)
