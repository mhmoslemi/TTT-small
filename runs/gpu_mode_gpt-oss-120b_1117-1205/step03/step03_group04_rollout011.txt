def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Input layernorm
    # input_norm = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # convert to half
    x = F.layer_norm(input_tensor, (dim,),
                weight=weights["norm.weight"],
                bias=weights["norm.bias"]).to(torch.float16)

    # Linear projections (half)
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # Prepare mask dtype half if needed
    if not nomask:
        # make sure mask is half and contiguous
        mask_h = mask.to(torch.float16).contiguous()
    else:
        # placeholder dummy mask (unused)
        mask_h = left  # just any tensor

    # Allocate output tensor
    out = torch.empty_like(left)  # shape (B,N,N,hidden_dim) in half

    # Compute strides
    B, N, _, H = left.shape

    # Get strides as Python ints
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_left_gate_b, stride_left_gate_i, stride_left_gate_k, stride_left_gate_h = left_gate.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_right_gate_b, stride_right_gate_i, stride_right_gate_k, stride_right_gate_h = right_gate.stride()
    stride_out_b, stride_out_i, stride_out_k, stride_out_h = out.stride()
    stride_mask_b, stride_mask_i, stride_mask_j = mask_h.stride()

    # Launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * H)

    HAS_MASK = 0 if nomask else 1

        left, left_gate, right, right_gate, mask_h, out,
        B, N, H,

        stride_left_gate_b, stride_left_gate_i, stride_left_gate_k, stride_left_gate_h,

        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        HAS_MASK=HAS_MASK,
    )

"""
Highly‑optimized implementation of the outgoing TriMul operation.

Algorithm
---------
1. Layer‑norm the input tensor (float32 → float16).
2. Compute left/right projections and their gates with linear layers (FP16).
3. Fuse mask, left‑gate, right‑gate and the triple‑multiplicative contraction
   in a single Triton kernel.  The kernel loops over the summed axis *k* in
   tiles, loads the needed slices of `left`, `right`, the gate tensors and the
   optional binary mask, applies the element‑wise products and accumulates the
   dot‑product using `tl.dot`.  The hidden dimension *d* is treated as a batch
   axis, so each kernel instance processes a single (batch, hidden) pair.
4. Apply a second LayerNorm on the contracted result, multiply by the output
   gate and project back to the original channel dimension.
5. Return the result in float32.

The heavy N³ contraction is performed entirely on‑device, avoiding the
extra memory traffic of the reference PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused mask + gates + triple‑multiplicative contraction
# ----------------------------------------------------------------------
@triton.jit
def outgoing_trimul_kernel(
    # tensor pointers
    left_ptr, left_gate_ptr, right_ptr, right_gate_ptr, mask_ptr, out_ptr,
    # problem size
    B, N, H,
    # strides for each tensor (int32)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_left_gate_b, stride_left_gate_i, stride_left_gate_k, stride_left_gate_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_right_gate_b, stride_right_gate_i, stride_right_gate_k, stride_right_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid_i = tl.program_id(0)          # tile over output rows   (i)
    pid_j = tl.program_id(1)          # tile over output cols   (j)
    pid_bh = tl.program_id(2)         # batch * hidden index   (b, h)

    # decode batch and hidden index
    b = pid_bh // H
    d = pid_bh % H

    # start indices of the tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # offsets within the tile
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator (FP32 for accuracy)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)

    for k_tile in range(num_k_tiles):
        k_start = k_tile * BLOCK_K
        mask_k = (k_start + offs_k) < N

        # ------------------------------------------------------------------
        # Load left and its gate, apply mask if present
        # ------------------------------------------------------------------
        # left[b,i,k,d]
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + d * stride_left_h
            + offs_i[:, None] * stride_left_i
            + (k_start + offs_k)[None, :] * stride_left_k
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # left_gate[b,i,k,d]
        left_gate_ptrs = (
            left_gate_ptr
            + b * stride_left_gate_b
            + d * stride_left_gate_h
            + offs_i[:, None] * stride_left_gate_i
            + (k_start + offs_k)[None, :] * stride_left_gate_k
        )
        left_gate = tl.load(
            left_gate_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        left_eff = left * left_gate

        if HAS_MASK:
            mask_i_ptrs = (
                mask_ptr
                + b * stride_mask_b
                + offs_i[:, None] * stride_mask_i
                + (k_start + offs_k)[None, :] * stride_mask_j
            )
            mask_i_val = tl.load(
                mask_i_ptrs,
                mask=mask_i[:, None] & mask_k[None, :],
                other=0.0,
            ).to(tl.float32)
            left_eff = left_eff * mask_i_val

        # ------------------------------------------------------------------
        # Load right (transposed) and its gate, apply mask if present
        # right_T[k,j,d] = right[b, j, k, d]
        # ------------------------------------------------------------------
        right_T_ptrs = (
            right_ptr
            + b * stride_right_b
            + d * stride_right_h
            + (k_start + offs_k)[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_i
        )
        right_T = tl.load(
            right_T_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        right_gate_T_ptrs = (
            right_gate_ptr
            + b * stride_right_gate_b
            + d * stride_right_gate_h
            + (k_start + offs_k)[:, None] * stride_right_gate_k
            + offs_j[None, :] * stride_right_gate_i
        )
        right_gate_T = tl.load(
            right_gate_T_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        right_eff = right_T * right_gate_T

        if HAS_MASK:
            mask_j_ptrs = (
                mask_ptr
                + b * stride_mask_b
                + offs_j[:, None] * stride_mask_i
                + (k_start + offs_k)[None, :] * stride_mask_j
            )
            mask_j_val = tl.load(
                mask_j_ptrs,
                mask=mask_j[:, None] & mask_k[None, :],
                other=0.0,
            ).to(tl.float32)
            right_eff = right_eff * mask_j_val

        # ------------------------------------------------------------------
        # Accumulate block result: (M,K) dot (K,N) -> (M,N)
        # ------------------------------------------------------------------
        acc += tl.dot(left_eff, right_eff)

    # ----------------------------------------------------------------------
    # Store the result (cast back to FP16)
    # ----------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + d * stride_out_h
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )

# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32/bool) or None
            - weights: dict of torch.Tensor model parameters (float32)
            - config: dict containing at least ``dim`` and ``hidden_dim`` and
                      optionally ``nomask`` (bool)

    Returns:
        torch.Tensor: output of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 → FP16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Linear projections (FP16)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3) Gate values (FP16)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 4) Prepare mask (FP16) – broadcasted over the hidden dimension
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask.to(torch.float16).contiguous()
    else:
        # dummy mask – never read when HAS_MASK == 0
        mask_h = left  # any tensor of compatible dtype on the same device

    # ------------------------------------------------------------------
    # 5) Allocate output tensor for the contracted result
    # ------------------------------------------------------------------
    out = torch.empty_like(left)  # [B, N, N, hidden_dim] in FP16

    # ------------------------------------------------------------------
    # 6) Gather strides (required by the kernel)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    stride_left = left.stride()
    stride_left_gate = left_gate.stride()
    stride_right = right.stride()
    stride_right_gate = right_gate.stride()
    stride_out = out.stride()
    stride_mask = mask_h.stride()

    (s_l_b, s_l_i, s_l_k, s_l_h) = stride_left
    (s_lg_b, s_lg_i, s_lg_k, s_lg_h) = stride_left_gate
    (s_r_b, s_r_i, s_r_k, s_r_h) = stride_right
    (s_rg_b, s_rg_i, s_rg_k, s_rg_h) = stride_right_gate
    (s_o_b, s_o_i, s_o_j, s_o_h) = stride_out
    (s_m_b, s_m_i, s_m_j) = stride_mask

    # ------------------------------------------------------------------
    # 7) Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    HAS_MASK = 0 if nomask else 1

    outgoing_trimul_kernel[grid](
        left,
        left_gate,
        right,
        right_gate,
        mask_h,
        out,
        B,
        N,
        hidden_dim,
        s_l_b,
        s_l_i,
        s_l_k,
        s_l_h,
        s_lg_b,
        s_lg_i,
        s_lg_k,
        s_lg_h,
        s_r_b,
        s_r_i,
        s_r_k,
        s_r_h,
        s_rg_b,
        s_rg_i,
        s_rg_k,
        s_rg_h,
        s_m_b,
        s_m_i,
        s_m_j,
        s_o_b,
        s_o_i,
        s_o_j,
        s_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HAS_MASK=HAS_MASK,
    )

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, gating and final projection (still in FP16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Convert back to float32 as required by the spec
    return out.to(torch.float32)
