3) The remaining LayerNorms, gating and final projection stay in PyTorch
   (FP16) to avoid extra data movement.

All tensors are kept in FP16 after the initial LayerNorm to maximize
throughput; the final result is returned in FP32 as required by the spec.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# 1. Fuse mask + gates (optional)
# --------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half*   [total]
    right_ptr,         # half*   [total]
    left_gate_ptr,     # half*   [total]
    right_gate_ptr,    # half*   [total]
    mask_ptr,          # half*   [mask_total] (broadcast)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask is present else 0
    H: tl.constexpr,          # hidden dimension
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elements

    left = tl.load(left_ptr + offs, mask=m, other=0.0)
    right = tl.load(right_ptr + offs, mask=m, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=m, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=m, other=0.0)

    if HAS_MASK:
        # each mask element corresponds to H consecutive hidden entries
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=m, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=m)
    tl.store(right_ptr + offs, right, mask=m)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place gating (and optional mask) using a Triton kernel.
    All tensors must be contiguous torch.float16 on the same CUDA device.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when HAS_MASK==0

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # operation is in‑place; no return


# --------------------------------------------------------------
# 2. Batched GEMM for the core contraction
# --------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    left_ptr,    # half*   [batch, M, K]
    right_ptr,   # half*   [batch, K, N] (already transposed)
    out_ptr,     # half*   [batch, M, N]
    BATCH: tl.int32,
    M: tl.int32,
    N: tl.int32,
    K: tl.int32,
    stride_l_batch: tl.int32,
    stride_l_m: tl.int32,
    stride_l_k: tl.int32,
    stride_r_batch: tl.int32,
    stride_r_k: tl.int32,
    stride_r_n: tl.int32,
    stride_o_batch: tl.int32,
    stride_o_m: tl.int32,
    stride_o_n: tl.int32,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)        # batch index
    pid_m = tl.program_id(0)        # block row of output
    pid_n = tl.program_id(1)        # block col of output

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # base pointers for the current batch
    left_batch_ptr  = left_ptr  + pid_b * stride_l_batch
    right_batch_ptr = right_ptr + pid_b * stride_r_batch
    out_batch_ptr   = out_ptr   + pid_b * stride_o_batch

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left block: (BLOCK_M, BLOCK_K)
        left_ptrs = left_batch_ptr + (offs_m[:, None] * stride_l_m + offs_k[None, :] * stride_l_k)
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)

        # right block: (BLOCK_K, BLOCK_N) – note K dimension first
        right_ptrs = right_batch_ptr + (offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_n)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0)

        acc += tl.dot(left, right)   # (M, N) accumulation

    # write out
    out_ptrs = out_batch_ptr + (offs_m[:, None] * stride_o_m + offs_n[None, :] * stride_o_n)
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------
# 3. Main entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------
    # 1. Input LayerNorm (float32 → half)
    # ------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------
    # 2. Linear projections (half)
    # ------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------
    # 3. Gate values (half)
    # ------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------
    # 4. Fuse mask (if any) with left/right gates via Triton
    # ------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B,N,N] → [B,N,N,1] → flat
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        # flatten everything
        left_f = left.view(-1)
        right_f = right.view(-1)
        left_gate_f = left_gate.view(-1)
        right_gate_f = right_gate.view(-1)
        mask_f = mask_h.view(-1)    # size = B*N*N
        _apply_gate_and_mask(
            left_f, right_f,
            left_gate_f, right_gate_f,
            mask_f,
            hidden_dim,
        )
        # reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask → simple elementwise multiplication
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------
    # 5. Core contraction via a custom batched GEMM kernel
    # ------------------------------------------------------------
    batch, seq_len, _, _ = left.shape  # batch = B, seq_len = N

    # transpose right to (K,N) layout needed for matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) where first N is K

    batch_total = batch * hidden_dim          # B * H
    M = N = K = seq_len

    # reshape to (batch_total, M, K) and (batch_total, K, N)
    left_mat = left_perm.view(batch_total, M, K)
    right_mat = right_perm.view(batch_total, K, N)

    # allocate output
    out_mat = torch.empty_like(left_mat)

    # strides (int32)
    stride_l_batch, stride_l_m, stride_l_k = left_mat.stride()
    stride_r_batch, stride_r_k, stride_r_n = right_mat.stride()
    stride_o_batch, stride_o_m, stride_o_n = out_mat.stride()

    # Convert strides to int32 for Triton
    stride_l_batch = int(stride_l_batch)
    stride_l_m = int(stride_l_m)
    stride_l_k = int(stride_l_k)
    stride_r_batch = int(stride_r_batch)
    stride_r_k = int(stride_r_k)
    stride_r_n = int(stride_r_n)
    stride_o_batch = int(stride_o_batch)
    stride_o_m = int(stride_o_m)
    stride_o_n = int(stride_o_n)

    # Kernel launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch_total,
    )
    _tri_mul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch_total,
        M,
        N,
        K,
        stride_l_batch,
        stride_l_m,
        stride_l_k,
        stride_r_batch,
        stride_r_k,
        stride_r_n,
        stride_o_batch,
        stride_o_m,
        stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_mat.view(batch, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return in float32 as required by the spec
    return out.to(torch.float32)
```
