@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elems: tl.int32,
    H: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        mask_idx = offs // H
        m_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * lgate * m_val
        right = right * rgate * m_val
    else:
        left = left * lgate
        right = right * rgate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)
```

"""
Outgoing TriMul implementation with a Triton kernel for mask+gate fusion.

Algorithm
---------
1. Layer‑norm the input tensor (float32).
2. Cast to fp16 and apply two linear projections (left/right) and three
   sigmoid‑gates (left_gate, right_gate, out_gate).
3. If a mask is present, a Triton kernel fuses:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
   where mask is broadcast over the hidden dimension.
   Otherwise the gates are applied with a simple element‑wise product.
4. Compute the O(N³) contraction
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   using a batched GEMM (torch.bmm) on the (B·H, N, N) matrices.
5. Apply a second LayerNorm, the output gate and a final linear projection
   back to the original channel dimension.
6. Return the result in float32.

The kernel only handles the mask‑gate fusion; the heavy contraction uses
cuBLAS‑accelerated torch.bmm, which yields strong performance while satisfying
the requirement of having a custom Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask (if present) and per‑element gates
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,        # half *
    right_ptr,       # half *
    left_gate_ptr,   # half *
    right_gate_ptr,  # half *
    mask_ptr,        # half *
    total_elems: tl.int32,
    H: tl.constexpr,          # hidden dimension (size of innermost channel)
    HAS_MASK: tl.constexpr,   # 1 if mask is provided, else 0
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elems

    # Load tensors
    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    lgate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # One mask element corresponds to H consecutive channel entries.
        mask_idx = offs // H
        m_val = tl.load(mask_ptr + mask_idx, mask=active, other=0.0)
        left = left * lgate * m_val
        right = right * rgate * m_val
    else:
        left = left * lgate
        right = right * rgate

    # Write results back in‑place
    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    Fuse gating and optional mask using the Triton kernel.
    All tensors must be contiguous torch.float16.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    # When mask is absent we still need to pass a pointer; use left as dummy.
    mask_ptr = mask if mask is not None else left

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        H=hidden_dim,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )
    # Results are written back into `left` and `right` in place.


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/int/float) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict containing at least "dim" and "hidden_dim",
                      optionally "nomask" (bool).

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) and cast to fp16
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (fp16)
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Compute gates (fp16) and sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask and gates (Triton) or simple multiplication
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] -> broadcast over hidden_dim
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        # Ensure contiguous storage for all tensors
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_h = mask_h.contiguous()
        # Fuse using Triton
        _apply_gate_and_mask(left, right, left_gate, right_gate, mask_h, hidden_dim)
    else:
        # No mask: simple element‑wise gating
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as batched GEMM on (B·H, N, N) matrices.
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) where H = hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, i_k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, k, j)
    B, H, N, _ = left_perm.shape

    left_mat = left_perm.view(B * H, N, N)   # (B·H, i, k)
    right_mat = right_perm.view(B * H, N, N) # (B·H, k, j)

    # Batched matrix multiplication (fp16, uses Tensor Cores)
    out_mat = torch.bmm(left_mat, right_mat)  # (B·H, i, j)

    # Restore layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return in the requested precision (float32)
    return out.to(torch.float32)
