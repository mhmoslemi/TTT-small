"""
Outgoing TriMul forward pass (AlphaFold3)

The operator works on a 4‑D tensor x of shape [B, N, N, C] (C = dim).
It performs:
  1. LayerNorm over the channel dimension.
  2. Linear projections to a hidden dimension H (left/right).
  3. Gating (sigmoid) on the projected tensors, optionally masked.
  4. A batched contraction:
        out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
     which is a batch of H independent GEMMs (implemented with torch.bmm).
  5. LayerNorm over the hidden dimension, an output gate and a final linear
     projection back to dim.

The mask‑and‑gate fusion (step 3) is implemented as a custom Triton kernel.
All heavy arithmetic is done in float16 for speed; the final result is
returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------------
# Triton kernel: fuses mask (if present) and element‑wise gating
# ------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,           # half *
    right_ptr,          # half *
    left_gate_ptr,      # half *
    right_gate_ptr,     # half *
    mask_ptr,           # half *
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    H: tl.constexpr,    # hidden dimension (size of innermost channel)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Broadcast mask over the hidden dimension
    mask_idx = offs // H                     # which (b,i,k) element we are at
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    # Fuse: left = left * lgate * mask, right = right * rgate * mask
    left = left * lgate * mask_val
    right = right * rgate * mask_val

    # Store back
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of left/right proj, gate and (optional) mask using Triton.
    All tensors must be contiguous torch.float16.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask,
        total,
        BLOCK=BLOCK,
        H=hidden_dim,
    )
    # Results are written back into `left` and `right`.

# Entry point
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or numeric) or None
            - weights: dict of model parameters (torch.Tensor)
            - config: dict with keys:
                * "dim" (int): input channel dimension C
                * "hidden_dim" (int): projection dimension H
                * "nomask" (bool, optional): True => ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> half)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half precision)
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid) and optional mask fusion via Triton
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    if not nomask:
        # mask shape: [B, N, N] → broadcast over hidden_dim inside the kernel
        mask_h = mask.to(torch.float16).contiguous()

        # Flatten tensors for the kernel launch
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask_h.view(-1)                     # length = B * N * N

        _apply_gate_and_mask(
            left_f, right_f,
            left_gate_f, right_gate_f,
            mask_f,
            hidden_dim,
        )
        # Reshape back to the original 4‑D tensors
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask → simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Core contraction (batched GEMM over the sequence dimension)
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) where H = hidden_dim.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) i.e. (k,j)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # Half‑precision batched matrix multiplication (uses cuBLAS tensor cores)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate                     # element‑wise, half precision
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return result in the requested precision
    return out.to(torch.float32)
