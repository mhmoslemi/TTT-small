"""
TriMul (outgoing) forward pass with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor (float32 → half).
2. Linear projections to the hidden dimension for the left/right tensors.
3. Compute left/right/out gates (sigmoid of a linear).  Fuse the optional mask
   with the left/right gates in a custom Triton kernel that updates the left
   and right tensors in‑place.
4. Core contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is realized as many independent matrix‑multiplications
   (B·H of size N×N) via a batched GEMM.
5. Apply output LayerNorm, point‑wise multiply by the output gate and a final
   linear projection back to the original feature dimension.
6. Return the result in float32.

Only the mask‑gate fusion is performed with Triton; all other operations
use highly‑optimised PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: in‑place fusion of mask (optional) with left/right gates
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half*  [total]
    right_ptr,         # half*  [total]
    left_gate_ptr,     # half*  [total]
    right_gate_ptr,    # half*  [total]
    mask_ptr,          # half*  [mask_total] (broadcast over hidden)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present else 0
    H: tl.constexpr,          # hidden dimension
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # Each mask element corresponds to H consecutive hidden entries
        mask_offs = offs // H
        mask_val = tl.load(mask_ptr + mask_offs, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    Fuse ``left = left * left_gate * mask`` and
    ``right = right * right_gate * mask`` (if mask provided) in‑place.
    All tensors must be torch.float16 and contiguous.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy, never read when HAS_MASK=0

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # tensors are modified in‑place; nothing to return


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or numeric) or None
            - weights: dict mapping parameter names to torch.Tensor
            - config: dict containing ``dim`` (int), ``hidden_dim`` (int),
                      and optionally ``nomask`` (bool, default True)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(dtype=torch.float16, device=device),
        bias=weights["norm.bias"].to(dtype=torch.float16, device=device),
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half‑precision)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(dtype=torch.float16, device=device),
    )  # [B,N,N,hidden_dim]
    right = F.linear(
        x,
        weights["right_proj.weight"].to(dtype=torch.float16, device=device),
    )

    # ------------------------------------------------------------------
    # 3. Gate values (half‑precision, sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(dtype=torch.float16, device=device),
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(dtype=torch.float16, device=device),
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(dtype=torch.float16, device=device),
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) with gates via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # Broadcast mask over hidden dimension and flatten tensors
        mask_h = mask.unsqueeze(-1).to(torch.float16, device=device).contiguous()
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        lg_f = left_gate.contiguous().view(-1)
        rg_f = right_gate.contiguous().view(-1)
        mask_f = mask_h.view(-1)   # length B*N*N
        _apply_gate_and_mask(left_f, right_f, lg_f, rg_f, mask_f, hidden_dim)
        # Reshape back (views already share storage)
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # Simple in‑place multiplication when no mask is present
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as batched GEMM over (B·hidden_dim) matrices of size N×N.
    # ------------------------------------------------------------------
    # Bring hidden dimension to the batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) = transposed k dimension

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Batched matrix multiplication (half precision, float32 accumulation)
    out_mat = torch.bmm(left_mat, right_mat)   # (B*H, N, N), half

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype=torch.float16, device=device),
        bias=weights["to_out_norm.bias"].to(dtype=torch.float16, device=device),
    )
    out = out * out_gate   # element‑wise gating (half precision)
    out = F.linear(
        out,
        weights["to_out.weight"].to(dtype=torch.float16, device=device),
    )

    # Return float32 as required
    return out.to(torch.float32)
