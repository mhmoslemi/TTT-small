def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm
    x = torch.nn.functional.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # Cast to float16
    x = x.to(torch.float16)

    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = torch.nn.functional.linear(x, weights["right_proj.weight"].to(torch.float16))

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"].to(torch.float16)))

    # Apply mask
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_h
        right = right * mask_h

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Permute to batch*hidden_dim first
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, k, j)

    B, H, N, _ = left_perm.shape
    # flatten batch*H
    left_mat = left_perm.view(-1, N, N)
    right_mat = right_perm.view(-1, N, N)

    out_mat = torch.empty_like(left_mat) # float16

    _batched_matmul_triton(left_mat, right_mat, out_mat)

    # reshape back
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # output LayerNorm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,),
                                         weight=weights["to_out_norm.weight"].to(torch.float16),
                                         bias=weights["to_out_norm.bias"].to(torch.float16))

    # Multiply by out_gate
    out = out * out_gate

    # Final linear projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return float32
    return out.to(torch.float32)

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The kernel computes the heavy O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
as a batched matrix multiplication using Triton.  The rest of the
operations (layer‑norms, linear projections, sigmoids, masking) are
performed with PyTorch in fp16 for speed, and the final output is
returned in fp32.

The Triton kernel implements a standard tiled GEMM:
    (B·H, M=N, K=N) × (B·H, K=N, N) → (B·H, M=N, N)
with float16 inputs and float32 accumulation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------
# Triton batched GEMM kernel (left @ right)
# ------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers to A, B, C (fp16)
    M, N, K,                      # matrix dimensions
    stride_ab, stride_am, stride_ak,  # A: (batch, M, K) strides
    stride_bb, stride_bk, stride_bn,  # B: (batch, K, N) strides
    stride_cb, stride_cm, stride_cn,  # C: (batch, M, N) strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(2)          # batch dimension (B·H)
    pid_m = tl.program_id(0)             # row tile
    pid_n = tl.program_id(1)             # column tile

    # -------- tile offsets ----------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # -------- accumulation ----------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------- loop over K ----------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (BLOCK_M x BLOCK_K)
        a = tl.load(
            a_ptr
            + batch_id * stride_ab
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile (BLOCK_K x BLOCK_N)
        b = tl.load(
            b_ptr
            + batch_id * stride_bb
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Accumulate
        acc += tl.dot(a, b)

    # -------- write result ----------
    c = acc.to(tl.float16)
    c_ptrs = (
        c_ptr
        + batch_id * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor, out: torch.Tensor):
    """
    Launch the batched GEMM kernel.

    Arguments:
        left:  (B·H, M, K)  fp16 tensor
        right: (B·H, K, N)  fp16 tensor
        out:   (B·H, M, N)  fp16 tensor (output)
    """
    assert left.is_contiguous() and right.is_contiguous() and out.is_contiguous()
    batch, M, K = left.shape
    _, K2, N = right.shape
    assert K == K2

    # Tunable tile sizes – chosen to give good occupancy across test sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch,
    )
    _batched_matmul_kernel[grid](
        left,
        right,
        out,
        M,
        N,
        K,
        left.stride(0),
        left.stride(1),
        left.stride(2),
        right.stride(0),
        right.stride(1),
        right.stride(2),
        out.stride(0),
        out.stride(1),
        out.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )


# ------------------------------------------------------------
# Entry point
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping of parameter names to torch.Tensor
        - config_dict  : contains "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------
    # 1. Input LayerNorm (float32 → fp16)
    # --------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # --------------------------------------------------------
    # 2. Linear projections (fp16)
    # --------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------
    # 3. Gates (sigmoid)
    # --------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(torch.float16)))

    # --------------------------------------------------------
    # 4. Optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # --------------------------------------------------------
    # 5. Apply gates
    # --------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------
    # 6. Core contraction via batched GEMM (Trition)
    # --------------------------------------------------------
    #   left:  (B, N, N, H)  → (B, H, i, k)
    #   right: (B, N, N, H)  → (B, H, k, j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, k, j)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(-1, N, N)   # (B·H, M=N, K=N)
    right_mat = right_perm.view(-1, N, N)  # (B·H, K=N, N=N)

    out_mat = torch.empty_like(left_mat)   # fp16

    _batched_matmul_triton(left_mat, right_mat, out_mat)

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final projection
    # --------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # --------------------------------------------------------
    # 8. Return float32 tensor as specified
    # --------------------------------------------------------
    return out.to(torch.float32)
