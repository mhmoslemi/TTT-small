import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements, hidden_dim,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def custom_kernel(data):
    """
    Outgoing TriMul forward pass implemented with fused gating/mask kernel.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict: mapping of parameter names to torch.Tensor
            - config_dict: must contain "dim" and "hidden_dim", may contain "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # shape [B,N,N,dim], float32

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for the rest of the pipeline
    # -----------------------------------------------------------------
    x_h = x.to(torch.float16)

    B, N, _, _ = x_h.shape
    total_pairs = B * N * N

    # -----------------------------------------------------------------
    # 3. Combined linear projections for left & left_gate
    # -----------------------------------------------------------------
    # Concatenate weights for left_proj and left_gate
    left_weight_concat = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)
    # Flatten input for linear
    x_flat = x_h.view(total_pairs, dim)

    left_concat = F.linear(x_flat, left_weight_concat)  # (total_pairs, 2*hidden_dim)
    left_proj = left_concat[:, :hidden_dim]
    left_gate = left_concat[:, hidden_dim:]

    # -----------------------------------------------------------------
    # 4. Combined linear projections for right & right_gate
    # -----------------------------------------------------------------
    right_weight_concat = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)
    right_concat = F.linear(x_flat, right_weight_concat)
    right_proj = right_concat[:, :hidden_dim]
    right_gate = right_concat[:, hidden_dim:]

    # Reshape back to 4‑D tensors
    left_proj = left_proj.view(B, N, N, hidden_dim).contiguous()
    left_gate = left_gate.view(B, N, N, hidden_dim).contiguous()
    right_proj = right_proj.view(B, N, N, hidden_dim).contiguous()
    right_gate = right_gate.view(B, N, N, hidden_dim).contiguous()

    # -----------------------------------------------------------------
    # 5. Out gate (no mask applied)
    # -----------------------------------------------------------------
    out_gate = F.linear(
        x_h, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()  # (B,N,N,hidden_dim)

    # -----------------------------------------------------------------
    # 6. Fuse mask and gating (Triton)
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).contiguous().view(-1)  # (B*N*N,)
        # Flatten tensors for kernel
        left_f = left_proj.view(-1)
        right_f = right_proj.view(-1)
        left_gate_f = left_gate.view(-1)
        right_gate_f = right_gate.view(-1)
        total_elements = left_f.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)

        HAS_MASK = 1
        _gate_mask_fuse_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_h,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            HAS_MASK=HAS_MASK,
        )
        # reshape back (in‑place data already modified)
        left_proj = left_f.view_as(left_proj)
        right_proj = right_f.view_as(right_proj)
        # gates no longer needed
        del left_gate, right_gate
    else:
        # No mask: simple elementwise multiplication
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate
        del left_gate, right_gate

    # -----------------------------------------------------------------
    # 7. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as batched matrix multiplication.
    # -----------------------------------------------------------------
    left_perm = left_proj.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_proj.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)

    # Use torch.matmul which supports 4‑D batched matmul
    out = torch.matmul(left_perm, right_perm)  # (B, H, N, N)

    # Restore original layout (B, N, N, H)
    out = out.permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 8. Output LayerNorm, gating, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out, weights["to_out.weight"].to(torch.float16)
    )  # (B,N,N,dim) half

    # Cast back to float32 as required by the API
    return out.to(torch.float32)

"""
Outgoing TriMul forward pass.

The implementation follows the reference PyTorch version but
* uses half‑precision for the heavy linear algebra,
* fuses the mask and gating multiplications in a single Triton kernel,
* concatenates the projection and gate weight matrices to halve the number
  of GEMM calls, and
* performs the O(N³) contraction as a batched matrix multiplication
  (torch.matmul) on fp16 tensors.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse left/right projections with their gates and an
# optional mask (mask is broadcast over the hidden dimension).
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # fp16 *
    right_ptr,              # fp16 *
    left_gate_ptr,          # fp16 *
    right_gate_ptr,         # fp16 *
    mask_ptr,               # fp16 *
    total_elements,         # int32
    hidden_dim,             # int32
    BLOCK: tl.constexpr,    # compile‑time block size
    HAS_MASK: tl.constexpr  # 1 if mask present, 0 otherwise
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # Each mask element corresponds to `hidden_dim` consecutive channel entries
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store the fused results back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict: mapping of parameter names to torch.Tensor
            - config_dict: must contain "dim" and "hidden_dim". May contain "nomask".

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, cfg = data
    device = x.device

    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim], float32

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for the remaining heavy work
    # -----------------------------------------------------------------
    x_h = x.to(torch.float16)                     # [B, N, N, dim], fp16
    B, N, _, _ = x_h.shape
    total_pairs = B * N * N

    # -----------------------------------------------------------------
    # 3. Combined projections for left & left_gate
    # -----------------------------------------------------------------
    # Concatenate the two weight matrices (out_features stacked)
    left_weight_cat = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                           # (2*hidden_dim, dim)

    # Flatten the (B,N,N) spatial dimensions for a single GEMM
    x_flat = x_h.view(total_pairs, dim)           # (B*N*N, dim)

    left_cat = F.linear(x_flat, left_weight_cat)  # (B*N*N, 2*hidden_dim)
    left_proj = left_cat[:, :hidden_dim].view(B, N, N, hidden_dim).contiguous()
    left_gate = left_cat[:, hidden_dim:].view(B, N, N, hidden_dim).contiguous()
    del left_cat  # free intermediate

    # -----------------------------------------------------------------
    # 4. Combined projections for right & right_gate
    # -----------------------------------------------------------------
    right_weight_cat = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                           # (2*hidden_dim, dim)

    right_cat = F.linear(x_flat, right_weight_cat)  # (B*N*N, 2*hidden_dim)
    right_proj = right_cat[:, :hidden_dim].view(B, N, N, hidden_dim).contiguous()
    right_gate = right_cat[:, hidden_dim:].view(B, N, N, hidden_dim).contiguous()
    del right_cat

    # -----------------------------------------------------------------
    # 5. Out gate (no mask applied)
    # -----------------------------------------------------------------
    out_gate = F.linear(
        x_h, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()                                   # (B, N, N, hidden_dim)

    # -----------------------------------------------------------------
    # 6. Fuse mask and gating via Triton (if a mask is present)
    # -----------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_h = mask.to(torch.float16).contiguous().view(-1)   # (B*N*N,)

        # Flatten tensors for the kernel
        left_f = left_proj.view(-1)
        right_f = right_proj.view(-1)
        left_gate_f = left_gate.view(-1)
        right_gate_f = right_gate.view(-1)

        total_elems = left_f.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total_elems, BLOCK),)

        _gate_mask_fuse_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_h,
            total_elems,
            hidden_dim,
            BLOCK=BLOCK,
            HAS_MASK=1,
        )

        # Reshape back (the kernel wrote in‑place)
        left_proj = left_f.view_as(left_proj)
        right_proj = right_f.view_as(right_proj)
        # Gates are no longer needed
        del left_gate, right_gate
    else:
        # Simple element‑wise fusion when no mask is used
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate
        del left_gate, right_gate

    # -----------------------------------------------------------------
    # 7. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched matrix multiplication.
    # -----------------------------------------------------------------
    # Rearrange to (B, H, N, N) where H = hidden_dim
    left_perm = left_proj.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_proj.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)

    # Batched matmul; each (B, H) slice computes L @ R
    out = torch.matmul(left_perm, right_perm)                # (B, H, N, N)

    # Restore (B, N, N, H)
    out = out.permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate multiplication, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out, weights["to_out.weight"].to(torch.float16)
    )                                                    # (B, N, N, dim), fp16

    # Cast back to float32 as required by the API
    return out.to(torch.float32)
