@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,
    right_ptr,
    left_gate_ptr,
    right_gate_ptr,
    mask_ptr,
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # apply gates
    left = left * left_gate
    right = right * right_gate

    if HAS_MASK:
        mask_idx = offs // H
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * m
        right = right * m

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)
```

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _fuse_gate_mask_kernel(left_ptr, right_ptr,
                           left_gate_ptr, right_gate_ptr,
                           mask_ptr,
                           total: tl.int32,
                           BLOCK: tl.constexpr,
                           HAS_MASK: tl.constexpr,
                           H: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    l = l * lg
    r = r * rg

    if HAS_MASK:
        mask_idx = offs // H
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        l = l * m
        r = r * m

    tl.store(left_ptr + offs, l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)

        left, right,

        mask_ptr,
        total,

        HAS_MASK=has_mask,
        H=hidden_dim,
    )

    """
    Outgoing TriMul operator forward pass.

    Implements the AlphaFold3 TriMul module using half precision kernels for
    gate+mask fusion and a batched matrix multiplication for the O(N³) contraction.
    The final output is returned in float32.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. LayerNorm on input (float32)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # cast to fp16 for rest of pipeline
    x = x.to(torch.float16)

    # 2. Linear projections to hidden dim (fp16)
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # 3. Gating tensors (fp16)
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # 4. Fuse masks and gates via Triton kernel (in‑place)
    if not nomask:
        # mask shape [B, N, N]; broadcast over hidden dim
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        # flatten tensors for kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask_h.view(-1)  # shape B*N*N
        _apply_gate_mask(left_f, right_f, left_gate_f, right_gate_f, mask_f, hidden_dim)

        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # gates no longer needed
    del left_gate, right_gate, x

    # 5. Core contraction: out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
    # Use batched matmul across batch and hidden dimension.
    # left_perm: [B, H, N, N] (i,k), right_perm: [B, H, N, N] (k,j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)
    out_perm = torch.matmul(left_perm, right_perm)      # (B, H, N, N)
    # bring back to original layout [B, N, N, H]
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # 6. Output LayerNorm (fp16)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    # 7. Apply out gate (fp16) and final linear projection
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Cast back to float32 as required
    return out.to(torch.float32)

"""
Outgoing TriMul forward pass.

The implementation follows the AlphaFold3 TriMul module:
1. LayerNorm on the input (float32).
2. Linear projections to a hidden dimension (fp16).
3. Gate tensors via linear+sigmoid (fp16).
4. In‑place fusion of mask (optional) and gates using a Triton kernel.
5. Core O(N³) contraction computed as a batched matrix multiplication:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   performed with torch.matmul on half‑precision tensors.
6. Output LayerNorm, element‑wise out‑gate, final linear projection.
7. Result is returned in float32.

Only the mask‑gate fusion is done in Triton; the heavy GEMM leverages
cuBLAS/Tensor‑cores via torch.matmul for best throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half*  (flattened left tensor)
    right_ptr,         # half*  (flattened right tensor)
    left_gate_ptr,     # half*  (flattened left_gate)
    right_gate_ptr,    # half*  (flattened right_gate)
    mask_ptr,          # half*  (flattened mask, only used if HAS_MASK)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present, else 0
    H: tl.constexpr,          # hidden dimension (size of the last axis)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load data
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    if HAS_MASK:
        # One mask element corresponds to H consecutive hidden entries
        mask_idx = offs // H
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * m
        right = right * m

    # Store the fused results back
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of left/right with their gates and an optional mask.
    All tensors must be contiguous half‑precision and flattened.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when mask absent

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict containing model parameters (float32)
            - config: dict with keys 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm on input (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16 for the heavy lifting
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (fp16)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Compute gating tensors (fp16)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) and gates via Triton (in‑place)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> broadcast over hidden_dim
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        # Flatten everything for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask_h.view(-1)                     # shape B*N*N
        _apply_gate_and_mask(
            left_f, right_f,
            left_gate_f, right_gate_f,
            mask_f,
            hidden_dim,
        )
        # Reshape back to original layout
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # Simple element‑wise gate multiplication
        left = left * left_gate
        right = right * right_gate

    # Release unnecessary tensors
    del left_gate, right_gate, x

    # ------------------------------------------------------------------
    # 5. Core O(N³) contraction via batched matmul (half)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # left_perm: (B, H, N, N) where rows=i, cols=k
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right_perm: (B, H, N, N) where rows=k, cols=j (i.e., transposed)
    right_perm = right.permute(0, 3, 2, 1).contiguous()
    # Batched matrix multiplication on the last two dims
    out_perm = torch.matmul(left_perm, right_perm)   # (B, H, N, N)
    # Back to [B, N, N, H]
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection (half)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 7. Return in float32 as required
    # ------------------------------------------------------------------
    return out.to(torch.float32)
