    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------

        weight=weights["norm.weight"],
        bias=weights["norm.bias"],

    # ------------------------------------------------------------------

    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    right_flat = right.view(-1)
    left_gate_flat = left_gate.view(-1)
    right_gate_flat = right_gate.view(-1)

        left_gate_flat,

"""
TriMul (outgoing) forward pass.

Algorithm:
1. Layer‑norm the input tensor.
2. Linear projections to a hidden dimension (half‑precision).
3. Compute three gating tensors (sigmoid of linear projections).
4. Fuse left/right gates (and optional mask) with a Triton kernel –
   in‑place multiplication: left←left·left_gate·mask, right←right·right_gate·mask.
5. Core O(N³) contraction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched half‑precision GEMM (torch.bmm) after
   appropriate transposes.
6. Output LayerNorm → element‑wise out‑gate → final linear projection.
7. Return the result in float32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,        # half*
    right_ptr,       # half*
    left_gate_ptr,   # half*
    right_gate_ptr,  # half*
    mask_ptr,        # half* (only read if HAS_MASK)
    total_elements,  # int32
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,  # hidden dimension
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load tensors (half precision)
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # One mask element is broadcast over H hidden entries
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of left/right with their gates and (optionally) a mask.
    All tensors must be half‑precision, contiguous, and flattened.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    has_mask = 1 if mask is not None else 0
    # Provide a dummy tensor when mask is unused to satisfy the call signature
    mask_tensor = mask if mask is not None else left
    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_tensor,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )


def custom_kernel(data):
    """
    Outgoing TriMul forward pass.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/int/float) – may be unused
            - weights: dict of model parameters (float32)
            - config: dict with keys 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (half)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Gating tensors (half)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse gates (and mask) via Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        # Broadcast mask over hidden dimension and cast to half
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        mask_flat = mask_h.view(-1)  # [B * N * N]
    else:
        mask_flat = None

    # Ensure contiguity before flattening
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    left_flat = left.view(-1)
    right_flat = right.view(-1)
    left_gate_flat = left_gate.view(-1)
    right_gate_flat = right_gate.view(-1)

    _apply_gate_and_mask(
        left_flat,
        right_flat,
        left_gate_flat,
        right_gate_flat,
        mask_flat,
        hidden_dim,
    )

    # Reshape back to original layout
    left = left_flat.view_as(left)
    right = right_flat.view_as(right)

    # ------------------------------------------------------------------
    # 5. O(N³) contraction via batched GEMM (half precision)
    # ------------------------------------------------------------------
    # left: [B, N, N, H] → (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: swap the two N axes to align the summed dimension k
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Batched matrix multiplication: (i,k) × (k,j) → (i,j)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N), half

    # Restore shape [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise gating, half precision
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return in float32 as required by the API
    return out.to(torch.float32)
