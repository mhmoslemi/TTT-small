"""
TriMul forward (outgoing) implementation with a small Triton kernel.

The heavy O(N³·H) contraction is performed with a batched GEMM
(left @ rightᵀ) using torch.bmm on FP16 tensors.
A Triton kernel fuses the optional mask with the left/right gating
operations to avoid extra memory passes.
All other operations (LayerNorms, linear projections, sigmoid gates,
final LayerNorm, and output projection) are done with PyTorch,
using half precision where possible and converting the final
result back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,           # half*
    right_ptr,          # half*
    left_gate_ptr,      # half*
    right_gate_ptr,     # half*
    mask_ptr,           # half* (broadcasted)
    total_elements,     # i32
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present else 0
    H: tl.constexpr,          # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    if HAS_MASK:
        # each H entries share the same mask element
        mask_offs = offs // H
        mask_val = tl.load(mask_ptr + mask_offs, mask=valid, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """In‑place fusion of gating (and optional mask) using a Triton kernel."""
    total = left.numel()
    BLOCK = 8192
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when HAS_MASK==0

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # No return – tensors are modified in place


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or numeric) – may be None
            - weights: dict with model weights (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True) or mask is None

    # ------------------ 1. Input LayerNorm ------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.half()  # to FP16 for the heavy ops

    # ------------------ 2. Linear projections ------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )  # [B, N, N, H]
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------ 3. Gates ------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------ 4. Fuse mask with gates ---------------
    if not nomask:
        # mask: [B, N, N] -> half and flatten
        mask_h = mask.to(torch.float16).contiguous().view(-1)
        # flatten tensors for kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_h,
            hidden_dim,
        )
        # reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # simple element‑wise gating (in‑place)
        left = left * left_gate
        right = right * right_gate

    # free intermediate tensors we no longer need
    del left_gate, right_gate, x

    # ------------------ 5. Core contraction (batched GEMM) ----------
    # reshape to (B, H, N, N) for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)       # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)     # (B*H, N, N)

    # out = left @ rightᵀ for each head
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------ 6. Output processing --------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise scaling
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )
    return out.float()
