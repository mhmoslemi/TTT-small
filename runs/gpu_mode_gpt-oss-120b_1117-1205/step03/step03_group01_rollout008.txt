def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16 after LN
    x = x.to(torch.float16)

    # Linear projections to hidden_dim
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # Gates computed from original LN output x (still fp16)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(torch.float16)))

    # Apply mask before gating if present
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_h
        right = right * mask_h

    # Apply gating
    left = left * left_gate
    right = right * right_gate

    # Prepare for batched GEMM for outgoing version:
    B, N, _, _ = left.shape
    # permute to (B, hidden, N, N) and view as (B*hidden, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(-1, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(-1, N, N)

    # allocate output
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    _batched_gemm_triton(left_mat, right_mat, out_mat)

    # Reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Output layernorm over hidden_dim
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16)
    )

    # Multiply by out_gate (broadcast over hidden_dim)
    out = out * out_gate

    # Final linear projection back to dim
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return float32
    return out.to(torch.float32)

"""
Optimized outgoing TriMul forward pass.

The expensive step is the N³ contraction:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,k,j,d]
For each hidden dimension `d` this is a regular matrix multiplication
(left_{i,k} @ right_{k,j}).  We fuse this computation across the batch
and hidden dimensions using a Triton batched GEMM kernel.  All other
operations (LayerNorms, Linear projections, gating, masking) are kept in
PyTorch for simplicity; they run in fp16 to reduce memory traffic.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton batched GEMM kernel (outgoing version)
# -------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,            # pointers to fp16 tensors
    M, N, K,                        # dimensions of each matmul (M×K)·(K×N) → (M×N)
    stride_Ab, stride_Am, stride_Ak,  # strides for A: (batch, M, K)
    stride_Bb, stride_Bk, stride_Bn,  # strides for B: (batch, K, N)
    stride_Cb, stride_Cm, stride_Cn,  # strides for C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(2)      # batch index (B*hidden)
    pid_m = tl.program_id(0)         # tile row index
    pid_n = tl.program_id(1)         # tile col index

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: (BLOCK_M, BLOCK_K)
        a = tl.load(
            A_ptr
            + batch_id * stride_Ab
            + offs_m[:, None] * stride_Am
            + offs_k[None, :] * stride_Ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile: (BLOCK_K, BLOCK_N)
        b = tl.load(
            B_ptr
            + batch_id * stride_Bb
            + offs_k[:, None] * stride_Bk
            + offs_n[None, :] * stride_Bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Compute block dot product
        acc += tl.dot(a, b)

    # Write back result in fp16
    c = acc.to(tl.float16)
    c_ptrs = (
        C_ptr
        + batch_id * stride_Cb
        + offs_m[:, None] * stride_Cm
        + offs_n[None, :] * stride_Cn
    )
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

def _batched_gemm_triton(A: torch.Tensor, B: torch.Tensor, C: torch.Tensor):
    """
    Launches the Triton batched GEMM kernel for tensors:
        A: (batch, M, K)  fp16
        B: (batch, K, N)  fp16
        C: (batch, M, N)  fp16 (output)
    """
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, "Inner dimensions must match"

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch,
    )
    _batched_gemm_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        A.stride(0),
        A.stride(1),
        A.stride(2),
        B.stride(0),
        B.stride(1),
        B.stride(2),
        C.stride(0),
        C.stride(1),
        C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

# Entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: mapping of parameter names to torch.Tensor
            - config: dict containing "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------------------------------------------------------
    # 1. Input LayerNorm
    # ---------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16 for the rest of the pipeline
    x = x.to(torch.float16)

    # ---------------------------------------------------------
    # 2. Linear projections to hidden dimension
    # ---------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # ---------------------------------------------------------
    # 3. Gating signals (sigmoid)
    # ---------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x, weights["left_gate.weight"].to(torch.float16))
    )
    right_gate = torch.sigmoid(
        F.linear(x, weights["right_gate.weight"].to(torch.float16))
    )
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"].to(torch.float16))
    )

    # ---------------------------------------------------------
    # 4. Apply optional mask (before gating, as in reference)
    # ---------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_h
        right = right * mask_h

    # ---------------------------------------------------------
    # 5. Apply gates
    # ---------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------
    # 6. Core contraction (outgoing): out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,k,j,d]
    #    Implemented as a batched GEMM across (batch * hidden_dim).
    # ---------------------------------------------------------
    B, N, _, _ = left.shape  # N is sequence length
    # Rearrange to (B, hidden, N, N) and flatten batch*hidden for GEMM
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(-1, N, N)   # (B*H, M=N, K=N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(-1, N, N) # (B*H, K=N, N=N)

    out_mat = torch.empty_like(left_mat, dtype=torch.float16)          # (B*H, N, N)
    _batched_gemm_triton(left_mat, right_mat, out_mat)

    # Restore shape to (B, N, N, hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------------------------------------------------------
    # 7. Output layer norm, out‑gate and final projection
    # ---------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ---------------------------------------------------------
    # Return in float32 as specified
    # ---------------------------------------------------------
    return out.to(torch.float32)
