@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask)
    right = tl.load(right_ptr + offs, mask=mask)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask)

    if HAS_MASK:
        # using broadcast: each mask element corresponds to H consecutive hidden values
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

"""
TriMul (outgoing) forward pass – highly‑optimized hybrid implementation.

Key ideas
---------
* Fuse the five linear projections (left, right, left_gate, right_gate, out_gate)
  into a single matrix multiplication to reduce memory traffic.
* Fuse element‑wise gating and the optional pairwise mask with a tiny Triton
  kernel (in‑place on the projected tensors).
* Perform the heavy N³ contraction with a batched GEMM (torch.bmm) on
  FP16 tensors – this maps to cuBLAS‑strided‑batched matmul which is
  near‑optimal for many small matrices.
* Keep LayerNorms and the final projection in PyTorch (FP16) and return
  the result in FP32 as required.

The implementation respects the required signature:
    custom_kernel(data) → torch.Tensor
where `data` = (input, mask, weights, config).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – fuse mask (if any) with left/right gating
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half*   [total]
    right_ptr,         # half*   [total]
    left_gate_ptr,     # half*   [total]
    right_gate_ptr,    # half*   [total]
    mask_ptr,          # half*   [mask_total] (broadcast)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask is present else 0
    H: tl.constexpr,          # hidden dimension (size of gate vector)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # mask has shape [B, N, N] -> broadcast over H
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place: left = left * left_gate * mask; right = right * right_gate * mask
    All tensors must be contiguous torch.float16 on the same CUDA device.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when HAS_MASK==0

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # No return; operation performed in‑place


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/float) or None
            - weights      : dict of model parameters (float32 tensors)
            - config       : dict with keys ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    #########################
    # 1. Input LayerNorm
    #########################
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    ##########################################
    # 2. Fused projection for left/right/gates
    ##########################################
    # Concatenate the five weight matrices (out_features first)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # shape (5*hidden_dim, dim)

    # Single linear pass
    proj = F.linear(x_h, combined_weight)  # (B, N, N, 5*hidden_dim)

    # Split into individual tensors
    H = hidden_dim
    left = proj[..., 0 * H : 1 * H]
    right = proj[..., 1 * H : 2 * H]
    left_gate = proj[..., 2 * H : 3 * H]
    right_gate = proj[..., 3 * H : 4 * H]
    out_gate = proj[..., 4 * H : 5 * H]

    # Sigmoids for gates
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    # ---------------------------------------------------------------
    # 3. Fuse mask (if any) with the element‑wise gating
    # ---------------------------------------------------------------
    if not nomask and mask is not None:
        # mask broadcasting over hidden_dim
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        # Flatten everything for the kernel
        left_flat = left.view(-1)
        right_flat = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)
        mask_flat = mask_h.view(-1)  # size = B * N * N
        _apply_gate_and_mask(
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            hidden_dim,
        )
        # Reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # Simple element‑wise multiplication when no mask
        left = left * left_gate
        right = right * right_gate

    ##########################################################
    # 4. Core contraction  out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as a batched GEMM (B*H, N, N) x (B*H, N, N)^T
    ##########################################################
    # Bring hidden dimension to batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) with k dimension transposed

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # Batched matrix multiplication (FP16 inputs, FP32 accumulation)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N), dtype=half

    # Restore original layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    ######################################
    # 5. Output LayerNorm, out‑gate, proj
    ######################################
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
    )

    # Return in float32 as required
    return out.to(torch.float32)
