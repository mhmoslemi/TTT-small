@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)  # batch index
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # Offsets for each block
    block_row = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    block_col = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # within A and B: iterate over K
    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute offsets for A (pid_b, block_row, k_offset)
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # Load tile from A and B with mask
        a_mask = (block_row < M) & (k_offset < K)
        b_mask = (k_offset < K) & (block_col < N)
        A = tl.load(A_ptr + pid_b * stride_Ab + block_row[:, None] * stride_Am + k_offset[None, :] * stride_Ak, mask=a_mask, other=0.0)
        B = tl.load(B_ptr + pid_b * stride_Bb + k_offset[:, None] * stride_Bk + block_col[None, :] * stride_Bn, mask=b_mask, other=0.0)
        # dot

        # increment pointers? Not needed because we compute offsets each iteration
    # Write out result

    tl.store(C_ptr + pid_b * stride_Cb + block_row[:, None] * stride_Cm + block_col[None, :] * stride_Cn, acc.to(tl.float16), mask=c_mask)

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

Algorithm
---------
1. LayerNorm the input (float32) → half.
2. Linear projection to hidden dimension for left/right tensors (half).
3. Compute left/right/out gates (half, sigmoid) and optionally apply mask.
4. Core contraction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication L @ Rᵀ for each head h.
   Implemented as a custom Triton kernel that processes all (batch * hidden)
   matrices in parallel.
5. Output LayerNorm over hidden dimension, multiply by out_gate, final linear
   projection back to the original dimension.
6. Return float32 output.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,           # pointers
    M, N, K,                       # matrix sizes (all int32)
    stride_Ab, stride_Am, stride_Ak,   # strides for A: (batch, row, col)
    stride_Bb, stride_Bk, stride_Bn,   # strides for B: (batch, row, col)
    stride_Cb, stride_Cm, stride_Cn,   # strides for C: (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B   (A,B,C are [batch, M, K], [batch, K, N], [batch, M, N])"""
    pid_b = tl.program_id(0)   # batch index
    pid_m = tl.program_id(1)   # block row index
    pid_n = tl.program_id(2)   # block col index

    # Compute start offsets for this block
    block_row = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    block_col = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows/cols
    mask_row = block_row < M
    mask_col = block_col < N

    # Accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        block_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = block_k < K

        # Load tiles of A (M × K) and B (K × N)
        a_ptrs = (
            A_ptr
            + pid_b * stride_Ab
            + block_row[:, None] * stride_Am
            + block_k[None, :] * stride_Ak
        )
        b_ptrs = (
            B_ptr
            + pid_b * stride_Bb
            + block_k[:, None] * stride_Bk
            + block_col[None, :] * stride_Bn
        )
        a = tl.load(a_ptrs, mask=mask_row[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_col[None, :], other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Write out result in float16
    c_ptrs = (
        C_ptr
        + pid_b * stride_Cb
        + block_row[:, None] * stride_Cm
        + block_col[None, :] * stride_Cn
    )
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_row[:, None] & mask_col[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or numeric) or None
            - weights: mapping of parameter names to torch.Tensor
            - config: dict containing ``dim`` (input dimension) and
                      ``hidden_dim`` (hidden dimension). May contain ``nomask``.

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) -> half
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # float32
    x = x.to(torch.float16)  # half for the rest

    # ------------------------------------------------------------------
    # 2. Linear projections (half)
    # ------------------------------------------------------------------
    left = F.linear(
        x, weights["left_proj.weight"].to(torch.float16)
    )  # [B,N,N,hidden]
    right = F.linear(
        x, weights["right_proj.weight"].to(torch.float16)
    )

    # ------------------------------------------------------------------
    # 3. Gates (half, sigmoid)
    # ------------------------------------------------------------------
    left_gate = (
        F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    )
    right_gate = (
        F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    )
    out_gate = (
        F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()
    )

    # ------------------------------------------------------------------
    # 4. Apply mask (if any) and gates
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B,N,N] -> [B,N,N,1]
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        left = left * left_gate * mask_h
        right = right * right_gate * mask_h
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: batched GEMM via Triton
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as left @ rightᵀ for each head.
    # ------------------------------------------------------------------
    # Permute to shape (B, H, N, N) where the last two dims are the matrices.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) = right transposed

    B, H, N, _ = left_perm.shape
    batch = B * H

    # reshape to [batch, N, N]
    A = left_perm.view(batch, N, N)
    B_mat = right_perm.view(batch, N, N)

    # allocate output
    C = torch.empty_like(A)  # half, shape (batch, N, N)

    # Block sizes (tune for your hardware; 128x128 works well on A100/B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _batched_matmul_kernel[grid](
        A_ptr=A.data_ptr(),
        B_ptr=B_mat.data_ptr(),
        C_ptr=C.data_ptr(),
        M=N,
        N=N,
        K=N,
        stride_Ab=A.stride(0),
        stride_Am=A.stride(1),
        stride_Ak=A.stride(2),
        stride_Bb=B_mat.stride(0),
        stride_Bk=B_mat.stride(1),
        stride_Bn=B_mat.stride(2),
        stride_Cb=C.stride(0),
        stride_Cm=C.stride(1),
        stride_Cn=C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B,N,N,H)
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise gating (half)

    out = F.linear(
        out, weights["to_out.weight"].to(torch.float16)
    )  # (B,N,N,dim), half

    # Convert back to float32 as required
    return out.to(torch.float32)
