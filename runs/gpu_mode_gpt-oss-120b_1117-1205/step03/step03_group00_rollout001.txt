"""
TriMul forward (outgoing) with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor (float32) → half.
2. Linear projections to a hidden dimension `H` (half precision):
   left = x·W_left,   right = x·W_right.
3. Compute three sigmoid gates (half):
   left_gate, right_gate, out_gate.
4. Fuse mask (if present) and the left/right gates in a single Triton kernel,
   writing the results back in‑place on `left` and `right`.
5. Core contraction:
   out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   which is a batched GEMM:
   (B, H, N, N) → (B·H, N, N) @ (B·H, N, N)^T.
6. Output LayerNorm, scale by `out_gate`, final linear projection back to `dim`.
7. Return float32 tensor.

Only the mask‑gate fusion is performed in Triton; the O(N³) contraction uses
high‑performance cuBLAS via `torch.bmm`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: in‑place fusion of left/right gates and optional mask.
# --------------------------------------------------------------------------- #
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half*
    right_ptr,         # half*
    left_gate_ptr,     # half*
    right_gate_ptr,    # half*
    mask_ptr,          # half* (broadcast over hidden dim)
    total_elements,    # i32
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present, else 0
    H: tl.constexpr,          # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=m, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=m, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=m, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=m, other=tl.float16(0.0))

    if HAS_MASK:
        # each hidden channel shares the same mask entry
        mask_off = offs // H
        mask_val = tl.load(mask_ptr + mask_off, mask=m, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Write‑back
    tl.store(left_ptr + offs, left, mask=m)
    tl.store(right_ptr + offs, right, mask=m)


def _fuse_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place element‑wise multiplication:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    `mask` may be None → only gates are applied.
    All tensors must be contiguous half‑precision.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when no mask

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
        num_warps=4,
    )
    # No return; tensors modified in‑place


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.
    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (bool or numeric) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------ #
    # 1. Input LayerNorm (float32 → half)
    # ------------------------------------------------------------------ #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 2. Linear projections (half)
    # ------------------------------------------------------------------ #
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------ #
    # 3. Gates (half)
    # ------------------------------------------------------------------ #
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------ #
    # 4. Fuse mask (if any) and gates via Triton
    # ------------------------------------------------------------------ #
    if not nomask and mask is not None:
        # mask: [B, N, N] → flatten, half
        mask_h = mask.to(torch.float16).contiguous().view(-1)  # [B*N*N]
        # Flatten tensors for kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _fuse_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_h,
            hidden_dim,
        )
        # Reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask → simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # Gates are no longer needed
    del left_gate, right_gate, x_h

    # ------------------------------------------------------------------ #
    # 5. Core O(N³) contraction via batched GEMM
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------ #
    # (B, N, N, H) → (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)       # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)     # (B*H, N, N)

    # Batched matrix multiply with transposition on the second operand
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------ #
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------ #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise scaling
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in original dtype
    return out.to(torch.float32)
