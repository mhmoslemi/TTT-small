@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr,
    mask_ptr, total_elements: tl.int32, hidden_dim: tl.int32,
    HAS_MASK: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

"""
TriMul (outgoing) forward pass with a Triton kernel that fuses mask and gating.

Algorithm
---------
1. LayerNorm on the input tensor (float32) → half precision.
2. Linear projections `left` and `right` to `hidden_dim` (half precision).
3. Compute three gates (left, right, out) via linear + sigmoid (half).
4. Fuse mask (if present) and left/right gating into a single Triton kernel
   that operates in‑place on the flattened tensors.
5. Core contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   implemented as a batched half‑precision GEMM (left @ rightᵀ) using torch.bmm.
6. Apply a second LayerNorm, the output gate, and the final linear projection.
7. Return the result in float32.

Only the mask‑gate fusion is performed in Triton; the remaining heavy
computation uses cuBLAS (torch.bmm), which is already optimal on B200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse optional mask with left/right gating (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # half*  (flattened left)
    right_ptr,           # half*  (flattened right)
    left_gate_ptr,       # half*  (flattened left_gate)
    right_gate_ptr,      # half*  (flattened right_gate)
    mask_ptr,            # half*  (flattened mask, broadcasted over hidden_dim)
    total_elements: tl.int32,   # B * N * N * hidden_dim
    hidden_dim: tl.int32,       # H
    HAS_MASK: tl.constexpr,     # 1 if mask present, 0 otherwise
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # each mask entry corresponds to `hidden_dim` consecutive channel values
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(
    left: torch.Tensor,
    right: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor | None,
    hidden_dim: int,
    has_mask: bool,
) -> None:
    """
    In‑place fusion of:
        left = left * left_gate * mask
        right = right * right_gate * mask
    (mask optional). All tensors must be contiguous half‑precision.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    mask_ptr = mask if mask is not None else left  # dummy when HAS_MASK=0

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        hidden_dim,
        HAS_MASK=1 if has_mask else 0,
        BLOCK=BLOCK,
    )
    # No return; tensors are modified in place.


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict containing
                * "dim" (int) – input channel size C
                * "hidden_dim" (int) – projection size H
                * "nomask" (bool, optional) – True if mask should be ignored

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 → half)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias = weights["norm.bias"].to(torch.float16)

    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x_h = x.to(torch.float16).contiguous()   # [B, N, N, dim] half

    # ------------------------------------------------------------------
    # 2. Linear projections (half‑precision)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_weight)   # [B, N, N, hidden_dim] half
    right = F.linear(x_h, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Gating tensors (half‑precision)
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_h, left_gate_weight).sigmoid()
    right_gate = F.linear(x_h, right_gate_weight).sigmoid()
    out_gate = F.linear(x_h, out_gate_weight).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) and gating via Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape [B, N, N] → broadcast over hidden_dim
        mask_h = mask.to(torch.float16).contiguous()
        # Flatten tensors for the kernel
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_h.view(-1)   # [B*N*N]

        _apply_gate_and_mask(
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            hidden_dim,
            has_mask=True,
        )
        # Reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask: simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # Gates no longer needed
    del left_gate, right_gate, x_h

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) and transpose right for GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # Batched matrix multiplication in half precision
    out_mat = torch.bmm(left_mat, right_mat)   # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate   # element‑wise gating (half)

    to_out_weight = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_weight)   # (B, N, N, dim) half

    # ------------------------------------------------------------------
    # 7. Return in float32 as required
    # ------------------------------------------------------------------
    return out.to(torch.float32)
