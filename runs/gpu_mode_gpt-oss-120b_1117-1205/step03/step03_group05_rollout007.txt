"""
TriMul (outgoing) forward pass.

The implementation fuses:
  * Five linear projections (left/right and three gates) into a single matmul.
  * Gate multiplication together with optional mask using a tiny Triton kernel.
The O(N³) contraction is performed with a batched gemm (torch.bmm) on half‑precision
tensors, which is highly optimized in cuBLAS.  The final layer‑norm, gating and
output projection are carried out in PyTorch.  The result is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half*  (flattened left tensor)
    right_ptr,         # half*  (flattened right tensor)
    left_gate_ptr,     # half*  (flattened left_gate)
    right_gate_ptr,    # half*  (flattened right_gate)
    mask_ptr,          # half*  (flattened mask) – only read if HAS_MASK=1
    total_elems: tl.int32,
    H: tl.constexpr,   # hidden dimension
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    if HAS_MASK:
        # One mask element corresponds to H consecutive hidden entries
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * mask_val
        right = right * mask_val

    # Write back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask):
    """
    In‑place fusion of left/right with their gates and an optional mask.
    All tensors must be half‑precision and contiguous.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when no mask

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        left.shape[-1],          # H
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
    )


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or numeric) – may be ignored
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    ##################################################################
    # 1. Input LayerNorm (float32) → half
    ##################################################################
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    ##################################################################
    # 2. Fuse all five linear projections (left, right, left_gate,
    #    right_gate, out_gate) into a single matmul.
    ##################################################################
    # Concatenate weight matrices: each is (hidden_dim, dim)
    w_left = weights["left_proj.weight"]
    w_right = weights["right_proj.weight"]
    w_lgate = weights["left_gate.weight"]
    w_rgate = weights["right_gate.weight"]
    w_outgate = weights["out_gate.weight"]
    weight_concat = torch.cat([w_left, w_right, w_lgate, w_rgate, w_outgate], dim=0)  # (5*H, dim)
    weight_concat = weight_concat.to(torch.float16)

    # Flatten input and apply the big linear
    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)                               # (B*N*N, dim)
    y = F.linear(x_flat, weight_concat)                    # (B*N*N, 5*H)
    y = y.view(B, N, N, hidden_dim * 5)                    # (B, N, N, 5*H)

    # Split into the five components
    left = y[..., 0 * hidden_dim : 1 * hidden_dim]
    right = y[..., 1 * hidden_dim : 2 * hidden_dim]
    left_gate = y[..., 2 * hidden_dim : 3 * hidden_dim].sigmoid()
    right_gate = y[..., 3 * hidden_dim : 4 * hidden_dim].sigmoid()
    out_gate = y[..., 4 * hidden_dim : 5 * hidden_dim].sigmoid()

    ##################################################################
    # 3. Apply gates (and optional mask) in‑place via Triton kernel
    ##################################################################
    if not nomask and mask is not None:
        # Broadcast mask over hidden dimension and flatten
        mask_h = mask.to(torch.float16).unsqueeze(-1).contiguous()
        mask_flat = mask_h.view(-1)               # length B*N*N
    else:
        mask_flat = None

    # Ensure contiguity before passing to kernel
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    _apply_gate_and_mask(left, right, left_gate, right_gate, mask_flat)

    # Gates are no longer needed
    del left_gate, right_gate, x

    ##################################################################
    # 4. Core O(N³) contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    ##################################################################
    # Re‑order tensors to (B, H, N, N) for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()     # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)                # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)              # (B*H, N, N)

    # Batched half‑precision GEMM (uses cuBLAS)
    out_mat = torch.bmm(left_mat, right_mat)              # (B*H, N, N)

    # Restore original layout (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    ##################################################################
    # 5. Output LayerNorm, out_gate scaling, final projection
    ##################################################################
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate                                 # element‑wise gating
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return as float32
    return out.to(torch.float32)
