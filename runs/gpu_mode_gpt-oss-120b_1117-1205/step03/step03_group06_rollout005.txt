```python
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,
    right_ptr,
    left_gate_ptr,
    right_gate_ptr,
    mask_ptr,
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    lgate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * lgate * mask_val
        right = right * rgate * mask_val
    else:
        left = left * lgate
        right = right * rgate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)
```

"""
Outgoing TriMul forward pass.

- LayerNorm on the input (float32), then cast to fp16.
- Linear projections to hidden dimension and gate values (fp16).
- Optional mask + gate fusion performed by a Triton kernel (in‑place).
- Core contraction: for each batch and hidden channel compute
  out_{i,j} = Σ_k left_{i,k} * right_{j,k}  -> implemented as
  batched GEMM (left @ rightᵀ) using torch.bmm (fp16, tensor‑cores).
- Output LayerNorm, gating and final linear projection back to original dim.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # fp16 *
    right_ptr,         # fp16 *
    left_gate_ptr,     # fp16 *
    right_gate_ptr,    # fp16 *
    mask_ptr,          # fp16 *
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present else 0
    H: tl.constexpr,          # hidden dimension (channels)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # Load data
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    lgate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # One mask element corresponds to H consecutive channel entries.
        mask_idx = offs // H
        mval = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * lgate * mval
        right = right * rgate * mval
    else:
        left = left * lgate
        right = right * rgate

    # Store results back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of ``left = left * left_gate * mask`` and
    ``right = right * right_gate * mask`` (mask optional) using Triton.
    All tensors must be contiguous and dtype torch.float16.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when unused

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict with the following keys (float32 tensors):
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) -> fp16
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (fp16)
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Gate values (fp16)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) and gates via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> broadcast over hidden_dim
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        # Flatten tensors for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask_h.view(-1)               # shape [B*N*N]
        _apply_gate_and_mask(
            left_f, right_f,
            left_gate_f, right_gate_f,
            mask_f,
            hidden_dim,
        )
        # Reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # Simple elementwise multiplication when no mask
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as batched matmul: out = left @ rightᵀ
    # ------------------------------------------------------------------
    # Bring hidden dimension in front for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N) where axes are (i, k)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N) where axes are (j, k)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)               # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)             # (B*H, N, N)

    # Batched matmul with transpose on the right operand
    out_mat = torch.bmm(left_mat, right_mat.transpose(2, 1))  # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # elementwise fp16
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return in float32 as required
    return out.to(torch.float32)
