"""
TriMul (outgoing) forward pass – Triton‑accelerated mask & gating.

Algorithm
---------
1. LayerNorm on the input (float32 → float16 for the heavy part).
2. Linear projections to hidden_dim (half‑precision).
3. Compute left/right gates (sigmoid of linear) – also half.
4. Fuse mask (if present) and gating into the tensors with a tiny Triton
   kernel (in‑place). This satisfies the “must‑use‑a‑kernel” requirement.
5. Core N³ contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication:
       out = (left_permuted) @ (right_permuted)ᵀ
   performed via torch.bmm on (B·H) matrices of size N×N.
6. Output LayerNorm, element‑wise out‑gate multiplication and final linear
   projection back to dim.
7. Cast the result to float32.

The implementation keeps the large O(N³) work on the GPU‑optimized cuBLAS
(bmm) and uses Triton only for the mask‑gate fusion, which is the
most expensive element‑wise step when a mask is present.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # fp16 *
    right_ptr,         # fp16 *
    left_gate_ptr,     # fp16 *
    right_gate_ptr,    # fp16 *
    mask_ptr,          # fp16 * (optional)
    total_elements,    # int32
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask is present, else 0
    H: tl.constexpr,          # hidden dimension (size of innermost channel)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load data
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each mask element repeats H times along the hidden dimension
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store back (in‑place)
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of:
        left  = left  * left_gate  * (mask)
        right = right * right_gate * (mask)

    All tensors must be contiguous and of dtype torch.float16.
    `mask` may be None (no mask). `hidden_dim` is the size of the innermost
    channel dimension.
    """
    total = left.numel()
    BLOCK = 1024  # reasonable trade‑off for large 1‑D launch
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    # Provide a dummy pointer when mask is not used (will never be dereferenced)
    mask_ptr = mask if mask is not None else left

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # The operation is done in‑place; nothing is returned.


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor:  torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict: mapping of parameter names to torch.Tensor
            - config_dict: must contain "dim" and "hidden_dim";
                           may contain "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 → keep as float32 for LN)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to half‑precision for the heavy work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (half)
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 4. Gates (half, sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 5. Fuse mask and gates with a Triton kernel (if needed)
    # ------------------------------------------------------------------
    use_mask = (not nomask) and (mask_tensor is not None)
    if use_mask:
        mask_h = mask_tensor.to(torch.float16).contiguous()
        # flatten tensors for the 1‑D kernel
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_h.view(-1)  # shape [B*N*N]

        _apply_gate_and_mask(
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            hidden_dim,
        )
        # reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask – simple element‑wise gating
        left = left * left_gate
        right = right * right_gate

    # Gates are no longer needed
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6. Core N³ contraction via batched GEMM:
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    → left_permuted @ right_permutedᵀ
    # ------------------------------------------------------------------
    # left: (B, N, N, H) → (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: (B, N, N, H) → (B, H, N, N) with i/j swapped (transpose on the
    #   two sequence dimensions) to implement the required inner‑product.
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape
    # reshape to batched matrices of shape (B*H, N, N)
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Batched matrix multiplication (FP16 inputs, FP16 output)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16)
    )

    # Return float32 as required by the API
    return out.to(torch.float32)
