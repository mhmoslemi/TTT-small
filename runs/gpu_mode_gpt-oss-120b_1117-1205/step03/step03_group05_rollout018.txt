"""
TriMul (outgoing) forward pass – Triton‑accelerated contraction.

Algorithm
---------
1. LayerNorm on the input tensor (float32 → float16).
2. Linear projections to a hidden dimension `H` (half precision).
3. Compute three sigmoid gates (half precision).
4. Apply mask (if present) and element‑wise gates (still in PyTorch).
5. Core O(N³) contraction:
     out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a custom Triton kernel that tiles the (i,j) output
   plane (BLOCK_M × BLOCK_N) and accumulates over k in blocks
   (BLOCK_K).  The hidden dimension `H` is folded into the first grid
   dimension (batch × hidden).
6. Output LayerNorm, element‑wise out‑gate and final linear projection
   back to the original dimension.
7. Return float32 tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Grid:
        program_id(0) -> batch * hidden dimension (b, h)
        program_id(1) -> i‑tile
        program_id(2) -> j‑tile
    """
    pid0 = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # decode batch and hidden index from pid0
    b = pid0 // H
    h = pid0 % H

    # tile offsets
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # range within the tile
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction dimension k
    for k_start in range(0, N, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + (
            b * stride_l_b
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k
            + h * stride_l_h
        )
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right tile: (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + (
            b * stride_r_b
            + offs_k[:, None] * stride_r_k
            + offs_j[None, :] * stride_r_j
            + h * stride_r_h
        )
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        acc += tl.dot(left, right)

    # store the result
    out_ptrs = out_ptr + (
        b * stride_o_b
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
        + h * stride_o_h
    )
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (half)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Gating tensors (half)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 4. Apply mask (if any) and gates (element‑wise, cheap)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.unsqueeze(-1).to(torch.float16)
        left = left * left_gate * mask_h
        right = right * right_gate * mask_h
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³) contraction via Triton kernel
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty((B, N, N, H), dtype=torch.float16, device=device)

    # strides for the contiguous tensors
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # block sizes (tuned for typical N≤1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                                 # batch * hidden dimension
        triton.cdiv(N, BLOCK_M),               # i‑tiles
        triton.cdiv(N, BLOCK_N),               # j‑tiles
    )
    _tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 7. Return in float32
    # ------------------------------------------------------------------
    return out.to(torch.float32)
