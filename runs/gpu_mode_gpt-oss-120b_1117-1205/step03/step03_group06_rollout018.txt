def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape (or just consider shape)
    hidden_dim = config["hidden_dim"]
    dim = config["dim"]
    nomask = config.get("nomask", True)

    # Stage 1: LN and cast to half
    # Using torch.nn.functional.layer_norm (float32 support). We'll compute LN in float32 then cast to half for fast ops.

    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x_h = x.to(torch.float16)

    # linear projections
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # gates
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # fuse gating and mask
    if not nomask and mask is not None:
        # mask shape (B,N,N) -> unsqueeze(-1)
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()  # shape (B,N,N,1)
        # flatten all but last? Actually mask broadcast across hidden_dim, but we will treat mask as shape (B,N,N) for kernel (no channel).
        # We'll pass mask without the last dim; kernel treats mask length = B*N*N
        mask_flat = mask_h.view(-1)  # shape B*N*N

        left_view = left.contiguous().view(-1)
        right_view = right.contiguous().view(-1)
        left_gate_view = left_gate.contiguous().view(-1)
        right_gate_view = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(left_view, right_view, left_gate_view, right_gate_view, mask_flat, hidden_dim)

        left = left_view.view_as(left)
        right = right_view.view_as(right)
    else:
        # no mask: just elementwise multiply with gates
        left = left * left_gate
        right = right * right_gate

    # Maybe out_gate shape: (B,N,N,hidden_dim). Out gate applied after output norm, so we keep.

    # tri-multiplicative contraction:
    # permute left to (B, hidden_dim, N, N)
    left_perm = left.permute(0,3,1,2).contiguous()
    # right permute: (B, hidden_dim, N, N) -> we need right transpose on the second N dimension (k dimension)
    # In code, they used right.permute(0,3,2,1) which transposes the two spatial dims; this yields shape (B, hidden_dim, N, N) where axes swapped.
    right_perm = right.permute(0,3,2,1).contiguous()

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Use torch.bmm which uses half if inputs half.
    out_mat = torch.bmm(left_mat, right_mat)  # shape (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    # out shape (B,N,N,H)

    # output layernorm
    out = F.layer_norm(out, (hidden_dim,),
                       weight=weights["to_out_norm.weight"].to(torch.float16),
                       bias=weights["to_out_norm.bias"].to(torch.float16))

    # out_gate multiplication (out_gate already shape (B,N,N,H))

    # final linear projection to dim
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Cast to float32 for output
    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass optimized with Triton.

Algorithm
---------
1. Layer‑norm on the input tensor (float32) and cast to fp16.
2. Linear projections `left` and `right` from `dim` → `hidden_dim`.
3. Compute three gate tensors (sigmoid of linear projections):
   `left_gate`, `right_gate`, `out_gate`.
4. Fuse element‑wise mask (if present) and gate multiplications into
   `left` and `right` with a custom Triton kernel.  The kernel processes
   the flattened tensors: for each element `e` belonging to a spatial
   position (b,i,j) it computes
        left[e]  = left[e]  * left_gate[e]  * mask[b,i,j]
        right[e] = right[e] * right_gate[e] * mask[b,i,j]
   If no mask is supplied the kernel only applies the gate.
5. Perform the core contraction
        out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   implemented as a batched GEMM: after permuting dimensions,
   `left` and `right` become (B*hidden_dim, N, N) matrices and are
   multiplied with `torch.bmm`.
6. Apply a second LayerNorm on `out`, multiply by `out_gate`,
   and linearly project back to `dim`.
7. Return the result in float32.

Only the mask‑gate fusion is written in Triton; the heavy O(N³) contraction
uses cuBLAS‑accelerated batched matrix multiplication.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------------
# Triton kernel: fuse mask and gate multiplications (fp16)
# ---------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,       # fp16 *
    right_ptr,      # fp16 *
    left_gate_ptr,  # fp16 *
    right_gate_ptr, # fp16 *
    mask_ptr,       # fp16 * (optional)
    total_elems: tl.int32,          # total number of elements in left/right
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,         # 1 if mask is present, else 0
    H: tl.constexpr,                # hidden dimension (size of innermost channel)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elems

    # Load tensors
    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    lgate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    if HAS_MASK:
        # One mask entry per spatial position (B,N,N); each position spans H channels.
        mask_idx = offs // H
        mval = tl.load(mask_ptr + mask_idx, mask=valid, other=0.0)
        left = left * lgate * mval
        right = right * rgate * mval
    else:
        left = left * lgate
        right = right * rgate

    # Store the results back in‑place
    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of mask and gates using the Triton kernel.
    All tensors must be contiguous ``torch.float16``.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when unused

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # The tensors `left` and `right` are modified in place.


# ---------------------------------------------------------------
# Entry point
# ---------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or numeric) or None
            - weights: dict mapping parameter names to tensors (float32)
            - config: dict with keys ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    with torch.no_grad():
        input_tensor, mask, weights, config = data
        device = input_tensor.device

        dim = config["dim"]
        hidden_dim = config["hidden_dim"]
        nomask = config.get("nomask", True)

        # ------------------------------------------------------------------
        # 1. Input LayerNorm (float32) and cast to fp16
        # ------------------------------------------------------------------
        x = F.layer_norm(
            input_tensor,
            (dim,),
            weight=weights["norm.weight"],
            bias=weights["norm.bias"],
        )
        x_h = x.to(torch.float16)

        # ------------------------------------------------------------------
        # 2. Linear projections (fp16)
        # ------------------------------------------------------------------
        left_proj_w = weights["left_proj.weight"].to(torch.float16)
        right_proj_w = weights["right_proj.weight"].to(torch.float16)

        left = F.linear(x_h, left_proj_w)   # (B,N,N,hidden_dim)
        right = F.linear(x_h, right_proj_w)

        # ------------------------------------------------------------------
        # 3. Gate values (fp16)
        # ------------------------------------------------------------------
        left_gate_w = weights["left_gate.weight"].to(torch.float16)
        right_gate_w = weights["right_gate.weight"].to(torch.float16)
        out_gate_w = weights["out_gate.weight"].to(torch.float16)

        left_gate = F.linear(x_h, left_gate_w).sigmoid()   # (B,N,N,hidden_dim)
        right_gate = F.linear(x_h, right_gate_w).sigmoid()
        out_gate = F.linear(x_h, out_gate_w).sigmoid()     # used later

        # ------------------------------------------------------------------
        # 4. Fuse mask (if any) and gates via Triton
        # ------------------------------------------------------------------
        if not nomask and mask is not None:
            # mask: (B,N,N) -> fp16, broadcast over hidden_dim
            mask_h = mask.to(torch.float16).contiguous()
            mask_flat = mask_h.view(-1)  # length B*N*N

            # Flatten tensors for the 1‑D kernel
            left_f = left.contiguous().view(-1)
            right_f = right.contiguous().view(-1)
            left_gate_f = left_gate.contiguous().view(-1)
            right_gate_f = right_gate.contiguous().view(-1)

            _apply_gate_and_mask(
                left_f,
                right_f,
                left_gate_f,
                right_gate_f,
                mask_flat,
                hidden_dim,
            )
            # Reshape back to original tensors
            left = left_f.view_as(left)
            right = right_f.view_as(right)
        else:
            # Simple element‑wise gate multiplication (no mask)
            left = left * left_gate
            right = right * right_gate

        # ------------------------------------------------------------------
        # 5. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
        #    Implemented as a batched GEMM (B*hidden_dim, N, N)
        # ------------------------------------------------------------------
        # (B, N, N, H) -> (B, H, N, N)
        left_perm = left.permute(0, 3, 1, 2).contiguous()
        # right needs k‑dimension transposed: (B, N, N, H) -> (B, H, N, N) with axes (j,k) swapped
        right_perm = right.permute(0, 3, 2, 1).contiguous()

        B, H, N, _ = left_perm.shape
        left_mat = left_perm.view(B * H, N, N)
        right_mat = right_perm.view(B * H, N, N)

        # Batched matrix multiplication (uses Tensor Cores for fp16)
        out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)

        # Restore (B, N, N, H) layout
        out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

        # ------------------------------------------------------------------
        # 6. Output LayerNorm, out‑gate, final linear projection
        # ------------------------------------------------------------------
        to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
        to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

        out = F.layer_norm(
            out,
            (hidden_dim,),
            weight=to_out_norm_w,
            bias=to_out_norm_b,
        )

        out = out * out_gate  # element‑wise gating (fp16)

        to_out_w = weights["to_out.weight"].to(torch.float16)
        out = F.linear(out, to_out_w)

        # ------------------------------------------------------------------
        # 7. Cast back to float32 for the final output
        # ------------------------------------------------------------------
        return out.to(torch.float32)
