@triton.jit
def fuse_gate_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    N: tl.constexpr, H: tl.constexpr, HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    # we treat linearized index over elements
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = N * N * H
    mask_len = N * N
    mask_valid = offs < total
    # load
    left = tl.load(left_ptr + offs, mask=mask_valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask_valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask_valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask_valid, other=0.0)
    if HAS_MASK:
        mask_idx = offs // H  # each mask element corresponds to position (i,k) or (j,k)
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate
    tl.store(left_ptr + offs, left, mask=mask_valid)
    tl.store(right_ptr + offs, right, mask=mask_valid)
```

"""
TriMul (outgoing) forward pass.

- Performs LayerNorm on the input.
- Computes left/right projections and gating in half precision.
- Fuses mask and gate multiplication with a tiny Triton kernel.
- Executes the O(N³) contraction using torch.einsum (batched matrix multiply).
- Applies output LayerNorm, out‑gate, and final projection.
All heavy arithmetic is in float16; the result is returned as float32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse gate and optional mask (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # float16 *
    right_ptr,         # float16 *
    left_gate_ptr,     # float16 *
    right_gate_ptr,    # float16 *
    mask_ptr,          # float16 *
    total_elements: tl.int32,
    H: tl.constexpr,          # hidden dimension
    HAS_MASK: tl.constexpr,   # 1 if mask present else 0
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # each mask element corresponds to H consecutive channel entries
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor:  torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict: mapping parameter name -> torch.Tensor (float32)
            - config_dict: must contain "dim" and "hidden_dim",
                           may contain "nomask" (bool)
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim] float32

    # ------------------------------------------------------------------
    # 2. Cast to half precision for the heavy work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    B, N, _, _ = x_h.shape
    # flatten the first three dims for the linear layers
    x_flat = x_h.reshape(-1, dim)                    # (B*N*N, dim)

    # ------------------------------------------------------------------
    # 3. Linear projections (half)
    # ------------------------------------------------------------------
    left = F.linear(x_flat, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_flat, weights["right_proj.weight"].to(torch.float16))
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4. Gating (half)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_flat, weights["left_gate.weight"].to(torch.float16)
    ).view(B, N, N, hidden_dim).sigmoid()
    right_gate = F.linear(
        x_flat, weights["right_gate.weight"].to(torch.float16)
    ).view(B, N, N, hidden_dim).sigmoid()
    out_gate = F.linear(
        x_flat, weights["out_gate.weight"].to(torch.float16)
    ).view(B, N, N, hidden_dim).sigmoid()

    # ------------------------------------------------------------------
    # 5. Fuse mask and gating via Triton (if mask is present)
    # ------------------------------------------------------------------
    if (not nomask) and (mask is not None):
        # mask shape [B,N,N] -> broadcast on hidden_dim
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        # flatten everything for the kernel
        left_f = left.view(-1)
        right_f = right.view(-1)
        left_gate_f = left_gate.view(-1)
        right_gate_f = right_gate.view(-1)
        mask_f = mask_h.view(-1)               # length B*N*N

        total = left_f.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)

        _gate_mask_fuse_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            total,
            H=hidden_dim,
            HAS_MASK=1,
            BLOCK=BLOCK,
        )
        # After kernel, left/right already contain the gated&masked values
    else:
        # No mask → simple elementwise multiplication
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented via torch.einsum which maps to an efficient batched GEMM.
    # ------------------------------------------------------------------
    # all tensors are half, result stays half
    out = torch.einsum("bikd,bjkd->bijd", left, right)  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final projection (half)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 8. Cast back to float32 for the API contract
    # ------------------------------------------------------------------
    return out.to(torch.float32)
