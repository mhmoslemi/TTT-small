    input_tensor, mask, weights, config = data
    # ensure device
    device = input_tensor.device
    # dtype to half
    dtype = torch.float16

    # load weights, cast to half
    norm_weight = weights["norm.weight"].to(dtype)
    norm_bias = weights["norm.bias"].to(dtype)
    left_proj_weight = weights["left_proj.weight"].to(dtype)   # shape (hidden_dim, dim?) Actually nn.Linear(dim, hidden_dim) weight shape (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"].to(dtype)
    left_gate_weight = weights["left_gate.weight"].to(dtype)
    right_gate_weight = weights["right_gate.weight"].to(dtype)
    out_gate_weight = weights["out_gate.weight"].to(dtype)
    to_out_norm_weight = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_bias = weights["to_out_norm.bias"].to(dtype)
    to_out_weight = weights["to_out.weight"].to(dtype) # shape (dim, hidden_dim)

    # step 1: layernorm
    x = F.layer_norm(
        input_tensor,
        (config["dim"],),
        weight=norm_weight,
        bias=norm_bias,
    )
    # cast to half
    x_h = x.to(dtype)

    # step 2: linear projections
    # left, right shape B,N,N,hidden_dim
    left = F.linear(x_h, left_proj_weight, bias=None)
    right = F.linear(x_h, right_proj_weight, bias=None)

    # step 3: gates
    left_gate = F.linear(x_h, left_gate_weight, bias=None).sigmoid()
    right_gate = F.linear(x_h, right_gate_weight, bias=None).sigmoid()
    out_gate = F.linear(x_h, out_gate_weight, bias=None).sigmoid()

    # step 4: fuse mask + gating via Triton
    if not config.get("nomask", True):
        # mask is (B,N,N) maybe bool or numeric
        mask_h = mask.to(dtype).contiguous().view(-1)  # flatten
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # step 5: core contraction O(N^3) using torch.bmm
    # permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

        bias=to_out_norm_bias,
    )

    out = F.linear(out, to_out_weight, bias=None)

    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

Algorithm
---------
1. Input LayerNorm (float16).
2. Linear projections to hidden dimension: left, right.
3. Compute sigmoid gates (left_gate, right_gate, out_gate).
4. If a mask is present, fuse mask, left_gate and right_gate using a Triton
   kernel (in‑place scaling of left/right).
   If no mask, apply gates with simple element‑wise multiplication.
5. Core cubic contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM (torch.bmm) over the hidden dimension,
   which maps to highly tuned cuBLAS kernels.
6. Output LayerNorm, element‑wise scaling by out_gate, final linear projection
   back to the original dimension.
7. Return result as float32.

Only the mask+gate fusion is performed in Triton; the heavy O(N³) contraction
uses cuBLAS via torch.bmm, which is optimal on modern GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half*  [total]
    right_ptr,         # half*  [total]
    left_gate_ptr,     # half*  [total]
    right_gate_ptr,    # half*  [total]
    mask_ptr,          # half*  [mask_total] (broadcast over hidden dim)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    H: tl.constexpr,   # hidden dimension
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    # Broadcast mask over hidden dimension
    mask_idx = offs // H
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))

    # Apply mask and gates
    left = left * left_gate * mask_val
    right = right * right_gate * mask_val

    # Store the results back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of masking and gating:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    All tensors must be contiguous half‑precision.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask,
        total,
        BLOCK=BLOCK,
        H=hidden_dim,
    )
    # No return needed – operation is performed in‑place

            - mask: torch.Tensor of shape [B, N, N] (bool or numeric) – may be None
            - weights: dictionary mapping parameter names to torch.Tensors (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float16

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. Input LayerNorm (convert to half for speed)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )
    x_h = x.to(dtype)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension
    # ------------------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(dtype),  # (hidden_dim, dim)
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(dtype),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Compute gates
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(dtype),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(dtype),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(dtype),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) with gates via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] -> flatten and cast to half
        mask_h = mask.to(dtype).contiguous().view(-1)  # [B*N*N]
        # Flatten tensors for the kernel (must be contiguous)
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_h,
            hidden_dim,
        )
        # Reshape back to original layout
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask: simple element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as a batched GEMM (torch.bmm) over the hidden dimension.
    # ------------------------------------------------------------------
    # Bring hidden dimension to the second axis for batch GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)               # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)             # (B*H, N, N)

    # Batched GEMM with transpose on the second operand
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(dtype),  # (dim, hidden_dim)
        bias=None,
    )

    # Return as float32 (original precision)
    return out.to(torch.float32)
```
