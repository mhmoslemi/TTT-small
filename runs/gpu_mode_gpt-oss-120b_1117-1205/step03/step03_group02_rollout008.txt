    right = F.linear(x, right_proj_w)

    right_gate = F.linear(x, right_gate_w).sigmoid()

    _gate_and_mask_fuse(left, right, left_gate, right_gate, mask_f, hidden_dim)

"""
TriMul (outgoing) forward pass – fast Triton‑fused mask & gate.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32 → half) using the provided weights.
2. Linear projections to the hidden dimension for the left/right tensors.
3. Compute left/right/out gates (linear + sigmoid) in half precision.
4. Fuse optional mask, gate and element‑wise multiplication into a single
   Triton kernel that updates ``left`` and ``right`` in‑place.
5. Compute the core O(N³) contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   using a single ``torch.einsum`` which avoids costly data reshapes.
6. Apply the output LayerNorm, multiply by the out‑gate, and project back
   to the original feature dimension.
7. Return the result in float32.

Only the mask‑gate fusion is performed with Triton; the remaining
operations use highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
@triton.jit

    right_ptr,              # half*   right tensor (flattened)

    right_gate_ptr,         # half*   right‑gate tensor (flattened)
    mask_ptr,               # half*   mask tensor (flattened, length B*N*N)
    total_elements: tl.int32,

    HAS_MASK: tl.constexpr,  # 1 if mask present, else 0
    H: tl.constexpr,         # hidden dimension
):

    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)

        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)

    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32 → half)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections (half‑precision)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 3. Gate values (half‑precision, sigmoid)
    # --------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # --------------------------------------------------------------
    # 4. Fuse mask (if any) with gates via Triton
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).contiguous()
    else:
        mask_f = None

    _fuse_gate_and_mask(left, right, left_gate, right_gate, mask_f, hidden_dim)

    # --------------------------------------------------------------
    # 5. Core O(N³) contraction using a single einsum (no extra copies)
    # --------------------------------------------------------------
    # out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 7. Cast back to float32 as required
    # --------------------------------------------------------------
    return out.to(torch.float32)
```
