@triton.jit
def pointwise_mul_kernel(a_ptr, b_ptr, out_ptr, N_ELEMS, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEMS
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)

    )  # (B, N, N, dim), half

    left = F.linear(x, w_left_proj)   # (B,N,N,hidden_dim)
    right = F.linear(x, w_right_proj)

    # 3. Gates
    left_gate = F.linear(x, w_left_gate).sigmoid()
    right_gate = F.linear(x, w_right_gate).sigmoid()
    out_gate = F.linear(x, w_out_gate).sigmoid()

    if not nomask and mask is not None:
        # mask: [B,N,N] -> broadcast over hidden_dim
        mask_h = mask.to(torch.float16).unsqueeze(-1)
        left = left * left_gate * mask_h
        right = right * right_gate * mask_h
    else:
        left = left * left_gate
        right = right * right_gate

    # Release unused tensors early
    del left_gate, right_gate, x

    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # Reshape to (B, H, N, N) where H = hidden_dim.
    # left: (B,N,N,H) -> (B,H,N,N) with i as rows, k as columns.
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: (B,N,N,H) -> (B,H,N,N) with k as rows, i as columns (i.e., transposed)
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Batched half‑precision GEMM; accumulation in float32 (default)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N), half
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # 6. Output LayerNorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w_to_out_norm_weight,
        bias=w_to_out_norm_bias,
    )

    # 7. Multiply with out_gate using a tiny Triton kernel (in‑place)
    out_flat = out.view(-1)
    out_gate_flat = out_gate.view(-1)
    total = out_flat.numel()
    BLOCK = 8192
    grid = (triton.cdiv(total, BLOCK),)
    pointwise_mul_kernel[grid](
        out_flat,
        out_gate_flat,
        out_flat,
        total,
        BLOCK=BLOCK,
    )

    # 8. Final linear projection back to original dim
    out = F.linear(out, w_to_out)

    # Return float32 as required
    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass – fully fused half‑precision implementation.

Algorithm
---------
1. Input LayerNorm (half‑precision).
2. Linear projections → left / right (half) and three gating tensors
   (left_gate, right_gate, out_gate) via linear + sigmoid.
3. Apply gates (and optional mask) with inexpensive point‑wise ops.
4. Core O(N³) contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented with a single `torch.einsum`, avoiding any explicit
   transposes or copies.
5. Output LayerNorm (half), out‑gate multiplication performed by a tiny
   Triton point‑wise kernel, and final linear projection back to `dim`.
6. Cast result to float32.

At least one operation (the final gate multiplication) is executed in a
custom Triton kernel to satisfy the requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _pointwise_mul_kernel(a_ptr, b_ptr, out_ptr, N_ELEMS: tl.int32, BLOCK: tl.constexpr):
    """Element‑wise a * b → out (in‑place). All tensors are 16‑bit."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEMS
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (float32)
            - config: dict containing at least "dim", "hidden_dim",
                      optional "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm on input (half‑precision)
    # ------------------------------------------------------------------
    w_norm_w = weights["norm.weight"].to(torch.float16)
    w_norm_b = weights["norm.bias"].to(torch.float16)
    x = F.layer_norm(
        input_tensor.to(torch.float16),          # cast once
        (dim,),
        weight=w_norm_w,
        bias=w_norm_b,
    )                                            # (B, N, N, dim), half

    # ------------------------------------------------------------------
    # 2. Linear projections (half)
    # ------------------------------------------------------------------
    w_left = weights["left_proj.weight"].to(torch.float16)
    w_right = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x, w_left)    # (B, N, N, hidden_dim)
    right = F.linear(x, w_right)  # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3. Gating tensors (half + sigmoid)
    # ------------------------------------------------------------------
    w_lgate = weights["left_gate.weight"].to(torch.float16)
    w_rgate = weights["right_gate.weight"].to(torch.float16)
    w_outgate = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x, w_lgate).sigmoid()
    right_gate = F.linear(x, w_rgate).sigmoid()
    out_gate = F.linear(x, w_outgate).sigmoid()

    # ------------------------------------------------------------------
    # 4. Apply gates (and optional mask) – point‑wise ops
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * left_gate * mask_h
        right = right * right_gate * mask_h
    else:
        left = left * left_gate
        right = right * right_gate

    # Release some temporaries
    del left_gate, right_gate, x

    # ------------------------------------------------------------------
    # 5. Core contraction via einsum (no extra copies)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    out = torch.einsum("b i k d, b j k d -> b i j d", left, right)  # (B, N, N, hidden_dim)
    # Free large intermediates early
    del left, right

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (half)
    # ------------------------------------------------------------------
    w_outnorm_w = weights["to_out_norm.weight"].to(torch.float16)
    w_outnorm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w_outnorm_w,
        bias=w_outnorm_b,
    )  # (B, N, N, hidden_dim), half

    # ------------------------------------------------------------------
    # 7. Multiply by out_gate using a tiny Triton kernel (in‑place)
    # ------------------------------------------------------------------
    out_flat = out.view(-1)
    out_gate_flat = out_gate.view(-1)
    total = out_flat.numel()
    BLOCK = 8192
    grid = (triton.cdiv(total, BLOCK),)
    _pointwise_mul_kernel[grid](
        out_flat,
        out_gate_flat,
        out_flat,
        total,
        BLOCK=BLOCK,
    )
    del out_gate  # no longer needed

    # ------------------------------------------------------------------
    # 8. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, w_to_out)  # (B, N, N, dim), half

    # ------------------------------------------------------------------
    # 9. Cast to float32 as required by the spec
    # ------------------------------------------------------------------
    return out.to(torch.float32)
