Implementation skeleton:

def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    b = pid_bh // H
    d = pid_bh % H

    # calculate start indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Offsets within tile
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # acc
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(0, num_k):
        k_start = k_iter * BLOCK_K
        offs_kk = k_start + offs_k
        mask_k = offs_kk < N

        # Load left tile (M,K)
        left_ptrs = left_ptr + b*stride_left_b + d*stride_left_h + (offs_i[:, None] * stride_left_i) + (offs_kk[None, :] * stride_left_k)
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right tile transposed (K,N) i.e., (K,N) shape
        right_ptrs = right_ptr + b*stride_right_b + d*stride_right_h + (offs_kk[:, None] * stride_right_k) + (offs_j[None, :] * stride_right_i)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Compute dot product
        acc += tl.dot(left, right)

    # Store result
    out_ptrs = out_ptr + b*stride_out_b + d*stride_out_h + (offs_i[:, None] * stride_out_i) + (offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) forward pass implemented with Triton.

Algorithm:
1. LayerNorm over the input (float32).
2. Cast to fp16 and compute left/right projections and gating linear layers.
3. Fuse gating and optional mask in a Triton kernel (in‑place on left/right).
4. Compute the core O(N³) contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   using a custom Triton kernel that works directly on the (B,N,N,H) layout,
   avoiding costly transposes and extra memory copies.
5. Apply output LayerNorm, element‑wise out‑gate, and final linear projection.
6. Return the result in float32.

The implementation uses mixed precision (fp16 for the bulk of the work) and
keeps the heavy tensor contraction entirely on the GPU in a fused kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: fuse left/right projections with their gates and (optional) mask
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # fp16 *
    right_ptr,           # fp16 *
    left_gate_ptr,       # fp16 *
    right_gate_ptr,      # fp16 *
    mask_ptr,            # fp16 *
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present, 0 otherwise
    H: tl.constexpr,          # hidden dimension
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    # load
    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    if HAS_MASK:
        # mask element repeats H times (once per hidden channel)
        mask_index = offs // H
        mask_val = tl.load(mask_ptr + mask_index, mask=valid, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # store in‑place
    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """In‑place fusion of gating (and optional mask) using Triton."""
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy ptr when mask not used

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # The operation is performed in‑place; nothing is returned.


# ----------------------------------------------------------------------
# Kernel 2: outgoing TriMul contraction (batched GEMM over (i,j) for each
#          batch and hidden channel) without any transposes.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,          # fp16 *
    right_ptr,         # fp16 *
    out_ptr,           # fp16 *
    B: tl.constexpr,   # batch size
    N: tl.constexpr,   # sequence length
    H: tl.constexpr,   # hidden dimension
    # strides (in elements)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)   # tile over output rows (i)
    pid_j = tl.program_id(1)   # tile over output cols (j)
    pid_bh = tl.program_id(2)  # combined batch * hidden index

    # decode batch and hidden index
    b = pid_bh // H
    h = pid_bh % H

    # start offsets for this tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # range vectors
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k_tile in range(0, num_k):
        k_start = k_tile * BLOCK_K
        k_idx = k_start + offs_k
        mask_k = k_idx < N

        # ----- load left tile (M x K) -----
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + h * stride_left_h
            + (offs_i[:, None] * stride_left_i)
            + (k_idx[None, :] * stride_left_k)
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16
        left = left.to(tl.float32)

        # ----- load right tile transposed (K x N) -----
        # we want right[k, j] so k is the row dimension, j the column
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + h * stride_right_h
            + (k_idx[:, None] * stride_right_k)
            + (offs_j[None, :] * stride_right_i)
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )  # fp16
        right = right.to(tl.float32)

        # ----- dot product (M x K) * (K x N) -----
        acc += tl.dot(left, right)

    # ----- store result (M x N) -----
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + h * stride_out_h
        + (offs_i[:, None] * stride_out_i)
        + (offs_j[None, :] * stride_out_j)
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor:  torch.Tensor [B, N, N] (float32/bool) or None
            - weights_dict: mapping of parameter names to torch.Tensor
            - config_dict: must contain "dim" and "hidden_dim",
                           may contain "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim] float32

    # --------------------------------------------------------------
    # 2. Cast to half for the heavy work
    # --------------------------------------------------------------
    x_h = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Linear projections (half)
    # --------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 4. Gating projections (half)
    # --------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # --------------------------------------------------------------
    # 5. Fuse gate (and optional mask) with a Triton kernel
    # --------------------------------------------------------------
    if mask is not None:
        mask_h = mask.to(torch.float16).contiguous()
    else:
        mask_h = None

    _apply_gate_and_mask(
        left,
        right,
        left_gate,
        right_gate,
        mask_h,
        hidden_dim,
    )
    # gates are no longer needed
    del left_gate, right_gate

    # --------------------------------------------------------------
    # 6. Core O(N³) contraction using a custom Triton kernel
    # --------------------------------------------------------------
    B, N, _, _ = left.shape  # left shape is (B, N, N, H)
    assert left.shape == right.shape
    out = torch.empty_like(left)  # fp16, same layout

    # strides (in elements)
    stride_left_b = left.stride(0)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)
    stride_left_h = left.stride(3)

    stride_right_b = right.stride(0)
    stride_right_i = right.stride(1)   # corresponds to the 'j' axis for right
    stride_right_k = right.stride(2)
    stride_right_h = right.stride(3)

    stride_out_b = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_h = out.stride(3)

    # tile sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid_i = triton.cdiv(N, BLOCK_M)
    grid_j = triton.cdiv(N, BLOCK_N)
    grid_bh = B * hidden_dim
    grid = (grid_i, grid_j, grid_bh)

    _trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_i,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Free left/right as they are no longer needed
    del left, right

    # --------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final linear projection (half precision)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 8. Cast back to float32 as required by the API
    # --------------------------------------------------------------
    return out.to(torch.float32)
