    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_h = tl.program_id(2) # index over B*H
    b = batch_h // H
    h = batch_h % H

    pid_m = tl.program_id(0) # row block (i)
    pid_n = tl.program_id(1) # col block (j)

    # Determine block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute mask for within N
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k offsets
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        left_ptrs = left_ptr + b*stride_left_b + h*stride_left_h + (offs_m[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k)

"""
TriMul (outgoing) forward pass optimized with Triton.

The implementation follows the reference PyTorch TriMul module but
accelerates the cubic contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
using a custom Triton kernel.  LayerNorms, linear projections and the
gating operations remain in PyTorch (FP16) for simplicity, while the
contraction is performed directly on the (B, N, N, H) layout without
creating large transposed copies.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication for the TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def trmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_m = tl.program_id(0)          # tile over output i dimension
    pid_n = tl.program_id(1)          # tile over output j dimension
    pid_h = tl.program_id(2)          # batch * hidden index

    # map pid_h to batch and hidden channel
    b = pid_h // H
    h = pid_h % H

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: (M, K)
        left_offset = (
            b * stride_left_b + h * stride_left_h
            + offs_m[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        left = tl.load(left_ptr + left_offset,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right tile: (N, K)  (j dimension is the row)
        right_offset = (
            b * stride_right_b + h * stride_right_h
            + offs_n[:, None] * stride_right_j
            + offs_k[None, :] * stride_right_k
        )
        right = tl.load(right_ptr + right_offset,
                        mask=mask_n[:, None] & mask_k[None, :],
                        other=0.0).to(tl.float32)

        # dot over k -> (M, N)
        acc += tl.dot(left, right)

    # write back result in fp16
    out_offset = (
        b * stride_out_b + h * stride_out_h
        + offs_m[:, None] * stride_out_i
        + offs_n[None, :] * stride_out_j
    )
    tl.store(out_ptr + out_offset,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (bool or numeric) or None
            - weights: dictionary of model parameters (float32 tensors)
            - config: dictionary with keys ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm on input (FP16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device).to(torch.float16)
    norm_bias = weights["norm.bias"].to(device).to(torch.float16)
    x = input_tensor.to(torch.float16)
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2. Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left = F.linear(x, left_proj_w)   # (B, N, N, hidden_dim)
    right = F.linear(x, right_proj_w) # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3. Gating (FP16)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4. Apply gates (and optional mask) – pure PyTorch
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate
    if not nomask and mask is not None:
        mask_h = mask.unsqueeze(-1).to(device).to(torch.float16)
        left = left * mask_h
        right = right * mask_h

    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 5. Core TriMul contraction via Triton
    # ------------------------------------------------------------------
    out = torch.empty_like(left)  # (B, N, N, hidden_dim), fp16

"""
TriMul (outgoing) forward pass optimized with Triton.

The implementation follows the reference PyTorch TriMul module but
accelerates the cubic contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
using a custom Triton kernel.  LayerNorms, linear projections and the
gating operations remain in PyTorch (FP16) for simplicity, while the
contraction is performed directly on the (B, N, N, H) layout without
creating large transposed copies.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication for the TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def trmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute, for each batch `b` and hidden channel `h`,
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    The tensors are stored in the native layout (B,N,N,H) where H is the
    hidden dimension.  The kernel tiles the i‑ and j‑axes (M×N) and streams
    over the k‑axis (K) using a standard GEMM‑style algorithm.
    """
    # program ids
    pid_m = tl.program_id(0)      # tile over i (row)
    pid_n = tl.program_id(1)      # tile over j (col)
    pid_h = tl.program_id(2)      # batch*hidden index

    # map pid_h -> (batch, hidden)
    b = pid_h // H
    h = pid_h % H

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator (fp32 for precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (M, K)
        left_offset = (
            b * stride_left_b + h * stride_left_h
            + offs_m[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        left = tl.load(left_ptr + left_offset,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right tile: shape (N, K) – j dimension is the row index
        right_offset = (
            b * stride_right_b + h * stride_right_h
            + offs_n[:, None] * stride_right_j
            + offs_k[None, :] * stride_right_k
        )
        right = tl.load(right_ptr + right_offset,
                        mask=mask_n[:, None] & mask_k[None, :],
                        other=0.0).to(tl.float32)

        # accumulate dot over the K dimension
        acc += tl.dot(left, right)   # (M, N)

    # write the result back in fp16
    out_offset = (
        b * stride_out_b + h * stride_out_h
        + offs_m[:, None] * stride_out_i
        + offs_n[None, :] * stride_out_j
    )
    tl.store(out_ptr + out_offset,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] (bool or numeric) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (FP16)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float16)
    norm_b = weights["norm.bias"].to(device).to(torch.float16)
    x = input_tensor.to(torch.float16)
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------
    # 2. Linear projections (FP16)
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left = F.linear(x, left_proj_w)   # (B, N, N, hidden_dim)
    right = F.linear(x, right_proj_w) # (B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 3. Gates (FP16)
    # --------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 4. Apply gates and optional mask (pure PyTorch)
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate
    if not nomask and mask is not None:
        mask_h = mask.unsqueeze(-1).to(device).to(torch.float16)
        left = left * mask_h
        right = right * mask_h

    left = left.contiguous()
    right = right.contiguous()

    # --------------------------------------------------------------
    # 5. Core TriMul contraction via Triton
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    out = torch.empty_like(left)  # (B, N, N, hidden_dim), fp16

    # strides for kernel (int64)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )
    trmul_outgoing_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection (FP16)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 7. Cast back to float32 (as required by the spec)
    # --------------------------------------------------------------
    return out.to(torch.float32)
