def _gate_mask_fuse_kernel(left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr, total_elements, BLOCK, HAS_MASK: tl.constexpr, H: tl.constexpr):

    right = tl.load(right_ptr + offs, mask=mask, other=0.0)

        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)  # default mask absent

    # Layer norm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # half precision
    x_h = x.to(torch.float16)

    # linear projections
    left = torch.nn.functional.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = torch.nn.functional.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # gates
    left_gate = torch.nn.functional.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # Fuse gating (and mask if present)
    has_mask = 0
    mask_f = None

    if not nomask and mask_tensor is not None:
        # mask present
        has_mask = 1
        mask_f = mask_tensor.to(torch.float16).contiguous()
    # Ensure contiguous flatten for all
    left_f = left.contiguous().view(-1)
    right_f = right.contiguous().view(-1)
    left_gate_f = left_gate.contiguous().view(-1)
    right_gate_f = right_gate.contiguous().view(-1)

    total = left_f.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    # dummy mask pointer if no mask: pass left_f pointer (any pointer)
    mask_ptr = mask_f if has_mask else left_f

    _gate_mask_fuse_kernel[grid](
        left_f,
        right_f,
        left_gate_f,
        right_gate_f,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )

    # reshape back
    left = left_f.view_as(left)
    right = right_f.view_as(right)

    # free gates left/right
    del left_gate, right_gate

    # Now compute contraction via batched matmul
    # Permute for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) with j<->i swapped

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # Output layer norm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16)
    )
    # out gate applied
    out = out * out_gate

    # final linear projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16))

    # cast to float32 output
    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass.

Algorithm:
1. LayerNorm on the input tensor (float32).
2. Cast to float16 for the heavy work.
3. Linear projections (left/right) and gating linear layers (half‑precision).
4. Fuse gating and optional mask using a small Triton kernel
   (left = left * left_gate * mask, right = right * right_gate * mask).
5. Compute the N³ contraction as a batched matrix multiplication:
   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]   (implemented with torch.bmm).
6. Output LayerNorm, out‑gate multiplication and final linear projection (half‑precision).
7. Cast the result back to float32.
The kernel in step 4 satisfies the “must‑use‑a‑kernel” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # float16 *
    right_ptr,         # float16 *
    left_gate_ptr,     # float16 *
    right_gate_ptr,    # float16 *
    mask_ptr,          # float16 *
    total_elements,    # int32
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present else 0
    H: tl.constexpr,          # hidden dimension (channel size)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    if HAS_MASK:
        # each mask element corresponds to H consecutive channel entries
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict: mapping of parameter names to torch.Tensor
            - config_dict: must contain "dim" and "hidden_dim",
                           may contain "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to half‑precision for the remaining heavy work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (half)
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 4. Gates (half)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 5. Fuse gating (and optional mask) via Triton kernel
    # ------------------------------------------------------------------
    has_mask = 0
    mask_f = None
    if not nomask and mask_tensor is not None:
        has_mask = 1
        mask_f = mask_tensor.to(torch.float16).contiguous()
    # Flatten tensors for the kernel
    left_f = left.contiguous().view(-1)
    right_f = right.contiguous().view(-1)
    left_gate_f = left_gate.contiguous().view(-1)
    right_gate_f = right_gate.contiguous().view(-1)

    total = left_f.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    # Dummy mask pointer when mask not used (won't be dereferenced)
    mask_ptr = mask_f if has_mask else left_f

    _gate_mask_fuse_kernel[grid](
        left_f,
        right_f,
        left_gate_f,
        right_gate_f,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )

    # Reshape back to original layout
    left = left_f.view_as(left)
    right = right_f.view_as(right)

    # ------------------------------------------------------------------
    # 6. Core N³ contraction via batched GEMM:
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) for efficient batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    # right needs to be transposed on the two inner axes
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return float32 as required by the API
    return out.to(torch.float32)
