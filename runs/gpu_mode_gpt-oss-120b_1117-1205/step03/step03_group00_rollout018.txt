@triton.jit
def outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    N: tl.int32,
    stride_batch: tl.int32, stride_i: tl.int32, stride_k: tl.int32,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # Compute tile offsets
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Bound checks
    mask_i = i_offset < N
    mask_j = j_offset < N
    mask = mask_i[:, None] & mask_j[None, :]

    # Compute base pointer offset for this batch_head
    # base offset = pid_bh * stride_batch
    bh_offset = pid_bh * stride_batch

    # Initialize accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # Load left tile (M x K)
        left_ptrs = left_ptr + bh_offset + i_offset[:, None] * stride_i + k_offset[None, :] * stride_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right tile (N x K)
        right_ptrs = right_ptr + bh_offset + j_offset[:, None] * stride_i + k_offset[None, :] * stride_k
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # Accumulate dot
        acc += tl.dot(left, right)   # left: (M,K); right: (N,K) -> (M,N) sum over K

    # Store result
    out_ptrs = out_ptr + bh_offset + i_offset[:, None] * stride_i + j_offset[None, :] * stride_i
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask)

def outgoing_kernel(left_ptr, right_ptr, out_ptr,
                    N: tl.int32,
                    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)
    # Compute offsets
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for bounds
    mask_i = i < N
    mask_j = j < N
    mask = mask_i[:, None] & mask_j[None, :]

    # Offsets for batch head
    # each batch_head segment size = N*N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k

        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left tile addresses

        mask_h = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_h
        right = right * mask_h

    # gates
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # core contraction via Triton
    # permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, N2 = left_perm.shape
    assert N == N2

    # flatten batch and heads
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # output placeholder
    out_flat = torch.empty_like(left_flat)  # same dtype half

    # launch kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * H)
    outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # output layernorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    # apply out_gate and final linear
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    return out.to(torch.float32)
```

```
@triton.jit
def outgoing_kernel(left_ptr, right_ptr, out_ptr, N,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """Compute out = sum_k left[..., i, k, :] * right[..., j, k, :] for each head.

    left, right, out are 3D tensors of shape [batch_head, N, N] with dtype
    float16. The result out has the same dtype.
    """
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # Offsets
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N
    mask = mask_i[:, None] & mask_j[None, :]

    # base offset for this batch_head
    bh_offset = pid_bh * N * N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # k-loop
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left tile (M x K)
        left_offset = bh_offset + i[:, None] * N + k[None, :]
        left = tl.load(left_ptr + left_offset,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right tile (N x K)
        right_offset = bh_offset + j[:, None] * N + k[None, :]
        right = tl.load(right_ptr + right_offset,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0)

        # dot over K dimension
        acc += tl.dot(left, right)   # (M,N) accumulator in fp32

    # store result as float16
    out_offset = bh_offset + i[:, None] * N + j[None, :]
    tl.store(out_ptr + out_offset, acc.to(tl.float16), mask=mask)

"""
TriMul outgoing operator (AlphaFold3) forward pass.

The core cubic contraction
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
is executed by a custom Triton kernel that performs a tiled
matrix‑multiply‑accumulate using `tl.dot`.  All other
operations (LayerNorm, linear projections, gates, mask handling
and output projection) are done in PyTorch with mixed‑precision
(half‑precision for the heavy tensor work).  The final result is
returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel for the O(N³) contraction
# --------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr,          # *float16, shape [B*H, N, N]
    right_ptr,         # *float16, shape [B*H, N, N]
    out_ptr,           # *float16, shape [B*H, N, N]
    N: tl.int32,       # sequence length (square matrix side)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out = Σ_k left[..., i, k, :] * right[..., j, k, :]"""

    pid_i = tl.program_id(0)   # block row   (i dimension)
    pid_j = tl.program_id(1)   # block col   (j dimension)
    pid_bh = tl.program_id(2)  # batch * head index

    # ----------------------------------------------------------
    # Tile offsets and bounds
    # ----------------------------------------------------------
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N
    mask_ij = mask_i[:, None] & mask_j[None, :]

    # Each batch‑head slice has size N*N
    bh_offset = pid_bh * N * N

    # Accumulator in fp32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------------------------------------------------------
    # Loop over the reduction dimension k
    # ----------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # Load a (M x K) tile from left
        left_tile_ptr = left_ptr + bh_offset + i[:, None] * N + k[None, :]
        left_tile = tl.load(
            left_tile_ptr,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load a (N x K) tile from right
        right_tile_ptr = right_ptr + bh_offset + j[:, None] * N + k[None, :]
        right_tile = tl.load(
            right_tile_ptr,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        )

        # tl.dot sums over K: (M,K)·(N,K) → (M,N)
        acc += tl.dot(left_tile, right_tile)   # fp32 accumulation

    # ----------------------------------------------------------
    # Write back the result as float16
    # ----------------------------------------------------------
    out_tile_ptr = out_ptr + bh_offset + i[:, None] * N + j[None, :]
    tl.store(out_tile_ptr, acc.to(tl.float16), mask=mask_ij)


# --------------------------------------------------------------
# Entry point expected by the evaluator
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (ignored if config["nomask"] is True)
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # ----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ----------------------------------------------------------
    # 2. Linear projections to hidden dimension (half)
    # ----------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ----------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # ----------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1]
        mask_h = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_h
        right = right * mask_h

    # ----------------------------------------------------------
    # 4. Gating (half precision)
    # ----------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------
    # 5. Core O(N³) contraction via Triton
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ----------------------------------------------------------
    # Move head dimension to axis 1 for a contiguous layout
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N)  # (B*H, N, N)

    out_flat = torch.empty_like(left_flat)  # allocate result (half)

    # Tile sizes – chosen to give good occupancy on B200 GPUs
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )
    _tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to original layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ----------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling, final projection
    # ----------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise scaling
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in float32 as expected by downstream code
    return out.to(torch.float32)
