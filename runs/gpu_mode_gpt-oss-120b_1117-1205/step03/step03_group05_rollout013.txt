@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_abatch, stride_bbatch, stride_cbatch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(2)
    block_i = tl.program_id(1)  # M dimension
    block_j = tl.program_id(0)  # N dimension

    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for i/j
    mask_i = offs_i < M
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # compute pointers
        a_ptrs = a_ptr + batch_id * stride_abatch + offs_i[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + batch_id * stride_bbatch + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=(mask_i[:, None] & (offs_k < K)[None, :]), other=0.0)
        b = tl.load(b_ptrs, mask=((offs_k < K)[:, None] & mask_j[None, :]), other=0.0)

        acc += tl.dot(a, b)  # a: (BLOCK_M, BLOCK_K) b: (BLOCK_K, BLOCK_N)
    # write back
    c_ptrs = c_ptr + batch_id * stride_cbatch + offs_i[:, None] * stride_cm + offs_j[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    B,N,N2,dim_check = input_tensor.shape
    assert N == N2

    # LayerNorm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # Proj
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # Apply gates and mask via Triton fusion if mask present
    if not nomask:
        # ensure contiguous
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_exp = mask.unsqueeze(-1).to(torch.float16).contiguous()
        # flatten for kernel
        total = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)
        gate_mask_fuse_kernel[grid](
            left, right,
            left_gate, right_gate,
            mask_exp,
            total,
            BLOCK=BLOCK,
            HAS_MASK=1,
            H=hidden_dim,
        )
        # Multiply left/right by gates & mask is done inside kernel
    else:
        left = left * left_gate
        right = right * right_gate

    # No longer need gate tensors
    del left_gate, right_gate, x

    # left: [B, N, N, H] -> (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()

    stride_abatch = left_mat.stride(0)

    stride_bn = right_mat.stride(2)
    stride_bbatch = right_mat.stride(0)

"""
Custom Triton + PyTorch implementation of the outgoing TriMul operation.
...
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    if HAS_MASK:
        # each H contiguous entries correspond to one mask element
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * mask_val
        right = right * mask_val

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_abatch, stride_bbatch, stride_cbatch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Batch id is the third program dimension
    batch_id = tl.program_id(2)
    block_m = tl.program_id(1)  # row tile
    block_n = tl.program_id(0)  # col tile

    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Pointers to the tiles of A and B
        a_ptrs = a_ptr + batch_id * stride_abatch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + batch_id * stride_bbatch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=(mask_m[:, None] & (offs_k < K)[None, :]), other=0.0)
        b = tl.load(b_ptrs, mask=((offs_k < K)[:, None] & mask_n[None, :]), other=0.0)

        acc += tl.dot(a, b)  # a: (BLOCK_M, BLOCK_K), b: (BLOCK_K, BLOCK_N)

    # Write back
    c_ptrs = c_ptr + batch_id * stride_cbatch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.
    Performs layer‑norm, linear projections, gate & mask fusion,
    a batched half‑precision matrix multiplication (the N³ contraction),
    output layer‑norm, out‑gate multiplication and final linear projection.
    Returns a float32 tensor of shape [B, N, N, dim].
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> half)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half precision)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # -----------------------------------------------------------------
    # 3. Gating tensors (half)
    # -----------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # -----------------------------------------------------------------
    # 4. Fuse mask (if any) and gates via Triton (in‑place)
    # -----------------------------------------------------------------
    if not nomask:
        # Expand mask to match hidden dimension (broadcast)
        mask_exp = mask.unsqueeze(-1).to(torch.float16).contiguous()
        # Flatten tensors to 1‑D for the kernel
        total = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)
        # Triton kernel expects mask pointer; we provide the expanded mask
        gate_mask_fuse_kernel[grid](
            left, right,
            left_gate, right_gate,
            mask_exp,
            total,
            BLOCK=BLOCK,
            HAS_MASK=1,
            H=hidden_dim,
        )
    else:
        # No mask: simple element‑wise gate multiplication
        left = left * left_gate
        right = right * right_gate

    # Gate tensors no longer needed
    del left_gate, right_gate, x

    # -----------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as batched half‑precision GEMM (B*H independent matmuls)
    # -----------------------------------------------------------------
    # Rearrange to (B, H, N, N) where the two N axes are ready for GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()        # (B, H, N, N) with swapped N axes

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)
    out_mat = torch.empty_like(left_mat)  # will hold the result in fp16

    # Strides for the batched GEMM
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)
    stride_abatch = left_mat.stride(0)

    stride_bk = right_mat.stride(1)
    stride_bn = right_mat.stride(2)
    stride_bbatch = right_mat.stride(0)

    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)
    stride_cbatch = out_mat.stride(0)

    # Tile sizes – chosen to fit into shared memory on typical GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_N),   # number of column tiles
        triton.cdiv(N, BLOCK_M),   # number of row tiles
        B * H,                     # batch dimension (B*H independent matmuls)
    )
    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,  # M

"""
custom_kernel
--------------

Triton‑accelerated forward pass of the *outgoing* TriMul operator (AlphaFold3).

Algorithm
---------
1. Input LayerNorm (float32) → half precision.
2. Linear projections `left` and `right` to `hidden_dim`.
3. Compute gating tensors (`left_gate`, `right_gate`, `out_gate`) via linear+sigmoid.
4. Fuse mask (if present) and gating into `left` / `right` **in‑place** with a
   Triton kernel (`gate_mask_fuse_kernel`).
5. Perform the N³ contraction
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   as a batched half‑precision GEMM using a custom Triton kernel
   (`batched_matmul_kernel`).  The hidden dimension is fused into the batch
   dimension, yielding (B·H) independent N×N matrix multiplications.
6. Output LayerNorm, element‑wise `out_gate`, and final linear projection back
   to `dim`.
7. Return the result in float32.

Only the mask‑gate fusion and the batched GEMM are executed on Triton; all
other operations use PyTorch (half precision where possible).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def gate_mask_fuse_kernel(
    left_ptr,        # half*
    right_ptr,       # half*
    left_gate_ptr,   # half*
    right_gate_ptr,  # half*
    mask_ptr,        # half*   (broadcasted mask, length = B·N·N)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 → mask present, 0 → no mask
    H: tl.constexpr,          # hidden dimension size
):
    """In‑place multiplication of left/right with their gates and an optional mask."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    # apply gates
    left = left * left_gate
    right = right * right_gate

    if HAS_MASK:
        # one mask element per hidden‑dim block
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=0.0)
        left = left * mask_val
        right = right * mask_val

    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_abatch, stride_bbatch, stride_cbatch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched half‑precision GEMM.
    Computes C[batch, i, j] = Σₖ A[batch, i, k] * B[batch, k, j]
    for all batches = B·H.
    """
    batch_id = tl.program_id(2)      # batch dimension
    block_i = tl.program_id(1)       # tile along M
    block_j = tl.program_id(0)       # tile along N

    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        a_ptrs = a_ptr + batch_id * stride_abatch \
                 + offs_i[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + batch_id * stride_bbatch \
                 + offs_k[:, None] * stride_bk \
                 + offs_j[None, :] * stride_bn

        a = tl.load(a_ptrs,
                    mask=(mask_i[:, None] & (offs_k < K)[None, :]),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=((offs_k < K)[:, None] & mask_j[None, :]),
                    other=0.0)

        acc += tl.dot(a, b)    # (BLOCK_M, BLOCK_K) × (BLOCK_K, BLOCK_N)

    c_ptrs = c_ptr + batch_id * stride_cbatch \
             + offs_i[:, None] * stride_cm \
             + offs_j[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) – may be ignored
            - weights: dict containing model parameters (float32 tensors)
            - config: dict with keys 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 3. Gating tensors (half)
    # --------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # --------------------------------------------------------------
    # 4. Fuse mask (if any) and gates via Triton (in‑place)
    # --------------------------------------------------------------
    if not nomask:
        # expand mask to broadcast over hidden dimension
        mask_exp = mask.unsqueeze(-1).to(torch.float16).contiguous()
        total = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)
        gate_mask_fuse_kernel[grid](
            left, right,
            left_gate, right_gate,
            mask_exp,
            total,
            BLOCK=BLOCK,
            HAS_MASK=1,
            H=hidden_dim,
        )
    else:
        # simple element‑wise gate multiplication
        left = left * left_gate
        right = right * right_gate

    # Gates no longer needed
    del left_gate, right_gate, x

    # --------------------------------------------------------------
    # 5. Core contraction (batched GEMM)
    # --------------------------------------------------------------
    # bring hidden dimension to the second axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) with swapped N

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B·H, N, N)
    right_mat = right_perm.view(B * H, N, N) # (B·H, N, N)
    out_mat = torch.empty_like(left_mat)     # destination (half)

    # strides for the batched GEMM
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)
    stride_abatch = left_mat.stride(0)

    stride_bk = right_mat.stride(1)
    stride_bn = right_mat.stride(2)
    stride_bbatch = right_mat.stride(0)

    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)
    stride_cbatch = out_mat.stride(0)

    # tile sizes (tuned for typical GPU shared‑memory limits)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_N),   # tiles along N (columns)
        triton.cdiv(N, BLOCK_M),   # tiles along M (rows)
        B * H,                     # batch dimension (B·H independent matmuls)
    )
    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,               # M
        N,               # N
        N,               # K
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_abatch, stride_bbatch, stride_cbatch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(

        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
