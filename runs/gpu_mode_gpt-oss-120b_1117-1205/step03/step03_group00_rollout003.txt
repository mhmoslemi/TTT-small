"""
TriMul outgoing operator – forward pass.

The heavy N³ contraction is performed as a batched GEMM (cuBLAS) on
half‑precision tensors:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

A small Triton kernel fuses the optional mask with the left/right gating
element‑wise multiplications, avoiding a separate Python kernel launch per
tensor. All other linear layers and LayerNorms are done with torch.nn.functional
using half‑precision for speed; the final result is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half*   [total]
    right_ptr,         # half*   [total]
    left_gate_ptr,     # half*   [total]
    right_gate_ptr,    # half*   [total]
    mask_ptr,          # half*   [mask_total] (broadcast)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present else 0
    H: tl.constexpr,          # hidden dimension (size of innermost channel)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=valid, other=0.0, dtype=tl.float16)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0, dtype=tl.float16)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0, dtype=tl.float16)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0, dtype=tl.float16)

    if HAS_MASK:
        # Broadcast mask across hidden dimension
        mask_offs = offs // H
        mask_val = tl.load(mask_ptr + mask_offs, mask=valid, other=0.0, dtype=tl.float16)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask

    All tensors must be contiguous half‑precision.  `mask` is a flattened
    half‑tensor of shape [B*N*N]; it is broadcast across the hidden dimension.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    # dummy pointer when mask is not used (triton requires an argument)
    mask_ptr = mask if mask is not None else left

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # No return value – operation performed in‑place


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor of shape [B, N, N] (bool or numeric) or None
            - weights_dict: mapping from parameter names to torch.Tensor (float32)
            - config_dict: dict with keys
                * "dim"          – input channel dimension
                * "hidden_dim"   – hidden dimension (output of projections)
                * "nomask"       – bool, True if mask should be ignored (default True)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision for speed
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half precision)
    # ------------------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gates (half precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) with gating via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → flatten, half precision
        mask_h = mask_tensor.to(torch.float16).contiguous().view(-1)  # [B*N*N]
        # Flatten tensors for kernel launch
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_h,
            hidden_dim,
        )
        # Reshape back to original layout
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask – simple element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate

    # Release resources we no longer need
    del left_gate, right_gate, x_h

    # ------------------------------------------------------------------
    # 5. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as a batched GEMM (cuBLAS) on half‑precision tensors.
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)              # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)            # (B*H, N, N)

    # Batched matmul: (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore (B, N, N, H) layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise scaling
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in original dtype (float32)
    return out.to(torch.float32)
