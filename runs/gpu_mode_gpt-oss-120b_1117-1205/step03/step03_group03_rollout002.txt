"""
TriMul (outgoing) forward pass with Triton‑fused mask & gating.

Algorithm
---------
1. LayerNorm on the input tensor (float32 → half).
2. Linear projections `left` and `right` to `hidden_dim` (half‑precision).
3. Compute gating tensors via linear + sigmoid (half‑precision).
4. Fuse optional mask with left/right gating in a single Triton kernel
   (in‑place on flattened tensors).
5. Core O(N³) contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM (cuBLAS) after reshaping to
   (B*hidden_dim, N, N).
6. Apply second LayerNorm, output gate, and final linear projection.
7. Return float32 output.

Both mask‑plus‑gate fusion and the heavy contraction are performed on the GPU,
ensuring the implementation meets the performance requirements.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse optional mask with left/right gating
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # half*  (flattened left)
    right_ptr,           # half*  (flattened right)
    left_gate_ptr,       # half*  (flattened left_gate)
    right_gate_ptr,      # half*  (flattened right_gate)
    mask_ptr,            # half*  (flattened mask, broadcasted)
    total_elements: tl.int32,
    hidden_dim: tl.int32,
    HAS_MASK: tl.constexpr,   # 1 if mask present, 0 otherwise
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK == 1:
        # each mask entry corresponds to `hidden_dim` consecutive channel values
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of:
        left = left * left_gate * mask
        right = right * right_gate * mask
    (mask optional). All tensors must be contiguous half‑precision.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    HAS_MASK = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when no mask

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        hidden_dim,
        HAS_MASK=HAS_MASK,
        BLOCK=BLOCK,
    )
    # tensors are modified in place


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict containing keys:
                * "dim" (int) – input channel size C
                * "hidden_dim" (int) – projection size H
                * "nomask" (bool, optional) – True if mask should be ignored

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> half)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half)
    # ------------------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gating tensors (half)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) and gate multiplications via Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape: [B, N, N] → broadcast over hidden_dim
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        # Flatten tensors for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask_h.view(-1)                # [B*N*N]

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            hidden_dim,
        )
        # Reshape back to original layout
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask: simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # Gates are no longer needed
    del left_gate, right_gate, x_h

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as batched GEMM (cuBLAS) after transposes.
    # ------------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: (B, N, N, H) -> (B, H, N, N) with axes (k, j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # Batched matrix multiplication (FP16 accumulation in FP32)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)

    # Restore layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise gating (half)

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )  # (B, N, N, dim) half

    # ------------------------------------------------------------------
    # 7. Return in float32 as required
    # ------------------------------------------------------------------
    return out.to(torch.float32)
