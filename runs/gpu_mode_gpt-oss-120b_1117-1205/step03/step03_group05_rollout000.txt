"""
TriMul (outgoing) forward pass with Triton fusion.

Algorithm:
1. LayerNorm on the input (float32 → half).
2. Linear projections to hidden dimension (half precision).
3. Compute three gate tensors via linear+sigmoid (half).
4. Fuse mask (if present) and the left/right gate multiplications in-place
   using a Triton kernel (operates on flattened half tensors).
5. Core contraction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication per hidden channel:
      out_{b,h} = left_{b,h} @ right_{b,h}.T
   Implemented with torch.bmm (cuBLAS) on half tensors.
6. Output LayerNorm, element‑wise out‑gate, final linear projection.
7. Return float32 tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half* (flattened)
    right_ptr,         # half* (flattened)
    left_gate_ptr,     # half* (flattened)
    right_gate_ptr,    # half* (flattened)
    mask_ptr,          # half* (flattened, length = B*N*N)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present, 0 otherwise
    H: tl.constexpr,          # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load data
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    if HAS_MASK:
        # Each mask element corresponds to H consecutive hidden entries
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * mask_val
        right = right * mask_val

    # Store results back in-place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of left/right with their gates and optional mask.
    All tensors must be torch.float16 and contiguous.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    HAS_MASK = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when no mask

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=HAS_MASK,
        H=hidden_dim,
    )


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys 'dim', 'hidden_dim', 'nomask' (bool)

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> half)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (half)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Gating tensors (half)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) and gate multiplications via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # broadcast mask over hidden dimension
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()  # [B,N,N,1]
        # flatten tensors for kernel
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_h.view(-1)  # length B*N*N

        _apply_gate_and_mask(
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            hidden_dim,
        )
        # reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # simple element‑wise gating (no mask)
        left = left * left_gate
        right = right * right_gate

    # free gates not needed later
    del left_gate, right_gate, x

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()        # [B, H, N, N] (j,k) -> (k,j)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)                     # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)                   # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)                   # (B*H, N, N) half
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise (half)
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 7. Return float32 tensor
    # ------------------------------------------------------------------
    return out.to(torch.float32)
