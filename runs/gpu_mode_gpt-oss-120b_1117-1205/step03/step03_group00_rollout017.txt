def matmul_batch_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_lm, stride_li, stride_lk,
    stride_rm, stride_rj, stride_rk,   # right, note dims: (M,N,N) with columns k dimension stride_rk
    stride_om, stride_oi, stride_oj,
    N, BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(2) // for batch? Actually we get 3D grid: i_tile = pid0; j_tile = pid1; batchhead = pid2.
    i0 = pid0 * BLOCK_M
    j0 = pid1 * BLOCK_N
    # compute pointers base for this batch m
    left_batch_ptr = left_ptr + batchhead * stride_lm
    right_batch_ptr = right_ptr + batchhead * stride_rm
    out_batch_ptr = out_ptr + batchhead * stride_om

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)  # accumulate in fp32 for accuracy

    # loop over k tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        # offsets within block
        i_offsets = i0 + tl.arange(0, BLOCK_M)
        j_offsets = j0 + tl.arange(0, BLOCK_N)
        k_offsets = k0 + tl.arange(0, BLOCK_K)

        # mask for bounds
        i_mask = i_offsets < N
        j_mask = j_offsets < N
        k_mask = k_offsets < N

        # Load A block: left[i, k]
        a_ptrs = left_batch_ptr + i_offsets[:, None] * stride_li + k_offsets[None, :] * stride_lk
        a = tl.load(a_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0).to(tl.float32)

        # Load B block: right transpose: we need right[j, k] but we interpret B as (k, j)
        # right shape [M,N,N] with dims (j,k); we want B block of shape (k, j)
        b_ptrs = right_batch_ptr + k_offsets[:, None] * stride_rk + j_offsets[None, :] * stride_rj
        b = tl.load(b_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0).to(tl.float32)

        # dot product
        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)

    # store result
    out_ptrs = out_batch_ptr + i_offsets[:, None] * stride_oi + j_offsets[None, :] * stride_oj
    tl.store(out_ptrs, acc.to(tl.float16), mask=i_mask[:, None] & j_mask[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # Input layernorm
    # Use float32 weights; input tensor may be float32
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to FP16 for subsequent ops
    x = x.to(torch.float16)

    # Projections
    left = torch.nn.functional.linear(
        x, weights["left_proj.weight"].to(torch.float16), bias=None
    )  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(
        x, weights["right_proj.weight"].to(torch.float16), bias=None
    )

    left_gate = torch.nn.functional.linear(
        x, weights["left_gate.weight"].to(torch.float16), bias=None
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x, weights["right_gate.weight"].to(torch.float16), bias=None
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x, weights["out_gate.weight"].to(torch.float16), bias=None
    ).sigmoid()

    # Apply mask (if any)
    if not nomask:
        # mask: [B,N,N] -> expand to hidden_dim
        mask_h = mask.to(torch.float16).contiguous().view(-1)  # flatten B,N,N
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        # use Triton kernel to multiply left & left_gate & mask etc.
        # we can copy earlier kernel (gate_mask_fuse) but using half.
        # We'll define kernel _gate_mask_fuse_kernel same as previous but with mask optional.
        # Use BLOCK = 1024
        total = left_f.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)
        HAS_MASK = 1
        # If mask is present we pass mask pointer.
        _gate_mask_fuse_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_h,
            total,
            BLOCK=BLOCK,
            HAS_MASK=HAS_MASK,
            H=hidden_dim,
        )
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # Free no longer needed
    del left_gate, right_gate, x

    # Permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    M = B * H

    # reshape to (M, N, N)
    left_mat = left_perm.view(M, N, N)
    right_mat = right_perm.view(M, N, N)
    # allocate output
    out_mat = torch.empty_like(left_mat)  # same dtype (fp16) and device

    # get strides
    # left_mat stride: (stride0, stride1, stride2)
    stride_lm, stride_li, stride_lk = left_mat.stride()
    stride_rm, stride_rj, stride_rk = right_mat.stride()
    stride_om, stride_oi, stride_oj = out_mat.stride()

    # Convert stride to int64 for kernel (they expect 32-bit?). pass as python ints.
    # Ensure all strides are in elements (not bytes).
    # Launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        M,
    )

        left_mat,
        right_mat,
        out_mat,
        stride_lm,
        stride_li,
        stride_lk,
        stride_rm,
        stride_rj,
        stride_rk,
        stride_om,
        stride_oi,
        stride_oj,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, H, N, N)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output layernorm

"""
TriMul (outgoing) forward pass with a custom Triton kernel.

The core O(N³) contraction:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
is implemented as a batched matrix multiplication (left @ rightᵀ) for each
head h using a fused Triton GEMM.  Mask‑and‑gate fusion (when a mask is
present) is also performed with Triton to avoid extra memory passes.
All other linear / LayerNorm operations are done in PyTorch (FP16) and the
final result is returned in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # [total] fp16 *
    right_ptr,              # [total] fp16 *
    left_gate_ptr,          # [total] fp16 *
    right_gate_ptr,         # [total] fp16 *
    mask_ptr,               # [mask_total] fp16 *
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,    # 1 if mask present else 0
    H: tl.constexpr,           # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elements

    left = tl.load(left_ptr + offs, mask=m, other=0.0)
    right = tl.load(right_ptr + offs, mask=m, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=m, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=m, other=0.0)

    if HAS_MASK:
        # broadcast mask across the hidden dimension
        mask_offs = offs // H
        mask_val = tl.load(mask_ptr + mask_offs, mask=m, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=m)
    tl.store(right_ptr + offs, right, mask=m)


@triton.jit
def _batched_matmul_kernel(
    left_ptr,   # [M, N, N] fp16 *
    right_ptr,  # [M, N, N] fp16 *
    out_ptr,    # [M, N, N] fp16 *
    stride_lm, stride_li, stride_lk,
    stride_rm, stride_rj, stride_rk,
    stride_om, stride_oi, stride_oj,
    N: tl.int32,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_i = tl.program_id(0)   # tile row
    pid_j = tl.program_id(1)   # tile col
    pid_m = tl.program_id(2)   # batch * head dimension

    # starting offsets of the tile
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # pointer to the current batch/head slice
    left_batch = left_ptr + pid_m * stride_lm
    right_batch = right_ptr + pid_m * stride_rm
    out_batch = out_ptr + pid_m * stride_om

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K

        # offsets
        i_offs = i_base + tl.arange(0, BLOCK_M)
        j_offs = j_base + tl.arange(0, BLOCK_N)
        k_offs = k_base + tl.arange(0, BLOCK_K)

        # masks for out‑of‑bounds
        i_mask = i_offs < N
        j_mask = j_offs < N
        k_mask = k_offs < N

        # load A tile: left[i, k]
        a_ptrs = left_batch + i_offs[:, None] * stride_li + k_offs[None, :] * stride_lk
        a = tl.load(a_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)
        a = a.to(tl.float32)

        # load B tile: right[j, k]  (we need rightᵀ, so load as (k, j))
        b_ptrs = right_batch + k_offs[:, None] * stride_rk + j_offs[None, :] * stride_rj
        b = tl.load(b_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0)
        b = b.to(tl.float32)

        # dot product
        acc += tl.dot(a, b)

    # store result
    out_ptrs = out_batch + i_offs[:, None] * stride_oi + j_offs[None, :] * stride_oj
    tl.store(out_ptrs, acc.to(tl.float16), mask=i_mask[:, None] & j_mask[None, :])


# ----------------------------------------------------------------------
# Forward entry point
# ----------------------------------------------------------------------


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor of shape [B, N, N] (float32) or None
            - weights_dict: mapping from weight names to torch.Tensors (float32)
            - config_dict: contains keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32) → half precision
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )  # [B, N, N, H]
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gates (half precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask and gates (if mask exists) via Triton kernel
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # flatten everything
        mask_f = mask.to(torch.float16).contiguous().view(-1)          # [B*N*N]
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        total = left_f.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)
        _gate_mask_fuse_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            total,
            BLOCK=BLOCK,
            HAS_MASK=1,
            H=hidden_dim,
        )
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # simple element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate

    # release unused tensors
    del left_gate, right_gate, x

    # ------------------------------------------------------------------
    # 5. Core O(N³) contraction using a fused Triton batched GEMM
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) and then merge batch & head
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    M = B * H

    left_mat = left_perm.view(M, N, N)   # (M, N, N)
    right_mat = right_perm.view(M, N, N) # (M, N, N)
    out_mat = torch.empty_like(left_mat)  # allocate output

    # strides in elements (int)
    stride_lm, stride_li, stride_lk = left_mat.stride()
    stride_rm, stride_rj, stride_rk = right_mat.stride()
    stride_om, stride_oi, stride_oj = out_mat.stride()

    # launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        M,
    )
    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_lm,
        stride_li,
        stride_lk,
        stride_rm,
        stride_rj,
        stride_rk,
        stride_om,
        stride_oi,
        stride_oj,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # return in FP32 as required
    return out.to(torch.float32)
