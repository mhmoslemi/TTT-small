```python
"""
High‑performance outgoing TriMul forward pass.

Algorithm
---------
1. LayerNorm on the input (float32 → half).
2. Linear projections `left` and `right` (half).
3. Compute gating tensors and optionally a binary mask.
4. Fuse mask, left‑gate and right‑gate multiplications into a single
   Triton kernel that updates `left` and `right` in‑place.
5. Core O(N³) contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM (cuBLAS) after transposing `right`.
6. Second LayerNorm, output‑gate, and final linear projection.
7. Return float32 tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask (optional) and left/right gating
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # half*
    right_ptr,           # half*
    left_gate_ptr,       # half*
    right_gate_ptr,      # half*
    mask_ptr,            # half* (broadcasted mask)
    total_elements: tl.int32,
    hidden_dim: tl.int32,
    HAS_MASK: tl.constexpr,   # 1 if mask provided, else 0
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    # Load data
    left = tl.load(left_ptr + offs, mask=valid, other=tl.float16(0.0))

    # ------------------------------------------------------------------

    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()

        right_shape = right.shape

        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)

    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

"""
High‑performance outgoing TriMul forward pass.

Algorithm
---------
1. LayerNorm on the input (float32 → half).
2. Linear projections `left` and `right` (half).
3. Compute gating tensors and optionally a binary mask.
4. Fuse mask, left‑gate and right‑gate multiplications into a single
   Triton kernel that updates `left` and `right` in‑place.
5. Core O(N³) contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM (cuBLAS) after transposing `right`.
6. Second LayerNorm, output‑gate, and final linear projection.
7. Return float32 tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse optional mask with left/right gating.
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # half *
    right_ptr,           # half *
    left_gate_ptr,       # half *
    right_gate_ptr,      # half *
    mask_ptr,            # half * (broadcasted mask)
    total_elements: tl.int32,
    hidden_dim: tl.int32,
    HAS_MASK: tl.constexpr,   # 1 if mask present, else 0
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=valid, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=valid, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=tl.float16(0.0))

    if HAS_MASK:
        # each mask entry corresponds to `hidden_dim` consecutive channel values
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store back in‑place
    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)
