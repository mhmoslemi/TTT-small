def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Convert types
    # Compute layernorm: use torch.nn.functional.layer_norm
    # Because we cannot use Triton for that part easily.

    # Usually we prefer to keep FP16 for speed; we can convert to half early.

    # It's essential that weight tensors may be dtype float32; we move to half.

    # Step: layernorm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    x = x.to(torch.float16)

    # projections
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # gates
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # apply mask (optional) and gating
    if not config.get("nomask", True):
        # mask shape (B,N,N). Expand to hidden dim.
        mask_exp = mask.unsqueeze(-1).to(torch.float16)
        left = left * left_gate * mask_exp
        right = right * right_gate * mask_exp
    else:
        left = left * left_gate
        right = right * right_gate

    # Now contract
    # Permute to batch*H matrices.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B,H,N,N) where dimensions (b,h,k,j)

    B, H, N, _ = left_perm.shape
    batch = B * H
    left_mat = left_perm.view(batch, N, N)
    right_mat = right_perm.view(batch, N, N)

    # allocate output
    out_mat = torch.empty_like(left_mat)  # same dtype half

    # Launch Triton kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    # compute grid sizes
    grid_M = triton.cdiv(N, BLOCK_M)
    grid_N = triton.cdiv(N, BLOCK_N)
    # Use a 2D grid for M,N, and batch dimension as third axis
    grid = (grid_M, grid_N, batch)

    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        GROUP_SIZE_M=8,  # typical
    )
    # restructure output
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output layer norm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16)
    )

    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return float32
    return out.to(torch.float32)

@triton.jit
def batched_matmul_kernel(
    A,
    B,
    C,
    M,
    N,
    K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    GROUP_SIZE_M: tl.constexpr
):
    # Determine block indices
    pid = tl.program_id(0)
    pid_n = tl.program_id(1)
    batch = tl.program_id(2)

    # compute block row index
    # Use block mapping to improve L2 reuse:
    # first compute block linear id
    # This code often uses "pid_m = (pid * GROUP_SIZE_M) % num_pid_m"
    # We'll compute num_pid_m = tl.cdiv(M, BLOCK_M)
    # and num_pid_n = tl.cdiv(N, BLOCK_N)
    # and then compute group of M:
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    # Use pid to compute group and then m
    group_id = pid // num_pid_n
    pid_m = group_id % num_pid_m
    pid_n = pid % num_pid_n

    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for A and B start of this block (including batch offset)
    a_ptrs = A + batch * stride_ab + (offs_m[:, None] * stride_am) + (tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = B + batch * stride_bb + (tl.arange(0, BLOCK_K)[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute offsets for current k block
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K)[None, :] < K - k * BLOCK_K), other=0.0)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K)[:, None] < K - k * BLOCK_K) & mask_n[None, :], other=0.0)
        # Dot product
        acc += tl.dot(a, b)

        # Increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write back
    c_ptrs = C + batch * stride_cb + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    # Cast accumulator to destination dtype
    c = acc.to(C.dtype)   # But need to convert dtype: acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

def batched_matmul_kernel(A, B, C,
                          M, N, K,
                          stride_am, stride_ak, stride_ab,
                          stride_bk, stride_bn, stride_bb,
                          stride_cm, stride_cn, stride_cb,
                          BLOCK_M: tl.constexpr,
                          BLOCK_N: tl.constexpr,
                          BLOCK_K: tl.constexpr):

    pid_n = tl.program_id(1)
    batch = tl.program_id(2)

    # Compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for boundaries
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for A and B
    # A: [batch, M, K]
    a_ptrs = A + batch * stride_ab + (offs_m[:, None] * stride_am) + (tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    # B: [batch, K, N]
    b_ptrs = B + batch * stride_bb + (tl.arange(0, BLOCK_K)[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load A and B blocks
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K)[None, :] < K - k*BLOCK_K), other=0.0)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K)[:, None] < K - k*BLOCK_K) & mask_n[None, :], other=0.0)
        # accumulate
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # Cast acc to c's dtype (fp16)
    acc_f16 = acc.to(C.dtype)  # convert to dtype of C

                          M, N, K,
                          stride_am, stride_ak, stride_ab,
                          stride_bk, stride_bn, stride_bb,
                          stride_cm, stride_cn, stride_cb,
                          BLOCK_M: tl.constexpr,

    pid_n = tl.program_id(1)
    batch = tl.program_id(2)

    # compute block base offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

"""
TriMul (outgoing) forward pass – Triton implementation.

The heavy N³ contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is realized as a batched matrix multiplication:
    left(b,d)  ∈ ℝ^{N×N},  right(b,d) ∈ ℝ^{N×N}
    out(b,d)  = left(b,d) @ right(b,d).T
A custom Triton kernel computes all B·hidden_dim GEMMs in one launch,
operating on FP16 tensors with FP32 accumulation.  The surrounding
normalizations, linear projections, gating and optional masking are kept
in PyTorch (FP16) for simplicity.  This satisfies the requirement of
using at least one Triton kernel while delivering a fast end‑to‑end
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel: batched GEMM (A @ B) where A,B,C have shape
# (batch, M, K), (batch, K, N), (batch, M, N) respectively.
# -------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A, B, C,
    M, N, K,
    stride_am, stride_ak, stride_ab,   # A: (batch, M, K)
    stride_bk, stride_bn, stride_bb,   # B: (batch, K, N)
    stride_cm, stride_cn, stride_cb,   # C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col
    batch = tl.program_id(2)          # batch index

    # off‑sets for the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the block (including batch offset)
    a_ptrs = A + batch * stride_ab + (offs_m[:, None] * stride_am) + (tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = B + batch * stride_bb + (tl.arange(0, BLOCK_K)[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    for _k in range(0, tl.cdiv(K, BLOCK_K)):
        k_mask = tl.arange(0, BLOCK_K) < K - _k * BLOCK_K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & k_mask[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=k_mask[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b, out_dtype=tl.float32)

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write back
    c_ptrs = C + batch * stride_cb + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(C.dtype), mask=mask_m[:, None] & mask_n[None, :])

# -------------------------------------------------------------
# Entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------------------------------------------------------
    # 1. LayerNorm on the channel dimension (float32 -> fp16)
    # ---------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ---------------------------------------------------------
    # 2. Linear projections (fp16)
    # ---------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # ---------------------------------------------------------
    # 3. Gates (fp16)
    # ---------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ---------------------------------------------------------
    # 4. Apply gates and optional mask (in‑place, fp16)
    # ---------------------------------------------------------
    if not nomask and mask is not None:
        mask_exp = mask.unsqueeze(-1).to(torch.float16)          # (B,N,N,1)
        left = left * left_gate * mask_exp
        right = right * right_gate * mask_exp
    else:
        left = left * left_gate
        right = right * right_gate

    # ---------------------------------------------------------
    # 5. Core contraction via batched GEMM (Triton)
    #    left:  (B,N,N,H) → (B,H,N,N) → (B*H,N,N)
    #    right: (B,N,N,H) → (B,H,N,N) with K,J swapped → (B*H,N,N)
    # ---------------------------------------------------------
    # Permute to (B, H, N, N) and make contiguous for good stride layout
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    batch = B * H

    left_mat = left_perm.view(batch, N, N)   # (batch, M, K)
    right_mat = right_perm.view(batch, N, N) # (batch, K, N)
    out_mat = torch.empty_like(left_mat)     # (batch, M, N)

    # Block sizes – tuned for Hopper/B200
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        batch,                     # batch dimension
    )

    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        left_mat.stride(1), left_mat.stride(2), left_mat.stride(0),   # A strides
        right_mat.stride(1), right_mat.stride(2), right_mat.stride(0),# B strides
        out_mat.stride(1), out_mat.stride(2), out_mat.stride(0),      # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection (fp16)
    # ---------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Cast back to float32 for the final output
    return out.to(torch.float32)
