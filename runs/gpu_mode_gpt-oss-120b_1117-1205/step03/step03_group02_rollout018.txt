```python
@triton.jit
def _apply_gate_mask_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr,
    mask_ptr, total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)
```

    B, N, _, _ = x.shape
    M = B * N * N

    # Flatten
    x_flat = x.reshape(M, dim)

    # Combine projection weights
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    # All are (hidden, dim) float32; cast to half
    combined_weight = torch.cat(
        [
            left_proj_w,
            right_proj_w,
            left_gate_w,
            right_gate_w,
            out_gate_w,

    left_gate = left_gate_raw.sigmoid()
    right_gate = right_gate_raw.sigmoid()
    out_gate = out_gate_raw.sigmoid()

    right_raw = right_raw.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    grid = (triton.cdiv(total, BLOCK),)

    return out_final.to(torch.float32)

"""
TriMul (outgoing) forward pass – highly fused implementation.

Algorithm
---------
1. Input LayerNorm (float16).
2. Fuse the five linear projections (left, right, left_gate, right_gate, out_gate)
   into a single matmul by concatenating their weight matrices.
3. Split the result, apply sigmoid to gates.
4. Fuse element‑wise gating (and optional mask) for left/right with a
   Triton kernel (in‑place, half precision).
5. Compute the core O(N³) contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   performed as a batched GEMM on (B·H) matrices of size N×N.
6. Output LayerNorm, multiply by out_gate, final linear projection back to dim.
7. Return float32 output.

Only the gate‑mask fusion (step 4) is written in Triton; all other
operations use highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half*  [total]
    right_ptr,         # half*  [total]
    left_gate_ptr,     # half*  [total]
    right_gate_ptr,    # half*  [total]
    mask_ptr,          # half*  [M]  (broadcasted)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present else 0
    H: tl.constexpr,          # hidden dimension
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each mask element corresponds to H consecutive hidden entries
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) or None
            - weights: dict of parameter tensors
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)
    use_mask = (not nomask) and (mask is not None)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias = weights["norm.bias"].to(torch.float16)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    x = x.to(torch.float16)                     # [B, N, N, dim]

    B, N, _, _ = x.shape
    M = B * N * N                               # number of (i,j) pairs per batch
    x_flat = x.reshape(M, dim)                  # [M, dim]

    # ------------------------------------------------------------------
    # 2. Fused linear projections (5 * hidden output channels)
    # ------------------------------------------------------------------
    # Gather weight matrices (shape: hidden x dim) and concatenate
    w_left = weights["left_proj.weight"]
    w_right = weights["right_proj.weight"]
    w_lgate = weights["left_gate.weight"]
    w_rgate = weights["right_gate.weight"]
    w_outgate = weights["out_gate.weight"]
    combined_w = torch.cat([w_left, w_right, w_lgate, w_rgate, w_outgate], dim=0)
    combined_w = combined_w.to(torch.float16)   # [5*hidden, dim]

    # Single matmul
    proj = F.linear(x_flat, combined_w)         # [M, 5*hidden]

    # ------------------------------------------------------------------
    # 3. Split projections and compute gate sigmoids
    # ------------------------------------------------------------------
    H = hidden
    left_raw   = proj[:, 0:H]
    right_raw  = proj[:, H:2*H]
    left_gate_raw  = proj[:, 2*H:3*H]
    right_gate_raw = proj[:, 3*H:4*H]
    out_gate_raw   = proj[:, 4*H:5*H]

    left_gate  = left_gate_raw.sigmoid()
    right_gate = right_gate_raw.sigmoid()
    out_gate   = out_gate_raw.sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse gating (and optional mask) with a Triton kernel
    # ------------------------------------------------------------------
    left_raw  = left_raw.contiguous()
    right_raw = right_raw.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    total = left_raw.numel()                     # M * hidden
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    if use_mask:
        mask_h = mask.to(torch.float16).contiguous().view(-1)   # [M]
    else:
        mask_h = left_raw   # dummy pointer (won’t be read)

    _gate_mask_fuse_kernel[grid](
        left_raw,
        right_raw,
        left_gate,
        right_gate,
        mask_h,
        total,
        BLOCK=BLOCK,
        HAS_MASK=1 if use_mask else 0,
        H=hidden,
    )
    # left_raw/right_raw now hold the gated (and masked) tensors
    left = left_raw.view(B, N, N, hidden)
    right = right_raw.view(B, N, N, hidden)

    # ------------------------------------------------------------------
    # 5. Core O(N³) contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as batched GEMM over (B·H) matrices of size N×N.
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) = right^T
    B_, H_, N_, _ = left_perm.shape

    left_mat = left_perm.view(B_ * H_, N_, N_)
    right_mat = right_perm.view(B_ * H_, N_, N_)

    out_mat = torch.bmm(left_mat, right_mat)            # (B*H, N, N) half
    out = out_mat.view(B_, H_, N_, N_).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out_gate multiplication, final projection
    # ------------------------------------------------------------------
    out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    out_norm_bias   = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(out, (hidden,), weight=out_norm_weight, bias=out_norm_bias)

    out_gate = out_gate.view(B, N, N, hidden)          # reshape to match out
    out = out * out_gate                               # element‑wise gating

    out_flat = out.view(M, hidden)
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out_proj = F.linear(out_flat, to_out_w)            # (M, dim) half
    out_final = out_proj.view(B, N, N, dim)

    return out_final.to(torch.float32)
