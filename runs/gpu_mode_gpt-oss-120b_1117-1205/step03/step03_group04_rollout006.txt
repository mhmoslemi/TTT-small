```python
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seqlen, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # dtype half for speed
    dtype = torch.float16
    # 1. Norm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # float32
    x_h = x.to(dtype)
    # 2. Projections
    left = F.linear(x_h, weights["left_proj.weight"].to(dtype))
    right = F.linear(x_h, weights["right_proj.weight"].to(dtype))
    # 3. Gates
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(dtype)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(dtype)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(dtype)).sigmoid()
    # 4. Apply mask and gates
    if mask is not None and not config.get("nomask", True):
        # mask shape (bs, seqlen, seqlen) -> broadcast
        mask_h = mask.to(dtype).unsqueeze(-1)  # (bs, N, N, 1)
        # Flatten to 1D for kernel
        left_flat = left.reshape(-1)
        right_flat = right.reshape(-1)
        left_gate_flat = left_gate.reshape(-1)
        right_gate_flat = right_gate.reshape(-1)
        mask_flat = mask_h.reshape(-1)   # length = bs*N*N
        _apply_gate_and_mask(
            left_flat, right_flat,
            left_gate_flat, right_gate_flat,
            mask_flat,
            hidden_dim,
        )
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask: elementwise multiply
        left = left * left_gate
        right = right * right_gate
    # 5. Core contraction using einsum
    # Expect half dtype; keep half to preserve speed
    out = torch.einsum('bikd,bjkd->bijd', left, right)
    # 6. Apply final layernorm, out_gate, final linear
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(dtype))
    # Return float32
    return out.to(torch.float32)
```

```python
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements: tl.int32,
    H: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)
```

"""
TriMul (outgoing) forward pass with Triton‑fused gate/mask.

Algorithm
---------
1. Layer‑Norm the input (float32) → half.
2. Linear projections to hidden_dim (half precision, bias‑free).
3. Compute left/right/out gates (half) with a sigmoid.
4. Fuse optional pairwise mask with left/right gate multiplication
   using a tiny Triton kernel (in‑place, no extra alloc).
5. Core contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   performed with a single torch.einsum which avoids costly permute+copy.
6. Apply output LayerNorm, multiply by out_gate, and final projection.
7. Return result in float32.

Only the gate‑mask fusion is implemented in Triton; the heavy
contraction uses torch.einsum which maps to an efficient cuBLAS kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,           # half*  [total]
    right_ptr,          # half*  [total]
    left_gate_ptr,      # half*  [total]
    right_gate_ptr,     # half*  [total]
    mask_ptr,           # half*  [mask_total] (broadcasted)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present else 0
    H: tl.constexpr,          # hidden dimension (compile‑time)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # Each mask entry corresponds to H consecutive hidden entries
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of mask (if any) and gates:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    All tensors must be contiguous half‑precision on the same CUDA device.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when HAS_MASK==0

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool / numeric) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections (half precision, bias‑free)
    # -----------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # -----------------------------------------------------------------
    # 3. Gate values (half precision)
    # -----------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # -----------------------------------------------------------------
    # 4. Fuse optional mask with gates (Triton)
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → broadcast over hidden_dim
        mask_h = mask.to(torch.float16).unsqueeze(-1).contiguous()
        # Flatten tensors for the kernel
        left_flat = left.view(-1)
        right_flat = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)
        mask_flat = mask_h.view(-1)  # size = B * N * N

        _apply_gate_and_mask(
            left_flat, right_flat,
            left_gate_flat, right_gate_flat,
            mask_flat,
            hidden_dim,
        )
        # Reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask → simple elementwise multiplication
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented with torch.einsum to avoid costly permutes.
    # -----------------------------------------------------------------
    out = torch.einsum("bikd,bjkd->bijd", left, right)

    # -----------------------------------------------------------------
    # 6. Output layer‑norm, out‑gate, final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return as float32 as required by the spec
    return out.to(torch.float32)
