Define kernel:

@triton.jit
def gate_mask_fuse_kernel(left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr, total_elements, BLOCK: tl.constexpr, HAS_MASK: tl.constexpr, H: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements
    # load
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    if HAS_MASK:
        # mask broadcast: each mask element corresponds to H consecutive hidden dim entries
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # layernorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x = x.to(torch.float16)

    # linear projections
    # Build combined weight for left_proj, right_proj, left_gate, right_gate, out_gate
    # trans weight shape: (out_features, in_features) => (hidden_dim, dim)
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)

    # We'll compute left_proj, right_proj, left_gate, right_gate, out_gate via separate linear calls (faster to code).
    # But we also can combine for less overhead. Let's use separate calls to reduce intermediate memory consumption.

    # Flatten input for linear: (B*N*N, dim)
    B, N, _, _ = input_tensor.shape
    x_flat = x.reshape(-1, dim) # (B*N*N, dim)

    left = F.linear(x_flat, left_proj_weight) # (B*N*N, hidden_dim)
    right = F.linear(x_flat, right_proj_weight)
    left_gate = F.linear(x_flat, left_gate_weight).sigmoid()
    right_gate = F.linear(x_flat, right_gate_weight).sigmoid()
    out_gate = F.linear(x_flat, out_gate_weight).sigmoid()

    # Reshape back to 4D
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)
    left_gate = left_gate.view(B, N, N, hidden_dim)
    right_gate = right_gate.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # Apply gating & mask
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1) # shape [B,N,N,1]
        # Flatten everything
        left_flat = left.reshape(-1)
        right_flat = right.reshape(-1)
        left_gate_flat = left_gate.reshape(-1)
        right_gate_flat = right_gate.reshape(-1)

"""
TriMul (outgoing) forward pass.

The implementation follows the AlphaFold3 reference:
1. LayerNorm on the input pairwise features.
2. Linear projections for left/right values and three gating tensors.
3. Optional mask‑gate fusion implemented with a tiny Triton kernel.
4. Contraction over the third sequence dimension using a batched GEMM
   (left @ rightᵀ) for each hidden channel.
5. Output LayerNorm, output gate, and final linear projection.

All heavy matmul work stays in PyTorch (cuBLAS) while the mask‑gate
fusion is off‑loaded to Triton. The final result is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask (if present) with left/right gating.
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,           # half*  [total]
    right_ptr,          # half*  [total]
    left_gate_ptr,      # half*  [total]
    right_gate_ptr,     # half*  [total]
    mask_ptr,           # half*  [mask_total] (broadcast)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask is given, else 0
    H: tl.constexpr,          # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=m, other=0.0)
    right = tl.load(right_ptr + offs, mask=m, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=m, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=m, other=0.0)

    if HAS_MASK:
        # each mask element corresponds to H consecutive hidden entries
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=m, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=m)
    tl.store(right_ptr + offs, right, mask=m)

# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (FP32 -> FP16)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)                       # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2. Linear projections & gates (all FP16)
    # --------------------------------------------------------------
    B, N, _, _ = x.shape
    x_flat = x.reshape(-1, dim)                  # (B*N*N, dim)

    # Individual linear layers (bias=False)
    left_proj   = F.linear(x_flat, weights["left_proj.weight"].to(torch.float16))
    right_proj  = F.linear(x_flat, weights["right_proj.weight"].to(torch.float16))
    left_gate   = F.linear(x_flat, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate  = F.linear(x_flat, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate    = F.linear(x_flat, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # Reshape back to 4‑D tensors
    left   = left_proj.view(B, N, N, hidden_dim)
    right  = right_proj.view(B, N, N, hidden_dim)
    left_gate  = left_gate.view(B, N, N, hidden_dim)
    right_gate = right_gate.view(B, N, N, hidden_dim)
    out_gate   = out_gate.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 3. Fuse mask (if any) with gating using Triton
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B,N,N] -> broadcast over hidden dimension
        mask_h = mask.to(torch.float16).unsqueeze(-1).contiguous()   # [B,N,N,1]

        # Flatten everything
        left_flat      = left.reshape(-1)
        right_flat     = right.reshape(-1)
        left_gate_flat = left_gate.reshape(-1)
        right_gate_flat= right_gate.reshape(-1)
        mask_flat      = mask_h.reshape(-1)                         # size B*N*N

        total = left_flat.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)

        _gate_mask_fuse_kernel[grid](
            left_flat, right_flat,
            left_gate_flat, right_gate_flat,
            mask_flat,
            total,
            BLOCK=BLOCK,
            HAS_MASK=1,
            H=hidden_dim,
        )
        # Restore shapes
        left  = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask: simple element‑wise multiplication
        left  = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 4. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as batched GEMM (left @ rightᵀ)
    # --------------------------------------------------------------
    # Bring hidden dim to 2nd axis for batched matmul
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B_, H_, N_, _ = left_perm.shape
    left_mat  = left_perm.view(B_ * H_, N_, N_)          # (B*H, N, N)
    right_mat = right_perm.view(B_ * H_, N_, N_).transpose(1, 2)  # (B*H, N, N) -> (k,j)

    out_mat = torch.bmm(left_mat, right_mat)            # (B*H, N, N)  FP16
    out = out_mat.view(B_, H_, N_, N_).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # --------------------------------------------------------------
    # 5. Output LayerNorm, out‑gate, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return in float32 as required
    return out.to(torch.float32)
