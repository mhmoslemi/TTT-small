1. Layer‑norm the input tensor (float32 → float16).
2. Linear projections (half‑precision) to the hidden dimension for left/right.
3. Compute left/right/out gates (sigmoid of a linear) in half.
4. If a mask is present, fuse mask, left‑gate and right‑gate in a single
   Triton kernel that updates the left and right tensors in‑place.
   Otherwise multiply by the gates directly.
5. Core O(N³) contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Realised as a batched GEMM on (B·H) matrices of size N×N using torch.bmm.
6. Output LayerNorm, element‑wise gating by out_gate, final linear projection.
7. Return the result in float32.

Only the mask‑gate fusion is performed in Triton; all other operations use
highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: in‑place fusion of mask and left/right gates
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half*  [total]
    right_ptr,         # half*  [total]
    left_gate_ptr,     # half*  [total]
    right_gate_ptr,    # half*  [total]
    mask_ptr,          # half*  [mask_total] (broadcast)
    total_elts: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 -> mask present, 0 -> no mask
    H: tl.constexpr,          # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elts

    # Load tensors (masked loads return 0.0 for out‑of‑range)
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each mask entry corresponds to H consecutive hidden values
        mask_idx = offs // H
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * m
        right = right * right_gate * m
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _fuse_mask_and_gates(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of mask (optional) and gates for left/right tensors.
    All tensors must be contiguous torch.float16.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when HAS_MASK==0

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # tensors are updated in‑place; no return value


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor:  torch.Tensor of shape [B, N, N] (bool/int/float)
            - weights: dict with parameter tensors (float32)
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (half precision)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16)
    )  # shape [B, N, N, hidden_dim]
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16)
    )

    # ------------------------------------------------------------------
    # 3. Gate values (half precision, sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) with gates via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> broadcast over hidden_dim
        mask_h = mask_tensor.unsqueeze(-1).to(torch.float16).contiguous()
        # Flatten all tensors for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        lg_f = left_gate.contiguous().view(-1)
        rg_f = right_gate.contiguous().view(-1)
        mask_f = mask_h.view(-1)  # length B*N*N
        _fuse_mask_and_gates(left_f, right_f, lg_f, rg_f, mask_f, hidden_dim)
        # Reshape back to original layout
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask: simple element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as batched GEMM: (B·H) matrices of size N×N.
    # ------------------------------------------------------------------
    # Prepare tensors for batched matrix multiplication
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)  i,k
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)  k,j
    B, H, N, _ = left_perm.shape

    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # Batched GEMM: half‑precision with float32 accumulation (default for torch.bmm)
    out_mat = torch.bmm(left_mat, right_mat)   # (B*H, N, N), dtype=half

    # Restore original layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate   # element‑wise gating (half)
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16)
    )                     # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 7. Cast back to float32 as required by the interface
    # ------------------------------------------------------------------
    return out.to(torch.float32)
