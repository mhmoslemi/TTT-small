```python
import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)
```

A `@triton.jit` function is NOT a Python function. It is launched with a grid
subscript, and its `tl.constexpr` parameters exist only inside it:

```python

def my_kernel(x_ptr, y_ptr, n, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < n
    tl.store(y_ptr + offs, tl.load(x_ptr + offs, mask=m, other=0.0), mask=m)
```

```python
TriMul forward pass with a fused Triton kernel.

The heavy part of TriMul is:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

We compute:
* LayerNorm and linear projections in PyTorch (float16 for speed).
* A small Triton kernel fuses the optional mask with the left/right
  gating (element‑wise multiplication by sigmoid gates and mask).
* The O(N³) contraction is performed as a batched GEMM using torch.bmm,
  which maps to highly tuned cuBLAS kernels.
* Output LayerNorm, out‑gate scaling and the final linear projection
  complete the forward pass.

The kernel operates on flattened tensors, broadcasting the mask over the
hidden dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # [total] half*
    right_ptr,              # [total] half*
    left_gate_ptr,          # [total] half*
    right_gate_ptr,         # [total] half*
    mask_ptr,               # [mask_total] half* (broadcast)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,    # 1 if mask is present, else 0
    H: tl.constexpr,           # hidden_dim (size of innermost channel)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each hidden channel group shares the same mask entry
        mask_offs = offs // H
        mask_val = tl.load(mask_ptr + mask_offs, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):

    In‑place fusion of:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    `mask` may be None (no masking). All tensors must be contiguous
    half‑precision.

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when HAS_MASK==0

        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )

            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    hidden_dim = config["hidden_dim"]

    # 1. Input LayerNorm (float32) → half precision

    # ------------------------------------------------------------------
    left = F.linear(

        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # 3. Gates (half precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(

        weights["left_gate.weight"].to(torch.float16),

    right_gate = F.linear(

        weights["right_gate.weight"].to(torch.float16),
        bias=None,

    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) with gates via Triton
    # ------------------------------------------------------------------
    if not nomask:

        mask_h = mask.to(torch.float16).contiguous().view(-1)  # [B*N*N]
        # flatten tensors for kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(

            right_f,
            left_gate_f,
            right_gate_f,
            mask_h,

        right = right_f.view_as(right)

        right = right * right_gate

    # Permute to (B, hidden_dim, N, N) for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    # Collapse batch & head dimensions
    left_mat = left_perm.view(B * H, N, N)               # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)             # (B*H, N, N)

    # Batched GEMM with transposition on the second operand
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(

        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )
```

## Strategy from the reasoning-model planning stage

<strategy>
**1. Approach and Rationale**
The TriMul operator, as defined in the AlphaFold3 paper, requires a specific computation that is inherently expensive due to its dependence on computing pairwise products over a third dimension. This triple nested loop structure leads to high computational complexity, particularly when the sequence length (N) is large. To optimize this, we will implement the operation using a batched matrix multiplication (batched GEMM) with explicit management of the input distribution (Normal or Cauchy) and optional masking. This approach leverages PyTorch's `torch.bmm` for efficient handling of the large O(N^3) complexity while retaining controls over floating point precision and additional transformations such as gating.

Our strategy centers on these steps:
- We avoid using trivial operations like average pool and reduce multiple recomputations by batching operations.
- We apply LayerNorm, gating, masking, and final projections in a controlled sequence. Matrix multiplications are batched to improve performance.
- We perform the core contraction (batched GEMM) only once, as specified, to simplify the computation structure and avoid duplicated work, especially when dealing with large tensors.

**2. Mathematical Specification**
The core of the TriMul operator is defined over a 4D tensor `x` of shape \([B, N, N, C]\), following sequential and channel labels. The operation involves:

1. LayerNorm Application:
   \[
   x_{\text{norm}} = \text{LayerNorm}(x, \text{norm\_weight}, \text{norm\_bias})
   \]

2. Projection to Hidden Dimensions:
   \[
   \text{left}_{\text{proj}} = \text{Linear}_\text{proj}(x_{\text{norm}}, \text{left\_proj\_weight})
   \]
   \[
   \text{right}_{\text{proj}} = \text{Linear}_\text{proj}(x_{\text{norm}}, \text{right\_proj\_weight})
   \]

3. Gate Application and Masking:
   \[
   \text{left\_gate} = \text{Linear}_\text{gate}(x_{\text{norm}}, \text{left\_gate\_weight}).\text{sigmoid()}
   \]
   \[
   \text{right\_gate} = \text{Linear}_\text{gate}(x_{\text{norm}}, \text{right\_gate\_weight}).\text{sigmoid()}
   \]
   The masking operation (if applicable) scales the left and right projections by their respective gates and the mask tensor.

4. TriMul Contraction:
   \[
   \text{out} = \text{torch.einsum}('...ikd,...jkd->...ijd', \text{left\_proj}, \text{right\_proj})
   \]
   This step computes the summation over k from 0 to N-1.

5. Output LayerNorm and Scaling:
   \[
   \text{out\_norm} = \text{LayerNorm}(\text{out}, \text{to\_out\_norm\_weight}, \text{to\_out\_norm\_bias})
   \]
   \[
   \text{out\_final} = \text{Linear}(\text{out\_norm}, \text{to\_out\_weight}) \times \text{out\_gate}
   \]

**3. Implementation Procedure**

**4. Search Choices and Useful Variations**
- **Masking**: The presence or absence of a mask can impact the gating mechanism. The mask value is only used during contraction (step 4) and does not require storage beyond the current computation.
- **Data Types**: The use of float16 throughout most stages ensures performance but defeats debugging tools like torch.compile. Converting back to float32 for the final output ensures correct arithmetic, and some intermediate steps may use float32 where necessary (LayerNorm, final Gate).
- **Permutations**: The permutation order ( `[0,2,1,3]` ) ensures efficient tensor reshaping for matrix multiplication. Alternative orders include `[0,3,1,2]` or `[0,1,3,2]`, each with its own cache behavior.
