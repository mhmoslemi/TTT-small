```python
import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)
```

## Triton launch convention

Every failure mode below has appeared in this search. Read this before writing.

A `@triton.jit` function is NOT a Python function. It is launched with a grid
subscript, and its `tl.constexpr` parameters exist only inside it:

```python
@triton.jit
def my_kernel(x_ptr, y_ptr, n, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < n
    tl.store(y_ptr + offs, tl.load(x_ptr + offs, mask=m, other=0.0), mask=m)

def custom_kernel(data):
    ...
    BLOCK = 128                                   # a host variable YOU define
    grid = (triton.cdiv(n, BLOCK),)               # host-side, use triton.cdiv
    my_kernel[grid](x, y, n, BLOCK=BLOCK)         # note the [grid] subscript
```

- `my_kernel(...)` without `[grid]` raises "Cannot call @triton.jit'd outside
  the scope of a kernel".
- `BLOCK`, `BLOCK_M`, `BLOCK_N` are not defined on the host unless you define
  them there. Referencing one you only declared as a `tl.constexpr` parameter
  raises NameError.
- `tl.cdiv` is device-side. On the host use `triton.cdiv`.
- Define the kernel at module top level, never nested inside `custom_kernel`.

You do not have to move everything into Triton. A correct submission that keeps
most of the computation in PyTorch and moves one hot region into a kernel scores;
one that fails to launch scores zero.

TriMul (outgoing) forward pass with Triton‑fused mask & gating.

Algorithm
---------
1. LayerNorm on the input tensor (float32 → half).
2. Linear projections `left` and `right` to `hidden_dim` (half‑precision).
3. Compute four gating tensors via a linear + sigmoid (half‑precision).
4. Fuse mask (if present) and the `left_gate` / `right_gate` multiplications
   into a single Triton kernel that works in‑place on the flattened tensors.
5. Perform the O(N³) contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiply for each head `d`.  We reshape the
   tensors to (B*hidden_dim, N, N) and use `torch.bmm` (cuBLAS) which is
   already highly optimized.
6. Apply a second LayerNorm, the output gate, and the final linear projection.
7. Return the result in float32.

Only the mask‑plus‑gate fusion is executed in a custom Triton kernel,
satisfying the requirement to use a kernel while keeping the overall
runtime low.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse optional mask with left/right gating
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # half*  (flattened left)
    right_ptr,           # half*  (flattened right)
    left_gate_ptr,       # half*  (flattened left_gate)
    right_gate_ptr,      # half*  (flattened right_gate)
    mask_ptr,            # half*  (flattened mask, broadcasted)
    total_elements: tl.int32,
    hidden_dim: tl.int32,
    HAS_MASK: tl.constexpr,   # 1 if mask present, 0 otherwise
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # each mask entry corresponds to `hidden_dim` consecutive channel values
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

        left = left * left_gate * mask

    mask_ptr = mask if mask is not None else left  # dummy when HAS_MASK=0

        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        hidden_dim,
        HAS_MASK=has_mask,

    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)

            - mask: torch.Tensor [B, N, N] (bool or float) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict containing
                * "dim" (int) – input channel size C
                * "hidden_dim" (int) – projection size H
                * "nomask" (bool, optional) – True if mask should be ignored

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)

    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 → half)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half‑precision)
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Gating tensors (half‑precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) and gate multiplications via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → broadcast over hidden_dim
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        # Flatten tensors for the kernel
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_h.view(-1)  # [B*N*N]

        # In‑place fusion
        _apply_gate_and_mask(
            left_flat, right_flat,
            left_gate_flat, right_gate_flat,
            mask_flat,
            hidden_dim,
        )
        # Reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask: simple elementwise multiplication
        left = left * left_gate
        right = right * right_gate

    # Gates are no longer needed
    del left_gate, right_gate, x_h

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as batched GEMM (cuBLAS) after appropriate transposes.
    # ------------------------------------------------------------------
    # Transpose right so that we can use standard bmm (left @ right^T)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) – already transposed

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Batched matrix multiplication in half precision (float32 accumulation)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise gating (half)
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 7. Return in float32 as required by the specification
    # ------------------------------------------------------------------
    return out.to(torch.float32)

### Approach and Rationale
We'll use a hybrid method combining parts of the operations in PyTorch and the core contraction in a Triton kernel using low-precision arithmetic. The main operations (normalization and projections) remain in PyTorch, while the expensive O(N³) computation moves to a Triton kernel optimized with blocking for tensor cores.

The core section of the TriMul operator is:
- `left = left_proj(x)`
- `right = right_proj(x)`
- (Optional) Mask application
- `out[b, i, j, d] = sum_{k} left[b, i, k, d] * right[b, j, k, d]`

We optimize this contraction using a blocked, occupancy-based kernel that aligns with Triton's tensor core-friendly execution.

### Mathematical Specification
**Input:** `x` (normalized input) ∈ ℝ<sup>B×N×N×dim</sup>
**Output:** `out` ∈ ℝ<sup>B×N×N×hidden_dim</sup>

For each batch element `b`, each pair `(i, j)`, and each head `d`:
```
out_b[i, j, d] = sum_{k=0}^{N-1} left_b[i, k, d] * right_b[j, k, d]
```
And finally:
```
out[b, i, j, :] = out[b, i, j, :] -> multiplied by out_gate[b, i, j, :]
```

We'll use:
- LayerNorm (float32)
- Linear projections (float16)
- Mask gating (float16)
- Core contraction (float16)
- Final gating (float16)

import torch
import triton
import triton.language as tl

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, input_dim = input.shape
    hidden_dim = config["hidden_dim"]

    # stage 1: normalization layer (float32)
    # LayerNorm.stats[x] -> mean, variance for x dimension

    # These include LayerNorm done in float32, then convert to float16
    # weights and biases are taken

    norm_weight = weights["norm.weight"].half()
    norm_bias = weights["norm.bias"].half()
    
    # When using LayerNorm, we need mean/variance over ...,-1
    if input.dtype == torch.float16:
        # Can do LayerNorm in float16, but precision takes a hit
        # We go with original design of float32 normalization
        input_full = input.to(torch.float32)
    else:
        input_full = input.clone().to(torch.float32)

    # Compute mean and variance
    mean = input_full.mean(-1)
    inv_std = 1 / torch.rsqrt(mean.var(-1) + 1e-6)
    input_norm = (input_full - mean) * inv_std

    # Stage 2: projection layers (float16)
    left_proj_weight = weights["left_proj.weight"].half()
    right_proj_weight = weights["right_proj.weight"].half()
    
    # Flatten to B*N*N, input_dim -> see common kernel usage for high overlap
    input_view = input_norm.view(batch_size * seq_len * seq_len, -1)
    left = torch.empty(batch_size * seq_len * seq_len, hidden_dim, dtype=torch.float16, device=input.device)
