import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(left_ptr, right_ptr, left_gate_ptr, right_gate_ptr,
                          mask_ptr, total_elements: tl.int32,
                          BLOCK: tl.constexpr, HAS_MASK: tl.constexpr, H: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        mask_offs = offs // H
        mask_val = tl.load(mask_ptr + mask_offs, mask=mask, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer
    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
```

def custom_kernel(data):
    """
    Outgoing TriMul forward pass using fused Triton kernel for mask and gating.
    ...
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm (half)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float16),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float16)
    )
    x = x.to(torch.float16)

    # Linear projections
    left = F.linear(x, weights["left_proj.weight"].to(device=device, dtype=torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(device=device, dtype=torch.float16))

    # Gates
    left_gate = F.linear(x, weights["left_gate.weight"].to(device=device, dtype=torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(device=device, dtype=torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(device=device, dtype=torch.float16)).sigmoid()

    # Fuse mask + gates
    if not nomask:
        mask_h = mask.unsqueeze(-1).to(device=device, dtype=torch.float16).contiguous()
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        lg_f = left_gate.contiguous().view(-1)
        rg_f = right_gate.contiguous().view(-1)
        mask_f = mask_h.view(-1)
        _apply_gate_and_mask(left_f, right_f, lg_f, rg_f, mask_f, hidden_dim)
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask: can also use kernel for in-place multiplication
        left = left * left_gate
        right = right * right_gate

    # Core contraction: out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    # left shape (B,N,N, H)
    # right shape (B,N,N, H)
    # need left[b,i,k,d] and right[b,j,k,d], treat as matmul per head: left @ right^T
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)
    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N) half
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # Output LayerNorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device=device, dtype=torch.float16),
        bias=weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    )
    out = out * out_gate  # gating
    out = F.linear(out, weights["to_out.weight"].to(device=device, dtype=torch.float16))

    return out.to(torch.float32)
```

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor (float32 → float16).
2. Linear projections to the hidden dimension (half‑precision).
3. Compute left/right/out gates (sigmoid of a linear, half‑precision).
4. Fuse optional mask, left‑gate and right‑gate multiplication in a single
   Triton kernel (in‑place on the projected tensors).
5. Core contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix‑multiply over the hidden dimension and is
   executed with torch.bmm (half‑precision, float32 accumulation).
6. Output LayerNorm, out‑gate multiplication and final linear projection
   back to the original feature dimension.
7. Return a float32 tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: in‑place fusion of mask (optional) and left/right gates
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # [total]   half*
    right_ptr,              # [total]   half*
    left_gate_ptr,          # [total]   half*
    right_gate_ptr,         # [total]   half*
    mask_ptr,               # [mask_total] half* (broadcast over hidden)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,     # 1 if a mask is present, else 0
    H: tl.constexpr,            # hidden_dim (compile‑time constant)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load tensors (half)
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # One mask entry per (b,i,k) -> repeat over H hidden values
        mask_offs = offs // H
        mask_val = tl.load(mask_ptr + mask_offs, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store back (in‑place)
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    (mask is optional). All tensors must be torch.float16 and contiguous.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    # Dummy pointer when mask is absent – never dereferenced because HAS_MASK==0
    mask_ptr = mask if mask is not None else left

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # The operation is performed in place; no return value.


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) – may be None
            - weights: mapping of parameter names to torch.Tensor
            - config: dict with keys ``dim`` and ``hidden_dim`` and optional
                      ``nomask`` (bool, default True)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32 → float16)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float16),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float16),
    )
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections (half‑precision)
    # --------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(device=device, dtype=torch.float16),
    )  # [B, N, N, hidden_dim]
    right = F.linear(
        x,
        weights["right_proj.weight"].to(device=device, dtype=torch.float16),
    )

    # --------------------------------------------------------------
    # 3. Gate values (half‑precision, sigmoid)
    # --------------------------------------------------------------
    left_gate = (
        F.linear(
            x,
            weights["left_gate.weight"].to(device=device, dtype=torch.float16),
        )
        .sigmoid()
    )
    right_gate = (
        F.linear(
            x,
            weights["right_gate.weight"].to(device=device, dtype=torch.float16),
        )
        .sigmoid()
    )
    out_gate = (
        F.linear(
            x,
            weights["out_gate.weight"].to(device=device, dtype=torch.float16),
        )
        .sigmoid()
    )

    # --------------------------------------------------------------
    # 4. Fuse mask (if any) with gates via Triton
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → broadcast over hidden_dim
        mask_h = mask.unsqueeze(-1).to(device=device, dtype=torch.float16).contiguous()
        # Flatten tensors for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask_h.view(-1)  # length B*N*N
        _apply_gate_and_mask(
            left_f, right_f, left_gate_f, right_gate_f, mask_f, hidden_dim
        )
        # Reshape back to original layout
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask – simple element‑wise multiplication (still in‑place)
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction:
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as batched GEMM (left @ right^T) per hidden channel.
    # --------------------------------------------------------------
    # Transform to (B, H, N, N) where H = hidden_dim.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) = right transposed
    B, H, N, _ = left_perm.shape

    # Collapse batch & head dimensions for torch.bmm
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Batched matrix multiplication (half, float32 accumulation)
    out_mat = torch.bmm(left_mat, right_mat)   # (B*H, N, N)

    # Restore layout: (B, N, N, H)
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"]
        .to(device=device, dtype=torch.float16),
        bias=weights["to_out_norm.bias"]
        .to(device=device, dtype=torch.float16),
    )
    out = out * out_gate  # element‑wise gating (half‑precision)

    out = F.linear(
        out,
        weights["to_out.weight"].to(device=device, dtype=torch.float16),
    )

    # Return float32 as required by the spec
    return out.to(torch.float32)
