"""
TriMul (outgoing) forward pass – Triton accelerated version.

Algorithm
---------
1. Layer‑norm the input (float32).
2. Cast to FP16 and compute the four linear projections:
   * left/right projections
   * left/right gates
   * output gate
3. Apply optional mask and the two gates (pure PyTorch elementwise ops).
4. Rearrange tensors so that the contraction
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   becomes a batched matrix multiplication:
      (B·D) matrices of shape (N,N) × (N,N)ᵀ .
5. Compute the batched GEMM with a custom Triton kernel
   (FP16 × FP16 → FP32 accumulation → FP16 store).  The kernel
   processes the BH = B·hidden_dim dimension as the program‑id(2)
   axis and tiles the N‑N output with BLOCK_M × BLOCK_N tiles.
6. Reshape the result back to (B,N,N,hidden_dim).
7. Output LayerNorm, elementwise output‑gate multiplication and final
   linear projection back to the original channel dimension.
8. Return a float32 tensor.

Only the O(N³) contraction is implemented in Triton; everything else
uses high‑performance PyTorch ops.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton batched GEMM kernel
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col
    pid_b = tl.program_id(2)   # batch * hidden index

    #------------------------------
    # Offsets & masks
    #------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the current batch matrix
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    # Pointers for the block (row tile, col tile) – K dimension will be stepped in the loop
    a_ptrs = a_batch + offs_m[:, None] * stride_a_row + tl.arange(0, BLOCK_K)[None, :] * stride_a_col
    b_ptrs = b_batch + tl.arange(0, BLOCK_K)[:, None] * stride_b_row + offs_n[None, :] * stride_b_col

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k = tl.cdiv(K, BLOCK_K)
    for _ in range(num_k):
        k_off = tl.arange(0, BLOCK_K)
        # masks for the current K‑tile
        mask_k = k_off + _ * BLOCK_K < K
        a_mask = mask_m[:, None] & mask_k[None, :]
        b_mask = mask_k[:, None] & mask_n[None, :]

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        acc += tl.dot(a, b)          # FP16 → FP32 accumulation

        a_ptrs += BLOCK_K * stride_a_col
        b_ptrs += BLOCK_K * stride_b_row

    # Store the result
    c_mask = mask_m[:, None] & mask_n[None, :]
    c = acc.to(tl.float16)
    c_ptrs = c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, c, mask=c_mask)


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (float32) or None
            - weights_dict: mapping from parameter name to torch.Tensor (float32)
            - config_dict: must contain "dim" (int) and "hidden_dim" (int);
                           may contain "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim] float32

    # --------------------------------------------------------------
    # 2. Cast to half and compute projections / gates
    # --------------------------------------------------------------
    x_h = x.to(torch.float16)

    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))          # [B,N,N,hidden]
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # --------------------------------------------------------------
    # 3. Optional mask and gating fusion (pure pytorch – inexpensive)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 4. Rearrange for batched GEMM
    # --------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right needs transposition of its inner N dimensions: (B, H, N, N) with swapped 2nd/3rd axes
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape
    BH = B * H

    # Collapse batch & channel into a single matrix batch dimension
    left_mat = left_perm.view(BH, N, N)
    right_mat = right_perm.view(BH, N, N)

    # Allocate output buffer
    out_mat = torch.empty_like(left_mat)

    # --------------------------------------------------------------
    # 5. Launch Triton batched GEMM
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        BH,
    )
    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,            # M
        N,            # N
        N,            # K
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 6. Reshape back to (B,N,N,hidden)
    # --------------------------------------------------------------
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,hidden]

    # --------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 8. Cast back to float32 and return
    # --------------------------------------------------------------
    return out.to(torch.float32)
