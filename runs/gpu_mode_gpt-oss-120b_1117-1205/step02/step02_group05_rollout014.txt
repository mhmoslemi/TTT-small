"""
Triton‑accelerated outgoing TriMul (AlphaFold3) forward pass.

Algorithm
---------
1. Layer‑norm over the channel dimension (float32).
2. Fuse the five linear layers (left_proj, right_proj, left_gate, right_gate, out_gate)
   into a single matrix multiplication to avoid multiple reads of the large input.
3. Slice the fused result, apply sigmoid to the three gate tensors.
4. (Optional) apply the pairwise mask to left/right before gating.
5. Multiply left/right by their respective gates.
6. Core contraction: for each (batch, hidden) pair compute
       out[i,j] = Σ_k left[i,k] * right_T[k,j]
   implemented as a batched GEMM (A @ B) in Triton.
7. Layer‑norm over the hidden dimension, multiply by out_gate and project back
   to the original dimension.

All heavy arithmetic is performed in FP16 with FP32 accumulation; final output
is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Batched GEMM kernel: C = A @ B   (A: (batch, M, K), B: (batch, K, N))
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)          # which (batch * hidden) GEMM
    pid_m = tl.program_id(1)             # tile row
    pid_n = tl.program_id(2)             # tile column

    # -------------------- tile offsets --------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this GEMM
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------- loop over K --------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        # B tile: (BLOCK_K, BLOCK_N) – B already stores the transposed operand
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        # block‑wise matrix product
        acc += tl.dot(a, b)

    # -------------------- write back --------------------
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def triton_batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ B for a batch of independent GEMMs using Triton.

    Parameters
    ----------
    A : torch.Tensor (batch, M, K)  FP16, contiguous
    B : torch.Tensor (batch, K, N)  FP16, contiguous (already transposed)

    Returns
    -------
    C : torch.Tensor (batch, M, N)  FP16, contiguous
    """
    assert A.dtype == B.dtype == torch.float16
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tile sizes – chosen for good occupancy on B200 (FP16, 64 KB shared memory)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

    right_flat = right_perm.view(B * H, N, N)  # (B*H, N, N)   B (already transposed)

    # Compute C = A @ B   → (B*H, N, N)
    out_flat = triton_batched_matmul(left_flat, right_flat)  # FP16

    # Restore (B, N, N, H) layout and cast back to FP32
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 8. Final LayerNorm, out‑gate multiplication, and projection back
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
```
