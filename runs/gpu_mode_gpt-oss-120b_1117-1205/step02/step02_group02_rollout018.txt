Define kernel signature:

@triton.jit
def triangle_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    b_id = tl.program_id(0)  # batch dimension
    h_id = tl.program_id(1)  # hidden dim
    pid_m = tl.program_id(2)
    pid_n = tl.program_id(3)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this (b,h)
    a_base = left_ptr + b_id * stride_left_b + h_id * stride_left_h
    b_base = right_ptr + b_id * stride_right_b + h_id * stride_right_h
    c_base = out_ptr + b_id * stride_out_b + h_id * stride_out_h

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K (dimension k)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # mask for k
        mask_k = offs_k < N

        # Load tiles
        # A tile: rows = offs_m, cols = offs_k
        a_ptrs = a_base + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B tile: we need B^T, i.e., rows = k, cols = n (j)
        # Original right has layout (b, j, k, h). To get element (k, j) we need
        b_ptrs = b_base + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)  # fp16 * fp16 -> fp32

    # Store result
    c_ptrs = c_base + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # get program ids
    b_id = tl.program_id(0)
    h_id = tl.program_id(1)
    pid_m = tl.program_id(2)
    pid_n = tl.program_id(3)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # compute base pointers
    a_base = left_ptr + b_id * stride_left_b + h_id * stride_left_h
    b_base = right_ptr + b_id * stride_right_b + h_id * stride_right_h
    c_base = out_ptr + b_id * stride_out_b + h_id * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        mask_k = offs_k < N

        # Load A tile: (M,K)
        a_ptrs = a_base + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile: B is transposed, need (K,N)
        b_ptrs = b_base + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    # Store to out
    c_ptrs = c_base + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

    # Linear projections
    left = torch.nn.functional.linear(
        x_f16,
        weights["left_proj.weight"].to(torch.float16)
    )   # shape [B,N,N,hidden_dim] fp16

    right = torch.nn.functional.linear(
        x_f16,
        weights["right_proj.weight"].to(torch.float16)
    )   # same shape

    if not nomask:

        mask_f16 = mask.to(torch.float16).unsqueeze(-1)   # shape [B,N,N,1]
        left = left * mask_f16

    # Gating

        x_f16,
        weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()

    right_gate = torch.nn.functional.linear(
        x_f16,
        weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()

    out_gate = torch.nn.functional.linear(
        x_f16,
        weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()  # shape [B,N,N,hidden_dim] fp16

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Core contraction via Triton
    # allocate output tensor
    out = torch.empty_like(left, dtype=torch.float16, device=device)

    # precompute strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    B = left.shape[0]

    H = left.shape[3]  # hidden_dim

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # launch kernel
    grid = (
        B,
        H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),

        left,
        right,
        out,

        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # convert out to float32 for next steps
    out_f32 = out.to(torch.float32)

    )

"""
Optimized outgoing TriMul forward pass.

The heavy O(N³·H) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is expressed as a batched matrix multiplication A @ Bᵀ for every hidden
dimension d.  The kernel fuses the transpose of the right operand by using
custom strides, thus avoiding any data copies.  All lightweight operations
(LayerNorm, Linear projections, gating, mask) are performed in PyTorch
while the core contraction is executed in a fused Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # left strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # right strides (original layout, we read it transposed)
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    # out strides
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute for a single (batch, hidden) pair:
        out[i, j] = Σ_k left[i, k] * right[j, k]   (right is read transposed)
    where left and right are fp16 tensors and the accumulation is fp32.
    """
    # program ids
    b_id = tl.program_id(0)          # batch index
    h_id = tl.program_id(1)          # hidden dimension index
    pid_m = tl.program_id(2)         # output row block id
    pid_n = tl.program_id(3)         # output col block id

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this (batch, hidden) pair
    a_base = left_ptr + b_id * stride_left_b + h_id * stride_left_h
    b_base = right_ptr + b_id * stride_right_b + h_id * stride_right_h
    c_base = out_ptr + b_id * stride_out_b + h_id * stride_out_h

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile (M x K) from left: rows=i, cols=k
        a_ptrs = a_base + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile (K x N) from right **transposed**:
        # original right layout: (b, j, k, h)
        # we need rows = k, cols = j → stride_row = stride_right_k, stride_col = stride_right_j
        b_ptrs = b_base + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # fp16 * fp16 -> fp32

    # Store the result (fp16)
    c_ptrs = c_base + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

    # 3. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16)
    )  # [B, N, N, hidden_dim] fp16

    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16)
    )  # [B, N, N, hidden_dim] fp16

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask is [B, N, N]; broadcast to hidden dim and cast to fp16
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 5. Gating tensors (fp16)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()

    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()

    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()  # shape [B, N, N, hidden_dim] fp16

    # Apply gates to the projected tensors
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Core contraction via Triton: batched matmul with transposed right
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # B, N, N, hidden_dim
    assert H == hidden_dim, "hidden dimension mismatch"

    out = torch.empty_like(left, dtype=torch.float16, device=device)  # [B,N,N,H]

    # Strides (in elements)
    s_left_b, s_left_i, s_left_k, s_left_h = left.stride()
    s_right_b, s_right_j, s_right_k, s_right_h = right.stride()
    s_out_b, s_out_i, s_out_j, s_out_h = out.stride()

    # Tile sizes – tuned for NVIDIA B200 (Ampere)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B,                     # batch dimension
        H,                     # hidden dimension
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    tri_mul_outgoing_kernel[grid](
        left, right, out,
        B, N, H,
        s_left_b, s_left_i, s_left_k, s_left_h,
        s_right_b, s_right_j, s_right_k, s_right_h,
        s_out_b, s_out_i, s_out_j, s_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Convert back to fp32 for the remaining layers
    # ------------------------------------------------------------------
    out_f32 = out.to(torch.float32)   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out_f32 = F.layer_norm(
        out_f32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out_f32 = out_f32 * out_gate.to(torch.float32)
    out_f32 = F.linear(
        out_f32,
        weights["to_out.weight"].to(torch.float32)
    )   # final shape [B, N, N, dim]

    return out_f32
