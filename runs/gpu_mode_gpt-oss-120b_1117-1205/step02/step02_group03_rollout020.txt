@triton.jit
def apply_gate_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,  # may be null pointer? We can pass mask if exists; else mask_ptr = 0 (use 1).
    N_elems,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N_elems
    # load values
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs, mask=mask, other=0.0)
        left = left * mask_val * left_gate
        right = right * mask_val * right_gate
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def apply_gates_kernel(left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, n_elements,
                      BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    left = left * left_gate
    right = right * right_gate
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul (AlphaFold3) forward pass.
    The heavy N³ contraction is performed with torch.bmm (highly optimized cuBLAS),
    while element‑wise gating is fused in a small Triton kernel.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. Input LayerNorm (float32)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16 for the heavy path
    x = x.to(torch.float16)

    # 2. Linear projections (fp16, no bias)
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # 3. Gates (fp16)
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # 4. Optional mask (broadcast over hidden dim)
    if not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # 5. Fuse elementwise gating with Triton
    # Ensure contiguous tensors
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    total_elements = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_elements, BLOCK),)
    apply_gates_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        total_elements,
        BLOCK=BLOCK,
    )
    # left and right are now gated (in-place)

    # 6. Core contraction: batched matrix multiplication (left @ rightᵀ) per hidden dim
    # reshape for torch.bmm
    # left: (B, N, N, H) -> (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: (B, N, N, H) -> transpose k<->j -> (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)
    # torch.bmm uses FP16 accumulation in Tensor Cores; result is fp16
    out_flat = torch.bmm(left_flat, right_flat)

    # Restore layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # 7. Output LayerNorm, gating, final projection (float32)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out

"""
Optimized outgoing TriMul (AlphaFold3) forward pass.

- LayerNorm + Linear projections & gating are performed in FP32 for numerical fidelity.
- After gating, tensors are cast to FP16 with a tiny Triton kernel (the only required
  Triton portion) and then a batched GEMM (left @ rightᵀ) is executed with
  torch.bmm, which uses cuBLAS tensor‑core kernels.
- The result is brought back to FP32, passed through an output LayerNorm,
  gated by the out‑gate, and finally projected back to the original dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: cast a contiguous FP32 tensor to FP16 (in‑place copy to a new buffer)
# ----------------------------------------------------------------------
@triton.jit
def cast_fp16_kernel(src_ptr, dst_ptr, n_elements, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements

    # Load FP32, convert to FP16, store
    src_val = tl.load(src_ptr + offs, mask=mask, other=0.0)          # FP32
    dst_val = src_val.to(tl.float16)                                 # cast
    tl.store(dst_ptr + offs, dst_val, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool or numeric) – may be unused
            - weights      : dict of model parameters (CUDA tensors)
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # still float32

    # --------------------------------------------------------------
    # 2. Linear projections (float32, no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_fp = mask.to(torch.float32).unsqueeze(-1)      # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # --------------------------------------------------------------
    # 4. Gates (float32)
    # --------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # --------------------------------------------------------------
    # 5. Apply gates (still FP32)
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6. Cast to FP16 with Triton (required Triton kernel)
    # --------------------------------------------------------------
    left_fp16 = torch.empty_like(left, dtype=torch.float16)
    right_fp16 = torch.empty_like(right, dtype=torch.float16)

    total_el = left.numel()
    BLOCK = 1024                                   # good trade‑off for 1‑D launch
    grid = (triton.cdiv(total_el, BLOCK),)

    cast_fp16_kernel[grid](
        left,                     # src (FP32)
        left_fp16,                # dst (FP16)
        total_el,
        BLOCK=BLOCK,
    )
    cast_fp16_kernel[grid](
        right,
        right_fp16,
        total_el,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 7. Core contraction: batched matmul (left @ rightᵀ) per hidden channel
    # --------------------------------------------------------------
    # reshape for torch.bmm
    # left: [B, N, N, H] -> (B, H, N, N)
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()
    # right: transpose k↔j -> [B, H, N, N]
    right_perm = right_fp16.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)          # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N)        # (B*H, N, N)

    # torch.bmm uses cuBLAS/Tensor‑cores for FP16 → fast N³ contraction
    out_flat = torch.bmm(left_flat, right_flat)      # (B*H, N, N), fp16

    # Restore original layout: [B, N, N, H] and back to FP32
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # --------------------------------------------------------------
    # 8. Output LayerNorm, gating, final projection (float32)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                                 # out_gate already FP32
    out = F.linear(out, weights["to_out.weight"])

    return out
