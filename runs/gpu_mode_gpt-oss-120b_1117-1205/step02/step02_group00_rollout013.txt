def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm input
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # Fuse linear projections of left, right, gates
    # Concatenate weights
    weight_list = [
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ]
    fused_weight = torch.cat([w.to(torch.float16) for w in weight_list], dim=0)  # shape (5*hidden_dim, dim)

    # Flatten inputs for linear
    B, N, _, _ = x_h.shape
    total = B * N * N
    x_flat = x_h.reshape(total, dim)  # (B*N*N, dim)

    fused_out = F.linear(x_flat, fused_weight)  # (total, 5*hidden_dim) half
    # reshape to (B, N, N, 5, hidden_dim)
    fused_out = fused_out.view(B, N, N, 5, hidden_dim)

    left_proj = fused_out[..., 0, :].contiguous()
    right_proj = fused_out[..., 1, :].contiguous()
    left_gate_pre = fused_out[..., 2, :].contiguous()
    right_gate_pre = fused_out[..., 3, :].contiguous()
    out_gate_pre = fused_out[..., 4, :].contiguous()

    # Sigmoid gates
    left_gate = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()
    out_gate = out_gate_pre.sigmoid()

    # Apply mask and gating via Triton kernel
    # left_proj and right_proj are inputs to kernel, will be overwritten with gated values
    has_mask = 0 if nomask else 1
    mask_h = None
    if not nomask:
        # mask shape [B,N,N] -> broadcast to hidden_dim
        mask_h = mask.to(torch.float16).unsqueeze(-1).contiguous()  # shape (B,N,N,1)
        # flatten mask but we need pointer per hidden dim broadcast
        mask_h_flat = mask_h.view(-1)  # length B*N*N
    # Flatten tensors for kernel
    left_flat = left_proj.view(-1)
    right_flat = right_proj.view(-1)
    left_gate_flat = left_gate.view(-1)
    right_gate_flat = right_gate.view(-1)
    mask_flat = mask_h_flat if (not nomask) else None

    _apply_gate_and_mask(
        left_flat, right_flat,
        left_gate_flat, right_gate_flat,
        mask_flat,
        hidden_dim,
        has_mask,
    )
    # reshape back
    left = left_flat.view_as(left_proj)
    right = right_flat.view_as(right_proj)

    # Now perform contraction: out = sum_k left[..., k, :] * right[..., k, :]
    # We need to transpose right as described.
    # Permute to (B, hidden_dim, N, N) for both
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) but (j,k) ; we need right transposed: we will compute bmm(left_perm, right_perm)
    # Actually we need to compute left @ right_transpose where right_transpose dims (K,J) = (k,j). So right_perm currently shape (B, H, N (j), N (k)). To get (k, j) we need to transpose last two dims: right_perm_T = right_perm.transpose(-2, -1). That yields shape (B, H, N (k), N (j)) which matches requirement.
    right_perm_T = right_perm.transpose(-2, -1).contiguous()
    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm_T.view(B * H, N, N)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N), half precision
    # reshape back
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # Output layer norm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # elementwise

    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    return out.to(torch.float32)

"""
custom_kernel implements the outgoing TriMul operator (AlphaFold3).

Key ideas
---------

* The expensive N³ contraction is performed with a batched GEMM
  (torch.bmm) on half‑precision tensors.
* The four per‑position linear projections (left_proj, right_proj,
  left_gate, right_gate) together with the output gate are fused into a
  single matrix multiplication, reducing kernel launch overhead.
* A tiny Triton kernel fuses element‑wise mask application and the
  gate multiplications (left = left_proj * left_gate * mask,
  right = right_proj * right_gate * mask).  The kernel works with an
  optional mask and runs over the flattened tensor.
* All heavy arithmetic runs in fp16; the final result is cast back to
  fp32 as required by the API.

The implementation is fully self‑contained, uses only one Triton kernel
(the mask‑gate fuse kernel) and should work for all configurations
listed in the problem statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: fuse mask (optional) + left/right gates
# --------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # fp16 * total_elements
    right_ptr,         # fp16 * total_elements
    left_gate_ptr,     # fp16 * total_elements
    right_gate_ptr,    # fp16 * total_elements
    mask_ptr,          # fp16 * (total_elements // H)  (broadcast)
    total_elements: tl.int32,
    H: tl.constexpr,               # hidden_dim
    HAS_MASK: tl.constexpr,        # 1 if mask is present, else 0
    BLOCK: tl.constexpr,           # block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load operands
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK == 1:
        # mask is broadcast over the hidden dimension
        mask_index = offs // H
        mask_val = tl.load(mask_ptr + mask_index, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store the results back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim, has_mask):
    """
    In‑place fusion of ``left = left * left_gate * mask`` and
    ``right = right * right_gate * mask`` (mask optional) using Triton.

    All tensors must be contiguous fp16.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    # dummy mask pointer when mask is not used (kernel never reads it)
    mask_ptr = mask if mask is not None else left
    _mask_gate_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        H=hidden_dim,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )
    # results are written in‑place; no return value


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool / numeric)
            - weights: dict with parameter tensors
            - config: dict containing "dim", "hidden_dim", and optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    ##################################################################
    # 1. Input LayerNorm (keep in FP32 for numerical stability)
    ##################################################################
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)                     # [B,N,N,dim] fp16

    ##################################################################
    # 2. Fuse the five per‑position linear projections into one GEMM
    ##################################################################
    # concatenate weights -> shape (5*hidden_dim, dim)
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"].to(torch.float16),
            weights["right_proj.weight"].to(torch.float16),
            weights["left_gate.weight"].to(torch.float16),
            weights["right_gate.weight"].to(torch.float16),
            weights["out_gate.weight"].to(torch.float16),
        ],
        dim=0,
    )  # (5*hidden_dim, dim)

    B, N, _, _ = x_h.shape
    total_positions = B * N * N

    # Flatten to (B*N*N, dim) for a single linear call
    x_flat = x_h.view(total_positions, dim)        # fp16

    # Single large linear projection
    fused_out = F.linear(x_flat, fused_weight)    # (total_positions, 5*hidden_dim) fp16

    # Reshape to (B, N, N, 5, hidden_dim)
    fused_out = fused_out.view(B, N, N, 5, hidden_dim)

    # Split the five components
    left_proj = fused_out[..., 0, :].contiguous()
    right_proj = fused_out[..., 1, :].contiguous()
    left_gate_pre = fused_out[..., 2, :].contiguous()
    right_gate_pre = fused_out[..., 3, :].contiguous()
    out_gate_pre = fused_out[..., 4, :].contiguous()

    # Sigmoid for gates (still fp16)
    left_gate = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()
    out_gate = out_gate_pre.sigmoid()

    ##################################################################
    # 3. Apply mask (if any) and multiply by gates – Triton kernel
    ##################################################################
    has_mask = 0 if nomask else 1
    mask_fp16 = None
    if has_mask:
        # mask: [B,N,N] -> broadcast over hidden_dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1).contiguous()
        mask_flat = mask_fp16.view(-1)            # length B*N*N
    else:
        mask_flat = None

    # Flatten tensors for the kernel
    left_flat = left_proj.view(-1)
    right_flat = right_proj.view(-1)
    left_gate_flat = left_gate.view(-1)
    right_gate_flat = right_gate.view(-1)

    # In‑place fuse mask & gate multiplications
    _apply_gate_and_mask(
        left_flat,
        right_flat,
        left_gate_flat,
        right_gate_flat,
        mask_flat,
        hidden_dim,
        has_mask,
    )

    # Recover the shaped tensors
    left = left_flat.view_as(left_proj)    # now contains left_proj * left_gate * mask
    right = right_flat.view_as(right_proj)

    ##################################################################
    # 4. Core N³ contraction: batched GEMM (left @ rightᵀ)
    ##################################################################
    # reshape to (B, hidden_dim, N, N) for batched gemm
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()        # (B, H, N, N) ==> (j,k)
    # we need Rᵀ i.e. swap last two axes
    right_perm_T = right_perm.transpose(-2, -1).contiguous()  # (B, H, N, N) == (k,j)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.reshape(B * H, N, N)
    right_mat = right_perm_T.reshape(B * H, N, N)

    # Batched half‑precision GEMM (cuBLAS)
    out_mat = torch.bmm(left_mat, right_mat)                 # (B*H, N, N) fp16

    # Back to original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    ##################################################################
    # 5. Output LayerNorm, out‑gate, final projection
    ##################################################################
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate                     # element‑wise (fp16)

    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return as float32 as specified by the API
    return out.to(torch.float32)
