@triton.jit
def fuse_gate_and_mask_kernel(left_ptr, left_gate_ptr, right_ptr, right_gate_ptr, mask_ptr, out_left_ptr, out_right_ptr, total_elems, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems
    left = tl.load(left_ptr + offs, mask=mask)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask)
    right = tl.load(right_ptr + offs, mask=mask)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask)
    if mask_ptr is not None:
        mask_val = tl.load(mask_ptr + offs, mask=mask)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate
    tl.store(out_left_ptr + offs, left, mask=mask)
    tl.store(out_right_ptr + offs, right, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    # Input LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x_fp16 = x.to(torch.float16)

    # Linear projections (combined)
    left_right_weight = torch.cat([weights["left_proj.weight"], weights["right_proj.weight"]], dim=0).to(torch.float16)
    proj = F.linear(x_fp16, left_right_weight)  # shape [B,N,N,2*hidden_dim]
    left = proj[..., :hidden_dim]
    right = proj[..., hidden_dim:]

    # Gates (combined)
    gates_weight = torch.cat([weights["left_gate.weight"], weights["right_gate.weight"], weights["out_gate.weight"]], dim=0).to(torch.float16)
    gates = F.linear(x_fp16, gates_weight)  # shape [B,N,N,3*hidden_dim]
    left_gate = gates[..., :hidden_dim].sigmoid()
    right_gate = gates[..., hidden_dim:2*hidden_dim].sigmoid()
    out_gate = gates[..., 2*hidden_dim:].sigmoid()

    # Prepare mask
    if nomask:
        mask_fp16 = torch.ones_like(left, dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16).expand(-1, -1, -1, hidden_dim).contiguous()
    # Allocate fused outputs
    left_fused = torch.empty_like(left)
    right_fused = torch.empty_like(right)

    # Launch Triton kernel
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    fuse_gate_mask_kernel[grid](
        left_ptr=left,
        left_gate_ptr=left_gate,
        right_ptr=right,
        right_gate_ptr=right_gate,
        mask_ptr=mask_fp16,
        out_left_ptr=left_fused,
        out_right_ptr=right_fused,
        total_elems=total,
        BLOCK=BLOCK,
    )

    # Batched matmul using torch.bmm
    B, N, _, _ = left_fused.shape
    # Permute to (B, hidden, N, N)
    left_perm = left_fused.permute(0, 3, 1, 2).contiguous()
    right_perm = right_fused.permute(0, 3, 2, 1).contiguous()
    # flatten batch*hidden
    left_flat = left_perm.view(B * hidden_dim, N, N)
    right_flat = right_perm.view(B * hidden_dim, N, N)
    out_flat = torch.bmm(left_flat, right_flat)  # fp16 result

    # reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden_dim]

    # Convert to fp32 for layernorm and final linear
    out_fp32 = out.to(torch.float32)

    # Output layernorm
    out_fp32 = F.layer_norm(out_fp32, (hidden_dim,),
                            weight=weights["to_out_norm.weight"],
                            bias=weights["to_out_norm.bias"])

    # Apply out gate (cast)
    out_fp32 = out_fp32 * out_gate.to(torch.float32)

    # Final linear projection
    out_fp32 = F.linear(out_fp32, weights["to_out.weight"])

    return out_fp32

"""
Outgoing TriMul (AlphaFold3) forward pass.

The implementation fuses:
  * Two linear projections (left/right) into a single matmul.
  * Three gating linear layers into a single matmul.
  * Masking and gating element‑wise multiplication into a tiny Triton kernel.
The heavy N³ contraction is performed with a batched torch.bmm (cuBLAS) which
significantly out‑performs a pure Triton GEMM on B200 GPUs.
All other LayerNorms and the final linear projection are kept in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse left/right gating and optional mask
# ----------------------------------------------------------------------
@triton.jit
def fuse_gate_mask_kernel(
    left_ptr,            # *fp16   left projected tensor
    left_gate_ptr,       # *fp16   left gating tensor
    right_ptr,           # *fp16   right projected tensor
    right_gate_ptr,      # *fp16   right gating tensor
    mask_ptr,            # *fp16   broadcast mask (1.0 or 0.0)
    out_left_ptr,        # *fp16   output left after gating/mask
    out_right_ptr,       # *fp16   output right after gating/mask
    total_elems: tl.constexpr,  # total number of elements to process
    BLOCK: tl.constexpr         # tile size (chosen at launch)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    l   = tl.load(left_ptr + offs,        mask=mask, other=0.0)
    lg  = tl.load(left_gate_ptr + offs,   mask=mask, other=0.0)
    r   = tl.load(right_ptr + offs,       mask=mask, other=0.0)
    rg  = tl.load(right_gate_ptr + offs,  mask=mask, other=0.0)
    m   = tl.load(mask_ptr + offs,        mask=mask, other=1.0)  # 1.0 for “no‑mask”

    l_out = l * lg * m
    r_out = r * rg * m

    tl.store(out_left_ptr  + offs, l_out, mask=mask)
    tl.store(out_right_ptr + offs, r_out, mask=mask)


# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor : torch.Tensor [B, N, N] (bool or numeric)
            - weights_dict: dict of torch.Tensors (all float32)
            - config_dict : must contain 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )                               # [B, N, N, dim]  float32
    x_fp16 = x.to(torch.float16)   # fp16 for the heavy linear ops

    # ------------------------------------------------------------------
    # 2. Combined projection: left & right (half‑precision)
    # ------------------------------------------------------------------
    # concat weight matrices on the output dimension
    proj_weight = torch.cat(
        [weights["left_proj.weight"], weights["right_proj.weight"]], dim=0
    ).to(torch.float16)          # (2*hidden, dim)
    proj = F.linear(x_fp16, proj_weight)      # [B, N, N, 2*hidden]
    left  = proj[..., :hidden_dim]            # [B, N, N, hidden]
    right = proj[..., hidden_dim:]            # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 3. Combined gating: left_gate, right_gate, out_gate (half‑precision)
    # ------------------------------------------------------------------
    gate_weight = torch.cat(
        [
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"]
        ],
        dim=0,
    ).to(torch.float16)          # (3*hidden, dim)
    gates = F.linear(x_fp16, gate_weight)    # [B, N, N, 3*hidden]
    left_gate  = gates[..., :hidden_dim].sigmoid()
    right_gate = gates[..., hidden_dim:2 * hidden_dim].sigmoid()
    out_gate   = gates[..., 2 * hidden_dim:].sigmoid()   # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 4. Fuse mask and gating with a tiny Triton kernel
    # ------------------------------------------------------------------
    if nomask:
        mask_fp16 = torch.ones_like(left, dtype=torch.float16, device=device)
    else:
        # mask: [B, N, N]  ->  [B, N, N, hidden] (broadcast)
        mask_fp16 = mask_tensor.unsqueeze(-1).to(torch.float16)
        mask_fp16 = mask_fp16.expand(-1, -1, -1, hidden_dim).contiguous()

    # allocate output tensors for the fused results
    left_fused = torch.empty_like(left)
    right_fused = torch.empty_like(right)

    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    fuse_gate_mask_kernel[grid](
        left, left_gate,
        right, right_gate,
        mask_fp16,
        left_fused, right_fused,
        total,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 5. Core contraction: batched matmul (A @ B) for every hidden dim
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    => treat (b, d) as a batch dimension and use torch.bmm
    # ------------------------------------------------------------------
    B, N, _, _ = left_fused.shape

    # (B, hidden, N, N) layout required for batched GEMM
    left_perm  = left_fused.permute(0, 3, 1, 2).contiguous()
    right_perm = right_fused.permute(0, 3, 2, 1).contiguous()  # transpose k↔j

    # flatten batch*hidden -> (B*hidden, N, N)
    left_flat  = left_perm.view(B * hidden_dim, N, N)
    right_flat = right_perm.view(B * hidden_dim, N, N)

    # batched matrix multiplication (fp16 inputs, fp32 accumulation inside cuBLAS)
    out_flat = torch.bmm(left_flat, right_flat)   # [B*hidden, N, N] fp16

    # reshape to original layout
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection (float32)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    out_fp32 = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    out_fp32 = out_fp32 * out_gate.to(torch.float32)

    out_fp32 = F.linear(out_fp32, weights["to_out.weight"])

    return out_fp32
