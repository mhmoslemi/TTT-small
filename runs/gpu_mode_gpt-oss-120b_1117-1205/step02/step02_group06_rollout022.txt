@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)  # batch dimension (B*hidden)
    pid_m = tl.program_id(1)     # tile row
    pid_n = tl.program_id(2)     # tile col

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)

        a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)
        b = tl.load(b_ptrs,

                    other=0.0)

             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _triton_batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched fp16 matrix multiplication C = A @ B where
    A: (batch, M, K), B: (batch, K, N) are contiguous fp16 tensors.
    Returns C in fp16.
    """
    assert A.is_contiguous() and B.is_contiguous()
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tunable block sizes – chosen to be multiples of 16 for tensor cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64   # 64 gives good K‑loop reduction for typical N (256‑1024)

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward pass.

    Arguments:
        data: tuple (input_tensor, mask,
                     weights_dict, config_dict)
        - input_tensor: torch.Tensor [B, N, N, dim] (float32)
        - mask: torch.Tensor [B, N, N] (bool/float) – may be empty
        - weights_dict: mapping from parameter name to torch.Tensor
        - config_dict: must contain 'dim' and 'hidden_dim'.
                       Optional 'nomask' (bool) indicates mask should be ignored.

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. LayerNorm on channel dim
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # work in fp16 for the heavy part
    x_fp16 = x.to(torch.float16)

    # 2. Linear projections (no bias)
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # 3. Gating projections
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # 4. Apply mask (if any) and gates
    if not nomask:
        # mask broadcast to hidden dim
        mask_fp = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp
        right = right * mask_fp

    left = left * left_gate
    right = right * right_gate

    # 5. Core contraction: batched matmul A @ B^T  (out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d])
    # Permute to bring hidden dim before the two spatial dims
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) where second N is the K axis

    B_, H, N, _ = left_perm.shape
    # flatten batch and hidden dim
    A_flat = left_perm.view(B_ * H, N, N)   # (B*H, M, K)
    B_flat = right_perm.view(B_ * H, N, N)  # (B*H, K, N)

    # perform the batched matmul in fp16 using Triton
    out_flat = _triton_batched_matmul(A_flat, B_flat)   # (B*H, N, N) fp16

    # reshape back to (B, N, N, H)
    out = out_flat.view(B_, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # 6. Output layer‑norm, gating and final linear projection
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out

Outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‑norm the input across the channel dimension.
2. Project the normalized tensor to a hidden dimension with two linear layers
   (left_proj, right_proj) – all in fp16.
3. Compute three sigmoid gates (left_gate, right_gate, out_gate) also in fp16.
4. Apply optional mask, then element‑wise multiply the projections by their
   corresponding gates.
5. Core contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication A @ Bᵀ where
   A = left.permute(0,3,1,2) → (B, H, N, N)
   B = right.permute(0,3,2,1) → (B, H, N, N)

   (B·H, N, N) × (B·H, N, N)ᵀ.  The GEMM is performed by a custom Triton
   kernel that tiles the N dimension (128×128 tiles) and reduces the K‑loop
   using a 64‑wide tile, thus exploiting Tensor‑Core instructions.
6. Layer‑norm the result, multiply by the output gate, and finally project
   back to the original channel dimension.

The implementation keeps the heavy arithmetic in fp16 and only materialises
the final tensor in fp32, matching the reference PyTorch behavior.

    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tile sizes (multiples of 16 → Tensor Core friendly)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64   # reduces K‑loops for all supported N (256‑1024)

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operation.

    Args:
        data: (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored if config['nomask'] is True
            - weights_dict: mapping of parameter name → torch.Tensor
            - config_dict: must contain 'dim' and 'hidden_dim'.
                           Optional key 'nomask' (bool) controls mask usage.

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # Cast to fp16 for the heavy path
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Gating projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 4. Optional mask and gate application
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: batched GEMM A @ Bᵀ
    #    left_perm: (B, H, N, N)  where H = hidden_dim
    #    right_perm: (B, H, N, N) with axes (k, j) = transpose of the second N dim
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    # Collapse batch and hidden dimensions for the kernel
    A_flat = left_perm.view(B * H, N, N)       # (B*H, M=N, K=N)
    B_flat = right_perm.view(B * H, N, N)      # (B*H, K=N, N)

    # Perform the batched matmul in fp16 (Trition kernel)
    out_flat = _triton_batched_matmul(A_flat, B_flat)   # (B*H, N, N) fp16

    # Reshape back to (B, N, N, H) and cast to fp32 for the remaining ops
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output layer‑norm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
