```
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def fused_gate_mask_kernel(
    out_ptr,
    proj_ptr,
    gate_ptr,
    mask_ptr,
    total,
    BLOCK_SIZE: tl.constexpr,
    H: tl.constexpr,
    USE_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # Load projection and gate pre-sigmoid
    proj = tl.load(proj_ptr + offs, mask=mask)
    gate_pre = tl.load(gate_ptr + offs, mask=mask)

    # sigmoid
    gate = 1.0 / (1.0 + tl.exp(-gate_pre))

    if USE_MASK:
        mask_idx = offs // H
        m = tl.load(mask_ptr + mask_idx, mask=mask)
        out = proj * gate * m
    else:
        out = proj * gate

    tl.store(out_ptr + offs, out, mask=mask)

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operation.
    The kernel fuses elementwise mask, gate (sigmoid) and projection multiplication.
    Linear projections are merged (proj+gate) to halve the number of GEMMs.
    The cubic contraction is performed as a batched GEMM (left @ right.T) using
    cuBLAS, which is much faster than a naïve Triton implementation.
    The final layer‑norm and output projection are kept in PyTorch for
    numerical stability.
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    
    # ------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x_norm = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    # Cast to fp16 for the heavy work
    x = x_norm.to(torch.float16)
    
    # ------------------------------------------------------------
    # 2. Project to hidden dimension (proj + gate fused)
    # ------------------------------------------------------------
    # concatenate weight matrices for proj and gate (no bias)
    left_comb_weight = torch.cat([weights["left_proj.weight"],
                                  weights["left_gate.weight"]], dim=0).to(device).to(torch.float16)
    right_comb_weight = torch.cat([weights["right_proj.weight"],
                                   weights["right_gate.weight"]], dim=0).to(device).to(torch.float16)
    
    # linear gives (B,N,N,2*hidden_dim)
    left_comb = F.linear(x, left_comb_weight)   # fp16
    right_comb = F.linear(x, right_comb_weight)
    
    # split into projection and gate pre‑sigmoid parts
    proj_left = left_comb[..., :hidden_dim]
    gate_left_pre = left_comb[..., hidden_dim:]
    proj_right = right_comb[..., :hidden_dim]
    gate_right_pre = right_comb[..., hidden_dim:]
    
    # ------------------------------------------------------------
    # 3. Out‑gate
    # ------------------------------------------------------------
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)
    out_gate_pre = F.linear(x, out_gate_weight)   # fp16
    out_gate = torch.sigmoid(out_gate_pre)        # fp16
    
    # ------------------------------------------------------------
    # 4. Fuse mask, gate (sigmoid) and projection multiplication via Triton
    # ------------------------------------------------------------
    if mask_tensor is None:
        use_mask = False
        mask_fp16 = torch.empty(0, dtype=torch.float16, device=device)
    else:
        use_mask = True
        mask_fp16 = mask_tensor.to(torch.float16).to(device)
    
    # allocate fused outputs
    left_out = torch.empty_like(proj_left)
    right_out = torch.empty_like(proj_right)
    
    total = proj_left.numel()                     # B * N * N * hidden_dim
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    
    # flatten tensors for kernel
    proj_left_flat = proj_left.contiguous().view(-1)
    gate_left_pre_flat = gate_left_pre.contiguous().view(-1)
    proj_right_flat = proj_right.contiguous().view(-1)
    gate_right_pre_flat = gate_right_pre.contiguous().view(-1)
    mask_flat = mask_fp16.contiguous().view(-1) if use_mask else torch.empty(0, dtype=torch.float16, device=device)
    
    # launch for left branch
    fused_gate_mask_kernel[grid](
        left_out,
        proj_left_flat,
        gate_left_pre_flat,
        mask_flat,
        total,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
        USE_MASK=use_mask,
    )
    # launch for right branch
    fused_gate_mask_kernel[grid](
        right_out,
        proj_right_flat,
        gate_right_pre_flat,
        mask_flat,
        total,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
        USE_MASK=use_mask,
    )
    
    # ------------------------------------------------------------
    # 5. Cubic contraction via batched GEMM (left @ right^T)
    # ------------------------------------------------------------
    # shape: (B, N, N, H) -> (B, H, N, N)
    left_perm = left_out.permute(0, 3, 1, 2).contiguous()
    right_perm = right_out.permute(0, 3, 2, 1).contiguous()
    
    LH = B * hidden_dim
    left_mat = left_perm.view(LH, N, N)
    right_mat = right_perm.view(LH, N, N)
    
    # batched matrix multiplication (fp16)
    out_mat = torch.bmm(left_mat, right_mat)    # (B*H, N, N)
    
    # reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    
    # ------------------------------------------------------------
    # 6. Post‑processing (LayerNorm, out‑gate & final projection)
    # ------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    
    # apply final gating
    out_norm = out_norm * out_gate.to(torch.float32)
    
    to_out_w = weights["to_out.weight"].to(device).to(torch.float32)
    output = F.linear(out_norm, to_out_w)  # (B, N, N, dim)
    return output

"""
TriMul (outgoing) forward pass with Triton acceleration.

The kernel fuses the three element‑wise operations:
    proj * sigmoid(gate) * mask
for both the left and right branches.  Projection and gate linear
weights are concatenated so that a single GEMM produces both the
projection and the pre‑sigmoid gate, halving the number of matrix
multiplications.  The cubic contraction is expressed as a batched
matrix‑multiply (left @ rightᵀ) and is executed by cuBLAS via
torch.bmm, which is far faster than a naïve Triton implementation.
LayerNorms and the final projection remain in PyTorch for numerical
stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def fused_gate_mask_kernel(
    out_ptr,        # fp16 output
    proj_ptr,       # fp16 projection tensor
    gate_ptr,       # fp16 pre‑sigmoid gate tensor
    mask_ptr,       # fp16 mask tensor (flattened, length = B*N*N)
    total,          # total number of elements = B*N*N*H
    BLOCK_SIZE: tl.constexpr,
    H: tl.constexpr,           # hidden dimension (size of innermost axis)
    USE_MASK: tl.constexpr,    # 1 => apply mask, 0 => skip mask
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total

    # load projection and gate pre‑sigmoid
    proj = tl.load(proj_ptr + offs, mask=active)
    gate_pre = tl.load(gate_ptr + offs, mask=active)

    # sigmoid(gate_pre)
    gate = 1.0 / (1.0 + tl.exp(-gate_pre))

    if USE_MASK:
        # each group of H consecutive elements shares one mask entry
        mask_idx = offs // H
        m = tl.load(mask_ptr + mask_idx, mask=active)
        out = proj * gate * m
    else:
        out = proj * gate

    tl.store(out_ptr + offs, out, mask=active)


def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor : torch.Tensor of shape [B, N, N] (float32) or None
            - weights_dict: contains the model weights
            - config_dict : {"dim": int, "hidden_dim": int}
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)  # heavy work in fp16

    # -----------------------------------------------------------------
    # 2. Linear projection + gate (concatenated weight matrices)
    # -----------------------------------------------------------------
    left_comb_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device).to(torch.float16)          # (2*H, dim)
    right_comb_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device).to(torch.float16)          # (2*H, dim)

    left_comb = F.linear(x, left_comb_weight)   # (B,N,N,2*H), fp16
    right_comb = F.linear(x, right_comb_weight) # (B,N,N,2*H), fp16

    # split projection / gate-pre‑sigmoid
    proj_left = left_comb[..., :hidden_dim]
    gate_left_pre = left_comb[..., hidden_dim:]
    proj_right = right_comb[..., :hidden_dim]
    gate_right_pre = right_comb[..., hidden_dim:]

    # -----------------------------------------------------------------
    # 3. Out‑gate (separate linear + sigmoid)
    # -----------------------------------------------------------------
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)
    out_gate_pre = F.linear(x, out_gate_weight)       # (B,N,N,H), fp16
    out_gate = torch.sigmoid(out_gate_pre)            # (B,N,N,H), fp16

    # -----------------------------------------------------------------
    # 4. Fuse mask, gate (sigmoid) and projection via Triton
    # -----------------------------------------------------------------
    if mask_tensor is None:
        use_mask = 0
        mask_fp16 = torch.empty(0, dtype=torch.float16, device=device)
    else:
        use_mask = 1
        mask_fp16 = mask_tensor.to(dtype=torch.float16, device=device)

    # allocate fused outputs
    left_out = torch.empty_like(proj_left)   # (B,N,N,H), fp16
    right_out = torch.empty_like(proj_right) # (B,N,N,H), fp16

    total = proj_left.numel()                # B * N * N * H
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    # flatten tensors for kernel launch
    proj_left_f = proj_left.contiguous().view(-1)
    gate_left_f = gate_left_pre.contiguous().view(-1)
    proj_right_f = proj_right.contiguous().view(-1)
    gate_right_f = gate_right_pre.contiguous().view(-1)
    mask_f = mask_fp16.contiguous().view(-1) if use_mask else torch.empty(0, dtype=torch.float16, device=device)

    # left branch
    fused_gate_mask_kernel[grid](
        left_out,
        proj_left_f,
        gate_left_f,
        mask_f,
        total,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
        USE_MASK=use_mask,
    )
    # right branch
    fused_gate_mask_kernel[grid](
        right_out,
        proj_right_f,
        gate_right_f,
        mask_f,
        total,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
        USE_MASK=use_mask,
    )

    # -----------------------------------------------------------------
    # 5. Cubic contraction via batched GEMM (left @ rightᵀ)
    # -----------------------------------------------------------------
    # shape to (B, H, N, N) for bmm; right needs transposition of the
    # last two axes to obtain the inner‑product over k.
    left_perm = left_out.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_out.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)

    LH = B * hidden_dim
    left_mat = left_perm.view(LH, N, N)
    right_mat = right_perm.view(LH, N, N)

    out_mat = torch.bmm(left_mat, right_mat)   # (B*H, N, N), fp16

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # -----------------------------------------------------------------
    # 6. Post‑processing: LayerNorm, out‑gate, final projection
    # -----------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # apply final gating
    out_norm = out_norm * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float32)  # (dim, H)
    output = F.linear(out_norm, to_out_w)  # (B, N, N, dim), float32

    return output
