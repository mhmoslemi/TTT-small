                          stride_a_batch, stride_a_m, stride_a_k,
                          stride_b_batch, stride_b_k, stride_b_n,
                          stride_c_batch, stride_c_m, stride_c_n,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):

    pid_n = tl.program_id(2)

        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def triton_matmul_fp16_to_fp32(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B where

    Returns C: (batch, M, N) float32
    """

    B_total, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    c = torch.empty((B_total, M, N), dtype=torch.float32, device=a.device)

    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_k, stride_b_n = b.stride()
    stride_c_batch, stride_c_m, stride_c_n = c.stride()

    # Choose tile sizes adaptively
    if M <= 256:
        BLOCK_M = 64
    else:
        BLOCK_M = 128
    if N <= 256:
        BLOCK_N = 64
    else:
        BLOCK_N = 128
    BLOCK_K = 32  # good for most K

    grid = (
        B_total,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](

        b_ptr=b,
        c_ptr=c,
        M=M,
        N=N,
        K=K,

        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )
    return c

def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Arguments:
        data: Tuple (input, mask, weights, config)
            input: torch.Tensor [B, N, N, dim] (float32)
            mask: torch.Tensor [B, N, N] (bool or float)
            weights: dict of model parameters (torch.Tensor)
            config: dict containing 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim], float32

    # -------------------------------------------------
    # 2. Cast to fp16 for heavy linear ops
    # -------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -------------------------------------------------
    # 3. Linear projections (fp16)
    # -------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # -------------------------------------------------
    # 4. Apply mask if present
    # -------------------------------------------------
    if not nomask:
        # mask may be bool; cast to float16 and broadcast
        mask_fp = mask.unsqueeze(-1).to(torch.float16)  # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # -------------------------------------------------
    # 5. Gating (fp16) and apply
    # -------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 6. Core contraction via Triton
    # -------------------------------------------------
    # Rearrange tensors to (B, hidden, N, N) order for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) with (k,j)

    B, H, N, _ = left_perm.shape
    # Flatten batch & head dimension
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Perform batched matmul (fp16 inputs, fp32 output)
    out_flat = triton_matmul_fp16_to_fp32(left_flat, right_flat)  # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    # out currently float32

    # -------------------------------------------------
    # 7. Output LayerNorm, out‑gate and final projection
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out

"""
Triton‑accelerated forward pass for the outgoing TriMul operator (AlphaFold3).

Algorithm
---------
1. Layer‑norm the input (float32).
2. Cast to fp16 and apply linear projections (left/right) and gating.
3. Optional element‑wise mask is applied in fp16.
4. Core contraction:
       out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   is a batched matrix multiplication of shape
   (B·H, N, N) @ (B·H, N, N)^T.
   The multiplication is performed by a custom Triton kernel that
   uses Tensor‑Core‑accelerated `tl.dot` (fp16 inputs, fp32 accumulation).
5. Reshape the result back to (B, N, N, H) and apply the output layer‑norm,
   out‑gate, and final linear projection (float32).

The heavy O(N³) contraction is the only part off‑loaded to Triton; all
other operations use high‑performance PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Batched GEMM kernel (C = A @ B) where B is already transposed.
# A: (batch, M, K) fp16
# B: (batch, K, N) fp16  (here B = right_perm)
# C: (batch, M, N) fp32
# -------------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)       # combined batch*head dimension
    pid_m = tl.program_id(1)          # block row
    pid_n = tl.program_id(2)          # block column

    # offsets inside the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # load A tile (M x K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        # load B tile (K x N) – B is already transposed
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        # multiply‑accumulate
        acc += tl.dot(a, b)

    # store result (fp32)
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def triton_matmul_fp16_to_fp32(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched matmul C = A @ B using Triton.
    A, B : (batch, M, K) and (batch, K, N) in fp16.
    Returns C : (batch, M, N) in fp32.
    """
    assert a.dtype == b.dtype == torch.float16
    B_tot, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    c = torch.empty((B_tot, M, N), dtype=torch.float32, device=a.device)

    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_k, stride_b_n = b.stride()
    stride_c_batch, stride_c_m, stride_c_n = c.stride()

    # heuristic tile sizes
    BLOCK_M = 128 if M > 256 else 64
    BLOCK_N = 128 if N > 256 else 64
    BLOCK_K = 32

    grid = (
        B_tot,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=M,
        N=N,
        K=K,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        # launch config
        num_warps=8,
        num_stages=3,
    )
    return c


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool or numeric)
            - weights_dict : mapping of parameter names to torch.Tensor
            - config_dict  : must contain 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim]

    # -------------------------------------------------
    # 2. Cast to fp16 for the heavy linear work
    # -------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -------------------------------------------------
    # 3. Linear projections (fp16)
    # -------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # -------------------------------------------------
    # 4. Optional mask (broadcast over hidden dim)
    # -------------------------------------------------
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)   # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # -------------------------------------------------
    # 5. Gating (fp16) and apply
    # -------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 6. Core contraction via Triton (fp16 → fp32)
    # -------------------------------------------------
    # Reorder to (B, H, N, N) where H = hidden_dim.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) = (B, H, K, J)

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # Batched matmul: (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    out_flat = triton_matmul_fp16_to_fp32(left_flat, right_flat)  # fp32

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------
    # 7. Output LayerNorm, out‑gate and final projection (float32)
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
