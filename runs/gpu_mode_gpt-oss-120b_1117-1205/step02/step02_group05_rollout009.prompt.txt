```python
import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)
```

A `@triton.jit` function is NOT a Python function. It is launched with a grid
subscript, and its `tl.constexpr` parameters exist only inside it:

```python

def my_kernel(x_ptr, y_ptr, n, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < n
    tl.store(y_ptr + offs, tl.load(x_ptr + offs, mask=m, other=0.0), mask=m)
```

```python
TriMul (outgoing) forward pass with a fused Triton batched matrix multiplication.

Algorithm:
1. Apply LayerNorm to the input (float32).
2. Project to hidden dimension with two linear layers (half‑precision).
3. Optional mask multiplication.
4. Compute gating values (half‑precision sigmoid) and apply them.
5. Core contraction:
   For each batch element and hidden channel d,
   compute out[...,i,j,d] = Σ_k left[...,i,k,d] * right[...,j,k,d]
   which is a batched matrix multiplication left @ right.T.
   This is performed by a custom Triton kernel that processes
   (B·hidden_dim) independent GEMMs of size N×N.
6. Apply LayerNorm on the hidden dimension, multiply by out‑gate,
   and project back to the original dimension.
All heavy arithmetic is in FP16 with FP32 accumulation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton batched GEMM kernel (A @ B.T) for the core contraction
# --------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)          # which (batch, hidden) pair
    pid_m = tl.program_id(1)             # tile row
    pid_n = tl.program_id(2)             # tile column

    # Block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    # Accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Main loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        # Load B tile (K x N) – note that B already holds the transposed operand
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        # Compute block‑wise dot product
        acc += tl.dot(a, b)

    # Write back the result (FP16)
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def triton_batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B where
        A : (batch, M, K)    FP16
        B : (batch, K, N)    FP16   (the second operand is already transposed)
    Returns C : (batch, M, N) FP16
    """
    assert A.dtype == B.dtype == torch.float16
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tile sizes – chosen for good occupancy on B200 (FP16, 64 KB shared memory)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

def custom_kernel(data):

    Triton‑accelerated forward pass for the outgoing TriMul operation.

    Arguments:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) – may be ignored
            - weights: dict of model parameters
            - config: dict containing 'dim', 'hidden_dim', and optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    # Configuration
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. LayerNorm over the channel dimension (in float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # Cast to half for all subsequent heavy arithmetic
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections: left and right (half‑precision, no bias)
    # --------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → [B, N, N, 1] for broadcasting
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 4. Gating values (half‑precision sigmoid) and application
    # --------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction: batched GEMM (left @ right.T) in Triton
    # --------------------------------------------------------------
    # Rearrange tensors to (batch, hidden, N, N) layout
    # left: (B, N, N, H) → (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right needs transposition of the two sequence axes: (B, N, N, H) → (B, H, N, N) with axes (k, j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape
    # Merge batch and hidden dimensions for the batched GEMM
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Compute out_flat = left @ right.T where right_flat already stores the transposed operand
    out_flat = triton_batched_matmul(left_flat, right_flat)   # (B*H, N, N), FP16

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)   # back to FP32 for the final layers

    # --------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate multiplication, and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
```

## Strategy from the reasoning-model planning stage

<strategy>
**1. Approach and Rationale**

The core operation of TriMul involves computing a batched matrix multiplication of two matrices with shared dimensions along the sequence, while maintaining separability across the batch and hidden dimensions. This can be viewed as a contraction over the third dimension (k) of two 4D tensors left and right of shape [B, N, N, hidden_dim]. 

The key design decision is to:
- Break down the large matrix multiplication into smaller tiles to manage memory and improve caching
- Utilize asynchronous memory operations for loading data from different blocks to reduce latency
- Implement bounds checking and proper masking to prevent out-of-bound accesses

This approach will:
- Reduce the operation time complexity from O(N^3) per element to better pipelined execution
- Minimize register usage by tightly coupling with input shape constraints
- Handle the large N dimensions (up to 1024) effectively
- Maintain proper hidden_dim dimension (128) as the tile size in computations

**2. Mathematical Specification**

Given inputs:
- `x`: tensor of shape [B, N, N, dim]
- `mask`: optional tensor of shape [B, N, N]

We perform:
1. Layer normalization along channel dimension
2. Projection into hidden_dim space: two separate linear projections
3. Optional masking with broadcasting
4. Compute gating values: three sigmoid activations
5. Matrix multiplication contraction: compute 
   `y[b,i,j,d] = left[b,i,k,d] * right[b,j,k,d] summed over k`
6. Final LayerNorm and linear projection back to dim

The output is:
- `y`: tensor of shape [B, N, N, dim]

Where d is the hidden dimension index (varying from 0 to hidden_dim-1), and k is summed from 0 to N-1 for each b,i,j,d.
