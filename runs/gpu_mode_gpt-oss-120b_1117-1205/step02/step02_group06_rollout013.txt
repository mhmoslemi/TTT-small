"""
TriMul (outgoing) forward pass – Triton‑accelerated core contraction.

Algorithm
---------
1. LayerNorm over the input channel dimension.
2. Two linear projections (left/right) into the hidden dimension.
3. Optional mask applied to the projections.
4. Three gating signals (left_gate, right_gate, out_gate) computed from the
   normalized input and passed through a sigmoid.
5. Gates are applied element‑wise to the projected tensors.
6. Core contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is a batched matrix multiplication A @ Bᵀ where
   A = left.permute(0,3,1,2) and B = right.permute(0,3,1,2).
   The multiplication is performed by a Triton kernel that
   processes all (batch * hidden_dim) matrices in parallel,
   using fp16 inputs, fp32 accumulation and fp16 output.
7. The result is reshaped back to [B, N, N, hidden_dim],
   LayerNormed, multiplied by out_gate and finally projected
   back to the original channel dimension.

The implementation keeps all non‑heavy operations in PyTorch and
accelerates only the O(N³·hidden_dim) contraction with Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # matrix dimensions
    stride_a_batch, stride_a_m, stride_a_k,  # strides for A (batch, row, col)
    stride_b_batch, stride_b_k, stride_b_n,  # strides for B (batch, row=k, col=j)
    stride_c_batch, stride_c_m, stride_c_n,  # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)   # flattened (batch * hidden_dim)
    pid_m = tl.program_id(1)      # block row
    pid_n = tl.program_id(2)      # block col

    # Block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        # Load B tile (K x N) – note that B is stored transposed (row = k)
        b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        # Dot product (uses Tensor‑Core when possible)
        acc += tl.dot(a, b)

    # Write result back in FP16
    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _triton_batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ Bᵀ.
    A, B : (batch, M, K)  with dtype torch.float16 and contiguous.
    Returns C : (batch, M, N)  (here M == N == K) with dtype torch.float16.
    """
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, "Inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides (in elements)
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_i, stride_b_j = B.stride()
    # B is stored as (i, j) = (row, col) = (N, K)
    # We need strides for Bᵀ where rows correspond to K and cols to N
    stride_b_k = stride_b_j          # row stride = original column stride (1)
    stride_b_n = stride_b_i          # col stride = original row stride (N)

    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Heuristic block size selection
    if M <= 256:
        BLOCK_M = BLOCK_N = 64
    else:
        BLOCK_M = BLOCK_N = 128
    BLOCK_K = 32  # works for all supported sizes and keeps K multiple of 8

    grid = (
        batch,                                 # batch * hidden_dim
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul (outgoing) entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator with a Triton‑fused
    batched GEMM.  Returns a tensor of shape [bs, seq_len, seq_len, dim].

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim]  (float32)
            - mask_tensor : torch.Tensor [B, N, N]      (bool/float) – may be ignored
            - weights_dict: mapping of parameter name → torch.Tensor
            - config_dict : must contain keys 'dim' and 'hidden_dim';
                            optional 'nomask' (bool).

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    x_fp16 = x.to(torch.float16)   # use FP16 for the heavy path

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [B, N, N, 1] and cast to FP16
        mask_fp = mask_tensor.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 4. Gating signals (computed in FP32 for better stability, then cast)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid().to(torch.float16)
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid().to(torch.float16)
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid().to(torch.float16)

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction – batched GEMM via Triton
    # ------------------------------------------------------------------
    # Permute to (B, H, N, N) where the last two axes are the matrix dimensions
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    # Flatten batch and hidden dimensions for the kernel
    A = left_perm.view(B * H, N, N)
    B_ = right_perm.view(B * H, N, N)

    # Batched matmul: out = A @ Bᵀ
    out_flat = _triton_batched_matmul(A, B_)               # (B*H, N, N), fp16

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # back to FP32 for the remaining ops

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out_gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"])

    return out
