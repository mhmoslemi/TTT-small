    a_ptr, b_ptr, c_ptr,
    B, H, N,
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_k, stride_b_j,
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    MASK: tl.constexpr
):
    pid = tl.program_id(0) # batch*head
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    batch_head = pid
    batch = batch_head // H
    head = batch_head % H

    # compute offsets within current batch and head
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this batch/head
    a_base = a_ptr + batch * stride_a_batch + head * stride_a_head
    b_base = b_ptr + batch * stride_b_batch + head * stride_b_head
    c_base = c_ptr + batch * stride_c_batch + head * stride_c_head

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load A tile
        a_ptrs = a_base + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

    # store
    c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j

    x = F.layer_norm(

    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))

and the kernel receives the true strides of the inputs.  LayerNorms,
linear projections and gating remain in PyTorch, while the O(N³) contraction
runs entirely in Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    B, H, N,                                 # dimensions
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_k, stride_b_j,
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid = tl.program_id(0)           # combined batch * head
    pid_m = tl.program_id(1)         # tile row (i)
    pid_n = tl.program_id(2)         # tile col (j)

    # split combined dimension back into batch and head
    batch = pid // H
    head = pid % H

    # tile offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (batch, head)
    a_base = a_ptr + batch * stride_a_batch + head * stride_a_head
    b_base = b_ptr + batch * stride_b_batch + head * stride_b_head
    c_base = c_ptr + batch * stride_c_batch + head * stride_c_head

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction axis k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A tile : (i,k)
        a_ptrs = a_base + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # B tile : (k,j)
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_j[None, :] * stride_b_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # fused multiply‑add
        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    store_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=store_mask)


def tri_mul_gemm(left: torch.Tensor, right: torch.Tensor,
                 BLOCK_M: int = 64, BLOCK_N: int = 64, BLOCK_K: int = 32) -> torch.Tensor:
    """
    Batched GEMM for the TriMul contraction.
    left/right : (B, H, N, N)   dtype=torch.float16
    returns    : (B, H, N, N)   dtype=torch.float16
    """
    B, H, N, _ = left.shape
    assert right.shape == (B, H, N, N)

    out = torch.empty_like(left)

    # strides (in elements)
    a_batch, a_head, a_i, a_k = left.stride()
    b_batch, b_head, b_k, b_j = right.stride()
    c_batch, c_head, c_i, c_j = out.stride()

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        B,
        H,
        N,
        a_batch,
        a_head,
        a_i,
        a_k,
        b_batch,
        b_head,
        b_k,
        b_j,
        c_batch,
        c_head,
        c_i,
        c_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Triton‑enabled forward of the outgoing TriMul module
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
            - mask         : torch.Tensor [B, N, N]      (bool or numeric)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16 for the heavy path
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias, fp16)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 4. Gating (fp16)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core N³ contraction via Triton
    # ------------------------------------------------------------------
    # Permute to (batch, hidden, N, N) where the right tensor is stored as (k, j)
    left_perm = left.permute(0, 3, 1, 2)   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1)  # (B, H, N, N)  (k, j)

    N = left_perm.shape[2]
    # Heuristic block sizes
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
    else:
        BLOCK_M = BLOCK_N = 128
    BLOCK_K = 32

    # Batched GEMM
    out_perm = tri_mul_gemm(left_perm, right_perm,
                            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K)   # (B, H, N, N), fp16

    # Back to original layout [B, N, N, H]
    out = out_perm.permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
