@triton.jit
def batched_matmul_kernel(a_ptr, b_ptr, c_ptr, M, N, K,
                         stride_a_batch, stride_a_m, stride_a_k,
                         stride_b_batch, stride_b_k, stride_b_n,
                         stride_c_batch, stride_c_m, stride_c_n,
                         BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # x
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch_ptr = a_ptr + batch_id * stride_a_batch
    b_batch_ptr = b_ptr + batch_id * stride_b_batch
    c_batch_ptr = c_ptr + batch_id * stride_c_batch

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute k range
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # mask for k dimension
        mask_k = offs_k < K

        # load A sub-tile (M x K)
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=tl.float16(0.0))

        # load B sub-tile (K x N)
        b_ptrs = b_batch_ptr + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=tl.float16(0.0))

        acc += tl.dot(a, b)  # produce shape (BLOCK_M, BLOCK_N)

    # store result
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def triton_matmul(a, b):
    # a: (batch, M, K) fp16 contiguous
    # b: (batch, K, N) fp16 contiguous
    # returns c (batch, M, N) fp16

    assert a.dtype == torch.float16 and b.dtype == torch.float16
    assert a.is_contiguous() and b.is_contiguous()
    B_total, M, K = a.shape
    assert b.shape[0] == B_total
    assert b.shape[1] == K
    N = b.shape[2]

    # allocate output
    c = torch.empty((B_total, M, N), dtype=torch.float16, device=a.device)

    # compute strides in elements
    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_k, stride_b_n = b.stride()
    stride_c_batch, stride_c_m, stride_c_n = c.stride()

    # Define block sizes; choose 128 for M/N if possible, else adjust near N size.
    # Could compute dynamic block sizes based on M,N.
    # For simplicity:
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Align to not exceed dimension size (in case N small)
    # We could adjust block sizes to min(dim, blocksize).
    # We'll handle via masking, so fine.

    grid = (B_total,
            triton.cdiv(M, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    batched_matmul_kernel[grid](
        a, b, c,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c

"""
Triton‑accelerated forward pass for the outgoing TriMul operation.

Algorithm
---------
1. LayerNorm over the input channels (float32).
2. Linear projections (left/right) and gating (left_gate, right_gate, out_gate)
   are performed in fp16 for speed.
3. Optional mask is applied (broadcast over the hidden dimension).
4. Core contraction:
       out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   This is a batched matrix multiplication for each (b,d) pair:
       (N×N) @ (N×N)^T
   Implemented with a custom Triton kernel (fp16 inputs, fp32
   accumulation, fp16 output).
5. Output LayerNorm, out‑gate scaling and final linear projection back to
   the original channel dimension (float32).

The kernel processes a batch of (B*hidden_dim) matrix multiplications in
parallel, handling arbitrary sequence lengths up to 1024 with block sizes
tuned for NVIDIA Hopper (B200).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------
# Triton batched GEMM kernel (C = A @ B)
#   A : (batch, M, K)   fp16
#   B : (batch, K, N)   fp16 (already transposed for the TriMul contraction)
#   C : (batch, M, N)   fp16  (accumulated in fp32)
# ------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs
    batch_id = tl.program_id(0)      # batch dimension (B * hidden_dim)
    pid_m = tl.program_id(1)         # block row
    pid_n = tl.program_id(2)         # block column

    # tile coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        mask_k = offs_k < K

        # load A tile (M x K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=tl.float16(0.0))

        # load B tile (K x N)
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=tl.float16(0.0))

        # block‑wise dot product
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N) in fp32

    # write back result (convert to fp16)
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def triton_batched_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B (fp16 inputs, fp32 accumulation).

    Args:
        a: Tensor of shape (batch, M, K), dtype torch.float16, contiguous.
        b: Tensor of shape (batch, K, N), dtype torch.float16, contiguous.

    Returns:
        Tensor c of shape (batch, M, N), dtype torch.float16.
    """
    assert a.dtype == b.dtype == torch.float16, "Kernel expects fp16 inputs"
    assert a.is_contiguous() and b.is_contiguous(), "Inputs must be contiguous"

    batch, M, K = a.shape
    _, Kb, N = b.shape
    assert K == Kb, "Inner dimensions must match"

    # allocate output
    c = torch.empty((batch, M, N), dtype=torch.float16, device=a.device)

    # strides in elements
    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_k, stride_b_n = b.stride()
    stride_c_batch, stride_c_m, stride_c_n = c.stride()

    # block sizes – tuned for up to N=1024 on Hopper
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_matmul_kernel[grid](
        a, b, c,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c


# ------------------------------------------------------------
# Entry point: custom_kernel
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric)
            - weights_dict: dict with the following keys:
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config_dict: must contain "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] float32

    # --------------------------------------------------------
    # 2. Cast to fp16 for heavy linear ops
    # --------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------
    # 3. Linear projections (fp16)
    # --------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------
    # 4. Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] -> [B, N, N, 1]
        mask_fp = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_fp
        right = right * mask_fp

    # --------------------------------------------------------
    # 5. Gating (fp16) + sigmoid
    # --------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------
    # 6. Core contraction via Triton batched GEMM
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # --------------------------------------------------------
    # reshape to (B, H, N, N) then flatten batch & heads
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, i) -> (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, k, j)

    B, H, N, _ = left_perm.shape
    batch_total = B * H

    left_flat = left_perm.view(batch_total, N, N)   # (B*H, N, K)
    right_flat = right_perm.view(batch_total, N, N)  # (B*H, K, N)

    # perform batched matmul (fp16 in/out, fp32 accumulation)
    out_flat = triton_batched_matmul(left_flat, right_flat)  # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # convert to fp32 for following layers

    # --------------------------------------------------------
    # 7. Output LayerNorm, out‑gate scaling and final projection
    # --------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
