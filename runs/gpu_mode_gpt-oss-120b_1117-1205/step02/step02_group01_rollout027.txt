                          left_ptr, right_ptr,
                          left_gate_ptr, right_gate_ptr,
                          mask_ptr,
                          total, hidden_dim,

    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    right_val = tl.load(right_ptr + offs, mask=mask)

"""
TriMul (outgoing) forward pass, fused implementation.

The heavy linear transformations are collapsed into a single matrix multiply
by concatenating the five projection weights (left_proj, left_gate, right_proj,
right_gate, out_gate).  This reduces five separate GEMM calls to one.
A custom Triton kernel fuses the element‑wise mask, left‑gate and right‑gate
multiplications for both the left and right branches, producing the
masked‑gated tensors `left` and `right`.  The cubic contraction
out[i,j] = Σ_k left[i,k]·right[j,k] is performed as a batched GEMM
(`torch.bmm`), which uses cuBLAS and is highly optimized.
Finally a layer‑norm, output gating and linear projection produce the result.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate multiplications.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    out_left_ptr,   # [total] left output
    out_right_ptr,  # [total] right output
    left_ptr,       # [total] left_proj
    right_ptr,      # [total] right_proj
    left_gate_ptr,  # [total] left_gate (sigmoid)
    right_gate_ptr, # [total] right_gate (sigmoid)
    mask_ptr,       # [mask_elements] mask broadcast over hidden dim
    total,          # total number of elements (B*N*N*hidden)
    BLOCK_SIZE: tl.constexpr,
    H: tl.constexpr,  # hidden dimension (fast axis)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # load tensors
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # broadcast mask over hidden dimension
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # fused multiply‑accumulate
    out_l = l * lg * m
    out_r = r * rg * m

    # store results
    tl.store(out_left_ptr + offs, out_l, mask=mask)
    tl.store(out_right_ptr + offs, out_r, mask=mask)

    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor: torch.Tensor [B, N, N, dim] (float32)
            mask_tensor : torch.Tensor [B, N, N] (float32) or None
            weights_dict : contains all linear / norm weights (FP32)
            config_dict  : {"dim": int, "hidden_dim": int}
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm  (float32 for stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for all subsequent heavy work
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Single fused linear for 5 heads (left/right proj & gates)
    # ------------------------------------------------------------------
    # collect and convert each weight to fp16
    w_left_proj   = weights["left_proj.weight"].to(device).to(torch.float16)
    w_left_gate   = weights["left_gate.weight"].to(device).to(torch.float16)
    w_right_proj  = weights["right_proj.weight"].to(device).to(torch.float16)
    w_right_gate  = weights["right_gate.weight"].to(device).to(torch.float16)
    w_out_gate    = weights["out_gate.weight"].to(device).to(torch.float16)

    # concatenate along the output‑feature dimension
    combined_weight = torch.cat(
        [w_left_proj, w_left_gate, w_right_proj, w_right_gate, w_out_gate],
        dim=0,
    )  # shape (5*hidden_dim, dim)

    # fused linear: (B,N,N,dim) @ (5*hidden_dim, dim).T -> (B,N,N,5*hidden_dim)
    combined = F.linear(x_fp16, combined_weight)  # fp16
    combined = combined.view(B, N, N, 5, hidden_dim)  # [B,N,N,5,hidden]

    left_proj   = combined[..., 0, :]  # [B,N,N,hidden]
    left_gate   = combined[..., 1, :]  # [B,N,N,hidden]
    right_proj  = combined[..., 2, :]  # [B,N,N,hidden]
    right_gate  = combined[..., 3, :]  # [B,N,N,hidden]
    out_gate    = combined[..., 4, :]  # [B,N,N,hidden]

    # sigmoid gates (still fp16)
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # ------------------------------------------------------------------
    # 4. Fuse mask & gate multiplication with a single Triton kernel
    # ------------------------------------------------------------------
    if mask_tensor is None:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(device).to(torch.float16)

    # flatten tensors for the kernel
    left_proj_f   = left_proj.contiguous().view(-1)
    right_proj_f  = right_proj.contiguous().view(-1)
    left_gate_f   = left_gate.contiguous().view(-1)
    right_gate_f  = right_gate.contiguous().view(-1)
    mask_f        = mask_fp16.contiguous().view(-1)

    # output buffers
    left_out_f  = torch.empty_like(left_proj_f)
    right_out_f = torch.empty_like(right_proj_f)

    # launch Triton kernel (one program processes BLOCK elements)
    BLOCK = 1024
    total = left_proj_f.numel()
    grid = (triton.cdiv(total, BLOCK),)
    _mask_gate_fuse_kernel[grid](
        left_out_f,
        right_out_f,
        left_proj_f,
        right_proj_f,
        left_gate_f,
        right_gate_f,
        mask_f,
        total,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
    )

    # reshape back to [B,N,N,hidden]
    left  = left_out_f.view(B, N, N, hidden_dim)
    right = right_out_f.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5. Cubic contraction via batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    # reshape to (B*hidden, N, N) for bmm
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)
    out_mat = torch.bmm(left_mat, right_mat)                      # fp16

    # back to [B,N,N,hidden]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Post‑processing: layer‑norm, output gate, final linear (float32)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float32)
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out_norm = out_norm * out_gate.to(torch.float32)   # final gating

    to_out_w = weights["to_out.weight"].to(device).to(torch.float32)  # (dim, hidden)
    output = F.linear(out_norm, to_out_w)  # [B,N,N,dim] float32

    return output
```
