"""
Triton‑accelerated outgoing TriMul forward pass.

Algorithm:
1. LayerNorm over the channel dimension (dim) – FP32.
2. Linear projections to hidden_dim (no bias) – FP16.
3. Optional mask multiplication (broadcast over hidden dim) – FP16.
4. Gating values (linear + sigmoid) – FP16, applied to the projections.
5. Core contraction:
   For each batch element and hidden channel compute
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication left @ rightᵀ.
   Implemented with a fused Triton kernel that processes
   (B·hidden_dim) independent GEMMs of size N×N.
6. LayerNorm on the hidden dimension, multiply by out‑gate, and final
   linear projection back to dim – FP32.
All heavy arithmetic is in FP16 with FP32 accumulation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton batched GEMM kernel (A @ Bᵀ) for the core contraction
# --------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute C = A @ Bᵀ for a batch of matrices.
    A : (batch, M, K)   FP16
    B : (batch, K, N)   FP16  (already transposed operand)
    C : (batch, M, N)   FP16
    """
    batch_id = tl.program_id(0)    # which (batch·hidden) pair
    pid_m = tl.program_id(1)       # tile row index
    pid_n = tl.program_id(2)       # tile column index

    # ----------- tile offsets ----------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch element
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- main K loop ----
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile (M×K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k < K),
            other=0.0,
        )

        # Load B tile (K×N) – note B already holds the transposed operand
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(
            b_ptrs,
            mask=(offs_k < K)[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Block‑wise dot product
        acc += tl.dot(a, b)

    # Write back result in FP16
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def triton_batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ Bᵀ.
    A : (batch, M, K)   FP16, contiguous
    B : (batch, K, N)   FP16, contiguous (already transposed)
    Returns C : (batch, M, N) FP16
    """
    assert A.dtype == B.dtype == torch.float16
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides (in elements)
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tile sizes – tuned for modern NVIDIA GPUs (e.g., Hopper B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# --------------------------------------------------------------
# Entry point expected by the evaluator
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operation.

    Arguments:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (bool / uint8 / float) – may be ignored
            - weights: dict of model parameters
            - config: dict containing 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. LayerNorm over channel dimension (FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # shape [B, N, N, dim], float32

    # --------------------------------------------------------------
    # 2. Cast to FP16 for heavy arithmetic
    # --------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Linear projections to hidden_dim (no bias) – FP16
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x_fp16, right_proj_w)  # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 4. Optional mask
    # --------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → [B, N, N, 1] for broadcasting
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 5. Gating values (sigmoid) – FP16
    # --------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()  # shape [B, N, N, hidden_dim]

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6. Core contraction: batched GEMM left @ rightᵀ (Triton)
    # --------------------------------------------------------------
    # Rearrange to (B, hidden, N, N) layout and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, k, j) = transposed for GEMM

    B, H, N, _ = left_perm.shape
    # Merge batch and hidden dimensions → (B*H, N, N)
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Compute batched matrix multiplication (left @ rightᵀ)
    out_flat = triton_batched_matmul(left_flat, right_flat)   # (B*H, N, N), FP16

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # FP32 for subsequent LayerNorm & projection

    # --------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate scaling, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"])  # (dim, hidden_dim) weight

    return out
