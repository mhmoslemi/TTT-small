Now implement the kernel:

```
@triton.jit
def mask_gate_fuse_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    N: tl.int32,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < N
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + offs, mask=mask, other=0.0)
    out_left = left * left_gate * mask_val
    out_right = right * right_gate * mask_val
    tl.store(out_left_ptr + offs, out_left, mask=mask)
    tl.store(out_right_ptr + offs, out_right, mask=mask)
```

    device = input_tensor.device

            mask_flat.data_ptr(),

"""
Optimized outgoing TriMul forward pass.

Algorithm:
1. LayerNorm on the input (dim).
2. Linear projections (left/right) and gating projections are computed in fp16.
3. Optional mask is fused together with left/right gating via a small Triton kernel.
4. Core contraction is performed as a batched matrix multiplication
   (left @ right^T) using torch.bmm (cuBLAS) which is highly optimized.
5. Output LayerNorm, gating and final linear projection back to the original dim.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def mask_gate_fuse_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    N: tl.int32,                 # total number of elements
    BLOCK_SIZE: tl.constexpr,    # compile‑time block size
):
    """Fuse mask, left‑gate and right‑gate multiplications."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < N

    left = tl.load(left_ptr + offs,   mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + offs, mask=mask, other=0.0)

    out_left = left * left_gate * mask_val
    out_right = right * right_gate * mask_val

    tl.store(out_left_ptr + offs, out_left, mask=mask)
    tl.store(out_right_ptr + offs, out_right, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/float)
            - weights_dict : mapping from parameter name to torch.Tensor
            - config_dict  : must contain 'dim' and 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    x_fp16 = x.to(torch.float16)  # subsequent linear ops in fp16

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias, fp16)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # (B, N, N, hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Gate projections (fp16) and optional mask fusion via Triton
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()   # (B, N, N, hidden_dim)

    if not nomask:
        # expand mask to hidden_dim, cast to fp16
        mask_fp = mask.to(torch.float16).unsqueeze(-1)                 # (B, N, N, 1)
        mask_fp = mask_fp.expand(-1, -1, -1, hidden_dim).contiguous()  # (B, N, N, hidden_dim)

        # flatten tensors for the kernel
        left_flat = left.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_fp.contiguous().view(-1)

        out_left = torch.empty_like(left_flat)
        out_right = torch.empty_like(right_flat)

        total = left_flat.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)
        mask_gate_fuse_kernel[grid](
            left_flat.data_ptr(),
            left_gate_flat.data_ptr(),
            right_flat.data_ptr(),
            right_gate_flat.data_ptr(),
            mask_flat.data_ptr(),
            out_left.data_ptr(),
            out_right.data_ptr(),
            N=total,
            BLOCK_SIZE=BLOCK,
        )
        left = out_left.view_as(left)
        right = out_right.view_as(right)
    else:
        # simple element‑wise gating (no mask)
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Core contraction: batched GEMM (left @ right^T)
    # ------------------------------------------------------------------
    #   left : (B, N, N, H) -> (B, H, N, N)
    #   right: (B, N, N, H) -> (B, H, N, N) with axes (k, j) -> (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    A = left_perm.view(B * H, N, N)        # (B*H, N, N)
    B_mat = right_perm.view(B * H, N, N)   # (B*H, N, N)

    # cuBLAS batched matmul, fp16 inputs, fp16 output
    out_flat = torch.bmm(A, B_mat)        # (B*H, N, N), fp16
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 5. Output layer‑norm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)

    # final linear: hidden_dim -> dim
    out = F.linear(out, weights["to_out.weight"])

    return out
