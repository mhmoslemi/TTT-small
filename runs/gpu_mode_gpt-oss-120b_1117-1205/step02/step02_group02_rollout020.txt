def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']; hidden_dim = config['hidden_dim']; nomask = config.get('nomask', True)

    # LayerNorm on input
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights['norm.weight'],
                     bias=weights['norm.bias'])
    # Convert to half for heavy ops
    x_h = x.to(torch.float16)

    # Linear projections (no bias)
    left = F.linear(x_h, weights['left_proj.weight'].to(torch.float16))
    right = F.linear(x_h, weights['right_proj.weight'].to(torch.float16))

    # Gating linear projections (sigmoid)
    left_gate = F.linear(x_h, weights['left_gate.weight'].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights['right_gate.weight'].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights['out_gate.weight'].to(torch.float16)).sigmoid()

    # Apply mask + gates to left/right using Triton kernel
    # Ensure tensors contiguous
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    # mask condition: if nomask => treat mask as all ones.
    if nomask:
        mask_fp = torch.ones((input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]), dtype=torch.float16, device=input_tensor.device)
    else:
        mask_fp = mask.to(torch.float16)

    # Expand mask to have last dim 1 for broadcasting shape (B,N,N,1)
    mask_fp = mask_fp.unsqueeze(-1).contiguous()

    # Flatten to 1D pointer
    total = left.numel()
    BLOCK_SIZE = 32768  # choose large for memory heavy
    grid = (triton.cdiv(total, BLOCK_SIZE),)
    gate_apply_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        left_gate.data_ptr(),
        right_gate.data_ptr(),
        mask_fp.data_ptr(),
        total,
        hidden_dim,
        mask_fp.numel(),
        has_mask=int(not nomask),
        BLOCK_SIZE=BLOCK_SIZE
    )

    # Compute core contraction using torch.bmm
    # Permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # now shape (B, H, N, N) with k as dim0
    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)
    # Use torch.bmm for batched matmul
    out_flat = torch.bmm(left_flat, right_flat)  # half precision
    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # shape (B,N,N,H)

    # Cast to float32 for layer norm and final linear
    out = out.to(torch.float32)

    # Output layer norm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights['to_out_norm.weight'],
        bias=weights['to_out_norm.bias']
    )
    # Apply out_gate (convert out_gate to float32)
    out_gate_f = out_gate.to(torch.float32)
    out = out * out_gate_f

    # Final projection back to dim
    out = F.linear(out, weights['to_out.weight'].to(torch.float32))

    return out

"""
TriMul (outgoing) forward pass.

Algorithm
---------
1. Layer‑Norm on the input (FP32).
2. Cast to FP16 and apply the four linear projections (no bias).
3. Compute three gate tensors with a sigmoid (FP16).
4. Fuse mask (if present) and gate multiplication into the left/right tensors
   with a small Triton element‑wise kernel (in‑place).
5. Perform the N³ contraction as a batched matrix multiplication:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented with torch.bmm on tensors of shape (B·H, N, N) which is
   highly optimized in cuBLAS.
6. Layer‑Norm on the hidden dimension, apply the out‑gate, and a final
   linear projection back to the original channel dimension.

A Triton kernel is used (step 4); the heavy contraction is delegated to cuBLAS
(batched GEMM) for maximal throughput on the B200 GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def gate_apply_kernel(
    left_ptr,               # fp16*[total_elements]
    right_ptr,              # fp16*[total_elements]
    left_gate_ptr,          # fp16*[total_elements]
    right_gate_ptr,         # fp16*[total_elements]
    mask_ptr,               # fp16*[mask_elements] (optional)
    TOTAL_ELEMS: tl.constexpr,   # B * N * N * H
    HIDDEN_DIM: tl.constexpr,    # H
    MASK_ELEMS: tl.constexpr,    # B * N * N (0 if no mask)
    HAS_MASK: tl.constexpr,      # 0/1 compile‑time flag
    BLOCK_SIZE: tl.constexpr,    # threads per program
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < TOTAL_ELEMS

    # Load the four tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # Broadcast mask over the hidden dimension
        mask_idx = offs // HIDDEN_DIM          # index into mask (size B*N*N)
        mask_val = tl.load(mask_ptr + mask_idx,
                           mask=mask_idx < MASK_ELEMS,
                           other=1.0)
        left_out = left * left_gate * mask_val
        right_out = right * right_gate * mask_val
    else:
        left_out = left * left_gate
        right_out = right * right_gate

    tl.store(left_ptr + offs, left_out, mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor of shape [B, N, N] (bool/float)
            - weights_dict: mapping from parameter name to torch.Tensor
            - config_dict: must contain 'dim' and 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )                                 # [B, N, N, dim], float32

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for the heavy ops
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 4. Gate projections + sigmoid (still FP16)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 5. Fuse mask (if any) and gate multiplication into left/right tensors
    # ------------------------------------------------------------------
    # Ensure contiguous layout for the kernel
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    if nomask:
        has_mask = 0
        mask_fp = torch.empty(0, dtype=torch.float16, device=device)   # dummy
        mask_elements = 0
    else:
        has_mask = 1
        # mask: [B, N, N] -> [B, N, N, 1] for broadcasting
        mask_fp = mask_tensor.to(torch.float16).unsqueeze(-1).contiguous()
        mask_elements = mask_fp.numel() // hidden_dim   # B*N*N
        # sanity: mask_fp shape should be (B, N, N, 1)

    total_elements = left.numel()            # B * N * N * hidden_dim
    BLOCK_SIZE = 1024                         # fits within hardware limits
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

    gate_apply_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_fp,
        total_elements,
        hidden_dim,
        mask_elements,
        has_mask,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # ------------------------------------------------------------------
    # 6. Core contraction: batched GEMM (A @ Bᵀ) using cuBLAS via torch.bmm
    #    left:  [B, N, N, H] -> (B, H, N, N) where last two dims are (i,k)
    #    right: [B, N, N, H] -> (B, H, N, N) where last two dims are (k,j)
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, i/k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, k/j)

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N)  # (B*H, N, N)

    # Batched matrix multiplication (FP16 inputs, FP32 accumulation inside cuBLAS)
    out_flat = torch.bmm(left_flat, right_flat)   # (B*H, N, N), still FP16

    # Restore the original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = out.to(torch.float32)  # switch to FP32 for the remaining ops

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    out = out * out_gate.to(torch.float32)   # element‑wise gating after LN

    out = F.linear(out, weights["to_out.weight"].to(torch.float32))  # final proj

    return out
