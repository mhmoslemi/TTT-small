import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    total_elems,
    B, N, H,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # compute mask index
    # b = offs // (N * N * H)
    # i = (offs % (N * N * H)) // (N * H)
    # j = (offs % (N * H)) // H
    tmp = N * N * H
    b = offs // tmp
    rem = offs - b * tmp
    tmp2 = N * H
    i = rem // tmp2
    rem2 = rem - i * tmp2
    j = rem2 // H

    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    m = tl.load(mask_ptr + mask_off, mask=mask)

    l_out = l * lg * m
    r_out = r * rg * m

    tl.store(left_out_ptr + offs, l_out, mask=mask)
    tl.store(right_out_ptr + offs, r_out, mask=mask)
```

    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
Outgoing TriMul (AlphaFold3) forward pass.

The heavy N³ contraction 
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed as a large batched GEMM (B*hidden_dim independent 
matrix multiplications).  The GEMM is executed with torch.bmm (cuBLAS 
tensor‑core accelerated) while the element‑wise mask and gate fusion is
offloaded to a tiny Triton kernel.  The rest of the module (LayerNorm,
linear projections, final gating and projection) stays in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    total_elems,
    B, N, H,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK: tl.constexpr,
):
    """Fuse mask, left‑gate and right‑gate into left/right tensors.

    Arguments
    ----------
    left_ptr, right_ptr            : fp16 source tensors, shape (B,N,N,H) flattened.
    left_gate_ptr, right_gate_ptr  : fp16 gate tensors, same shape.
    mask_ptr                       : fp16 mask tensor, shape (B,N,N) (no H dim).
    left_out_ptr, right_out_ptr    : fp16 output tensors, same shape as inputs.
    total_elems                    : total number of elements in each tensor.
    B, N, H                        : compile‑time constants for dimensions.
    stride_mask_*                  : strides of the mask tensor (B,N,N) in elements.
    BLOCK                          : number of elements processed per program.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elems

    # load the four source tensors (contiguous layout)
    l = tl.load(left_ptr + offs, mask=valid)
    r = tl.load(right_ptr + offs, mask=valid)
    lg = tl.load(left_gate_ptr + offs, mask=valid)
    rg = tl.load(right_gate_ptr + offs, mask=valid)

    # --- compute mask index (b,i,j) from flattened offset ---
    # flattening order: [b, i, j, h]  (h = hidden dim)
    # stride = N * N * H for batch, N * H for row, H for col
    tmp1 = N * N * H
    b = offs // tmp1
    rem1 = offs - b * tmp1
    tmp2 = N * H
    i = rem1 // tmp2
    rem2 = rem1 - i * tmp2
    j = rem2 // H

    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    m = tl.load(mask_ptr + mask_off, mask=valid)

    # apply mask and gates
    l_out = l * lg * m
    r_out = r * rg * m

    # store results
    tl.store(left_out_ptr + offs, l_out, mask=valid)
    tl.store(right_out_ptr + offs, r_out, mask=valid)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool or numeric)
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections (fp16, bias=False)
    # --------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 3. Gates (fp16)
    # --------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # --------------------------------------------------------------
    # 4. Mask handling (fp16)
    # --------------------------------------------------------------
    if nomask:
        mask_fp = torch.ones(
            (input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]),
            dtype=torch.float16,
            device=device,
        )
    else:
        mask_fp = mask.to(torch.float16)

    # --------------------------------------------------------------
    # 5. Fuse mask + left/right gates via Triton (elementwise)
    # --------------------------------------------------------------
    # Ensure contiguous layout for flat indexing
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_fp = mask_fp.contiguous()

    left_gated = torch.empty_like(left)
    right_gated = torch.empty_like(right)

    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    B, N, N2, H = left.shape
    assert N == N2, "expected square pairwise dimensions"
    stride_mask_b, stride_mask_i, stride_mask_j = mask_fp.stride()  # (B,N,N)

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_fp,
        left_gated,
        right_gated,
        total,
        B,
        N,
        H,
        stride_mask_b,
        stride_mask_i,
        stride_mask_j,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 6. Core N³ contraction as batched GEMM (torch.bmm, fp16)
    #    left_gated: (B,N,N,H) -> (B,H,N,N)
    #    right_gated: (B,N,N,H) -> (B,H,N,N) then transpose (k,j)
    # --------------------------------------------------------------
    left_perm = left_gated.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_gated.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) where dim2 is k, dim3 is j

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N)  # (B*H, N, N)

    # Batched matmul: (B*H, N, N) @ (B*H, N, N)ᵀ  -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat)  # fp16 result

    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)
    out = out.to(torch.float32)

    # --------------------------------------------------------------
    # 7. Output LayerNorm, gating and final projection (float32)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"])

    return out
