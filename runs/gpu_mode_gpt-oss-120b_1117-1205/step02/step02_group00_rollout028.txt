@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b_left, stride_i_left, stride_k_left, stride_h_left,
    stride_b_right, stride_i_right, stride_k_right, stride_h_right,
    stride_b_out, stride_i_out, stride_j_out, stride_h_out,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # block row index
    pid_n = tl.program_id(1)  # block column index
    pid_z = tl.program_id(2)  # batch * head index

    # compute batch and head
    b = pid_z // H
    h = pid_z % H

    # Compute start offsets for this batch_head
    # pointers base offset
    left_base = left_ptr + b * stride_b_left + h * stride_h_left
    right_base = right_ptr + b * stride_b_right + h * stride_h_right
    out_base = out_ptr + b * stride_b_out + h * stride_h_out

    # Offsets for rows and columns within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, N, BLOCK_K):
        # k block offsets
        offs_k = k + tl.arange(0, BLOCK_K)
        # mask for k dimension
        mask_k = offs_k < N

        # Compute pointers for A (left) tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + \
            (offs_m[:, None] * stride_i_left) + \
            (offs_k[None, :] * stride_k_left)
        # Compute pointers for B (right) tile (transposed): shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_base + \
            (offs_k[:, None] * stride_k_right) + \
            (offs_n[None, :] * stride_i_right)  # using stride_i as column stride

        # Load tiles with appropriate masks
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Write back to output
    c_ptrs = out_base + \
        (offs_m[:, None] * stride_i_out) + \
        (offs_n[None, :] * stride_j_out)
    # Apply mask for output
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x_h = x.to(torch.float16)
    # Projections
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))
    # Gates
    left_gate = torch.sigmoid(F.linear(x_h, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_h, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].to(torch.float16)))
    # Apply mask and gates
    left = left * left_gate
    right = right * right_gate
    if not nomask:
        mask_h = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_h
        right = right * mask_h
    # contraction via Triton kernel
    out = torch.empty_like(left)  # same shape B,N,N,H, half
    # compute strides
    B, N, _, H = left.shape
    # stride order for left
    stride_b_left, stride_i_left, stride_k_left, stride_h_left = left.stride()
    stride_b_right, stride_i_right, stride_k_right, stride_h_right = right.stride()
    stride_b_out, stride_i_out, stride_j_out, stride_h_out = out.stride()
    # set block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * H)
    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_b_left, stride_i_left, stride_k_left, stride_h_left,
        stride_b_right, stride_i_right, stride_k_right, stride_h_right,
        stride_b_out, stride_i_out, stride_j_out, stride_h_out,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Post processing
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"].to(torch.float16), bias=weights["to_out_norm.bias"].to(torch.float16))
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))
    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass optimized with Triton.
- LayerNorm + linear projections & gating are handled in PyTorch (float16).
- The O(N³) contraction   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
  is implemented as a fused batched GEMM (A @ Bᵀ) in a single Triton kernel,
  avoiding the costly permute‑and‑contiguous copies required by the naive
  torch.bmm implementation.
- The kernel works for arbitrary batch size (≤2) and hidden dimension
  (≤384). It respects an optional mask and returns a float32 tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_gemm_kernel(
    left_ptr,               # half*  [B,N,N,H]
    right_ptr,              # half*  [B,N,N,H]
    out_ptr,                # half*  [B,N,N,H]
    B, N, H,                # int32
    # left strides
    stride_b_left, stride_i_left, stride_k_left, stride_h_left,
    # right strides
    stride_b_right, stride_i_right, stride_k_right, stride_h_right,
    # out strides
    stride_b_out, stride_i_out, stride_j_out, stride_h_out,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched gemm: out = left @ rightᵀ  for each (batch, head)."""
    pid_i = tl.program_id(0)        # block row (i dimension)
    pid_j = tl.program_id(1)        # block col (j dimension)
    pid_z = tl.program_id(2)        # fused batch*head index

    # ----- decode batch and head indices ----- #
    b = pid_z // H                  # batch index
    h = pid_z - b * H               # head index

    # base pointers for this (b,h) slice
    left_base  = left_ptr  + b * stride_b_left  + h * stride_h_left
    right_base = right_ptr + b * stride_b_right + h * stride_h_right
    out_base   = out_ptr   + b * stride_b_out   + h * stride_h_out

    # ----- tile offsets ----- #
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- K‑loop ----- #
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)  from left[i,k]
        a_ptrs = left_base + \
            (offs_i[:, None] * stride_i_left) + (offs_k[None, :] * stride_k_left)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # rightᵀ tile: shape (BLOCK_K, BLOCK_N)  from right[j,k] (transposed)
        b_ptrs = right_base + \
            (offs_k[:, None] * stride_k_right) + (offs_j[None, :] * stride_i_right)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # accumulate
        acc += tl.dot(a, b)          # fp16×fp16 → fp32

    # ----- write result ----- #
    c_ptrs = out_base + \
        (offs_i[:, None] * stride_i_out) + (offs_j[None, :] * stride_j_out)
    mask = mask_i[:, None] & mask_j[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) – may be None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> half)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # 2. Linear projections (half precision)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(torch.float16)))

    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate
    if not nomask and mask is not None:
        mask_h = mask.unsqueeze(-1).to(torch.float16)  # [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM (left @ rightᵀ) in Triton.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # allocate result tensor (half)

    # Strides (int64)
    stride_b_left, stride_i_left, stride_k_left, stride_h_left = left.stride()
    stride_b_right, stride_i_right, stride_k_right, stride_h_right = right.stride()
    stride_b_out, stride_i_out, stride_j_out, stride_h_out = out.stride()

    # Block sizes – tuned for RTX/AMPere (adjustable if needed)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),        # i‑tiles
        triton.cdiv(N, BLOCK_N),        # j‑tiles
        B * H,                          # batch*head dimension
    )

    _tri_mul_gemm_kernel[grid](
        left, right, out,
        B, N, H,
        stride_b_left, stride_i_left, stride_k_left, stride_h_left,
        stride_b_right, stride_i_right, stride_k_right, stride_h_right,
        stride_b_out, stride_i_out, stride_j_out, stride_h_out,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return in float32 as required
    return out.to(torch.float32)
