@triton.jit
def fuse_gate_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    mask_present: tl.constexpr,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_out_left_b, stride_out_left_i, stride_out_left_k, stride_out_left_h,
    stride_out_right_b, stride_out_right_i, stride_out_right_k, stride_out_right_h,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    # compute linear offset for element
    # Use block of 1D arrangement across total elements: total = B*N*N*H
    # Compute global index based on block and thread.
    i = pid * BLOCK + tl.arange(0, BLOCK)  # flattened index
    # Determine mask for valid indices
    total = B * N * N * H
    mask = i < total
    # Compute multi-dimensional indices from flattened index.
    # Need to map index to (b,i_idx,k_idx,h_idx). Use integer division.
    # Implementation: compute idx = i // H => (b,i,k) part, and h = i % H
    # Then for each (b,i,k) part, map to dimensions: Let's flatten as ((b * N + i_idx) * N + k_idx) * H + h_idx.
    # So we can convert: tmp = i
    tmp = i
    h = tmp % H
    tmp = tmp // H
    k = tmp % N
    tmp = tmp // N
    i_idx = tmp % N
    b = tmp // N

    # Compute offsets
    left_offset = b * stride_left_b + i_idx * stride_left_i + k * stride_left_k + h * stride_left_h
    right_offset = b * stride_right_b + i_idx * stride_right_i + k * stride_right_k + h * stride_right_h
    left_gate_offset = b * stride_lgate_b + i_idx * stride_lgate_i + k * stride_lgate_k + h * stride_lgate_h
    right_gate_offset = b * stride_rgate_b + i_idx * stride_rgate_i + k * stride_rgate_k + h * stride_rgate_h
    mask_offset = b * stride_mask_b + i_idx * stride_mask_i + k * stride_mask_k
    out_left_offset = b * stride_out_left_b + i_idx * stride_out_left_i + k * stride_out_left_k + h * stride_out_left_h
    out_right_offset = b * stride_out_right_b + i_idx * stride_out_right_i + k * stride_out_right_k + h * stride_out_right_h

    # Load
    left_val = tl.load(left_ptr + left_offset, mask=mask, other=0.0)

"""
Optimized outgoing TriMul (AlphaFold3) forward pass.

1. LayerNorm on the input (float32).
2. FP16 linear projections for left/right and their gates.
3. A tiny Triton kernel fuses:
      * optional mask broadcasting,
      * element‑wise mask·gate multiplication for left & right tensors.
4. Batched matrix multiplication (left @ rightᵀ) is performed with
   torch.matmul which uses cuBLAS tensor‑core kernels.
5. Output LayerNorm, out‑gate scaling and final FP32 linear projection.
The Triton kernel only touches the hot element‑wise region, keeping launch
overhead minimal while the heavy N³ contraction is handled by highly
optimized cuBLAS.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    total_elements,                     # scalar
    H: tl.constexpr,                    # hidden dimension (for mask broadcasting)
    HAS_MASK: tl.constexpr,             # 1 if mask is present, 0 otherwise
    BLOCK: tl.constexpr,                # thread block size
):
    """Fuse mask (if present) and gate multiplication for left/right tensors."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    # load raw tensors (contiguous layout B,N,N,H -> flattened)
    left = tl.load(left_ptr + offs, mask=active, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=active, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=tl.float16(0.0))

    if HAS_MASK:
        # mask is stored without the hidden dimension → broadcast
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=tl.float16(0.0))
        left = left * mask_val
        right = right * mask_val

    out_left = left * left_gate
    out_right = right * right_gate

    tl.store(out_left_ptr + offs, out_left, mask=active)
    tl.store(out_right_ptr + offs, out_right, mask=active)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/float)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for cheap heavy path
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 4. Gate values (FP16)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 5. Fuse mask and gate multiplication via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # left.shape = (B, N, N, hidden_dim)

    total = B * N * N * hidden_dim
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    # flatten mask to 1‑D (no hidden dim) – Triton will broadcast internally
    if not nomask:
        mask_fp16 = mask.to(torch.float16).reshape(-1)  # shape B*N*N
    else:
        # dummy tensor – never read when HAS_MASK==0
        mask_fp16 = left  # any fp16 tensor of sufficient size

    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)

    gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_fp16,
        left_out,
        right_out,
        total,
        H=hidden_dim,
        HAS_MASK=int(not nomask),
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 6. Core contraction: batched GEMM (left @ rightᵀ)
    #    left_out: (B, N, N, H) → (B, H, N, N)
    #    right_out: (B, N, N, H) → (B, H, N, N) with axes (k, j)
    # ------------------------------------------------------------------
    left_perm = left_out.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_out.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    # torch.matmul performs batched matrix multiplication using cuBLAS/Tensor Cores
    out_perm = torch.matmul(left_perm, right_perm)          # (B, H, N, N)

    out = out_perm.permute(0, 2, 3, 1).contiguous()        # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate scaling and final projection (float32)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    out_scaled = out_norm * out_gate.to(torch.float32)

    result = F.linear(out_scaled, weights["to_out.weight"])

    return result
