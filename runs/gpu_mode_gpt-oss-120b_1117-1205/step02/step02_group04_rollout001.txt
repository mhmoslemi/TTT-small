"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The kernel accelerates the O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
by treating each (batch, hidden) pair as an independent matrix multiplication
(left @ rightᵀ) performed with a tiled Triton GEMM in FP16 with FP32
accumulation. All surrounding operations (LayerNorm, gating, final projection)
are kept in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton batched GEMM kernel (C = A @ B) for many batches.
# A: (B, M, K), B: (B, K, N)  ->  C: (B, M, N)
# --------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ah,
    stride_bk, stride_bn, stride_bh,
    stride_cm, stride_cn, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for this batch
    a_batch = a_ptr + batch_id * stride_ah
    b_batch = b_ptr + batch_id * stride_bh
    c_batch = c_ptr + batch_id * stride_ch

    # Accumulator (FP32 for precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K chunks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k < K),
            other=0.0,
        )

        # Load B tile (K x N)
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(
            b_ptrs,
            mask=(offs_k < K)[:, None] & mask_n[None, :],
            other=0.0,
        )

        acc += tl.dot(a, b)   # dot yields FP16, accumulation is FP32

    # Write back result (convert to FP16)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def triton_batched_gemm(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B using Triton.
    A: (batch, M, K)  FP16, contiguous
    B: (batch, K, N)  FP16, contiguous
    Returns C: (batch, M, N) FP16
    """
    assert A.is_contiguous() and B.is_contiguous()
    assert A.dtype == B.dtype == torch.float16

    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides in element counts
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bk, stride_bn = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    # Tile sizes – chosen for good occupancy on B200
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_gemm_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_am,
        stride_ak,
        stride_ah,
        stride_bk,
        stride_bn,
        stride_bh,
        stride_cm,
        stride_cn,
        stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

# Entry point: custom_kernel

    nomask = config.get("nomask", True)

    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to half‑precision for the heavy parts
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)    # (B, N, N, hidden_dim)
    right = F.linear(x_fp16, right_proj_w)  # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3) Optional masking
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)  # (B, N, N, 1)
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core O(N³) contraction via Triton batched GEMM
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    This is equivalent to left @ rightᵀ for each (b,d) pair.
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden_dim, N, N) where the last two axes are the
    # matrix dimensions to be multiplied.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) = rightᵀ

    B, H, N, _ = left_perm.shape
    total_batches = B * H

    # Collapse batch and hidden dimensions → (total_batches, N, N)
    left_flat = left_perm.view(total_batches, N, N)
    right_flat = right_perm.view(total_batches, N, N)

    # Perform batched GEMM in FP16
    out_flat = triton_batched_gemm(left_flat, right_flat)  # (total_batches, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # subsequent layers expect FP32

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(out.dtype)  # apply output gate
    out = F.linear(out, weights["to_out.weight"])  # (B, N, N, dim)

    return out
```
