@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, hidden_dim,
    stride_a_batch, stride_a_hidden,
    stride_a_i, stride_a_k,
    stride_b_batch, stride_b_hidden,
    stride_b_k, stride_b_i,
    stride_c_batch, stride_c_hidden,
    stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_hidden = tl.program_id(0) # combined index across B*hidden_dim
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute batch and hidden indices
    hidden = batch_hidden % hidden_dim
    batch = batch_hidden // hidden_dim

    # Offsets for M and N dim
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers
    a_base = a_ptr + batch * stride_a_batch + hidden * stride_a_hidden
    b_base = b_ptr + batch * stride_b_batch + hidden * stride_b_hidden
    c_base = c_ptr + batch * stride_c_batch + hidden * stride_c_hidden

    # Initialize accumulator FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile (i,k)
        a_ptrs = a_base + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B tile (k,j) i.e., (k, j)
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_j[None, :] * stride_b_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Write to C
    c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

        bias=weights["to_out_norm.bias"]

"""
Triton‑accelerated forward pass for the outgoing TriMul operator.

Algorithm
---------
1. Apply LayerNorm on the input (float32).
2. Cast to fp16 and project with two linear layers (left/right) to hidden_dim.
3. Apply optional mask (broadcast over hidden_dim).
4. Compute three sigmoid gates from the normalized input and apply them.
5. Core contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched GEMM (left @ rightᵀ) with a custom Triton kernel.
   The kernel works directly on the original (B, N, N, H) layout, avoiding any
   costly permute/contiguous copies.
6. LayerNorm on the hidden dimension, multiply by the output gate.
7. Final linear projection back to the original dimension.

All heavy arithmetic is carried out in fp16 with fp32 accumulation. The final
output is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_a_batch, stride_a_hidden,
    stride_a_i, stride_a_k,
    stride_b_batch, stride_b_hidden,
    stride_b_k, stride_b_i,
    stride_c_batch, stride_c_hidden,
    stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Combined (batch, hidden) index
    batch_hidden = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Recover batch and hidden indices
    hidden = batch_hidden % H
    batch = batch_hidden // H

    # Tile offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this (batch, hidden) pair
    a_base = a_ptr + batch * stride_a_batch + hidden * stride_a_hidden
    b_base = b_ptr + batch * stride_b_batch + hidden * stride_b_hidden
    c_base = c_ptr + batch * stride_c_batch + hidden * stride_c_hidden

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the shared K dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile: rows i, cols k
        a_ptrs = a_base + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile: rows k, cols j  (right matrix is already transposed)
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_j[None, :] * stride_b_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    # Write back the result in fp16
    c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])
