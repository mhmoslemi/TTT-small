```python
"""
Triton-accelerated outgoing TriMul (Triangle Multiplicative Update).

The heavy O(N³ * hidden_dim) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is computed as a batched matrix multiplication of shape
    (batch*hidden, N, N) @ (batch*hidden, N, N) → (batch*hidden, N, N)
using a custom Triton GEMM kernel with FP16 inputs and FP32 accumulation.
All other operations (LayerNorm, gating, final projection) are performed
in PyTorch (mixed precision) for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------------
# Triton kernel: batched GEMM (C = A @ B) for many independent batches.
# A: (B, M, K), B: (B, K, N) -> C: (B, M, N)
# ------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ah,
    stride_bk, stride_bn, stride_bh,
    stride_cm, stride_cn, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + batch_id * stride_ah
    b_batch = b_ptr + batch_id * stride_bh
    c_batch = c_ptr + batch_id * stride_ch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k < K),
            other=0.0,
        )

        # Load B tile (K x N)
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(
            b_ptrs,
            mask=(offs_k < K)[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Compute partial product
        acc += tl.dot(a, b)

    # Store result in FP16
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )

def triton_batched_gemm(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.
    A: (batch, M, K)  fp16, contiguous
    B: (batch, K, N)  fp16, contiguous
    Returns C: (batch, M, N) fp16.
    """
    assert A.is_contiguous() and B.is_contiguous()
    assert A.dtype == B.dtype == torch.float16
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # strides in elements
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bk, stride_bn = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    # Tunable tile sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_gemm_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_am,
        stride_ak,
        stride_ah,
        stride_bk,
        stride_bn,
        stride_bh,
        stride_cm,
        stride_cn,
        stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

# ------------------------------------------------------------------
# Entry point
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.
    Args:
        data: (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/int) – optional
            - weights: dict of model parameters (float32)
            - config: dict with keys 'dim', 'hidden_dim', optional 'nomask'
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] float32

    # Cast to fp16 for the heavy compute
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (fp16, no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x_fp16, right_proj_w)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)  # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 4. Gating (fp16)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton batched GEMM
    #    Compute out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    # Rearrange tensors so that each hidden dimension becomes an independent batch.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)  where H=hidden_dim
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)  (K,J)

    B, H, N, _ = left_perm.shape
    total_batch = B * H

    # Collapse batch and hidden dimensions.
    left_flat = left_perm.view(total_batch, N, N)
    right_flat = right_perm.view(total_batch, N, N)

    # Batched GEMM:   (total_batch, N, N) @ (total_batch, N, N)  -> (total_batch, N, N)
    out_flat = triton_batched_gemm(left_flat, right_flat)

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # subsequent ops use FP32

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------

"""
Triton‑accelerated outgoing TriMul (Triangle Multiplicative Update).

The O(N³·hidden_dim) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is reformulated as a batched GEMM:
    (batch*hidden, N, N) @ (batch*hidden, N, N) → (batch*hidden, N, N)
and computed with a custom Triton kernel (FP16 inputs, FP32 accumulation).
All surrounding LayerNorms, gates and final projection stay in PyTorch
(mixed‑precision) for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ B
# A: (B, M, K)  B: (B, K, N)  -> C: (B, M, N)
# --------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ah,
    stride_bk, stride_bn, stride_bh,
    stride_cm, stride_cn, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + batch_id * stride_ah
    b_batch = b_ptr + batch_id * stride_bh
    c_batch = c_ptr + batch_id * stride_ch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)

        # Load A tile (M×K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k < K),
            other=0.0,
        )

        # Load B tile (K×N)
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(
            b_ptrs,
            mask=(offs_k < K)[:, None] & mask_n[None, :],
            other=0.0,
        )

        acc += tl.dot(a, b)   # FP32 accumulation

    # Store C tile (FP16)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )

def triton_batched_gemm(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication using the kernel above.
    A: (batch, M, K) fp16, contiguous
    B: (batch, K, N) fp16, contiguous
    Returns C: (batch, M, N) fp16.
    """
    assert A.is_contiguous() and B.is_contiguous()
    assert A.dtype == B.dtype == torch.float16
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # strides (in elements)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bk, stride_bn = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    # tile sizes – chosen for good occupancy on B200
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_gemm_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_am,
        stride_ak,
        stride_ah,
        stride_bk,
        stride_bn,
        stride_bh,
        stride_cm,
        stride_cn,
        stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

# --------------------------------------------------------------
# Entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/int), may be ignored when nomask=True
            - weights: dict mapping parameter names to torch.Tensor (float32)
            - config: dict with keys 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] float32

    # Cast to fp16 for the expensive path
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------
    # 2. Linear projections (fp16, bias=False)
    # --------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)    # [B, N, N, hidden_dim]
    right = F.linear(x_fp16, right_proj_w)  # [B, N, N, hidden_dim]

    # --------------------------------------------------
    # 3. Mask (optional)
    # --------------------------------------------------
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)  # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # --------------------------------------------------
    # 4. Gating (fp16)
    # --------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------
    # 5. Core contraction via Triton batched GEMM
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # --------------------------------------------------
    # Rearrange so each hidden dimension becomes an independent batch.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)  (K,J)

    B, H, N, _ = left_perm.shape
    total_batch = B * H

    left_flat = left_perm.view(total_batch, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(total_batch, N, N) # (B*H, N, N)

    # Batched GEMM: (total_batch, N, N) @ (total_batch, N, N) → (total_batch, N, N)
    out_flat = triton_batched_gemm(left_flat, right_flat)

    # Restore layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # subsequent ops use FP32

    # --------------------------------------------------
    # 6. Output LayerNorm, gate, final projection
    # --------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(out.dtype)
    out = F.linear(out, weights["to_out.weight"])   # (B, N, N, dim)

    return out
```
