"""
TriMul (outgoing) forward pass with a fused Triton batched GEMM.

Algorithm:
1️⃣ Layer‑norm the input (float32).
2️⃣ Cast to fp16 and apply two linear projections (left / right) → (B,N,N,H).
3️⃣ (Optional) mask the projected tensors.
4️⃣ Compute left/right gates with a sigmoid, apply them to the projected tensors.
5️⃣ Core contraction:
   out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   This is a batched matrix multiplication for each head h.
   Implemented with a custom Triton kernel (fp16 inputs, fp32 accumulation).
6️⃣ Cast result back to float32, reshape to (B,N,N,H).
7️⃣ Output layer‑norm, multiply by an out‑gate (computed in float32),
   and final linear projection back to the original channel dimension.
The final tensor has shape (B, N, N, dim) and dtype float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------
# Triton batched GEMM kernel (C = A @ B, where B is already transposed)
# ------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """C = A @ B (batched). All tensors are fp16, accumulation in fp32."""
    batch_id = tl.program_id(0)          # batch dimension (B*H)
    pid_m = tl.program_id(1)             # block row of C
    pid_n = tl.program_id(2)             # block col of C

    # Compute offsets for the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    # Accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k < K),
            other=0.0,
        )

        # Load B tile (K x N) – B is already transposed (k, j)
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(
            b_ptrs,
            mask=(offs_k < K)[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Compute outer product for the tile
        acc += tl.dot(a, b)

    # Write back (cast to fp16)
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def triton_batched_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM with Triton.
    a: (batch, M, K)  fp16, contiguous
    b: (batch, K, N)  fp16, contiguous
    Returns c: (batch, M, N) fp16
    """
    assert a.dtype == b.dtype == torch.float16
    assert a.is_contiguous() and b.is_contiguous()
    B_total, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    c = torch.empty((B_total, M, N), dtype=torch.float16, device=a.device)

    # Strides in element units
    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_k, stride_b_n = b.stride()
    stride_c_batch, stride_c_m, stride_c_n = c.stride()

    # Tunable block sizes (work well for N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B_total,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        a,
        b,
        c,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return c


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/int/float) or None
            - weights_dict: mapping of parameter names to torch.Tensor
            - config_dict: must contain 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = int(config["dim"])
    hidden_dim = int(config["hidden_dim"])
    nomask = bool(config.get("nomask", True))

    # -------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim], float32

    # -------------------------------------------------
    # 2. Cast to fp16 for heavy linear work
    # -------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -------------------------------------------------
    # 3. Linear projections (fp16, bias=False)
    # -------------------------------------------------
    left = F.linear(
        x_fp16, weights["left_proj.weight"].to(torch.float16), bias=None
    )
    right = F.linear(
        x_fp16, weights["right_proj.weight"].to(torch.float16), bias=None
    )

    # -------------------------------------------------
    # 4. Optional mask (broadcast over hidden dim)
    # -------------------------------------------------
    if not nomask and mask is not None:
        # Ensure mask has shape [B, N, N, 1] and dtype fp16
        mask_fp = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp
        right = right * mask_fp

    # -------------------------------------------------
    # 5. Gating (fp16) and apply
    # -------------------------------------------------
    left_gate = F.linear(
        x_fp16, weights["left_gate.weight"].to(torch.float16), bias=None
    ).sigmoid()
    right_gate = F.linear(
        x_fp16, weights["right_gate.weight"].to(torch.float16), bias=None
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 6. Core contraction via Triton batched GEMM
    #    Compute out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    # -------------------------------------------------
    # Rearrange tensors to (B, H, N, N) where the last two dims are (i,k) and (k,j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N) -> (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) -> (k,j)

    B, H, N, _ = left_perm.shape
    # Flatten batch and head dimensions for the kernel
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Batched matrix multiplication (fp16 inputs, fp32 accumulation)
    out_flat = triton_batched_matmul(left_flat, right_flat)  # (B*H, N, N), fp16

    # Reshape back to (B, N, N, H) and cast to float32
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # -------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final linear projection (float32)
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # Compute out‑gate in float32 (more accurate) and apply
    out_gate = F.linear(
        x, weights["out_gate.weight"], bias=None
    ).sigmoid()  # shape [B, N, N, H], float32
    out = out * out_gate

    # Final linear projection back to original dim
    out = F.linear(out, weights["to_out.weight"], bias=None)

    return out
