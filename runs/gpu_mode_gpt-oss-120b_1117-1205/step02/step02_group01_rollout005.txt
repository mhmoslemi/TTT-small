import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def mul_gate_mask_kernel(out_ptr, a_ptr, gate_ptr, mask_ptr, total,
                        BLOCK_SIZE: tl.constexpr,
                        H: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    a = tl.load(a_ptr + offs, mask=mask)
    g = tl.load(gate_ptr + offs, mask=mask)
    # Compute mask index by dividing by hidden dim H
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    tl.store(out_ptr + offs, a * g * m, mask=mask)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def mul_gate_mask_kernel(
    out_ptr,          # ptr to output (flattened)
    a_ptr,            # ptr to input tensor (flattened)
    gate_ptr,         # ptr to gate tensor (flattened)
    mask_ptr,         # ptr to mask tensor (flattened, broadcast over H)
    total,            # total number of elements
    BLOCK_SIZE: tl.constexpr,
    H: tl.constexpr,  # hidden dimension (size of innermost axis)
):
    """Fused elementwise multiplication:
       out = a * gate * mask   (mask broadcast over hidden dim)"""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    a = tl.load(a_ptr + offs, mask=mask)
    g = tl.load(gate_ptr + offs, mask=mask)

    # Mask has shape (B, N, N) → flatten index = offs // H
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    tl.store(out_ptr + offs, a * g * m, mask=mask)
