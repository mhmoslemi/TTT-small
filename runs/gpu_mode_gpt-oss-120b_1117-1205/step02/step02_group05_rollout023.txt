"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm:
1. LayerNorm on the input (float32).
2. Linear projections to hidden_dim (float16) for the left and right tensors.
3. Optional mask multiplication (float16).
4. Gating (sigmoid) applied to the projected tensors (float16).
5. Core contraction:
   For each batch element b and hidden channel h,
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (A @ Bᵀ) with a custom Triton kernel.
6. LayerNorm on the hidden dimension (float32), multiply by out_gate,
   and project back to the original dimension.

The heavy N³ work in step 5 is executed entirely on the GPU using
Triton, avoiding the costly Python loops or torch.einsum.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication left @ rightᵀ
# ----------------------------------------------------------------------
@triton.jit
def batched_tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    total_batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_k, stride_b_n, stride_b_h,
    stride_c_batch, stride_c_m, stride_c_n, stride_c_h,
    hidden_dim: tl.constexpr,               # compile‑time constant for modulo/div
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)                 # flattened (batch, hidden) index
    b_idx = pid // hidden_dim               # original batch index
    h_idx = pid % hidden_dim                # hidden channel index

    # Base pointers for this (batch, hidden) pair
    a_base = a_ptr + b_idx * stride_a_batch + h_idx * stride_a_h
    b_base = b_ptr + b_idx * stride_b_batch + h_idx * stride_b_h
    c_base = c_ptr + b_idx * stride_c_batch + h_idx * stride_c_h

    # Tile offsets
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Main K‑loop
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0,
                    dtype=tl.float16)

        # Load B tile (transposed operand): shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Block‑wise dot product, accumulation in fp32
        acc += tl.dot(a, b)

    # Write result (fp16) respecting masks
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def triton_tri_mul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    left, right : (B, N, N, H)  fp16
    Returns out  : (B, N, N, H)  fp16
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # Strides for left (A)
    stride_a_batch, stride_a_m, stride_a_k, stride_a_h = left.stride()
    # Strides for right (B) – we treat it as (K,N) with K = original third dim
    stride_b_batch, stride_b_m, stride_b_k, stride_b_h = right.stride()
    # For Bᵀ we need stride for K (k) and N (j)
    stride_b_k = stride_b_k          # original third dim (k)
    stride_b_n = stride_b_m          # original second dim (j)

    # Strides for output C (same layout as left/right)
    stride_c_batch, stride_c_m, stride_c_k, stride_c_h = out.stride()

    total_batch = B * H
    M = N
    K = N
    N_dim = N

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        total_batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N_dim, BLOCK_N),
    )
    batched_tri_mul_kernel[grid](
        left,
        right,
        out,
        total_batch,
        M,
        N_dim,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_a_h,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_b_h,
        stride_c_batch,
        stride_c_m,
        stride_c_k,
        stride_c_h,
        H,                              # hidden_dim as constexpr
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operation.

    Arguments:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool or numeric)
            - weights      : dict of model parameters
            - config       : dict with keys 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. LayerNorm over channel dimension (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # Cast to fp16 for heavy arithmetic
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections (fp16, bias=False)
    # --------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 4. Gating values (sigmoid, fp16) and element‑wise application
    # --------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction: batched GEMM in Triton
    # --------------------------------------------------------------
    # left, right : [B, N, N, hidden_dim] (fp16)
    out_hidden = triton_tri_mul(left, right)               # fp16, shape (B,N,N,hidden_dim)

    # Cast back to fp32 for the final LayerNorm & projection
    out = out_hidden.to(torch.float32)

    # --------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
