"""
TriMul (outgoing) forward pass with a Triton‑accelerated batched GEMM.

Algorithm:
1. LayerNorm over the channel dimension (float32).
2. Linear projections to a hidden dimension (FP16, no bias).
3. Optional mask multiplication (broadcast over hidden dim).
4. Gate values (FP16 sigmoid) and element‑wise gating.
5. Core contraction:
   For each batch and hidden channel we compute
       out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   which is a batched matrix multiplication left @ right.T.
   Implemented by a Triton kernel that processes (B·H) independent
   GEMMs of size N×N with FP32 accumulation and FP16 results.
6. LayerNorm over the hidden dimension, out‑gate multiplication,
   and final linear projection back to the original dimension.
All heavy arithmetic is in FP16; final output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ B) where B is already transposed
#   A: (batch, M, K)   FP16
#   B: (batch, K, N)   FP16   (the K‑N matrix is the transposed operand)
#   C: (batch, M, N)   FP16   (FP32 accumulation)
# ------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)          # which (batch, hidden) pair
    pid_m    = tl.program_id(1)          # tile row
    pid_n    = tl.program_id(2)          # tile column

    # offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for the current batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile (K x N) – B already holds the transposed operand
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Compute block‑wise dot product
        acc += tl.dot(a, b)

    # Write back result (cast to FP16)
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def triton_batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM C = A @ B   (B is already transposed)
    A : (batch, M, K)   FP16
    B : (batch, K, N)   FP16
    Returns C : (batch, M, N) FP16
    """
    assert A.is_contiguous() and B.is_contiguous()
    assert A.dtype == B.dtype == torch.float16
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # strides
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # tile sizes (tuned for Hopper B200, FP16)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

# ------------------------------------------------------------------
# Entry point
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operation.

    Arguments:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) – may be ignored
            - weights: dict of model parameters (torch.Tensors)
            - config: dict containing 'dim', 'hidden_dim', and optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. LayerNorm over channel dim (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # fp16 for heavy ops
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 3. Optional masking (broadcast over hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [B, N, N, 1]
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 4. Gating values (sigmoid) and application
    # --------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction: batched GEMM (left @ right.T) via Triton
    # --------------------------------------------------------------
    # reshape to (B, H, N, N) where H = hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, k, j) -> transposed operand

    B, H, N, _ = left_perm.shape
    # flatten batch*hidden into a single batch dimension for the kernel
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Perform batched matmul: (B*H, N, N) @ (B*H, N, N) -> (B*H, N, N)
    out_flat = triton_batched_matmul(left_flat, right_flat)   # FP16

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # back to FP32 for the final layers

    # --------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate multiplication, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
