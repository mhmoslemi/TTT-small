def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Access weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # 1. LayerNorm (float32)
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # Convert to half
    x_h = x.to(torch.float16)

    # Concatenate projection and gate weights for left
    left_concat_weight = torch.cat([left_proj_weight, left_gate_weight], dim=0).to(torch.float16)  # (2*hidden_dim, dim)
    right_concat_weight = torch.cat([right_proj_weight, right_gate_weight], dim=0).to(torch.float16)  # (2*hidden_dim, dim)

    # Linear projection + gate pre-sigmoid
    left_concat = F.linear(x_h, left_concat_weight)  # shape (B,N,N,2*hidden_dim)
    right_concat = F.linear(x_h, right_concat_weight)

    # Split
    left = left_concat[..., :hidden_dim]
    left_gate_pre = left_concat[..., hidden_dim:]
    right = right_concat[..., :hidden_dim]
    right_gate_pre = right_concat[..., hidden_dim:]

    # Apply sigmoid to gates
    left_gate = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()

    # Compute out_gate (used later)
    out_gate = F.linear(x_h, out_gate_weight.to(torch.float16)).sigmoid()

    # Apply mask and gates to left/right
    if not nomask and mask is not None:
        # mask shape [B, N, N]
        mask_h = mask.to(torch.float16).contiguous()
        # Flatten
        total = left.numel()
        # Ensure contiguous
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_h.view(-1)  # shape B*N*N

        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)
        HAS_MASK = 1
        _gate_mask_fuse_kernel[grid](
            left_flat, right_flat,
            left_gate_flat, right_gate_flat,
            mask_flat,
            total,
            BLOCK=BLOCK,
            HAS_MASK=HAS_MASK,
            H=hidden_dim,
        )
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
        # gates no longer needed; but we already applied them
    else:
        left = left * left_gate
        right = right * right_gate

    # Free gate tensors (optional)
    del left_gate, right_gate, left_gate_pre, right_gate_pre

    # 5. Core contraction via batched GEMM
    # Permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape

    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)  # half

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # Output LayerNorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight.to(torch.float16),
        bias=to_out_norm_bias.to(torch.float16)
    )
    # Multiply by out_gate
    out = out * out_gate

    # Final linear projection to dim
    out = F.linear(out, to_out_weight.to(torch.float16))

    # Return float32
    return out.to(torch.float32)

def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,

        right = right * rg * m

        left = left * lg
        right = right * rg

    tl.store(right_ptr + offs, right, mask=mask)

"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

The heavy N³ contraction is performed with a batched GEMM (torch.bmm) on
half‑precision data.  A small Triton kernel fuses the optional mask with the
left/right gating multiplications, avoiding extra memory passes.  The remaining
linear layers and LayerNorms are executed in PyTorch (half where possible)
and the final result is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: fuse mask (optional) + left/right gates
# --------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # [total] float16
    right_ptr,              # [total] float16
    left_gate_ptr,          # [total] float16
    right_gate_ptr,         # [total] float16
    mask_ptr,               # [mask_total] float16 (broadcasted)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,                # threads per program
    HAS_MASK: tl.constexpr,             # 1 if mask present, else 0
    H: tl.constexpr,                    # hidden_dim (size of innermost dim)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    lgate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    if HAS_MASK:
        # One mask element corresponds to H consecutive hidden entries
        mask_idx = offs // H
        m = tl.load(mask_ptr + mask_idx, mask=valid, other=0.0)
        left = left * lgate * m
        right = right * rgate * m
    else:
        left = left * lgate
        right = right * rgate

    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (bool or float) or None
            - weights: Mapping of parameter names to torch.Tensor
            - config: Dictionary with keys ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2. Convert to half for the heavy ops
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections + gate pre‑sigmoid (fused via concatenated weight)
    # ------------------------------------------------------------------
    # concatenate proj and gate weights to amortise the input read
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*hidden_dim, dim)
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*hidden_dim, dim)

    left_concat = F.linear(x_h, left_concat_w)   # [B, N, N, 2*hidden_dim]
    right_concat = F.linear(x_h, right_concat_w)  # [B, N, N, 2*hidden_dim]

    left = left_concat[..., :hidden_dim]                    # proj
    left_gate_pre = left_concat[..., hidden_dim:]           # gate pre‑sigmoid
    right = right_concat[..., :hidden_dim]
    right_gate_pre = right_concat[..., hidden_dim:]

    left_gate = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()

    # ------------------------------------------------------------------
    # 4. Out‑gate (used after the contraction)
    # ------------------------------------------------------------------
    out_gate = F.linear(
        x_h, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 5. Apply mask (if any) and gates in‑place via Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).contiguous()   # [B, N, N]
        total = left.numel()  # B * N * N * hidden_dim

        # flatten everything for the kernel
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        lgate_flat = left_gate.contiguous().view(-1)
        rgate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_h.view(-1)                     # [B*N*N]

        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)
        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            lgate_flat,
            rgate_flat,
            mask_flat,
            total,
            BLOCK=BLOCK,
            HAS_MASK=1,
            H=hidden_dim,
        )
        # reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
        # gates have been fused; they can be discarded
        del left_gate, right_gate
    else:
        # no mask → simple elementwise multiplication
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6. Core contraction: batched matrix‑multiply over the middle axis
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) where the last two dims are matrices
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) = right transposed
    B, H, N, _ = left_perm.shape

    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # half‑precision batched GEMM
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N), dtype=float16

    # restore original layout (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    # Cast to float32 for the remaining lightweight ops (more accurate)
    out_f32 = out.float()
    out_gate_f32 = out_gate.float()

    out_f32 = F.layer_norm(
        out_f32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_f32 = out_f32 * out_gate_f32
    out_f32 = F.linear(out_f32, weights["to_out.weight"].to(torch.float16))

    # Return float32 as required
    return out_f32.to(torch.float32)
