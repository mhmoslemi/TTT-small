"""
Highly‑optimized implementation of the outgoing TriMul operator (AlphaFold3).
The heavy linear projections and the three gating / mask multiplications are
fused into a single large linear layer (by concatenating weights) and the
elementwise gate·mask fusion is performed by a tiny Triton kernel.  The cubic
contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is carried out as a batched GEMM (torch.bmm) on FP16 tensors, which fully
leverages cuBLAS.  All remaining layer‑norms and the final projection are
done in float32 for numerical stability.

The function `custom_kernel` follows the required signature:
    (input_tensor, mask_tensor, weights_dict, config_dict) -> output_tensor
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fused elementwise multiplication of tensor, gate and mask
# ----------------------------------------------------------------------
@triton.jit
def mul_gate_mask_kernel(
    out_ptr,      # pointer to output (flattened)
    a_ptr,        # pointer to tensor (flattened)
    gate_ptr,     # pointer to gate (flattened)
    mask_ptr,     # pointer to mask (flattened, broadcast over hidden dim)
    total,        # total number of elements in a/gate tensors (B * N * N * H)
    BLOCK_SIZE: tl.constexpr,
    H: tl.constexpr,            # hidden dimension
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total

    a = tl.load(a_ptr + offs, mask=valid)
    g = tl.load(gate_ptr + offs, mask=valid)

    # each hidden element shares the same mask entry
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=valid)

    tl.store(out_ptr + offs, a * g * m, mask=valid)

    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple containing
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)

            - weights     : dict with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight",
                "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config      : dict with at least "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """

    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Fuse all linear projections / gates into one matmul (FP16)
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)
    x_flat = x_fp16.view(-1, dim)                 # (B*N*N, dim)

    # concatenate weight matrices: proj / gate for left & right, plus out_gate
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    combined_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0
    )  # shape (5*hidden_dim, dim)

    # single large linear layer
    combined_out = F.linear(x_flat, combined_weight)  # (B*N*N, 5*hidden_dim)
    combined_out = combined_out.view(B, N, N, 5 * hidden_dim)

    # split into the individual tensors
    left_proj, left_gate, right_proj, right_gate, out_gate = torch.split(
        combined_out, hidden_dim, dim=-1
    )  # each: [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 3. Sigmoid for gates
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)  # used after the contraction

    # --------------------------------------------------------------
    # 4. Fuse mask and gate via Triton (left and right branches)
    # --------------------------------------------------------------
    if mask_tensor is None or config.get("nomask", False):
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(torch.float16).to(device)

    # flatten tensors for the kernel
    left_proj_flat = left_proj.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_proj_flat = right_proj.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)
    mask_flat = mask_fp16.contiguous().view(-1)

    total = left_proj_flat.numel()            # B * N * N * hidden_dim
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    left_out_flat = torch.empty_like(left_proj_flat)
    right_out_flat = torch.empty_like(right_proj_flat)

    # left = left_proj * left_gate * mask
    mul_gate_mask_kernel[grid](
        left_out_flat,
        left_proj_flat,
        left_gate_flat,
        mask_flat,
        total,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
    )
    # right = right_proj * right_gate * mask
    mul_gate_mask_kernel[grid](
        right_out_flat,
        right_proj_flat,
        right_gate_flat,
        mask_flat,
        total,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
    )

    left = left_out_flat.view(B, N, N, hidden_dim)
    right = right_out_flat.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 5. Cubic contraction as batched GEMM (FP16)
    # --------------------------------------------------------------
    # shape (B, hidden_dim, N, N) for each operand
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) -> transpose k & j

    LH = B * hidden_dim
    left_mat = left_perm.view(LH, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(LH, N, N) # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N), FP16
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 6. Post‑processing: LayerNorm, out‑gate, final projection
    # --------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float32)
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # apply final gating
    out_norm = out_norm * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float32)  # (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)  # [B, N, N, dim], float32

    return output
```
