"""
TriMul (outgoing) forward pass with a fused Triton batched GEMM.

Algorithm
---------
1. Layer‑norm the input tensor (float32).
2. Cast to fp16 and apply three linear projections:
   * left  = x @ W_left
   * right = x @ W_right
   * left_gate, right_gate, out_gate = sigmoid(x @ W_gate)
3. (Optional) element‑wise mask is applied to left/right.
4. Apply the gates: left *= left_gate, right *= right_gate.
5. Core contraction:
   out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   which is a batched matrix multiplication
   (B·H, N, N) @ (B·H, N, N)^T.
   Implemented with a Triton kernel that tiles the M/N dimensions
   (BLOCK_M × BLOCK_N) and the reduction dimension K (BLOCK_K).
6. Cast the result to float32, apply a second LayerNorm, multiply by
   out_gate, and a final linear projection back to the original channel
   dimension.
All heavy compute (step 5) runs in fp16 with fp32 accumulation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton batched GEMM kernel (C = A @ B)  (A: (M,K), B: (K,N))
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_fp16_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_stages: tl.constexpr, num_warps: tl.constexpr
):
    batch_id = tl.program_id(0)           # batch dimension (B*H)
    pid_m = tl.program_id(1)              # block row
    pid_n = tl.program_id(2)              # block column

    # Offsets for rows and columns of the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds indices
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the current batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles of size BLOCK_K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        # Load B tile (K x N) – B is already transposed
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    # Write back the result tile
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def triton_batched_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B (both fp16).  A and B have shape
    (batch, M, K) and (batch, K, N) respectively.  Returns C of shape
    (batch, M, N) in fp16.
    """
    assert a.dtype == b.dtype == torch.float16
    assert a.is_contiguous() and b.is_contiguous()
    B_total, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    c = torch.empty((B_total, M, N), dtype=torch.float16, device=a.device)

    # Strides in elements
    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_k, stride_b_n = b.stride()
    stride_c_batch, stride_c_m, stride_c_n = c.stride()

    # Tunable tile sizes (work well for N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    NUM_WARPS = 8
    NUM_STAGES = 2

    grid = (
        B_total,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_fp16_kernel[grid](
        a, b, c,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=NUM_WARPS,
        num_stages=NUM_STAGES,
    )
    return c

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric). Ignored when
                    config['nomask'] is True.
            - weights_dict: mapping from parameter name to torch.Tensor.
            - config_dict: must contain 'dim', 'hidden_dim', optional 'nomask'.

    Returns:
        torch.FloatTensor of shape [B, N, N, dim].
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the heavy linear ops
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (fp16)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # [B, N, N, H]
    right = F.linear(x_fp16, right_proj_w)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4. Optional element‑wise mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] -> [B, N, N, 1] -> fp16
        mask_fp = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 5. Gating (fp16) and apply
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Core contraction via Triton batched GEMM
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Treat each (b,d) as an independent batch.
    # ------------------------------------------------------------------
    # Rearrange tensors to (B, H, N, N) where the last two dims are
    # the matrix dimensions required for GEMM.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, K) with K==N
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, N)

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)      # (B*H, M, K)
    right_flat = right_perm.view(B * H, N, N)    # (B*H, K, N)

    # Batched multiplication in fp16 -> fp16
    out_flat = triton_batched_matmul(left_flat, right_flat)  # (B*H, N, N)

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final projection (float32)
    # ------------------------------------------------------------------
    out_f32 = out.to(torch.float32)

    out_f32 = F.layer_norm(
        out_f32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    # Apply out‑gate (cast to float32)
    out_f32 = out_f32 * out_gate.to(torch.float32)

    # Final linear projection back to original dimension
    out_f32 = F.linear(out_f32, weights["to_out.weight"])

    return out_f32
