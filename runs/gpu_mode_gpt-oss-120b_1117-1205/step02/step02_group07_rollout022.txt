def triton_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    assert a.dtype == b.dtype == torch.float16
    B, H, M, K = a.shape
    _, _, K2, N = b.shape
    assert K == K2
    # allocate output
    c = torch.empty((B, H, M, N), dtype=torch.float16, device=a.device)   # shape (B,H,N,N)
    # compute strides (elements) for each dimension
    stride_a_batch, stride_a_head, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_head, stride_b_k, stride_b_n = b.stride()
    stride_c_batch, stride_c_head, stride_c_m, stride_c_n = c.stride()
    # Choose block sizes

    BLOCK_N = 128
    BLOCK_K = 32
    # grid dimensions:
    # number of combined batch (B*H) training
    total_batches = B * H
    grid = (
        total_batches,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        a, b, c,
        M, N, K,

        stride_b_batch, stride_b_head, stride_b_k, stride_b_n,

        HEADS=H,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c

"""
Triton‑accelerated forward pass for the outgoing TriMul operator.

The heavy part of the computation is the contraction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
which is a batched matrix multiplication for each head d.
We perform:
  1. LayerNorm on the input (float32).
  2. Linear projections + gating (float16).
  3. Optional mask.
  4. Core contraction via a custom Triton kernel that computes
     a batched GEMM for every (batch, head) pair.
  5. Output LayerNorm, output gate, final linear projection (float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_head, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_head, stride_c_m, stride_c_n,
    HEADS: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM C = A @ B  where B is already transposed (K×N).

    A has shape (batch, head, M, K)   – row‑major (M is rows, K is cols)
    B has shape (batch, head, K, N)   – row‑major (K is rows, N is cols)
    C has shape (batch, head, M, N)
    """
    batch_id = tl.program_id(0)          # combined batch*head index
    pid_m = tl.program_id(1)             # block row
    pid_n = tl.program_id(2)             # block col

    # Decode batch and head indices
    b = batch_id // HEADS
    h = batch_id - b * HEADS

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this (batch, head)
    a_base = a_ptr + b * stride_a_batch + h * stride_a_head
    b_base = b_ptr + b * stride_b_batch + h * stride_b_head
    c_base = c_ptr + b * stride_c_batch + h * stride_c_head

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Load A tile (M × K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        # Load B tile (K × N) – already transposed
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    # Store the result
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def triton_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B (float16 inputs, fp32 accumulation)
    left : (B, H, N, N)  – interpreted as (batch, head, i, k)
    right: (B, H, N, N)  – interpreted as (batch, head, k, j)
    returns C of shape (B, H, N, N) (float16)
    """
    assert left.dtype == right.dtype == torch.float16
    B, H, N, K = left.shape
    _, _, K2, N2 = right.shape
    assert K == K2 and N == N2

    # Output tensor (contiguous)
    out = torch.empty((B, H, N, N), dtype=torch.float16, device=left.device)

    # Strides (in elements)
    stride_a_batch, stride_a_head, stride_a_m, stride_a_k = left.stride()
    stride_b_batch, stride_b_head, stride_b_k, stride_b_n = right.stride()
    stride_c_batch, stride_c_head, stride_c_m, stride_c_n = out.stride()

    # Tunable block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    total_batches = B * H
    grid = (
        total_batches,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        left, right, out,
        N, N, N,                           # M, N, K
        stride_a_batch, stride_a_head, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_head, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_head, stride_c_m, stride_c_n,
        HEADS=H,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool or numeric)
            - weights      : dict of parameter tensors
            - config       : dict with keys 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                           # [B, N, N, dim]  float32

    # Cast to fp16 for the heavy linear work
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections (fp16)
    # -----------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # -----------------------------------------------------------------
    # 3. Optional mask
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)   # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # -----------------------------------------------------------------
    # 4. Gating (fp16) and apply
    # -----------------------------------------------------------------
    left_gate = F.linear(
        x_fp16, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = F.linear(
        x_fp16, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = F.linear(
        x_fp16, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction via Triton batched GEMM
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # -----------------------------------------------------------------
    # Shape permutations to obtain (B, H, N, N) views
    left_perm = left.permute(0, 3, 1, 2)   # (B, H, N, N)  (i,k)
    right_perm = right.permute(0, 3, 2, 1) # (B, H, N, N)  (k,j)

    # Batched matmul (fp16 in, fp16 out)
    out_perm = triton_matmul(left_perm, right_perm)   # (B, H, N, N)

    # Back to original layout and cast to float32 for the remaining ops
    out = out_perm.permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)
    out = out.to(torch.float32)

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection (float32)
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
