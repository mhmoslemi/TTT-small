def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Step 1: Normalize input
    x = F.layer_norm(input_tensor, (dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])
    # Cast to half
    x = x.to(torch.float16)

    # step 2: Projections
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))
    # Step 3: mask
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_fp
        right = right * mask_fp
    # Step 4: gating
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # Step 5: core contraction using Triton
    # reshape for batched matmul
    left_per = left.permute(0, 3, 1, 2).contiguous()
    right_per = right.permute(0, 3, 2, 1).contiguous()
    B, H, N, _ = left_per.shape
    left_flat = left_per.view(B * H, N, N)
    right_flat = right_per.view(B * H, N, N)

    # Perform batched gemm: left_flat @ right_flat (right already transposed)
    out_flat = triton_matmul(left_flat, right_flat)  # returns half

    # Reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # Step 6: Output layer norm, gating, final linear
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    out = out * out_gate  # out_gate is half? Should be float32; convert
    out = out.to(torch.float32) # ensure dtype
    # final linear
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass using fused gating/mask kernel and cuBLAS batched GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def gate_mask_kernel(
    x_ptr, gate_ptr, mask_ptr, out_ptr,
    total_elements, hidden_dim: tl.constexpr, BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # broadcast mask over hidden dimension
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = x * g * m
    tl.store(out_ptr + offs, out, mask=mask)

def custom_kernel(data):
    """
    Outgoing TriMul forward pass.

    Arguments:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric)
            - weights: dict of tensors
            - config: dict with keys 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------- LayerNorm ----------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ---------- Linear projections ----------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)

    # ---------- Gates ----------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()   # [B, N, N, hidden_dim]

    # ---------- Apply mask + gates (fused Triton kernel when mask present) ----------
    if not nomask:
        # mask is broadcast over hidden dimension
        mask_fp = mask.to(torch.float16).contiguous()
        total = left.numel()
        BLOCK = 65536   # ~64K threads per launch
        grid = (triton.cdiv(total, BLOCK),)

        left_gated = torch.empty_like(left)
        right_gated = torch.empty_like(right)

        # fuse left * left_gate * mask
        gate_mask_kernel[grid](
            left.view(-1), left_gate.view(-1), mask_fp.view(-1), left_gated.view(-1),
            total, hidden_dim, BLOCK=BLOCK,
        )
        # fuse right * right_gate * mask
        gate_mask_kernel[grid](
            right.view(-1), right_gate.view(-1), mask_fp.view(-1), right_gated.view(-1),
            total, hidden_dim, BLOCK=BLOCK,
        )
        left = left_gated
        right = right_gated
    else:
        left = left * left_gate
        right = right * right_gate

    # ---------- Core contraction: batched GEMM ----------
    # Rearrange for batched matrix multiplication per hidden channel
    # left: (B, N, N, H) -> (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()      # (B, H, N, N)
    # right needs to be transposed on the inner dimension: (B, H, N, N) where dims are (B, H, K, J)
    right_perm = right.permute(0, 3, 2, 1).contiguous()    # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)   # already transposed (K, J)

    # Use cuBLAS batched GEMM (torch.bmm) – much faster than a generic Triton implementation
    out_flat = torch.bmm(left_flat, right_flat)   # (B*H, N, N), dtype=half

    # Restore original layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ---------- Output LayerNorm + final gating + linear ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out

"""
Outgoing TriMul (AlphaFold3) forward pass.

- Applies LayerNorm, linear projections and sigmoid gates.
- When a mask is provided we fuse: left = (x @ W_left) * gate_left * mask
  and similarly for the right tensor using a small Triton kernel.
- The heavy N³ contraction is performed as a batched matrix multiplication
  using torch.bmm (cuBLAS), which is much faster than a generic Triton GEMM.
- Finally applies a second LayerNorm, an output gate and a linear projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: fuse element‑wise mask and gate multiplication
# --------------------------------------------------------------
@triton.jit
def gate_mask_kernel(
    x_ptr,                # input tensor (e.g. left_proj)
    gate_ptr,             # gate tensor (sigmoid output)
    mask_ptr,             # mask tensor (broadcast over hidden dim)
    out_ptr,              # output tensor
    total_elements: tl.int64,       # total number of elements in the tensor
    hidden_dim: tl.constexpr,       # hidden dimension (channel size)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elements

    # Load values
    x = tl.load(x_ptr + offs, mask=m, other=0.0)
    g = tl.load(gate_ptr + offs, mask=m, other=0.0)

    # Broadcast mask over the hidden dimension
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=m, other=1.0)

    # Compute gated + masked value
    out = x * g * mask_val
    tl.store(out_ptr + offs, out, mask=m)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric)
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)       # cast to fp16 for the heavy path

    # --------------------------------------------------------------
    # 2. Linear projections (fp16, no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 3. Gates (fp16)
    # --------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # --------------------------------------------------------------
    # 4. Optional mask – fuse with gates via Triton
    # --------------------------------------------------------------
    if not nomask:
        # Prepare mask as half‑precision contiguous tensor
        mask_fp = mask.to(torch.float16).contiguous()
        total = left.numel()                     # B * N * N * hidden_dim
        BLOCK = 65536                             # ~64 K threads per launch
        grid = (triton.cdiv(total, BLOCK),)

        # left = left * left_gate * mask
        left_gated = torch.empty_like(left)
        gate_mask_kernel[grid](
            left.view(-1),
            left_gate.view(-1),
            mask_fp.view(-1),
            left_gated.view(-1),
            total,
            hidden_dim,
            BLOCK=BLOCK,
        )

        # right = right * right_gate * mask
        right_gated = torch.empty_like(right)
        gate_mask_kernel[grid](
            right.view(-1),
            right_gate.view(-1),
            mask_fp.view(-1),
            right_gated.view(-1),
            total,
            hidden_dim,
            BLOCK=BLOCK,
        )

        left = left_gated
        right = right_gated
    else:
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction: batched GEMM using cuBLAS (torch.bmm)
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # --------------------------------------------------------------
    # Rearrange tensors to (batch * hidden, N, N) where the second operand
    # is already transposed on the inner dimension.
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) == (K, J)

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)   # (B*H, M, K)
    right_flat = right_perm.view(B * H, N, N) # (B*H, K, N)

    # Batched matrix multiplication (fp16 -> fp16, accumulation in fp32 internally)
    out_flat = torch.bmm(left_flat, right_flat)   # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection (float32)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
