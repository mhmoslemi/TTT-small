def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # cast to fp16
    x_fp16 = x.to(torch.float16)

    # Linear projections
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # mask
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_fp
        right = right * mask_fp

    # gating
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Core contraction: A @ B for each hidden_dim
    # Permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # transpose last two dims

    B, H, N, _ = left_perm.shape
    # flatten head and batch into one dimension for kernel
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # call kernel
    out_flat = batched_matmul_triton(left_flat, right_flat)  # on GPU

    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # Output LayerNorm
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    out = out * out_gate.to(torch.float32)

    out = F.linear(out, weights["to_out.weight"])

    return out
```

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        a = tl.load(a_base + (offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k),
                    mask=mask_m[:, None] & (offs_k < K), other=0.0, dtype=tl.float16)
        b = tl.load(b_base + (offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n),
                    mask=(offs_k < K)[:, None] & mask_n[None, :], other=0.0, dtype=tl.float16)

        acc += tl.dot(a, b)

    c = c_base + (offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n)
    tl.store(c, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

    assert a.dtype == torch.float16 and b.dtype == torch.float16
    B, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    c = torch.empty((B, M, N), dtype=torch.float16, device=a.device)

    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_k, stride_b_n = b.stride()
    stride_c_batch, stride_c_m, stride_c_n = c.stride()

    BLOCK_N = 64
    BLOCK_K = 32

    batched_matmul_kernel[grid](
        a, b, c,

"""
Triton‑accelerated outgoing TriMul forward pass.

Algorithm:
1. LayerNorm on the input (float32).
2. Linear projections (left/right) and gating linear layers are executed in
   fp16 for speed.
3. Optional element‑wise mask is applied to the left/right projections.
4. Gating (sigmoid) modulates left/right tensors.
5. Core contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication A @ B where
   A = left (i,k) and B = right (k,j) for each (b,d) pair.
   The contraction is performed by a custom Triton kernel that computes
   fp16 * fp16 → fp32 accumulation.
6. The result is reshaped back to (B,N,N,hidden_dim) and cast to float32.
7. Output LayerNorm, out‑gate scaling and final linear projection back to
   the original channel dimension are performed in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------
# Triton batched GEMM kernel (C = A @ B) with fp16 inputs,
# fp32 accumulation and fp16 output.
# ------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)   # batch dimension (B*H)
    pid_m = tl.program_id(1)      # block row
    pid_n = tl.program_id(2)      # block column

    # Compute offsets for the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a = tl.load(
            a_base + (offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k),
            mask=mask_m[:, None] & (offs_k < K),
            other=0.0,
            dtype=tl.float16,
        )

        # Load B tile (K x N) – B is not transposed in memory
        b = tl.load(
            b_base + (offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n),
            mask=(offs_k < K)[:, None] & mask_n[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # Multiply‑accumulate
        acc += tl.dot(a, b)   # fp16*fp16 → fp32

    # Write back the result tile
    c_ptrs = c_base + (offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n)
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def batched_matmul_triton(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B for fp16 tensors using Triton.

    Args:
        a: Tensor of shape (batch, M, K), dtype fp16, contiguous.
        b: Tensor of shape (batch, K, N), dtype fp16, contiguous.

    Returns:
        c: Tensor of shape (batch, M, N), dtype fp16, contiguous.
    """
    assert a.is_contiguous() and b.is_contiguous()
    assert a.dtype == torch.float16 and b.dtype == torch.float16
    B, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    c = torch.empty((B, M, N), dtype=torch.float16, device=a.device)

    # strides in elements
    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_k, stride_b_n = b.stride()
    stride_c_batch, stride_c_m, stride_c_n = c.stride()

    # Tuned tile sizes (work well for N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        a,
        b,
        c,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric)
            - weights_dict: mapping of parameter names to torch.Tensor
            - config_dict: must contain 'dim', 'hidden_dim' and optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim], float32

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for heavy linear operations
    # -----------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections (fp16)
    # -----------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # -----------------------------------------------------------------
    # 4. Optional mask (broadcast over hidden dimension)
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)  # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # -----------------------------------------------------------------
    # 5. Gating (fp16) and apply
    # -----------------------------------------------------------------
    left_gate = F.linear(
        x_fp16, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = F.linear(
        x_fp16, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = F.linear(
        x_fp16, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6. Core contraction via Triton batched GEMM
    #    Compute out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Equivalent to batched matrix multiplication per (b,d) head.
    # -----------------------------------------------------------------
    # Rearrange tensors to (B, H, N, N) where H = hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N) → (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) → (k,j)

    B, H, N, _ = left_perm.shape
    # Collapse batch and head dimensions for the kernel
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Perform batched GEMM: (B*H, N, N) @ (B*H, N, N)ᵀ
    out_flat = batched_matmul_triton(left_flat, right_flat)  # fp16 output

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # -----------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate scaling, final projection (float32)
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"])

    return out
