import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def mul_inplace_kernel(a_ptr, b_ptr, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    out = a * b
    tl.store(a_ptr + offs, out, mask=mask)

def custom_kernel(data):
    """
    Outgoing TriMul forward pass.

    Steps:
    1. LayerNorm on input (dim).
    2. Linear projections (left, right) and gate projections (left_gate, right_gate,
       out_gate) in fp16.
    3. Apply gating via a simple Triton element‑wise kernel.
    4. Optional mask multiplication (broadcasted).
    5. Core contraction: for each hidden dimension compute
          out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
       which is a batched matrix multiply left @ right^T. Executed with
       torch.bmm (cuBLAS) on tensors reshaped to (B*H, N, N).
    6. LayerNorm on the hidden dimension, multiply by out_gate, and final
       linear projection back to dim.
    Returns a float32 tensor of shape [B, N, N, dim].
    """
    # Unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # Cast to fp16 for all subsequent linear ops
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (fp16, no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Gate projections (fp16) and sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 4. Apply element‑wise gating using a tiny Triton kernel
    #    (in‑place multiplication to save memory)
    # ------------------------------------------------------------------
    numel = left.numel()
    BLOCK = 8192  # threads per program
    grid = (triton.cdiv(numel, BLOCK),)

    mul_inplace_kernel[grid](left.data_ptr(), left_gate.data_ptr(), numel, BLOCK=BLOCK)
    mul_inplace_kernel[grid](right.data_ptr(), right_gate.data_ptr(), numel, BLOCK=BLOCK)

    # Optionally apply mask (broadcast over hidden dim)
    if not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5. Core contraction: batched matmul left @ right^T
    #    left  : (B, N, N, H)
    #    right : (B, N, N, H)
    #    After permute we get (B, H, N, N) for both.
    #    Flatten batch*head dimension for torch.bmm.
    # ------------------------------------------------------------------
    # permute to (B, H, N, N) where H = hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) = right^T

    B, H, N, _ = left_perm.shape
    # reshape to (B*H, N, N)
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Batched matrix multiplication using cuBLAS (torch.bmm)
    out_flat = torch.bmm(left_flat, right_flat)   # (B*H, N, N) fp16

    # Restore original layout (B, N, N, H) and cast to fp32
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output layer‑norm, gate, and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    # out_gate is fp16, convert to fp32 for scaling
    out = out * out_gate.to(torch.float32)

    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
```

"""
Outgoing TriMul (TriMul‑out) forward pass.

The implementation follows the reference PyTorch code but accelerates the
element‑wise gating (left*left_gate and right*right_gate) with a tiny Triton
kernel and performs the heavy O(N³) contraction with a batched
torch.bmm (cuBLAS).  All tensors are kept in float32 for numerical fidelity;
only the gating kernel operates in‑place on the same data to avoid extra
allocations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def mul_inplace_kernel(a_ptr, b_ptr, size, BLOCK: tl.constexpr):
    """In‑place element‑wise multiplication a *= b."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(a_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or float), may be unused
            - weights_dict: mapping from parameter name to torch.Tensor (float32)
            - config_dict: contains at least 'dim' and 'hidden_dim', optionally 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – keep float32 for exactness
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 4. Apply element‑wise gating using a tiny Triton kernel (in‑place)
    # ------------------------------------------------------------------
    numel = left.numel()
    BLOCK = 8192  # threads per program – a safe value for simple kernels
    grid = (triton.cdiv(numel, BLOCK),)

    # left *= left_gate  (in‑place)
    mul_inplace_kernel[grid](left, left_gate, numel, BLOCK=BLOCK)
    # right *= right_gate (in‑place)
    mul_inplace_kernel[grid](right, right_gate, numel, BLOCK=BLOCK)

    # ------------------------------------------------------------------
    # 5. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp = mask.to(torch.float32).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 6. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    This is a batched matrix multiply left @ rightᵀ.
    # ------------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # rightᵀ: swap the two N axes before permuting
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape
    # Collapse batch and head dimensions for torch.bmm
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Batched GEMM using cuBLAS (torch.bmm)
    out_flat = torch.bmm(left_flat, right_flat)  # (B*H, N, N), float32

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, gate, and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate  # broadcasting over hidden dimension
    out = F.linear(out, weights["to_out.weight"])

    return out
