"""
Outgoing TriMul forward pass (AlphaFold3 style).

Algorithm
---------
1. Layer‑norm the input (float16).
2. Fuse the five linear projections (left, right, three gates) into a single
   matmul: concat the weight matrices (shape (5*hidden, dim)) and split the
   result.
3. Apply sigmoid to the three gate tensors.
4. Fuse mask (if present) and gating into a single Triton kernel that updates
   `left` and `right` in‑place:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
   (mask is omitted when `nomask=True`).
5. Contract over the summation dimension `k`:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM computed with `torch.bmm`.
6. Output LayerNorm, multiply by `out_gate`, and final linear projection back
   to the original dimension.
The implementation keeps all heavy compute in half‑precision and uses only a
single Triton kernel (step 4) to satisfy the “must use Triton” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask (optional) + left/right gates
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # [total] float16
    right_ptr,         # [total] float16
    left_gate_ptr,     # [total] float16
    right_gate_ptr,    # [total] float16
    mask_ptr,          # [mask_total] float16 (broadcasted)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 = mask present, 0 = no mask
    H: tl.constexpr,          # hidden dimension (size of channel axis)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # One mask value is shared by H consecutive entries.
        mask_off = offs // H
        mask_val = tl.load(mask_ptr + mask_off, mask=mask, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    All tensors are 1‑D contiguous float16.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when no mask

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # No return – the operation is performed in‑place.


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or numeric) or None
            - weights: mapping of parameter names to torch.Tensor (float32)
            - config: dict with keys ``dim``, ``hidden_dim``, optional ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (half precision)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(torch.float16).contiguous(),
        bias=weights["norm.bias"].to(torch.float16).contiguous(),
    )
    x_h = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Fused linear projections + gates (single matmul)
    # --------------------------------------------------------------
    # Concatenate weight matrices: [left_proj, right_proj, left_gate,
    # right_gate, out_gate]  -> shape (5*hidden, dim)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16).contiguous()

    # One large linear transform
    proj_all = F.linear(x_h, w_concat)  # (B, N, N, 5*hidden)

    h = hidden_dim
    left = proj_all[..., :h]                     # (B, N, N, h)
    right = proj_all[..., h : 2 * h]            # (B, N, N, h)
    left_gate = proj_all[..., 2 * h : 3 * h]    # (B, N, N, h)
    right_gate = proj_all[..., 3 * h : 4 * h]   # (B, N, N, h)
    out_gate = proj_all[..., 4 * h : 5 * h]    # (B, N, N, h)

    # Sigmoid on gates
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    # --------------------------------------------------------------
    # 3. Fuse mask and gating into left/right (Triton)
    # --------------------------------------------------------------
    if not nomask:
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        # Flatten everything for the kernel
        left_flat = left.view(-1)
        right_flat = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)
        mask_flat = mask_h.view(-1)  # shape (B*N*N)
        _apply_gate_and_mask(
            left_flat, right_flat,
            left_gate_flat, right_gate_flat,
            mask_flat,
            hidden_dim,
        )
        # Restore shapes
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # Simple element‑wise multiplication when there is no mask
        left = left * left_gate
        right = right * right_gate

    # Gates no longer needed
    del left_gate, right_gate, proj_all, x_h

    # --------------------------------------------------------------
    # 4. Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # Rearrange for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, h, N, N)  -> (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, h, N, N)  -> (k,j)
    B, H, N, _ = left_perm.shape

    left_mat = left_perm.reshape(B * H, N, N)   # (B*H, i, k)
    right_mat = right_perm.reshape(B * H, N, N) # (B*H, k, j)

    out_mat = torch.bmm(left_mat, right_mat)   # (B*H, i, j)

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, h)

    # --------------------------------------------------------------
    # 5. Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16).contiguous(),
        bias=weights["to_out_norm.bias"].to(torch.float16).contiguous(),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16).contiguous())

    # Return in float32 as required
    return out.to(torch.float32)
