    right = tl.load(right_ptr + offs, mask=mask)

                          left_ptr, right_ptr,

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def fused_gate_mask_kernel(
    out_left_ptr, out_right_ptr,
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements,
    hidden_dim: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load values
    left = tl.load(left_ptr + offs, mask=mask)
    right = tl.load(right_ptr + offs, mask=mask)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask)

    # mask broadcast over hidden dimension
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Multiply
    out_l = left * left_gate * m
    out_r = right * right_gate * m

    # Store
    tl.store(out_left_ptr + offs, out_l, mask=mask)
    tl.store(out_right_ptr + offs, out_r, mask=mask)

def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.
    The heavy elementwise gating+masking is fused in a single Triton kernel,
    while the cubic contraction is performed with a batched fp16 GEMM,
    giving a fast implementation without gradients.
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. Input layer‑norm (FP32)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # [B,N,N,dim] FP32

    # -----------------------------------------------------------------
    # 2. Cast to FP16 for the rest of the pipeline
    # -----------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # [B,N,N,hidden] FP16
    right = F.linear(x_fp16, right_proj_w)  # [B,N,N,hidden] FP16

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid) – also FP16
    # -----------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))   # [B,N,N,hidden] FP16

    # -----------------------------------------------------------------
    # 5. Fuse mask & gates with a Triton kernel
    # -----------------------------------------------------------------
    if mask_tensor is None:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(torch.float16).to(device)

    total = B * N * N * hidden

    # Flatten tensors (contiguous view)
    left_flat = left.contiguous().view(-1)
    right_flat = right.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)
    mask_flat = mask_fp16.contiguous().view(-1)  # length B*N*N

    left_out_flat = torch.empty_like(left_flat)
    right_out_flat = torch.empty_like(right_flat)

    # Choose a block size that is a multiple of 32 for good coalescing.
    BLOCK = 65536  # 64k elements per program
    grid = (triton.cdiv(total, BLOCK),)

    fused_gate_mask_kernel[grid](
        left_out_flat,
        right_out_flat,
        left_flat,
        right_flat,
        left_gate_flat,
        right_gate_flat,
        mask_flat,
        total,
        hidden,
        BLOCK_SIZE=BLOCK,
    )

    # Reshape back to [B,N,N,hidden]
    left = left_out_flat.view(B, N, N, hidden)
    right = right_out_flat.view(B, N, N, hidden)

    # Free intermediate tensors that are no longer needed
    del left_flat, right_flat, left_gate_flat, right_gate_flat, left_gate, right_gate, left, right, x_fp16, x

    # -----------------------------------------------------------------
    # 6. Cubic contraction via batched GEMM (fp16)
    # -----------------------------------------------------------------
    # left: (B,N,N,hidden) -> (B, hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right transposed on the two N dimensions: (B,N,N,hidden) -> (B, hidden, N, N) where we swap the inner N dims
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    BH = B * hidden
    left_mat = left_perm.view(BH, N, N)
    right_mat = right_perm.view(BH, N, N)

    out_mat = torch.bmm(left_mat, right_mat)   # (B*hidden, N, N) fp16

    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden] fp16

    # -----------------------------------------------------------------
    # 7. Post‑processing (LayerNorm, out‑gate, final linear)
    # -----------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    out_norm = F.layer_norm(out_fp32, (hidden,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply final gating
    out_norm = out_norm * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"].to(device)   # (dim, hidden)
    output = F.linear(out_norm, to_out_w)           # [B,N,N,dim] float32

    return output
```

"""
TriMul (outgoing) forward pass with Triton acceleration.

The heavy element‑wise part (mask × gate × tensor for the left and right
branches) is fused into a single Triton kernel.  The cubic contraction
`\sum_k left[...,i,k,d] * right[...,j,k,d]` is performed as a batched
matrix‑multiply (`torch.bmm`) on fp16 tensors, which leverages cuBLAS
Tensor‑Core kernels.  The remaining linear layers and LayerNorms are
executed in PyTorch.  The final output is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def fused_gate_mask_kernel(
    out_left_ptr, out_right_ptr,      # output pointers
    left_ptr, right_ptr,              # projected tensors
    left_gate_ptr, right_gate_ptr,    # gate tensors (sigmoid outputs)
    mask_ptr,                         # mask tensor (broadcast over hidden dim)
    total_elements,                   # total number of elements = B*N*N*H
    hidden_dim: tl.constexpr,         # size of the innermost (channel) dimension
    BLOCK_SIZE: tl.constexpr,         # compile‑time constant block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load the four tensors
    left = tl.load(left_ptr + offs, mask=mask)
    right = tl.load(right_ptr + offs, mask=mask)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask)

    # Broadcast mask over the hidden dimension
    mask_idx = offs // hidden_dim          # integer division, maps each element to its (b,i,j) mask
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Fuse the three element‑wise multiplications
    out_l = left * left_gate * m
    out_r = right * right_gate * m

    # Write the results
    tl.store(out_left_ptr + offs, out_l, mask=mask)
    tl.store(out_right_ptr + offs, out_r, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor : torch.Tensor [B, N, N] (float32) or None
            - weights_dict: model weights with keys:
                'norm.weight', 'norm.bias',
                'left_proj.weight', 'right_proj.weight',
                'left_gate.weight', 'right_gate.weight', 'out_gate.weight',
                'to_out_norm.weight', 'to_out_norm.bias',
                'to_out.weight'
            - config_dict : {'dim': int, 'hidden_dim': int}

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # [B,N,N,dim]

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for the bulk of the computation
    # -----------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)     # [B,N,N,hidden] (fp16)
    right = F.linear(x_fp16, right_proj_w)   # [B,N,N,hidden] (fp16)

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid) – also fp16
    # -----------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))   # [B,N,N,hidden] fp16

    # -----------------------------------------------------------------
    # 5. Fuse mask + gates with a Triton kernel
    # -----------------------------------------------------------------
    if mask_tensor is None:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(torch.float16).to(device)

    total = B * N * N * hidden  # total number of elements to process

    # Flatten everything (contiguous view) for the 1‑D kernel
    left_flat = left.contiguous().view(-1)
    right_flat = right.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)
    mask_flat = mask_fp16.contiguous().view(-1)   # length = B*N*N

    left_out_flat = torch.empty_like(left_flat)
    right_out_flat = torch.empty_like(right_flat)

    # Block size – a multiple of 32 for good memory coalescing
    BLOCK = 65536
    grid = (triton.cdiv(total, BLOCK),)

    fused_gate_mask_kernel[grid](
        left_out_flat,
        right_out_flat,
        left_flat,
        right_flat,
        left_gate_flat,
        right_gate_flat,
        mask_flat,
        total,
        hidden,
        BLOCK_SIZE=BLOCK,
    )

    # Reshape back to the original 4‑D layout
    left = left_out_flat.view(B, N, N, hidden)
    right = right_out_flat.view(B, N, N, hidden)

    # -----------------------------------------------------------------
    # 6. Cubic contraction via batched fp16 GEMM
    # -----------------------------------------------------------------
    # left:   (B,N,N,hidden) -> (B,hidden,N,N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right:  (B,N,N,hidden) -> (B,hidden,N,N) with inner N’s swapped (j,k) -> (k,j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    BH = B * hidden
    left_mat = left_perm.view(BH, N, N)     # (B*hidden, N, N)
    right_mat = right_perm.view(BH, N, N)   # (B*hidden, N, N)

    out_mat = torch.bmm(left_mat, right_mat)   # (B*hidden, N, N), fp16

    # Back to [B,N,N,hidden]
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # -----------------------------------------------------------------
    # 7. Post‑processing: LayerNorm, final gate, output projection
    # -----------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    out_norm = F.layer_norm(out_fp32, (hidden,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply the final gating
    out_norm = out_norm * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"].to(device)   # shape (dim, hidden)
    output = F.linear(out_norm, to_out_w)           # [B,N,N,dim], float32

    return output
