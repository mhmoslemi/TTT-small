"""
TriMul (outgoing) – fast forward pass.

The heavy O(N³·C) contraction
    out[b, i, j, c] = Σ_k left[b, i, k, c] * right[b, j, k, c]
is realized as a batched matrix multiplication for each hidden channel:
    left_perm  : (B, C, N, N)  with axes (i,k)
    right_perm : (B, C, N, N)  with axes (k,j)
    out_perm   = batched_gemm(left_perm, right_perm)  → (B, C, N, N)
After the contraction a layer‑norm, an output‑gate and a final linear
projection restore the original dimension.

All element‑wise mask & gate fusions are performed with a tiny Triton kernel,
while the cubic contraction uses the highly‑optimised cuBLAS batched GEMM
(`torch.bmm`).  The final tensor is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate into the projected tensors.
# ----------------------------------------------------------------------
@triton.jit
def apply_mask_gate_kernel(
    left_ptr,                     # fp16 * (B,N,N,C)
    right_ptr,                    # fp16 * (B,N,N,C)
    mask_ptr,                     # fp16 * (B,N,N,1)  broadcast over C
    left_gate_ptr,                # fp16 * (B,N,N,C)
    right_gate_ptr,               # fp16 * (B,N,N,C)
    B, N, C,                      # runtime constants
    # strides for left   (b,i,k,c)
    stride_left_b, stride_left_i, stride_left_k, stride_left_c,
    # strides for right  (b,i,k,c) – same layout as left
    stride_right_b, stride_right_i, stride_right_k, stride_right_c,
    # strides for mask   (b,i,k,1) – ignore last dim stride
    stride_mask_b, stride_mask_i, stride_mask_k,
    # strides for left‑gate  (b,i,k,c)
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_c,
    # strides for right‑gate (b,i,k,c)
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_c,
    BLOCK: tl.constexpr,
):
    """In‑place: left = left * mask * left_gate; right = right * mask * right_gate."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * C
    valid = offs < total

    # -------------------- decode linear index --------------------
    b = offs // (N * N * C)
    rem = offs % (N * N * C)
    i = rem // (N * C)
    rem2 = rem % (N * C)
    k = rem2 // C
    c = rem2 % C

    # -------------------- compute pointers --------------------
    left_off = (
        b * stride_left_b
        + i * stride_left_i
        + k * stride_left_k
        + c * stride_left_c
    )
    right_off = (
        b * stride_right_b
        + i * stride_right_i
        + k * stride_right_k
        + c * stride_right_c
    )
    lg_off = (
        b * stride_lg_b
        + i * stride_lg_i
        + k * stride_lg_k
        + c * stride_lg_c
    )
    rg_off = (
        b * stride_rg_b
        + i * stride_rg_i
        + k * stride_rg_k
        + c * stride_rg_c
    )
    mask_off = b * stride_mask_b + i * stride_mask_i + k * stride_mask_k  # broadcast over C

    # -------------------- load, compute, store --------------------
    l = tl.load(left_ptr + left_off, mask=valid, other=0.0)
    r = tl.load(right_ptr + right_off, mask=valid, other=0.0)
    lg = tl.load(left_gate_ptr + lg_off, mask=valid, other=0.0)
    rg = tl.load(right_gate_ptr + rg_off, mask=valid, other=0.0)
    m = tl.load(mask_ptr + mask_off, mask=valid, other=1.0)   # 1.0 when mask is omitted

    l_new = l * lg * m
    r_new = r * rg * m

    tl.store(left_ptr + left_off, l_new, mask=valid)
    tl.store(right_ptr + right_off, r_new, mask=valid)


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (tensor, mask, weights, config)
            tensor : torch.Tensor of shape [B, N, N, dim] (float32)
            mask   : torch.Tensor of shape [B, N, N]   (bool or numeric) or None
            weights: dict of model parameters (float32)
            config : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] float32

    # --------------------------------------------------------------
    # 2. Cast to fp16 for the heavy path
    # --------------------------------------------------------------
    x_h = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 4. Gate vectors (fp16) – one per hidden channel
    # --------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # --------------------------------------------------------------
    # 5. Fuse mask and gates with a tiny Triton kernel
    # --------------------------------------------------------------
    B, N, _, _ = left.shape  # B,N,N,hidden_dim
    if (mask is None) or nomask:
        mask_fp = torch.ones((B, N, N, 1), dtype=torch.float16, device=device)
    else:
        # mask may be bool, int or float – cast to fp16 and add trailing dim
        mask_fp = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]

    # Prepare strides
    left_stride = left.stride()          # (b,i,k,c)
    right_stride = right.stride()
    mask_stride = mask_fp.stride()       # (b,i,k,1)
    lg_stride = left_gate.stride()
    rg_stride = right_gate.stride()

    total_elements = B * N * N * hidden_dim
    BLOCK = 128
    grid = (triton.cdiv(total_elements, BLOCK),)

    apply_mask_gate_kernel[grid](
        left,
        right,
        mask_fp,
        left_gate,
        right_gate,
        B, N, hidden_dim,
        *left_stride,
        *right_stride,
        mask_stride[0], mask_stride[1], mask_stride[2],
        *lg_stride,
        *rg_stride,
        BLOCK=BLOCK,
    )
    # At this point `left` and `right` already contain mask & gate factors.

    # --------------------------------------------------------------
    # 6. Core O(N³) contraction via batched GEMM (torch.bmm)
    #    out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
    # --------------------------------------------------------------
    # Rearrange to (B, C, N, N) where the inner dimensions are (i,k) and (k,j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, C, N, i)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, C, k, j)

    B, C, N, _ = left_perm.shape
    # Flatten batch & channel → (B*C, N, N)
    left_flat = left_perm.view(B * C, N, N)
    right_flat = right_perm.view(B * C, N, N)

    # Batched matrix multiplication (fp16 → fp16, accumulation in fp32)
    out_flat = torch.bmm(left_flat, right_flat)  # (B*C, N, N)

    # Restore original layout (B, N, N, C)
    out = out_flat.view(B, C, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, C)
    out = out.to(torch.float32)  # go back to fp32 for the following ops

    # --------------------------------------------------------------
    # 7. Output LayerNorm + output gate
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)

    # --------------------------------------------------------------
    # 8. Final projection back to original dimension
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
