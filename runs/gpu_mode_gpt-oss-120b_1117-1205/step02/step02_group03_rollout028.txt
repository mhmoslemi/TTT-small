"""
Triton‑accelerated outgoing TriMul (AlphaFold3).

The N³ contraction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
is realized as a batched matrix‑multiply (left @ rightᵀ) for each
hidden channel d.  An element‑wise Triton kernel fuses the mask and
gate multiplications to keep the heavy path in FP16.  The remaining
operations (LayerNorms, Linear projections, final projection) are
performed with PyTorch’s highly‑optimized kernels.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask (if any) and gate multiplications
# ----------------------------------------------------------------------
@triton.jit
def fuse_mul_kernel(
    inp_ptr,                # tensor to be multiplied in‑place (fp16)
    gate_ptr,               # gate tensor (fp16)
    mask_ptr,               # mask tensor broadcasted over hidden dim (fp16)
    out_ptr,                # output pointer (fp16, same as inp_ptr for in‑place)
    total_elems,            # total number of elements in inp/gate
    H: tl.constexpr,        # hidden dimension (size of the broadcasted dim)
    HAS_MASK: tl.constexpr, # 1 if mask should be applied, 0 otherwise
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    m = offs < total_elems

    a = tl.load(inp_ptr + offs, mask=m, other=tl.float16(0.0))
    g = tl.load(gate_ptr + offs, mask=m, other=tl.float16(0.0))
    out = a * g
    if HAS_MASK:
        # index into the mask (mask does not have the hidden dim)
        idx_mask = offs // H
        mask_val = tl.load(mask_ptr + idx_mask, mask=m, other=tl.float16(1.0))
        out = out * mask_val
    tl.store(out_ptr + offs, out, mask=m)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weight_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool or numeric)
            - weight_dict  : mapping from parameter names to tensors
            - config_dict  : contains "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to FP16 for the heavy path
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (FP16, no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Gate projections (FP16)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) and gate multiplications via Triton
    # ------------------------------------------------------------------
    BLOCK = 1024
    HAS_MASK = 0 if nomask else 1

    if not nomask:
        # mask broadcasted over hidden dimension
        mask_fp = mask.unsqueeze(-1).to(torch.float16).contiguous()
        mask_arg = mask_fp
    else:
        # dummy tensor; will never be read because HAS_MASK=0
        mask_arg = torch.empty((1,), dtype=torch.float16, device=device)

    # left = left * left_gate * mask
    total = left.numel()
    grid = (triton.cdiv(total, BLOCK),)
    fuse_mul_kernel[grid](
        left,
        left_gate,
        mask_arg,
        left,
        total,
        H=hidden_dim,
        HAS_MASK=HAS_MASK,
        BLOCK_SIZE=BLOCK,
    )

    # right = right * right_gate * mask
    total = right.numel()
    grid = (triton.cdiv(total, BLOCK),)
    fuse_mul_kernel[grid](
        right,
        right_gate,
        mask_arg,
        right,
        total,
        H=hidden_dim,
        HAS_MASK=HAS_MASK,
        BLOCK_SIZE=BLOCK,
    )

    # ------------------------------------------------------------------
    # 5. Core contraction: batched GEMM (left @ rightᵀ) per hidden channel
    #    left, right : [B, N, N, H] (FP16)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # Rearrange to (B, H, N, N) and make contiguous for cuBLAS
    left_perm = left.permute(0, 3, 1, 2).contiguous()      # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()    # (B, H, N, N) -> K dimension first

    # Collapse batch & hidden dim -> (B*H, N, N)
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Batched matrix multiply (FP16 -> FP32 accumulation internally)
    out_flat = torch.bmm(left_flat, right_flat)  # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # back to FP32 for the remaining layers

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final linear (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
