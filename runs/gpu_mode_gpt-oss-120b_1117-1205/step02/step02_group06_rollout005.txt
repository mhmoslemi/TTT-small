def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. layernorm input (float32)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    ) # keep as float32

    # Cast to half for projection and gating
    x_fp16 = x.half()  # to fp16

    # Linear projections (no bias)
    left = torch.nn.functional.linear(x_fp16, weights["left_proj.weight"].half())
    right = torch.nn.functional.linear(x_fp16, weights["right_proj.weight"].half())
    # left/right shape [B,N,N,hidden_dim], dtype half

    # Apply mask if needed
    if not nomask:
        mask_fp = mask.to(x.dtype).unsqueeze(-1)  # float32? mask may be bool; we can cast to half
        mask_fp16 = mask_fp.half()
        left = left * mask_fp16
        right = right * mask_fp16

    # Compute gates (half)
    left_gate = torch.nn.functional.linear(x_fp16, weights["left_gate.weight"].half()).sigmoid()
    right_gate = torch.nn.functional.linear(x_fp16, weights["right_gate.weight"].half()).sigmoid()
    out_gate = torch.nn.functional.linear(x_fp16, weights["out_gate.weight"].half()).sigmoid()
    # out_gate shape [B,N,N,hidden_dim], dtype half.

    left = left * left_gate
    right = right * right_gate

    # Now compute core contraction using triton batched matmul with half inputs
    # Permute to batch dimension for matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # B, H, N, N
    right_perm = right.permute(0, 3, 2, 1).contiguous() # B, H, N, N

    Bsize, Hsize, N, _ = left_perm.shape
    # flatten batch and head:
    A = left_perm.view(Bsize * Hsize, N, N)

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

"""
TriMul (outgoing) forward pass with a fused Triton batched GEMM.

The core contraction:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is reformulated as a batched matrix multiplication:
    out_{b,h} = left_{b,h} @ right_{b,h}^T,
where left_{b,h} and right_{b,h} are N×N matrices.
The kernel `_batched_matmul_kernel` efficiently computes the
batched GEMM in FP16 with FP32 accumulation.
All remaining operations (LayerNorm, Linear projections, gating,
masking, final projection) are performed in PyTorch using mixed
precision for speed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ B   (A: (B, M, K), B: (B, K, N))
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute C = A @ B for a batch of matrices."""
    batch_id = tl.program_id(0)          # batch dimension (B*H)
    pid_m = tl.program_id(1)            # block row
    pid_n = tl.program_id(2)            # block column

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A tile [BLOCK_M, BLOCK_K]
        a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        # Load B tile [BLOCK_K, BLOCK_N]
        b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate product
        acc += tl.dot(a, b)

    # Write back FP16 result
    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _triton_batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B using the Triton kernel above.
    A : (batch, M, K), dtype=half, contiguous
    B : (batch, K, N), dtype=half, contiguous
    Returns C : (batch, M, N), dtype=half
    """
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tunable block sizes (work well for N up to 1024)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) – may be ignored
            - weights_dict: mapping from parameter name to torch.Tensor
            - config_dict: must contain keys 'dim' and 'hidden_dim'.
              Optional key 'nomask' (bool) indicates that mask should be ignored.

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. LayerNorm on the input channel dimension (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # shape [B, N, N, dim], dtype float32

    # --------------------------------------------------------------
    # 2. Cast to half and perform linear projections (no bias)
    # --------------------------------------------------------------
    x_fp16 = x.half()  # FP16 for the heavy compute

    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()

    left = F.linear(x_fp16, left_proj_w)   # [B, N, N, H]
    right = F.linear(x_fp16, right_proj_w)  # [B, N, N, H]

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 4. Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()

    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()   # [B, N, N, H]
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()  # [B, N, N, H]
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()     # [B, N, N, H]

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction: batched GEMM using Triton
    #    out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    # --------------------------------------------------------------
    # Permute to (B, H, N, N) where the second operand is already transposed.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) = right^T

    B_sz, H_sz, N, _ = left_perm.shape
    # Flatten batch & head dimension for the kernel
    A = left_perm.view(B_sz * H_sz, N, N)
    B = right_perm.view(B_sz * H_sz, N, N)

    # Batched matmul in FP16 (result still FP16)
    out_flat = _triton_batched_matmul(A, B)   # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B_sz, H_sz, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # convert to FP32 for the rest of the pipeline

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gate, final linear projection (all FP32)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)   # broadcast multiplication

    out = F.linear(out, weights["to_out.weight"])  # final projection to dim

    return out
