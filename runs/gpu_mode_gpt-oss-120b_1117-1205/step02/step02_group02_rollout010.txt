"""
TriMul (outgoing) forward pass with a Triton‑accelerated batched GEMM.

The core contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is equivalent to a batched matrix multiplication A @ Bᵀ for each
hidden dimension d.  We first apply LayerNorm, two linear projections,
optional masking and sigmoid gating (all in PyTorch, using fp16 where
possible).  The remaining N×N×N reduction is performed by a custom
Triton kernel that computes many independent GEMMs in parallel:
    A = left.permute(0,3,1,2)  # (B, D, N, N)   (i,k)
    B = right.permute(0,3,2,1) # (B, D, N, N)   (k,j)
    out = batched_matmul(A, B) # (B, D, N, N)   (i,j)
Finally we apply a second LayerNorm, an output gate and a linear
projection back to the original dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ B) for tensors of shape
#   (batch, M, K) and (batch, K, N)  (all fp16, fp32 accumulation)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix dimensions
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # --------------------------------------------------------------
    # program ids
    # --------------------------------------------------------------
    batch_id = tl.program_id(0)          # which (B * D) batch
    pid_m = tl.program_id(1)             # row tile
    pid_n = tl.program_id(2)             # col tile

    # --------------------------------------------------------------
    # tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over K dimension
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # load A tile (M × K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=tl.float16(0.0))

        # load B tile (K × N)
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=tl.float16(0.0))

        # fp16 dot → fp32 accumulation
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # write back result
    # --------------------------------------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def triton_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM C = A @ B where A, B are (batch, M, K) and (batch, K, N)
    with dtype torch.float16. Returns C of shape (batch, M, N) (float16).
    """
    assert A.dtype == B.dtype == torch.float16
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, f"K mismatch: {K} vs {K2}"

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # strides in elements
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tuned tile sizes (good for fp16 tensor‑core on Ampere/A100)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor:  torch.Tensor [B, N, N] (bool or numeric)
            - weights_dict: mapping from parameter name to torch.Tensor
            - config_dict: must contain 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the heavy sub‑graph
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (bias‑free, fp16)
    # ------------------------------------------------------------------
    left = F.linear(
        x_fp16,
        weights["left_proj.weight"].to(torch.float16)
    )  # [B, N, N, hidden_dim]

    right = F.linear(
        x_fp16,
        weights["right_proj.weight"].to(torch.float16)
    )  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 4. Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)   # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 5. Gating (fp16)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_fp16,
        weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()

    right_gate = F.linear(
        x_fp16,
        weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()

    out_gate = F.linear(
        x_fp16,
        weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()  # [B, N, N, hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Core contraction via Triton: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    We treat each (b, d) pair as an independent batch and perform
    #    a standard GEMM A @ B where A = left[b,d,i,k] and B = right[b,d,k,j].
    # ------------------------------------------------------------------
    #   (B, N, N, D) → (B, D, N, N)   (i,k) for left, (k,j) for right
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, D, N, N)   (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, D, N, N)   (k,j)

    B_, D_, N_, _ = left_perm.shape
    left_flat = left_perm.view(B_ * D_, N_, N_)   # (B*D, M=N, K=N)
    right_flat = right_perm.view(B_ * D_, N_, N_) # (B*D, K=N, N=N)

    # Batched matrix multiplication using Triton (fp16 in, fp16 out)
    out_flat = triton_matmul(left_flat, right_flat)  # (B*D, N, N) fp16

    # Reshape back to (B, N, N, D)
    out = out_flat.view(B_, D_, N_, N_).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # convert to fp32 for the remaining ops

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
