@triton.jit
def batched_matmul_kernel(a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_ptrs = a_ptr + batch_id * stride_a_batch
    b_ptrs = b_ptr + batch_id * stride_b_batch
    c_ptrs = c_ptr + batch_id * stride_c_batch

    # compute tile pointers
    # A tile: (BLOCK_M x BLOCK_K)
    # B tile: (BLOCK_K x BLOCK_N)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        a_tile_ptr = a_ptrs + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_tile_ptr = b_ptrs + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        a = tl.load(a_tile_ptr,
                    mask=(mask_m[:, None] & (offs_k < K)),
                    other=0.0)
        b = tl.load(b_tile_ptr,
                    mask=((offs_k < K)[:, None] & mask_n[None, :]),
                    other=0.0)

        acc += tl.dot(a, b)  # fp16*fp16 : fp32

    # store results
    c_tile_ptr = c_ptrs + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_tile_ptr, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # layernorm
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])
    # convert to fp16
    x_fp16 = x.to(torch.float16)

    # compute projections
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # mask if needed
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)  # (B,N,N,1)
        left = left * mask_fp
        right = right * mask_fp

    # compute gates
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # core contraction
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) = transpose k,j

    B_, H_, N_, _ = left_mat.shape
    left_flat = left_mat.view(B_ * H_, N_, N_)
    right_flat = right_mat.view(B_ * H_, N_, N_)

    out_flat = triton_matmul(left_flat, right_flat)  # (B*H, N, N), fp16

    out = out_flat.view(B_, H_, N_, N_).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)
    out = out.to(torch.float32)

    # output layernorm
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"])

    return out
```

"""
TriMul (outgoing) forward pass with Triton-accelerated batched GEMM.

The heavy N³ contraction:
    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
is implemented as a batched matrix multiplication A @ Bᵀ for each
hidden dimension d.  All surrounding operations (LayerNorms,
linear projections, gating and final projection) are performed in PyTorch.
The core contraction is fused into a single Triton kernel that works on
half‑precision inputs (fp16) with float‑32 accumulation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (C = A @ B) for tensors of shape
#   A : (batch, M, K)  fp16
#   B : (batch, K, N)  fp16
#   C : (batch, M, N)  fp16 (float32 accumulator)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)                     # fp16*fp16 -> fp32

    # store result
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def triton_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM C = A @ B  (fp16 inputs, fp32 accumulation, fp16 output)

    Args:
        A: (batch, M, K)  torch.float16
        B: (batch, K, N)  torch.float16

    Returns:
        C: (batch, M, N) torch.float16
    """
    assert A.is_contiguous() and B.is_contiguous()
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # strides (in elements)
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # tile sizes (tuned for Ampere B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor:  torch.Tensor [B, N, N] (bool / numeric)
            - weights_dict: mapping from parameter name to torch.Tensor
            - config_dict: must contain 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim]  float32

    # convert to fp16 for the heavy part
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (fp16, bias=False)
    # ------------------------------------------------------------------
    left = F.linear(
        x_fp16,
        weights["left_proj.weight"].to(torch.float16)
    )  # [B, N, N, hidden_dim]
    right = F.linear(
        x_fp16,
        weights["right_proj.weight"].to(torch.float16)
    )  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)  # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 4. Gating tensors (fp16)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_fp16,
        weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = F.linear(
        x_fp16,
        weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = F.linear(
        x_fp16,
        weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()  # [B, N, N, hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: batched GEMM via Triton
    #    Compute out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    => for each (b,d) do A @ Bᵀ where
    #       A = left_{b,:, :, d}  (i × k)
    #       Bᵀ = right_{b,:, :, d}.transpose(-1,-2)  (k × j)
    # ------------------------------------------------------------------
    #   Rearrange to (B, H, N, N) where H = hidden_dim
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, i/k)
    right_mat = right.permute(0, 3, 2, 1).contiguous()  # (B, H, k, j)

    B_, H_, N_, _ = left_mat.shape
    left_flat = left_mat.view(B_ * H_, N_, N_)   # (B*H, N, N)
    right_flat = right_mat.view(B_ * H_, N_, N_)  # (B*H, N, N)   (already transposed)

    out_flat = triton_matmul(left_flat, right_flat)  # (B*H, N, N) fp16

    # reshape back to (B, N, N, H)
    out = out_flat.view(B_, H_, N_, N_).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # cast back to fp32 for the rest

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"])

    return out
