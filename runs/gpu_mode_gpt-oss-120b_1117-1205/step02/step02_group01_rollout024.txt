def mul_gate_mask_kernel(

"""
TriMul (outgoing) forward pass – Triton‑accelerated elementwise
mask‑gate fusion.

Algorithm
---------
1. Layer‑norm the input (FP32).
2. Cast to FP16 and compute linear projections:
   left = x·W_left  ,  right = x·W_right
3. Compute gating vectors (sigmoid(linear(x))) in FP16.
4. Fuse “tensor * gate * mask” for left and right branches with a
   single Triton kernel (mask broadcast over the hidden dimension).
5. Perform the cubic contraction
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   as a batched GEMM (torch.bmm) on the (B·H, N, N) view.
6. Layer‑norm over the hidden dimension (FP32), apply the output gate,
   and linearly project back to the original dimension.

The Triton kernel only handles the elementwise multiplication;
all heavy linear work and the matrix multiplication are left to
highly‑optimized PyTorch/cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def mul_gate_mask_kernel(
    out_ptr,          # pointer to output (flattened)
    a_ptr,            # pointer to tensor (flattened)
    gate_ptr,         # pointer to gate (flattened)
    mask_ptr,         # pointer to mask (flattened, broadcast over hidden dim)
    total,            # total number of elements (int)
    H,                # hidden dimension (int, constexpr)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # Load data
    a = tl.load(a_ptr + offs, mask=mask)
    g = tl.load(gate_ptr + offs, mask=mask)

    # Compute mask index (broadcast over H)
    idx = offs // H
    m = tl.load(mask_ptr + idx, mask=mask)

    # Store result
    tl.store(out_ptr + offs, a * g * m, mask=mask)


def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor : torch.Tensor [B, N, N] (float32) or None
            - weights_dict: dict containing model weights
            - config_dict : {"dim": int, "hidden_dim": int, "nomask": bool}
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # --------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # [B,N,N,dim]

    # --------------------------------------------------------------------
    # 2. Cast to FP16 for heavy work
    # --------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------------
    # 3. Linear projections (no bias)
    # --------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)    # [B,N,N,hidden_dim], fp16
    right = F.linear(x_fp16, right_proj_w)  # [B,N,N,hidden_dim], fp16

    # --------------------------------------------------------------------
    # 4. Gating vectors (sigmoid)
    # --------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w)) # [B,N,N,hidden_dim]
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))    # [B,N,N,hidden_dim]

    # --------------------------------------------------------------------
    # 5. Prepare mask (broadcasted over hidden dim)
    # --------------------------------------------------------------------
    if nomask or mask_tensor is None:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(torch.float16).to(device)

    # --------------------------------------------------------------------
    # 6. Fuse tensor * gate * mask with Triton
    # --------------------------------------------------------------------
    # Flatten tensors for the kernel
    left_flat = left.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_flat = right.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)
    mask_flat = mask_fp16.contiguous().view(-1)

    total_elems = left_flat.numel()       # B * N * N * hidden_dim
    BLOCK = 1024
    grid = (triton.cdiv(total_elems, BLOCK),)

    left_out_flat = torch.empty_like(left_flat)
    right_out_flat = torch.empty_like(right_flat)

    # Left branch
    mul_gate_mask_kernel[grid](
        left_out_flat,
        left_flat,
        left_gate_flat,
        mask_flat,
        total_elems,
        H=hidden_dim,
        BLOCK_SIZE=BLOCK,
    )
    # Right branch
    mul_gate_mask_kernel[grid](
        right_out_flat,
        right_flat,
        right_gate_flat,
        mask_flat,
        total_elems,
        H=hidden_dim,
        BLOCK_SIZE=BLOCK,
    )

    # Reshape back
    left = left_out_flat.view(B, N, N, hidden_dim)
    right = right_out_flat.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------------
    # 7. Cubic contraction via batched GEMM (torch.bmm)
    # --------------------------------------------------------------------
    # left: (B, N, N, H) -> (B, H, N, N)
    # right: (B, N, N, H) -> (B, H, N, N) with swapped axes for contraction
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)    # (B*H, N, N)
    right_mat = right_perm.view(BH, N, N)  # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N), fp16

    # Restore [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------------
    # 8. Post‑processing: LayerNorm, output gate, final linear
    # --------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float32)

    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply output gate (cast to FP32)
    out_norm = out_norm * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float32)  # (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)  # [B, N, N, dim]

    return output
