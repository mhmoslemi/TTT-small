"""
TriMul (outgoing) forward pass optimized with a Triton batched GEMM.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension (float32).
2. Linear projections to the hidden dimension (float32).
3. Optional mask multiplication.
4. Compute three sigmoid gates (float32) and apply them.
5. Fuse the core contraction:
   For each batch element and hidden channel d,
   compute   out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]
   which is a batched matrix multiplication left @ right.T over the
   sequence length N.  This is performed by a Triton kernel that processes
   (B·hidden_dim) independent GEMMs of size N×N with FP16 inputs and FP32
   accumulation.
6. Layer‑norm on the hidden dimension, apply the out‑gate, and a final
   linear projection back to the original channel dimension (float32).

All heavy arithmetic is done in FP16 to exploit Tensor‐cores; the final
output is returned in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton batched GEMM kernel (A @ B) for the core contraction
# --------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute C = A @ B for a batch of independent matrices.
    A: (batch, M, K)  FP16
    B: (batch, K, N)  FP16
    C: (batch, M, N)  FP16 (stored after FP32 accumulation)
    """
    batch_id = tl.program_id(0)       # which GEMM in the batch
    pid_m = tl.program_id(1)          # tile row
    pid_n = tl.program_id(2)          # tile column

    # Offsets within the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Boundary masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch element
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension with tiling
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        # Load B tile (K x N)
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        # Matrix‑multiply‑accumulate
        acc += tl.dot(a, b)

    # Write out C (convert back to FP16)
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def triton_batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM: C = A @ B where
        A : (batch, M, K)    FP16
        B : (batch, K, N)    FP16   (second operand already transposed)
    Returns
        C : (batch, M, N)    FP16
    """
    assert A.dtype == B.dtype == torch.float16
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides for contiguous tensors
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tile sizes (tuned for NVIDIA B200 Tensor‑cores)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# --------------------------------------------------------------
# Entry point expected by the evaluator
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operation.

    Arguments:
        data (tuple): (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric)
            - weights: dict of model parameters (float32 tensors)
            - config: dict containing at least 'dim' and 'hidden_dim',
              optionally 'nomask' (bool, default True)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    # ------------------------------------------------------------------
    # Configuration
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm over the channel dimension (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (float32)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"]) # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Optional mask multiplication
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [B, N, N, 1] for broadcasting
        mask_fp = mask.to(torch.float32).unsqueeze(-1)
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 4. Gating (float32)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton batched GEMM
    # ------------------------------------------------------------------
    # Cast to FP16 just before the GEMM to exploit Tensor‑cores
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Rearrange to (B, hidden_dim, N, N)
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()
    right_perm = right_fp16.permute(0, 3, 2, 1).contiguous()  # transpose the two seq axes

    Bsz, H, N, _ = left_perm.shape
    # Merge batch and hidden dimensions to obtain independent GEMMs
    left_flat = left_perm.view(Bsz * H, N, N)
    right_flat = right_perm.view(Bsz * H, N, N)

    # Batched matrix multiplication: (B*H, N, N) @ (B*H, N, N) -> (B*H, N, N)
    out_flat = triton_batched_matmul(left_flat, right_flat)   # FP16

    # Restore original layout: (B, N, N, hidden_dim)
    out = out_flat.view(Bsz, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)   # back to FP32 for the final layers

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate multiplication, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate  # both are float32
    out = F.linear(out, weights["to_out.weight"])  # final projection to dim

    return out
