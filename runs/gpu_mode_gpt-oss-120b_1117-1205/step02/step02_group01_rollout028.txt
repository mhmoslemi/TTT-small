"""
Optimized Triton‑accelerated implementation of the outgoing Triangle
Multiplicative Update (TriMul) from AlphaFold3.

Algorithm
---------
1. Layer‑Norm the input (float32).
2. Cast to float16 for all heavy linear work.
3. Fuse the left/right projections into a single linear layer
   (weight = [left_proj; right_proj]).
4. Fuse the left/right gates into a single linear layer
   (weight = [left_gate; right_gate]) followed by sigmoid.
5. One Triton kernel fuses the three element‑wise multiplications:
        left = left_proj * left_gate * mask
        right = right_proj * right_gate * mask
6. Batched matrix multiplication (torch.bmm) implements the cubic
   contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   using half‑precision.
7. Layer‑Norm over the hidden dimension, final gating, and output
   projection back to the original dimension (float32).

The kernel only performs the elementwise fusion; all linear
operations use highly‑optimized PyTorch/cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fusion_kernel(
    left_raw_ptr, right_raw_ptr,          # input tensors (flattened)
    left_gate_ptr, right_gate_ptr,        # gate tensors (flattened)
    mask_ptr,                             # mask tensor (flattened, broadcast over hidden)
    left_out_ptr, right_out_ptr,          # output tensors (flattened)
    total,                                # total number of elements (B*N*N*H)
    BLOCK_SIZE: tl.constexpr,             # threads per block
    H: tl.constexpr,                      # hidden dimension
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # Load operands
    left_raw = tl.load(left_raw_ptr + offs, mask=mask)
    right_raw = tl.load(right_raw_ptr + offs, mask=mask)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask)

    # Broadcast mask over the hidden dimension
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Fuse multiplications
    tl.store(left_out_ptr + offs, left_raw * left_gate * m, mask=mask)
    tl.store(right_out_ptr + offs, right_raw * right_gate * m, mask=mask)

    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2) Cast to float16 for the heavy path
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Fused projections: left & right
    # ------------------------------------------------------------------
    left_w = weights["left_proj.weight"].to(device).to(torch.float16)   # (H, dim)
    right_w = weights["right_proj.weight"].to(device).to(torch.float16) # (H, dim)
    proj_w = torch.cat([left_w, right_w], dim=0)  # (2*H, dim)

    proj = F.linear(x_h, proj_w)                     # [B,N,N,2*H]
    left_raw = proj[..., :H]
    right_raw = proj[..., H:]

    # ------------------------------------------------------------------
    # 4) Fused gates: left_gate & right_gate
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)   # (H, dim)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16) # (H, dim)
    gate_w = torch.cat([left_gate_w, right_gate_w], dim=0)  # (2*H, dim)

    gate = torch.sigmoid(F.linear(x_h, gate_w))        # [B,N,N,2*H]
    left_gate = gate[..., :H]
    right_gate = gate[..., H:]

    # ------------------------------------------------------------------
    # 5) Mask handling
    # ------------------------------------------------------------------
    if mask_tensor is None or config.get("nomask", False):
        mask_h = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_h = mask_tensor.to(dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 6) Triton kernel: elementwise left = left_raw * left_gate * mask,
    #    right = right_raw * right_gate * mask
    # ------------------------------------------------------------------
    left_raw_f = left_raw.contiguous().view(-1)
    right_raw_f = right_raw.contiguous().view(-1)
    left_gate_f = left_gate.contiguous().view(-1)
    right_gate_f = right_gate.contiguous().view(-1)
    mask_f = mask_h.contiguous().view(-1)

    left_f = torch.empty_like(left_raw_f)
    right_f = torch.empty_like(right_raw_f)

    total = left_raw_f.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    _gate_mask_fusion_kernel[grid](
        left_raw_f,
        right_raw_f,
        left_gate_f,
        right_gate_f,
        mask_f,
        left_f,
        right_f,
        total,
        BLOCK_SIZE=BLOCK,
        H=H,
    )

    left = left_f.view(B, N, N, H)
    right = right_f.view(B, N, N, H)

    # ------------------------------------------------------------------
    # 7) Cubic contraction via batched GEMM (half‑precision)
    # ------------------------------------------------------------------
    # left: (B,N,N,H) -> (B,H,N,N)   rows = i, cols = k
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    # right: (B,N,N,H) -> (B,H,N,N)   rows = k, cols = j
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)

    batch = B * H
    left_mat = left_perm.view(batch, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(batch, N, N) # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N), half
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 8) Post‑processing: LayerNorm over hidden, final gate, output proj
    # ------------------------------------------------------------------
    # Convert to float32 for the final LayerNorm
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    out_norm = F.layer_norm(out_fp32, (H,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Final gating
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))      # (B,N,N,H)
    out_norm = out_norm * out_gate.to(torch.float32)

    # Output projection back to dim
    to_out_w = weights["to_out.weight"].to(device)          # (dim, H)
    output = F.linear(out_norm, to_out_w)  # (B,N,N,dim), float32

    return output
```
