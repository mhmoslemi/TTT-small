@triton.jit
def mul_gate_mask_twobranch_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr,
    mask_ptr, left_out_ptr, right_out_ptr,
    total, hidden_dim,
    BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=offs<total)
    a = tl.load(left_ptr + offs, mask=offs<total)
    b = tl.load(right_ptr + offs, mask=offs<total)
    gl = tl.load(left_gate_ptr + offs, mask=offs<total)
    gr = tl.load(right_gate_ptr + offs, mask=offs<total)
    left_out = a * gl * m
    right_out = b * gr * m
    tl.store(left_out_ptr + offs, left_out, mask=offs<total)
    tl.store(right_out_ptr + offs, right_out, mask=offs<total)

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    
    # 1 LayerNorm (input)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    
    # Cast to fp16
    x_fp16 = x.to(torch.float16)
    
    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    # shape (hidden_dim, dim) maybe default? In PyTorch's Linear weight uses (out_features, in_features)
    # So F.linear(input, weight) does input @ weight.T
    left = torch.nn.functional.linear(x_fp16, left_proj_weight.to(torch.float16))
    right = torch.nn.functional.linear(x_fp16, right_proj_weight.to(torch.float16))
    
    # Gate weights
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, left_gate_weight.to(torch.float16)))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, right_gate_weight.to(torch.float16)))

    output = torch.nn.functional.linear(out_norm, to_out_weight)  # weight shape (dim, hidden_dim)
    
    return output

"""
TriMul (outgoing) forward pass with a Triton‑accelerated elementwise
gate‑and‑mask fusion.  All LayerNorms, linear projections and the cubic
contraction are performed in PyTorch (mixed FP16/FP32).  The expensive
elementwise part

    left = left_proj(x) * left_gate(x) * mask
    right = right_proj(x) * right_gate(x) * mask

is fused into a single Triton kernel that processes the flattened
tensors in one launch.  The N³ contraction is expressed as a batched
matrix multiply (torch.bmm) which leverages cuBLAS.  This yields a fast
forward implementation of the outgoing TriMul operator.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def fused_gate_mask_kernel(
    left_ptr,        # fp16, [total]
    right_ptr,       # fp16, [total]
    left_gate_ptr,   # fp16, [total]
    right_gate_ptr,  # fp16, [total]
    mask_ptr,        # fp16, [B*N*N]
    left_out_ptr,    # fp16, [total]
    right_out_ptr,   # fp16, [total]
    total: tl.constexpr,          # total number of elements = B*N*N*H
    hidden_dim: tl.constexpr,     # H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total

    # index of the corresponding mask element (broadcast over H)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid)
    l = tl.load(left_ptr + offs, mask=valid)
    r = tl.load(right_ptr + offs, mask=valid)
    gl = tl.load(left_gate_ptr + offs, mask=valid)
    gr = tl.load(right_gate_ptr + offs, mask=valid)

    tl.store(left_out_ptr + offs,  l * gl * m, mask=valid)
    tl.store(right_out_ptr + offs, r * gr * m, mask=valid)


def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor: torch.Tensor [B, N, N, dim] (float32)
            mask_tensor  : torch.Tensor [B, N, N] (float32) or None
            weights_dict : dict with keys matching the TriMul module
            config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool}
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- 1. Input LayerNorm ----------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # [B,N,N,dim]

    # ---------- 2. Cast to FP16 for the heavy work ----------
    x_fp16 = x.to(torch.float16)

    # ---------- 3. Linear projections + gates (merged proj+gate) ----------
    # concatenate proj and gate weights to reduce the number of matmuls
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device).to(torch.float16)                      # (2*H, dim)
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device).to(torch.float16)                     # (2*H, dim)

    # combined matmuls
    left_combined = F.linear(x_fp16, left_concat_w)    # [B,N,N,2*H]
    right_combined = F.linear(x_fp16, right_concat_w)  # [B,N,N,2*H]

    # split into proj / gate and apply sigmoid to gates
    left_proj, left_gate = torch.split(left_combined, hidden_dim, dim=-1)
    right_proj, right_gate = torch.split(right_combined, hidden_dim, dim=-1)

    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    # out gate (separate linear)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))  # [B,N,N,H]

    # ---------- 4. Optional mask ----------
    if mask_tensor is None or nomask:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(device).to(torch.float16)

    # ---------- 5. Fuse gate & mask multiplication ----------
    left_flat = left_proj.contiguous().view(-1)
    right_flat = right_proj.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)
    mask_flat = mask_fp16.contiguous().view(-1)

    left_out_flat = torch.empty_like(left_flat)
    right_out_flat = torch.empty_like(right_flat)

    total = left_flat.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    fused_gate_mask_kernel[grid](
        left_flat,
        right_flat,
        left_gate_flat,
        right_gate_flat,
        mask_flat,
        left_out_flat,
        right_out_flat,
        total,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
    )

    # reshape back
    left = left_out_flat.view(B, N, N, hidden_dim)
    right = right_out_flat.view(B, N, N, hidden_dim)

    # ---------- 6. Cubic contraction via batched GEMM ----------
    # left: (B,N,N,H) -> (B,H,N,N) where rows i, cols k
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    # right: (B,N,N,H) -> (B,H,N,N) where rows k, cols j
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)

    LH = B * hidden_dim
    left_mat = left_perm.view(LH, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(LH, N, N) # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N), fp16

    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B,N,N,H]  fp16

    # ---------- 7. Post‑processing ----------
    # to_out_norm
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    out_fp32 = out.to(torch.float32)  # layernorm works in FP32 for stability
    out_norm = F.layer_norm(
        out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b
    )

    # final gating
    out_norm = out_norm * out_gate.to(torch.float32)

    # final linear projection
    to_out_w = weights["to_out.weight"].to(device)  # (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)  # [B,N,N,dim], float32

    return output
```
