"""
TriMul (outgoing) forward pass with a fused Triton mask kernel.
The heavy N³ contraction is performed as a batched matrix multiplication
using PyTorch's cuBLAS‑accelerated `torch.bmm` on FP16 tensors (Tensor Cores).
A small Triton kernel is used to apply the optional pairwise mask in‐place,
broadcasted over the hidden dimension.
All layer‑norms and the final projection run in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def apply_mask_kernel(
    src_ptr,               # input tensor (flattened)   [total]
    mask_ptr,              # mask tensor (flattened)    [mask_len]
    dst_ptr,               # output tensor (flattened)  [total]
    total,                 # number of elements in src
    hidden_dim: tl.constexpr,  # hidden dimension (inner stride)
    BLOCK: tl.constexpr,       # block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # Load source value (FP16)
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)

    # Compute mask index (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    # Apply mask
    out = src * m
    tl.store(dst_ptr + offs, out, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric)
            - weights_dict: mapping of parameter names to torch.Tensor
            - config_dict: must contain 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] FP32

    # -----------------------------------------------------------------
    # 2. Cast to FP16 for the heavy linear/gate ops
    # -----------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections (FP16)
    # -----------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # -----------------------------------------------------------------
    # 4. Optional mask (broadcast over hidden_dim) using a Triton kernel
    # -----------------------------------------------------------------
    if not nomask:
        # Prepare mask in FP16 and flatten
        mask_fp16 = mask.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)

        total = left.numel()  # B * N * N * hidden_dim
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)

        # In‑place mask application for left and right
        apply_mask_kernel[grid](
            left, mask_flat, left, total,
            hidden_dim=hidden_dim,
            BLOCK=BLOCK,
        )
        apply_mask_kernel[grid](
            right, mask_flat, right, total,
            hidden_dim=hidden_dim,
            BLOCK=BLOCK,
        )

    # -----------------------------------------------------------------
    # 5. Gating (FP16) and apply
    # -----------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6. Core contraction: batched GEMM using torch.bmm (FP16 inputs, FP32 accumulation)
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    -> left (i,k) @ right (k,j) for each (b,d)
    # -----------------------------------------------------------------
    # Rearrange tensors to (B, H, N, N) where the last two dims are (i,k) for left
    # and (k,j) for right.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)  (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)  (k,j)

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)   # (B*H, N, N)   (i,k)
    right_flat = right_perm.view(B * H, N, N)  # (B*H, N, N)   (k,j)

    # Batched matrix multiplication (Tensor Core accelerated)
    out_flat = torch.bmm(left_flat, right_flat)  # (B*H, N, N), FP16

    # Reshape back to (B, N, N, H) and cast to FP32 for the remaining ops
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # -----------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final projection (FP32)
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
