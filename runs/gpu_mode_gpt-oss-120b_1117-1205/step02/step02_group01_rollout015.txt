import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def mul_gate_mask_kernel(out_ptr, a_ptr, gate_ptr, mask_ptr, total, BLOCK_SIZE: tl.constexpr, H: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    a = tl.load(a_ptr + offs, mask=mask)
    g = tl.load(gate_ptr + offs, mask=mask)
    # Compute mask index (across batch-seq dims)
    idx_mask = offs // H
    m = tl.load(mask_ptr + idx_mask, mask=mask)
    tl.store(out_ptr + offs, a * g * m, mask=mask)
```

    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"].to(device), bias=weights["norm.bias"].to(device))
    # Convert to half
    x_fp16 = x.to(torch.float16)
    # Flatten for linear
    x_flat = x_fp16.view(-1, dim) # shape (B*N*N, dim)
    # Prepare combined weight
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    # Combine left + left_gate weights
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)  # (2*hid, dim)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*hid, dim)

    # Compute combined linear results
    left_comb_out = F.linear(x_flat, left_comb_w)  # (BNN, 2*hid)
    right_comb_out = F.linear(x_flat, right_comb_w)  # (BNN, 2*hid)

    # Split
    left, left_gate = torch.split(left_comb_out, hidden_dim, dim=1)  # each (BNN, hid)
    right, right_gate = torch.split(right_comb_out, hidden_dim, dim=1)

    # Out gate
    out_gate_pre = F.linear(x_flat, out_gate_w)  # (BNN, hid)
    out_gate = torch.sigmoid(out_gate_pre)

    # Apply sigmoid to left_gate, right_gate
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    # Reshape to 4D
    left = left.view(B, N, N, hidden_dim)
    left_gate = left_gate.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)
    right_gate = right_gate.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # Process mask

    # Flatten for kernel
    total = left.numel()
    left_flat = left.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_flat = right.contiguous().view(-1)

        right_gate_flat,

    to_out_w = weights["to_out.weight"].to(device).to(torch.float32)

"""
Highly‑optimized forward pass of the TriMul (outgoing) module.

Algorithm
---------

1. **LayerNorm** on the input (float32 for numerical stability).
2. **Mixed‑precision linear projections**:
   - Fuse `left_proj` with `left_gate` and `right_proj` with `right_gate`
     into two `F.linear` calls by concatenating the weight matrices.
   - A third `F.linear` yields the `out_gate`.
3. **Sigmoid** on the three gate tensors.
4. **Mask‑and‑gate fusion** using a tiny Triton kernel that computes
   `out = tensor * gate * mask` for the left and right branches.
5. **Cubic contraction** `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`
   is performed as a batched matrix multiplication (`torch.bmm`) after
   suitable transpositions – the operation reduces to `L @ Rᵀ` per hidden
   dimension and exploits tensor‑core acceleration.
6. **Post‑processing**:
   - LayerNorm on the hidden dimension.
   - Element‑wise multiplication with `out_gate`.
   - Final linear projection back to `dim`.

All heavy compute is in float16; final output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def mul_gate_mask_kernel(
    out_ptr,          # flattened output pointer (half)
    a_ptr,            # flattened tensor pointer (half)
    gate_ptr,         # flattened gate pointer (half)
    mask_ptr,         # flattened mask pointer (half) – broadcast over H
    total,            # total number of elements (int)
    BLOCK_SIZE: tl.constexpr,
    H: tl.constexpr,  # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    a = tl.load(a_ptr + offs, mask=mask)        # tensor value
    g = tl.load(gate_ptr + offs, mask=mask)     # gate value
    # mask index = element // hidden_dim (broadcast over H)
    idx_mask = offs // H
    m = tl.load(mask_ptr + idx_mask, mask=mask)
    tl.store(out_ptr + offs, a * g * m, mask=mask)

    """
    Forward pass of the TriMul (outgoing) operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

            - mask_tensor : torch.Tensor [B, N, N] (float32) or None

    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)

    # -----------------------------------------------------------------

    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)   # (H, dim)

    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16) # (H, dim)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)     # (H, dim)

    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2H, dim)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2H, dim)

    # compute fused linear outputs
    left_comb_out = F.linear(x_flat, left_comb_w)    # [(B*N*N), 2H]
    right_comb_out = F.linear(x_flat, right_comb_w)  # [(B*N*N), 2H]

    right, right_gate = torch.split(right_comb_out, hidden_dim, dim=1)

    left_gate = left_gate.view(B, N, N, hidden_dim)

    right_gate = right_gate.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # 4. Fuse mask, gate and projection via Triton kernel
    # -----------------------------------------------------------------
    if mask_tensor is None:

    left_flat = left.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_flat = right.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)
    mask_flat = mask_fp16.contiguous().view(-1)         # B*N*N

    right_out_flat = torch.empty_like(right_flat)

        mask_flat,
        total,
        BLOCK_SIZE=BLOCK_SIZE,
        H=hidden_dim,
    )
    mul_gate_mask_kernel[grid](
        right_out_flat,
        right_flat,
        right_gate_flat,
        mask_flat,
        total,
        BLOCK_SIZE=BLOCK_SIZE,
        H=hidden_dim,
    )

    # reshape back
    left = left_out_flat.view(B, N, N, hidden_dim)
    right = right_out_flat.view(B, N, N, hidden_dim)

    # -----------------------------------------------------------------
    # 5. Cubic contraction via batched GEMM (L @ Rᵀ per hidden dim)
    # -----------------------------------------------------------------
    # (B, H, N, N) layout for both tensors
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()        # (B, H, N, N) -> (k, j)

    LH = B * hidden_dim
    left_mat = left_perm.view(LH, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(LH, N, N) # (B*H, N, N)

    # batched matrix multiplication (fp16)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)

    # restore original layout [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Post‑processing (LayerNorm, final gate, output projection)
    # -----------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float32)
    out_norm = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    out_norm = out_norm * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float32)  # (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)  # final [B, N, N, dim]

    return output
```
