"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm:
1. Layer‑norm the input (float32).
2. Linear projections → `left` & `right` (float16, bias‑free).
3. Optional mask applied (broadcast over hidden dimension).
4. Gating linear layers with sigmoid (float16) and element‑wise multiplication.
5. Core contraction:
   out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   For each (b,d) this is a matrix multiplication A @ Bᵀ where
   A = left_{b,:, :, d} (N×N) and B = right_{b,:, :, d} (N×N).
   We flatten (b,d) as a batch dimension and invoke a Triton batched GEMM.
6. Layer‑norm over the hidden dimension, multiply by `out_gate`, final linear projection
   back to the original dimension (float32).

All heavy matrix multiplications are executed in half‑precision with fp32 accumulation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton batched GEMM kernel (A @ Bᵀ) for tensors of shape (batch, M, K) and (batch, K, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Program IDs
    batch_id = tl.program_id(0)          # batch dimension (B*H)
    pid_m = tl.program_id(1)             # row tile
    pid_n = tl.program_id(2)             # column tile

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch entry
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        # Load B tile (K x N) - B is already transposed layout for A @ Bᵀ
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # fp16 * fp16 → fp32

    # Store result (fp16)
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def triton_batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM C = A @ B   (A: (batch, M, K), B: (batch, K, N))
    All tensors must be contiguous torch.float16.
    Returns C of shape (batch, M, N) as torch.float16.
    """
    assert A.dtype == B.dtype == torch.float16
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides (in elements)
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tile sizes tuned for Ampere B200
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor:  torch.Tensor [B, N, N] (bool or numeric)
            - weights_dict: mapping from parameter name to torch.Tensor
            - config_dict: contains 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim]  float32

    # --------------------------------------------------------------
    # 2. Linear projections (bias=False) → fp16 for heavy ops
    # --------------------------------------------------------------
    # Convert weights once to fp16
    left_proj_w_fp16 = weights["left_proj.weight"].to(torch.float16, non_blocking=True)
    right_proj_w_fp16 = weights["right_proj.weight"].to(torch.float16, non_blocking=True)

    # Project
    left = F.linear(x.to(torch.float16), left_proj_w_fp16)   # [B, N, N, hidden_dim] fp16
    right = F.linear(x.to(torch.float16), right_proj_w_fp16) # [B, N, N, hidden_dim] fp16

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [B, N, N, 1] and cast to fp16
        mask_fp = mask.unsqueeze(-1).to(torch.float16, non_blocking=True)
        left = left * mask_fp
        right = right * mask_fp

    # --------------------------------------------------------------
    # 4. Gates (sigmoid) – compute in fp16 for consistency
    # --------------------------------------------------------------
    left_gate_w_fp16 = weights["left_gate.weight"].to(torch.float16, non_blocking=True)
    right_gate_w_fp16 = weights["right_gate.weight"].to(torch.float16, non_blocking=True)
    out_gate_w_fp16 = weights["out_gate.weight"].to(torch.float16, non_blocking=True)

    left_gate = F.linear(x.to(torch.float16), left_gate_w_fp16).sigmoid()   # [B,N,N,hidden_dim] fp16
    right_gate = F.linear(x.to(torch.float16), right_gate_w_fp16).sigmoid()
    out_gate = F.linear(x.to(torch.float16), out_gate_w_fp16).sigmoid()     # fp16

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction via Triton batched GEMM (outgoing TriMul)
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    For each (b,d) this is A @ Bᵀ where A = left[..., :, d] and B = right[..., :, d]
    # --------------------------------------------------------------
    # Rearrange tensors to (B, H, N, N) where H = hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, K) where K=N
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, K, N)

    B_, H_, N_, _ = left_perm.shape  # B_, H_, N_, K (K==N_)
    # Flatten batch and hidden dims for batched GEMM
    left_flat = left_perm.view(B_ * H_, N_, N_)   # (B*H, M=N, K=N)
    right_flat = right_perm.view(B_ * H_, N_, N_) # (B*H, K=N, N=N)

    # Perform batched matrix multiplication using Triton (fp16 inputs → fp16 output)
    out_flat = triton_batched_matmul(left_flat, right_flat)  # (B*H, N, N) fp16

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B_, H_, N_, N_).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden_dim] fp16

    # --------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection (float32)
    # --------------------------------------------------------------
    # Convert to float32 for the final stages
    out = out.to(torch.float32)

    # LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    # Apply out_gate (cast to float32)
    out = out * out_gate.to(torch.float32)

    # Final linear projection back to dim
    out = F.linear(out, weights["to_out.weight"])

    return out
