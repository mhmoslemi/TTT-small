def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B,N,N,dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # 1. LayerNorm
    # Convert weight and bias to appropriate dtype
    norm_w = weights["norm.weight"].to(device, dtype=torch.float16)
    norm_b = weights["norm.bias"].to(device, dtype=torch.float16)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # fp16

    # 2. Linear proj (no bias) using half weights
    left_proj_w = weights["left_proj.weight"].to(device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device, dtype=torch.float16)

    # Use torch.nn.functional.linear: F.linear(x, weight) expects weight shape (out_features, in_features); we have left_proj.weight shape (hidden_dim, dim). So F.linear(x, left_proj_w) reduces last dim (dim) to hidden_dim.

    left = F.linear(x, left_proj_w)  # shape [B,N,N,hidden_dim], half
    right = F.linear(x, right_proj_w)

    # 3. Gates:
    left_gate_w = weights["left_gate.weight"].to(device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device, dtype=torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # 4. Mask handling
    if mask_tensor is None:

    else:
        mask_fp16 = mask_tensor.to(dtype=torch.float16, device=device)

    # 5. Fuse elementwise multiply of left/right with gates and mask using Triton

def fused_mul_gate_mask_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    total: tl.int64,
    BLOCK_SIZE: tl.constexpr,
    H: tl.constexpr,

"""
Fast TriMul (outgoing) forward pass.

The heavy computation consists of:
  1. LayerNorm on the input (float16).
  2. Linear projections to hidden dimension (float16, cuBLAS).
  3. A fused Triton kernel that applies the three element‑wise
     operations   out = tensor * gate * mask   for both left and right branches.
  4. The cubic “tri‑multiplication” is realised as a batched GEMM:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
     which becomes B*hidden_dim independent matrix‑multiplications
     of size N×N and is executed by torch.bmm (cuBLAS).
  5. LayerNorm on the hidden dimension, a final gating, and a linear
     projection back to the original dimension (all in float16).

Only the element‑wise mask/gate fusion is written in Triton; all
large matrix multiplications are delegated to cuBLAS, which is much
faster than a pure‑Triton implementation for the considered shapes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def fused_mul_gate_mask_kernel(
    left_ptr,                # float16*  (B,N,N,hidden)
    left_gate_ptr,           # float16*  (B,N,N,hidden)
    right_ptr,               # float16*  (B,N,N,hidden)
    right_gate_ptr,          # float16*  (B,N,N,hidden)
    mask_ptr,                # float16*  (B,N,N)
    total: tl.int64,         # total number of elements (= B*N*N*hidden)
    BLOCK_SIZE: tl.constexpr,
    H: tl.constexpr,         # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # Load the four tensors
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    # Broadcast mask (shape without hidden)
    mask_idx = offs // H                      # integer divide
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))

    # In‑place fused multiplication
    tl.store(left_ptr + offs, left * left_gate * m, mask=mask)
    tl.store(right_ptr + offs, right * right_gate * m, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
            mask_tensor  : torch.Tensor of shape [B, N, N] (float32) or None
            weights_dict : dict with the model parameters (see problem statement)
            config_dict  : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1. Layer‑norm on the input (use half precision for speed)
    # --------------------------------------------------------------
    x = input_tensor.to(torch.float16)

    norm_w = weights["norm.weight"].to(device, dtype=torch.float16)
    norm_b = weights["norm.bias"].to(device, dtype=torch.float16)
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------
    # 2. Linear projections to the hidden dimension (no bias)
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device, dtype=torch.float16)

    left = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 3. Gating (sigmoid) – still in half precision
    # --------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device, dtype=torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 4. Mask preparation (half‑precision broadcast)
    # --------------------------------------------------------------
    if mask_tensor is None:
        mask = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask = mask_tensor.to(dtype=torch.float16, device=device)

    # --------------------------------------------------------------
    # 5. Fuse element‑wise mask, left‑gate and right‑gate in Triton
    # --------------------------------------------------------------
    total = left.numel()                     # B * N * N * hidden_dim
    BLOCK = 8192                              # chosen to give good occupancy
    grid = (triton.cdiv(total, BLOCK),)

    fused_mul_gate_mask_kernel[grid](
        left_ptr=left,
        left_gate_ptr=left_gate,
        right_ptr=right,
        right_gate_ptr=right_gate,
        mask_ptr=mask,
        total=total,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
    )
    # After the kernel `left` and `right` contain the masked & gated values.

    # --------------------------------------------------------------
    # 6. Cubic contraction via a batched GEMM (cuBLAS)
    # --------------------------------------------------------------
    # Rearrange tensors so that each (batch, hidden) pair becomes a
    # separate matrix multiplication of shape (N, N).
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, hidden, N, N)

    LH = B * hidden_dim
    left_mat = left_perm.view(LH, N, N)    # (B*hidden, N, N)
    right_mat = right_perm.view(LH, N, N)  # (B*hidden, N, N)

    # Batched matrix multiplication: (i,k) @ (k,j) -> (i,j)
    out_mat = torch.bmm(left_mat, right_mat)   # (B*hidden, N, N), half

    # Restore the original layout
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7. Post‑processing: hidden‑dim LayerNorm, final gating and projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype=torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply the output gate (still half‑precision)
    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(device, dtype=torch.float16)
    out = F.linear(out, to_out_w)    # (B, N, N, dim), half

    # Cast back to float32 for the final result
    return out.to(torch.float32)

"""
Fast TriMul (outgoing) forward pass.

The core of the operator is a cubic contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which we realise as a batched matrix‑multiply (torch.bmm).  To avoid
multiple launches we fuse the three element‑wise operations
    tensor * gate * mask
for the left and right branches in a single Triton kernel.
All heavy linear work (projections, batch GEMM, final projection) is
performed with cuBLAS in float16; the final result is cast back to
float32.  At least one Triton kernel is used, as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def fused_mul_gate_mask_kernel(
    left_ptr,                # float16*  (B, N, N, hidden)
    left_gate_ptr,           # float16*  (B, N, N, hidden)
    right_ptr,               # float16*  (B, N, N, hidden)
    right_gate_ptr,          # float16*  (B, N, N, hidden)
    mask_ptr,                # float16*  (B, N, N)
    total: tl.int64,         # total elements = B*N*N*hidden
    BLOCK_SIZE: tl.constexpr,
    H: tl.constexpr,         # hidden dimension
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # Load the four tensors (half‑precision)
    left      = tl.load(left_ptr + offs,      mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right     = tl.load(right_ptr + offs,     mask=mask, other=tl.float16(0.0))
    right_gate= tl.load(right_gate_ptr + offs,mask=mask, other=tl.float16(0.0))

    # Broadcast mask (shape without hidden) over the hidden dimension
    mask_idx = offs // H                       # integer division
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))

    # In‑place fused multiplication
    tl.store(left_ptr + offs,  left * left_gate * m,  mask=mask)
    tl.store(right_ptr + offs, right * right_gate * m, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, dim] (float32)
            mask_tensor  : torch.Tensor [B, N, N] (float32) or None
            weights_dict : dict containing the model parameters:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            config_dict  : {"dim": int, "hidden_dim": int}
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the input (float16 for speed)
    # ------------------------------------------------------------------
    x = input_tensor.to(torch.float16)

    norm_w = weights["norm.weight"].to(device, dtype=torch.float16)
    norm_b = weights["norm.bias"].to(device, dtype=torch.float16)
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (no bias)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device, dtype=torch.float16)

    left  = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim], half
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid) – stays in half precision
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device, dtype=torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device, dtype=torch.float16)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 4. Mask (float16, broadcasted over hidden)
    # ------------------------------------------------------------------
    if mask_tensor is None:
        mask = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask = mask_tensor.to(dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 5. Fuse (tensor * gate * mask) for left and right in a single Triton kernel
    # ------------------------------------------------------------------
    total = left.numel()                     # B * N * N * hidden_dim
    BLOCK = 8192                              # good occupancy for large tensors
    grid = (triton.cdiv(total, BLOCK),)

    fused_mul_gate_mask_kernel[grid](
        left_ptr=left,
        left_gate_ptr=left_gate,
        right_ptr=right,
        right_gate_ptr=right_gate,
        mask_ptr=mask,
        total=total,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
    )
    # left and right now hold the masked & gated values (in‑place).

    # ------------------------------------------------------------------
    # 6. Cubic contraction via batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    # Rearrange to (B*hidden, N, N) where each batch performs a matmul
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, hidden, N, N)

    LH = B * hidden_dim
    left_mat  = left_perm.view(LH, N, N)   # (B*hidden, N, N)
    right_mat = right_perm.view(LH, N, N)  # (B*hidden, N, N)

    out_mat = torch.bmm(left_mat, right_mat)   # (B*hidden, N, N), half

    # Restore [B, N, N, hidden] layout
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Post‑processing: hidden‑dim LayerNorm, final gate, output projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype=torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply the output gate (still half‑precision)
    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(device, dtype=torch.float16)  # (dim, hidden)
    out = F.linear(out, to_out_w)    # [B, N, N, dim], half

    # Cast back to float32 for the final result
    return out.to(torch.float32)
