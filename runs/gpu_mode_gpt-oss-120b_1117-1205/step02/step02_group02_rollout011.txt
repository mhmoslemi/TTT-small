def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Computes C = A @ B for a batch of matrices.
    A : (batch, M, K)
    B : (batch, K, N)
    C : (batch, M, N)
    All tensors are stored in row‑major order.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K sized chunks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load tile of A [BLOCK_M, BLOCK_K]
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        # Load tile of B [BLOCK_K, BLOCK_N]
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Write back the result (cast to fp16)
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def triton_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B for FP16 tensors.
    A : (batch, M, K)   fp16
    B : (batch, K, N)   fp16
    Returns C : (batch, M, N) fp16
    """
    assert A.dtype == B.dtype == torch.float16
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides in elements
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tile sizes (tuned for RTX‑A6000 / B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator with Triton‑accelerated core.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor:  torch.Tensor [B, N, N] (bool or numeric)
            - weights_dict: mapping from parameter name to torch.Tensor
            - config_dict: must contain keys 'dim' and 'hidden_dim';
                           optional key 'nomask' (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )   # [B, N, N, dim], float32

    # Cast to fp16 for the heavy part
    x_fp16 = x.half()               # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – fp16
    # ------------------------------------------------------------------
    left = F.linear(
        x_fp16,
        weights["left_proj.weight"].to(torch.float16)
    )   # [B, N, N, hidden_dim], fp16

    right = F.linear(
        x_fp16,
        weights["right_proj.weight"].to(torch.float16)
    )   # [B, N, N, hidden_dim], fp16

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)    # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – fp16
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_fp16,
        weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()   # [B, N, N, hidden_dim]

    right_gate = F.linear(
        x_fp16,
        weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()

    out_gate = F.linear(
        x_fp16,
        weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()   # [B, N, N, hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched GEMM per hidden dimension:
    #        left_perm : (B, H, N, N)   -> rows i, cols k
    #        right_perm: (B, H, N, N)   -> rows k, cols j
    #    Then out_perm = left_perm @ right_perm  (batch = B*H)
    # ------------------------------------------------------------------
    #   Rearrange tensors to (B, H, N, N) where
    #   left_perm[b,h,i,k] = left[b,i,k,h]
    #   right_perm[b,h,k,j] = right[b,j,k,h]
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)

    B_, H_, N_, _ = left_perm.shape

    # Flatten batch and hidden dimensions for the batched GEMM
    left_flat = left_perm.view(B_ * H_, N_, N_)   # (B*H, M=N, K=N)
    right_flat = right_perm.view(B_ * H_, N_, N_) # (B*H, K=N, N=N)

    # Batched matrix multiplication via Triton
    out_flat = triton_matmul(left_flat, right_flat)  # (B*H, N, N), fp16

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B_, H_, N_, N_).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)
    out = out.float()   # convert to fp32 for the remaining layers

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.float()
    out = F.linear(
        out,
        weights["to_out.weight"]
    )   # final projection to dim, bias is None

    return out
