"""
Optimized outgoing TriMul forward pass.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32).
2. Cast to fp16 and compute the four linear projections (left, right,
   left_gate, right_gate, out_gate) in half precision.
3. Apply optional mask (broadcast over hidden_dim) and gating.
4. Core contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication (B·H, N, N) @ (B·H, N, N)^T.
   Implemented with a custom Triton kernel (fp16 × fp16 → fp32 accumulation).
5. Convert result to float32, apply output LayerNorm, scale by out_gate,
   and a final linear projection back to the original channel dimension.

All heavy arithmetic (steps 4‑5) is done in fp16/fp32 to reduce memory
traffic; the kernel is tiled with BLOCK_M/N/K = 128/128/32 which fits well
on Hopper B200 GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------
# Triton batched GEMM kernel (C = A @ B^T) for the core contraction
# ------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Computes C = A @ B^T for a batch of matrices.
    A: (batch, M, K)   fp16
    B: (batch, K, N)   fp16   (already transposed, i.e. original B^T)
    C: (batch, M, N)   fp16   (fp32 accumulator, cast back to fp16)
    """
    batch_id = tl.program_id(0)      # batch dimension (B*H)
    pid_m = tl.program_id(1)         # block row
    pid_n = tl.program_id(2)         # block col

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the beginning of this batch
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Load A tile (M x K_tile)
        a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        # Load B tile (K_tile x N) – note B is already transposed
        b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # fp16×fp16 → fp32

    # Write back C tile
    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def triton_batched_gemm(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Wrapper around the Triton kernel.
    a: (batch, M, K)  fp16, contiguous
    b: (batch, K, N)  fp16, contiguous (already transposed)
    Returns c: (batch, M, N) fp16, contiguous
    """
    assert a.is_contiguous() and b.is_contiguous()
    assert a.dtype == b.dtype == torch.float16
    B, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    c = torch.empty((B, M, N), dtype=torch.float16, device=a.device)

    # strides in *elements*
    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_k, stride_b_n = b.stride()
    stride_c_batch, stride_c_m, stride_c_n = c.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (B, triton.cdiv(M, BLOCK_M), triton.cdiv(N, BLOCK_N))
    batched_gemm_kernel[grid](
        a, b, c,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) or None
            - weights_dict: mapping of parameter names to torch.Tensor (float32)
            - config_dict: must contain 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----------------------------------------------
    # 1. Input LayerNorm (float32)
    # ----------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim] float32

    # ----------------------------------------------
    # 2. Cast to fp16 for heavy ops
    # ----------------------------------------------
    x_fp16 = x.to(torch.float16)

    # Pre‑cast weight matrices to fp16 (once)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # ----------------------------------------------
    # 3. Linear projections (fp16)
    # ----------------------------------------------
    left  = F.linear(x_fp16, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x_fp16, right_proj_w)  # [B, N, N, hidden_dim]

    # ----------------------------------------------
    # 4. Optional mask (broadcast over hidden_dim)
    # ----------------------------------------------
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)   # [B, N, N, 1]
        left  = left * mask_fp
        right = right * mask_fp

    # ----------------------------------------------
    # 5. Gating (fp16) and apply
    # ----------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()   # [B, N, N, hidden_dim]

    left  = left * left_gate
    right = right * right_gate

    # ----------------------------------------------
    # 6. Core contraction via Triton batched GEMM
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ----------------------------------------------
    # Rearrange to (batch, head, N, N) then flatten batch*head
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N) – (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) – (k,j) already transposed

    B, H, N, _ = left_perm.shape
    left_flat  = left_perm.view(B * H, N, N)    # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N)   # (B*H, N, N) – ready as B^T

    # Batched GEMM: (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    out_flat = triton_batched_gemm(left_flat, right_flat)   # fp16

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ----------------------------------------------
    # 7. Output LayerNorm, out‑gate, final projection (float32)
    # ----------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"])

    return out
