```
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,
):
    pid = tl.program_id(0)
    offset = pid * BLOCK + tl.arange(0, BLOCK)
    mask_off = offset // H  # integer division
    valid = offset < total_elements

    # Load
    left = tl.load(left_ptr + offset, mask=valid, other=tl.float16(0.0))
    right = tl.load(right_ptr + offset, mask=valid, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offset, mask=valid, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offset, mask=valid, other=tl.float16(0.0))

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + mask_off, mask=valid, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offset, left, mask=valid)
    tl.store(right_ptr + offset, right, mask=valid)
```

"""
Outgoing TriMul (AlphaFold3) – fast forward pass.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Fuse the two projection linear layers (left/right) into one GEMM and
   split the result. Same for the two gating linear layers.
3. Apply mask (if present) and the sigmoid gates in a single Triton kernel
   (in‑place on the projected tensors).
4. Compute the O(N³) contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is implemented as a batched matrix multiplication:
   left_permuted  : (B, H, N, N)   → (i,k)
   right_permuted : (B, H, N, N)   → (k,j)  (right transposed)
   out = bmm(left_permuted, right_permuted)
5. Apply the output‑norm, out‑gate, and the final linear projection.
6. Return a float‑32 tensor.

Only the element‑wise mask/gate fusion is done in Triton; the heavy
linear and GEMM operations rely on cuBLAS / torch kernels which are
already highly optimized.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # half*
    right_ptr,              # half*
    left_gate_ptr,          # half*
    right_gate_ptr,         # half*
    mask_ptr,               # half* (broadcasted)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask is present else 0
    H: tl.constexpr,          # hidden_dim
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask_offs = offs // H               # one mask per H entries
    valid = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=valid, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=valid, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=tl.float16(0.0))

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + mask_offs, mask=valid, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place: left = left * left_gate * mask
             right = right * right_gate * mask
    All tensors must be contiguous half‑precision.
    """
    total = left.numel()
    BLOCK = 1024                     # max threads per triton block
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy; never read when has_mask==0

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # operation is in‑place; no return


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool / numeric) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm (float32 → half)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float16, device=device)
    norm_bias = weights["norm.bias"].to(torch.float16, device=device)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Combined linear projections for left/right and for left_gate/right_gate
    # ------------------------------------------------------------------
    # concatenate projection weights
    left_proj_w = weights["left_proj.weight"].to(torch.float16, device=device)
    right_proj_w = weights["right_proj.weight"].to(torch.float16, device=device)
    proj_weight = torch.cat([left_proj_w, right_proj_w], dim=0)          # (2*H, dim)

    # concatenate gate weights
    left_gate_w = weights["left_gate.weight"].to(torch.float16, device=device)
    right_gate_w = weights["right_gate.weight"].to(torch.float16, device=device)
    gate_weight = torch.cat([left_gate_w, right_gate_w], dim=0)         # (2*H, dim)

    # project
    proj = F.linear(x, proj_weight)                                     # [B,N,N,2H]
    left, right = torch.split(proj, hidden_dim, dim=-1)                 # each [B,N,N,H]

    gate = F.linear(x, gate_weight)                                     # [B,N,N,2H]
    left_gate, right_gate = torch.split(gate, hidden_dim, dim=-1)
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()

    # out‑gate (separate)
    out_gate_w = weights["out_gate.weight"].to(torch.float16, device=device)
    out_gate = F.linear(x, out_gate_w).sigmoid()                        # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3. Fuse mask and gates (Triton)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16, device=device).unsqueeze(-1).contiguous()
        _apply_gate_and_mask(
            left.view(-1), right.view(-1),
            left_gate.view(-1), right_gate.view(-1),
            mask_h.view(-1),
            hidden_dim,
        )
        # reshape back (view is cheap as we used the same memory)
        left = left.view_as(left)
        right = right.view_as(right)
    else:
        # no mask → simple elementwise multiplication
        left = left * left_gate
        right = right * right_gate

    # free intermediate gate tensors early
    del left_gate, right_gate, gate, proj, gate_weight, proj_weight

    # ------------------------------------------------------------------
    # 4. Core O(N³) contraction via batched GEMM
    # ------------------------------------------------------------------
    # bring hidden dimension to batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N) – (i,k)
    # transpose the second pair of axes of right to get (k,j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # (B, H, N, N) – (k,j)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # batched matrix multiply (half×half → float32 accumulation)
    out_mat = torch.bmm(left_mat, right_mat)                    # (B*H, N, N)

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Output norm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16, device=device)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16, device=device)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate                       # elementwise gating (half)

    to_out_w = weights["to_out.weight"].to(torch.float16, device=device)
    out = F.linear(out, to_out_w)               # (B,N,N,dim) half

    # ------------------------------------------------------------------
    # 6. Cast back to float32 as required by the API
    # ------------------------------------------------------------------
    return out.to(torch.float32)
