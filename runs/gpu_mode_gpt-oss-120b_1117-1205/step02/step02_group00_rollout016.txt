def custom_kernel(data):
    x, mask, weights, config = data
    device = x.device
    bs, N, _, dim = x.shape
    hidden = config["hidden_dim"]
    # Apply input LayerNorm (float32) and convert to half
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x_h = x_norm.to(torch.float16)

    # Combine weights for left_proj, right_proj, left_gate, right_gate
    # Each weight: shape (hidden, dim)
    # We'll stack them: total_weight = torch.cat([left_proj_weight, right_proj_weight, left_gate_weight, right_gate_weight], dim=0) shape (4*hidden, dim)
    # We'll also include out_gate weight separately.

    total_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # shape (4*hidden, dim)

    # Reshape x_h to (bs*N*N, dim)
    x_flat = x_h.view(-1, dim)  # (bs*N*N, dim)
    # Perform single linear
    combined_out = torch.nn.functional.linear(x_flat, total_weight)  # shape (bs*N*N, 4*hidden)
    combined_out = combined_out.view(bs, N, N, 4 * hidden)  # shape (bs,N,N,4*hidden)

    # Split
    left = combined_out[..., 0:hidden]
    right = combined_out[..., hidden:2*hidden]
    left_gate_pre = combined_out[..., 2*hidden:3*hidden]
    right_gate_pre = combined_out[..., 3*hidden:4*hidden]

    # Compute out_gate separately:
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.nn.functional.linear(x_flat, out_gate_weight).view(bs, N, N, hidden)

    # Convert mask to half and broadcast
    if mask is not None: mask_h = mask.unsqueeze(-1).to(torch.float16) else: mask_h = None

    # Apply gating and mask using Triton kernel
    # We'll implement kernel that takes left, right, left_gate_pre, right_gate_pre, mask_h(optional) and outputs:
    # left = left * sigmoid(left_gate_pre) * mask_h (if exists)
    # right similarly.
    # out_gate = sigmoid(out_gate) (does not need mask)
    # We could just compute sigmoid in torch, not in kernel.
    
    # We'll compute left_gate = torch.sigmoid(left_gate_pre) (half).
    # Same for right_gate.
    # Then left = left * left_gate; etc.
    
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate)

    # Apply mask if needed: left = left * mask_h; right = right * mask_h ; mask broadcast
    if mask_h is not None:
        left = left * left_gate * mask_h
        right = right * right_gate * mask_h
    else:
        left = left * left_gate
        right = right * right_gate

    # Now compute contraction via batched gemm
    # Permute to (bs, hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2)  # (bs, hidden, N, N)
    right_perm = right.permute(0, 3, 2, 1)  # (bs, hidden, N, N)
    B, H, N, _ = left_perm.shape
    left_mat = left_perm.reshape(B*H, N, N)

    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    # Cast to float32
    return out.to(torch.float32)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _mask_gate_kernel(left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr, total_elements, hidden, BLOCK: tl.constexpr, HAS_MASK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # compute offset for mask if present
    # mask length = total_elements // hidden (since mask broadcast across hidden)
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # mask index is offs // hidden
        mask_idx = offs // hidden
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)  # if mask missing, treat as 1?
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

"""
Optimized outgoing TriMul forward pass.

Key ideas:
1. Fuse the five linear layers (left/right projections + left/right gates)
   into a single matrix multiplication by concatenating the weight matrices.
   This drastically reduces memory traffic (the input tensor is read only once).
2. Fuse mask and gating element‑wise multiplication into a tiny Triton kernel.
   The kernel works on the flattened tensors and broadcasts the mask over
   the hidden dimension.
3. Perform the N³ contraction with a batched GEMM (torch.bmm) on half‑precision
   tensors – the most efficient path on modern GPUs.
4. All remaining operations (layer‑norms, final projection) are kept in
   half‑precision; the final result is cast back to float32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,               # half*: left tensor
    right_ptr,              # half*: right tensor
    left_gate_ptr,          # half*: left gate tensor
    right_gate_ptr,         # half*: right gate tensor
    mask_ptr,               # half*: mask tensor (flattened) – valid only if HAS_MASK==1
    total_elements: tl.int32,
    hidden: tl.int32,       # hidden_dim (size of innermost channel)
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr   # 1 if mask is present, 0 otherwise
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load data
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # Each mask element corresponds to `hidden` consecutive channel entries
        mask_idx = offs // hidden
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store results back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_mask_and_gate(left, right, left_gate, right_gate, mask, hidden):
    """
    In‑place elementwise fusion:
        left  = left * left_gate * mask
        right = right * right_gate * mask
    If `mask` is None the mask multiplication is omitted.
    All tensors must be contiguous ``torch.float16``.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when HAS_MASK==0

    _mask_gate_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        hidden,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
    )
    # No return – operation performed in‑place


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool or float) – may be None
            - weights      : dict of model parameters (float32 tensors)
            - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    bs, N, _, dim = x.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 → half)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Fuse five linear layers (2 proj + 2 gate + out_gate) into
    #    two matrix multiplications: one for proj+gate, one for out_gate.
    # ------------------------------------------------------------------
    # Combine weights for left/right projections and their gates.
    proj_gate_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                        # (4*hidden, dim)

    # Flatten x for the batched GEMM.
    x_flat = x_h.reshape(-1, dim)               # (B*N*N, dim)

    # Single GEMM producing concatenated tensors.
    combined = F.linear(x_flat, proj_gate_weight)  # (B*N*N, 4*hidden)
    combined = combined.view(bs, N, N, 4 * hidden)  # (B, N, N, 4*hidden)

    # Slice the combined result.
    left = combined[..., 0:hidden]
    right = combined[..., hidden:2 * hidden]
    left_gate_pre = combined[..., 2 * hidden:3 * hidden]
    right_gate_pre = combined[..., 3 * hidden:4 * hidden]

    # Separate linear for out_gate.
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)  # (hidden, dim)
    out_gate = F.linear(x_flat, out_gate_weight)                    # (B*N*N, hidden)
    out_gate = out_gate.view(bs, N, N, hidden)

    # ------------------------------------------------------------------
    # 3. Apply sigmoid to gates.
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate)

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) and gate multiplication via Triton.
    # ------------------------------------------------------------------
    if mask is not None:
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()  # (B, N, N, 1)
        _apply_mask_and_gate(
            left, right, left_gate, right_gate, mask_h, hidden
        )
    else:
        # No mask – just apply the gates.
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core N³ contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as a batched GEMM.
    # ------------------------------------------------------------------
    # Arrange tensors as (B, hidden, N, N) for batched matmul.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, hidden, N, N) (right transposed)

    B, H, _, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B*hidden, N, N)
    right_mat = right_perm.view(B * H, N, N) # (B*hidden, N, N)

    # Batched matrix multiplication in half‑precision.
    out_mat = torch.bmm(left_mat, right_mat)            # (B*hidden, N, N)

    # Reshape back to (B, N, N, hidden)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection.
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 7. Cast back to float32 as required by the API.
    # ------------------------------------------------------------------
    return out.to(torch.float32)
