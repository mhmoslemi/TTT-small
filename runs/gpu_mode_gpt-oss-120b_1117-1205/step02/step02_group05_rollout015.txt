@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets in each dimension
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for out-of-bound rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # load A tile
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0)

        # load B tile
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & mask_n[None, :], other=0.0)

        # dot product
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs
    batch_id = tl.program_id(0)  # which (batch*hidden) GEMM
    pid_m = tl.program_id(1)     # tile row
    pid_n = tl.program_id(2)     # tile column

    # offsets for M and N
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this GEMM
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for kp in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kp * BLOCK_K + tl.arange(0, BLOCK_K)
        # mask for K dimension
        mask_k = offs_k < K

        # Load A tile (M x K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile (K x N)
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Compute block dot product
        acc += tl.dot(a, b)

    # Write back result (store as FP16)
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def triton_batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ B for a batch of matrices using Triton.
    A : (batch, M, K) float16
    B : (batch, K, N) float16 (second operand is not transposed)
    Returns C : (batch, M, N) float16
    """
    assert A.is_contiguous() and B.is_contiguous()
    assert A.dtype == B.dtype == torch.float16
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tile sizes (tuned for NVIDIA B200, fp16)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor of shape [B, N, N] (bool or float)
            - weights_dict: mapping from weight name to torch.Tensor
            - config_dict: contains "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack arguments
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------
    # 1) Input layer‑norm (float32)
    # ------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # Cast to FP16 for the heavy work
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------
    # 2) Linear projections to hidden dimension
    # ------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # [B, N, N, H]
    right = F.linear(x_fp16, right_proj_w)  # [B, N, N, H]

    # ------------------------------------------------------------
    # 3) Optional mask
    # ------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [B, N, N, 1] (float16)
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------
    # 4) Gating values (sigmoid) and application
    # ------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()   # [B, N, N, H]
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------
    # 5) Core contraction: batch GEMM left @ right.T
    #    left:  (B, N, N, H) -> (B, H, N, N)
    #    right: (B, N, N, H) -> (B, H, N, N) with axes (k, j) swapped
    # ------------------------------------------------------------
    # Permute to bring the hidden dimension before the two sequence axes.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    # right needs to be transposed on the sequence axes: (B, H, k, j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    # Fuse batch and hidden dimensions for a single Triton batched GEMM
    left_flat = left_perm.view(B * H, N, N)   # (B*H, M=N, K=N)
    right_flat = right_perm.view(B * H, N, N) # (B*H, K=N, N)

    # Triton batched matrix multiplication (FP16, FP32 accumulation)
    out_flat = triton_batched_matmul(left_flat, right_flat)   # (B*H, N, N), FP16

    # Reshape back to (B, N, N, H)
    out_hidden = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # Convert to FP32 for the remaining LayerNorm / Linear
    out_hidden = out_hidden.to(torch.float32)

    # ------------------------------------------------------------
    # 6) Output LayerNorm, gating, and final projection
    # ------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    # Apply out‑gate (already in half, convert to FP32)
    out_hidden = out_hidden * out_gate.to(torch.float32)

    # Final linear projection back to original dim
    out = F.linear(out_hidden, weights["to_out.weight"].to(torch.float32))

    return out

"""
TriMul (outgoing) forward pass with a Triton‑fused batched GEMM.

Algorithm
---------
1. LayerNorm over the channel dimension (float32).
2. Cast to fp16 and apply two linear projections (left/right) to the hidden
   dimension.
3. Optional element‑wise mask (broadcast over the hidden dimension).
4. Compute three sigmoid gates from the normalized input and apply them to
   left, right and the later output.
5. Core contraction:
   For each batch element and hidden channel h, compute
       out[..., i, j, h] = Σ_k left[..., i, k, h] * right[..., j, k, h]
   which is a batched matrix multiplication left @ rightᵀ.
   This is realized by a custom Triton kernel that processes
   (batch * hidden) independent GEMMs of size N×N using fp16 arithmetic
   with fp32 accumulation.
6. LayerNorm over the hidden dimension, multiply by the output gate and
   project back to the original dimension with a final linear layer.
   The result is returned as float32.
"""

    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # Cast to fp16 for the heavy work
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------
    # 2) Linear projections to hidden dimension
    # ------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)    # [B, N, N, H]
    right = F.linear(x_fp16, right_proj_w) # [B, N, N, H]

    # ------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------
    # 4) Gating values (sigmoid) and application
    # ------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()   # [B, N, N, H]
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------
    # 5) Core contraction via Triton batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------
    # Rearrange to (B, H, N, N) layout; right needs a transpose on the
    # two sequence axes to become (K, N) for the GEMM.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) = (B, H, K, J)

    B, H, N, _ = left_perm.shape
    # Fuse batch and hidden dimensions
    left_flat = left_perm.view(B * H, N, N)   # (B*H, M=N, K=N)
    right_flat = right_perm.view(B * H, N, N) # (B*H, K=N, N)

    # Perform the batched matmul in fp16 (accumulated in fp32)
    out_flat = triton_batched_matmul(left_flat, right_flat)  # (B*H, N, N) fp16

    # Restore shape (B, N, N, H)
    out_hidden = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # Convert to float32 for the remaining operations
    out_hidden = out_hidden.to(torch.float32)

    # ------------------------------------------------------------
    # 6) Output LayerNorm, out‑gate multiplication and final projection
    # ------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    # Apply the output gate (cast to float32)
    out_hidden = out_hidden * out_gate.to(torch.float32)

    # Final linear back to `dim`
    out = F.linear(out_hidden, weights["to_out.weight"].to(torch.float32))

    return out
```
