```
@triton.jit
def fused_gate_mask_kernel(
    left_in_ptr,     # [B,N,N,hidden_dim] left projected
    left_gate_ptr,   # [B,N,N,hidden_dim] left gate values
    right_in_ptr,    # [B,N,N,hidden_dim] right projected
    right_gate_ptr,  # [B,N,N,hidden_dim] right gate values
    mask_ptr,        # [B,N,N] mask broadcast for hidden_dim
    left_out_ptr,    # output left
    right_out_ptr,   # output right
    total_elems,     # total number of elements (B*N*N*hidden_dim)
    hidden_dim: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    left = tl.load(left_in_ptr + offs, mask=mask)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask)
    right = tl.load(right_in_ptr + offs, mask=mask)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask)
    # compute mask index
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask)
    # broadcast mask to hidden dim
    # because we have mask per [B,N,N] element, each hidden entry uses same mask
    left_out = left * left_gate * mask_val
    right_out = right * right_gate * mask_val
    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)
```

"""
TriMul (outgoing) forward pass with Triton‑fused mask+gate multiplication.

Algorithm
---------
1. Layer‑norm the input tensor (float32).
2. Cast to FP16 and apply three linear projections:
   * left_proj, right_proj  -> tensors of shape [B,N,N,H]
   * left_gate, right_gate, out_gate -> same shape, followed by sigmoid.
3. Fuse elementwise `* mask * gate` for left and right branches using a
   single Triton kernel (avoids two separate launches and extra memory traffic).
   The mask (shape [B,N,N]) is broadcast over the hidden dimension H.
4. Perform the cubic contraction
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   by reshaping each hidden slice as a separate batch and calling `torch.bmm`,
   which uses cuBLAS for maximal performance.
5. Apply LayerNorm over hidden dim, multiply by `out_gate`, and final
   linear projection back to `dim` (float32).

The heavy linear work stays in PyTorch (cuBLAS), while the elementwise
fusion is accelerated by Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def fused_gate_mask_kernel(
    left_ptr,          # [B,N,N,H] left projection (flat)
    left_gate_ptr,    # [B,N,N,H] left gate (flat)
    right_ptr,         # [B,N,N,H] right projection (flat)
    right_gate_ptr,    # [B,N,N,H] right gate (flat)
    mask_ptr,          # [B,N,N] mask (flat)
    left_out_ptr,      # output left after gate*mask
    right_out_ptr,     # output right after gate*mask
    total_elems,       # total number of elements = B*N*N*H
    hidden_dim: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elems

    # Load tensors
    left = tl.load(left_ptr + offs, mask=valid)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid)
    right = tl.load(right_ptr + offs, mask=valid)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid)

    # Compute mask index (broadcast over hidden dimension)
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=valid)

    # Fuse multiplications
    left_out = left * left_gate * mask_val
    right_out = right * right_gate * mask_val

    # Store results
    tl.store(left_out_ptr + offs, left_out, mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)


def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor : torch.Tensor [B, N, N] (float32) or None
            - weights_dict: dict containing all required weight tensors
            - config_dict : {"dim": int, "hidden_dim": int}
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # [B,N,N,dim]

    # -----------------------------------------------------------------
    # 2. Cast to FP16 for heavy work
    # -----------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections (no bias) in FP16
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)      # [B,N,N,hidden_dim], fp16
    right = F.linear(x_fp16, right_proj_w)    # [B,N,N,hidden_dim], fp16

    # -----------------------------------------------------------------
    # 4. Gates (sigmoid) in FP16
    # -----------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w)) # [B,N,N,hidden_dim]
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))    # [B,N,N,hidden_dim]

    # -----------------------------------------------------------------
    # 5. Optional mask handling
    # -----------------------------------------------------------------
    if mask_tensor is None:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(device).to(torch.float16)

    # Ensure contiguity for flat views
    left_flat = left.contiguous().view(-1)
    right_flat = right.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)
    mask_flat = mask_fp16.contiguous().view(-1)

    # Allocate output buffers for the fused operation
    left_out_flat = torch.empty_like(left_flat)
    right_out_flat = torch.empty_like(right_flat)

    # -----------------------------------------------------------------
    # 6. Triton kernel: fuse gate & mask multiplication for left/right
    # -----------------------------------------------------------------
    total = left_flat.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    fused_gate_mask_kernel[grid](
        left_ptr=left_flat,
        left_gate_ptr=left_gate_flat,
        right_ptr=right_flat,
        right_gate_ptr=right_gate_flat,
        mask_ptr=mask_flat,
        left_out_ptr=left_out_flat,
        right_out_ptr=right_out_flat,
        total_elems=total,
        hidden_dim=hidden_dim,
        BLOCK_SIZE=BLOCK,
    )

    # Reshape back to [B,N,N,hidden_dim]
    left = left_out_flat.view(B, N, N, hidden_dim)
    right = right_out_flat.view(B, N, N, hidden_dim)

    # -----------------------------------------------------------------
    # 7. Cubic contraction via batched GEMM (cuBLAS)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # -----------------------------------------------------------------
    # Prepare tensors: left (i,k) ; right (k,j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) -> (k,j)

    batch = B * hidden_dim
    left_mat = left_perm.view(batch, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(batch, N, N) # (B*H, N, N)

    # Batched matrix multiplication (FP16, cuBLAS)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)

    # Restore original layout [B,N,N,hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden_dim)

    # -----------------------------------------------------------------
    # 8. Post‑processing: LayerNorm (hidden dim), gating, final linear
    # -----------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply final gate (broadcast over hidden dim)
    out_norm = out_norm * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"].to(device)  # (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)  # [B, N, N, dim], float32

    return output
