import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Optional, Tuple, Dict

@triton.jit
def _triton_batched_matmul_block_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,           # Matrix dimensions
    stride_am, stride_ak,  # Strides for matrix A: [batch_size * hidden_dim, M, K]
    stride_bk, stride_bn,  # Strides for matrix B: [batch_size * hidden_dim, K, N]
    BLOCK: tl.constexpr   # Tile size for M and K dimensions
):
    # Compute batch index and channel index
    batch_id = tl.program_id(0)
    channel = tl.program_id(1)
    
    # Base pointer for this (batch, channel)
    a_batch = a_ptr + (batch_id * hidden_dim + channel) * stride_am
    b_batch = b_ptr + (batch_id * hidden_dim + channel) * stride_bk
    c_batch = c_ptr + (batch_id * hidden_dim + channel) * stride_cm
    
    # Compute offsets for the tile
    offs_m = channel * BLOCK + tl.arange(0, BLOCK) % M     # Offset within batch * hidden_dim * M * K
    offs_k = tl.arange(0, BLOCK) // M * M + tl.arange(0, M) % BLOCK + (tl.program_id(2) * M)  # More complex index calculation for heatmap
    offs_n = tl.arange(0, N)
    
    # Masks to avoid out-of-bound
    offs_m_mask = offs_m < M
    offs_k_mask = offs_k < K
    
    # Initialize accumulator
    acc = tl.zeros((BLOCK, N), dtype=tl.float32)
    
    # Iterate over k for the block
    for k_idx in range(0, K, BLOCK):
        # Integer division and offset index for k
        k_offsets = k_idx + tl.arange(0, BLOCK)
        
        # Update pointers
        a_ptrs = a_batch + k_offsets * stride_ak + offs_m[:, None] * stride_am
        b_ptrs = b_batch + k_offsets * stride_bk + offs_n[None, :] * stride_bn
        
        # Load a block of A
        a_slice = tl.load(a_ptrs, mask=offs_m_mask, other=0.0)
        
        # Load a block of B
        b_slice = tl.load(b_ptrs, mask=offs_k_mask, other=0.0)
        
        # Perform dot product
        acc += tl.dot(a_slice, b_slice, allow_approx=True)
    
    # Write the result
    c_ptrs = c_batch + offs_k[None, :] * stride_cm + offs_m[:, None] * stride_am
    tl.store(c_ptrs, acc, mask=offs_m_mask[:, None] & (offs_k[:, None] < K))
