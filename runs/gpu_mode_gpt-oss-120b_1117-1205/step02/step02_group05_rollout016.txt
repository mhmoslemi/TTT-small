import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def fused_mul_inplace_kernel(a_ptr, b_ptr, n_elements, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < n_elements
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(a_ptr + offs, a * b, mask=mask)

def fused_mul_inplace(a, b):
    # a and b must be contiguous, same shape, on cuda, same dtype
    assert a.is_cuda and b.is_cuda
    assert a.shape == b.shape
    n = a.numel()
    BLOCK = 1024
    grid = (triton.cdiv(n, BLOCK),)
    fused_mul_inplace_kernel[grid](a, b, n, BLOCK=BLOCK)
    # no return needed; a modified in-place

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm on input (float32)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # Cast to fp16 for heavy ops
    x_fp16 = x.to(torch.float16)

    # Linear projections (no bias)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left = F.linear(x_fp16, left_proj_w)   # shape [B,N,N,hidden_dim]
    right = F.linear(x_fp16, right_proj_w)

    # Gating linear layers (sigmoid)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()

    # Apply gating via in-place Triton kernel
    fused_mul_inplace(left, left_gate)   # left = left * left_gate
    fused_mul_inplace(right, right_gate) # right = right * right_gate

    # Optional mask (broadcast over hidden_dim)
    if not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # shape [B,N,N,1]
        left.mul_(mask_fp16)
        right.mul_(mask_fp16)

    # Core contraction: batch matrix multiplication left @ right^T
    # Rearrange to (batch, hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)        # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N)      # (B*H, N, N)

    # Batched matrix multiplication (A @ B^T)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N), fp16

    # Restore shape (B,N,N,H) and convert to fp32
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # Output LayerNorm over hidden_dim
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    # Multiply by out_gate (fp32)
    out = out * out_gate.to(torch.float32)

    # Final projection back to dim
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
Fast Triton‑accelerated forward pass for the outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor (float32).
2. Cast to fp16 and linearly project to a hidden dimension (left/right).
3. Compute sigmoid gates (left_gate, right_gate, out_gate) in fp16.
4. Fuse the element‑wise multiplication left←left*left_gate and right←right*right_gate
   with a small Triton kernel (in‑place, no extra memory).
5. Optionally apply the mask (broadcast over the hidden dimension) with a cheap
   PyTorch in‑place multiplication.
6. Core contraction: for each batch and hidden channel compute
      out[...,i,j,d] = Σ_k left[...,i,k,d] * right[...,j,k,d]
   implemented as a batched GEMM (left @ rightᵀ) using torch.bmm on fp16 tensors
   (Tensor‑core accelerated).
7. Layer‑norm over the hidden dimension, multiply by out_gate, and linearly project
   back to the original dimension (float32).

Only the element‑wise gating step is performed with Triton; the large matrix
multiplication leverages PyTorch’s highly‑optimized cuBLAS batched GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: in‑place element‑wise multiplication a *= b
# ----------------------------------------------------------------------
@triton.jit
def _fused_mul_inplace_kernel(
    a_ptr, b_ptr, n_elements, BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < n_elements

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(a_ptr + offs, a * b, mask=mask)


def _fused_mul_inplace(a: torch.Tensor, b: torch.Tensor) -> None:
    """
    In‑place element‑wise multiplication a *= b using Triton.
    Both tensors must be on CUDA, contiguous, have the same shape and dtype.
    """
    assert a.is_cuda and b.is_cuda
    assert a.shape == b.shape
    n = a.numel()
    BLOCK = 1024
    grid = (triton.cdiv(n, BLOCK),)
    _fused_mul_inplace_kernel[grid](a, b, n, BLOCK=BLOCK)


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool or numeric)
        - weights      : dict of model parameters
        - config       : dict containing 'dim', 'hidden_dim' and optional 'nomask'

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 and linear projections (no bias)
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)    # [B, N, N, hidden_dim], fp16
    right = F.linear(x_fp16, right_proj_w)  # [B, N, N, hidden_dim], fp16

    # ------------------------------------------------------------------
    # 3. Gating values (sigmoid, fp16)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()   # fp16
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()  # fp16
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()     # fp16

    # ------------------------------------------------------------------
    # 4. Apply gates with an in‑place Triton kernel
    # ------------------------------------------------------------------
    _fused_mul_inplace(left, left_gate)    # left ← left * left_gate
    _fused_mul_inplace(right, right_gate)  # right ← right * right_gate

    # ------------------------------------------------------------------
    # 5. Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] → fp16
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left.mul_(mask_fp16)   # in‑place broadcast multiplication
        right.mul_(mask_fp16)

    # ------------------------------------------------------------------
    # 6. Core contraction: batched GEMM left @ rightᵀ (fp16, Tensor‑core)
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden_dim, N, N) for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)            # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N)          # (B*H, N, N)

    # Batched matrix multiplication: A @ Bᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # fp16

    # Restore original layout [B, N, N, hidden_dim] and promote to fp32
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate multiplication, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    out = out * out_gate.to(torch.float32)

    out = F.linear(out, weights["to_out.weight"])   # final projection to dim

    return out
