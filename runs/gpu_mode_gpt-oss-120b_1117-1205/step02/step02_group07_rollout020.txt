```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    a_batch_base = a_ptr + batch_id * stride_a_batch
    b_batch_base = b_ptr + batch_id * stride_b_batch
    c_batch_base = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        a_ptrs = a_batch_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_ptrs = b_batch_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    c_ptrs = c_batch_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def triton_batched_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched matmul using Triton: C = A @ B where B is already transposed (KxN).
    All tensors must be contiguous, dtype float16.
    Shapes:
        a: (batch, M, K)
        b: (batch, K, N)
    Returns:
        c: (batch, M, N) float16
    """
    assert a.dtype == b.dtype == torch.float16
    assert a.is_contiguous() and b.is_contiguous()
    batch, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    c = torch.empty((batch, M, N), dtype=torch.float16, device=a.device)

    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_k, stride_b_n = b.stride()
    stride_c_batch, stride_c_m, stride_c_n = c.stride()

    # Tune block sizes – 64 works well for a wide range of N
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        a, b, c,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=2,
    )
    return c

def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.
    Arguments:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # shape [B, N, N, dim] float32

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for the heavy linear layers
    # -----------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Combine left projection & left gate, right projection & right gate
    #    to reduce # of GEMM calls.
    # -----------------------------------------------------------------
    # Concatenate weights for left projection + left gate
    left_combined_w = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(torch.float16)
    right_combined_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(torch.float16)

    # Single matmul for left_proj + left_gate
    left_combined = F.linear(x_fp16, left_combined_w)  # shape [B, N, N, 2*hidden_dim]
    left_proj = left_combined[..., :hidden_dim]
    left_gate_pre = left_combined[..., hidden_dim:]
    left_gate = left_gate_pre.sigmoid()

    # Single matmul for right_proj + right_gate
    right_combined = F.linear(x_fp16, right_combined_w)
    right_proj = right_combined[..., :hidden_dim]
    right_gate_pre = right_combined[..., hidden_dim:]
    right_gate = right_gate_pre.sigmoid()

    # -----------------------------------------------------------------
    # 4. Apply optional mask
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)  # [B, N, N, 1]
        left_proj = left_proj * mask_fp
        right_proj = right_proj * mask_fp

    # -----------------------------------------------------------------
    # 5. Apply gates
    # -----------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 6. Compute out_gate (separate linear)
    # -----------------------------------------------------------------
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # -----------------------------------------------------------------
    # 7. Core contraction via batched GEMM (Triton)
    #    left: (B, N, N, hidden) -> (B, hidden, N, N)
    #    right: (B, N, N, hidden) -> (B, hidden, N, N) and transpose K,J
    # -----------------------------------------------------------------
    # Prepare tensors for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)   (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)   (k,j)

    B, H, N, _ = left_perm.shape
    # Flatten batch and head dimensions
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Batched matmul: (B*H, N, N) @ (B*H, N, N) -> (B*H, N, N)
    out_flat = triton_batched_matmul(left_flat, right_flat)

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # -----------------------------------------------------------------
    # 8. Output LayerNorm, out_gate scaling and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
```

dot products.

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ B (B is already transposed)
# -------------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)          # combined batch * head dimension
    pid_m = tl.program_id(1)             # tile row
    pid_n = tl.program_id(2)             # tile column

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k < K),
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k < K)[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

    # Store the result
    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def triton_batched_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B using Triton.
    A: (batch, M, K)   fp16
    B: (batch, K, N)   fp16   (already transposed for the contraction)
    Returns C: (batch, M, N)   fp16
    """
    assert a.dtype == b.dtype == torch.float16
    assert a.is_contiguous() and b.is_contiguous()
    batch, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    c = torch.empty((batch, M, N), dtype=torch.float16, device=a.device)

    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_k, stride_b_n = b.stride()
    stride_c_batch, stride_c_m, stride_c_n = c.stride()

    # Tile sizes – 64 works well for the range of N in the tests
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        a,
        b,
        c,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=2,
    )
    return c


# -------------------------------------------------------------------------
# Entry point expected by the evaluator
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator, using Triton for the
    expensive N‑by‑N contraction.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor : torch.Tensor of shape [B, N, N] (bool or numeric)
            - weights_dict: mapping from parameter names to torch.Tensors
            - config_dict : must contain 'dim' and 'hidden_dim' and optionally 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim], float32

    # -------------------------------------------------
    # 2. Cast to fp16 for heavy linear work
    # -------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -------------------------------------------------
    # 3. Fuse left/right projections with their gates
    # -------------------------------------------------
    # left_proj + left_gate
    left_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)
    left_combined = F.linear(x_fp16, left_w)  # [B, N, N, 2*hidden]
    left_proj = left_combined[..., :hidden_dim]
    left_gate = left_combined[..., hidden_dim:].sigmoid()

    # right_proj + right_gate
    right_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)
    right_combined = F.linear(x_fp16, right_w)  # [B, N, N, 2*hidden]
    right_proj = right_combined[..., :hidden_dim]
    right_gate = right_combined[..., hidden_dim:].sigmoid()

    # -------------------------------------------------
    # 4. Optional mask (broadcast over hidden)
    # -------------------------------------------------
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)  # [B, N, N, 1]
        left_proj = left_proj * mask_fp
        right_proj = right_proj * mask_fp

    # -------------------------------------------------
    # 5. Apply gates
    # -------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # -------------------------------------------------
    # 6. Compute out_gate (separate linear)
    # -------------------------------------------------
    out_gate = F.linear(
        x_fp16, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()  # [B, N, N, hidden]

    # -------------------------------------------------
    # 7. Core contraction via Triton batch matmul
    #    left: (B, N, N, H) -> (B, H, N, N)   (i,k)
    #    right: (B, N, N, H) -> (B, H, N, N)   (k,j)  by swapping N dims
    # -------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2)          # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1)        # (B, H, N, N) = (k,j)

    B, H, N, _ = left_perm.shape
    # Merge batch and head dimensions (no copy – view is possible after the permute)
    left_flat = left_perm.reshape(B * H, N, N)
    right_flat = right_perm.reshape(B * H, N, N)

    out_flat = triton_batched_matmul(left_flat, right_flat)  # (B*H, N, N)

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # -------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final projection
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
