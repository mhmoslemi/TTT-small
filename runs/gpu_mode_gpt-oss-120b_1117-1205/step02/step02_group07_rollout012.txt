def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ah,
    stride_bk, stride_bn, stride_bh,
    stride_cm, stride_cn, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)   # batch dimension (B*H)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block start offset
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute base pointer for each batch
    a_batch_ptr = a_ptr + batch_id * stride_ah
    b_batch_ptr = b_ptr + batch_id * stride_bh
    c_batch_ptr = c_ptr + batch_id * stride_ch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_gemm_kernel(a_ptr, b_ptr, c_ptr,
                        M, N, K,
                        stride_a_batch, stride_a_m, stride_a_k,
                        stride_b_batch, stride_b_k, stride_b_n,
                        stride_c_batch, stride_c_m, stride_c_n,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def batched_gemm(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM C = A @ B where
        A: (batch, M, K) FP16
        B: (batch, K, N) FP16
    Returns C: (batch, M, N) FP16
    """
    assert a.dtype == torch.float16 and b.dtype == torch.float16
    batch, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    c = torch.empty((batch, M, N), dtype=torch.float16, device=a.device)

    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_k, stride_b_n = b.stride()
    stride_c_batch, stride_c_m, stride_c_n = c.stride()

    # tuned block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_gemm_kernel[grid](
        a, b, c,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    The implementation follows the reference PyTorch code but off‑loads the
    expensive contraction over the sequence dimension to a custom batched GEMM
    kernel written in Triton. All other linear projections, gating and layer‑norm
    steps are performed with PyTorch for simplicity.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # (B, N, N, dim), float32

    # -------------------------------------------------
    # 2. Cast to FP16 for heavy work
    # -------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -------------------------------------------------
    # 3. Linear projections (no bias)
    # -------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # -------------------------------------------------
    # 4. Mask (optional)
    # -------------------------------------------------
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_fp
        right = right * mask_fp

    # -------------------------------------------------
    # 5. Gating
    # -------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 6. Core contraction via Triton batched GEMM
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # -------------------------------------------------
    # Rearrange to (batch, head, N, N) where head == hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)   (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)   (k,j)

    B, H, N, _ = left_perm.shape

    # flatten batch*head dimension
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Perform batched matmul: (B*H, N, N) @ (B*H, N, N)ᵀ
    out_flat = batched_gemm(left_flat, right_flat)  # (B*H, N, N), fp16

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # -------------------------------------------------
    # 7. Output LayerNorm, out‑gate and final projection
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out

    _, K2, N = b.shape
    assert K == K2

    c = torch.empty((batch, M, N), dtype=torch.float16, device=a.device)

    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_k, stride_b_n = b.stride()

    # Rearrange tensors to (B, H, N, N) where H = hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N) – (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) – (k,j)

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)     # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N)   # (B*H, N, N)

    # Batched matmul: (B*H, N, N) @ (B*H, N, N)^T  → (B*H, N, N)
    out_flat = _batched_gemm(left_flat, right_flat)  # FP16

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # -------------------------------------------------
    # 7️⃣ Output LayerNorm, out‑gate, final projection
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
```
