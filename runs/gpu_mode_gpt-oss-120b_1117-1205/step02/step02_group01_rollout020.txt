"""
TriMul (outgoing) forward pass.

The kernel fuses three elementwise operations:
    out = tensor * gate * mask
for the left and right branches.  The heavy cubic contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is realized as a batched matrix‑multiply (left @ rightᵀ) on FP16,
leveraging cuBLAS.  All LayerNorms and linear projections are performed
with PyTorch (FP32 for the final stages, FP16 for the bulk of the work).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def mul_gate_mask_kernel(
    out_ptr,            # pointer to output (flattened FP16)
    a_ptr,              # pointer to tensor (flattened FP16)
    gate_ptr,           # pointer to gate tensor (flattened FP16)
    mask_ptr,           # pointer to mask tensor (flattened FP16, broadcast over H)
    total,              # total number of elements (B*N*N*H)
    BLOCK_SIZE: tl.constexpr,
    H: tl.constexpr,    # hidden dimension (size of innermost axis)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    a = tl.load(a_ptr + offs, mask=mask)
    g = tl.load(gate_ptr + offs, mask=mask)

    # broadcast mask over hidden dimension
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    tl.store(out_ptr + offs, a * g * m, mask=mask)


def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            mask_tensor : torch.Tensor of shape [B, N, N] (float32) or None
            weights_dict : model weights
            config_dict : {"dim": int, "hidden_dim": int, "nomask": bool (optional)}
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # --------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_b = weights["norm.bias"].to(device, dtype=torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # [B,N,N,dim]

    # --------------------------------------------------------------------
    # 2. Cast to FP16 for the costly work
    # --------------------------------------------------------------------
    x_fp16 = x.half()

    # --------------------------------------------------------------------
    # 3. Linear projections (no bias) → [B,N,N,hidden]
    # --------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device, dtype=torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # [B,N,N,hidden]
    right = F.linear(x_fp16, right_proj_w)  # [B,N,N,hidden]

    # --------------------------------------------------------------------
    # 4. Gates (sigmoid) → [B,N,N,hidden]
    # --------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device, dtype=torch.float16)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))   # used later

    # --------------------------------------------------------------------
    # 5. Optional mask (broadcast over hidden) – treat missing mask as all‑ones
    # --------------------------------------------------------------------
    if mask_tensor is None or nomask:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(device).to(torch.float16)

    # --------------------------------------------------------------------
    # 6. Fuse element‑wise multiplication: tensor * gate * mask
    # --------------------------------------------------------------------
    # flatten everything for the kernel (contiguous layout assumed)
    left_flat = left.view(-1)
    left_gate_flat = left_gate.view(-1)
    right_flat = right.view(-1)
    right_gate_flat = right_gate.view(-1)
    mask_flat = mask_fp16.view(-1)

    total = left_flat.numel()                     # B * N * N * hidden
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    # left = left * left_gate * mask
    mul_gate_mask_kernel[grid](
        left_flat,
        left_flat,
        left_gate_flat,
        mask_flat,
        total,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
    )
    # right = right * right_gate * mask
    mul_gate_mask_kernel[grid](
        right_flat,
        right_flat,
        right_gate_flat,
        mask_flat,
        total,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
    )

    # reshape back
    left = left_flat.view(B, N, N, hidden_dim)
    right = right_flat.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------------
    # 7. Cubic contraction via batched GEMM: out = left @ rightᵀ
    # --------------------------------------------------------------------
    # (B, hidden, N, N) where axes are (batch, head, i, k) and (batch, head, k, j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) -> (B, H, k, i)

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)   # (BH, N, N)
    right_mat = right_perm.view(BH, N, N) # (BH, N, N)

    out_mat = torch.bmm(left_mat, right_mat)  # (BH, N, N), FP16
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden)

    # --------------------------------------------------------------------
    # 8. Post‑processing: LayerNorm on hidden dim, final gate, output projection
    # --------------------------------------------------------------------
    out_fp32 = out.float()  # switch to FP32 for the last two linear ops

    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype=torch.float32)

    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # apply final gating
    out_norm = out_norm * out_gate.float()

    to_out_w = weights["to_out.weight"].to(device, dtype=torch.float32)  # (dim, hidden)
    output = F.linear(out_norm, to_out_w)   # (B, N, N, dim)

    return output
