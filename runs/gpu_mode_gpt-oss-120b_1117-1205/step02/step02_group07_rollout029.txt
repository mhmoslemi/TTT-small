```
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers
    a_ptrs = a_ptr + batch_id * stride_a_batch
    b_ptrs = b_ptr + batch_id * stride_b_batch
    c_ptrs = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Load A tile
        a = tl.load(
            a_ptrs + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k,
            mask=mask_m[:, None] & (offs_k < K),
            other=0.0
        )
        # Load B tile
        b = tl.load(
            b_ptrs + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n,
            mask=(offs_k < K)[:, None] & mask_n[None, :],
            other=0.0
        )

        acc += tl.dot(a, b)

    # store
    c = acc.to(tl.float16)
    tl.store(
        c_ptrs + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n,
        c,
        mask=mask_m[:, None] & mask_n[None, :]
    )
```

```
def triton_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    # a,b both (B_total, M, K/N) same shape; a dtype fp16
    B_total, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    c = torch.empty((B_total, M, N), dtype=torch.float16, device=a.device)

    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_k, stride_b_n = b.stride()
    stride_c_batch, stride_c_m, stride_c_n = c.stride()

    # block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (B_total, triton.cdiv(M, BLOCK_M), triton.cdiv(N, BLOCK_N))

    batched_matmul_kernel[grid](
        a, b, c,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8
    )
    return c
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # cast to fp16
    x_fp16 = x.to(torch.float16)
    # weight casts
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # linear projections and gating
    left = F.linear(x_fp16, left_proj_w)   # [B,N,N,H]
    right = F.linear(x_fp16, right_proj_w)

    if not config.get("nomask", True):
        mask_fp = mask.unsqueeze(-1).to(torch.float16)  # [B,N,N,1]
        left = left * mask_fp
        right = right * mask_fp

    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # contraction via Triton
    # permute and reshape
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    out_flat = triton_matmul(left_flat, right_flat)  # fp16

    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # output norm
    out = F.layer_norm(out, (hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])

    out = out * out_gate.to(torch.float32)

    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out

"""
TriMul (outgoing) forward pass with a fused Triton batched GEMM.

The algorithm follows the reference PyTorch implementation:
1. Layer‑norm over the channel dimension (float32).
2. Linear projections & gating are performed in fp16.
3. An optional binary mask is applied (broadcast over the hidden dim).
4. The core contraction
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   is a batched matrix multiplication of shape (B·H, N, N) @ (B·H, N, N).
   This is executed by a custom Triton kernel (fp16 inputs, fp32 accumulation).
5. Output layer‑norm, out‑gate and final projection back to the original
   channel dimension are done in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------
# Triton batched GEMM kernel (C = A @ B)
# ------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matrix multiplication C = A @ B
    A : (batch, M, K)   fp16
    B : (batch, K, N)   fp16
    C : (batch, M, N)   fp16 (stored)  with fp32 accumulation
    """
    batch_id = tl.program_id(0)       # batch dimension = B * hidden_dim
    pid_m = tl.program_id(1)          # block row
    pid_n = tl.program_id(2)          # block column

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    # Accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a = tl.load(
            a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k,
            mask=mask_m[:, None] & (offs_k < K),
            other=0.0
        )
        # Load B tile (K x N)
        b = tl.load(
            b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n,
            mask=(offs_k < K)[:, None] & mask_n[None, :],
            other=0.0
        )
        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # Write back the result (convert to fp16)
    c = acc.to(tl.float16)
    tl.store(
        c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n,
        c,
        mask=mask_m[:, None] & mask_n[None, :]
    )


def triton_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.
    a, b : float16 tensors of shape (batch, M, K) and (batch, K, N)
    Returns c : float16 tensor of shape (batch, M, N)
    """
    assert a.dtype == b.dtype == torch.float16
    assert a.is_contiguous() and b.is_contiguous()
    batch, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    c = torch.empty((batch, M, N), dtype=torch.float16, device=a.device)

    # Strides in elements
    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_k, stride_b_n = b.stride()
    stride_c_batch, stride_c_m, stride_c_n = c.stride()

    # Block sizes – tuned for GPUs up to 1024×1024
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        a, b, c,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return c


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric)
            - weights_dict: mapping of parameter names to torch.Tensor
            - config_dict: must contain 'dim', 'hidden_dim', and optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim] float32

    # Cast to fp16 for heavy linear work
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections (fp16)
    # -----------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden_dim)
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)   # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # -----------------------------------------------------------------
    # 4. Gating (fp16) and apply
    # -----------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: batched GEMM via Triton
    #    Compute out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    # -----------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N) where dimensions correspond to (b, h, i, k)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N) – (i,k)
    # (B, N, N, H) -> (B, H, N, N) where last two dims are (k,j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) – (k,j)

    B, H, N, _ = left_perm.shape
    # Merge batch and head dimensions for the kernel
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Batched matrix multiplication (fp16 in, fp16 out)
    out_flat = triton_matmul(left_flat, right_flat)   # (B*H, N, N)

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection (float32)
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
