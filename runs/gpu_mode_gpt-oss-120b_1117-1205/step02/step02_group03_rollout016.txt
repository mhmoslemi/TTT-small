"""
Triton‑accelerated outgoing TriMul (AlphaFold3).

The heavy N³ contraction
    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
is implemented as a batched GEMM where each (batch, hidden) pair is a
regular matrix multiplication (N×N)·(N×N)ᵀ.  By using the original tensor
layout we avoid costly permutes and extra memory copies.
All other operations (LayerNorm, Linear projections, gates, final Linear)
are performed in PyTorch (fp16/fp32) while the core contraction runs in a
custom Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,                     # runtime scalars
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Parameters
    ----------
    left_ptr, right_ptr, out_ptr : pointers to the tensors with layout
        (B, N, N, H) – contiguous in memory.
    B, N, H : batch size, sequence length, hidden dimension.
    stride_* : strides for the respective tensors (as returned by .stride()).
    BLOCK_*: compile‑time tile sizes.
    """
    # combined batch id = b * H + d
    pid_batch = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # decompose into batch index b and hidden index d
    b = pid_batch // H
    d = pid_batch % H

    # tile offsets for i and j dimensions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (b, d)
    left_base  = left_ptr  + b * stride_lb + d * stride_ld
    right_base = right_ptr + b * stride_rb + d * stride_rd
    out_base   = out_ptr   + b * stride_ob + d * stride_od

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = kk * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i, k] tile
        left_ptrs = left_base + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        left_tile = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right[j, k] tile (note the transposed access: rows = k, cols = j)
        right_ptrs = right_base + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_rj
        right_tile = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # accumulate dot product (uses Tensor Cores)
        acc += tl.dot(left_tile, right_tile)

    # write the result tile
    out_ptrs = out_base + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold3).

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/float) – may be ignored if nomask=True
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)  # switch to fp16 for the heavy path

    # ------------------------------------------------------------------
    # 2. Linear projections (fp16, bias=False)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)  # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 4. Gates (fp16)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton (batched GEMM)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # B, N, N, hidden_dim
    out = torch.empty_like(left, dtype=torch.float16, device=device)

    # strides for the original layout (B, N, N, H)
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    # tile sizes – tuned for fp16 TensorCores on Hopper (B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_lb,
        stride_li,
        stride_lk,
        stride_ld,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rd,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection (float32)
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))
    return out
