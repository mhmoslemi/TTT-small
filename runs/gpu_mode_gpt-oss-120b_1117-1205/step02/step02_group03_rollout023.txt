    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_jl, stride_dl,

"""
Outgoing Triangle Multiplicative Update (TriMul) kernel.

The core of TriMul is a contraction:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

We fuse this N³ reduction into a single Triton kernel.  The kernel
processes a tile of the output (i, j) while looping over the hidden
dimension (d) and the reduction dimension (k).  All arithmetic is
performed in fp16 with fp32 accumulation for Tensor‑Core efficiency.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fused batched contraction across the hidden dimension
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,                      # dimensions
    stride_bl, stride_il, stride_jl, stride_dl,  # left strides
    stride_br, stride_ir, stride_jr, stride_dr,  # right strides
    stride_bo, stride_io, stride_jo, stride_do,  # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out = Σ_k left[...,k,:] * right[...,k,:]"""
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # tile offsets for i and j dimensions
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_offset < N
    mask_j = j_offset < N

    # base pointers for this batch
    left_batch  = left_ptr  + pid_b * stride_bl
    right_batch = right_ptr + pid_b * stride_br
    out_batch   = out_ptr   + pid_b * stride_bo

    # loop over hidden channels
    for d in range(0, H):
        # accumulator for a single (i, j) tile at channel d
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # reduction over k
        for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
            k_offset = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = k_offset < N

            # left slice: shape (M, K)
            left_ptrs = (
                left_batch
                + i_offset[:, None] * stride_il
                + k_offset[None, :] * stride_jl
                + d * stride_dl
            )
            left = tl.load(
                left_ptrs,
                mask=mask_i[:, None] & mask_k[None, :],
                other=0.0,
            )

            # right slice: shape (K, N)  (k dimension first)
            right_ptrs = (
                right_batch
                + k_offset[:, None] * stride_jr
                + j_offset[None, :] * stride_ir
                + d * stride_dr
            )
            right = tl.load(
                right_ptrs,
                mask=mask_k[:, None] & mask_j[None, :],
                other=0.0,
            )

            # fused matmul for this channel
            acc += tl.dot(left, right)   # (M,N) accumulated in fp32

        # store result for channel d
        out_ptrs = (
            out_batch
            + i_offset[:, None] * stride_io
            + j_offset[None, :] * stride_jo
            + d * stride_do
        )
        tl.store(
            out_ptrs,
            acc.to(tl.float16),
            mask=mask_i[:, None] & mask_j[None, :],
        )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using Triton for the
    heavy N³ contraction.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool or numeric), may be unused
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16 for the heavy path
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections (fp16, no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)   # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # --------------------------------------------------------------
    # 4. Gates (fp16)
    # --------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous layout for Triton
    left = left.contiguous()
    right = right.contiguous()

    # --------------------------------------------------------------
    # 5. Core contraction: fused Triton kernel
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    out = torch.empty_like(left)  # [B, N, N, H], fp16

    # Strides (batch, i, j, d)
    s_bl, s_il, s_jl, s_dl = left.stride()
    s_br, s_ir, s_jr, s_dr = right.stride()
    s_bo, s_io, s_jo, s_do = out.stride()

    # Tile sizes (tuned for Hopper Tensor Cores)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_bl,
        s_il,
        s_jl,
        s_dl,
        s_br,
        s_ir,
        s_jr,
        s_dr,
        s_bo,
        s_io,
        s_jo,
        s_do,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Convert back to float32 for the remaining operations
    out = out.to(torch.float32)

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection (float32)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"])

    return out
