@triton.jit
def apply_gate_mask_kernel(
    left_ptr, left_gate_ptr, mask_ptr,
    right_ptr, right_gate_ptr,
    out_left_ptr, out_right_ptr,
    total_elements,
    stride_left, stride_left_gate, stride_mask,
    stride_right, stride_right_gate,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load
    l = tl.load(left_ptr + offs * stride_left, mask=mask)
    lg = tl.load(left_gate_ptr + offs * stride_left_gate, mask=mask)
    if mask_ptr != 0: # optional? but we can't condition at runtime maybe use a boolean flag to pass zeros
        m = tl.load(mask_ptr + offs * stride_mask, mask=mask)
    else:
        m = tl.full([BLOCK], 1.0, dtype=tl.float16)
    # Multiply
    out_l = l * lg * m

    # Right side
    r = tl.load(right_ptr + offs * stride_right, mask=mask)
    rg = tl.load(right_gate_ptr + offs * stride_right_gate, mask=mask)
    out_r = r * rg * m

    # Store
    tl.store(out_left_ptr + offs * stride_left, out_l, mask=mask)
    tl.store(out_right_ptr + offs * stride_right, out_r, mask=mask)
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm (float32)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # Cast to fp16 for heavy path
    x_fp16 = x.to(torch.float16)

    # Linear projections (no bias)
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # Gates
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # Mask (broadcast)
    if nomask:
        # create mask of ones with shape (B,N,N,1)
        mask_fp = torch.ones((left.shape[0], left.shape[1], left.shape[2], 1), dtype=torch.float16, device=left.device)
    else:
        # mask is [B,N,N]; unsqueeze to broadcast
        mask_fp = mask.unsqueeze(-1).to(torch.float16)

    # Ensure contiguous
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_fp = mask_fp.contiguous()

    # Apply gates and mask via Triton kernel
    total = left.numel()
    out_left = torch.empty_like(left)
    out_right = torch.empty_like(right)

    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    apply_gate_mask_kernel[grid](
        left,
        left_gate,
        mask_fp,
        right,
        right_gate,
        out_left,
        out_right,
        total,
        BLOCK=BLOCK,
    )

    # Release some tensors to free memory
    del left, right, left_gate, right_gate, mask_fp

    # Compute batched matmul using torch.bmm (could also use custom kernel)
    B, N, _, H = out_left.shape
    # Permute to (B, H, N, N)
    left_perm = out_left.permute(0, 3, 1, 2).contiguous()
    right_perm = out_right.permute(0, 3, 2, 1).contiguous()
    # Flatten batch*head
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)
    # Batched matmul
    out_flat = torch.bmm(left_flat, right_flat)  # (B*H,N,N)
    # Restore shape
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # Output LayerNorm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # Gate and final linear
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))
    return out
```

"""
TriMul (outgoing) forward implementation.

The heavy O(N³) contraction
    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
is a batched matrix‑multiply (left @ rightᵀ) for every hidden channel d.
We accelerate the element‑wise gating+masking with a tiny Triton kernel and
use the highly tuned cuBLAS batched GEMM (torch.bmm) for the N×N matrix
multiplications.  The overall pipeline is

    1) LayerNorm over the last dimension (float32)
    2) Linear projections → left/right (fp16)
    3) Gate/Mask fusion via Triton → gated_left/gated_right (fp16)
    4) Batched GEMM: (B*H, N, N) @ (B*H, N, N)ᵀ  (fp16, accum. fp32)
    5) Output LayerNorm, out‑gate, final Linear (float32)

Only the gating‑mask fusion is performed in Triton, satisfying the
“must use a kernel” requirement while keeping the dominant N³ work in the
highly‑optimized cuBLAS path.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: element‑wise gate + optional mask fusion
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    left_ptr, left_gate_ptr, mask_ptr,
    right_ptr, right_gate_ptr,
    out_left_ptr, out_right_ptr,
    total_elems: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # load tensors (all fp16)
    l = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    r = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    m = tl.load(mask_ptr + offs, mask=mask, other=tl.float16(1.0))

    out_l = l * lg * m
    out_r = r * rg * m

    tl.store(out_left_ptr + offs, out_l, mask=mask)
    tl.store(out_right_ptr + offs, out_r, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or numeric) – may be ignored if config["nomask"] is True
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16 for the heavy path
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias, fp16)
    # ------------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Gates (fp16)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()  # fp16, later cast

    # ------------------------------------------------------------------
    # 4. Optional mask handling
    # ------------------------------------------------------------------
    if nomask:
        # create a dummy mask of all ones that matches the left/right shape
        mask_fp = torch.ones_like(left, dtype=torch.float16, device=device)
    else:
        # mask shape [B, N, N] → broadcast to [B, N, N, hidden_dim]
        mask_fp = mask.unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous()
        mask_fp = mask_fp.to(torch.float16)

    # ------------------------------------------------------------------
    # 5. Fuse mask & gates with a Triton kernel
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_fp = mask_fp.contiguous()

    total = left.numel()                     # B * N * N * hidden_dim
    out_left = torch.empty_like(left)        # fp16
    out_right = torch.empty_like(right)      # fp16

    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    _gate_mask_kernel[grid](
        left,
        left_gate,
        mask_fp,
        right,
        right_gate,
        out_left,
        out_right,
        total,
        BLOCK=BLOCK,
    )
    # Release intermediate tensors (helps memory pressure)
    del left, right, left_gate, right_gate, mask_fp

    # ------------------------------------------------------------------
    # 6. Core contraction: batched GEMM using torch.bmm (cuBLAS)
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    B, N, _, H = out_left.shape  # (B, N, N, hidden_dim)

    # reshape to (B, H, N, N) and make right transposed (k,j)
    left_perm = out_left.permute(0, 3, 1, 2).contiguous()      # (B, H, N, N)  i,k
    right_perm = out_right.permute(0, 3, 2, 1).contiguous()    # (B, H, N, N)  k,j

    # flatten batch*head => (B*H, N, N)
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # batched matrix multiply (fp16, accumulation in fp32)
    out_flat = torch.bmm(left_flat, right_flat)               # (B*H, N, N)

    # reshape back to original layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 7. Output LayerNorm + out‑gate + final projection (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)   # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
