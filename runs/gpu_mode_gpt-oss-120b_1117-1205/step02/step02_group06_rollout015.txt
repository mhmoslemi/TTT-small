```
@triton.jit
def mask_gate_apply_kernel(left_ptr, right_ptr, mask_ptr,
                           left_gate_ptr, right_gate_ptr,
                           stride_left_b, stride_left_i, stride_left_j, stride_left_h,
                           stride_right_b, stride_right_i, stride_right_j, stride_right_h,
                           stride_mask_b, stride_mask_i, stride_mask_j,
                           stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
                           stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
                           N, H,
                           total_elements,
                           BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    b = offs // (N * N * H)
    rem = offs % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    j = rem2 // H
    h = rem2 % H

    # compute offsets
    left_off = left_ptr + b * stride_left_b + i * stride_left_i + j * stride_left_j + h * stride_left_h
    right_off = right_ptr + b * stride_right_b + i * stride_right_i + j * stride_right_j + h * stride_right_h
    left_gate_off = left_gate_ptr + b * stride_lgate_b + i * stride_lgate_i + j * stride_lgate_j + h * stride_lgate_h
    right_gate_off = right_gate_ptr + b * stride_rgate_b + i * stride_rgate_i + j * stride_rgate_j + h * stride_rgate_h
    mask_off = mask_ptr + b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    mask_val = tl.load(mask_off, mask=mask, other=0.0)
    left_val = tl.load(left_off, mask=mask, other=0.0)
    right_val = tl.load(right_off, mask=mask, other=0.0)
    left_gate_val = tl.load(left_gate_off, mask=mask, other=0.0)
    right_gate_val = tl.load(right_gate_off, mask=mask, other=0.0)

    # apply mask then gate
    left_out = left_val * mask_val * left_gate_val
    right_out = right_val * mask_val * right_gate_val

    tl.store(left_off, left_out, mask=mask)
    tl.store(right_off, right_out, mask=mask)

"""
TriMul (outgoing) forward pass – highly‑optimized implementation.

Algorithm
---------
1. Layer‑norm on the input tensor (float32 → float16).
2. Linear projections (left/right) and three gating linear layers
   (all in fp16).  Gates are passed through a sigmoid.
3. If a mask is present, a custom Triton kernel fuses:
      left  = left  * mask * left_gate
      right = right * mask * right_gate
   (in‑place, fp16).
   When `nomask` is True the mask step is skipped and only the gates are
   applied with a cheap PyTorch element‑wise multiply.
4. Core contraction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication A @ B where
      A = left.permute(0,3,1,2)  (B, H, N, N)
      B = right.permute(0,3,2,1) (B, H, N, N)
   We flatten the first two dimensions and call `torch.bmm`, which
   leverages cuBLAS Tensor‑Cores (fp16×fp16 → fp16 with fp32 accumulation).
5. Layer‑norm on the hidden dimension, multiplication with `out_gate`,
   and a final linear projection back to `dim`.
   The final output is float32.

Only the mask‑and‑gate fusion is performed with Triton; the heavy
batched GEMM uses the highly‑optimised cuBLAS kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _mask_gate_kernel(
    left_ptr, right_ptr, mask_ptr,
    left_gate_ptr, right_gate_ptr,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    N: tl.constexpr, H: tl.constexpr,
    total_elems,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elems

    # decode linear index -> (b,i,j,h)
    b = offs // (N * N * H)
    rem = offs % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    j = rem2 // H
    h = rem2 % H

    # compute raw pointers (int64 offsets)
    left_off = left_ptr + b * stride_left_b + i * stride_left_i + j * stride_left_j + h * stride_left_h
    right_off = right_ptr + b * stride_right_b + i * stride_right_i + j * stride_right_j + h * stride_right_h

    lgate_off = left_gate_ptr + b * stride_lgate_b + i * stride_lgate_i + j * stride_lgate_j + h * stride_lgate_h
    rgate_off = right_gate_ptr + b * stride_rgate_b + i * stride_rgate_i + j * stride_rgate_j + h * stride_rgate_h

    mask_off = mask_ptr + b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    # loads
    m = tl.load(mask_off, mask=valid, other=0.0)          # mask value (fp16)
    lv = tl.load(left_off, mask=valid, other=0.0)        # left element (fp16)
    rv = tl.load(right_off, mask=valid, other=0.0)       # right element (fp16)
    lgv = tl.load(lgate_off, mask=valid, other=0.0)      # left‑gate (fp16)
    rgv = tl.load(rgate_off, mask=valid, other=0.0)      # right‑gate (fp16)

    # apply mask and gates in‑place
    tl.store(left_off,  lv * m * lgv, mask=valid)
    tl.store(right_off, rv * m * rgv, mask=valid)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask: torch.Tensor, shape [B, N, N] (bool or float); may be ignored.
            - weights_dict: mapping from parameter names to torch.Tensor (float32)
            - config_dict: must contain 'dim' and 'hidden_dim';
                           optional 'nomask' (bool, default True).

    Returns:
        torch.Tensor of shape [B, N, N, dim], dtype float32.
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – fp16
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16)
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16)
    )

    # ------------------------------------------------------------------
    # 3. Gating (fp16) – sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Optional mask + gate fusion via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # ensure contiguous tensors
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp16 = mask.to(torch.float16).contiguous()

        B, N, _, H = left.shape
        total_elements = B * N * N * H
        BLOCK = 1024  # 32‑warps, good for fp16

        grid = (triton.cdiv(total_elements, BLOCK),)

        _mask_gate_kernel[grid](
            left, right, mask_fp16,
            left_gate, right_gate,
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
            left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
            right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
            N, H,
            total_elements,
            BLOCK=BLOCK,
        )
    else:
        # only apply gates (elementwise)
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction – batched GEMM via torch.bmm (fp16)
    # ------------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) = (k,j)

    B_batch, H_batch, N_seq, _ = left_perm.shape
    A = left_perm.view(B_batch * H_batch, N_seq, N_seq)
    B_mat = right_perm.view(B_batch * H_batch, N_seq, N_seq)

    # torch.bmm uses cuBLAS Tensor‑Cores for fp16 → fp16
    out_flat = torch.bmm(A, B_mat)                     # (B*H, N, N) fp16
    out = out_flat.view(B_batch, H_batch, N_seq, N_seq) \
                    .permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16)
    )
    out = out * out_gate                     # both fp16
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))  # -> float32

    return out
