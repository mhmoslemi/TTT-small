@triton.jit
def tri_mul_out_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,  # dimensions (int)
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    H: tl.constexpr
):
    batch_h = tl.program_id(0)  # combined (b*H + h)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # decode b and h
    b = batch_h // H
    h = batch_h - b * H

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers (including batch and hidden offsets)
    base_left = left_ptr + b * stride_left_batch + h * stride_left_h
    base_right = right_ptr + b * stride_right_batch + h * stride_right_h
    base_out = out_ptr + b * stride_out_batch + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # Load A tile (M x K) from left: dims (i,k)
        a_ptrs = base_left + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < N),
                    other=0.0)

        # Load B tile (K x N) from right as transposed: we need B_T[k,j] = right[b, j, k, h]
        # Right stride for row dimension (k) = stride_right_k, col dimension (j) = stride_right_i

        b = tl.load(b_ptrs,
                    mask=(offs_k < N)[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    # Store result
    out_ptrs = base_out + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs,
            acc.to(tl.float16),

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

Algorithm:
1. LayerNorm on the channel dimension (float32).
2. Linear projections (left/right) and gating (left_gate/right_gate/out_gate) in fp16.
3. Optional mask multiplication (broadcasted over the hidden dimension).
4. Core contraction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication (left @ rightᵀ) for each
   (batch, hidden) pair.  The kernel works on the original memory layout
   (no permute/contiguous copies) and uses block‑wise tiling with fp16
   loads and fp32 accumulation.
5. LayerNorm on the hidden dimension, out‑gate multiplication,
   and final linear projection back to the original dimension.
All heavy arithmetic is in fp16 with fp32 accumulation; the final output
is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction
# --------------------------------------------------------------
@triton.jit
def tri_mul_out_kernel(
    left_ptr, right_ptr, out_ptr,            # ptrs
    B, N,                                    # int scalars
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_k, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    H: tl.constexpr,                         # hidden dim (compile‑time)
):
    # Program IDs
    batch_h_id = tl.program_id(0)            # combined (batch, hidden)
    pid_m = tl.program_id(1)                 # tile row index
    pid_n = tl.program_id(2)                 # tile col index

    # Decode batch and hidden indices
    b = batch_h_id // H
    h = batch_h_id - b * H

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for this (b,h) pair
    base_left = left_ptr + b * stride_left_batch + h * stride_left_h
    base_right = right_ptr + b * stride_right_batch + h * stride_right_h
    base_out = out_ptr + b * stride_out_batch + h * stride_out_h

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile: left[b, i, k, h]
        a_ptrs = base_left + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < N),
                    other=0.0)

        # Load B tile (transposed): right_T[k, j] = right[b, j, k, h]
        b_ptrs = base_right + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_i
        b = tl.load(b_ptrs,
                    mask=(offs_k < N)[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # Store result as FP16
    out_ptrs = base_out + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_k
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def triton_tri_mul_out(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the TriMul contraction using the Triton kernel.
    left / right: tensors of shape [B, N, N, H] (fp16, contiguous)
    returns: tensor of shape [B, N, N, H] (fp16)
    """
    assert left.dtype == right.dtype == torch.float16
    B, N, N2, H = left.shape
    assert N == N2

    # Output tensor
    out = torch.empty_like(left)

    # Strides (as int64)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_ok, s_oh = out.stride()

    # Tile sizes (tuned for Hopper B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,                          # combined batch*hidden dimension
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    tri_mul_out_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_ri, s_rk, s_rh,
        s_ob, s_oi, s_ok, s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        H=H,
    )
    return out


# --------------------------------------------------------------
# Entry point expected by the evaluator
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operation.

    Arguments:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) – may be ignored
            - weights: dict of model parameters
            - config: dict containing 'dim', 'hidden_dim', and optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    # Configuration
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. LayerNorm over the channel dimension (FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections (no bias) – FP16
    # --------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 4. Gating values (sigmoid) – FP16
    # --------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction: batched GEMM (left @ rightᵀ) in Triton
    # --------------------------------------------------------------
    out_fp16 = triton_tri_mul_out(left, right)          # [B, N, N, H] fp16

    # --------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate multiplication, final projection
    # --------------------------------------------------------------
    out_fp32 = out_fp16.to(torch.float32)
    out_fp32 = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out_fp32 = out_fp32 * out_gate.to(torch.float32)
    out_fp32 = F.linear(out_fp32, weights["to_out.weight"].to(torch.float32))

    return out_fp32
