"""
TriMul (outgoing) forward pass with Triton‑accelerated elementwise fusion.

Algorithm
---------
1. Layer‑normalize the input tensor (FP32).
2. Cast to FP16 and compute the four linear projections:
   * left_proj, right_proj : (dim → hidden_dim)
   * left_gate, right_gate : (dim → hidden_dim) followed by sigmoid
   * out_gate              : (dim → hidden_dim) followed by sigmoid
3. Fuse elementwise multiplication of each projection with its gate and the
   optional mask in a single Triton kernel (two launches: left & right).
4. Perform the cubic contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batch of independent matrix multiplications for each
   hidden dimension, implemented with a single `torch.bmm` after reshaping.
5. Apply a second LayerNorm on the hidden dimension, multiply by `out_gate`,
   and project back to the original dimensionality with a final linear layer
   (all in FP32).

The kernel only handles the elementwise “* gate * mask” fusion; the heavy
linear layers and the batched GEMM rely on cuBLAS via PyTorch which is already
highly optimized.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def mul_gate_mask_kernel(
    out_ptr,        # pointer to output (flattened)
    a_ptr,          # pointer to tensor (flattened)
    gate_ptr,       # pointer to gate (flattened)
    mask_ptr,       # pointer to mask (flattened, broadcast over hidden dim)
    total,          # total number of elements
    BLOCK_SIZE: tl.constexpr,
    H: tl.constexpr,  # hidden dimension (size of the innermost axis)
):
    """Fuse `tensor * gate * mask` where mask is broadcast over the hidden axis."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    m = offs < total

    # load tensor and gate
    a = tl.load(a_ptr + offs, mask=m)
    g = tl.load(gate_ptr + offs, mask=m)

    # broadcast mask over the hidden dimension
    mask_index = offs // H                # index of (b,i,j) for this element
    mask_val = tl.load(mask_ptr + mask_index, mask=m)

    tl.store(out_ptr + offs, a * g * mask_val, mask=m)


def custom_kernel(data):
    """
    TriMul outgoing forward pass.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor: torch.Tensor [B, N, N, dim] (float32)
            mask_tensor : torch.Tensor [B, N, N] (float32) or None
            weights_dict: {
                "norm.weight": Tensor,
                "norm.bias": Tensor,
                "left_proj.weight": Tensor,
                "right_proj.weight": Tensor,
                "left_gate.weight": Tensor,
                "right_gate.weight": Tensor,
                "out_gate.weight": Tensor,
                "to_out_norm.weight": Tensor,
                "to_out_norm.bias": Tensor,
                "to_out.weight": Tensor,
            }
            config_dict : {"dim": int, "hidden_dim": int}
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # --------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # [B,N,N,dim]

    # --------------------------------------------------------------------
    # 2. Cast to FP16 for heavy work
    # --------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------------
    # 3. Linear projections (no bias)
    # --------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # [B,N,N,hidden_dim]
    right = F.linear(x_fp16, right_proj_w)  # [B,N,N,hidden_dim]

    # --------------------------------------------------------------------
    # 4. Gating + optional mask (fused in a Triton kernel)
    # --------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))    # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))  # [B,N,N,hidden_dim]
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))     # [B,N,N,hidden_dim]

    # Prepare mask (broadcast over hidden dimension)
    if mask_tensor is None:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(torch.float16).to(device)

    # Flatten tensors for the triton kernel
    left_flat = left.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_flat = right.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)
    mask_flat = mask_fp16.contiguous().view(-1)

    # Allocate output buffers
    left_out_flat = torch.empty_like(left_flat)
    right_out_flat = torch.empty_like(right_flat)

    # Launch Triton kernel to compute left = left * left_gate * mask
    BLOCK = 1024
    total = left_flat.numel()
    grid = (triton.cdiv(total, BLOCK),)

    mul_gate_mask_kernel[grid](
        left_out_flat,
        left_flat,
        left_gate_flat,
        mask_flat,
        total,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
    )
    # Compute right = right * right_gate * mask
    mul_gate_mask_kernel[grid](
        right_out_flat,
        right_flat,
        right_gate_flat,
        mask_flat,
        total,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
    )

    # Reshape back to [B,N,N,hidden_dim]
    left = left_out_flat.view(B, N, N, hidden_dim)
    right = right_out_flat.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------------
    # 5. Cubic contraction via batched GEMM (torch.bmm)
    # --------------------------------------------------------------------
    # Rearrange tensors to (B, hidden, N, N) where the matmul is over the
    # N dimension. Right is transposed so that the inner dimension aligns.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, hidden, N, N)

    LH = B * hidden_dim
    left_mat = left_perm.view(LH, N, N)    # (B*hidden, N, N)
    right_mat = right_perm.view(LH, N, N)  # (B*hidden, N, N)

    out_mat = torch.bmm(left_mat, right_mat)  # (B*hidden, N, N)

    # Restore layout [B,N,N,hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------------
    # 6. Post‑processing (LayerNorm on hidden dim, final gate, output proj)
    # --------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float32)
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply the final gating (out_gate computed in FP16)
    out_norm = out_norm * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float32)  # (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)  # [B, N, N, dim], float32

    return output
