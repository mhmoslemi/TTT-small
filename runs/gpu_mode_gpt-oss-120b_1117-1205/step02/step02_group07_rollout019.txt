@triton.jit
def trmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_d,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_d,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)  # combined batch*head index
    pid_m = tl.program_id(1) # i block
    pid_n = tl.program_id(2) # j block

    total_heads = B * H
    batch_id = pid // H
    head_id = pid % H

    # offsets within i and j
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers
    left_base = left_ptr + batch_id * stride_left_batch + head_id * stride_left_d
    right_base = right_ptr + batch_id * stride_right_batch + head_id * stride_right_d
    out_base = out_ptr + batch_id * stride_out_batch + head_id * stride_out_d

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k tile
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile: shape (M, K)
        left_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(left_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # load right tile transposed: shape (K, N)
        right_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        b = tl.load(right_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    # store result
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])
    # Cast to fp16
    x_fp16 = x.to(torch.float16)

    # Linear projections
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # Optional mask
    if not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)   # shape (B,N,N,1)
        left = left * mask_fp
        right = right * mask_fp

    # Gating
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Core contraction: trigMul outgoing
    B, N, _, H = left.shape

    # Allocate output
    out_fp16 = torch.empty_like(left)  # same shape and dtype

    # Strides
    left_stride = left.stride()
    right_stride = right.stride()
    out_stride = out_fp16.stride()

    # Define block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (B * H, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    trmul_outgoing_kernel[grid](
        left, right, out_fp16,
        B, N, H,
        left_stride[0], left_stride[1], left_stride[2], left_stride[3],
        right_stride[0], right_stride[1], right_stride[2], right_stride[3],
        out_stride[0], out_stride[1], out_stride[2], out_stride[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert to float32 for post-processing
    out = out_fp16.to(torch.float32)

    # Output LayerNorm
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])

    # Apply out_gate
    out_gate_fp32 = out_gate.to(torch.float32)
    out = out * out_gate_fp32

    # Final linear projection back to dim
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input (float32).
2. Cast to fp16 and apply linear projections + sigmoid gates.
   Optional mask is applied element‑wise.
3. Core contraction:
   out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   is a batched matrix multiplication for each (b,d) pair.
   The Triton kernel computes this without any transposes or copies:
   - The input tensors are in layout (B, N, N, H) (contiguous, H fastest).
   - For each (b,d) a GEMM of shape (N×N) @ (N×N)^T is performed.
   - fp16 inputs, fp32 accumulation, fp16 output.
4. Cast back to float32, apply output LayerNorm, out‑gate and final linear
   projection to the original channel dimension.

All heavy work (step 3) is performed in a single Triton kernel,
eliminating the costly permute+contiguous transposes from the reference
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trmul_outgoing_kernel(
    # Pointers
    left_ptr, right_ptr, out_ptr,
    # Dimensions
    B, N, H,
    # Strides for left tensor (B, N, N, H)
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    # Strides for right tensor (B, N, N, H)
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
    # Strides for output tensor (B, N, N, H)
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    # Block sizes (compile‑time constants)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # -------------------------------------------------
    # program IDs
    # -------------------------------------------------
    pid_bh = tl.program_id(0)          # combined batch * head index
    pid_m  = tl.program_id(1)          # block row (i)
    pid_n  = tl.program_id(2)          # block column (j)

    # Decode batch and head indices
    batch_id = pid_bh // H
    head_id  = pid_bh %  H

    # -------------------------------------------------
    # tile offsets
    # -------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this (batch, head)
    left_base  = left_ptr  + batch_id * stride_l_batch  + head_id * stride_l_h
    right_base = right_ptr + batch_id * stride_r_batch + head_id * stride_r_h
    out_base   = out_ptr   + batch_id * stride_o_batch + head_id * stride_o_h

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------
    # K‑loop
    # -------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: (i, k)
        left_ptrs = left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        a = tl.load(left_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # right tile transposed: we need right[k, j]
        # right layout is (i, k) → we treat i as j here
        right_ptrs = right_base + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i
        b = tl.load(right_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # fused fp16‑fp16 → fp32 dot
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # -------------------------------------------------
    # Write result (fp16)
    # -------------------------------------------------
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool / numeric) – may be omitted via config
            - weights_dict: dict with all required weights
            - config_dict: must contain 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim]

    # -------------------------------------------------
    # 2. Cast to fp16 for heavy linear ops
    # -------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -------------------------------------------------
    # 3. Linear projections (fp16)
    # -------------------------------------------------
    left  = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # -------------------------------------------------
    # 4. Optional mask (broadcast over hidden dim)
    # -------------------------------------------------
    if not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_fp
        right = right * mask_fp

    # -------------------------------------------------
    # 5. Gating (fp16) and apply
    # -------------------------------------------------
    left_gate  = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate   = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 6. Core contraction via Triton (batched GEMM)
    # -------------------------------------------------
    B, N, _, H = left.shape  # left.shape == (B, N, N, H)

    # Allocate output buffer (fp16)
    out_fp16 = torch.empty_like(left)

    # Strides (in elements)
    l_str = left.stride()   # (stride_batch, stride_i, stride_k, stride_h)
    r_str = right.stride()
    o_str = out_fp16.stride()

    # Tuned block sizes – work well for N up to 1024
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Grid: (batch*head, row_blocks, col_blocks)
    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    trmul_outgoing_kernel[grid](
        left, right, out_fp16,
        B, N, H,
        l_str[0], l_str[1], l_str[2], l_str[3],
        r_str[0], r_str[1], r_str[2], r_str[3],
        o_str[0], o_str[1], o_str[2], o_str[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -------------------------------------------------
    # 7. Post‑processing (float32)
    # -------------------------------------------------
    out = out_fp16.to(torch.float32)

    # Output LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    # Apply out‑gate
    out = out * out_gate.to(torch.float32)

    # Final linear projection back to original dim
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
