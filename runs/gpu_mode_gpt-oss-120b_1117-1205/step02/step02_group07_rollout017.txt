@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Get combined batch index
    batch_id = tl.program_id(0)  # total batches = B*H
    # Compute batch and head (channel) indices
    b = batch_id // H
    h = batch_id % H

    # Compute base pointers for this batch and head
    left_base = left_ptr + b * stride_left_batch + h * stride_left_h
    right_base = right_ptr + b * stride_right_batch + h * stride_right_h
    out_base = out_ptr + b * stride_out_batch + h * stride_out_h

    # Get block row/col indices
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile (i,k)
        a_ptrs = left_base + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile (k,j) (right transposed)
        b_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Store result in out (i,j)
    out_ptrs = out_base + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j

"""
TriMul (outgoing) forward pass with a fused Triton batched GEMM.

The core of the operator is
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication of (i,k) @ (k,j)^T for each
(b,d).  We launch a 3‑D Triton kernel where the first axis indexes the
combined (batch, head) dimension, the second axis tiles the output rows
(i) and the third axis tiles the output columns (j).  The kernel works
directly on the original [B,N,N,H] layout, using only stride arithmetic
– no extra copies or transposes – and accumulates in FP32 while
operating in FP16.  All other lightweight operations (LayerNorm,
linear projections, gating and final projection) are kept in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # left strides
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    # right strides (note transposed loading)
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
    # output strides
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr = 4,
):
    # -------------------------------------------------------------
    # 1) Identify the (batch, head) pair for this program instance
    # -------------------------------------------------------------
    batch_id = tl.program_id(0)                     # 0 <= batch_id < B*H
    b = batch_id // H                               # original batch index
    h = batch_id % H                                # hidden‑dim (head) index

    # -------------------------------------------------------------
    # 2) Tile indices for the output matrix (i, j)
    # -------------------------------------------------------------
    pid_m = tl.program_id(1)                        # tile row
    pid_n = tl.program_id(2)                        # tile column

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # -------------------------------------------------------------
    # 3) Base pointers for the selected (b, h) slice
    # -------------------------------------------------------------
    left_base = left_ptr + b * stride_left_batch + h * stride_left_h
    right_base = right_ptr + b * stride_right_batch + h * stride_right_h
    out_base = out_ptr + b * stride_out_batch + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # 4) Loop over the reduction dimension k
    # -------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A tile : left[b,i,k,h]  (i,k) layout
        a_ptrs = left_base + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B tile : right[b,j,k,h] transposed -> (k,j) layout
        b_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # -------------------------------------------------------------
    # 5) Write the result (FP16 output, FP32 accumulation)
    # -------------------------------------------------------------
    out_ptrs = out_base + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM driver for the outgoing TriMul core.
    left/right: [B, N, N, H] (float16)
    returns:    out  : [B, N, N, H] (float16)
    """
    assert left.dtype == right.dtype == torch.float16
    B, N, N2, H = left.shape
    assert N == N2

    out = torch.empty_like(left)

    # Strides in element units
    sL = left.stride()
    sR = right.stride()
    sO = out.stride()

    # Block sizes (tuned for typical NVIDIA GPUs; adjust if needed)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        # left strides
        sL[0],
        sL[1],
        sL[2],
        sL[3],
        # right strides (j corresponds to original i)
        sR[0],
        sR[1],
        sR[2],
        sR[3],
        # output strides
        sO[0],
        sO[1],
        sO[2],
        sO[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/int) – may be ignored if config["nomask"]
            - weights_dict: mapping of parameter names to torch.Tensor
            - config_dict: must contain 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim], float32

    # -------------------------------------------------------------
    # 2. Cast to fp16 for heavy linear work
    # -------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -------------------------------------------------------------
    # 3. Linear projections (fp16)
    # -------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # -------------------------------------------------------------
    # 4. Optional mask (broadcast over hidden dimension)
    # -------------------------------------------------------------
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)  # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # -------------------------------------------------------------
    # 5. Gating (fp16) and apply
    # -------------------------------------------------------------
    left_gate = F.linear(
        x_fp16, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = F.linear(
        x_fp16, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = F.linear(
        x_fp16, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------
    # 6. Core contraction via Triton (fp16 in, fp16 out)
    # -------------------------------------------------------------
    out_fp16 = tri_mul_triton(left, right)          # [B, N, N, H] fp16

    # -------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, and final projection (float32)
    # -------------------------------------------------------------
    out = out_fp16.to(torch.float32)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
