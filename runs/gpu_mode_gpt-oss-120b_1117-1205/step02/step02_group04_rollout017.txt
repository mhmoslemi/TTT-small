import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_gemm_kernel(a_ptr, b_ptr, c_ptr,
                        M, N, K,
                        stride_am, stride_ak, stride_ah,
                        stride_bk, stride_bn, stride_bh,
                        stride_cm, stride_cn, stride_ch,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + batch * stride_ah
    b_batch = b_ptr + batch * stride_bh
    c_batch = c_ptr + batch * stride_ch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        # Load B tile (K x N)
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # accumulate in FP32

    # Store result
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def triton_batched_gemm(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B using Triton.
    A, B: (batch, M, K) and (batch, K, N) in fp16.
    Returns C: (batch, M, N) in fp16.
    """
    assert A.is_contiguous() and B.is_contiguous()
    assert A.dtype == B.dtype == torch.float16
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bk, stride_bn = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    # Tunable tile size (empirically good for B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (batch,
            triton.cdiv(M, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    batched_gemm_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_am,
        stride_ak,
        stride_ah,
        stride_bk,
        stride_bn,
        stride_bh,
        stride_cm,
        stride_cn,
        stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/int)
            - weights_dict: contains all required weight tensors
            - config_dict: must include 'dim' and 'hidden_dim', optionally 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16 for the heavy compute
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (fp16)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # (B, N, N, hidden_dim)
    right = F.linear(x_fp16, right_proj_w)   # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 4. Gating (fp16)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()   # (B, N, N, hidden_dim)

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N^3) contraction via Triton batched GEMM
    #    Compute out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    # Rearrange tensors to (B, H, N, N) where H = hidden_dim.
    # left: (i,k) , right: (k,j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    total_batch = B * H

    # Collapse batch and hidden dimensions.
    left_flat = left_perm.view(total_batch, N, N)
    right_flat = right_perm.view(total_batch, N, N)

    # Batched matrix multiplication in fp16 (FP32 accumulation).
    out_flat = triton_batched_gemm(left_flat, right_flat)   # (total_batch, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)   # subsequent ops use fp32

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # Apply the output gate
    out = out * out_gate.to(out.dtype)

    # Final linear projection back to original dimension
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The heavy O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched GEMM using a Triton kernel.
All other operations (LayerNorm, linear projections, gating) are performed with
PyTorch (mixed‑precision) for simplicity and numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ B   (A: (B, M, K), B: (B, K, N))
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ah,
    stride_bk, stride_bn, stride_bh,
    stride_cm, stride_cn, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ B for many independent batches.
    A, B, C are expected to be FP16 tensors; accumulation is FP32.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + batch_id * stride_ah
    b_batch = b_ptr + batch_id * stride_bh
    c_batch = c_ptr + batch_id * stride_ch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k < K),
            other=0.0,
        )

        # Load B tile (K x N)
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(
            b_ptrs,
            mask=(offs_k < K)[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Accumulate
        acc += tl.dot(a, b)   # fp16 dot, result accumulated into fp32

    # Store result (cast back to fp16)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )

    """

    assert A.dtype == B.dtype == torch.float16
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides in elements
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bk, stride_bn = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    # Tile sizes (tuned for the B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_gemm_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_am,
        stride_ak,
        stride_ah,
        stride_bk,
        stride_bn,
        stride_bh,
        stride_cm,
        stride_cn,
        stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/int); may be ignored if nomask=True
            - weights_dict: mapping from parameter names to torch.Tensors
            - config_dict: must contain 'dim' and 'hidden_dim', may contain 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_fp16 = x.to(torch.float16)   # cast to fp16 for the heavy work

    # ------------------------------------------------------------------
    # 2. Linear projections (fp16, no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # shape [B, N, N, hidden_dim]
    right = F.linear(x_fp16, right_proj_w)   # shape [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Optional masking
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 4. Gating (fp16)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()   # (B, N, N, hidden_dim)

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³) contraction via Triton batched GEMM
    #    Compute out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) where H = hidden_dim.
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N) : (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) : (k,j)

    B, H, N, _ = left_perm.shape
    total_batch = B * H

    # Collapse batch and hidden dimensions.
    left_flat = left_perm.view(total_batch, N, N)      # (total_batch, N, N)
    right_flat = right_perm.view(total_batch, N, N)    # (total_batch, N, N)

    # Batched GEMM in fp16 (with fp32 accumulation).
    out_flat = triton_batched_gemm(left_flat, right_flat)   # (total_batch, N, N)

    # Restore layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)   # subsequent layers work in fp32

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # Apply output gate (cast to fp32 for the multiplication)
    out = out * out_gate.to(out.dtype)

    # Final projection back to original `dim`
    out = F.linear(out, weights["to_out.weight"])

    return out
