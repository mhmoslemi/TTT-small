def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    # maybe more
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    a_ptr_batch = a_ptr + batch_id * batch_stride_a
    b_ptr_batch = b_ptr + batch_id * batch_stride_b
    c_ptr_batch = c_ptr + batch_id * batch_stride_c

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # k loop unrolled
    for k in range(0, K, BLOCK_K):
        a = tl.load(a_ptr_batch + (offs_m[:, None] * stride_am) + (k + tl.arange(0, BLOCK_K))[None, :] * stride_ak,

                         mask=mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)

    c = acc.to(tl.float16)
    tl.store(c_ptr_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn),

def custom_kernel(data):

    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    x_norm = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x_fp16 = x_norm.to(torch.float16)

    # linear
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # mask
    nomask = config.get("nomask", True)
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_fp
        right = right * mask_fp

    # gating
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    right = right * right_gate

    # Permute for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # B, H, N, N
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # B, H, N, N (k,j)

    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Use torch.bmm (cuBLAS)

    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

def fuse_mul_kernel(
    left_ptr, left_gate_ptr, right_ptr, right_gate_ptr,
    mask_ptr, out_left_ptr, out_right_ptr,

"""
TriMul (outgoing) forward pass.

Algorithm
---------
1. LayerNorm on the input (float32).
2. Linear projections to the hidden dimension in fp16.
3. Compute gating values (fp16 sigmoid).
4. Fuse mask (if any) and gating via a small Triton kernel
   `fuse_mul_kernel` which performs   out = proj * gate * mask   in one pass.
   This reduces the number of full‑tensor passes from three (mask, gate, multiply)
   to a single pass.
5. Core contraction: for every head `d` we need
       out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   which is a batched matrix multiplication (B·H independent GEMMs of size N×N).
   It is efficiently carried out by `torch.bmm`, which maps to cuBLAS tensor‑core
   GEMM and gives near‑peak performance.
6. Output LayerNorm (float32), out‑gate scaling and final linear projection
   back to the original dimension.

A Triton kernel is used for the fused elementwise multiplication (step 4), satisfying
the requirement that “part of the operations” are implemented in Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel that fuses: out = proj * gate * mask   (mask optional)
# ----------------------------------------------------------------------
@triton.jit
def fuse_mul_kernel(
    proj_ptr,          # projection tensor (fp16)
    gate_ptr,          # gate tensor (fp16)
    mask_ptr,          # mask tensor (fp16) – ignored when HAS_MASK==0
    out_ptr,           # output tensor (fp16)
    numel: tl.constexpr,          # total number of elements
    BLOCK_SIZE: tl.constexpr,     # thread‑block size
    HAS_MASK: tl.constexpr         # compile‑time flag (0/1)
):
    idx = tl.program_id(0) * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = idx < numel

    proj = tl.load(proj_ptr + idx, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + idx, mask=mask, other=0.0)

    if HAS_MASK:
        m = tl.load(mask_ptr + idx, mask=mask, other=0.0)
        out = proj * gate * m
    else:
        out = proj * gate

    tl.store(out_ptr + idx, out, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor : torch.Tensor of shape [B, N, N] (bool or numeric)
            - weights_dict: mapping from weight names to torch.Tensor
            - config_dict : contains 'dim', 'hidden_dim' and optionally 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim], float32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 and project
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    left_proj = F.linear(
        x_fp16, weights["left_proj.weight"].to(torch.float16)
    )  # [B, N, N, hidden_dim]
    right_proj = F.linear(
        x_fp16, weights["right_proj.weight"].to(torch.float16)
    )  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Gating (fp16 sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_fp16, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = F.linear(
        x_fp16, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = F.linear(
        x_fp16, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()  # will be used later (still fp16)

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) and gating via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left_proj.shape
    total_elems = left_proj.numel()  # B * N * N * hidden_dim

    # prepare mask tensor (broadcasted over hidden_dim) only when needed
    if not nomask:
        # shape [B, N, N, 1] -> broadcast to hidden_dim with stride 0 in last dim
        mask_fp = mask.unsqueeze(-1).to(torch.float16)
        mask_fp = mask_fp.expand(-1, -1, -1, hidden_dim).contiguous()
        mask_ptr = mask_fp
        has_mask = 1
    else:
        # dummy pointer – never read because HAS_MASK=0
        mask_ptr = left_proj  # any fp16 tensor of compatible size
        has_mask = 0

    BLOCK = 1024  # reasonable tile size for elementwise work
    grid = (triton.cdiv(total_elems, BLOCK),)

    # fused left
    left = torch.empty_like(left_proj)
    fuse_mul_kernel[grid](
        left_proj,
        left_gate,
        mask_ptr,
        left,
        numel=total_elems,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=has_mask,
    )
    # fused right
    right = torch.empty_like(right_proj)
    fuse_mul_kernel[grid](
        right_proj,
        right_gate,
        mask_ptr,
        right,
        numel=total_elems,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=has_mask,
    )
    # (proj, gate, mask tensors are now free for garbage collection)

    # ------------------------------------------------------------------
    # 5. Core contraction via batched GEMM (torch.bmm → cuBLAS)
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) where the second matrix is already transposed
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()        # (B, H, N, N) -> (K, N)

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)      # (B·H, N, K)
    right_flat = right_perm.view(B * H, N, N)    # (B·H, K, N)   (already transposed)

    # batched matrix multiplication in fp16, accumulation in fp32 (cuBLAS tensor cores)
    out_flat = torch.bmm(left_flat, right_flat)   # (B·H, N, N), fp16

    # reshape back to [B, N, N, H] and promote to float32 for the rest
    out = (
        out_flat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
