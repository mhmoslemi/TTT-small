def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)

    # Cast weights to half and appropriate device
    # Normalization weights
    norm_weight = weights["norm.weight"].to(device=device, dtype=torch.float16)
    norm_bias = weights["norm.bias"].to(device=device, dtype=torch.float16)
    # Linear projection combined weights
    left_combined_weight = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(device=device, dtype=torch.float16)
    right_combined_weight = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(device=device, dtype=torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_weight = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # 1. Input layernorm (float32 -> half)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight, bias=norm_bias
    )
    x_h = x.to(torch.float16)

    # 2. Compute out_gate early
    out_gate = torch.nn.functional.linear(x_h, out_gate_weight).sigmoid()  # shape [B,N,N,hidden]

    # 3. Combined projection for left (proj+gate) and right
    left_combined = torch.nn.functional.linear(x_h, left_combined_weight)  # [B,N,N,2H]
    right_combined = torch.nn.functional.linear(x_h, right_combined_weight)  # [B,N,N,2H]

    # split
    left = left_combined[..., :hidden_dim]   # [B,N,N,H]
    left_gate_logits = left_combined[..., hidden_dim:]  # [B,N,N,H]
    left_gate = left_gate_logits.sigmoid()

    right = right_combined[..., :hidden_dim]
    right_gate_logits = right_combined[..., hidden_dim:]
    right_gate = right_gate_logits.sigmoid()

    # 4. Apply mask if present (mask is [B,N,N] possibly)
    if not nomask:
        # mask may be boolean or numeric; cast to half
        mask_h = mask.unsqueeze(-1).to(dtype=torch.float16, device=device)
        left = left * mask_h
        right = right * mask_h

    # 5. Apply gates
    left = left * left_gate
    right = right * right_gate

    # free combined and gate logits maybe
    # 6. Core contraction using batched GEMM
    # Permute for batch GEMM
    B, N, _, _ = left.shape  # B,N,N,H
    # left_perm
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # transpose last two dims
    # reshape to (B*H, N, N)
    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)  # already transposed

    # batched matmul
    out_mat = torch.bmm(left_mat, right_mat)  # shape (B*H,N,N)
    # reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # 7. Output layernorm (half)
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias
    )

    # 8. Elementwise multiply with out_gate using Triton kernel
    out_scaled = torch.empty_like(out_norm)
    total = out_norm.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    _elementwise_mul_kernel[grid](
        out_norm, out_gate, out_scaled,
        total,
        BLOCK=BLOCK,
    )

    # 9. Final linear projection
    out_final = torch.nn.functional.linear(out_scaled, to_out_weight)

    # Return float32
    return out_final.to(torch.float32)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _elementwise_mul_kernel(a_ptr, b_ptr, out_ptr, total_elements: tl.int32, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements
    a = tl.load(a_ptr + offs, mask=mask, other=tl.float16(0.0))
    b = tl.load(b_ptr + offs, mask=mask, other=tl.float16(0.0))
    tl.store(out_ptr + offs, a * b, mask=mask)

def custom_kernel(data):
    """
    Outgoing TriMul forward pass (no gradient).
    Uses fused projection+gate linear layers, optional mask,
    batched contraction via torch.einsum, and a small Triton kernel
    for the final element‑wise scaling.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----- LayerNorm on input -----
    norm_weight = weights["norm.weight"].to(device=device, dtype=torch.float16)
    norm_bias = weights["norm.bias"].to(device=device, dtype=torch.float16)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x_h = x.to(torch.float16)

    # ----- Out gate (shared for all paths) -----
    out_gate_weight = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate = F.linear(x_h, out_gate_weight).sigmoid()   # [B, N, N, H]

    # ----- Combined left proj + left gate -----
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    left_combined_weight = torch.cat([left_proj_w, left_gate_w], dim=0).to(device=device, dtype=torch.float16)
    left_combined = F.linear(x_h, left_combined_weight)   # [B, N, N, 2H]
    left = left_combined[..., :hidden_dim]                # proj
    left_gate = left_combined[..., hidden_dim:].sigmoid() # gate

    # ----- Combined right proj + right gate -----
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    right_combined_weight = torch.cat([right_proj_w, right_gate_w], dim=0).to(device=device, dtype=torch.float16)
    right_combined = F.linear(x_h, right_combined_weight) # [B, N, N, 2H]
    right = right_combined[..., :hidden_dim]
    right_gate = right_combined[..., hidden_dim:].sigmoid()

    # ----- Optional mask -----
    if not nomask:
        # mask shape: [B, N, N] -> broadcast over hidden dim
        mask_h = mask.unsqueeze(-1).to(dtype=torch.float16, device=device)
        left = left * mask_h
        right = right * mask_h

    # ----- Apply gating -----
    left = left * left_gate
    right = right * right_gate

    # ----- Core contraction (batched outer product over k) -----
    # Use einsum to avoid extra data copies.
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)  # [B, N, N, H]

    # ----- Output LayerNorm -----
    to_out_norm_weight = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # ----- Element‑wise scaling with out_gate (Triton) -----
    out_scaled = torch.empty_like(out_norm)
    total = out_norm.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    _elementwise_mul_kernel[grid](
        out_norm,
        out_gate,
        out_scaled,
        total,
        BLOCK=BLOCK,
    )

    # ----- Final projection -----
    to_out_weight = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out_final = F.linear(out_scaled, to_out_weight)

    # Return float32 as required
    return out_final.to(torch.float32)

"""
Outgoing TriMul forward pass (no gradient).

Key ideas:
* Fuse the four projection/gate Linear layers into a single Linear call with
  a stacked weight matrix, reducing memory traffic.
* Use a small Triton kernel only for the final element‑wise scaling
  (out_norm * out_gate), satisfying the “must use a kernel” requirement.
* Perform the heavy N³ contraction with a single torch.einsum, which avoids
  explicit tensor transposes and extra copies.
* All heavy arithmetic is carried out in float16; the final result is cast
  back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _elementwise_mul_kernel(
    a_ptr,            # tensor a (half)
    b_ptr,            # tensor b (half)
    out_ptr,          # destination (half)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    a = tl.load(a_ptr + offs, mask=mask, other=tl.float16(0.0))
    b = tl.load(b_ptr + offs, mask=mask, other=tl.float16(0.0))
    tl.store(out_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or numeric) – may be ignored
            - weights: mapping of parameter names to torch.Tensor
            - config: contains "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Cast model weights to the compute dtype (float16)
    # ------------------------------------------------------------------
    w_norm = {
        "weight": weights["norm.weight"].to(device=device, dtype=torch.float16),
        "bias":   weights["norm.bias"].to(device=device, dtype=torch.float16),
    }
    w_to_out_norm = {
        "weight": weights["to_out_norm.weight"].to(device=device, dtype=torch.float16),
        "bias":   weights["to_out_norm.bias"].to(device=device, dtype=torch.float16),
    }
    w_out_gate = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    w_to_out = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # Stack the four projection/gate weights:
    #   left_proj, left_gate, right_proj, right_gate  (each [H, dim])
    w_stacked = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=w_norm["weight"],
        bias=w_norm["bias"],
    )
    x_h = x.to(torch.float16)  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Compute out‑gate (shared for all paths)
    # ------------------------------------------------------------------
    out_gate = F.linear(x_h, w_out_gate).sigmoid()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Combined projection & gating for left/right sides
    # ------------------------------------------------------------------
    combined = F.linear(x_h, w_stacked)  # [B, N, N, 4*H]

    # Split the concatenated result
    left = combined[..., :hidden_dim]                                 # proj
    left_gate = combined[..., hidden_dim:2 * hidden_dim].sigmoid()    # gate
    right = combined[..., 2 * hidden_dim:3 * hidden_dim]              # proj
    right_gate = combined[..., 3 * hidden_dim:].sigmoid()             # gate

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask.unsqueeze(-1).to(dtype=torch.float16, device=device)
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 5. Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Core N³ contraction:   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented with a single torch.einsum to avoid extra copies.
    # ------------------------------------------------------------------
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7. Output LayerNorm (half‑precision)
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w_to_out_norm["weight"],
        bias=w_to_out_norm["bias"],
    )

    # ------------------------------------------------------------------
    # 8. Element‑wise scaling with out_gate (Triton kernel)
    # ------------------------------------------------------------------
    out_scaled = torch.empty_like(out_norm)
    total = out_norm.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    _elementwise_mul_kernel[grid](
        out_norm,
        out_gate,
        out_scaled,
        total,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 9. Final Linear projection back to `dim`
    # ------------------------------------------------------------------
    out_final = F.linear(out_scaled, w_to_out)  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 10. Cast back to float32 as required by the spec
    # ------------------------------------------------------------------
    return out_final.to(torch.float32)
