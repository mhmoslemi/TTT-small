def _triton_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
   batch_id = tl.program_id(0)
   pid_i = tl.program_id(1)
   pid_j = tl.program_id(2)

   offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

   mask_i = offs_i < N
   mask_j = offs_j < N

   # base pointers for this batch
   left_batch_ptr = left_ptr + batch_id * stride_l_batch
   right_batch_ptr = right_ptr + batch_id * stride_r_batch
   out_batch_ptr = out_ptr + batch_id * stride_o_batch

   for h_start in range(0, H, BLOCK_H):
       cur_h = tl.minimum(tl.arange(0, BLOCK_H), H - h_start - 1) # not used for mask directly
       # but block may go out-of-range; compute mask_h
       mask_h = (h_start + tl.arange(0, BLOCK_H)) < H

       # accumulator
       acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

       for k in range(0, tl.cdiv(N, BLOCK_K)):
           k_offset = k * BLOCK_K
           offs_k = k_offset + tl.arange(0, BLOCK_K)
           mask_k = offs_k < N

           # left tile: shape (BLOCK_M, BLOCK_K, BLOCK_H)
           left_ptrs = left_batch_ptr + \
                       offs_i[:, None, None] * stride_l_i + \
                       offs_k[None, :, None] * stride_l_k + \
                       (h_start + tl.arange(0, BLOCK_H))[None, None, :] * stride_l_h
           left_tile = tl.load(left_ptrs,
                               mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                               other=0.0)

           # right tile: shape (BLOCK_K, BLOCK_N, BLOCK_H)
           # we need rows = k, cols = j
           right_ptrs = right_batch_ptr + \
                        offs_k[:, None, None] * stride_r_k + \
                        offs_j[None, :, None] * stride_r_i + \
                        (h_start + tl.arange(0, BLOCK_H))[None, None, :] * stride_r_h
           right_tile = tl.load(right_ptrs,
                                mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
                                other=0.0)

           # compute per hidden
           # loop over hidden inside block for accumulation
           for hh in range(0, BLOCK_H):
               acc[:, :, hh] += tl.dot(left_tile[:, :, hh], right_tile[:, :, hh])
       # store acc
       out_ptrs = out_batch_ptr + \
                 offs_i[:, None, None] * stride_o_i + \
                 offs_j[None, :, None] * stride_o_j + \
                 (h_start + tl.arange(0, BLOCK_H))[None, None, :] * stride_o_h
       tl.store(out_ptrs,
                acc.to(tl.float16),
                mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])

"""
TriMul outgoing forward pass (AlphaFold3 style)

Algorithm
---------
1. Layer‑norm the input over the ``dim`` channel.
2. FP16 linear projections (left, right) and three sigmoid gates.
3. Optional mask is applied to the projected tensors.
4. Gates are applied element‑wise.
5. Core contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for every hidden slice.
   The contraction is performed by a custom Triton kernel that
   works directly on the original [B, N, N, H] layout, avoiding the
   costly permutation / copy used in a naïve implementation.  The
   kernel processes a tile of size (BLOCK_M, BLOCK_N) of the i‑j
   plane and a small block (BLOCK_H) of the hidden dimension at a
   time, using fp16 arithmetic with fp32 accumulation.
6. fp16 → fp32, output LayerNorm, out‑gate multiplication and final
   linear projection back to ``dim`` channels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fused batched GEMM over the hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def _triton_outgoing_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    # ---------------------------- program ids ---------------------------
    batch_id = tl.program_id(0)          # batch index (0 … B‑1)
    pid_i    = tl.program_id(1)          # tile row   (0 … ceil(N/BLOCK_M)-1)
    pid_j    = tl.program_id(2)          # tile col   (0 … ceil(N/BLOCK_N)-1)

    # --------------------------- offsets --------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this batch
    left_batch  = left_ptr  + batch_id * stride_l_batch
    right_batch = right_ptr + batch_id * stride_r_batch
    out_batch   = out_ptr   + batch_id * stride_o_batch

    # iterate over hidden dimension in small blocks
    h_start = 0
    while h_start < H:
        # offsets for the current hidden block
        h_offsets = h_start + tl.arange(0, BLOCK_H)
        mask_h = h_offsets < H

        # accumulator for this hidden block
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

        # ----------- K‑loop (pairwise sequence reduction) ---------------
        for k in range(0, tl.cdiv(N, BLOCK_K)):
            k_off = k * BLOCK_K
            offs_k = k_off + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # ----- load left tile: shape (M, K, H) --------------------
            left_ptrs = (left_batch
                         + offs_i[:, None, None] * stride_l_i
                         + offs_k[None, :, None] * stride_l_k
                         + h_offsets[None, None, :] * stride_l_h)
            left_tile = tl.load(left_ptrs,
                                mask=mask_i[:, None, None] &
                                     mask_k[None, :, None] &
                                     mask_h[None, None, :],
                                other=0.0)

            # ----- load right tile (k as row, j as col): shape (K, N, H) -----
            right_ptrs = (right_batch
                          + offs_k[:, None, None] * stride_r_k
                          + offs_j[None, :, None] * stride_r_i
                          + h_offsets[None, None, :] * stride_r_h)
            right_tile = tl.load(right_ptrs,
                                 mask=mask_k[:, None, None] &
                                      mask_j[None, :, None] &
                                      mask_h[None, None, :],
                                 other=0.0)

            # ----- dot for each hidden slice ---------------------------------
            for hh in range(0, BLOCK_H):
                acc[:, :, hh] += tl.dot(left_tile[:, :, hh],
                                        right_tile[:, :, hh])

        # ---------------------- store the block --------------------------
        out_ptrs = (out_batch
                    + offs_i[:, None, None] * stride_o_i
                    + offs_j[None, :, None] * stride_o_j
                    + h_offsets[None, None, :] * stride_o_h)
        tl.store(out_ptrs,
                 acc.to(tl.float16),
                 mask=mask_i[:, None, None] &
                      mask_j[None, :, None] &
                      mask_h[None, None, :])

        h_start += BLOCK_H  # next hidden block

def _triton_outgoing_trimul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right: (B, N, N, H)  fp16 tensors
    returns out: (B, N, N, H)  fp16 tensor
    """
    assert left.dtype == torch.float16 and right.dtype == torch.float16
    B, N, N2, H = left.shape
    assert N == N2
    out = torch.empty((B, N, N, H), dtype=torch.float16, device=left.device)

    # strides for the three tensors (as Python integers)
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tunable tile sizes – chosen to fit the B200 memory hierarchy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_H = 8   # H ≤ 384, so 8‑way blocking gives good occupancy

    grid = (B,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    _triton_outgoing_trimul_kernel[grid](
        left, right, out,
        N, H,
        stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
        stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
        stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )
    return out

# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the *outgoing* TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored
            - weights_dict: mapping name → torch.Tensor (all float32)
            - config_dict: must contain ``dim`` and ``hidden_dim``.
              Optional ``nomask`` (bool) indicates that mask is not used.

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm  (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for all heavy ops
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1]
        mask_fp = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 5. Gates (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # Apply element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Core contraction using the fused Triton kernel
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    out = _triton_outgoing_trimul(left, right)               # fp16, shape (B,N,N,H)

    # ------------------------------------------------------------------
    # 7. Output processing (float32)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    out_fp32 = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out_fp32 = out_fp32 * out_gate.to(torch.float32)
    out_fp32 = F.linear(out_fp32, weights["to_out.weight"])

    return out_fp32
