"""Implementation of the outgoing TriMul operator with a fused Triton batched GEMM.
The forward pass follows:
1. LayerNorm on input (float32).
2. Linear projections (left/right) to hidden_dim, and gating linear layers.
3. Optional mask applied to left/right.
4. Contract = sum_k left[..., i, k, d] * right[..., j, k, d] computed as
   a batched matrix multiplication for each hidden dimension using a
   custom Triton kernel (fp16 inputs, fp32 accumulation, fp16 output).
5. Output LayerNorm, out_gate multiplication, and final linear projection
   back to the original dim.
All heavy compute (the N×N matrix multiplication) is performed in the
Triton kernel.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------
# Triton batched GEMM kernel (C = A @ B, where B is already transposed)
# ------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute C = A @ B for a batch of matrices.
    A has shape (batch, M, K), B has shape (batch, K, N), C will have shape (batch, M, N).
    All tensors are stored in row‑major order.
    """
    batch_id = tl.program_id(0)   # batch dimension
    pid_m = tl.program_id(1)      # block row
    pid_n = tl.program_id(2)      # block column

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles of size BLOCK_K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        # Load B tile (K x N) – note B is already transposed
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Write back the result in fp16
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def triton_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B for fp16 tensors.
    a: (batch, M, K) fp16
    b: (batch, K, N) fp16 (already transposed as needed)
    Returns:
        c: (batch, M, N) fp16
    """
    assert a.dtype == b.dtype == torch.float16, "batched_matmul expects fp16 tensors"
    assert a.is_contiguous() and b.is_contiguous()
    batch, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2, "Inner dimensions must match"

    c = torch.empty((batch, M, N), dtype=torch.float16, device=a.device)

    # Strides in elements
    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_k, stride_b_n = b.stride()
    stride_c_batch, stride_c_m, stride_c_n = c.stride()

    # Tile sizes – tuned for typical sequence lengths up to 1024
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        a, b, c,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric)
            - weights_dict: mapping of parameter names to torch.Tensor
            - config_dict: must contain 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -------------------------------------------------
    # LayerNorm over the last channel dimension
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim] float32

    # -------------------------------------------------
    # 2. Cast to fp16 for heavy linear ops
    # -------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -------------------------------------------------
    # 3. Linear projections (fp16)
    # -------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # -------------------------------------------------
    # 4. Optional mask (broadcast over hidden dim)
    # -------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] -> [B, N, N, 1]
        mask_fp = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp
        right = right * mask_fp

    # -------------------------------------------------
    # 5. Gates (fp16) and apply
    # -------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 6. Core contraction via Triton batched GEMM
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # -------------------------------------------------
    # Rearrange for batched GEMM: (B, H, N, N) where H = hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)  i,k
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)  k,j

    B, H, N, _ = left_perm.shape
    # Flatten batch and head dimensions
    left_flat = left_perm.view(B * H, N, N)   # (B*H, M=N, K=N)
    right_flat = right_perm.view(B * H, N, N) # (B*H, K=N, N)

    # Perform batched matrix multiplication (fp16 inputs, fp16 output)
    out_flat = triton_matmul(left_flat, right_flat)  # (B*H, N, N) fp16

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # Convert to fp32 for subsequent LayerNorm

    # -------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final projection (float32)
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)  # broadcast over last dim
    out = F.linear(out, weights["to_out.weight"])

    return out

import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (C = A @ B) where B is already transposed
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute C = A @ B for a batch of matrices.

    Shapes:
        A: (batch, M, K)
        B: (batch, K, N)   # B is already transposed compared to the original right tensor
        C: (batch, M, N)

    All tensors are row‑major.  Accumulation is performed in FP32 and the
    result is stored in FP16.
    """
    batch_id = tl.program_id(0)   # batch index
    pid_m = tl.program_id(1)      # block row index
    pid_n = tl.program_id(2)      # block column index

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for the current batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension in tiles of size BLOCK_K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        # Load B tile (K x N) – B is already transposed (k, n)
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # Write back the result as FP16
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def triton_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B using the Triton kernel above.
    Both inputs must be contiguous FP16 tensors.

    Args:
        a: (batch, M, K) FP16
        b: (batch, K, N) FP16 (already transposed as required)

    Returns:
        c: (batch, M, N) FP16
    """
    assert a.dtype == b.dtype == torch.float16, "triton_matmul expects FP16 tensors"
    assert a.is_contiguous() and b.is_contiguous()
    batch, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2, "inner dimensions must match"

    c = torch.empty((batch, M, N), dtype=torch.float16, device=a.device)

    # Strides in element counts (not bytes)
    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_k, stride_b_n = b.stride()
    stride_c_batch, stride_c_m, stride_c_n = c.stride()

    # Tile sizes that give good occupancy up to N=1024
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        a, b, c,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c

# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or numeric)
            - weights_dict: mapping of parameter names to torch.Tensor
            - config_dict: must contain 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # shape: [B, N, N, dim], dtype: float32

    # -------------------------------------------------
    # 2. Cast to FP16 for the heavy linear paths
    # -------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -------------------------------------------------
    # 3. Linear projections (left/right) – no bias
    # -------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # -------------------------------------------------
    # 4. Optional mask (broadcast over hidden dimension)
    # -------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [B, N, N, 1] and cast to fp16
        mask_fp = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp
        right = right * mask_fp

    # -------------------------------------------------
    # 5. Gating projections (sigmoid) – still in FP16
    # -------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 6. Core contraction via Triton batched GEMM
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # -------------------------------------------------
    # Rearrange tensors to (B, H, N, N) where H = hidden_dim:
    #   left  : (i, k)   -> (B, H, N, N)
    #   right : (j, k)   -> (B, H, N, N) after transposition
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)  i,k
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)  k,j

    B, H, N, _ = left_perm.shape
    # Collapse batch and hidden dimensions into a single batch for the kernel
    left_flat = left_perm.view(B * H, N, N)    # (B*H, M=N, K=N)
    right_flat = right_perm.view(B * H, N, N)  # (B*H, K=N, N)

    # Batched matrix multiplication (FP16 in/out, FP32 accumulation)
    out_flat = triton_matmul(left_flat, right_flat)  # (B*H, N, N), FP16

    # Reshape back to [B, N, N, H] and convert to FP32 for the remainder
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # -------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final projection (float32)
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)  # broadcast over the hidden dim
    out = F.linear(out, weights["to_out.weight"])  # (B, N, N, dim)

    return out
