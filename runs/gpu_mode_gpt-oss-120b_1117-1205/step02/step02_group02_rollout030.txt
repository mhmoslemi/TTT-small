    # get dims.

    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)

    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # Convert to fp16
    x_fp16 = x.to(torch.float16)

    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))

@triton.jit
def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                   M, N, K,
                   stride_left_batch, stride_left_i, stride_left_k, stride_left_d,
                   stride_right_batch, stride_right_i, stride_right_k, stride_right_d,
                   stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
                   hidden_dim,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # Program IDs
    batch_h = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Decode batch and hidden index
    batch_idx = batch_h // hidden_dim
    hidden_idx = batch_h % hidden_dim

    # Compute base pointers
    left_base = left_ptr + batch_idx * stride_left_batch + hidden_idx * stride_left_d
    right_base = right_ptr + batch_idx * stride_right_batch + hidden_idx * stride_right_d
    out_base = out_ptr + batch_idx * stride_out_batch + hidden_idx * stride_out_d

    # Offsets for block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        mask_k = offs_k < K

        # Load A tile (left)
        a_ptrs = left_base + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

"""
TriMul (outgoing) forward pass with a fused Triton batched GEMM.

The core operation
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batch of matrix multiplications A @ Bᵀ for every hidden dimension d.
We launch a single Triton kernel that processes all (batch, hidden_dim) pairs
as a flattened batch dimension, avoiding the expensive permute / view copies
used in a naïve implementation.
All other ops (LayerNorm, Linear, gating, mask) are performed with PyTorch
in FP16/F32 as appropriate.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,                     # matrix dimensions (all == seqlen)
    stride_left_batch, stride_left_i, stride_left_k, stride_left_d,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_d,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
    hidden_dim,                  # runtime hidden dimension (e.g. 128)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Computes a batched contraction over the k‑dimension.
    Grid layout:
        program_id(0) : flattened batch*hidden index (0 .. B*H-1)
        program_id(1) : block row (i)
        program_id(2) : block col (j)
    """
    batch_h = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # decode into (batch, hidden) pair
    batch_idx = batch_h // hidden_dim
    hidden_idx = batch_h % hidden_dim

    # base pointers for the selected (batch, hidden) slice
    left_base = left_ptr + batch_idx * stride_left_batch + hidden_idx * stride_left_d
    right_base = right_ptr + batch_idx * stride_right_batch + hidden_idx * stride_right_d
    out_base = out_ptr + batch_idx * stride_out_batch + hidden_idx * stride_out_d

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # right tile (already transposed): shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)        # fp16 * fp16 -> fp32 accumulation

    # write result (fp16)
    c_ptrs = out_base + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor:  torch.Tensor [B, N, N] (bool or numeric) or None
            - weights_dict: mapping from parameter name to torch.Tensor
            - config_dict: must contain 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )                                 # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the heavy ops
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (fp16, bias=False)
    # ------------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)   # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 5. Gating tensors (fp16)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Core contraction: batched GEMM via a fused Triton kernel
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    assert H == hidden_dim

    out_fp16 = torch.empty_like(left)  # same shape [B, N, N, H]

    # strides in element units (since Triton works with element offsets)
    stride_l_batch, stride_l_i, stride_l_k, stride_l_d = left.stride()
    stride_r_batch, stride_r_i, stride_r_k, stride_r_d = right.stride()
    stride_o_batch, stride_o_i, stride_o_j, stride_o_d = out_fp16.stride()

    # tiling parameters – tuned for B200 (Ampere)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64   # larger K‑tile reduces loop count

    grid = (
        B * H,                     # flattened (batch, hidden)
        triton.cdiv(N, BLOCK_M),   # i‑tiles
        triton.cdiv(N, BLOCK_N)    # j‑tiles
    )
    tri_mul_kernel[grid](
        left,
        right,
        out_fp16,
        N, N, N,                                   # M, N, K
        stride_l_batch, stride_l_i, stride_l_k, stride_l_d,
        stride_r_batch, stride_r_i, stride_r_k, stride_r_d,
        stride_o_batch, stride_o_i, stride_o_j, stride_o_d,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Cast back to fp32 and apply output transforms
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"])

    return out
