```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # batch index (each batch corresponds to one original batch*hidden pair)
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets of the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask to avoid OOB
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of a batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_base + (offs_m[:, None] * stride_a_m) + (offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None]) & (offs_k < K),
                    other=0.0)

        # Load B tile (K x N)
        b_ptrs = b_base + (offs_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        # Compute dot product for this tile
        acc += tl.dot(a, b)

    # Write back result
    c_ptrs = c_base + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def triton_batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B where:
        A: (batch, M, K)   FP16
        B: (batch, K, N)   FP16
    Returns:
        C: (batch, M, N)   FP16
    """
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # strides
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tile sizes – tuned for B200 (FP16)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (batch,
            triton.cdiv(M, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operation.

    Arguments:
        data (tuple): (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or numeric) – may be None
            - weights: dict of model parameters
            - config: dict containing 'dim', 'hidden_dim', and optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # Cast to half‑precision for all heavy ops
    x_fp16 = x.to(torch.float16)

    # -------------------------------------------------
    # 2. Linear projections (no bias)
    # -------------------------------------------------
    left = F.linear(
        x_fp16,
        weights["left_proj.weight"].to(torch.float16)
    )
    right = F.linear(
        x_fp16,
        weights["right_proj.weight"].to(torch.float16)
    )

    # -------------------------------------------------
    # 3. Apply mask if present
    # -------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [B, N, N, 1]
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 4. Gating (sigmoid) and apply to projected tensors
    # -------------------------------------------------
    left_gate = F.linear(
        x_fp16,
        weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = F.linear(
        x_fp16,
        weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = F.linear(
        x_fp16,
        weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 5. Core contraction: batched GEMM left @ right^T using Triton
    # -------------------------------------------------
    # Rearrange to (B, H, N, N) layout
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # Right needs transpose of the two sequence axes
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape
    # Collapse batch and hidden dimensions for the batched GEMM
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Perform batched matrix multiplication: (B*H, N, N) @ (B*H, N, N)^T
    out_flat = triton_batched_matmul(left_flat, right_flat)  # (B*H, N, N), fp16

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    # Cast back to fp32 for subsequent LayerNorm & projection
    out = out.to(torch.float32)

    # -------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    # Multiply by out_gate (convert to fp32)
    out = out * out_gate.to(torch.float32)
    # Final linear projection back to original dimension
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out

    right = F.linear(
        x_fp16,
        weights["right_proj.weight"].to(torch.float16)
    )

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid) and application
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_fp16,
        weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = F.linear(
        x_fp16,
        weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = F.linear(
        x_fp16,
        weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton batched GEMM
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Permute to (B, H, N, N) where H = hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right needs (k,j) ordering for the transpose inside the GEMM
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)   # (B*H, N, K)
    right_flat = right_perm.view(B * H, N, N) # (B*H, K, N) with K=N

    out_flat = triton_batched_matmul(left_flat, right_flat)   # (B*H, N, N), fp16

    # Restore (B, N, N, H) layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)                               # to FP32 for LayerNorm

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
```
