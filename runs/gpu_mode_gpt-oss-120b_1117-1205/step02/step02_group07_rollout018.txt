"""
Highly‑optimized forward pass for the outgoing TriMul operator.

Algorithm
---------
1. Layer‑norm the input (float32).
2. Project to hidden dimension with 2 linear layers (fp16).
3. (Optional) apply the pair mask.
4. Compute three element‑wise gates (left, right, out) and apply the first two.
5. Core contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication A @ B where
   A = left.permute(0,3,1,2)  -> (B, H, N, N)  (i,k)
   B = right.permute(0,3,2,1) -> (B, H, N, N)  (k,j)
   We flatten the (B,H) axes and launch a Triton kernel that computes
   C = A @ B using fp16 inputs and fp32 accumulation.
6. Layer‑norm the result, apply the output gate and a final linear projection
   back to the original channel dimension (float32).

The Triton kernel is a classic tiled GEMM with configurable block sizes.
It is invoked with a grid of (batch, M‑blocks, N‑blocks) where
batch = B * hidden_dim.
"""

):

    # ------------------------------------------------------------

    # Offsets within the block
    # ------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------
    # Base pointers for this batch
    # ------------------------------------------------------------
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        # Load B tile (K x N) – B is already in (K, N) order
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ------------------------------------------------------------
    # Write back
    # ------------------------------------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B where
        A: (batch, M, K)   fp16
        B: (batch, K, N)   fp16   (already transposed for the contraction)
    Returns
        C: (batch, M, N)   fp16
    """
    assert a.is_contiguous() and b.is_contiguous()
    assert a.dtype == b.dtype == torch.float16
    batch, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    c = torch.empty((batch, M, N), dtype=torch.float16, device=a.device)

    # Strides in elements (not bytes)
    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_k, stride_b_n = b.stride()
    stride_c_batch, stride_c_m, stride_c_n = c.stride()

    # Tunable tile sizes – work well for N up to 1024 and hidden_dim=128
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_gemm_kernel[grid](
        a, b, c,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c


# ------------------------------------------------------------
# Entry point – the custom kernel required by the grader
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float)
        - weights_dict : mapping name -> torch.Tensor
        - config_dict  : must contain 'dim' and 'hidden_dim',
                         optional 'nomask' (True/False)

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim] – float32

    # ------------------------------------------------------------
    # 2) Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------
    # 3) Linear projections (fp16, no bias)
    # ------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))   # [B,N,N,hidden]
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------
    # 4) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → [B,N,N,1] → fp16
        mask_fp = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------
    # 5) Gating – sigmoid on linear transforms
    # ------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------
    # 6) Core contraction via Triton batched GEMM
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------
    #   (b, i, k, d) → (b, d, i, k)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)   i‑k order
    #   (b, j, k, d) → (b, d, k, j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)   k‑j order

    B, H, N, _ = left_perm.shape
    # Flatten first two dimensions → (B*H, N, N)
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Batched matrix multiplication (fp16 in, fp16 out)
    out_flat = _batched_gemm(left_flat, right_flat)      # (B*H, N, N)

    # Reshape back to [B, N, N, H] and cast to float32 for the rest
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate and final projection (float32)
    # ------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
