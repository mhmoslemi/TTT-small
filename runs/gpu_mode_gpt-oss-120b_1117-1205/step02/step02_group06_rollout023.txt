@triton.jit
def batched_tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_k, stride_b_n, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_n, stride_c_h,
    hidden_dim,  # added
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_hidden_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute batch and hidden
    batch_id = batch_hidden_id // hidden_dim
    hidden_id = batch_hidden_id - batch_id * hidden_dim

    # Offsets for each batch+hidden slice
    a_batch = a_ptr + batch_id * stride_a_batch + hidden_id * stride_a_h
    b_batch = b_ptr + batch_id * stride_b_batch + hidden_id * stride_b_h
    c_batch = c_ptr + batch_id * stride_c_batch + hidden_id * stride_c_h

    # Compute block indices
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B tile: shape (BLOCK_K, BLOCK_N) but B is transposed; using row stride=K (k) and col stride=n (j)
        b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_j[None, :] * stride_b_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Multiply-accumulate
        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_batch + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
Optimized Triton implementation of the outgoing TriMul (AlphaFold3) forward pass.

Algorithm
---------
1. LayerNorm on the input (dim).
2. Linear projections (left/right) and gating (left_gate/right_gate/out_gate) in FP32.
3. Apply optional mask and element‑wise gates.
4. Cast the gated tensors to FP16.
5. Core contraction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication A @ Bᵀ where
   A = left (B,N,N,H) and Bᵀ is the transpose of right on the last two axes.
   The multiplication is performed by a custom Triton kernel that
   (a) fuses the batch and hidden dimensions (B*H virtual batch),
   (b) loads the right operand with swapped strides to avoid an explicit transpose,
   (c) accumulates in FP32 and stores FP16.
6. Convert the result back to FP32, apply the final LayerNorm, multiply by the
   output gate and a final linear projection to the original channel dimension.

The kernel operates on contiguous FP16 tensors and avoids any large temporary
copies of the input tensors.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul core contraction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions (all = seq_len)
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,   # left strides
    stride_b_batch, stride_b_k, stride_b_n, stride_b_h,   # right (transposed) strides
    stride_c_batch, stride_c_i, stride_c_n, stride_c_h,   # output strides
    hidden_dim,                         # total hidden channels
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # 1) Resolve virtual batch (batch * hidden) into concrete batch & channel
    # ------------------------------------------------------------------
    batch_hidden = tl.program_id(0)          # combined batch*hidden id
    pid_m = tl.program_id(1)                # block row
    pid_n = tl.program_id(2)                # block column

    batch_id = batch_hidden // hidden_dim
    hidden_id = batch_hidden - batch_id * hidden_dim

    # Base pointers for this (batch, hidden) slice
    a = a_ptr + batch_id * stride_a_batch + hidden_id * stride_a_h
    b = b_ptr + batch_id * stride_b_batch + hidden_id * stride_b_h
    c = c_ptr + batch_id * stride_c_batch + hidden_id * stride_c_h

    # ------------------------------------------------------------------
    # 2) Tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3) Main K‑loop
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_tiles):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        a_tile = tl.load(a_ptrs,
                         mask=mask_i[:, None] & mask_k[None, :],
                         other=0.0)

        # B tile (transposed on the fly): shape (BLOCK_K, BLOCK_N)
        # Strides: row -> k (stride_b_k), col -> j (stride_b_n)
        b_ptrs = b + offs_k[:, None] * stride_b_k + offs_j[None, :] * stride_b_n
        b_tile = tl.load(b_ptrs,
                         mask=mask_k[:, None] & mask_j[None, :],
                         other=0.0)

        acc += tl.dot(a_tile, b_tile)

    # ------------------------------------------------------------------
    # 4) Write back
    # ------------------------------------------------------------------
    c_ptrs = c + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def _tri_mul_outgoing_triton(left: torch.Tensor,
                             right: torch.Tensor,
                             hidden_dim: int,
                             BLOCK_M: int = 64,
                             BLOCK_N: int = 64,
                             BLOCK_K: int = 32) -> torch.Tensor:
    """
    Batched matrix multiplication C = left @ rightᵀ
    left/right : (B, N, N, H)  FP16
    Returns C  : (B, N, N, H)  FP16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, _, H = left.shape
    assert H == hidden_dim

    # output tensor
    out = torch.empty_like(left, dtype=torch.float16, device=left.device)

    # Strides (in elements, not bytes)
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h = left.stride()
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h = right.stride()
    # For the transposed view of right we need (row=k, col=j) strides
    stride_b_row = stride_b_k          # stride for k dimension
    stride_b_col = stride_b_j          # stride for j dimension

    stride_c_batch, stride_c_i, stride_c_j, stride_c_h = out.stride()

    # Grid: (combined batch*hidden, M‑blocks, N‑blocks)
    grid = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _tri_mul_outgoing_kernel[grid](
        left, right, out,
        N, N, N,
        stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
        stride_b_batch, stride_b_row, stride_b_col, stride_b_h,
        stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – optional
            - weights_dict : mapping name → torch.Tensor (float32)
            - config_dict  : must contain 'dim' and 'hidden_dim';
                             may contain 'nomask' (bool).

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2. Linear projections (float32) – no bias
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [B, N, N, 1]
        mask_f = mask.to(torch.float32).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gates (float32)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to FP16 for the heavy contraction
    # ------------------------------------------------------------------
    left_fp16 = left.half()
    right_fp16 = right.half()

    # ------------------------------------------------------------------
    # 6. Core contraction via Triton (left @ rightᵀ)
    # ------------------------------------------------------------------
    out_fp16 = _tri_mul_outgoing_triton(
        left_fp16,
        right_fp16,
        hidden_dim,
        BLOCK_M=64,
        BLOCK_N=64,
        BLOCK_K=32,
    )  # [B, N, N, hidden_dim] fp16

    # Free intermediate FP16 tensors early
    del left_fp16, right_fp16, left, right

    # ------------------------------------------------------------------
    # 7. Cast back to FP32 and apply output processing
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
