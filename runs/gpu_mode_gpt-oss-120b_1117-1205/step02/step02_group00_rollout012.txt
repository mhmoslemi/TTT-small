@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements: tl.int32,
    BLOCK: tl.constexpr,

):

    # load
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)

"""
Optimized outgoing TriMul forward pass.

The implementation follows the reference PyTorch code but:
1. Performs the element‑wise mask & gate fusion in a small Triton kernel
   (still works when there is no mask).
2. Executes the heavy O(N³) contraction as a batched matrix multiplication
   using torch.bmm (FP32), which maps to cuBLAS and provides near‑optimal
   performance.
All other operations (LayerNorms, Linear projections, final LayerNorm and
output projection) are performed with PyTorch primitives. The final output
is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: fuse mask (optional) + left/right gates
# --------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # ptr to left tensor   [total] float32
    right_ptr,              # ptr to right tensor  [total] float32
    left_gate_ptr,          # ptr to left gate    [total] float32
    right_gate_ptr,         # ptr to right gate   [total] float32
    mask_ptr,               # ptr to mask         [mask_total] float32 (broadcast)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,    # 1 if mask is present, 0 otherwise
    H: tl.constexpr,           # hidden_dim (size of innermost dimension)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    if HAS_MASK:
        # One mask element corresponds to H consecutive channel entries
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store results back in‑place
    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    All tensors must be contiguous torch.float32.
    """
    total = left.numel()
    BLOCK = 4096  # reasonable block size for fp32
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy, never read when HAS_MASK==0

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # No return — operation is performed in‑place

# Main entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor:  torch.Tensor [B, N, N] (bool or numeric)
            - weights: mapping of parameter names to torch.Tensor
            - config: dict with keys ``dim``, ``hidden_dim``, optional ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2. Projections (float32)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gates (float32)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) and gates via Triton (in‑place)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> unsqueeze to broadcast over hidden_dim
        mask_h = mask.to(torch.float32).unsqueeze(-1).contiguous()
        # Flatten tensors for the kernel
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_h.view(-1)                     # shape [B*N*N]
        # In‑place fusion
        _apply_gate_and_mask(
            left_flat, right_flat,
            left_gate_flat, right_gate_flat,
            mask_flat,
            hidden_dim,
        )
        # Reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask – simple element‑wise multiplication (still via kernel for consistency)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        _apply_gate_and_mask(
            left_flat, right_flat,
            left_gate_flat, right_gate_flat,
            None,
            hidden_dim,
        )
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)

    # Gates are no longer needed
    del left_gate, right_gate, x

    # ------------------------------------------------------------------
    # 5. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #   Implemented as batched matrix multiplication (B*H, N, N) @ (B*H, N, N)
    # ------------------------------------------------------------------
    #   left: (B, N, N, H)  -> (B, H, N, N) where axes = (i,k)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, i=k)
    #   right: (B, N, N, H) -> (B, H, N, N) where axes = (k,j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, k, j)

    B, H, N, _ = left_perm.shape
    # Collapse batch and head dimensions
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Batched GEMM; uses cuBLAS (FP32 accumulation)
    out_mat = torch.bmm(left_mat, right_mat)   # (B*H, N, N)

    # Restore the original layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # Clean up large intermediates early
    del left, right, left_perm, right_perm, left_mat, right_mat, out_mat

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    # Return in float32 as required
    return out
