def elementwise_mul_kernel(out_ptr, gate_ptr, out_ptr_res, numel: tl.int64, BLOCK_SIZE: tl.constexpr):

    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    device = input_tensor.device

    dim, hidden_dim = config["dim"], config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. LayerNorm
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])

    # Cast to fp16 for heavy compute
    x_fp16 = x.to(torch.float16)

    # linear projections
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # mask
    if not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # gates
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # core contraction using torch.bmm:
    # Rearrange left/right to (B, hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # transpose sequence dims

    B, H, N, _ = left_perm.shape
    # flatten batch*hidden
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # batched matrix multiplication
    out_flat = torch.bmm(left_flat, right_flat)  # shape (B*H,N,N), dtype FP16

    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # shape (B,N,N,H)

    # Convert to float32 for layernorm
    out_fp32 = out.to(torch.float32)

    # to_out_norm
    out_norm = F.layer_norm(out_fp32,
                            (hidden_dim,),
                            weight=weights["to_out_norm.weight"],
                            bias=weights["to_out_norm.bias"])

    # elementwise multiplication via Triton kernel
    # out_norm and out_gate are both (B,N,N,hidden_dim) float32? out_gate currently fp16; we need fp32.
    out_gate_fp32 = out_gate.to(torch.float32)

    out_norm_flat = out_norm.contiguous().view(-1)

    # allocate result tensor
    out_mul_flat = torch.empty_like(out_norm_flat)

"""
TriMul (outgoing) forward pass.

Algorithm:
1. LayerNorm over the channel dimension (float32).
2. Linear projections to a hidden dimension (FP16).
3. Optional mask application (FP16).
4. Gating via linear + sigmoid (FP16) and element‑wise multiplication.
5. Core contraction: for each hidden slice compute
   out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d].
   Implemented as a batched matrix multiplication using torch.bmm
   (highly‑optimized cuBLAS Tensor‑Core GEMM).
6. LayerNorm over the hidden dimension (float32).
7. Element‑wise multiplication with the output gate using a tiny Triton kernel.
8. Final linear projection back to the original dimension (float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_inplace_kernel(out_ptr, gate_ptr, numel: tl.int64, BLOCK_SIZE: tl.constexpr):
    """
    In‑place elementwise multiplication: out = out * gate
    All tensors are float32.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < numel
    out_val = tl.load(out_ptr + offs, mask=mask)
    gate_val = tl.load(gate_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, out_val * gate_val, mask=mask)

def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) – optional
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys 'dim', 'hidden_dim', and optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. LayerNorm over channel dimension (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Linear projections to hidden dimension (FP16)
    # --------------------------------------------------------------
    x_fp16 = x.to(torch.float16)
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 4. Gating values (sigmoid) and application
    # --------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction (batched GEMM) – use torch.bmm for speed
    # --------------------------------------------------------------
    # reshape to (B, hidden, N, N) where the two sequence axes are arranged
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) with (j,k) -> (k,j)

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # batched matrix multiplication: (B*H, N, N) @ (B*H, N, N)ᵀ  (right already transposed)
    out_flat = torch.bmm(left_flat, right_flat)   # FP16 result

    # restore original layout (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H) FP16

    # --------------------------------------------------------------
    # 6. Output LayerNorm (float32)
    # --------------------------------------------------------------
    out = out.to(torch.float32)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # --------------------------------------------------------------
    # 7. Multiply by output gate using a tiny Triton kernel
    # --------------------------------------------------------------
    out_gate_f32 = out_gate.to(torch.float32)  # (B, N, N, H)

    # flatten tensors for the kernel
    out_flat = out.view(-1)
    gate_flat = out_gate_f32.view(-1)
    numel = out_flat.numel()

    BLOCK_SIZE = 1024
    grid = (triton.cdiv(numel, BLOCK_SIZE),)
    elementwise_mul_inplace_kernel[grid](
        out_flat,
        gate_flat,
        numel,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # --------------------------------------------------------------
    # 8. Final projection back to original dimension (float32)
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
