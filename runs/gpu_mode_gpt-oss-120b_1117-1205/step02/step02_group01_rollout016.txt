@triton.jit
def mul_gate_mask_2out_kernel(
    left_out_ptr, right_out_ptr, left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total, hidden,
    BLOCK_SIZE: tl.constexpr,
    H: tl.constexpr, # same as hidden
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask_off = offs < total

    # Load elements
    l = tl.load(left_ptr + offs, mask=mask_off)
    r = tl.load(right_ptr + offs, mask=mask_off)
    lg = tl.load(left_gate_ptr + offs, mask=mask_off)
    rg = tl.load(right_gate_ptr + offs, mask=mask_off)

    # Compute mask index: each element corresponds to a position (b,i,k) ignoring hidden index.
    # hidden dimension size = H
    # offset // H gives index into mask flatten (b,N,N)
    mask_idx = offs // H
    m_val = tl.load(mask_ptr + mask_idx, mask=mask_off)

    left_out = l * lg * m_val
    right_out = r * rg * m_val

    tl.store(left_out_ptr + offs, left_out, mask=mask_off)
    tl.store(right_out_ptr + offs, right_out, mask=mask_off)

5. Triton kernel:  left = left * left_gate * mask
                  right = right * right_gate * mask  (in‑place).
6. Batched GEMM (`torch.bmm`) to compute the cubic contraction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a batch of matrix‑multiplications over the hidden
   dimension.
7. LayerNorm over the hidden dimension, final gating, and output linear
   projection back to `dim` (float32).

All tensors are kept on the GPU; the only Triton kernel required by the
task is the mask‑gate fusion.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mul_gate_mask_2out_inplace_kernel(
    left_ptr,          # [total] left tensor (will be overwritten)
    right_ptr,         # [total] right tensor (will be overwritten)
    left_gate_ptr,     # [total] left gate tensor
    right_gate_ptr,    # [total] right gate tensor
    mask_ptr,          # [B*N*N] mask tensor (broadcast over hidden dim)
    total,             # int: total number of elements in left/right tensors
    BLOCK_SIZE: tl.constexpr,
    H: tl.constexpr,   # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # Load data
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Broadcast mask over the hidden dimension
    mask_idx = offs // H               # index into mask (b, i, k)
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply mask and gate in‑place
    tl.store(left_ptr + offs,  l * lg * m, mask=mask)
    tl.store(right_ptr + offs, r * rg * m, mask=mask)


def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, dim] (float32)
            mask_tensor  : torch.Tensor [B, N, N] (float32) or None
            weights_dict : dict with the following keys
                - "norm.weight", "norm.bias"
                - "left_proj.weight", "right_proj.weight"
                - "left_gate.weight", "right_gate.weight", "out_gate.weight"
                - "to_out_norm.weight", "to_out_norm.bias"
                - "to_out.weight"
            config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool}
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # [B,N,N,dim]

    # -----------------------------------------------------------------
    # 2. Cast to half for the heavy part
    # -----------------------------------------------------------------
    x_h = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left = F.linear(x_h, left_proj_w)   # [B,N,N,hidden]
    right = F.linear(x_h, right_proj_w) # [B,N,N,hidden]

    # -----------------------------------------------------------------
    # 4. Gating (linear + sigmoid)
    # -----------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w)) # [B,N,N,hidden]
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))    # [B,N,N,hidden]

    # -----------------------------------------------------------------
    # 5. Mask handling (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if config.get("nomask", False) or mask_tensor is None:
        mask_h = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_h = mask_tensor.to(dtype=torch.float16, device=device)

    # Flatten tensors for the Triton kernel
    left_flat = left.reshape(-1).contiguous()
    right_flat = right.reshape(-1).contiguous()
    left_gate_flat = left_gate.reshape(-1).contiguous()
    right_gate_flat = right_gate.reshape(-1).contiguous()
    mask_flat = mask_h.reshape(-1).contiguous()

    total_elements = left_flat.numel()                     # B*N*N*hidden

    # -----------------------------------------------------------------
    # 6. Triton kernel: fuse mask & gate multiplication (in‑place)
    # -----------------------------------------------------------------
    BLOCK = 1024
    grid = (triton.cdiv(total_elements, BLOCK),)
    _mul_gate_mask_2out_inplace_kernel[grid](
        left_flat,
        right_flat,
        left_gate_flat,
        right_gate_flat,
        mask_flat,
        total_elements,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
    )

    # Reshape back to 4‑D tensors
    left = left_flat.view(B, N, N, hidden_dim)
    right = right_flat.view(B, N, N, hidden_dim)

    # -----------------------------------------------------------------
    # 7. Cubic contraction via batched GEMM (left @ right^T)
    # -----------------------------------------------------------------
    # Rearrange for batched bmm: (B, hidden, N, N) where rows=i/k and rows=k/j
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B, hidden, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, hidden, k, j)

    LH = B * hidden_dim
    left_mat = left_perm.view(LH, N, N)   # (B*hidden, N, N)
    right_mat = right_perm.view(LH, N, N) # (B*hidden, N, N)

    out_mat = torch.bmm(left_mat, right_mat)                # (B*hidden, N, N)

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden]

    # -----------------------------------------------------------------
    # 8. Post‑processing: LayerNorm, final gate, output projection
    # -----------------------------------------------------------------
    out_f32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float32)
    out_norm = F.layer_norm(out_f32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply the final gating (out_gate was computed in half)
    out_norm = out_norm * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float32)  # (dim, hidden)
    output = F.linear(out_norm, to_out_w)  # [B, N, N, dim]

    return output
```
