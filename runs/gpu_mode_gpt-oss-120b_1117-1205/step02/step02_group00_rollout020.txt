```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _tri_mul_outgoing_kernel(
    out_ptr, left_ptr, right_ptr,
    B, N, H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_i = tl.program_id(0)  # block row
    pid_j = tl.program_id(1)  # block col
    pid_h = tl.program_id(2)  # batch*head

    # decode batch and head
    b = pid_h // H
    h = pid_h % H

    # offsets within block
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute strides (contiguous layout)
    stride_batch = N * N * H
    stride_i = N * H
    stride_k = H   # also stride_j

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left: shape (M, K)
        left_ptrs = left_ptr + b * stride_batch + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k + h
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right: shape (K, N) (k as rows, j as cols)
        right_ptrs = right_ptr + b * stride_batch + offs_k[:, None] * stride_k + offs_j[None, :] * stride_i + h
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # matrix multiply-accumulate
        acc += tl.dot(left, right)

    # write output
    out_ptrs = out_ptr + b * stride_batch + offs_i[:, None] * stride_i + offs_j[None, :] * stride_k + h
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

    1) LayerNorm and two linear projections (half precision)
    2) Sigmoid gating and optional mask (elementwise, done with PyTorch)

"""
Outgoing TriMul (AlphaFold3) – forward only.

Algorithm
---------
1. Layer‑norm the 4‑D input.
2. Two linear projections (dim → hidden_dim) produce `left` and `right`.
3. Sigmoid gates are computed from the same normalized tensor.
   If a mask is supplied it is broadcast over the hidden dimension.
   Element‑wise multiplication ``left *= left_gate * mask`` and
   ``right *= right_gate * mask`` is done with plain PyTorch.
4. The heavy O(N³·hidden_dim) contraction
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is performed by a custom Triton kernel that treats each
   (batch, hidden) pair as an independent GEMM.  The kernel works
   on strided tensors directly, avoiding the expensive
   ``permute(...).contiguous()`` copies required for ``torch.bmm``.
   Accumulation is carried out in FP32 and the result is stored as
   FP16.
5. Output layer‑norm, out‑gate multiplication and the final linear
   projection (hidden_dim → dim) bring the tensor back to shape
   [B, N, N, dim].
All operations after the kernel use half‑precision; the final result
is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------
# Triton kernel: batched GEMM over the (batch, hidden) dimension.
# ------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    out_ptr, left_ptr, right_ptr,
    B, N, H,                     # problem sizes (int32)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    Layout of tensors (contiguous, row‑major):
        left / right / out  : [B, N, N, H]
        stride_batch = N * N * H
        stride_i     = N * H        (first N dimension)
        stride_k     = H            (second N dimension)
        stride_h     = 1            (hidden)

    The kernel tiles the (i, j) output plane with BLOCK_M×BLOCK_N
    and iterates over k in steps of BLOCK_K.
    """
    pid_i = tl.program_id(0)            # block row   (i)
    pid_j = tl.program_id(1)            # block col   (j)
    pid_h = tl.program_id(2)            # flattened (batch, hidden)

    # decode batch index and hidden index
    b = pid_h // H
    h = pid_h % H

    # offsets inside the tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # strides for a contiguous [B,N,N,H] tensor
    stride_batch = N * N * H
    stride_i = N * H          # stride for the first N dimension (i / j)
    stride_k = H              # stride for the second N dimension (k)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: shape (M, K)
        left_ptrs = (left_ptr + b * stride_batch
                     + offs_i[:, None] * stride_i
                     + offs_k[None, :] * stride_k
                     + h)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right block: shape (K, N)  (k as rows, j as cols)
        right_ptrs = (right_ptr + b * stride_batch
                      + offs_k[:, None] * stride_k
                      + offs_j[None, :] * stride_i
                      + h)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # matrix‑multiply‑accumulate
        acc += tl.dot(left, right)

    # store the result
    out_ptrs = (out_ptr + b * stride_batch
                + offs_i[:, None] * stride_i
                + offs_j[None, :] * stride_k
                + h)
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

# ------------------------------------------------------------
# Entry point
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args
    ----
    data : tuple (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool or numeric) or None
        - weights      : dict of model parameters
        - config       : dict with keys ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool) defaults to True

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)  # half‑precision for the rest

    # ------------------------------------------------------------------
    # 2. Linear projections (dim → hidden_dim)
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid) and optional mask
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    if not nomask and mask is not None:
        # mask shape [B,N,N] → broadcast over hidden_dim
        mask_h = mask.unsqueeze(-1).to(torch.float16)
        left = left * left_gate * mask_h
        right = right * right_gate * mask_h
    else:
        left = left * left_gate
        right = right * right_gate

    # ensure contiguous tensors for the kernel
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 4. Core contraction via Triton (O(N³·hidden_dim))
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape   # left shape is [B, N, N, hidden_dim]
    out = torch.empty_like(left)  # allocate FP16 output

    # tile sizes – tuned for the B200 GPU
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )
    _tri_mul_outgoing_kernel[grid](
        out,
        left,
        right,
        B,
        N,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 5. Output processing
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return in the requested dtype
    return out.to(torch.float32)
