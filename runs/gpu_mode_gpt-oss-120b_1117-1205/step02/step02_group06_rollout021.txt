def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute global offsets
    off_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # calculate mask for out-of-bounds
    mask_m = off_m < M
    mask_n = off_n < N

    # base pointers
    a_batch = a_ptr + batch * stride_a_batch
    b_batch = b_ptr + batch * stride_b_batch
    c_batch = c_ptr + batch * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        off_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # mask for k
        mask_k = off_k < K

        # loads A: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + off_m[:, None] * stride_a_m + off_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # loads B: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + off_k[:, None] * stride_b_k + off_n[None, :] * stride_b_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Store to C
    c_ptrs = c_batch + off_m[:, None] * stride_c_m + off_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass using a fused Triton batched GEMM.
- LayerNorm on input.
- Linear projections & sigmoid gates computed in fp16.
- Optional mask applied to projected tensors.
- Core contraction:
    out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
  implemented as a batched matrix multiplication (B*H independent matmuls)
  with a custom Triton kernel that tiles the M, N, K dimensions.
- Output LayerNorm, out‑gate multiplication and final linear projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for TriMul contraction.
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_hidden, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_hidden, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_hidden, stride_c_m, stride_c_n,
    hidden_dim,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute C = A @ B for many (batch, head) pairs.
    A: (batch, hidden, M, K)
    B: (batch, hidden, K, N)
    C: (batch, hidden, M, N)
    All tensors are stored in row‑major order; strides are given in
    element units.  The kernel tiles the computation with blocks of size
    BLOCK_M × BLOCK_N and iterates over K in steps of BLOCK_K.
    """
    # program ids
    batch_hidden = tl.program_id(0)   # 0 .. batch*hidden-1
    pid_m = tl.program_id(1)         # block row
    pid_n = tl.program_id(2)         # block col

    # decode batch and head indices
    b_idx = batch_hidden // hidden_dim
    h_idx = batch_hidden % hidden_dim

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this (batch, head)
    a_base = a_ptr + b_idx * stride_a_batch + h_idx * stride_a_hidden
    b_base = b_ptr + b_idx * stride_b_batch + h_idx * stride_b_hidden
    c_base = c_ptr + b_idx * stride_c_batch + h_idx * stride_c_hidden

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile (K x N)
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _batched_tri_mul_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper around the Triton kernel.
    left / right: tensors of shape (batch, hidden, N, N), dtype fp16.
    Returns tensor of same shape and dtype.
    """
    assert left.dtype == torch.float16 and right.dtype == torch.float16
    B, H, N, _ = left.shape
    assert right.shape == (B, H, N, N)

    # output tensor
    out = torch.empty_like(left)

    # Strides in element units (not bytes)
    stride_a_batch, stride_a_hidden, stride_a_m, stride_a_k = left.stride()
    stride_b_batch, stride_b_hidden, stride_b_k, stride_b_n = right.stride()
    stride_c_batch, stride_c_hidden, stride_c_m, stride_c_n = out.stride()

    # Tile sizes – tuned for typical sequence lengths
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _tri_mul_matmul_kernel[grid](
        left,
        right,
        out,
        N,    # M
        N,    # N
        N,    # K
        stride_a_batch,
        stride_a_hidden,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_hidden,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_hidden,
        stride_c_m,
        stride_c_n,
        H,    # hidden_dim
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) – may be unused
            - weights: dict mapping parameter names to tensors
            - config: dict with keys 'dim' and 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    x_fp16 = x.half()

    # --------------------------------------------------------------
    # 2. Linear projections (no bias) – fp16 for speed
    # --------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].half())
    right = F.linear(x_fp16, weights["right_proj.weight"].half())

    # --------------------------------------------------------------
    # 3. Optional mask
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [B, N, N, 1]
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 4. Gating (sigmoid after linear)
    # --------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].half()).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].half()).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].half()).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction: batched GEMM via Triton
    #    left_perm: (B, H, N, N)  where H = hidden_dim
    #    right_perm: (B, H, N, N) (already in K‑N order)
    # --------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2)    # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1)  # (B, H, N, N)

    out_perm = _batched_tri_mul_matmul(left_perm, right_perm)  # (B, H, N, N)

    # reshape back to (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)
    out = out.to(torch.float32)

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out

                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile (K × N).
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate.
        acc += tl.dot(a, b)

    # Store the result.
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _batched_tri_mul_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper that launches the Triton kernel.
    left/right: tensors of shape (B, H, N, N), dtype fp16.
    Returns a tensor of the same shape and dtype.
    """
    assert left.dtype == torch.float16 and right.dtype == torch.float16
    B, H, N, _ = left.shape
    assert right.shape == (B, H, N, N)

    out = torch.empty_like(left)

    # Strides in element units (not bytes).
    stride_a_batch, stride_a_hidden, stride_a_m, stride_a_k = left.stride()
    stride_b_batch, stride_b_hidden, stride_b_k, stride_b_n = right.stride()
    stride_c_batch, stride_c_hidden, stride_c_m, stride_c_n = out.stride()

    # Tile sizes – tuned for typical sequence lengths.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _tri_mul_matmul_kernel[grid](
        left,
        right,
        out,
        N,                # M
        N,                # N
        N,                # K
        stride_a_batch,
        stride_a_hidden,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_hidden,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_hidden,
        stride_c_m,
        stride_c_n,
        H,                # hidden_dim
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be unused
            - weights: dict of model parameters
            - config: dict with keys 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    x_fp16 = x.half()

    # --------------------------------------------------------------
    # 2. Linear projections (no bias) – fp16 for speed
    # --------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].half())
    right = F.linear(x_fp16, weights["right_proj.weight"].half())

    # --------------------------------------------------------------
    # 3. Optional mask
    # --------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 4. Gating (sigmoid after linear)
    # --------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].half()).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].half()).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].half()).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction via Triton batched GEMM
    # --------------------------------------------------------------
    # Shapes: (B, N, N, H) -> permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2)    # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1)  # (B, H, N, N)  (K‑N order)

    out_perm = _batched_tri_mul_matmul(left_perm, right_perm)  # (B, H, N, N)

    # Back to (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
