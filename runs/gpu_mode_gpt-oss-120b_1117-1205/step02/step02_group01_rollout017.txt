    # final linear projection

"""
TriMul (outgoing) forward pass – highly optimized Triton+PyTorch implementation.

Algorithm
---------
1. Input LayerNorm (float32).
2. Cast to fp16 for the bulk of the work.
3. Fuse the projection and its gate into a single linear layer:
       left_proj  + left_gate   →   concat(weight_left_proj, weight_left_gate)
       right_proj + right_gate  →   concat(weight_right_proj, weight_right_gate)
   This reduces the number of GEMM launches from 5 to 3.
4. Compute the output‑gate (sigmoid) separately (tiny cost).
5. Fuse element‑wise mask, gate‑sigmoid and multiplication for the left/right
   branches in a Triton kernel (`proj_gate_mask_kernel`).  The mask is broadcast
   over the hidden dimension.
6. Batched matrix multiplication (left @ rightᵀ) implements the cubic
   contraction using cuBLAS (`torch.bmm`).  Batch dimension = B * hidden.
7. Post‑layer‑norm on the hidden dimension, multiply by the output‑gate,
   and final linear projection back to the original channel dimension.
All heavy linear work stays in fp16; only the two layer‑norms and the final
output are in fp32 as required.

The kernel is launched once for the left branch and once for the right branch;
each launch processes `B * N * N * hidden` elements.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def proj_gate_mask_kernel(
    out_ptr,          # float16* – output buffer (flattened)
    proj_ptr,         # float16* – projected values (flattened)
    gate_pre_ptr,     # float16* – pre‑sigmoid gate values (flattened)
    mask_ptr,         # float16* – mask values (flattened, size = B*N*N)
    total,            # i32 – total number of elements = B*N*N*hidden
    BLOCK_SIZE: tl.constexpr,
    H: tl.constexpr,  # hidden dimension (size of innermost axis)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # Load projected value and pre‑sigmoid gate
    proj = tl.load(proj_ptr + offs, mask=mask)
    gate_pre = tl.load(gate_pre_ptr + offs, mask=mask)

    # sigmoid(x) = 1 / (1 + exp(-x))
    gate = 1.0 / (1.0 + tl.exp(-gate_pre))

    # Broadcasted mask: every H consecutive elements share the same mask entry
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Fused multiply: proj * sigmoid(gate) * mask
    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor: torch.Tensor [B, N, N, dim] (float32)
            mask_tensor : torch.Tensor [B, N, N] or None
            weights_dict : dict with required weight tensors
            config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool (optional)}
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # [B,N,N,dim] FP32

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for the expensive linear work
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Fuse projection + gate for left and right branches
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    left_combined_w = torch.cat([left_proj_w, left_gate_w], dim=0)  # (2*hidden, dim)

    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    right_combined_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*hidden, dim)

    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    # Combined linear layers produce concatenated (proj, gate_pre) tensors
    left_concat = F.linear(x, left_combined_w)   # [B,N,N,2*hidden]
    right_concat = F.linear(x, right_combined_w) # [B,N,N,2*hidden]
    out_gate_pre = F.linear(x, out_gate_w)       # [B,N,N,hidden]

    # Split concatenated tensors
    left_proj = left_concat[..., :hidden]        # [B,N,N,hidden]
    left_gate_pre = left_concat[..., hidden:]    # [B,N,N,hidden]

    right_proj = right_concat[..., :hidden]
    right_gate_pre = right_concat[..., hidden:]

    # Output gate (sigmoid) – used after the cubic contraction
    out_gate = torch.sigmoid(out_gate_pre)       # [B,N,N,hidden] fp16

    # ------------------------------------------------------------------
    # 4. Prepare mask (broadcast over hidden)
    # ------------------------------------------------------------------
    if mask_tensor is None or nomask:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(device).to(torch.float16)

    # Flatten tensors for Triton kernel
    total = left_proj.numel()  # B * N * N * hidden
    left_proj_flat = left_proj.contiguous().view(-1)
    left_gate_pre_flat = left_gate_pre.contiguous().view(-1)
    right_proj_flat = right_proj.contiguous().view(-1)
    right_gate_pre_flat = right_gate_pre.contiguous().view(-1)
    mask_flat = mask_fp16.contiguous().view(-1)  # size = B * N * N

    # Allocate output buffers for the fused multiply‑gate‑mask
    left_out_flat = torch.empty_like(left_proj_flat)
    right_out_flat = torch.empty_like(right_proj_flat)

    # ------------------------------------------------------------------
    # 5. Triton kernel: out = proj * sigmoid(gate_pre) * mask
    # ------------------------------------------------------------------
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    proj_gate_mask_kernel[grid](
        left_out_flat,
        left_proj_flat,
        left_gate_pre_flat,
        mask_flat,
        total,
        BLOCK_SIZE=BLOCK,
        H=hidden,
    )
    proj_gate_mask_kernel[grid](
        right_out_flat,
        right_proj_flat,
        right_gate_pre_flat,
        mask_flat,
        total,
        BLOCK_SIZE=BLOCK,
        H=hidden,
    )

    # Reshape back to [B, N, N, hidden]
    left = left_out_flat.view(B, N, N, hidden)
    right = right_out_flat.view(B, N, N, hidden)

    # ------------------------------------------------------------------
    # 6. Cubic contraction via batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    # (B, hidden, N, N) where hidden is fused into the batch dimension
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)    # (i,k)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden, N, N)  # (j,k)ᵀ

    out_mat = torch.bmm(left_mat, right_mat)  # (B*hidden, N, N) fp16
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden] fp16

    # ------------------------------------------------------------------
    # 7. Post‑processing: hidden‑dim LayerNorm, output‑gate, final projection
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float32)
    out_norm = F.layer_norm(out_fp32, (hidden,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply the final gating (out_gate computed in fp16)
    out_norm = out_norm * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float32)  # (dim, hidden)
    output = F.linear(out_norm, to_out_w)  # [B, N, N, dim] FP32

    return output
