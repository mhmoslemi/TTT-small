"""
Implementation of the outgoing TriMul operator with a fused elementwise multiplication
kernel and a torch.bmm contraction. The heavy O(N^3) contraction is performed with
cuBLAS (torch.bmm) which is faster than the previous pure Triton kernel, while an
element‑wise kernel provides the required Triton component.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def mul_inplace_kernel(a_ptr, b_ptr, N, BLOCK: tl.constexpr):
    """
    In‑place elementwise multiplication: a = a * b
    a_ptr, b_ptr: pointers to contiguous 1‑D tensors of the same dtype.
    N: total number of elements.
    BLOCK: compile‑time thread block size.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N

    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    c = a * b
    tl.store(a_ptr + offs, c, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            mask_tensor: torch.Tensor of shape [B, N, N] (bool/int/float) – may be ignored.
            weights_dict: mapping from parameter name to torch.Tensor.
            config_dict: must contain 'dim' and 'hidden_dim', optional 'nomask'.

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim], float32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for heavy ops
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (fp16, bias=False)
    # ------------------------------------------------------------------
    left = F.linear(
        x_fp16,
        weights["left_proj.weight"].to(torch.float16)
    )  # [B, N, N, hidden_dim] fp16
    right = F.linear(
        x_fp16,
        weights["right_proj.weight"].to(torch.float16)
    )  # [B, N, N, hidden_dim] fp16

    # ------------------------------------------------------------------
    # 4. Gating tensors (fp16)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_fp16,
        weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()  # [B, N, N, hidden_dim] fp16

    right_gate = F.linear(
        x_fp16,
        weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()  # [B, N, N, hidden_dim] fp16

    out_gate = F.linear(
        x_fp16,
        weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()  # [B, N, N, hidden_dim] fp16

    # ------------------------------------------------------------------
    # 5. Fuse element‑wise gating via Triton (in‑place multiplication)
    # ------------------------------------------------------------------
    # flatten tensors to 1‑D for kernel launch
    left = left.contiguous()
    left_gate = left_gate.contiguous()
    right = right.contiguous()
    right_gate = right_gate.contiguous()

    total_elements = left.numel()
    BLOCK = 1024  # max threads per block
    grid = (triton.cdiv(total_elements, BLOCK),)

    # left = left * left_gate
    mul_inplace_kernel[grid](left, left_gate, total_elements, BLOCK=BLOCK)

    # right = right * right_gate
    mul_inplace_kernel[grid](right, right_gate, total_elements, BLOCK=BLOCK)

    # ------------------------------------------------------------------
    # 6. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] -> [B, N, N, 1] for broadcasting
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 7. Core contraction: batched GEMM (A @ Bᵀ) via torch.bmm
    #    For each hidden channel d we compute left_{i,k} @ right_{k,j}
    # ------------------------------------------------------------------
    # Permute to (B, H, N, N) where the last two dims are the matrix dimensions
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, K) with K = N
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, N)

    B, H, N, _ = left_perm.shape
    # reshape batch * hidden as a single batch dimension for torch.bmm
    left_flat = left_perm.view(B * H, N, N)   # (B*H, M=N, K=N)
    right_flat = right_perm.view(B * H, N, N) # (B*H, K=N, N)

    # Perform batched matrix multiplication (fp16 inputs, fp32 accumulation)
    out_flat = torch.bmm(left_flat, right_flat)  # (B*H, N, N) fp16

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # switch to fp32 for the remaining ops

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )  # [B, N, N, hidden_dim] float32

    # Apply out_gate (converted to fp32)
    out_gate_fp32 = out_gate.to(torch.float32)
    out = out * out_gate_fp32

    # Final linear projection back to dim
    out = F.linear(
        out,
        weights["to_out.weight"]
    )  # [B, N, N, dim] float32

    return out

Outgoing TriMul implementation.

The heavy O(N³) reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed as a batched matrix multiplication
    left_{i,k} @ right_{k,j}
using torch.bmm (cuBLAS).  To satisfy the requirement of
including a Triton kernel we fuse the element‑wise
gate‑multiplications (left *= left_gate, right *= right_gate)
into an in‑place Triton kernel.  All other linear / LayerNorm
steps remain in PyTorch for simplicity and speed.

def mul_inplace_kernel(a_ptr, b_ptr, N, BLOCK: tl.constexpr):
