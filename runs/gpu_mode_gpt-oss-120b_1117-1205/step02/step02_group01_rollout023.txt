@triton.jit
def mul_gate_mask_fuse_kernel(out_left_ptr, out_right_ptr,
                              left_ptr, left_gate_ptr,
                              right_ptr, right_gate_ptr,
                              mask_ptr, total,
                              BLOCK_SIZE: tl.constexpr, H: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # Load values conditionally
    a = tl.load(left_ptr + offs, mask=mask)
    ag = tl.load(left_gate_ptr + offs, mask=mask)
    b = tl.load(right_ptr + offs, mask=mask)
    bg = tl.load(right_gate_ptr + offs, mask=mask)

    # compute mask index
    idx = offs // H
    m = tl.load(mask_ptr + idx, mask=mask)

    tl.store(out_left_ptr + offs, a * ag * m, mask=mask)
    tl.store(out_right_ptr + offs, b * bg * m, mask=mask)

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # LayerNorm
    norm_weight = weights["norm.weight"].to(device).to(torch.float32)
    norm_bias = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)  # float32

    # Cast to fp16
    x_fp16 = x.to(torch.float16)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)

    left = F.linear(x_fp16, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = F.linear(x_fp16, right_proj_weight)

    # Gates
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))

    # Mask
    if mask_tensor is None:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(device).to(torch.float16)

    # Flatten
    left_flat = left.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_flat = right.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)
    mask_flat = mask_fp16.contiguous().view(-1)

    total = left_flat.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    left_out_flat = torch.empty_like(left_flat)
    right_out_flat = torch.empty_like(right_flat)

    mul_gate_mask_fuse_kernel[grid](
        left_out_flat,
        right_out_flat,
        left_flat,
        left_gate_flat,
        right_flat,
        right_gate_flat,
        mask_flat,
        total,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
    )

    # Reshape back to [B,N,N,hidden_dim]
    left_out = left_out_flat.view(B, N, N, hidden_dim)
    right_out = right_out_flat.view(B, N, N, hidden_dim)

    # Cubic contraction via batched GEMM: out_{b,i,j,d} = sum_k left_{b,i,k,d} * right_{b,j,k,d}
    # Achieved by batched matmul of left (i,k) and right (j,k)^T per hidden dim.
    left_mat = left_out.permute(0, 3, 1, 2).reshape(-1, N, N)   # (B*H, N, N)
    right_mat = right_out.permute(0, 3, 2, 1).reshape(-1, N, N)  # (B*H, N, N)

"""
TriMul (outgoing) forward pass with a Triton‑accelerated elementwise fusion.
The heavy linear / projection / gate / mask multiplication is fused into a
single Triton kernel, while the cubic contraction is performed as a batched
matrix multiplication using torch.bmm (cuBLAS) on FP16 data for maximum
throughput. The final layers are done in FP32 to match the reference
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ---------------------------------------------------------------
# Triton kernel that fuses (tensor * gate * mask) for left and right
# ---------------------------------------------------------------
@triton.jit
def mul_gate_mask_fuse_kernel(
    out_left_ptr, out_right_ptr,          # output pointers (flattened)
    left_ptr, left_gate_ptr,              # left projection and its gate
    right_ptr, right_gate_ptr,            # right projection and its gate
    mask_ptr,                             # mask pointer (flattened, broadcast over H)
    total,                                # total number of elements (B*N*N*H)
    BLOCK_SIZE: tl.constexpr,             # compile‑time block size
    H: tl.constexpr,                      # hidden dimension (size of innermost axis)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # Load values
    a = tl.load(left_ptr + offs, mask=mask)
    ag = tl.load(left_gate_ptr + offs, mask=mask)
    b = tl.load(right_ptr + offs, mask=mask)
    bg = tl.load(right_gate_ptr + offs, mask=mask)

    # Broadcast mask over hidden dimension
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Store fused results
    tl.store(out_left_ptr + offs, a * ag * m, mask=mask)
    tl.store(out_right_ptr + offs, b * bg * m, mask=mask)


def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor: torch.Tensor [B, N, N, dim] (float32)
            mask_tensor : torch.Tensor [B, N, N] or None (float32)
            weights_dict: model weights as described in the prompt
            config_dict : {"dim": int, "hidden_dim": int}
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # [B,N,N,dim]

    # -----------------------------------------------------------------
    # 2. Cast to FP16 for all heavy work
    # -----------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # [B,N,N,hidden_dim]
    right = F.linear(x_fp16, right_proj_w)  # [B,N,N,hidden_dim]

    # -----------------------------------------------------------------
    # 4. Gating + optional mask (fused in a Triton kernel)
    # -----------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w)) # [B,N,N,hidden_dim]
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))    # [B,N,N,hidden_dim]

    # Prepare mask (all‑ones if none)
    if mask_tensor is None:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(device).to(torch.float16)

    # Flatten tensors for the kernel (contiguous view)
    left_flat = left.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_flat = right.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)
    mask_flat = mask_fp16.contiguous().view(-1)

    total = left_flat.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    left_out_flat = torch.empty_like(left_flat)
    right_out_flat = torch.empty_like(right_flat)

    mul_gate_mask_fuse_kernel[grid](
        left_out_flat,
        right_out_flat,
        left_flat,
        left_gate_flat,
        right_flat,
        right_gate_flat,
        mask_flat,
        total,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
    )

    # Restore the original layout
    left = left_out_flat.view(B, N, N, hidden_dim)
    right = right_out_flat.view(B, N, N, hidden_dim)

    # -----------------------------------------------------------------
    # 5. Cubic contraction via batched GEMM (torch.bmm with FP16)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # -----------------------------------------------------------------
    # Re‑order to (B*H, N, N) where each batch corresponds to a hidden dim
    left_mat = left.permute(0, 3, 1, 2).reshape(-1, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(-1, N, N)  # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)               # (B*H, N, N)

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Post‑processing: LayerNorm on hidden dim, final gate, output proj
    # -----------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float32)
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply the final gating (out_gate is still FP16)
    out_norm = out_norm * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float32)  # (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)  # [B, N, N, dim] (float32)

    return output
