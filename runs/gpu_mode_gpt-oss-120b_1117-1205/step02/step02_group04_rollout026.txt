"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The heavy O(N³) contraction is performed as a batched matrix multiplication
using torch.bmm (highly‑optimized TensorCore GEMM).  A tiny Triton kernel
fuses the element‑wise gating multiplications (left * left_gate,
right * right_gate).  All other operations (LayerNorms, linear projections,
masking) are carried out with PyTorch in mixed precision.  The final output
is FP32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------
# Triton kernel: element‑wise multiply (used for gating)
# ------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(
    a_ptr, b_ptr, out_ptr, size, BLOCK: tl.constexpr
):
    """
    out = a * b   (flat contiguous tensors)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def fused_elementwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Element‑wise multiply `a` and `b` using the Triton kernel."""
    assert a.is_contiguous() and b.is_contiguous()
    out = torch.empty_like(a)
    size = a.numel()
    BLOCK = 65536  # tuneable; large enough to keep kernel launch overhead low
    grid = (triton.cdiv(size, BLOCK),)
    elementwise_mul_kernel[grid](
        a, b, out, size, BLOCK=BLOCK
    )
    return out


# ------------------------------------------------------------
# Entry point
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/int) – may be None or unused
            - weights_dict: mapping of parameter names to torch.Tensor
            - config_dict: contains 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)  float32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (fp16)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # (B, N, N, hidden_dim)
    right = F.linear(x_fp16, right_proj_w)  # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcast)
    # ------------------------------------------------------------------
    if not nomask:
        # mask has shape (B, N, N); broadcast over hidden dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5. Gates (fp16) – sigmoid
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()   # (B, N, N, hidden_dim)
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()     # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 6. Fuse gating multiplications via Triton
    # ------------------------------------------------------------------
    left = fused_elementwise_mul(left, left_gate)   # (B, N, N, hidden_dim)
    right = fused_elementwise_mul(right, right_gate)

    # ------------------------------------------------------------------
    # 7. Core O(N³) contraction: batched GEMM using torch.bmm (Tensor Cores)
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to (B*hidden, N, N) for left and (B*hidden, N, N) for right (transposed)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)   # (i, k)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)  # (j, k)
    right_mat_T = right_mat.transpose(-2, -1)                                     # (k, j)

    out_mat = torch.bmm(left_mat, right_mat_T)   # (B*hidden, N, N), fp16

    # reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_norm = out_norm * out_gate.to(out_norm.dtype)   # apply out_gate (fp32)

    # final linear projection back to original dim
    out_final = F.linear(out_norm, weights["to_out.weight"])   # (B, N, N, dim), fp32

    return out_final
