"""
TriMul (outgoing) forward pass with a fused Triton batched GEMM.

Algorithm:
1. LayerNorm over the input channel dimension.
2. Linear projections (left, right) and their gates are computed in fp16.
3. Optional binary mask is applied (broadcast over the hidden dim).
4. Gates (sigmoid) are applied element‑wise.
5. Core contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication A @ B where
       A = left.permute(0,3,1,2)  (B, H, N, N)
       B = right.permute(0,3,2,1) (B, H, N, N) = right^T.
   The multiplication is performed by a custom Triton kernel that
   computes C = A @ B (both fp16) with float32 accumulation.
6. Output LayerNorm, element‑wise out‑gate, and final linear projection
   back to the original channel dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: batched matmul C = A @ B   (A: (B, M, K), B: (B, K, N))
# --------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                      # pointers
    M, N, K,                                   # matrix sizes
    stride_a_batch, stride_a_m, stride_a_k,    # strides for A
    stride_b_batch, stride_b_k, stride_b_n,    # strides for B
    stride_c_batch, stride_c_m, stride_c_n,    # strides for C
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)          # batch dimension (B * H)
    pid_m = tl.program_id(1)            # block row
    pid_n = tl.program_id(2)            # block col

    # Offsets for the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for this batch
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        # Load B tile (K x N)
        b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # fp16 dot, accumulates in fp32

    # Write back the result
    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def _triton_batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B (fp16 inputs, fp16 output)
    A : (batch, M, K)
    B : (batch, K, N)
    Returns C : (batch, M, N)
    """
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides (in elements)
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tunable tile sizes – chosen to fit typical B200 shared‑memory limits
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

# --------------------------------------------------------------
# Entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (bool/float)
            - weights_dict : mapping parameter name -> torch.Tensor
            - config_dict  : must contain 'dim' and 'hidden_dim'.
                            Optional 'nomask' (bool) indicates whether mask is ignored.

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    x_fp16 = x.to(torch.float16)

    # ----------------------------------------------------------
    # 2. Linear projections (no bias) – fp16 for speed
    # ----------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # ----------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ----------------------------------------------------------
    if not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)      # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # ----------------------------------------------------------
    # 4. Gates (sigmoid after linear) – fp16
    # ----------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------
    # 5. Core contraction – batched GEMM via Triton
    #    Compute out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    # ----------------------------------------------------------
    # Rearrange to (B, H, N, N); left: (i,k), right: (j,k) → right^T: (k,j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) (k,j)

    B, H, N, _ = left_perm.shape
    A_flat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    B_flat = right_perm.view(B * H, N, N)  # (B*H, N, N)

    # Perform batched matrix multiplication with Triton
    out_flat = _triton_batched_matmul(A_flat, B_flat)   # (B*H, N, N), fp16

    # Reshape back to (B, N, N, H) and cast to float32 for subsequent layers
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ----------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ----------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)               # element‑wise
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
