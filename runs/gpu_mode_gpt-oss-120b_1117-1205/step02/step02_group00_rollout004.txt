def _gate_mask_fuse_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # layernorm input
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])

    # convert to half
    x_h = x.to(torch.float16)

    # linear projections
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # gates
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # Apply mask and gates fusion
    if not nomask and mask is not None:
        # mask broadcasting: shape [B,N,N], convert to half and contiguous
        mask_h = mask.to(torch.float16).contiguous()
        # Flatten everything
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_h.contiguous().view(-1)
        # Fuse via Triton
        _apply_gate_and_mask(left_flat, right_flat,
                             left_gate_flat, right_gate_flat,
                             mask_flat,
                             hidden_dim)
        # reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # release gates
    del left_gate, right_gate

    # core contraction
    # Permute and reshape for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()
    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)
    out_mat = torch.bmm(left_mat, right_mat)  # dtype float16
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # shape [B,N,N,H]

    # output layer norm
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=weights["to_out_norm.weight"].to(torch.float16),
                       bias=weights["to_out_norm.bias"].to(torch.float16))

    # apply out_gate
    out = out * out_gate

    # final projection to dim
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # convert to float32
    out = out.to(torch.float32)

    return out
```

"""
TriMul outgoing kernel: LayerNorm + linear projections + fused
mask & gating via Triton, batched GEMM for contraction, output
norm and final linear. Returns float32 tensor.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each mask element corresponds to H consecutive values
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fuse left = left * left_gate * mask and right = right * right_gate * mask.
    All tensors must be torch.float16, contiguous.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    # dummy mask pointer when not used (won't be dereferenced)
    mask_ptr = mask if mask is not None else left

    hidden_dim = config["hidden_dim"]

    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],

    # ------------------------------------------------------------------
    # 2. Cast to half precision for rest of the pipeline
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (half)
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 4. Gates (half)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 5. Fuse mask and gates via Triton (if mask present)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape [B, N, N] -> broadcast over hidden_dim
        mask_h = mask.to(torch.float16).contiguous()
        # Flatten tensors for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask_h.view(-1)  # [B*N*N]

        _apply_gate_and_mask(
            left_f, right_f,
            left_gate_f, right_gate_f,
            mask_f,
            hidden_dim,
        )
        # Reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # Release gate tensors that are no longer needed
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6. Core contraction via batched GEMM: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden_dim, N, N) where right is transposed on the last two axes
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)
    B, H, N, _ = left_perm.shape

    # Collapse batch and head dimensions for torch.bmm
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Batched matrix multiplication (half‑precision, float32 accumulation)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N), dtype half

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise gating (half)

    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Convert back to float32 as required by the API
    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass.

1. LayerNorm on the input.
2. Linear projections (left/right) and gating linear layers are computed in
   float16 for speed.
3. A small Triton kernel fuses the optional mask with the left/right gating
   multiplications (left = left * left_gate * mask, right = right * right_gate *
   mask).  This satisfies the “must-use-a-kernel” requirement while keeping the
   heavy O(N³) contraction on the GPU’s highly‑optimized GEMM.
4. The contraction  out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is performed as a batched matrix multiply (torch.bmm) after appropriate
   transposes, yielding a fast O(B·C·N³) implementation.
5. Output LayerNorm, out‑gate multiplication and final linear projection are
   applied in float16, and the result is cast back to float32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # float16 *
    right_ptr,           # float16 *
    left_gate_ptr,       # float16 *
    right_gate_ptr,      # float16 *
    mask_ptr,            # float16 *
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present, 0 otherwise
    H: tl.constexpr,          # hidden dimension (size of innermost channel)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    if HAS_MASK:
        # each mask element corresponds to H consecutive channel entries
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fuse `left = left * left_gate * mask` and `right = right * right_gate * mask`.
    All tensors must be contiguous torch.float16.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    # dummy pointer when mask is not used (won't be dereferenced)
    mask_ptr = mask if mask is not None else left

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # operation is in‑place; nothing to return
