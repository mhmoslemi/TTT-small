"""
Highly‑optimized outgoing TriMul forward pass.

Algorithm
---------
1. Layer‑norm over the channel dimension (float32).
2. Linear projections to a hidden dimension (FP16, cuBLAS).
3. Optional mask broadcasting (FP16).
4. Gating: three linear layers + sigmoid, applied element‑wise (FP16).
5. Core contraction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (left @ rightᵀ) with a custom Triton kernel.
   The kernel works on the original [B,N,N,H] layout, avoiding costly permutes.
6. Cast to FP32, apply a second Layer‑norm, out‑gate multiplication and a
   final linear projection back to the original dimension.

All heavy arithmetic is in FP16 with FP32 accumulation. The kernel
processes (B·H) independent N×N GEMMs using a 3‑D grid:
   program_id(0) → (batch, hidden) pair,
   program_id(1) → tile‑row,
   program_id(2) → tile‑col.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel: batched GEMM for the outgoing TriMul contraction
# -------------------------------------------------------------
@triton.jit
def trimul_outgoing_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_j, stride_left_h,
    stride_right_batch, stride_right_i, stride_right_j, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # 0‑th program id = (batch * hidden) index
    bh_id = tl.program_id(0)
    # decode batch and hidden indices
    b = bh_id // H
    h = bh_id % H

    pid_m = tl.program_id(1)   # tile row
    pid_n = tl.program_id(2)   # tile col

    # ---------------------------------------------------------
    # Offsets for the current tile
    # ---------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for this (b,h) slice
    left_base  = left_ptr  + b * stride_left_batch  + h * stride_left_h
    right_base = right_ptr + b * stride_right_batch + h * stride_right_h
    out_base   = out_ptr   + bh_id * stride_out_batch  # out is flattened (B*H, N, N)

    # Accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------
    # Main K‑loop
    # ---------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- left tile: shape (BLOCK_M, BLOCK_K) -----
        # left[b, i, k, h]

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # ----- right tile (transposed) : shape (BLOCK_K, BLOCK_N) -----
        # right_T[k, j] = right[b, j, k, h]
        b_ptrs = right_base + offs_k[:, None] * stride_right_j + offs_n[None, :] * stride_right_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # ----- block‑wise dot product (FP16 * FP16 -> FP32) -----
        acc += tl.dot(a, b)

    # ---------------------------------------------------------
    # Write back
    # ---------------------------------------------------------
    c_ptrs = out_base + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def triton_outgoing_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for outgoing TriMul:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    left, right: [B, N, N, H] (torch.float16, contiguous)
    Returns out: [B*H, N, N] (torch.float16, contiguous)
    """
    assert left.dtype == torch.float16 and right.dtype == torch.float16
    B, N, _, H = left.shape
    total = B * H

    # Output buffer (flattened batch dimension)
    out = torch.empty((total, N, N), dtype=torch.float16, device=left.device)

    # Strides (as Python ints)
    s_left_b, s_left_i, s_left_j, s_left_h = left.stride()
    s_right_b, s_right_i, s_right_j, s_right_h = right.stride()
    s_out_b, s_out_i, s_out_j = out.stride()

    # Tile sizes – chosen for good occupancy on B200 (FP16, 64 KB shared memory)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        total,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    trimul_outgoing_gemm_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        s_left_b,  s_left_i,  s_left_j,  s_left_h,
        s_right_b, s_right_i, s_right_j, s_right_h,
        s_out_b,    s_out_i,   s_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# -------------------------------------------------------------
# Entry point expected by the evaluator
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operation.

    Arguments:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) – may be None
            - weights: dict of model parameters (torch.Tensor)
            - config: dict containing 'dim', 'hidden_dim', 'nomask' (optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ---------------------------------------------------------
    # Unpack arguments
    # ---------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------------------------------------------------------
    # 1. LayerNorm over the channel dimension (float32)
    # ---------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16 for the heavy arithmetic
    x_fp16 = x.to(torch.float16)

    # ---------------------------------------------------------
    # 2. Linear projections (no bias)
    # ---------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # ---------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # ---------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [B, N, N, 1] for broadcasting
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ---------------------------------------------------------
    # 4. Gating values (sigmoid) and application
    # ---------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------
    # 5. Core contraction using the Triton kernel
    # ---------------------------------------------------------
    # left/right are [B, N, N, H] fp16, contiguous after linear
    out_flat = triton_outgoing_gemm(left, right)   # shape (B*H, N, N), fp16

    # Restore original layout: (B, N, N, H)
    B, N, _, H = left.shape
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)   # back to FP32 for the final layers

    # ---------------------------------------------------------
    # 6. Output LayerNorm, out‑gate multiplication, final projection
    # ---------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"])   # weight is (dim, hidden_dim)

    return out
