"""
TriMul (outgoing) forward pass using Triton for the core contraction.
...

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Batch index for (batch, hidden) pair
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)   # tile row
    pid_n = tl.program_id(2)   # tile column

    # Compute offsets within the matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & (offs_k < K)),
                    other=0.0)

        # Load B tile (K x N)
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=((offs_k < K)[:, None] & mask_n[None, :]),
                    other=0.0)

        acc += tl.dot(a, b)

    # Write result back
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def triton_batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Compute batched matrix multiplication C = A @ B where
        A : (batch, M, K)  fp16
        B : (batch, K, N)  fp16
    Returns:
        C : (batch, M, N)  fp16
    """
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

    """
    Triton‑accelerated forward pass for the outgoing TriMul operation.

    Args:
        data (tuple): (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) – may be ignored when config["nomask"] is True
            - weights: dict containing model parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor: output of shape [B, N, N, dim] (float32)
    """

    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # --------------------------------------------------------------
    # 2. Cast to FP16 for heavy arithmetic
    # --------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Linear projections (left, right) – no bias
    # --------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 5. Gating values (sigmoid) and application
    # --------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6. Core contraction: batched GEMM left @ right^T via Triton
    # --------------------------------------------------------------
    # Rearrange to (batch, hidden, N, N) layout.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    # Right is transposed on the two sequence axes.
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) where last two dims are (j, i)

    B, H, N, _ = left_perm.shape

    # Merge batch and hidden dimensions for the batched GEMM.
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Compute the product. right_flat already stores the transposed operand.
    out_flat = triton_batched_matmul(left_flat, right_flat)   # (B*H, N, N), fp16

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate multiplication, final projection
    # --------------------------------------------------------------
    out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,),
                       weight=out_norm_weight,
                       bias=out_norm_bias)

    out = out * out_gate  # both fp16

    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Cast back to float32 as expected by downstream modules
    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass using Triton for the core contraction.

The operator works on a tensor of shape [B, N, N, C] and performs:
1. LayerNorm over the channel dimension.
2. Linear projections to a hidden dimension (left/right).
3. Optional masking.
4. Gated scaling (sigmoid gates).
5. Core contraction: out[...,i,j,d] = Σ_k left[...,i,k,d] * right[...,j,k,d].
   This is a batched matrix multiplication (left @ rightᵀ) performed with a
   custom Triton kernel to keep the heavy arithmetic on the GPU.
6. LayerNorm on the hidden dimension, multiplication by an output gate,
   and a final linear projection back to the original dimension.

All heavy arithmetic is carried out in FP16 with FP32 accumulation;
the final result is returned in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

    batch_id = tl.program_id(0)      # which (batch, hidden) pair
    pid_m = tl.program_id(1)         # tile row
    pid_n = tl.program_id(2)         # tile column

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch
