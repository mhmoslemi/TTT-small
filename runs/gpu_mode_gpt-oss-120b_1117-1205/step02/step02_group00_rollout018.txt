@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # load
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:

"""
Optimized Triton implementation of the outgoing TriMul module.
The kernel fuses element‑wise mask, left/right gating and optional mask
multiplication into a single pass over memory. The heavy O(N³) contraction
is performed with a batched GEMM (torch.bmm) which maps to cuBLAS
strided‑batched matrix multiplication.
All compute is done in float16 (FP16) except the very last conversion to
float32 for the returned tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # FP16 pointer
    right_ptr,              # FP16 pointer
    left_gate_ptr,          # FP16 pointer
    right_gate_ptr,         # FP16 pointer
    mask_ptr,               # FP16 pointer (broadcasted)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present, 0 otherwise
    H: tl.constexpr,          # hidden dimension
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # load tensors (half precision)
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # each hidden entry shares the same mask value
        mask_offs = offs // H
        mask_val = tl.load(mask_ptr + mask_offs, mask=mask, other=tl.float16(1.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    Fuse ``left = left * left_gate * mask`` and ``right = right * right_gate * mask``
    using a Triton kernel. All tensors must be contiguous and have dtype torch.float16.
    """
    total = left.numel()
    BLOCK = 16384  # adjust for occupancy
    grid = (triton.cdiv(total, BLOCK),)

    HAS_MASK = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when mask not used

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=HAS_MASK,
        H=hidden_dim,
    )

def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator implemented with Triton.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or numeric)
            - weights: dict mapping parameter names to torch.Tensor
            - config: dict containing ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Layer‑norm (half precision)
    # ------------------------------------------------------------------
    x = input_tensor.to(torch.float16)
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias = weights["norm.bias"].to(torch.float16)
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2. Combined linear projections (half)
    #    left: proj + gate, right: proj + gate
    # ------------------------------------------------------------------
    left_combined_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)          # shape (2*hidden_dim, dim)
    right_combined_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)          # shape (2*hidden_dim, dim)

    left_combined = F.linear(x, left_combined_weight)   # (B, N, N, 2*hidden_dim)
    right_combined = F.linear(x, right_combined_weight)

    # split into projection and gate components
    left_proj = left_combined[..., :hidden_dim].contiguous()
    left_gate = left_combined[..., hidden_dim:].contiguous().sigmoid()
    right_proj = right_combined[..., :hidden_dim].contiguous()
    right_gate = right_combined[..., hidden_dim:].contiguous().sigmoid()

    # ------------------------------------------------------------------
    # 3. Fuse mask (if present) and gating with Triton
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)   # shape (B*N*N,)
    else:
        mask_flat = None

    # Ensure tensors are contiguous before kernel launch
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    _apply_gate_and_mask(
        left_proj, right_proj,
        left_gate, right_gate,
        mask_flat,
        hidden_dim,
    )
    # After kernel, left_proj and right_proj contain the masked‑gated tensors

    # ------------------------------------------------------------------
    # 4. Core O(N³) contraction via batched GEMM
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) where H = hidden_dim
    left_perm = left_proj.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_proj.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) where k is leading dim

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.reshape(B * H, N, N)
    right_mat = right_perm.reshape(B * H, N, N)

    # batched matrix multiplication (FP16, internal FP32 accumulation)
    out_mat = torch.bmm(left_mat, right_mat)   # (B*H, N, N), dtype=half

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Output layer‑norm, out‑gate, final projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    out_gate = F.linear(x, out_gate_weight).sigmoid()   # shape (B,N,N,hidden_dim)
    out = out * out_gate

    to_out_weight = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_weight)

    # return in float32 as required by the specification
    return out.to(torch.float32)

"""
Optimized Triton implementation of the outgoing TriMul operator.
The Triton kernel fuses element‑wise mask application, left/right gating,
and optional mask multiplication into a single pass. The heavy O(N³)
contraction is performed with a batched GEMM (torch.bmm) which maps to
cuBLAS strided‑batched matrix multiplication. All compute is in FP16
except the final cast back to FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: fuse mask (optional) + left/right gates
# --------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # [total] fp16
    right_ptr,              # [total] fp16
    left_gate_ptr,          # [total] fp16
    right_gate_ptr,         # [total] fp16
    mask_ptr,               # [mask_total] fp16 (broadcasted)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask is present, 0 otherwise
    H: tl.constexpr,          # hidden dimension (size of innermost dim)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    # load tensors
    left = tl.load(left_ptr + offs, mask=valid, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=valid, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=tl.float16(0.0))

    if HAS_MASK:
        # one mask element corresponds to H consecutive hidden entries
        mask_offs = offs // H
        mask_val = tl.load(mask_ptr + mask_offs, mask=valid, other=tl.float16(1.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fuse ``left = left * left_gate * mask`` and
    ``right = right * right_gate * mask`` (if ``mask`` is provided).
    All tensors must be contiguous and of dtype ``torch.float16``.
    """
    total = left.numel()
    BLOCK = 16384
    grid = (triton.cdiv(total, BLOCK),)

    HAS_MASK = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when mask not used

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=HAS_MASK,
        H=hidden_dim,
    )


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or numeric)
            - weights: dict mapping parameter names to torch.Tensor
            - config: dict containing ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP16)
    # ------------------------------------------------------------------
    x = input_tensor.to(torch.float16)
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias = weights["norm.bias"].to(torch.float16)
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2. Combined linear projections (FP16)
    #    left: proj + gate, right: proj + gate
    # ------------------------------------------------------------------
    left_combined_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)          # (2*hidden_dim, dim)
    right_combined_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)          # (2*hidden_dim, dim)

    left_combined = F.linear(x, left_combined_weight)   # (B, N, N, 2*hidden_dim)
    right_combined = F.linear(x, right_combined_weight)

    # split into projection and gate components
    left_proj = left_combined[..., :hidden_dim].contiguous()
    left_gate = left_combined[..., hidden_dim:].contiguous().sigmoid()
    right_proj = right_combined[..., :hidden_dim].contiguous()
    right_gate = right_combined[..., hidden_dim:].contiguous().sigmoid()

    # ------------------------------------------------------------------
    # 3. Fuse mask (if any) and gating with Triton
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)          # (B*N*N,)
    else:
        mask_flat = None

    # Ensure contiguity before kernel launch
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    _apply_gate_and_mask(
        left_proj, right_proj,
        left_gate, right_gate,
        mask_flat,
        hidden_dim,
    )

    # ------------------------------------------------------------------
    # 4. Core O(N³) contraction via batched GEMM (FP16, FP32 accumulation)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N)
    left_perm = left_proj.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_proj.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) where k is leading dim

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)                # (B*H, N, N), fp16

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, out‑gate, and final linear projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    out_gate = F.linear(x, out_gate_weight).sigmoid()     # (B, N, N, hidden_dim)
    out = out * out_gate

    to_out_weight = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_weight)                    # (B, N, N, dim)

    # Return in FP32 as required
    return out.to(torch.float32)
