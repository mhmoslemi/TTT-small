Define @triton.jit kernel:

def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    BH, N, K,
    stride_left_bh, stride_left_m, stride_left_k,
    stride_right_bh, stride_right_n, stride_right_k,  # note right uses (BH,N,K) but we treat accordingly
    stride_out_bh, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_bh = tl.program_id(0)  # batch_head index
    pid_m = tl.program_id(1)   # block row
    pid_n = tl.program_id(2)   # block col

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # boundaries
    m_mask = offs_m < N
    n_mask = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = offs_k < K

        # compute pointers for left: shape (BH, M, K)
        left_offset = pid_bh * stride_left_bh + offs_m[:, None] * stride_left_m + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptr + left_offset, mask=m_mask[:, None] & k_mask[None, :], other=0.0).to(tl.float32)

        # compute pointers for right transposed: we need B_T shape (BH, K, N)
        right_offset = pid_bh * stride_right_bh + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_n
        # note: stride_right_k is stride of original B along last axis K; stride_right_n is stride along original N axis (for B_T's second dimension)
        right = tl.load(right_ptr + right_offset, mask=k_mask[:, None] & n_mask[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(left, right)  # expects shapes (BLOCK_M, BLOCK_K) and (BLOCK_K, BLOCK_N)
    # store result
    out_offset = pid_bh * stride_out_bh + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    tl.store(out_ptr + out_offset, acc.to(tl.float16), mask=m_mask[:, None] & n_mask[None, :])
```

@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,    # strides for A: batch, M, K
    stride_bm, stride_bk, stride_bb,    # strides for B: batch, K?, N? Actually B shape (batch, K, N) if we treat B as transposed? But we can pass strides accordingly.
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0) # batch dimension
    pid_m = tl.program_id(1)     # block row
    pid_n = tl.program_id(2)     # block col

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_ptr + pid_batch * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + pid_batch * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        # Wait, define B strides: stride_bk for K dimension, stride_bn for N dimension.
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # store
    c_ptrs = c_ptr + pid_batch * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ah, stride_am, stride_ak,
    stride_bh, stride_bk, stride_bn,
    stride_ch, stride_cm, stride_cn,

    pid_n = tl.program_id(2)

    mask_n = offs_n < N

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])
    x_h = x.to(torch.float16)

    # Projections
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # Gates
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # Apply mask & gates
    if not nomask:
        # mask shape: [B, N, N], broadcast over hidden_dim
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        # flatten
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_h.view(-1)
        _apply_gate_and_mask(
            left_flat, right_flat,
            left_gate_flat, right_gate_flat,
            mask_flat,
            hidden_dim,
        )
        # reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # Release unneeded
    del left_gate, right_gate, x_h

    # Permute for batched GEMM
    # left: (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right_T: (B, H, N, N) where dimension order is (k, j)
    right_T = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape
    BH = B * H

    # view as (BH, N, N)
    a = left_perm.view(BH, N, N)
    b = right_T.view(BH, N, N)  # b shape (BH, K=N, N)

    # allocate output
    c = torch.empty_like(a)  # same dtype (float16)

    # kernel launch
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (
        BH,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _tri_mul_gemm_kernel[grid](
        a, b, c,
        N, N, N,
        a.stride(0), a.stride(1), a.stride(2),
        b.stride(0), b.stride(1), b.stride(2),
        c.stride(0), c.stride(1), c.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # reshape back to (B, H, N, N)
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output layernorm and final projection
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=weights["to_out_norm.weight"].to(torch.float16),
                       bias=weights["to_out_norm.bias"].to(torch.float16))
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    return out.to(torch.float32)

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,
):

    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:

        mask_off = offs // H
        mask_val = tl.load(mask_ptr + mask_off, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

"""
Optimized TriMul (outgoing) forward pass.

The heavy O(N³) contraction is implemented as a custom Triton batched GEMM
kernel (C = A @ Bᵀ) where A = left_proj * left_gate * mask and Bᵀ is the
transposed right projection.  A separate Triton kernel fuses the optional mask
with the left/right gate multiplications.  All other operations (LayerNorm,
Linear, sigmoid) are performed with PyTorch in half‑precision; the final
output is cast back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Kernel: fuse mask (optional) with left/right gates
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # half*
    right_ptr,              # half*
    left_gate_ptr,          # half*
    right_gate_ptr,         # half*
    mask_ptr,               # half* (broadcasted)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present else 0
    H: tl.constexpr,          # hidden_dim (size of innermost channel)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each GROUP of H consecutive elements shares the same mask value
        mask_off = offs // H
        mask_val = tl.load(mask_ptr + mask_off, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fuse ``left = left * left_gate * mask`` and
    ``right = right * right_gate * mask`` (if mask is provided).
    All tensors must be contiguous ``torch.float16``.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when HAS_MASK==0

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # operation is in‑place; no return


# ----------------------------------------------------------------------
# Kernel: batched GEMM for the core N³ contraction (C = A @ Bᵀ)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # half* pointers
    M, N, K,                           # int dimensions (M=N=seq_len, K=N)
    stride_ah, stride_am, stride_ak,   # strides for A (batch, M, K)
    stride_bh, stride_bk, stride_bn,   # strides for B (batch, K, N) -- B already transposed
    stride_ch, stride_cm, stride_cn,   # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(0)   # batch‑head index
    pid_m = tl.program_id(1)    # block row
    pid_n = tl.program_id(2)    # block column

    # linear offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + pid_bh * stride_ah + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # B tile (already transposed): shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_ptr + pid_bh * stride_bh + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # write back C tile
    c_ptrs = c_ptr + pid_bh * stride_ch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) or None
            - weights: mapping of parameter names to torch.Tensor
            - config: dict with keys ``dim``, ``hidden_dim``, optional ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> half)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half‑precision)
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Gates (half‑precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) and gates via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> broadcast over hidden_dim
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        # flatten everything for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask_h.contiguous().view(-1)          # [B*N*N]
        _apply_gate_and_mask(
            left_f, right_f,
            left_gate_f, right_gate_f,
            mask_f,
            hidden_dim,
        )
        # reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask – simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # Clean up intermediate tensors
    del left_gate, right_gate, x_h

    # ------------------------------------------------------------------
    # 5. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as a batched GEMM (A @ Bᵀ) using Triton.
    # ------------------------------------------------------------------
    # left:  (B, N, N, H) -> (B, H, N, N)  (rows i, cols k)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: (B, N, N, H) -> (B, H, N, N) with axes (k, j) for GEMM
    right_T = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape
    BH = B * H

    # view as (BH, N, N) matrices
    A = left_perm.view(BH, N, N)
    B_mat = right_T.view(BH, N, N)   # already transposed for GEMM

    # output buffer
    C = torch.empty_like(A)          # (BH, N, N), half precision

    # launch Triton GEMM kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (
        BH,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _tri_mul_gemm_kernel[grid](
        A, B_mat, C,
        N, N, N,
        A.stride(0), A.stride(1), A.stride(2),
        B_mat.stride(0), B_mat.stride(1), B_mat.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 7. Cast back to float32 as required by the interface
    # ------------------------------------------------------------------
    return out.to(torch.float32)
