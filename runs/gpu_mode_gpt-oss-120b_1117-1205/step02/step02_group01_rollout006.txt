import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def mul_gate_mask_kernel(out_ptr, proj_ptr, gate_ptr, mask_ptr, total, hidden_dim, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total
    proj = tl.load(proj_ptr + offs, mask=mask)
    gate = tl.load(gate_ptr + offs, mask=mask)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask)
    result = proj * gate * m
    tl.store(out_ptr + offs, result, mask=mask)
```

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data

    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # layer norm
    norm_weight = weights["norm.weight"].to(device).to(torch.float32)
    norm_bias = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # cast to fp16
    x_fp16 = x.to(torch.float16)

    # linear projections
    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)

    left = F.linear(x_fp16, left_proj_weight)   # shape [B, N, N, hidden_dim]
    right = F.linear(x_fp16, right_proj_weight)

    # gating
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))

    # mask
    if mask_tensor is None:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(torch.float16).to(device)

    total_elements = left.numel()   # B * N * N * hidden_dim
    BLOCK = 1024  # adjust
    grid = (triton.cdiv(total_elements, BLOCK),)

    # flatten tensors
    left_flat = left.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_flat = right.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)
    mask_flat = mask_fp16.contiguous().view(-1)

    left_out_flat = torch.empty_like(left_flat)
    right_out_flat = torch.empty_like(right_flat)

    mul_gate_mask_kernel[grid](
        left_out_flat,
        left_flat,
        left_gate_flat,
        mask_flat,
        total_elements,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
    )
    mul_gate_mask_kernel[grid](
        right_out_flat,
        right_flat,
        right_gate_flat,
        mask_flat,
        total_elements,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
    )

    # reshape back
    left = left_out_flat.view(B, N, N, hidden_dim)
    right = right_out_flat.view(B, N, N, hidden_dim)

    # cubic contraction via batched GEMM
    # permute to shape (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden_dim, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, hidden_dim, N, N) with k as rows
    # reshape to (B*hidden_dim, N, N)
    LH = B * hidden_dim
    left_mat = left_perm.view(LH, N, N)   # (B*hidden_dim, N, N)
    right_mat = right_perm.view(LH, N, N) # same shape
    # batched matmul (left_mat @ right_mat)
    out_mat = torch.bmm(left_mat, right_mat)  # shape (B*hidden_dim, N, N)

    # reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # post processing: to_out_norm, out_gate, to_out linear
    # convert to fp32 for layernorm and final linear for better precision
    out_fp32 = out.to(torch.float32)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float32)
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # apply final gate (out_gate from fp16)
    out_gate_fp32 = out_gate.to(torch.float32)
    out_norm = out_norm * out_gate_fp32

    # final linear
    to_out_weight = weights["to_out.weight"].to(device).to(torch.float32)  # shape (dim, hidden_dim)
    output = F.linear(out_norm, to_out_weight)    # shape [B, N, N, dim]
    return output
```

"""
TriMul (outgoing) forward pass.

Core ideas:
- Perform the initial LayerNorm and all linear projections in PyTorch (FP16).
- Fuse the three element‑wise operations (proj * gate * mask) for the left and
  right branches with a tiny Triton kernel, avoiding three separate launches.
- Compute the cubic contraction with a batched GEMM:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
  by reshaping the tensors to (B*hidden, N, N) and calling `torch.bmm`.
- Apply a second LayerNorm, gating and final linear projection (FP32) to produce
  the result tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def mul_gate_mask_kernel(
    out_ptr, proj_ptr, gate_ptr, mask_ptr,  # pointers
    total_elems, hidden_dim,                # runtime scalars
    BLOCK_SIZE: tl.constexpr                # compile‑time constant
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # load projection and gate values
    proj = tl.load(proj_ptr + offs, mask=mask)
    gate = tl.load(gate_ptr + offs, mask=mask)

    # compute index for the broadcasted mask (mask has no hidden_dim)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # store the fused result
    tl.store(out_ptr + offs, proj * gate * m, mask=mask)


def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, dim] (float32)
            mask_tensor  : torch.Tensor [B, N, N] or None (float32)
            weights_dict : dict with the following keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            config_dict  : {"dim": int, "hidden_dim": int}
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------
    # 2. Cast to FP16 for heavy work
    # --------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Linear projections (no bias)
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # [B,N,N,hidden_dim], fp16
    right = F.linear(x_fp16, right_proj_w)  # [B,N,N,hidden_dim], fp16

    # --------------------------------------------------------------
    # 4. Gates (sigmoid) and optional mask – fused via Triton
    # --------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    if mask_tensor is None:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(torch.float16).to(device)

    # flatten tensors for the Triton kernel
    total = left.numel()                     # B * N * N * hidden_dim
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    left_flat = left.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_flat = right.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)
    mask_flat = mask_fp16.contiguous().view(-1)

    # In‑place fusion: left = left * left_gate * mask
    mul_gate_mask_kernel[grid](
        left_flat,          # out_ptr (in‑place)
        left_flat,          # proj_ptr
        left_gate_flat,     # gate_ptr
        mask_flat,          # mask_ptr
        total,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
    )
    # In‑place fusion: right = right * right_gate * mask
    mul_gate_mask_kernel[grid](
        right_flat,
        right_flat,
        right_gate_flat,
        mask_flat,
        total,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
    )

    # restore the original shapes (still FP16)
    left = left_flat.view(B, N, N, hidden_dim)
    right = right_flat.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 5. Cubic contraction via batched GEMM (torch.bmm)
    # --------------------------------------------------------------
    # reshape to (B, hidden_dim, N, N) and then to (B*hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden_dim, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, hidden_dim, N, N) with k as rows

    LH = B * hidden_dim
    left_mat = left_perm.view(LH, N, N)   # (B*hidden_dim, N, N)
    right_mat = right_perm.view(LH, N, N) # (B*hidden_dim, N, N)

    # batched matrix multiplication over k dimension
    out_mat = torch.bmm(left_mat, right_mat)   # (B*hidden_dim, N, N), fp16

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------
    # 6. Post‑processing: LayerNorm on hidden_dim, final gate, projection
    # --------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float32)
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # apply final gating (out_gate is fp16)
    out_norm = out_norm * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float32)  # (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)  # [B, N, N, dim], float32

    return output
