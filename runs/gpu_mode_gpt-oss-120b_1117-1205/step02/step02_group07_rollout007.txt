@triton.jit
def batched_matmul_kernel(a_ptr, b_ptr, c_ptr, M, N, K,
                          stride_a_batch, stride_a_m, stride_a_k,
                          stride_b_batch, stride_b_k, stride_b_n,
                          stride_c_batch, stride_c_m, stride_c_n,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    a_ptrs = a_ptr + batch*stride_a_batch
    b_ptrs = b_ptr + batch*stride_b_batch
    c_ptrs = c_ptr + batch*stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A tile
        a_tile_ptrs = a_ptrs + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_tile_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        # Load B tile: B is (K, N)
        b_tile_ptrs = b_ptrs + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_tile_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    # Store result
    c_tile_ptrs = c_ptrs + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_tile_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

    assert a.shape[0] == b.shape[0]
    B_total, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2
    c = torch.empty((B_total, M, N), dtype=torch.float16, device=a.device)
    # Strides
    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_k, stride_b_n = b.stride()
    stride_c_batch, stride_c_m, stride_c_n = c.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B_total,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),

        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c

"""
TriMul (outgoing) forward pass with a fused Triton batched GEMM.

Algorithm
---------
1. Layer‑norm over the channel dimension (float32).
2. Linear projections (left/right) and gating (left_gate/right_gate/out_gate) are
   performed in float16 for speed.  An optional mask is applied to the projected
   tensors before gating.
3. The core contraction
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   is a batched matrix multiplication for every (b,d):
        out_{b,d} = left_{b,d} @ right_{b,d}ᵀ
   where left_{b,d} and right_{b,d} are (N×N) matrices.
   This is accelerated with a Triton kernel that computes
   C = A @ B (A: M×K, B: K×N) with fp16 inputs and fp32 accumulation.
4. The result is reshaped back, followed by a second Layer‑norm, an output‑gate
   scaling, and a final linear projection back to the original dimension
   (float32).

The kernel is launched with block sizes tuned for N up to 1024
and hidden_dim (H) up to a few hundred.  The implementation works for
both masked and un‑masked inputs and supports batch sizes 1‑2.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------
# Triton batched GEMM kernel (C = A @ B, B is already transposed)
# ------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute C = A @ B for a batch of matrices.

    A: (batch, M, K)  fp16
    B: (batch, K, N)  fp16 (already transposed for the contraction)
    C: (batch, M, N)  fp16 (fp32 accumulation)
    """
    batch_id = tl.program_id(0)       # batch dimension (B * hidden_dim)
    pid_m = tl.program_id(1)          # block row
    pid_n = tl.program_id(2)          # block column

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension in tiles of size BLOCK_K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k < K),
            other=0.0
        )

        # Load B tile (K x N) – B is already transposed (k, n)
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(
            b_ptrs,
            mask=(offs_k < K)[:, None] & mask_n[None, :],
            other=0.0
        )

        # Accumulate
        acc += tl.dot(a, b)

    # Store the result
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :]
    )


def triton_batched_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Wrapper for the Triton batched GEMM.
    a: (B_total, M, K)  fp16
    b: (B_total, K, N)  fp16
    Returns:
        c: (B_total, M, N) fp16
    """
    assert a.dtype == b.dtype == torch.float16
    assert a.is_contiguous() and b.is_contiguous()
    B_total, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    c = torch.empty((B_total, M, N), dtype=torch.float16, device=a.device)

    # Strides in elements
    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_k, stride_b_n = b.stride()
    stride_c_batch, stride_c_m, stride_c_n = c.stride()

    # Tuned block sizes (work well for N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B_total,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        a, b, c,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c


# ============================================================
# Entry point
# ============================================================
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or numeric)
            - weights_dict: mapping from parameter names to torch.Tensor
            - config_dict: must contain 'dim' and 'hidden_dim'; may contain
                           'nomask' (bool, default True)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim], fp32

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for heavy linear work
    # -----------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections (fp16)
    # -----------------------------------------------------------------
    left = F.linear(
        x_fp16,
        weights["left_proj.weight"].to(torch.float16)
    )  # [B, N, N, hidden_dim]
    right = F.linear(
        x_fp16,
        weights["right_proj.weight"].to(torch.float16)
    )

    # -----------------------------------------------------------------
    # 4. Optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)  # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # -----------------------------------------------------------------
    # 5. Gating (fp16) and apply
    # -----------------------------------------------------------------
    left_gate = F.linear(
        x_fp16,
        weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = F.linear(
        x_fp16,
        weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = F.linear(
        x_fp16,
        weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6. Core contraction via Triton batched GEMM
    #    Compute out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # -----------------------------------------------------------------
    # Permute to bring hidden dimension into the batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)  (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)  (k,j)

    B, H, N, _ = left_perm.shape
    # Flatten batch and hidden dimensions for the kernel
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Batched matrix multiplication (fp16 in/out, fp32 accumulation)
    out_flat = triton_batched_matmul(left_flat, right_flat)  # (B*H, N, N)

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # -----------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final projection (float32)
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"])

    return out
