"""
Highly‑optimized Triton implementation of the outgoing TriMul operator.

Algorithm:
1. Layer‑norm + linear projections & gating are done in PyTorch (fp16 for speed).
2. The core contraction
        out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
   is a batched matrix multiplication for each hidden channel d.
   A custom Triton kernel fuses the hidden‑channel loop inside the kernel,
   drastically reducing kernel launch overhead while keeping the compute on Tensor‑cores.
3. The result (fp16) is cast back to fp32, followed by the output LayerNorm,
   out‑gate and final linear projection (all fp32).

The kernel iterates over the hidden dimension (d) with a simple while‑loop,
and over the reduction dimension (k) with a tiled for‑loop.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------
# Triton kernel for the TriMul contraction
# ------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(0)      # batch axis (0 … B‑1)
    pid_i = tl.program_id(1)      # block row of output
    pid_j = tl.program_id(2)      # block col of output

    # base pointers for this batch
    a_base = a_ptr + pid_b * stride_a_batch
    b_base = b_ptr + pid_b * stride_b_batch
    c_base = c_ptr + pid_b * stride_c_batch

    # thread‑block offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # iterate over hidden channels d = 0 … H‑1
    d = tl.zeros([], dtype=tl.int32)          # scalar = 0
    while d < H:
        # accumulator for this (i,j) tile and channel d
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # reduction over k
        for kk in range(0, tl.cdiv(N, BLOCK_K)):
            offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # left tile: shape (BLOCK_M, BLOCK_K)
            a_ptrs = a_base \
                     + offs_i[:, None] * stride_a_i \
                     + offs_k[None, :] * stride_a_k \
                     + d * stride_a_h
            a = tl.load(
                a_ptrs,
                mask=mask_i[:, None] & mask_k[None, :],
                other=0.0,
            )

            # right tile (transposed): shape (BLOCK_K, BLOCK_N)
            b_ptrs = b_base \
                     + offs_k[:, None] * stride_b_k \
                     + offs_j[None, :] * stride_b_j \
                     + d * stride_b_h
            b = tl.load(
                b_ptrs,
                mask=mask_k[:, None] & mask_j[None, :],
                other=0.0,
            )

            # matrix‑multiply‑accumulate (fp16⋅fp16 → fp32)
            acc += tl.dot(a, b)

        # store the result for channel d
        c_ptrs = c_base \
                 + offs_i[:, None] * stride_c_i \
                 + offs_j[None, :] * stride_c_j \
                 + d * stride_c_h
        tl.store(
            c_ptrs,
            acc.to(tl.float16),
            mask=mask_i[:, None] & mask_j[None, :],
        )
        d += 1


def triton_tri_mul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Triton‑accelerated contraction of the TriMul operator.
    left, right : [B, N, N, H]  (fp16)
    returns      : [B, N, N, H]  (fp16)
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape

    out = torch.empty_like(left)

    # strides in **elements** (not bytes)
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h = left.stride()
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h = right.stride()
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h = out.stride()

    # tile sizes – tuned for Tensor Cores on Ampere/A100-like GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_a_batch,
        stride_a_i,
        stride_a_k,
        stride_a_h,
        stride_b_batch,
        stride_b_j,
        stride_b_k,
        stride_b_h,
        stride_c_batch,
        stride_c_i,
        stride_c_j,
        stride_c_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out

# Main entry point required by the test harness
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim]   (float32)
            - mask       : torch.Tensor [B, N, N]        (bool or numeric)
            - weights    : dict of model parameters (torch.Tensors)
            - config     : dict with keys 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  float32

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for the heavy linear ops
    # -----------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections (fp16)
    # -----------------------------------------------------------------
    left = F.linear(
        x_fp16, weights["left_proj.weight"].to(torch.float16)
    )  # [B, N, N, hidden_dim]
    right = F.linear(
        x_fp16, weights["right_proj.weight"].to(torch.float16)
    )

    # -----------------------------------------------------------------
    # 4. Optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)  # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # -----------------------------------------------------------------
    # 5. Gating (fp16) and apply
    # -----------------------------------------------------------------
    left_gate = F.linear(
        x_fp16, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = F.linear(
        x_fp16, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = F.linear(
        x_fp16, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6. Core contraction via Triton
    # -----------------------------------------------------------------
    out_fp16 = triton_tri_mul(left, right)               # [B, N, N, hidden_dim] fp16
    out = out_fp16.to(torch.float32)                     # cast back to fp32 for the rest

    # -----------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate and final projection (float32)
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
