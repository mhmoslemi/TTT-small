@triton.jit
def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_x = tl.program_id(0)
    pid_y = tl.program_id(1)
    pid_z = tl.program_id(2)

    # compute batch/head indices
    bh = pid_z
    b = bh // H
    h = bh % H

    # compute base pointers
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # compute start offsets for i and j
    i0 = pid_x * BLOCK_M
    j0 = pid_y * BLOCK_N

    # Create ranges
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    # Masks for out-of-bounds rows and cols
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile (left)
        # shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + (offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B tile (right) but we need B_T: B has shape (j,k) but we want (k,j)
        b_ptrs = right_base + (offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Convert to float32 and compute dot
        acc += tl.dot(a, b)

    # Write back output
    c_ptrs = out_base + (offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""TriMul outgoing operator optimized using Triton for batched GEMM.

The kernel computes out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
directly on the original 4‑D layout, avoiding costly permutes/copies.
All other operations (LayerNorm, linear projections, gating, mask)
are performed with PyTorch in half precision.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_x = tl.program_id(0)  # row tile
    pid_y = tl.program_id(1)  # column tile
    pid_z = tl.program_id(2)  # batch * head

    bh = pid_z
    b = bh // H
    h = bh % H

    # base pointers for this (b, h)
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    i0 = pid_x * BLOCK_M
    j0 = pid_y * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a tile from left: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + (offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load a tile from right (transposed): shape (BLOCK_K, BLOCK_N)
        # right is indexed as (b, j, k, h); we need B_T[k, j]
        b_ptrs = right_base + (offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)

    c_ptrs = out_base + (offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """Forward pass of the outgoing TriangleMultiplicativeUpdate (TriMul).

    Args:
        data: Tuple of (input_tensor, mask_tensor, weights_dict, config_dict).
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32).
            - mask_tensor: torch.Tensor of shape [B, N, N] (bool or float), may be None.
            - weights: mapping from parameter names to torch.Tensor (float32).
            - config: dict containing ``dim`` and ``hidden_dim`` and optional ``nomask`` flag.

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. Input LayerNorm (float32)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # Cast to half for the rest of the pipeline
    x_h = x.to(torch.float16)

    # 2. Linear projections (half precision)
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # 3. Gating (half precision, sigmoid)
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # 4. Apply mask (if any) and gates
    # mask broadcast over hidden_dim
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h
    left = left * left_gate
    right = right * right_gate

    # Release intermediate tensors no longer needed
    del x, x_h, left_gate, right_gate

    # 5. Core contraction using Triton batched GEMM
    B, N, _, H = left.shape  # left.shape = (B, N, N, hidden_dim)
    # Allocate output tensor
    out = torch.empty_like(left)

    # Strides for the three tensors (in elements)
    left_stride_b, left_stride_i, left_stride_k, left_stride_h = left.stride()
    right_stride_b, right_stride_i, right_stride_k, right_stride_h = right.stride()
    out_stride_b, out_stride_i, out_stride_k, out_stride_h = out.stride()

    # Triton expects strides in number of elements, not bytes, so we can pass directly
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )
    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        left_stride_b, left_stride_i, left_stride_k, left_stride_h,
        right_stride_b, right_stride_i, right_stride_k, right_stride_h,
        out_stride_b, out_stride_i, out_stride_k, out_stride_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # 6. Output LayerNorm, out‑gate, final linear projection
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return as float32 as required
    return out.to(torch.float32)

    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_x = tl.program_id(0)  # tile row index
    pid_y = tl.program_id(1)  # tile col index
    pid_z = tl.program_id(2)  # batch * head index

    # ------------------------------------------------------------------
    # Decode batch and head from the third dimension
    # ------------------------------------------------------------------
    bh = pid_z
    b = bh // H
    h = bh % H

    # Base pointers for this (b, h) slice
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # Tile start offsets
    i0 = pid_x * BLOCK_M
    j0 = pid_y * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + (offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B tile transposed: shape (BLOCK_K, BLOCK_N)
        # right is indexed as (b, j, k, h); we need B_T[k, j]
        b_ptrs = right_base + (offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N) in FP32

    # Write back the result in FP16
    c_ptrs = out_base + (offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[:, None])


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor – torch.Tensor[B, N, N, dim] (float32)
            mask_tensor  – torch.Tensor[B, N, N] (bool/float) or None
            weights      – mapping name → torch.Tensor (float32)
            config       – dict with keys ``dim``, ``hidden_dim``, optional ``nomask``

    Returns:
        torch.Tensor[B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32 → half)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections (half precision)
    # --------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 3. Gates (half precision)
    # --------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # --------------------------------------------------------------
    # 4. Apply mask (if any) and element‑wise gating
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h
    left = left * left_gate
    right = right * right_gate

    # Release no‑longer‑needed tensors
    del x, x_h, left_gate, right_gate

    # --------------------------------------------------------------
    # 5. Core contraction via Triton (batched GEMM without copies)
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # left: [B, N, N, hidden_dim]
    out = torch.empty_like(left)

    # Strides (in element steps)
    lsb, lsi, lsk, lsh = left.stride()
    rsb, rsi, rsk, rsh = right.stride()
    osb, osi, osj, osh = out.stride()   # osi → i‑dim, osj → j‑dim

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )
    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        lsb, lsi, lsk, lsh,
        rsb, rsi, rsk, rsh,
        osb, osi, osj, osh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return as float32 as required by the interface
    return out.to(torch.float32)
