def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # Load weights and move to same device/dtype
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    # compute layernorm
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)  # float32
    # cast to fp16
    x_fp16 = x.to(torch.float16)
    # load projection weights
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)
    # gates
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))

        right,
        right_gate,
        mask_fp16,
        total,

"""
TriMul (outgoing) forward pass with a Triton‑accelerated elementwise
mask‑gate fusion.  The heavy linear & gating work is done in FP16 with
torch.nn.functional.linear.  The mask, gate and projection are fused in a
single Triton kernel to avoid extra memory passes.  The cubic contraction
is realized with a batched matrix‑multiply via torch.einsum, which maps
to cuBLAS’s batched GEMM and therefore runs efficiently on the GPU.
All final layers (LayerNorm, output Linear) operate in FP32 to match the
reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def mul_gate_mask_kernel(
    out_ptr,          # pointer to output tensor (in‑place)
    a_ptr,            # pointer to source tensor (same shape as out)
    gate_ptr,         # pointer to gate tensor (same shape)
    mask_ptr,         # pointer to mask tensor (shape B·N·N)
    total,            # total number of elements in out / a / gate
    BLOCK_SIZE: tl.constexpr,
    H: tl.constexpr,  # hidden dimension (size of innermost axis)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    a = tl.load(a_ptr + offs, mask=mask, other=tl.float16(0.0))
    g = tl.load(gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    # broadcast mask over the hidden dimension
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(1.0))

    tl.store(out_ptr + offs, a * g * m, mask=mask)


def custom_kernel(data):
    """
    TriMul forward pass (outgoing) using Triton for the mask‑gate multiplication.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            mask_tensor : torch.Tensor of shape [B, N, N] (float32) or None
            weights_dict: mapping of weight names to torch.Tensors
            config_dict : {"dim": int, "hidden_dim": int, "nomask": bool}
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0. unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the heavy work
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias) – FP16
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # [B,N,N,hidden_dim]
    right = F.linear(x_fp16, right_proj_w)  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid of a linear) – FP16
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w)) # [B,N,N,hidden_dim]
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))    # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 5. Optional mask (broadcasted over hidden dim)
    # ------------------------------------------------------------------
    if nomask or mask_tensor is None:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 6. Fuse mask * gate * projection with a Triton kernel (in‑place)
    # ------------------------------------------------------------------
    total = left.numel()                # B * N * N * hidden_dim
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    # left = left * left_gate * mask
    mul_gate_mask_kernel[grid](
        left,          # out_ptr (in‑place)
        left,          # a_ptr
        left_gate,     # gate_ptr
        mask_fp16,     # mask_ptr
        total,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
    )
    # right = right * right_gate * mask
    mul_gate_mask_kernel[grid](
        right,
        right,
        right_gate,
        mask_fp16,
        total,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
    )

    # ------------------------------------------------------------------
    # 7. Cubic contraction:   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched matmul via torch.einsum (maps to cuBLAS)
    # ------------------------------------------------------------------
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)  # fp16

    # ------------------------------------------------------------------
    # 8. Post‑processing (LayerNorm on hidden dim, final gate, out projection)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out_norm = out_norm * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"].to(device)  # (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)           # [B, N, N, dim] (float32)

    return output
