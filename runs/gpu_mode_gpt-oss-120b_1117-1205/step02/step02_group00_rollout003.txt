def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. Input layernorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # Convert to half for subsequent ops
    x_h = x.to(torch.float16)

    # 2. Linear projections
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # 3. Gates and out_gate
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # 4. Gate+mask fusion using kernel
    if not nomask and mask is not None:
        # mask shape: [B, N, N] broadcast to hidden_dim
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        # Flatten everything
        left_flat = left.view(-1).contiguous()
        right_flat = right.view(-1).contiguous()
        left_gate_flat = left_gate.view(-1).contiguous()
        right_gate_flat = right_gate.view(-1).contiguous()
        mask_flat = mask_h.view(-1).contiguous()  # shape B*N*N
        # Apply kernel
        _apply_gate_and_mask(left_flat, right_flat, left_gate_flat, right_gate_flat,
                             mask_flat, hidden_dim)
        # Reshape
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask case: just multiply by gates
        left = left * left_gate
        right = right * right_gate

    # Clean up gates and x_h references
    del left_gate, right_gate, x_h

    # 5. Contraction via batched GEMM
    # Permute to B, hidden_dim, N, N
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # For right we need to transpose the last two dims (N,N) to enable dot product
    # Actually as earlier derived, we need right_transposed = right_perm.transpose(-2, -1)
    right_trans = right_perm.transpose(-2, -1).contiguous()
    B, H, N, _ = left_perm.shape
    # reshape to (B*H, N, N)
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_trans.view(B * H, N, N)
    # Perform batched GEMM
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N) product
    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 6. Output layernorm
    out = F.layer_norm(out, (hidden_dim,),
                       weight=weights["to_out_norm.weight"].to(torch.float16),
                       bias=weights["to_out_norm.bias"].to(torch.float16))
    # Multiply out_gate
    out = out * out_gate
    # Final linear projection to dim
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return float32
    return out.to(torch.float32)

@triton.jit
def _gate_mask_fuse_kernel(left_ptr, right_ptr,
                          left_gate_ptr, right_gate_ptr,
                          mask_ptr,
                          total_elements: tl.int32,
                          BLOCK: tl.constexpr,
                          HAS_MASK: tl.constexpr,
                          H: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # load
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0))

    if HAS_MASK:
        # compute mask index
        mask_index = offs // H
        mask_val = tl.load(mask_ptr + mask_index, mask=mask, other=tl.float16(0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

"""
TriMul outgoing update (AlphaFold3) – fast forward pass.

The core O(N³·Cₓ) contraction is reduced to a batched GEMM:
    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
which is computed as (left @ rightᵀ) for each hidden dimension d.
All linear/projected tensors are kept in float16 to exploit tensor‑cores.
A small Triton kernel fuses mask‑application (when present) together
with the left/right gating multiplications, guaranteeing at least one
custom kernel in the implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: fuse mask (optional) with left/right gates
# --------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # half*
    right_ptr,              # half*
    left_gate_ptr,          # half*
    right_gate_ptr,         # half*
    mask_ptr,               # half*  (broadcasted over hidden dim)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present else 0
    H: tl.constexpr,          # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0))

    if HAS_MASK:
        # One mask element corresponds to H consecutive hidden entries
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store results back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fuse ``left = left * left_gate * mask`` and
    ``right = right * right_gate * mask`` (if ``mask`` is not None).
    All tensors must be contiguous torch.float16.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    HAS_MASK = 1
    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask,
        total,
        BLOCK=BLOCK,
        HAS_MASK=HAS_MASK,
        H=hidden_dim,
    )


# --------------------------------------------------------------
# Entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) – may be ignored.
            - weights: dict of model parameters (float32 tensors).
            - config: dict containing ``dim`` and ``hidden_dim`` and ``nomask``.

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----------------------------------------------------------
    # 1. Input LayerNorm (float32 → half)
    # ----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ----------------------------------------------------------
    # 2. Projections (half‑precision)
    # ----------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ----------------------------------------------------------
    # 3. Gates (half‑precision)
    # ----------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ----------------------------------------------------------
    # 4. Fuse mask (if any) and gates via Triton
    # ----------------------------------------------------------
    if (not nomask) and (mask is not None):
        # mask: [B, N, N] → broadcast over hidden_dim
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        # Flatten everything for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask_h.contiguous().view(-1)  # [B*N*N]
        # In‑place fusion
        _apply_gate_and_mask(
            left_f, right_f, left_gate_f, right_gate_f, mask_f, hidden_dim
        )
        # Reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask – simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # Gates are no longer needed
    del left_gate, right_gate, x_h

    # ----------------------------------------------------------
    # 5. Core contraction: batched GEMM (left @ rightᵀ)
    # ----------------------------------------------------------
    # Permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2)               # (B, H, N, N) – view is fine
    right_trans = right_perm.transpose(-2, -1)         # (B, H, N, N) – transposed K dimension
    B, H, N, _ = left_perm.shape

    # Collapse batch * hidden_dim -> (B*H, N, N)
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_trans.view(B * H, N, N)

    # Batched matrix multiply (half → half, float32 accumulation)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)

    # Restore layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ----------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return float32 as required
    return out.to(torch.float32)
