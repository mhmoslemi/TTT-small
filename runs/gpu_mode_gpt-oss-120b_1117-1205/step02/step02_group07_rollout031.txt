```
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_k, stride_right_i, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    H_TOTAL: tl.constexpr, # hidden dim
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute batch and head index
    b = batch_id // H_TOTAL
    h = batch_id - b * H_TOTAL

    # offsets for i and j
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this batch+head
    left_base = left_ptr + b * stride_left_batch + h * stride_left_h
    right_base = right_ptr + b * stride_right_batch + h * stride_right_h
    out_base = out_ptr + b * stride_out_batch + h * stride_out_h

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        mask_k = offs_k < N

        # load left tile (i,k)
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load right tile (k,j) see right stride: k dim stride= stride_right_k, j dim stride = stride_right_i
        b_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)

    # store result (i,j)
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

"""
Triton‑accelerated outgoing TriMul (AlphaFold3) forward pass.

Algorithm:
1. Layer‑norm the 4‑D input (float32).
2. Project to hidden_dim with three fp16 linear layers (left/right and three gates).
3. Optional pairwise mask applied to left/right.
4. Apply element‑wise sigmoid gates.
5. Core contraction:
   out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   This is a batched GEMM (B·hidden_dim, N, N) @ (B·hidden_dim, N, N)^T.
   Implemented as a custom Triton kernel that works on the original
   (B,N,N,hidden) layout – no extra copies or transposes.
6. Output LayerNorm, out‑gate, and final linear projection back to dim
   (all float32).  The final tensor shape is (B, N, N, dim).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication for the TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # dimensions
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_k, stride_right_i, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    H_TOTAL: tl.constexpr                  # same as H, but constexpr for compile‑time
):
    # ---------------------------------------------------------------
    # program IDs
    # ---------------------------------------------------------------
    batch_flat = tl.program_id(0)   # combines batch and head dimension
    pid_m = tl.program_id(1)        # block row (i)
    pid_n = tl.program_id(2)        # block col (j)

    # ---------------------------------------------------------------
    # derive batch (b) and head (h) index from the flattened ID
    # ---------------------------------------------------------------
    b = batch_flat // H_TOTAL
    h = batch_flat - b * H_TOTAL  # equivalent to batch_flat % H_TOTAL

    # ---------------------------------------------------------------
    # offsets in the i and j dimensions for this block
    # ---------------------------------------------------------------
    off_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = off_i < N
    mask_j = off_j < N

    # ---------------------------------------------------------------
    # base pointers for the current (b, h) slice
    # ---------------------------------------------------------------
    left_base  = left_ptr  + b * stride_left_batch  + h * stride_left_h
    right_base = right_ptr + b * stride_right_batch + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_batch   + h * stride_out_h

    # ---------------------------------------------------------------
    # accumulator (fp32)
    # ---------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # iterate over the reduction dimension k
    # ---------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        off_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = off_k < N

        # left tile: (i, k)
        a_ptrs = left_base + off_i[:, None] * stride_left_i + off_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # right tile: (k, j)  – note the order of strides (k first, then j)
        b_ptrs = right_base + off_k[:, None] * stride_right_k + off_j[None, :] * stride_right_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # fp16×fp16 → fp32 accumulation using tensor cores
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # ---------------------------------------------------------------
    # write back the result (fp16)
    # ---------------------------------------------------------------
    out_ptrs = out_base + off_i[:, None] * stride_out_i + off_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right: (B, N, N, H)  fp16
    returns out: (B, N, N, H) fp16
    """
    B, N, N2, H = left.shape
    assert N == N2, "input must be square in the pairwise dimensions"

    out = torch.empty_like(left)

    # strides (in elements)
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # tuning parameters – work well for N up to 1024 on B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,                     # flattened (batch, head) dimension
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N)    # cols
    )
    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
        stride_right_batch, stride_right_k, stride_right_i, stride_right_h,
        stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        H_TOTAL=H,
        num_warps=8,
        num_stages=3,
    )
    return out


# ----------------------------------------------------------------------
# Entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or numeric)
            - weights_dict: mapping from parameter names to torch.Tensor
            - config_dict: must contain 'dim', 'hidden_dim', optionally 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim]  float32

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for the heavy linear paths
    # -----------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections (fp16)
    # -----------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # -----------------------------------------------------------------
    # 4. Apply optional pairwise mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)   # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # -----------------------------------------------------------------
    # 5. Gating (fp16) and element‑wise multiplication
    # -----------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6. Core contraction via the Triton kernel
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # -----------------------------------------------------------------
    out_fp16 = tri_mul_triton(left, right)          # (B,N,N,hidden_dim) fp16

    # -----------------------------------------------------------------
    # 7. Cast back to float32 and apply output LayerNorm, out‑gate,
    #    and final projection to the original channel dimension.
    # -----------------------------------------------------------------
    out = out_fp16.to(torch.float32)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
