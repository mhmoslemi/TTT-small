"""
TriMul outgoing kernel (AlphaFold3).

Algorithm:
1. Layer‑norm the input (float32) and cast to float16.
2. Linear projections → left/right (float16, shape [B,N,N,H]).
3. Compute left/right gates (float16, sigmoid).
4. Fuse gate multiplication (and optional mask) with a Triton kernel:
   left  = left  * left_gate  * mask
   right = right * right_gate * mask
5. Perform the core N³ contraction as a batched matrix multiply:
   out_{i,j,d} = Σ_k left_{i,k,d} * right_{j,k,d}
   implemented as (B*H) x (N×N) bmm with right transposed.
6. Apply output LayerNorm, multiply by out_gate, and a final linear
   projection back to the original dimension.
7. Return float32 tensor of shape [B, N, N, dim].

The Triton kernel only handles the element‑wise gate+mask fusion,
ensuring the solution contains a custom GPU kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # float16 *
    right_ptr,         # float16 *
    left_gate_ptr,     # float16 *
    right_gate_ptr,    # float16 *
    mask_ptr,          # float16 * (broadcasted)
    total_elements,   # int32, number of elements in left/right
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present, 0 otherwise
    H: tl.constexpr,          # hidden dimension (size of innermost channel)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # mask is broadcasted over the hidden dimension
        mask_idx = offs // H          # int division
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * m
        right = right * right_gate * m
    else:
        left = left * left_gate
        right = right * right_gate

    # Store results back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place element‑wise multiplication:
        left  = left  * left_gate  * (mask if present)
        right = right * right_gate * (mask if present)

    All tensors must be contiguous float16.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    # Dummy pointer if mask is unused (kernel will not read it)
    mask_ptr = mask if mask is not None else left

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # No return – tensors are modified in‑place


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) or None
            - weights: dict mapping parameter names to torch.Tensor
            - config: dict with keys ``dim``, ``hidden_dim``, ``nomask``

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w)           # [B,N,N,hidden_dim]
    right = F.linear(x_h, right_proj_w)         # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3. Gates (half, sigmoid)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_h, left_gate_w).sigmoid()
    right_gate = F.linear(x_h, right_gate_w).sigmoid()
    out_gate = F.linear(x_h, out_gate_w).sigmoid()   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) and gates via Triton (in‑place)
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1).contiguous()  # [B,N,N,1]
    else:
        mask_h = None

    _apply_gate_and_mask(left, right, left_gate, right_gate, mask_h, hidden_dim)

    # ------------------------------------------------------------------
    # 5. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as batched GEMM: (B*H, N, N) × (B*H, N, N)^T
    # ------------------------------------------------------------------
    # Rearrange tensors for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) = transpose on last two dims

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)  # (B*H, N, N)

    # Batched matrix multiplication (half precision, float32 accumulation)
    out_mat = torch.bmm(left_mat, right_mat)   # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate  # element‑wise gating

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)   # [B,N,N,dim]

    # Return in float32 as required
    return out.to(torch.float32)
