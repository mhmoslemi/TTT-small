def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # layernorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x = x.to(torch.float16)

    # linear projections
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_fp
        right = right * mask_fp

    # gates
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = (left * left_gate).contiguous()
    right = (right * right_gate).contiguous()
    out_gate = out_gate.contiguous()

    # core contraction via Triton
    B, N, _, H = left.shape
    # left, right, out_gate are [B,N,N,H]
    out = torch.empty_like(left, dtype=torch.float16)

    # compute strides
    stride_left_batch = left.stride(0)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)
    stride_left_h = left.stride(3)

    stride_right_batch = right.stride(0)
    stride_right_j = right.stride(1)  # for column j offset
    stride_right_k = right.stride(2)
    stride_right_h = right.stride(3)

    stride_out_batch = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_h = out.stride(3)

    total_batches = B * H
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (total_batches, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

Triton‑accelerated outgoing TriMul (AlphaFold3).

The heavy part of TriMul is the N×N matrix multiplication per hidden channel:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
We fuse this contraction into a single Triton kernel that treats (batch, hidden)
as a combined batch dimension and computes a batched GEMM (A @ Bᵀ) with
tiling (BLOCK_M, BLOCK_N, BLOCK_K).  All other operations (LayerNorm,
linear projections, gating) stay in PyTorch.
"""

import triton.language as tl

# Triton kernel: batched GEMM for the outgoing TriMul contraction.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # batch, seq_len, hidden_dim
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
    Arguments:
        left/right/out: [B, N, N, H] tensors (half‑precision)
        Strides correspond to the contiguous layout (batch, i, k, hidden).
    """
    # program IDs
    pid_batch = tl.program_id(0)   # combined (b, h)
    pid_i = tl.program_id(1)       # tile row index (i)
    pid_j = tl.program_id(2)       # tile column index (j)

    # decode combined batch/hidden index
    b = pid_batch // H
    h = pid_batch % H

    # offsets for the output tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for the current (b, h) slice
    left_base  = left_ptr  + b * stride_left_batch  + h * stride_left_h
    right_base = right_ptr + b * stride_right_batch + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_batch   + h * stride_out_h

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: [i, k]   -> stride_i for rows, stride_k for cols
        left_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right tile: [k, j]   -> stride_k for rows, stride_j for cols
        right_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # accumulate the dot product
        acc += tl.dot(left, right)

    # store the result tile (converted back to fp16)
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool or numeric)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)  # fp16 for the heavy path

    # --------------------------------------------------------------
    # 2. Linear projections (fp16, bias=False)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)  # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # --------------------------------------------------------------
    # 4. Gating (fp16)
    # --------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = (left * left_gate).contiguous()
    right = (right * right_gate).contiguous()
    out_gate = out_gate.contiguous()

    # --------------------------------------------------------------
    # 5. Core contraction: batched GEMM via Triton
    # --------------------------------------------------------------
    B, N, _, H = left.shape   # left.shape == (B, N, N, hidden_dim)
    assert H == hidden_dim

    # allocate output tensor (half‑precision)
    out = torch.empty_like(left, dtype=torch.float16)

    # strides of the tensors (contiguous layout: [B, N, N, H])
    stride_left_batch = left.stride(0)
    stride_left_i     = left.stride(1)   # stride for the second N (i)
    stride_left_k     = left.stride(2)   # stride for the third N (k)
    stride_left_h     = left.stride(3)   # stride for hidden dim

    stride_right_batch = right.stride(0)
    stride_right_j     = right.stride(1)   # stride for the second N (j)
    stride_right_k     = right.stride(2)   # stride for the third N (k)
    stride_right_h     = right.stride(3)

    stride_out_batch = out.stride(0)
    stride_out_i     = out.stride(1)
    stride_out_j     = out.stride(2)
    stride_out_h     = out.stride(3)

    total_batches = B * H
    BLOCK_M = 64   # tile size for i dimension
    BLOCK_N = 64   # tile size for j dimension
    BLOCK_K = 32   # tile size for reduction over k

    grid = (
        total_batches,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_left_batch,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_batch,
        stride_right_j,
        stride_right_k,
        stride_right_h,
        stride_out_batch,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Convert to float32 for the rest of the pipeline
    out = out.to(torch.float32)

    # --------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final projection (float32)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"])

    return out
