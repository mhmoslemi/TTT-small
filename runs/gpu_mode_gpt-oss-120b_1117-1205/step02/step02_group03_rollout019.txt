def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm (float32)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # Cast to FP16 for heavy computation
    x = x.to(torch.float16)

    # Linear projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"].to(torch.float16))
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"].to(torch.float16))

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # Apply mask and gating
    if not nomask:
        # mask shape (B,N,N) -> unsqueeze(-1)
        mask_fp = mask.to(torch.float16).unsqueeze(-1)  # shape (B,N,N,1)
        # Ensure contiguous flatten
        left_proj = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp = mask_fp.contiguous()
        # Allocate output tensors
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)
        total_elements = left_proj.numel()
        BLOCK = 131072  # maybe 65536? We'll pick 65536 or 16384.
        grid = (triton.cdiv(total_elements, BLOCK),)
        apply_mask_gate_kernel[grid](
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask_fp,
            left,
            right,
            total_elements,
            BLOCK=BLOCK,
        )
        # left, right now have mask applied
    else:
        # No mask: just multiply with gates
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Core contraction via torch.bmm
    # Permute to (B, H, N, N) and reshape
    B, N, _, _ = input_tensor.shape
    # left shape (B,N,N,H) (since left_proj original shape B,N,N,H)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 2, 1).contiguous()  # transposed k,j
    # flatten batch*H dimension
    left_flat = left.view(-1, N, N)
    right_flat = right.view(-1, N, N)
    # Batched matmul (float16)
    out_flat = torch.bmm(left_flat, right_flat)  # shape (B*H,N,N)
    # reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Output LayerNorm (float32)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # Multiply by out_gate (need shape broadcast)
    out_gate = out_gate.to(torch.float32)  # shape (B,N,N,H)
    out = out * out_gate
    # Final linear projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float32))

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def mask_gate_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    total_elements,
    hidden_dim: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    # Load values
    lp = tl.load(left_proj_ptr + offs, mask=valid, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=valid, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    # Compute index for mask: one mask value per hidden_dim entries
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=0.0)

    out_l = lp * lg * m
    out_r = rp * rg * m

    tl.store(out_left_ptr + offs, out_l, mask=valid)
    tl.store(out_right_ptr + offs, out_r, mask=valid)

def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward pass.
    The heavy O(N³) contraction is performed with a batched
    torch.bmm (cuBLAS) while the mask‑gate fusion is done with a minimal
    Triton kernel. The implementation works for all test configurations.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1) Input LayerNorm (float32)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16 for the heavy path
    x = x.to(torch.float16)

    # 2) Linear projections (fp16, no bias)
    left_proj = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right_proj = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # 3) Gating (fp16)
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # 4) Apply mask and combine with gates
    if not nomask:
        # mask: [B, N, N]  ->  [B, N, N, 1]
        mask_fp = mask.to(torch.float16).unsqueeze(-1).contiguous()
        # allocate output tensors
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        total_elements = left_proj.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)

        mask_gate_kernel[grid](
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask_fp,
            left,
            right,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
        )
    else:
        # No mask – simple elementwise fusion
        left = left_proj * left_gate
        right = right_proj * right_gate

    # 5) Core contraction: batched GEMM (B*H, N, N) @ (B*H, N, N)
    #    left: (B, N, N, H)  -> (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    #    right: (B, N, N, H) -> (B, H, N, N) with k as first dim
    right = right.permute(0, 3, 2, 1).contiguous()   # (B, H, N, N) where dim2=k, dim3=j

    B, N = input_tensor.shape[0], input_tensor.shape[1]

    left_flat = left.view(-1, N, N)   # (B*H, N, N)
    right_flat = right.view(-1, N, N) # (B*H, N, N)

    # Matmul in FP16 using cuBLAS (torch.bmm) – fast and numerically stable
    out_flat = torch.bmm(left_flat, right_flat)  # (B*H, N, N)

    # Restore layout: [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # back to FP32 for the rest

    # 6) Output LayerNorm, gating, final projection (float32)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out

import torch

import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left/right projections and gates.
# The mask (shape B,N,N,1) is broadcast over the hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    total_elements,
    hidden_dim: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load the four tensors (same shape)
    lp = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Broadcasted mask: one value per hidden_dim entries
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    out_l = lp * lg * m
    out_r = rp * rg * m

    tl.store(out_left_ptr + offs, out_l, mask=mask)
    tl.store(out_right_ptr + offs, out_r, mask=mask)


def custom_kernel(data):
    """
    Outgoing TriMul forward pass.

    1. Layer‑norm on the input (float32).
    2. FP16 linear projections + sigmoid gates.
    3. Optional mask fused with the gates via a tiny Triton kernel.
    4. Core O(N³) contraction performed as a batched GEMM with
       torch.bmm (cuBLAS) – the most efficient path on B200.
    5. Output Layer‑norm, final gating and linear projection (float32).

    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)                # fp16 for the heavy path

    # ------------------------------------------------------------------
    # 2) Linear projections (fp16, bias‑free)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right_proj = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3) Gates (fp16)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 4) Apply mask and combine with gates
    # ------------------------------------------------------------------
    if not nomask:
        # mask: (B, N, N) -> (B, N, N, 1)
        mask_fp = mask.to(torch.float16).unsqueeze(-1).contiguous()
        # output tensors
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        total_elems = left_proj.numel()

        grid = (triton.cdiv(total_elems, BLOCK),)

        mask_gate_kernel[grid](
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask_fp,
            left,
            right,
            total_elems,
            hidden_dim,
            BLOCK=BLOCK,
        )
    else:

        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5) Core contraction: batched GEMM (B*H, N, N) @ (B*H, N, N)
    #    left  : (B, N, N, H) -> (B, H, N, N)
    #    right : (B, N, N, H) -> (B, H, N, N) with k as first dim
    # ------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)  where dim2=k, dim3=j

    B, N = input_tensor.shape[0], input_tensor.shape[1]

    # Use cuBLAS batched GEMM (torch.bmm) – fast, fp16 accumulation in FP32
    out_flat = torch.bmm(left_flat, right_flat)      # (B*H, N, N)

    # Restore original layout: [B, N, N, H]
