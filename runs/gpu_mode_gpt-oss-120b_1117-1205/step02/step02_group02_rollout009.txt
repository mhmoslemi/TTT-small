```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute tile coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # load A tile
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0)

        # load B tile
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & mask_n[None, :], other=0.0)

        # dot product
        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

Now wrapper:

```
def triton_matmul(A: torch.Tensor, B: torch.Tensor):
    """
    Batched matrix multiplication C = A @ B, where
    A: (batch, M, K)
    B: (batch, K, N)
    Returns C: (batch, M, N)
    """
    assert A.is_contiguous() and B.is_contiguous()
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (batch, triton.cdiv(M, BLOCK_M), triton.cdiv(N, BLOCK_N))
    batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C
```

```
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Input LayerNorm (float32)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B,N,N,dim] float32

    # Convert to fp16 for heavy ops
    x_f16 = x.to(torch.float16)

    # Linear projections without bias
    left = F.linear(x_f16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_f16, weights["right_proj.weight"].to(torch.float16))

    # Mask if needed
    if not nomask:
        # mask shape [B,N,N] possibly bool
        mask_f16 = mask.unsqueeze(-1).to(torch.float16)  # broadcast
        left = left * mask_f16
        right = right * mask_f16

    # Gates
    left_gate = F.linear(x_f16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_f16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_f16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Permute to (B, H, N, N) to get matrices for each hidden dim
    # left: (B,N,N,H) -> (B,H,N,N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: (B,N,N,H) -> (B,H,N,N) with dimensions (j,k) turned into (k,j) for B_T
    # we need right indexed as (B,H,K,N) where K is sum dimension (k) and N is output column (j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B,H,N,N) where dims are (k,j)

    B_, H_, N_, _ = left_perm.shape
    left_flat = left_perm.view(B_ * H_, N_, N_)  # (batch*H, M=N, K=N)
    right_flat = right_perm.view(B_ * H_, N_, N_)  # (batch*H, K=N, N)

    # batched GEMM using Triton
    out_flat = triton_matmul(left_flat, right_flat)  # (batch*H, N, N) fp16

    # reshape back to (B,N,N,H)
    out = out_flat.view(B_, H_, N_, N_).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # Output LayerNorm over hidden_dim
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    # Apply out_gate (need same dtype)
    out = out * out_gate.to(torch.float32)

    # Final linear projection
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
```

"""
TriMul (outgoing) forward pass with a Triton‑accelerated batched GEMM.

The heavy contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix multiplication A @ Bᵀ for every hidden dimension ``d``.
We fuse this N³ operation into a single Triton kernel:
    * input tensors are cast to float16 for bandwidth,
    * accumulation is done in float32,
    * each (batch, hidden) pair is processed as an independent GEMM.
All other linear / LayerNorm / gating steps stay in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (C = A @ B) where
#   A : (batch, M, K)
#   B : (batch, K, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A tile (M × K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        # Load B tile (K × N)
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)    # fp16 * fp16 -> fp32 accumulate

    # Store C (M × N) in fp16
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def triton_batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM C = A @ B using Triton.
    A: (batch, M, K), B: (batch, K, N), both float16, contiguous.
    Returns C: (batch, M, N) float16.
    """
    assert A.is_contiguous() and B.is_contiguous()
    assert A.dtype == B.dtype == torch.float16
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tuned tile sizes for B200
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator (AlphaFold3 style).

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor of shape [B, N, N] (bool or numeric)
            - weights_dict: mapping from parameter name to torch.Tensor
            - config_dict: must contain 'dim' and 'hidden_dim';
                           optional 'nomask' (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim] float32

    # Cast to float16 for heavy linear/GEMM work
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(
        x_fp16,
        weights["left_proj.weight"].to(torch.float16)
    )  # [B, N, N, hidden_dim]
    right = F.linear(
        x_fp16,
        weights["right_proj.weight"].to(torch.float16)
    )  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [B, N, N, 1]
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4. Gating tensors (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_fp16,
        weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = F.linear(
        x_fp16,
        weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = F.linear(
        x_fp16,
        weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()                     # [B, N, N, hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as a batched GEMM (A @ Bᵀ) per hidden dimension.
    # ------------------------------------------------------------------
    # Rearrange tensors to (B, H, N, N) where the last two dims are (i,k) and (k,j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N) -> (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) -> (k,j)

    B_, H_, N_, _ = left_perm.shape
    # Collapse batch and hidden dimensions for the kernel
    left_flat = left_perm.view(B_ * H_, N_, N_)   # (B*H, M=N, K=N)
    right_flat = right_perm.view(B_ * H_, N_, N_)  # (B*H, K=N, N=N)

    # Perform the batched matmul on float16 inputs, float32 accumulation
    out_flat = triton_batched_matmul(left_flat, right_flat)  # (B*H, N, N) fp16

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B_, H_, N_, N_).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
