def custom_kernel(data):
    input_, mask_, weights, config = data

    device = input_.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Layernorm on input
    x = F.layer_norm(
        input_,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Convert to half for speed
    x_h = x.to(torch.float16)

    # Combine linear weights for left_proj, right_proj, left_gate, right_gate, out_gate
    # All weights are stored as (out_features, in_features) and bias=False.
    # Ensure they are half and contiguous.
    w_left_proj = weights["left_proj.weight"].to(torch.float16)
    w_right_proj = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    combined_weight = torch.cat([w_left_proj, w_right_proj, w_left_gate, w_right_gate, w_out_gate], dim=0) # shape (5*H, dim)

    # Flatten input for linear
    B, N, _, _ = input_.shape
    # x_h shape (B,N,N,dim)
    x_flat = x_h.view(-1, dim)  # shape (B*N*N, dim)

    # Perform combined linear
    combined_out = torch.nn.functional.linear(x_flat, combined_weight)  # (B*N*N, 5*H)
    # Reshape back to (B,N,N,5*H)
    combined_out = combined_out.view(B, N, N, -1)

    # Split
    left_proj = combined_out[..., :hidden_dim]
    right_proj = combined_out[..., hidden_dim:2*hidden_dim]
    left_gate = combined_out[..., 2*hidden_dim:3*hidden_dim]
    right_gate = combined_out[..., 3*hidden_dim:4*hidden_dim]
    out_gate = combined_out[..., 4*hidden_dim:5*hidden_dim]

    # Convert to half, already half

    # Apply mask + gating (and sigmoid out_gate) via Triton kernel
    if not nomask:
        mask = mask_.unsqueeze(-1).to(torch.float16).contiguous()
    else:
        mask = None

    # flatten all
    left_flat = left_proj.contiguous().view(-1)
    right_flat = right_proj.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)
    out_gate_flat = out_gate.contiguous().view(-1)
    if mask is not None:
        mask_flat = mask.contiguous().view(-1)  # shape (B,N,N)
    else:
        mask_flat = None

    total = left_flat.numel()
    BLOCK = 1024*4  # 4096 maybe; larger block better for memory coalescing.
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask_flat is not None else 0
    mask_ptr = mask_flat if mask_flat is not None else left_flat  # dummy

    _gate_mask_fuse_kernel[grid](
        left_flat, right_flat,
        left_gate_flat, right_gate_flat,
        out_gate_flat,
        mask_ptr,
        total,
        hidden_dim,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
    )
    # Reshape after kernel
    left = left_flat.view_as(left_proj)
    right = right_flat.view_as(right_proj)
    out_gate = out_gate_flat.view_as(out_gate)

    # Now contraction
    # left: (B,N,N,hidden_dim), right: (B,N,N,hidden_dim)
    # Rearrange for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) transposed
    B_, H_, N_, _ = left_perm.shape
    left_mat = left_perm.view(B_ * H_, N_, N_)
    right_mat = right_perm.view(B_ * H_, N_, N_)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H,N,N)

    # Reshape back
    out = out_mat.view(B_, H_, N_, N_).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # Output layernorm and out gate and final linear
    # to_out_norm: LayerNorm over hidden_dim
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    # out_gate is already sigmoid
    out = out * out_gate  # elementwise
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16))

    # Convert back to float32
    return out.to(torch.float32)

"""
Highly optimized outgoing TriMul forward pass.

The computation consists of:
  1) LayerNorm on the input.
  2) A single fused linear projection that produces the two projections
     (left/right) and the three gate vectors (left/right/out) in one pass.
  3) A Triton kernel that
       * applies sigmoid to the three gate tensors,
       * multiplies left/right by their respective gates,
       * optionally masks left/right,
       * writes the sigmoid’ed out‑gate back.
  4) Batched matrix‑multiply (left @ rightᵀ) performed with torch.bmm.
  5) Output LayerNorm, out‑gate multiplication and final linear projection.
All heavy tensors are kept in fp16; only the final result is returned in fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, gate sigmoids and element‑wise multiplications
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # fp16 *, [total]
    right_ptr,         # fp16 *, [total]
    left_gate_ptr,     # fp16 *, [total]
    right_gate_ptr,    # fp16 *, [total]
    out_gate_ptr,      # fp16 *, [total]
    mask_ptr,          # fp16 *, [mask_total] (broadcasted)
    total_elements: tl.int64,
    hidden_dim: tl.int64,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present else 0
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elements

    # load tensors
    left      = tl.load(left_ptr      + offs, mask=m, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=m, other=0.0)
    lgate     = tl.load(left_gate_ptr + offs, mask=m, other=0.0)
    rgate     = tl.load(right_gate_ptr+ offs, mask=m, other=0.0)
    out_gate  = tl.load(out_gate_ptr  + offs, mask=m, other=0.0)

    # sigmoid in float32 for better accuracy
    lgate_f = lgate.to(tl.float32)
    rgate_f = rgate.to(tl.float32)
    out_gate_f = out_gate.to(tl.float32)

    lgate_sig_f = 1.0 / (1.0 + tl.exp(-lgate_f))
    rgate_sig_f = 1.0 / (1.0 + tl.exp(-rgate_f))
    out_gate_sig_f = 1.0 / (1.0 + tl.exp(-out_gate_f))

    lgate_sig = lgate_sig_f.to(tl.float16)
    rgate_sig = rgate_sig_f.to(tl.float16)
    out_gate_sig = out_gate_sig_f.to(tl.float16)

    if HAS_MASK:
        # mask is broadcasted over the hidden dimension
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=m, other=1.0)
        left  = left  * lgate_sig * mask_val
        right = right * rgate_sig * mask_val
    else:
        left  = left  * lgate_sig
        right = right * rgate_sig

    # store the results back in‑place
    tl.store(left_ptr     + offs, left,    mask=m)
    tl.store(right_ptr    + offs, right,   mask=m)
    tl.store(out_gate_ptr + offs, out_gate_sig, mask=m)


# ----------------------------------------------------------------------
# Helper to launch the fused kernel
# ----------------------------------------------------------------------
def _apply_gate_mask_and_sigmoid(
    left: torch.Tensor,
    right: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    out_gate: torch.Tensor,
    mask: torch.Tensor | None,
    hidden_dim: int,
):
    """
    In‑place: left ← left * sigmoid(left_gate) * mask
             right← right* sigmoid(right_gate) * mask
             out_gate ← sigmoid(out_gate)
    All tensors must be contiguous fp16.
    """
    total = left.numel()
    BLOCK = 8192                     # enough threads per program
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when no mask

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        out_gate,
        mask_ptr,
        total,
        hidden_dim,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
    )


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            input_tensor: torch.Tensor [B, N, N, dim] (float32)
            mask: torch.Tensor [B, N, N] (bool or numeric) or None
            weights: dict with the following keys (all tensors are float32):
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            config: dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> fp16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)   # fp16 for the rest

    # ------------------------------------------------------------------
    # 2. Fused linear that produces left/right projections and all gates
    # ------------------------------------------------------------------
    # gather all weight matrices and cast to fp16 once
    w_left_proj = weights["left_proj.weight"].to(torch.float16)
    w_right_proj = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    combined_weight = torch.cat(
        [w_left_proj, w_right_proj, w_left_gate, w_right_gate, w_out_gate],
        dim=0,
    )  # shape (5*hidden_dim, dim)

    B, N, _, _ = input_tensor.shape
    x_flat = x.view(-1, dim)                 # (B*N*N, dim)
    combined_out = F.linear(x_flat, combined_weight)   # (B*N*N, 5*hidden_dim)
    combined_out = combined_out.view(B, N, N, -1)

    # split the concatenated tensor
    left_proj = combined_out[..., :hidden_dim]
    right_proj = combined_out[..., hidden_dim : 2 * hidden_dim]
    left_gate = combined_out[..., 2 * hidden_dim : 3 * hidden_dim]
    right_gate = combined_out[..., 3 * hidden_dim : 4 * hidden_dim]
    out_gate = combined_out[..., 4 * hidden_dim : 5 * hidden_dim]

    # ------------------------------------------------------------------
    # 3. Fuse mask (if any) and sigmoid gates via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # mask broadcasted over the hidden dimension
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
    else:
        mask_h = None

    # flatten for the kernel (keep everything contiguous)
    left_flat = left_proj.contiguous().view(-1)
    right_flat = right_proj.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)
    out_gate_flat = out_gate.contiguous().view(-1)
    mask_flat = mask_h.contiguous().view(-1) if mask_h is not None else None

    _apply_gate_mask_and_sigmoid(
        left_flat,
        right_flat,
        left_gate_flat,
        right_gate_flat,
        out_gate_flat,
        mask_flat,
        hidden_dim,
    )

    # reshape back
    left = left_flat.view_as(left_proj)    # (B,N,N,hidden_dim) – masked and gated
    right = right_flat.view_as(right_proj)
    out_gate = out_gate_flat.view_as(out_gate)

    # ------------------------------------------------------------------
    # 4. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as batched GEMM (TensorCore fp16)
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) = rightᵀ
    B_, H_, N_, _ = left_perm.shape
    left_mat = left_perm.view(B_ * H_, N_, N_)
    right_mat = right_perm.view(B_ * H_, N_, N_)
    out_mat = torch.bmm(left_mat, right_mat)            # (B*H, N, N)

    out = (
        out_mat.view(B_, H_, N_, N_)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, out‑gate multiplication and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate                     # out‑gate is already sigmoid‑ed
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return in float32 as required
    return out.to(torch.float32)
