"""
TriMul (outgoing) forward pass with a fused Triton element‑wise kernel.
The heavy N³ contraction is performed with a highly optimized batched
matrix multiplication via torch.bmm (cuBLAS), while a tiny Triton kernel
applies the optional mask together with the left/right gates in fp16.
All other operations (LayerNorm, Linear, Sigmoid) stay in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _apply_mask_gate_kernel(
    a_ptr,                # left or right projection (fp16)
    gate_ptr,             # corresponding gate (fp16)
    mask_ptr,             # mask (fp16, shape B,N,N)
    out_ptr,              # output buffer (fp16)
    hidden_dim,           # hidden dimension (int)
    total_elems,          # total number of elements in a/gate/out (int)
    BLOCK: tl.constexpr,  # elements per program
):
    """out = a * gate * mask  (mask broadcast over hidden dim)"""
    idx = tl.program_id(0) * BLOCK + tl.arange(0, BLOCK)
    mask = idx < total_elems

    # Load a and gate (contiguous memory => 1‑D stride)
    a = tl.load(a_ptr + idx, mask=mask, other=0.0)
    g = tl.load(gate_ptr + idx, mask=mask, other=0.0)

    # Compute mask index: broadcast over hidden_dim (last dimension)
    mask_idx = idx // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = a * g * m
    tl.store(out_ptr + idx, out, mask=mask)


def _apply_mask_gate(a, gate, mask, hidden_dim):
    """Convenience wrapper for the Triton element‑wise kernel."""
    out = torch.empty_like(a)
    total = a.numel()
    BLOCK = 65536  # tuneable; keep within 2^16 for safety
    grid = (triton.cdiv(total, BLOCK),)
    _apply_mask_gate_kernel[grid](
        a, gate, mask, out,
        hidden_dim,
        total,
        BLOCK=BLOCK,
    )
    return out


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) – may be ignored
            - weights_dict: mapping from parameter name to torch.Tensor
            - config_dict: must contain keys 'dim' and 'hidden_dim'.
              Optional key 'nomask' (bool) indicates that mask should be ignored.

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) and cast to fp16 for speed
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)  # now fp16

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – fp16
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right_proj = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid) – fp16
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 4. Apply mask and gates (fused Triton kernel if mask is used)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> fp16, contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_mask_gate(left_proj, left_gate, mask_fp16, hidden_dim)
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16, hidden_dim)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: batched matrix multiplication
    #    out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    #    This is a batched GEMM A @ B where A = left.permute(0,3,1,2)
    #    and B = right.permute(0,3,2,1) (already transposed on the last two dims).
    # ------------------------------------------------------------------
    # Bring hidden dim to batch dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    A = left_perm.view(B * H, N, N)        # (B*H, N, N), fp16
    Bmat = right_perm.view(B * H, N, N)    # (B*H, N, N), fp16

    # Batched matmul using cuBLAS (torch.bmm) – fp16 inputs, fp32 accumulation internally
    out_flat = torch.bmm(A, Bmat)          # (B*H, N, N), fp16

    # Reshape back to (B, N, N, H) and cast to float32 for the remaining ops
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
