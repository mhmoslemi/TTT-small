"""
Implementation of the outgoing TriMul (Triangular Multiplicative Update) forward
pass using a Triton kernel for the heavy N³ contraction.

The kernel computes a batched matrix multiplication A @ Bᵀ for each hidden
dimension:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
where left and right are the projected and gated pairwise tensors.
All other operations (LayerNorms, linear projections, gating) are performed
in PyTorch. The contraction is performed in float16 with float32 accumulation
and the final result is converted back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for tensors of shape
# (batch, M, K) and (batch, K, N)  (all half‑precision)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets within the M and N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the beginning of the batch
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m) + (offs_k[None, :] * stride_a_k)
        b_ptrs = b_batch + (offs_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)

        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0)

        acc += tl.dot(a, b)   # fp16 * fp16 -> fp32 accumulation

    # Store result
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def triton_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM C = A @ Bᵀ where
        A shape = (batch, M, K)
        B shape = (batch, K, N)
    All tensors must be half‑precision and contiguous.
    Returns C with shape (batch, M, N) dtype float16.
    """
    assert A.dtype == B.dtype == torch.float16
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tile sizes – tuned for A200 series; adjust if needed
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------

            - mask_tensor: torch.Tensor [B, N, N] (bool or numeric)
            - weights_dict: mapping from parameter name to torch.Tensor (float32)
            - config_dict: must contain 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim], float32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for heavy ops
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (fp16, bias=False)
    # ------------------------------------------------------------------
    left = F.linear(
        x_fp16,
        weights["left_proj.weight"].to(torch.float16),
    )  # [B, N, N, hidden_dim]
    right = F.linear(
        x_fp16,
        weights["right_proj.weight"].to(torch.float16),
    )  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------

        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------

        x_fp16,
        weights["left_gate.weight"].to(torch.float16),
    ).sigmoid()
    right_gate = F.linear(
        x_fp16,
        weights["right_gate.weight"].to(torch.float16),
    ).sigmoid()
    out_gate = F.linear(
        x_fp16,
        weights["out_gate.weight"].to(torch.float16),
    ).sigmoid()  # [B, N, N, hidden_dim]

    right = right * right_gate

    # 6. Core contraction: batched GEMM via Triton
    #    Compute out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    # Bring the hidden dimension up front for easy flattening
    # left: (B, N, N, H) -> (B, H, N, N) where axes = (b, d, i, k)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: (B, N, N, H) -> (B, H, N, N) where axes = (b, d, k, j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    B_, H_, N_, _ = left_perm.shape  # B_, H_, N_, N_

    # Flatten batch and hidden dimensions: (B*H, N, N)
    left_flat = left_perm.view(B_ * H_, N_, N_)   # (batch, M=N, K=N)
    right_flat = right_perm.view(B_ * H_, N_, N_) # (batch, K=N, N=N)

    # Perform the batched matmul in fp16 (result still fp16)
    out_flat = triton_matmul(left_flat, right_flat)  # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B_, H_, N_, N_).permute(0, 2, 3, 1).contiguous()
    # Convert to float32 for the remaining layers
    out = out.to(torch.float32)

    # ------------------------------------------------------------------

        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out

"""
Implementation of the outgoing TriMul (Triangular Multiplicative Update) forward
pass using a Triton kernel for the heavy N³ contraction.

The kernel computes a batched matrix multiplication A @ Bᵀ for each hidden
dimension:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
where left and right are the projected and gated pairwise tensors.
All other operations (LayerNorms, linear projections, gating) are performed
in PyTorch. The contraction is performed in float16 with float32 accumulation
and the final result is converted back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for tensors of shape
# (batch, M, K) and (batch, K, N)  (all half‑precision)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets within the M and N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for the current batch
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m) + (offs_k[None, :] * stride_a_k)
        b_ptrs = b_batch + (offs_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)

        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0)

        acc += tl.dot(a, b)   # fp16 * fp16 -> fp32 accumulation

    # Store the result
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def triton_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM C = A @ Bᵀ where
        A shape = (batch, M, K)
        B shape = (batch, K, N)
    All tensors must be half‑precision and contiguous.
    Returns C with shape (batch, M, N) dtype float16.
    """
    assert A.dtype == B.dtype == torch.float16
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tile sizes – tuned for Ampere GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (bool or numeric)
            - weights_dict: mapping from parameter name to torch.Tensor (float32)
            - config_dict: must contain 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim], float32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the heavy operations
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (fp16, bias=False)
    # ------------------------------------------------------------------
    left = F.linear(
        x_fp16,
        weights["left_proj.weight"].to(torch.float16),
    )  # [B, N, N, hidden_dim]
    right = F.linear(
        x_fp16,
        weights["right_proj.weight"].to(torch.float16),
    )  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 5. Gating (fp16)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_fp16,
        weights["left_gate.weight"].to(torch.float16),
    ).sigmoid()
    right_gate = F.linear(
        x_fp16,
        weights["right_gate.weight"].to(torch.float16),
    ).sigmoid()
    out_gate = F.linear(
        x_fp16,
        weights["out_gate.weight"].to(torch.float16),
    ).sigmoid()  # [B, N, N, hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Core contraction: batched GEMM via Triton
    #    Compute out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    # Bring the hidden dimension to the second axis for easy flattening
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N) -> (b,d,i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) -> (b,d,k,j)

    B_, H_, N_, _ = left_perm.shape  # B_, H_, N_, N_

    # Flatten batch and hidden dimensions: (B*H, N, N)
    left_flat = left_perm.view(B_ * H_, N_, N_)   # (batch, M=N, K=N)
    right_flat = right_perm.view(B_ * H_, N_, N_) # (batch, K=N, N)

    # Perform the batched matmul in fp16 (result still fp16)
    out_flat = triton_matmul(left_flat, right_flat)  # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B_, H_, N_, N_).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # convert for subsequent ops

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
