"""
TriMul (outgoing) forward pass with Triton‑accelerated elementwise mask‑gate fusion.

Algorithm
---------
1. Layer‑norm the input (float32).
2. Cast to FP16 and apply three linear projections:
   * left_proj, right_proj   → hidden_dim
   * left_gate, right_gate, out_gate → hidden_dim (followed by sigmoid)
3. Fuse elementwise ``tensor * gate * mask`` for the left and right branches
   in a single Triton kernel (mask broadcast over the hidden dimension).
4. Perform the cubic contraction
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   as a batched matrix multiplication (torch.bmm) after reshaping
   each hidden slice to a separate batch.
5. Apply a second LayerNorm on the hidden dimension, multiply by out_gate,
   and project back to the original dim.

The kernel only handles the elementwise fusion; all linear / norm work uses
high‑performance PyTorch (cuBLAS / ATen). The final result is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def mul_gate_mask_kernel(
    out_ptr,                # pointer to output (flattened)
    a_ptr,                  # pointer to tensor (flattened)
    gate_ptr,               # pointer to gate (flattened)
    mask_ptr,               # pointer to mask (flattened, broadcast over H)
    total,                  # total number of elements (int)
    H: tl.constexpr,        # hidden dimension (size of last axis)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    a = tl.load(a_ptr + offs, mask=mask)
    g = tl.load(gate_ptr + offs, mask=mask)

    # Broadcast mask over hidden dimension
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    tl.store(out_ptr + offs, a * g * m, mask=mask)


def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            mask_tensor : torch.Tensor of shape [B, N, N] (float32) or None
            weights_dict: dict with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            config_dict : {"dim": int, "hidden_dim": int, "nomask": bool (optional)}
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # [B,N,N,dim]

    # --------------------------------------------------------------------
    # 2. Cast to FP16 for heavy work
    # --------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------------
    # 3. Linear projections (no bias)
    # --------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # [B,N,N,hidden_dim]
    right = F.linear(x_fp16, right_proj_w)  # [B,N,N,hidden_dim]

    # --------------------------------------------------------------------
    # 4. Gating + optional mask (fused via Triton)
    # --------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w)) # [B,N,N,hidden_dim]
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))    # [B,N,N,hidden_dim]

    # Prepare mask: all ones if no mask is required
    if mask_tensor is None or nomask:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(device).to(torch.float16)

    # Flatten tensors for kernel launch
    left_flat = left.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_flat = right.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)
    mask_flat = mask_fp16.contiguous().view(-1)

    # Allocate outputs for gated & masked tensors
    left_out = torch.empty_like(left_flat)
    right_out = torch.empty_like(right_flat)

    # Triton launch parameters
    BLOCK = 1024
    total = left_flat.numel()
    grid = (triton.cdiv(total, BLOCK),)

    # Launch kernel for left branch
    mul_gate_mask_kernel[grid](
        left_out,
        left_flat,
        left_gate_flat,
        mask_flat,
        total,
        H=hidden_dim,
        BLOCK_SIZE=BLOCK,
    )
    # Launch kernel for right branch
    mul_gate_mask_kernel[grid](
        right_out,
        right_flat,
        right_gate_flat,
        mask_flat,
        total,
        H=hidden_dim,
        BLOCK_SIZE=BLOCK,
    )

    # Reshape back to [B,N,N,hidden_dim]
    left = left_out.view(B, N, N, hidden_dim)
    right = right_out.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------------
    # 5. Cubic contraction via batched GEMM (torch.bmm)
    # --------------------------------------------------------------------
    # Transform to (B, hidden_dim, N, N) where the last two dims are matrices
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden_dim, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, hidden_dim, N, N)

    LH = B * hidden_dim
    left_mat = left_perm.view(LH, N, N)   # (B*hidden_dim, N, N)
    right_mat = right_perm.view(LH, N, N) # (B*hidden_dim, N, N)

    # Batched matrix multiply (fp16)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*hidden_dim, N, N)

    # Restore original layout [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------------
    # 6. Post‑processing (LayerNorm on hidden dim, final gate, output proj)
    # --------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float32)
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply final gate (out_gate was computed in fp16)
    out_norm = out_norm * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float32)  # (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)  # [B, N, N, dim], float32

    return output
