def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm
    x = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x_fp16 = x.half()

    # Linear projections
    left = F.linear(x_fp16, weights["left_proj.weight"].half())
    right = F.linear(x_fp16, weights["right_proj.weight"].half())

    if not nomask:
        mask_fp16 = mask.to(x.dtype).unsqueeze(-1)  # shape (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16

    # Gating
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].half()).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].half()).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].half()).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Core contraction
    # Permute
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B*H, N, N)
    right_flat = right_perm.view(B*H, N, N)

    out_flat = triton_batched_matmul(left_flat, right_flat)  # half

    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    # shape (B,N,N,H)

    # Convert to float32
    out = out.to(torch.float32)

    # Output LayerNorm
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    out = out * out_gate.to(out.dtype)

    out = F.linear(out, weights["to_out.weight"])

    return out
```

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    # Write output
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

def triton_batched_matmul(A, B):
    assert A.is_cuda and B.is_cuda
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, Kb, N = B.shape
    assert K == Kb
    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # strides
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (batch, triton.cdiv(M, BLOCK_M), triton.cdiv(N, BLOCK_N))
    batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

"""
TriMul (outgoing) forward pass with a fused Triton batched matrix multiplication.

Algorithm:
1. LayerNorm on the input tensor (float32).
2. Cast to float16 and apply two linear projections (left/right) without bias.
3. Optionally apply a mask (broadcast over the hidden dimension).
4. Compute three gating vectors (left_gate, right_gate, out_gate) via linear
   projections + sigmoid, and apply them to left/right.
5. Core contraction: for each head d, compute
   out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d].
   This is a batched matrix multiplication left @ right.T, implemented as a
   Triton kernel that processes (batch * hidden_dim) independent GEMMs of
   size N×N with FP16 inputs and FP32 accumulation.
6. LayerNorm on the hidden dimension, multiply by out_gate, and linearly project
   back to the original dimension.
All heavy arithmetic is performed in FP16; final output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton batched GEMM kernel (A @ B) for the core contraction
# --------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute C = A @ B for a batch of matrices.
    A: (batch, M, K)  FP16
    B: (batch, K, N)  FP16
    C: (batch, M, N)  FP16 (stored as FP16 after FP32 accumulation)
    """
    batch_id = tl.program_id(0)   # which (batch * hidden) pair
    pid_m = tl.program_id(1)      # tile row
    pid_n = tl.program_id(2)      # tile column

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Main loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        # Load B tile (K x N)
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        # Block-wise dot product (FP16 inputs, FP32 accumulation)
        acc += tl.dot(a, b)

    # Write back result (cast back to FP16)
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def triton_batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B where
        A : (batch, M, K) FP16
        B : (batch, K, N) FP16
    Returns
        C : (batch, M, N) FP16
    """
    assert A.is_cuda and B.is_cuda, "Tensors must be on CUDA"
    assert A.dtype == B.dtype == torch.float16, "Tensors must be float16"
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, "Inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tile sizes (tuned for FP16 tensor cores on B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# --------------------------------------------------------------
# Entry point expected by the evaluator
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operation.

    Arguments:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) – may be ignored
            - weights: dict of model parameters (float32 tensors)
            - config: dict containing 'dim', 'hidden_dim', and optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    # Configuration
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the channel dimension (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to half for heavy arithmetic
    x_fp16 = x.half()

    # ------------------------------------------------------------------
    # 2. Linear projections (left/right) – no bias, half precision
    # ------------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].half())
    right = F.linear(x_fp16, weights["right_proj.weight"].half())

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [B, N, N, 1] for broadcasting
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4. Gating values (sigmoid) and application
    # ------------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].half()).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].half()).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].half()).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction using a Triton batched GEMM
    #    Compute out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]
    # ------------------------------------------------------------------
    # Permute to (B, hidden, N, N) where right is transposed on the two
    # sequence axes so that a regular matrix multiplication suffices.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) with swapped seq axes

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Batched matrix multiplication (FP16 inputs, FP32 accumulation)
    out_flat = triton_batched_matmul(left_flat, right_flat)  # (B*H, N, N) FP16

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)   # Convert back to FP32 for the final layers

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate multiplication, and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(out.dtype)
    out = F.linear(out, weights["to_out.weight"])  # (dim, hidden_dim) weight, no bias

    return out
