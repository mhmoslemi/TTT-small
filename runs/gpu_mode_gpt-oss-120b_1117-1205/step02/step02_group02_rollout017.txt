`@triton.jit
def batched_matmul_kernel(a_ptr, b_ptr, c_ptr, M, N, K,
                           stride_a_batch, stride_a_m, stride_a_k,
                           stride_b_batch, stride_b_k, stride_b_n,
                           stride_c_batch, stride_c_m, stride_c_n,
                           BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):`

Inside, get program IDs:

`pid_batch = tl.program_id(0) # The batch index (b*h)
 pid_m = tl.program_id(1) # tile row
 pid_n = tl.program_id(2) # tile col`

Compute tile offsets for m and n: `offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)`
`offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)`

Set m mask, n mask: `mask_m = offs_m < M` ; `mask_n = offs_n < N`.

Base pointers:

`a_base = a_ptr + pid_batch * stride_a_batch`
`b_base = b_ptr + pid_batch * stride_b_batch`
`c_base = c_ptr + pid_batch * stride_c_batch`

Define accumulator: `acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)`

Iterate over K dimension:

`for k in range(0, tl.cdiv(K, BLOCK_K)):
    k_offset = k * BLOCK_K
    offs_k = k_offset + tl.arange(0, BLOCK_K)
    mask_k = offs_k < K`

    # Load A tile: shape (BLOCK_M, BLOCK_K)
    a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
    a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

    # Load B tile: shape (BLOCK_K, BLOCK_N)
    b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
    b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

    acc += tl.dot(a, b)`

After loop, store:

`c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n`

`tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])`

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel for batched matrix multiplication (float16 input, float32 accumulation)
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0)

        # Load B tile (K x N)
        b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)  # fp16*fp16 -> fp32

    # Store result
    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def triton_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B   (A: (B,M,K), B: (B,K,N))
    All tensors must be torch.float16 and contiguous.
    Returns C with shape (B,M,N) float16.
    """
    assert A.dtype == B.dtype == torch.float16
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tune tile sizes (generally good for Ampere)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.
    Implements the heavy N³ contraction with a fused batched GEMM kernel.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. Input LayerNorm (float32)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # shape [B, N, N, dim], dtype float32

    # 2. Linear projections (float32, bias=False)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # 3. Optional mask (float32)
    if not nomask:
        mask_f = mask.unsqueeze(-1).to(x.dtype)  # shape [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # 4. Gating tensors (float32)
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # 5. Core contraction via Triton (convert to fp16 for speed)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Permute to (B, H, i, k) and (B, H, k, j) respectively
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_fp16.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    B_, H_, N_, _ = left_perm.shape
    left_flat = left_perm.view(B_ * H_, N_, N_)
    right_flat = right_perm.view(B_ * H_, N_, N_)

    out_flat = triton_matmul(left_flat, right_flat)  # (B*H, N, N), fp16

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B_, H_, N_, N_).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # back to fp32 for the rest

    # 6. Output layer norm, gating, final linear
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate  # out_gate is float32
    out = F.linear(out, weights["to_out.weight"])
    return out
```

"""
Outgoing TriMul forward pass (AlphaFold3).

The heavy N³ contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is rewritten as a batched GEMM:
    for each (b,d) pair  A = left[b,:, :, d] (N×N)
                        B = right[b,:, :, d] (N×N)
    out[b,:, :, d] = A @ Bᵀ
We fuse this with a custom Triton kernel that performs a batched
matrix multiply on FP16 inputs with FP32 accumulation (Tensor‑Core
friendly). All remaining operations (LayerNorms, linear projections,
gates) are carried out with PyTorch in FP32 and only the contraction
is off‑loaded to Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (C = A @ B) for tensors
#   A : (batch, M, K)
#   B : (batch, K, N)
#   C : (batch, M, N)
# All tensors are FP16, accumulation is FP32.
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)          # batch index (B*H)
    pid_m = tl.program_id(1)              # tile row
    pid_n = tl.program_id(2)              # tile col

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K)[None, :],
                    other=0.0)

        # B tile: (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # fp16 * fp16 → fp32

    # write back
    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def triton_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM C = A @ B   (A: (B,M,K), B: (B,K,N))
    All tensors must be torch.float16 and contiguous.
    Returns C with shape (B,M,N) as torch.float16.
    """
    assert A.dtype == B.dtype == torch.float16
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tile sizes (tuned for Ampere B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.
    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor:  torch.Tensor [B, N, N] (bool or numeric)
            - weights_dict: mapping from parameter name to torch.Tensor
            - config_dict: must contain 'dim' and 'hidden_dim', optional 'nomask'
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim], dtype float32

    # ------------------------------------------------------------------
    # 2. Linear projections (float32, bias=False)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (float32)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.unsqueeze(-1).to(x.dtype)          # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating tensors (float32)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()   # [B,N,N,hidden]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton (FP16 for speed)
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # (B, H, i, k) and (B, H, k, j)
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()
    right_perm = right_fp16.permute(0, 3, 2, 1).contiguous()

    B_, H_, N_, _ = left_perm.shape
    left_flat = left_perm.view(B_ * H_, N_, N_)    # (B*H, M=N, K=N)
    right_flat = right_perm.view(B_ * H_, N_, N_)  # (B*H, K=N, N=N)

    out_flat = triton_matmul(left_flat, right_flat)   # (B*H, N, N), fp16

    # reshape back to [B, N, N, H]
    out = out_flat.view(B_, H_, N_, N_).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)                     # back to fp32

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
