"""
TriMul (outgoing) forward pass with a fused Triton batched GEMM.

Algorithm
---------
1. Layer‑norm over the channel dimension of the input (float32).
2. Two linear projections (left/right) and three gating projections are
   performed in fp16 for speed.
3. An optional mask is applied (broadcast on the hidden dimension).
4. The core contraction
       out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   is a batched matrix multiplication for each head:
       left  : (B, H, N, N)  with layout (i,k)
       right : (B, H, N, N)  with layout (k,j)  (already transposed)
   The multiplication is executed by a custom Triton kernel
   (fp16 inputs, fp32 accumulation, fp16 output).
5. Output layer‑norm, out‑gate, and final linear projection back to the
   original channel dimension are performed in float32.

The kernel is deliberately simple but uses large tile sizes (128×128)
that map well to the B200 GPU.  All heavy compute stays on the GPU; only
lightweight element‑wise operations remain in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton batched GEMM kernel (C = A @ B, where B is already transposed)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched fp16 GEMM with fp32 accumulation.

    A : (batch, M, K)   fp16
    B : (batch, K, N)   fp16 (already transposed)
    C : (batch, M, N)   fp16
    """
    batch_id = tl.program_id(0)      # which (B*H) instance
    pid_m = tl.program_id(1)         # tile row
    pid_n = tl.program_id(2)         # tile col

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch entry
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # A tile (M x K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        # B tile (K x N) – B is already transposed
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        # dot product; tl.dot on fp16 returns fp32 accumulation
        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def triton_batched_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Perform batched matrix multiplication C = A @ B using the Triton kernel.
    A: (batch, M, K)   fp16
    B: (batch, K, N)   fp16 (already transposed for the required contraction)
    Returns C: (batch, M, N) fp16
    """
    assert a.dtype == b.dtype == torch.float16, "Kernel expects fp16 tensors"
    assert a.is_contiguous() and b.is_contiguous(), "Inputs must be contiguous"
    batch, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2, "Inner dimensions must match"

    c = torch.empty((batch, M, N), dtype=torch.float16, device=a.device)

    # element strides (not byte strides)
    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_k, stride_b_n = b.stride()
    stride_c_batch, stride_c_m, stride_c_n = c.stride()

    # tile sizes – these work well up to N=1024 on Ada GPUs
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        a, b, c,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or float)
            - weights_dict: mapping from parameter name to torch.Tensor
            - config_dict: must contain 'dim', 'hidden_dim' and optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # shape: [B, N, N, dim], float32

    # --------------------------------------------------------------
    # 2. Cast to fp16 for heavy linear work
    # --------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Linear projections (fp16, no bias)
    # --------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 4. Optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] -> [B, N, N, 1] -> fp16
        mask_fp = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_fp
        right = right * mask_fp

    # --------------------------------------------------------------
    # 5. Gating (fp16) and application
    # --------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6. Core contraction via Triton batched GEMM
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # --------------------------------------------------------------
    # Rearrange to (B, H, N, N) where H == hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N) – (i,k)
    # Right needs (k,j) therefore transpose the last two axes
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) – (k,j)

    B, H, N, _ = left_perm.shape
    # Merge batch and head dimensions for the kernel
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Perform batched matmul (fp16 in, fp16 out)
    out_flat = triton_batched_matmul(left_flat, right_flat)   # (B*H, N, N)

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)   # convert to fp32 for the remaining ops

    # --------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final projection (float32)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
