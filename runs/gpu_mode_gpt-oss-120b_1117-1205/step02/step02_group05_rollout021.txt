    # A shape (B, H, M, K) not necessarily contiguous; similarly B shape (B,H,K,N).
    # Compute dims: Bsize = A.shape[0]; H = A.shape[1]; M = A.shape[2]; K = A.shape[3].

    # Compute strides:
    stride_a_batch, stride_a_hidden, stride_a_m, stride_a_k = A.stride()

    TOTAL = Bsize * H

        M, N, K,

        stride_c_batch, stride_c_hidden, stride_c_m, stride_c_n,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_hidden, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_hidden, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_hidden, stride_c_m, stride_c_n,
    hidden_dim,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # compute indexes
    combined_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute batch and hidden indices
    batch_idx = combined_id // hidden_dim
    hidden_idx = combined_id % hidden_dim

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers
    a_base = a_ptr + batch_idx * stride_a_batch + hidden_idx * stride_a_hidden
    b_base = b_ptr + batch_idx * stride_b_batch + hidden_idx * stride_b_hidden
    c_base = c_ptr + batch_idx * stride_c_batch + hidden_idx * stride_c_hidden

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # mask for k dimension
        mask_k = offs_k < K

        # pointers for A and B
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass with a fused Triton batched GEMM.

Algorithm
---------
1. LayerNorm over the channel dimension (float32).
2. Linear projections (half‑precision) to a hidden dimension.
3. Optional mask multiplication (broadcast over hidden dim).
4. Gating: linear + sigmoid (half‑precision) applied element‑wise.
5. Core contraction:
   For each batch element b and hidden channel h compute
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   which is a batched matrix multiplication left @ right^T.
   The contraction is performed by a custom Triton kernel that
   processes (B·hidden_dim) independent GEMMs of size N×N.
   The kernel works on the original tensor layout (no materialised
   permutes) by using the appropriate strides.
6. LayerNorm on the hidden dimension, multiplication by the output gate,
   and a final linear projection back to the original dimension.
All heavy arithmetic uses FP16 with FP32 accumulation; the final
output is FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: batched GEMM (A @ B) where A = (B, H, M, K)
#                                   B = (B, H, K, N)
# --------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions
    stride_a_batch, stride_a_hidden, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_hidden, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_hidden, stride_c_m, stride_c_n,
    hidden_dim,                        # runtime hidden dimension
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # combined index over (batch, hidden)
    combined_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # split into batch and hidden indices
    batch_idx = combined_id // hidden_dim
    hidden_idx = combined_id % hidden_dim

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this (batch, hidden) pair
    a_base = a_ptr + batch_idx * stride_a_batch + hidden_idx * stride_a_hidden
    b_base = b_ptr + batch_idx * stride_b_batch + hidden_idx * stride_b_hidden
    c_base = c_ptr + batch_idx * stride_c_batch + hidden_idx * stride_c_hidden

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (M x K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B tile: (K x N)
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # FP16 -> FP32 accumulation

    # write back result (FP16)
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def triton_batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM C = A @ B where
        A : (batch, hidden, M, K)    FP16
        B : (batch, hidden, K, N)    FP16
    Returns C : (batch, hidden, M, N) FP16
    """
    assert A.dtype == B.dtype == torch.float16
    assert A.is_cuda and B.is_cuda
    Bsize, H, M, K = A.shape
    _, _, K2, N = B.shape
    assert K == K2

    C = torch.empty((Bsize, H, M, N), dtype=torch.float16, device=A.device)

    # strides (int64)
    stride_a_batch, stride_a_hidden, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_hidden, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_hidden, stride_c_m, stride_c_n = C.stride()

    # tile sizes – tuned for Hopper (B200) FP16 TensorCore
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        Bsize * H,                 # combined (batch, hidden)
        triton.cdiv(M, BLOCK_M),   # tile rows
        triton.cdiv(N, BLOCK_N),   # tile cols
    )

    batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_hidden,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_hidden,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_hidden,
        stride_c_m,
        stride_c_n,
        H,                         # hidden_dim runtime argument
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# --------------------------------------------------------------
# Entry point expected by the evaluator
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operation.

    Arguments:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) – may be None/ignored
            - weights: dict of model parameters (tensors)
            - config: dict containing 'dim', 'hidden_dim', optional 'nomask'
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm over the channel dimension (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to half for the heavy part
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half precision, bias=False)
    # ------------------------------------------------------------------
    left = F.linear(
        x_fp16,
        weights["left_proj.weight"].to(torch.float16),
    )
    right = F.linear(
        x_fp16,
        weights["right_proj.weight"].to(torch.float16),
    )

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] -> [B, N, N, 1]
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4. Gating values (half‑precision sigmoid) and application
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_fp16,
        weights["left_gate.weight"].to(torch.float16),
    ).sigmoid()
    right_gate = F.linear(
        x_fp16,
        weights["right_gate.weight"].to(torch.float16),
    ).sigmoid()
    out_gate = F.linear(
        x_fp16,
        weights["out_gate.weight"].to(torch.float16),
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: batched GEMM (left @ right) in Triton
    #    left: (B, N, N, H) -> (B, H, N, N)
    #    right permuted to (B, H, N, N) with axes (k, i)
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2)   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1)  # (B, H, N, N) where inner dim is K

    # Perform the batched matrix multiplication using Triton
    out_perm = triton_batched_matmul(left_perm, right_perm)   # (B, H, N, N) FP16

    # ------------------------------------------------------------------
    # 6. Restore layout, apply output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = out_perm.permute(0, 2, 3, 1).to(torch.float32)   # (B, N, N, H)

    # Output LayerNorm over hidden dim
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # Multiply by output gate (cast to float32)
    out = out * out_gate.to(torch.float32)

    # Final linear projection back to original dim
    out = F.linear(out, weights["to_out.weight"])

    return out
```
