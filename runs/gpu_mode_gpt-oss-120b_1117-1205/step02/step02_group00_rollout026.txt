Implementation details for kernel:

```
@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,
    left_gate_raw_ptr, right_gate_raw_ptr,
    mask_ptr,
    total_elems: tl.int32,
    hidden_dim: tl.constexpr,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask_off = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask_off, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask_off, other=0.0)
    left_gate_raw = tl.load(left_gate_raw_ptr + offs, mask=mask_off, other=0.0)
    right_gate_raw = tl.load(right_gate_raw_ptr + offs, mask=mask_off, other=0.0)

    # sigmoid
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_raw))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))

    if HAS_MASK:
        # compute mask offset: each hidden_dim values share same mask
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask_off, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask_off)
    tl.store(right_ptr + offs, right, mask=mask_off)
```

"""
Optimized outgoing TriMul forward pass.

The heavy O(N³) contraction is performed with a single
batched GEMM (torch.bmm).  All linear projections and gate
computations are fused into one large matrix multiplication,
thereby reducing kernel launch overhead.  A small Triton
kernel fuses the element‑wise sigmoid, mask and gate
multiplications for the left/right branches.  The implementation
uses float16 throughout (except the final cast) to exploit
Tensor Cores on the B200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # [total] half
    right_ptr,              # [total] half
    left_gate_raw_ptr,      # [total] half
    right_gate_raw_ptr,     # [total] half
    mask_ptr,               # [mask_total] half (broadcasted)
    total_elements: tl.int32,
    hidden_dim: tl.constexpr,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present, else 0
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # load data
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate_raw = tl.load(left_gate_raw_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate_raw = tl.load(right_gate_raw_ptr + offs, mask=mask, other=tl.float16(0.0))

    # sigmoid for the gates (1 / (1 + exp(-x)))
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_raw))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))

    if HAS_MASK:
        # each consecutive hidden_dim elements share the same mask entry
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # write back
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate_raw, right_gate_raw, mask, hidden_dim):
    """
    In‑place fusion:
        left  = left * sigmoid(left_gate_raw)  * (mask?)
        right = right * sigmoid(right_gate_raw) * (mask?)
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy, never read when HAS_MASK=0

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate_raw,
        right_gate_raw,
        mask_ptr,
        total,
        hidden_dim=hidden_dim,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
    )
    # operation is in‑place; nothing is returned


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or numeric) or None
            - weights: mapping of parameter names to torch.Tensor
            - config: dict with keys ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------- 1. Input LayerNorm (float32) ----------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # cast to half for the heavy compute
    x = x.to(torch.float16)

    # ---------- 2. One fused linear for all projections/gates ----------
    # concatenate weights: left_proj, right_proj, left_gate, right_gate, out_gate
    w_left = weights["left_proj.weight"]
    w_right = weights["right_proj.weight"]
    w_lgate = weights["left_gate.weight"]
    w_rgate = weights["right_gate.weight"]
    w_outgate = weights["out_gate.weight"]
    weight_concat = torch.cat(
        [w_left, w_right, w_lgate, w_rgate, w_outgate], dim=0
    ).to(torch.float16)

    B, N, _, _ = input_tensor.shape
    P = B * N * N  # number of (b,i,j) positions
    x_flat = x.view(P, dim)  # (B*N*N, dim)

    # Linear: (P, dim) @ (dim, 5*hidden_dim).T -> (P, 5*hidden_dim)
    proj = F.linear(x_flat, weight_concat)  # no bias
    h = hidden_dim
    # split the concatenated output
    left_proj = proj[:, 0:h]
    right_proj = proj[:, h:2 * h]
    left_gate_raw = proj[:, 2 * h:3 * h]
    right_gate_raw = proj[:, 3 * h:4 * h]
    out_gate_raw = proj[:, 4 * h:5 * h]

    # reshape to original 4‑D layout
    left_proj = left_proj.view(B, N, N, h).contiguous()
    right_proj = right_proj.view(B, N, N, h).contiguous()
    left_gate_raw = left_gate_raw.view(B, N, N, h).contiguous()
    right_gate_raw = right_gate_raw.view(B, N, N, h).contiguous()
    out_gate = torch.sigmoid(out_gate_raw).view(B, N, N, h).contiguous()

    # ---------- 3. Fuse mask + gates (Triton) ----------
    if not nomask:
        # ensure mask is half and has a trailing singleton dimension for broadcasting
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        # flatten for the kernel
        mask_flat = mask_h.view(-1).contiguous()
    else:
        mask_flat = None

    _apply_gate_and_mask(
        left_proj,
        right_proj,
        left_gate_raw,
        right_gate_raw,
        mask_flat,
        hidden_dim=h,
    )
    # after this, left_proj and right_proj already contain the gated & masked values

    # ---------- 4. Core O(N³) contraction via batched GEMM ----------
    # left: (B, N, N, h) → (B, h, N, N)
    # right: (B, N, N, h) → (B, h, N, N) with j/k ordering swapped
    left_mat = left_proj.permute(0, 3, 1, 2).contiguous().view(B * h, N, N)
    right_mat = right_proj.permute(0, 3, 2, 1).contiguous().view(B * h, N, N)

    out_mat = torch.bmm(left_mat, right_mat)  # (B*h, N, N) in half
    out = out_mat.view(B, h, N, N).permute(0, 2, 3, 1).contiguous()   # (B, N, N, h)

    # ---------- 5. Output LayerNorm, out‑gate, final projection ----------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # cast back to float32 as required
    return out.to(torch.float32)
