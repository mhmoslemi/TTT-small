@triton.jit
def mul_inplace_kernel(out_ptr, a_ptr, b_ptr, total, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE
    idx = offset + tl.arange(0, BLOCK_SIZE)
    mask = idx < total
    a = tl.load(a_ptr + idx, mask=mask, other=0.0, dtype=tl.float16)
    b = tl.load(b_ptr + idx, mask=mask, other=0.0, dtype=tl.float16)

def custom_kernel(data):
    '''
    TriMul (outgoing) forward pass implemented with Triton elementwise
    multiplication for gating and masked scaling, while the heavy
    O(N³) contraction is performed via a batched fp16 GEMM using
    torch.bmm (cuBLAS). Final LayerNorm and projection are done in
    PyTorch, returning a float32 tensor of shape [B, N, N, dim].
    '''
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. Input LayerNorm (float32)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B,N,N,dim], float32

    # Cast to fp16 for heavy ops
    x_fp16 = x.half()

    # 2. Linear projections (fp16)
    left = F.linear(x_fp16, weights["left_proj.weight"].half())
    right = F.linear(x_fp16, weights["right_proj.weight"].half())

    # 3. Gating tensors (fp16)
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].half()).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].half()).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].half()).sigmoid()

    BLOCK = 1024

        left.data_ptr(),

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiplication (in‑place)
# Computes out = a * b   (all tensors are fp16 and contiguous)
# ----------------------------------------------------------------------
@triton.jit
def mul_inplace_kernel(out_ptr, a_ptr, b_ptr, total_elements, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE
    idx = offset + tl.arange(0, BLOCK_SIZE)                # thread‑local indices
    mask = idx < total_elements                           # out‑of‑bounds mask

    a = tl.load(a_ptr + idx, mask=mask, other=0.0, dtype=tl.float16)
    b = tl.load(b_ptr + idx, mask=mask, other=0.0, dtype=tl.float16)
    tl.store(out_ptr + idx, a * b, mask=mask)


def custom_kernel(data):
    """
    Outgoing TriMul forward pass.

    Steps
    -----
    1. Layer‑norm the input (float32).
    2. Linear projections to the hidden dimension (fp16).
    3. Compute three gating tensors (fp16) and apply them to the projections
       with a Triton element‑wise kernel.
    4. Optional mask multiplication (handled by a simple torch broadcast).
    5. Core contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
       implemented as a batched matrix multiplication
       (B·hidden_dim) × (N×N) using `torch.bmm`, which leverages cuBLAS/Tensor
       Cores.
    6. Output layer‑norm, out‑gate scaling and final linear projection
       back to `dim` (float32).
    """
    # unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )                                   # [B, N, N, dim], float32

    # Cast to fp16 for the heavy ops
    x_fp16 = x.half()                  # [B, N, N, dim], fp16

    # ------------------------------------------------------------------
    # 2. Linear projections (fp16, bias=False)
    # ------------------------------------------------------------------
    left  = F.linear(x_fp16, weights["left_proj.weight"].half())
    right = F.linear(x_fp16, weights["right_proj.weight"].half())

    # ------------------------------------------------------------------
    # 3. Gating tensors (fp16)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, weights["left_gate.weight"].half()).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].half()).sigmoid()
    out_gate   = F.linear(x_fp16, weights["out_gate.weight"].half()).sigmoid()

    # ------------------------------------------------------------------
    # 4. Apply gates using a Triton kernel (in‑place)
    # ------------------------------------------------------------------
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    # left = left * left_gate   (in‑place)
    mul_inplace_kernel[grid](
        left.data_ptr(),
        left.data_ptr(),
        left_gate.data_ptr(),
        total,
        BLOCK_SIZE=BLOCK,
    )
    # right = right * right_gate (in‑place)
    mul_inplace_kernel[grid](
        right.data_ptr(),
        right.data_ptr(),
        right_gate.data_ptr(),
        total,
        BLOCK_SIZE=BLOCK,
    )

    # ------------------------------------------------------------------
    # 5. Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → [B, N, N, 1] → fp16
        mask_fp = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 6. Core contraction: batched GEMM (fp16)
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    # reshape to (B * hidden_dim, N, N)
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # matrix multiplication: (B*H, N, N) @ (B*H, N, N)^T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to [B, N, N, hidden_dim] (still fp16)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)        # convert to fp32 for the remaining ops

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"])

    return out
