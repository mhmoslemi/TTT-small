"""
TriMul (outgoing) forward pass with Triton‑accelerated elementwise
mask‑gate‑sigmoid fusion.  The heavy linear projections are fused
(left+gate, right+gate) to reduce kernel launches.  The cubic
contraction is performed as a batched GEMM using cuBLAS (torch.bmm),
which is the optimal way to compute the
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
operation.  The final LayerNorm, output gate, and output projection
are kept in PyTorch (FP32 for numerical stability).  The kernel
fuses three operations per element:
    out = proj * mask * sigmoid(gate_raw)
which avoids separate Python launches for mask, gating and
multiplication.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def proj_gate_mask_kernel(
    out_ptr,          # pointer to output tensor (flattened)
    proj_ptr,         # pointer to projection tensor (flattened)
    gate_ptr,         # pointer to gate‑raw tensor (flattened)
    mask_ptr,         # pointer to mask tensor (flattened, broadcast over H)
    total,            # total number of elements = B*N*N*H
    H: tl.constexpr,  # hidden dimension (size of innermost axis)
    BLOCK_SIZE: tl.constexpr,
):
    """
    out = proj * mask * sigmoid(gate_raw)
    mask is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # load projection and raw gate
    proj = tl.load(proj_ptr + offs, mask=mask)
    gate_raw = tl.load(gate_ptr + offs, mask=mask)

    # compute mask index (broadcast over hidden dim)
    mask_idx = offs // H
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask)

    # sigmoid(gate_raw) = 1 / (1 + exp(-x))
    gate = 1.0 / (1.0 + tl.exp(-gate_raw))

    # fused multiply
    out = proj * mask_val * gate

    tl.store(out_ptr + offs, out, mask=mask)


def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor: torch.Tensor [B, N, N, dim] (float32)
            mask_tensor : torch.Tensor [B, N, N] (float32) or None
            weights_dict: dict with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight",
                "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            config_dict: {"dim": int, "hidden_dim": int, "nomask": bool}
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # LayerNorm on the input (float32 for stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    # x_norm: [B, N, N, dim]  (float32)
    x_norm = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to half for the bulk of the work
    x = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # Fuse left projection + left gate, right projection + right gate
    # ------------------------------------------------------------------
    # Prepare fused weights (out_features = 2*hidden_dim)
    left_fused_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device).to(torch.float16)   # shape (2*hidden_dim, dim)

    right_fused_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device).to(torch.float16)   # shape (2*hidden_dim, dim)

    # out_gate weight (hidden_dim, dim)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    # Linear passes (produces fused tensors)
    left_fused = F.linear(x, left_fused_w)   # [B, N, N, 2*hidden_dim]
    right_fused = F.linear(x, right_fused_w) # [B, N, N, 2*hidden_dim]
    out_gate_raw = F.linear(x, out_gate_w)  # [B, N, N, hidden_dim]

    # Split fused tensors into proj / gate raw
    left_proj_raw, left_gate_raw = torch.split(left_fused, hidden_dim, dim=-1)
    right_proj_raw, right_gate_raw = torch.split(right_fused, hidden_dim, dim=-1)

    # ------------------------------------------------------------------
    # Prepare mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask_tensor is None or nomask:
        mask = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask = mask_tensor.to(dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # Triton kernel: proj * mask * sigmoid(gate_raw) for left and right
    # ------------------------------------------------------------------
    total = left_proj_raw.numel()  # B * N * N * hidden_dim
    BLOCK = 1024  # reasonable block size
    grid = (triton.cdiv(total, BLOCK),)

    # Allocate output buffers
    left = torch.empty_like(left_proj_raw)
    right = torch.empty_like(right_proj_raw)

    # Launch kernel for left branch
    proj_gate_mask_kernel[grid](
        left,
        left_proj_raw,
        left_gate_raw,
        mask,
        total,
        H=hidden_dim,
        BLOCK_SIZE=BLOCK,
    )
    # Launch kernel for right branch
    proj_gate_mask_kernel[grid](
        right,
        right_proj_raw,
        right_gate_raw,
        mask,
        total,
        H=hidden_dim,
        BLOCK_SIZE=BLOCK,
    )

    # ------------------------------------------------------------------
    # Compute output gate (sigmoid) – this is cheap compared to the kernel
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_raw)  # [B, N, N, hidden_dim] (float16)

    # ------------------------------------------------------------------
    # Cubic contraction via batched GEMM (torch.bmm)
    # ------------------------------------------------------------------
    # left: (B, N, N, hidden) -> (B, hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: (B, N, N, hidden) -> (B, hidden, N, N) with swapped axes (k, j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    LH = B * hidden_dim
    left_mat = left_perm.view(LH, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(LH, N, N) # (B*H, N, N)

    # Batched matrix multiplication: (i,k) @ (k,j) -> (i,j)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N), fp16

    # Restore [B, N, N, hidden]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Post‑processing: LayerNorm on hidden dim, output gate, final linear
    # ------------------------------------------------------------------
    # Convert to FP32 for the final LayerNorm
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float32)
    out_norm = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # [B, N, N, hidden], float32

    # Apply output gate (cast gate to FP32)
    out_norm = out_norm * out_gate.to(torch.float32)

    # Final projection back to dim
    to_out_w = weights["to_out.weight"].to(device).to(torch.float32)  # (dim, hidden)
    output = F.linear(out_norm, to_out_w)  # [B, N, N, dim], float32

    return output
