    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', True)  # note param name may be config['nomask'] but also maybe config['nomask'].

    )

    # Projections
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)
    left = torch.nn.functional.linear(x_fp16, left_proj_weight)  # shape B,N,N,H
    right = torch.nn.functional.linear(x_fp16, right_proj_weight)
    # Mask
    if not nomask:
        # mask shape [B,N,N], unsqueeze last dim, cast to fp16
        mask_fp = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_fp
        right = right * mask_fp
    # Gates
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    left_gate = torch.nn.functional.linear(x_fp16, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_fp16, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_fp16, out_gate_weight).sigmoid()
    left = left * left_gate
    right = right * right_gate
    # Core contraction using Triton
    # Permute to [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()
    B, H, N, _ = left_perm.shape
    total_batch = B * H
    left_flat = left_perm.view(total_batch, N, N)
    right_flat = right_perm.view(total_batch, N, N)
    # compute gemm
    out_flat = triton_batched_gemm(left_flat, right_flat)  # shape (total_batch, N, N)
    # reshape back to (B, H, N, N) with permute to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    # Cast to float32 for subsequent layernorm and final linear
    out = out.to(torch.float32)

        bias=weights["to_out_norm.bias"]
    )
    # Apply out_gate (converted to same dtype)
    out = out * out_gate.to(out.dtype)
    # final linear projection to dim
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)  # shape B,N,N,dim

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

Core operation:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

The O(N³) contraction is expressed as many independent GEMMs:
    for each (b, d)  ->  out_{b,d} = left_{b,d} @ right_{b,d}
where left_{b,d} and right_{b,d} are (N, N) matrices.
These GEMMs are executed by a custom Triton kernel (batched GEMM) in
FP16 with FP32 accumulation. All other operations (LayerNorm,
linear projections, gating) are performed in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ B   (A: (B, M, K), B: (B, K, N))
# --------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ah,
    stride_bk, stride_bn, stride_bh,
    stride_cm, stride_cn, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + batch_id * stride_ah
    b_batch = b_ptr + batch_id * stride_bh
    c_batch = c_ptr + batch_id * stride_ch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)

        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    c = acc.to(tl.float16)

    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def triton_batched_gemm(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B (no bias) using Triton.
    A: (batch, M, K)   fp16, contiguous
    B: (batch, K, N)   fp16, contiguous
    Returns C: (batch, M, N) fp16.
    """
    assert A.is_cuda and B.is_cuda
    assert A.dtype == B.dtype == torch.float16
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # strides (in elements)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bk, stride_bn = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    # tile sizes (tuned for B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_gemm_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_am,
        stride_ak,
        stride_ah,
        stride_bk,
        stride_bn,
        stride_bh,
        stride_cm,
        stride_cn,
        stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# --------------------------------------------------------------
# Triton‑accelerated forward pass for the outgoing TriMul operator
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data: (input_tensor, mask, weights_dict, config_dict)
            input_tensor: torch.Tensor [B, N, N, dim] (float32)
            mask:         torch.Tensor [B, N, N] (bool/int) – might be omitted
            weights:      dict with all linear / norm parameters (float32)
            config:       dict with keys 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # ----------------------------------------------------------
    # 2. Linear projections (FP16)
    # ----------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # (B, N, N, hidden_dim)
    right = F.linear(x_fp16, right_proj_w)  # (B, N, N, hidden_dim)

    # ----------------------------------------------------------
    # 3. Optional mask
    # ----------------------------------------------------------
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)  # (B, N, N, 1)
        left = left * mask_fp
        right = right * mask_fp

    # ----------------------------------------------------------
    # 4. Gating (FP16)
    # ----------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------
    # 5. Core contraction via Triton batched GEMM
    #    Compute out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ----------------------------------------------------------
    # reshape to (B, hidden_dim, N, N) so each (b,d) pair is a matrix
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    batch_h = B * H

    left_flat = left_perm.view(batch_h, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(batch_h, N, N) # (B*H, N, N)

    out_flat = triton_batched_gemm(left_flat, right_flat)  # (B*H, N, N)

    # reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # subsequent layers work in FP32

    # ----------------------------------------------------------
    # 6. Output LayerNorm, gate, final linear projection
    # ----------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(out.dtype)
    out = F.linear(out, weights["to_out.weight"])   # (B, N, N, dim)

    return out
