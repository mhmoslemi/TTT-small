```python
import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)
```

A `@triton.jit` function is NOT a Python function. It is launched with a grid
subscript, and its `tl.constexpr` parameters exist only inside it:

```python

def my_kernel(x_ptr, y_ptr, n, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < n
    tl.store(y_ptr + offs, tl.load(x_ptr + offs, mask=m, other=0.0), mask=m)

def custom_kernel(data):
    ...
    BLOCK = 128                                   # a host variable YOU define
```

```python
TriMul (outgoing) forward pass with a fused Triton batched GEMM.

Algorithm:
1. LayerNorm on the input channel dimension.
2. Linear projections (left, right) and gating (left_gate, right_gate, out_gate)
   are computed in fp16 for speed.
3. Optional mask is applied to the projected tensors.
4. Gates are applied element‑wise.
5. Core contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication A @ B^T where
   A = left.permute(0,3,1,2)  (B, H, N, N) and
   B = right.permute(0,3,2,1) (B, H, N, N) = right^T.
   The multiplication is performed by a custom Triton kernel that
   computes C = A @ B (both fp16) with float32 accumulation.
6. The result is reshaped back, layer‑normed, multiplied by out_gate and
   linearly projected back to the original channel dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ B   (A: (B, M, K), B: (B, K, N))
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)          # batch dimension (B*H)
    pid_m = tl.program_id(1)            # block row
    pid_n = tl.program_id(2)            # block col

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        # Load B tile (K x N)
        b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _triton_batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B (fp16 inputs, fp16 output)
    A : (batch, M, K)
    B : (batch, K, N)
    Returns C : (batch, M, N)
    """
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tunable block sizes (fit within shared memory and occupancy)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or float) – may be unused
            - weights_dict: mapping from parameter name to torch.Tensor
            - config_dict: must contain keys 'dim' and 'hidden_dim'.
              Optional 'nomask' (bool) indicates that mask should be ignored.

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [B, N, N, 1]
        mask_fp = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: batched GEMM using Triton
    #    Compute out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    # ------------------------------------------------------------------
    # Permute to bring hidden dim to second axis:
    #   left_perm: (B, H, N, N)  where H = hidden_dim
    #   right_perm: (B, H, N, N) with K and J swapped = right^T
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) = (B, H, K, J)

    B, H, N, _ = left_perm.shape

    # Flatten batch and hidden dimension for the kernel
    A_flat = left_perm.view(B * H, N, N)
    B_flat = right_perm.view(B * H, N, N)

    # Perform batch matmul A @ B (B already transposed)
    out_flat = _triton_batched_matmul(A_flat, B_flat)   # (B*H, N, N), fp16

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output layer‑norm, gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
```

## Approach and rationale

The outgoing version of the Triangle Multiplicative Update operator is a 4D tensor operation with shape [B, N, N, C]. The key challenge is the computational complexity, O(N³), which is dominated by the large sequence length N. To mitigate this, we reframe the operation as a batched matrix multiplication between projected input tensors and then apply gating operations. This decomposition allows us to leverage Triton's ability to handle large matrix operations efficiently with tiling and concurrent thread blocks.

We first apply LayerNorm to the input tensor along the channel dimension. Then, we perform linear projections (left_proj, right_proj, and gating projections) to transform the input to the hidden_dim space. The optional masking is applied element-wise. The core operation is implemented using a segmented batched matrix multiplication kernel that operates on dimensions [batch, hidden_dim, N, N] and is launched with a configurable grid size to approximate the dense N³ operation while minimizing memory overhead. The kernel uses fp16 for speed and fp32 for output precision.

The main bottlenecks and optimizations:
1. Input tensor is normalized using LayerNorm
2. Sequential indexing is tiled across hidden_dim batches for concurrent computation
3. Masking is handled element-wise before gating
4. The expensive O(N³) part is parallelized using a configurable grid with tile sizes

## Mathematical specification

Input: x ∈ ℝ^[B, N, N, dim]  
The TriMul operation is defined as:  
output[b,i,j,h] = Σ_{k=0}^{N-1} left[b,i,k,h] * right[b,j,k,h] for all b, i, j, h  
where:
- left[b,i,k,h] = projection of x[b,i,k, :]  
- right[b,j,k,h] = projection of x[b,j,k, :]  

The operation can be interpreted as a special case of matrix multiplication A @ Aᵀ followed by reshaping and gating. More precisely, after linear projections:
- Let Γ^[B, N, N, hidden_dim] = (left_proj @ x) where x is reshaped as [B*N*N, dim] and projected
- Let Λ^[B, N, N, hidden_dim] = (right_proj @ x)
- Then, the core operation computes Ω^[B, N, N, hidden_dim] = (Γ @ Λ)  
- Finally, result = Ω @ (out_gate), with proper element-wise multiplication

## Implementation procedure

1. **Input normalization**:  
   Apply LayerNorm to input tensor's channel dimension (dim) using given weights/biases.

2. **Linear projections**:  
   Project input using left_proj and right_proj to get tensors of shape [B, N, N, hidden_dim]. Also project gating terms.

3. **Masking**:  
   If mask is not None (and not using 'nomask' option), apply element-wise masking to left and right projections. Mask is broadcast along hidden_dim.

4. **Gating**:  
   Apply sigmoid to gating projections (left_gate, right_gate, out_gate). Make sure to use they're associated with the correct hidden_dim dimension.

5. **Core operation**:  
   Launch a Triton kernel with tile sizes configured for optimal memory usage and concurrency. The kernel should:
   - Process input as [B * hidden_dim, N, N] (conceptually)
   - Use multiple thread blocks to tile the [B * hidden_dim] space
   - Ensure proper ordering and dimensions for the multiplication

   The kernel operation is:
   ```python
   # Batched matmul kernel for dimensions [N, hidden_dim] and [N, hidden_dim] yielding [N, hidden_dim]
   def kernel_matmul(a_ptr, b_ptr, c_ptr, M, N, K):
       ...
   ```

   We'll prepare the tensor data as:
   ```python
   # Reshape inputs for the kernel to have shape [B * hidden_dim, N] for both left and right projections
   left_reshaped = left.view(B * hidden_dim, N)
   right_reshaped = right.view(B * hidden_dim, N)
   ```

   Then, the kernel will compute:
   ```python
   out_copy = kernel_matmul(left_reshaped, right_reshaped, ...)
   ```

   followed by:
   ```python
   out = out_copy.view(B, N, hidden_dim)
   ```

   This approximate of the full NxNxhidden_dim operation ensures that the computations are parallelized across a manageable grid.

6. **Output processing**:  
   Apply output LayerNorm to the core result and then multiply by the out_gate element-wise.

After the core operation is efficiently handled by the Triton kernel, the output is projected back to the original channel dimension.

## Compute budget and robustness

The dense N³ operation is replaced by a segmented batched matrix multiplication. The performance should be comparable but more scalable than the previous reference implementation.
