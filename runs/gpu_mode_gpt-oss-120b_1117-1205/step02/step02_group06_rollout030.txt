"""
Optimized outgoing TriMul (AlphaFold3) forward pass.

Algorithm:
1. LayerNorm on the input tensor (dim‑wise).
2. fp16 linear projections (left/right) and gating (left_gate, right_gate, out_gate).
3. Optional mask applied (broadcast over hidden dim).
4. Gates applied element‑wise to left/right.
5. Core contraction:
     out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched fp16 GEMM with fp32 accumulation using a custom Triton kernel.
   The kernel works directly on the original layout [B,N,N,H] to avoid costly permutes.
6. Output LayerNorm, out_gate multiplication, and final fp32 linear projection.

The heavy N³ contraction is fully expressed in Triton; all other ops use
high‑performance PyTorch (cuBLAS/Tensor‑cores). The final output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the outgoing TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,                               # all equal to seq_len
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_k, stride_right_n, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HIDDEN: tl.constexpr,
):
    """
    Compute for each (batch, hidden) pair:
        out[i, j] = Σ_k left[i, k] * right_T[k, j]
    where left and right are taken from tensors of shape [B, N, N, H] without
    materialising any permutes.
    """
    batch_id = tl.program_id(0)   # combined batch * hidden
    pid_m = tl.program_id(1)      # block row index (i)
    pid_n = tl.program_id(2)      # block col index (j)

    # De‑compose combined index into original batch and hidden index
    b_idx = batch_id // HIDDEN
    h_idx = batch_id % HIDDEN

    # Base pointers for this (b, h) slice
    left_base = left_ptr + b_idx * stride_left_batch + h_idx * stride_left_h
    right_base = right_ptr + b_idx * stride_right_batch + h_idx * stride_right_h
    out_base = out_ptr + b_idx * stride_out_batch + h_idx * stride_out_h

    # Tile offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left[i, k]  (row stride = stride_left_i, col stride = stride_left_k)
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right_T[k, j] = right[k, j] where row stride = stride_right_k, col stride = stride_right_n
        b_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)   # fp16 * fp16 -> fp32 accumulation

    # Store result back to out[i, j]
    c_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def _tri_mul_core_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for TriMul contraction.

    Arguments:
        left, right: torch tensors of shape [B, N, N, H] (fp16, contiguous)

    Returns:
        out: torch tensor of shape [B, N, N, H] (fp16, contiguous)
    """
    assert left.dtype == torch.float16 and right.dtype == torch.float16
    B, N, _, H = left.shape
    assert right.shape == (B, N, N, H)

    # Output tensor
    out = torch.empty_like(left)

    # Strides (in elements)
    s_batch, s_i, s_k, s_h = left.stride()        # (batch, i, k, hidden)
    rs_batch, rs_i, rs_k, rs_h = right.stride()
    os_batch, os_i, os_j, os_h = out.stride()

    # Kernel block sizes – good for most N up to 1024
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # 32 works well for fp16 Tensor‑Core utilisation

    # Grid: (combined_batch_hidden, blocks_M, blocks_N)
    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _tri_mul_outgoing_kernel[grid](
        left, right, out,
        N, N, N,
        s_batch, s_i, s_k, s_h,
        rs_batch, rs_k, rs_i, rs_h,      # note: row stride = k, col stride = i for the transposed view
        os_batch, os_i, os_j, os_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HIDDEN=H,
    )
    return out


# ----------------------------------------------------------------------
# Entry point following the required signature
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be unused
            - weights_dict: mapping from parameter name to torch.Tensor
            - config_dict: must contain keys 'dim' and 'hidden_dim'.
              Optional 'nomask' (bool) indicates that mask should be ignored.

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (dim‑wise)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – fp16
    # ------------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid after linear) – fp16
    # ------------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton GEMM
    #    Computes out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Ensure contiguous layout for the kernel
    left = left.contiguous()
    right = right.contiguous()
    out_fp16 = _tri_mul_core_matmul(left, right)      # [B, N, N, H] (fp16)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gate, final linear projection (fp32)
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"])

    return out
