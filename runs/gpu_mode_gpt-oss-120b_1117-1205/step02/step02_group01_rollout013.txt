"""
TriMul (outgoing) forward pass with partial Triton acceleration.
- LayerNorm, Linear projections and the cubic contraction are performed with
  torch (FP16 where possible) using fused batched GEMM (torch.matmul).
- Element‑wise mask·gate fusion for the left and right branches is done in a
  custom Triton kernel, reducing three separate elementwise kernels to one.
- Weight concatenation fuses left/right projections with their gates, cutting
  the number of GEMMs from 5 to 3.
The implementation returns a float32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def mul_gate_mask_kernel(
    out_ptr,          # pointer to output (flattened)
    a_ptr,            # pointer to tensor (flattened)   (left or right)
    gate_ptr,         # pointer to gate  (flattened)
    mask_ptr,         # pointer to mask  (flattened, broadcast over H)
    total,            # total number of elements (B*N*N*H)
    BLOCK_SIZE: tl.constexpr,
    H: tl.constexpr,  # hidden dimension (size of innermost axis)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    a = tl.load(a_ptr + offs, mask=mask)
    g = tl.load(gate_ptr + offs, mask=mask)

    # broadcast mask over the hidden dimension
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    tl.store(out_ptr + offs, a * g * m, mask=mask)


def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor: torch.Tensor [B, N, N, dim] (float32)
            mask_tensor : torch.Tensor [B, N, N] (float32) or None
            weights_dict: dict with keys:
                norm.weight, norm.bias,
                left_proj.weight, right_proj.weight,
                left_gate.weight, right_gate.weight,
                out_gate.weight,
                to_out_norm.weight, to_out_norm.bias,
                to_out.weight
            config_dict: {"dim": int, "hidden_dim": int}
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------------
    # 1. LayerNorm on input (FP32)
    # --------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_b = weights["norm.bias"].to(device, dtype=torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # [B,N,N,dim]

    # --------------------------------------------------------------------
    # 2. Cast to FP16 for heavy linear work
    # --------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------------
    # 3. Fused linear + gate projections (concatenate weights)
    # --------------------------------------------------------------------
    # concatenate left projection & left gate weights
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device, dtype=torch.float16)                     # (2*hidden_dim, dim)
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device, dtype=torch.float16)                     # (2*hidden_dim, dim)

    # compute concatenated outputs
    left_concat = F.linear(x_fp16, left_concat_w)   # [B,N,N,2*hidden_dim]
    right_concat = F.linear(x_fp16, right_concat_w) # [B,N,N,2*hidden_dim]

    # split projections and gates
    left = left_concat[..., :hidden_dim]                     # [B,N,N,hidden_dim]
    left_gate_pre = left_concat[..., hidden_dim:]            # [B,N,N,hidden_dim]
    left_gate = torch.sigmoid(left_gate_pre)                # FP16

    right = right_concat[..., :hidden_dim]                  # [B,N,N,hidden_dim]
    right_gate_pre = right_concat[..., hidden_dim:]          # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(right_gate_pre)              # FP16

    # out gate (separate linear)
    out_gate_w = weights["out_gate.weight"].to(device, dtype=torch.float16)
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))  # [B,N,N,hidden_dim] FP16

    # --------------------------------------------------------------------
    # 4. Apply mask and gate with a Triton kernel (broadcast mask over H)
    # --------------------------------------------------------------------
    if mask_tensor is None:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(device, dtype=torch.float16)

    # flatten tensors for the kernel
    total = B * N * N * hidden_dim
    left_flat = left.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_flat = right.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)
    mask_flat = mask_fp16.contiguous().view(-1)  # size B*N*N

    # allocate output buffers
    left_out_flat = torch.empty_like(left_flat)
    right_out_flat = torch.empty_like(right_flat)

    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    mul_gate_mask_kernel[grid](
        left_out_flat,
        left_flat,
        left_gate_flat,
        mask_flat,
        total,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
    )
    mul_gate_mask_kernel[grid](
        right_out_flat,
        right_flat,
        right_gate_flat,
        mask_flat,
        total,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
    )

    # reshape back
    left_out = left_out_flat.view(B, N, N, hidden_dim)   # [B,N,N,hidden_dim] FP16
    right_out = right_out_flat.view(B, N, N, hidden_dim) # [B,N,N,hidden_dim] FP16

    # --------------------------------------------------------------------
    # 5. Cubic contraction via batched matrix multiplication (FP16)
    # --------------------------------------------------------------------
    # arrange tensors to (B, hidden_dim, N, N)
    left_perm = left_out.permute(0, 3, 1, 2)   # (B, H, N, N)
    right_perm = right_out.permute(0, 3, 1, 2) # (B, H, N, N)
    # transpose the last two dimensions of right
    right_t = right_perm.transpose(-2, -1)    # (B, H, N, N)

    # batched GEMM: (B, H, N, N) @ (B, H, N, N) -> (B, H, N, N)
    out_perm = torch.matmul(left_perm, right_t)  # FP16

    out = out_perm.permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden_dim] FP16

    # --------------------------------------------------------------------
    # 6. Post‑processing: LayerNorm, final gate, output projection (FP32)
    # --------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype=torch.float32)
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # apply final gating
    out_norm = out_norm * out_gate.to(torch.float32)

    # final linear projection back to original dim
    to_out_w = weights["to_out.weight"].to(device, dtype=torch.float32)  # (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)  # [B,N,N,dim] float32

    return output
```
