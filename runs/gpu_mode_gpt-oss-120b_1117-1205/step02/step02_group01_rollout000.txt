"""
TriMul (outgoing) forward pass with a fused Triton kernel.

The kernel fuses the element‑wise mask, gate and linear‑proj results for the
left and right branches:
    left_out  = left_proj  * left_gate  * mask
    right_out = right_proj * right_gate * mask
All three multiplications are performed in a single Triton launch, reducing
kernel‑launch overhead compared to separate element‑wise passes.

The cubic contraction
    out[b,i,j,d] = Σ_k left_out[b,i,k,d] * right_out[b,j,k,d]
is realized as a batch of GEMMs (torch.bmm), which maps to cuBLAS and is
highly optimized.  Remaining LayerNorms, the final gate, and output projection
are kept in PyTorch (FP32) for numerical stability.

The implementation supports optional masking (mask may be None) and works in
mixed precision (FP16 for the heavy ops, FP32 for the final output).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def fuse_left_right_kernel(
    left_ptr,          # fp16, [B,N,N,H] flattened
    left_gate_ptr,     # fp16, same shape
    right_ptr,         # fp16, same shape
    right_gate_ptr,    # fp16, same shape
    mask_ptr,          # fp16, [B,N,N] flattened
    left_out_ptr,      # fp16, same shape as left
    right_out_ptr,     # fp16, same shape as right
    total,             # total number of elements = B*N*N*H
    H: tl.constexpr,   # hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # load the four tensors
    left = tl.load(left_ptr + offs, mask=mask)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask)
    right = tl.load(right_ptr + offs, mask=mask)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask)

    # broadcast mask across H (hidden) dimension
    mask_idx = offs // H          # index into mask (size B*N*N)
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    left_out = left * left_gate * m
    right_out = right * right_gate * m

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor: torch.Tensor [B, N, N, dim] (float32)
            mask_tensor : torch.Tensor [B, N, N] or None
            weights_dict : dict with weight tensors (see problem statement)
            config_dict  : {"dim": int, "hidden_dim": int}
    Returns:
        torch.Tensor [B, N, N, dim] (float32) – the TriMul output.
    """
    torch.cuda.set_device(torch.cuda.current_device())
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for the heavy path
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias) – FP16
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)   # (H, dim)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16) # (H, dim)

    left = F.linear(x_fp16, left_proj_w)   # [B,N,N,hidden_dim] fp16
    right = F.linear(x_fp16, right_proj_w) # [B,N,N,hidden_dim] fp16

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – FP16
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w)) # [B,N,N,hidden_dim]
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))    # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 5. Prepare mask (broadcasted over hidden_dim)
    # ------------------------------------------------------------------
    if mask_tensor is None:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(device).to(torch.float16)

    # ------------------------------------------------------------------
    # 6. Fuse element‑wise mask and gate for left/right via Triton
    # ------------------------------------------------------------------
    # Flatten tensors for kernel launch
    total = B * N * N * hidden_dim
    left_flat = left.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_flat = right.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)
    mask_flat = mask_fp16.contiguous().view(-1)

    left_out_flat = torch.empty_like(left_flat)
    right_out_flat = torch.empty_like(right_flat)

    BLOCK = 8192  # multiple of 32, reasonable occupancy
    grid = (triton.cdiv(total, BLOCK),)

    fuse_left_right_kernel[grid](
        left_flat,
        left_gate_flat,
        right_flat,
        right_gate_flat,
        mask_flat,
        left_out_flat,
        right_out_flat,
        total,
        H=hidden_dim,
        BLOCK_SIZE=BLOCK,
    )

    # Reshape back to [B,N,N,hidden_dim]
    left_out = left_out_flat.view(B, N, N, hidden_dim)
    right_out = right_out_flat.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 7. Cubic contraction via batched GEMM (torch.bmm)
    # ------------------------------------------------------------------
    # Permute to (B, hidden_dim, N, N) and prepare for bmm
    left_perm = left_out.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_out.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) where k/j swapped

    LH = B * hidden_dim
    left_mat = left_perm.view(LH, N, N)   # (B*H, N, N), fp16
    right_mat = right_perm.view(LH, N, N) # (B*H, N, N), fp16

    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N), fp16

    # Restore layout to [B,N,N,hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 8. Post‑processing: LayerNorm over hidden dim, final gate, output proj
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float32)

    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply final gating (out_gate is fp16)
    out_norm = out_norm * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float32)  # (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)  # [B, N, N, dim] float32

    return output
```
