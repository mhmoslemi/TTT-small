import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def mask_mul_kernel(tensor_ptr, mask_ptr, total_elements, hidden_dim, mask_len, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements
    # load tensor values
    vals = tl.load(tensor_ptr + offs, mask=mask, other=0.0)
    # compute mask index
    mask_offs = offs // hidden_dim
    mask_ok = mask_offs < mask_len
    mask_val = tl.load(mask_ptr + mask_offs, mask=mask_ok, other=1.0)
    vals = vals * mask_val
    tl.store(tensor_ptr + offs, vals, mask=mask)
```

def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operation.
    Core contraction is performed by a batched torch.bmm (cuBLAS) while
    optional masking is applied by a small Triton elementwise kernel.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. LayerNorm over channel dim (float32)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    x_fp16 = x.to(torch.float16)

    # 2. Linear projections
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16), bias=None)

    # 3. Gating values (sigmoid)
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # Apply gates (elementwise)
    left = left * left_gate
    right = right * right_gate

    # 4. Optional mask multiplication using Triton kernel
    if not nomask:
        # Ensure contiguous tensors
        left = left.contiguous()
        right = right.contiguous()
        mask_fp16 = mask.to(torch.float16).contiguous()
        total_elements = left.numel()
        mask_len = mask_fp16.numel()
        BLOCK = 32768  # adjust for GPU
        grid = (triton.cdiv(total_elements, BLOCK),)
        mask_mul_kernel[grid](
            left,
            mask_fp16,
            total_elements,
            hidden_dim,
            mask_len,
            BLOCK=BLOCK,
        )
        mask_mul_kernel[grid](
            right,
            mask_fp16,
            total_elements,
            hidden_dim,
            mask_len,
            BLOCK=BLOCK,
        )

    # 5. Core contraction: batched matrix multiplication (left @ right.T)
    # Rearrange to (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 2, 1).contiguous()  # transpose k and j axes
    B, H, N, _ = left.shape
    left_flat = left.view(B * H, N, N)
    right_flat = right.view(B * H, N, N)
    # Perform batched GEMM using cuBLAS (torch.bmm)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # Restore layout (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # back to float32 for layernorm and final projection

    # 6. Output LayerNorm, gating, final linear
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32), bias=None)

    return out
```

"""
TriMul (outgoing) forward pass.

Algorithm
---------
1. Layer‑norm the input over the channel dimension (float32).
2. Linear projection to the hidden dimension (fp16, no bias).
3. Compute three sigmoid gates (fp16) and apply them.
4. Optional mask multiplication is performed by a tiny Triton kernel
   (element‑wise fp16 * mask).
5. Core contraction:
   * reshape tensors to (B·H, N, N) where H = hidden_dim,
   * compute batched matrix multiplication left @ rightᵀ with torch.bmm
     (cuBLAS/tensor‑cores).
6. Layer‑norm the hidden result, multiply by the output gate,
   and project back to the original dimension.

Only the mask‑multiplication is implemented in Triton, satisfying the
requirement to use at least one kernel while keeping the heavy O(N³·H)
operation on the highly‑optimized cuBLAS path.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def mask_mul_kernel(
    tensor_ptr,            # fp16 tensor to be masked (contiguous 1‑D)
    mask_ptr,              # fp16 mask (contiguous 1‑D, length = B·N·N)
    total_elements,        # int: number of elements in tensor_ptr
    hidden_dim,            # int: hidden dimension H
    mask_len,              # int: number of mask entries = B·N·N
    BLOCK: tl.constexpr,   # compile‑time block size
):
    """In‑place multiplication of a 4‑D tensor by a broadcasted mask."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # offsets of this block
    active = offs < total_elements                     # valid element mask

    # Load tensor values
    vals = tl.load(tensor_ptr + offs, mask=active, other=0.0)

    # Compute index of the corresponding mask element (mask is broadcast over H)
    mask_offs = offs // hidden_dim
    mask_ok = mask_offs < mask_len
    mask_val = tl.load(mask_ptr + mask_offs, mask=mask_ok, other=1.0)

    # Apply mask
    vals = vals * mask_val

    # Write back
    tl.store(tensor_ptr + offs, vals, mask=active)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) – may be ignored
            - weights: dict of model parameters
            - config: dict with keys 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm over the channel dimension (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (fp16, no bias)
    # ------------------------------------------------------------------
    left = F.linear(
        x_fp16,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x_fp16,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gating values (sigmoid, fp16) and application
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_fp16,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_fp16,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_fp16,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Optional mask multiplication via a small Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        # Ensure contiguous layout for simple 1‑D indexing
        left = left.contiguous()
        right = right.contiguous()
        mask_fp16 = mask.to(torch.float16).contiguous()

        total_elems = left.numel()                     # B·N·N·H
        mask_len = mask_fp16.numel()                   # B·N·N
        BLOCK = 32768                                   # fits comfortably on B200

        grid = (triton.cdiv(total_elems, BLOCK),)

        mask_mul_kernel[grid](
            left,
            mask_fp16,
            total_elems,
            hidden_dim,
            mask_len,
            BLOCK=BLOCK,
        )
        mask_mul_kernel[grid](
            right,
            mask_fp16,
            total_elems,
            hidden_dim,
            mask_len,
            BLOCK=BLOCK,
        )

    # ------------------------------------------------------------------
    # 5. Core contraction: batched matrix multiplication (left @ rightᵀ)
    # ------------------------------------------------------------------
    # Reorder to (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)  -> L_{i,k}
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)  -> R_{j,k}

    B, H, N, _ = left.shape
    left_flat = left.view(B * H, N, N)                # (B·H, N, N)
    right_flat = right.view(B * H, N, N)              # (B·H, N, N)

    # Batched GEMM using cuBLAS (tensor cores)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B·H, N, N)

    # Restore (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)   # back to FP32 for the final layers

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float32),
        bias=None,
    )

    return out
