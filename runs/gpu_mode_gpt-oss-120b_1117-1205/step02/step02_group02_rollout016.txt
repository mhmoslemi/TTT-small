"""
TriMul outgoing forward pass using a fused Triton batched GEMM.
The heavy contraction:
    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
is computed as a batched matrix multiplication A @ B where
A = left.permute(0,3,1,2)  # (B, H, N, K)
B = right.permute(0,3,2,1) # (B, H, K, N)
We flatten the leading batch+hidden dimensions (B*H) and launch a
custom Triton kernel that tiles the matrices (128×128×32) and uses
float16 inputs with float32 accumulation. All other operations
(layer‑norms, linear projections, sigmoids, gating) are performed in
PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton batched matrix multiplication kernel (A @ B)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute C = A @ B for a batch of matrices.
    A : (batch, M, K)  fp16
    B : (batch, K, N)  fp16
    C : (batch, M, N)  fp16 (stored as fp16, accumulated in fp32)
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets for the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # masks for K dimension
        mask_k = offs_k < K

        # Load tiles
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Compute dot product (fp16 * fp16 -> fp32)
        acc += tl.dot(a, b)

    # Store result (cast back to fp16)
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def triton_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B using Triton.
    A : (batch, M, K)  torch.float16, contiguous
    B : (batch, K, N)  torch.float16, contiguous
    Returns C : (batch, M, N) torch.float16
    """
    assert A.is_cuda and B.is_cuda, "Tensors must be on CUDA"
    assert A.dtype == B.dtype == torch.float16, "Triton kernel expects fp16"
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, "Inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides in elements
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tuned tile sizes (good for Ampere B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor:  torch.Tensor of shape [B, N, N] (bool or float)
            - weights_dict: mapping from parameter name to torch.Tensor
            - config_dict: contains 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the heavy parts
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias) to hidden_dim
    # ------------------------------------------------------------------
    left = F.linear(
        x_fp16,
        weights["left_proj.weight"].to(torch.float16)
    )  # [B, N, N, hidden_dim]
    right = F.linear(
        x_fp16,
        weights["right_proj.weight"].to(torch.float16)
    )  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 4. Apply mask if present
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] -> [B, N, N, 1]
        mask_fp = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 5. Gating tensors (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_fp16,
        weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()  # [B, N, N, hidden_dim]
    right_gate = F.linear(
        x_fp16,
        weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = F.linear(
        x_fp16,
        weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()  # [B, N, N, hidden_dim]

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Core contraction: batched GEMM via Triton
    #    For each (b, d) compute out_{i,j} = sum_k left_{i,k} * right_{j,k}
    # ------------------------------------------------------------------
    # Permute to bring hidden dimension next to batch for easier flattening
    # left_perm: (B, hidden_dim, N, N) where axes = (b, d, i, k)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right_perm: (B, hidden_dim, N, N) where axes = (b, d, k, j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    B_, H_, N_, _ = left_perm.shape  # B_, H_, N_, N_ (the last dimension is N_)
    # Flatten batch and hidden dims into a single batch for Triton kernel
    left_flat = left_perm.view(B_ * H_, N_, N_)   # (B*H, M=N, K=N)
    right_flat = right_perm.view(B_ * H_, N_, N_) # (B*H, K=N, N=N)

    # Perform batched matrix multiplication using Triton
    out_flat = triton_matmul(left_flat, right_flat)  # (B*H, N, N) fp16

    # Reshape back to original layout: (B, N, N, hidden_dim)
    out = out_flat.view(B_, H_, N_, N_).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # for the remaining ops in fp32

    # ------------------------------------------------------------------
    # 7. Output LayerNorm + out_gate + final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    # Apply out gate (make sure dtype matches)
    out = out * out_gate.to(torch.float32)

    # Final linear projection to dim
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float32)
    )
    return out

"""
TriMul (outgoing) forward pass accelerated with a Triton batched GEMM.

The core contraction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
is a batched matrix multiplication per (b, d):
    out_{b,d} = left_{b,d} @ right_{b,d}.T
where left_{b,d} and right_{b,d} are (N×N) matrices.
We flatten the leading (batch, hidden) dimensions and launch a
custom Triton kernel that tiles the matrices (128×128×32) and
uses float16 inputs with float32 accumulation.  All other
operations (LayerNorm, linear projections, gating) stay in
PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ B
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute C = A @ B for a batched set of matrices.
    A : (batch, M, K)  fp16
    B : (batch, K, N)  fp16
    C : (batch, M, N)  fp16 (accumulated in fp32)
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)        # fp16*fp16 -> fp32 accumulation

    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def triton_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B using the Triton kernel above.
    A : (batch, M, K)  torch.float16, contiguous, CUDA
    B : (batch, K, N)  torch.float16, contiguous, CUDA
    Returns C : (batch, M, N) torch.float16
    """
    assert A.is_cuda and B.is_cuda, "inputs must be CUDA tensors"
    assert A.dtype == B.dtype == torch.float16, "kernel expects fp16 tensors"
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, "inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tile sizes tuned for B200 (Ampere)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor:  torch.Tensor [B, N, N] (bool/float) – may be ignored
            - weights_dict: mapping from parameter name to torch.Tensor
            - config_dict: must contain 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the heavy sub‑graph
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias) to hidden_dim
    # ------------------------------------------------------------------
    left = F.linear(
        x_fp16,
        weights["left_proj.weight"].to(torch.float16)
    )  # [B, N, N, hidden_dim]
    right = F.linear(
        x_fp16,
        weights["right_proj.weight"].to(torch.float16)
    )  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 4. Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)   # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 5. Gating tensors (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_fp16,
        weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()  # [B, N, N, hidden_dim]

    right_gate = F.linear(
        x_fp16,
        weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()

    out_gate = F.linear(
        x_fp16,
        weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()   # [B, N, N, hidden_dim]

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Core contraction via Triton batched GEMM
    #    For each (b, d) compute out_{i,j} = Σ_k left_{i,k} * right_{j,k}
    # ------------------------------------------------------------------
    # Rearrange dimensions so that hidden becomes the second axis.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)  i,k
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)  k,j

    B_, H_, N_, _ = left_perm.shape
    left_flat = left_perm.view(B_ * H_, N_, N_)   # (B*H, M=N, K=N)
    right_flat = right_perm.view(B_ * H_, N_, N_) # (B*H, K=N, N=N)

    # Batched matrix multiplication (fp16 inputs, fp32 accumulation)
    out_flat = triton_matmul(left_flat, right_flat)   # (B*H, N, N) fp16

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B_, H_, N_, N_).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)   # subsequent ops in fp32

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float32)
    )
    return out
