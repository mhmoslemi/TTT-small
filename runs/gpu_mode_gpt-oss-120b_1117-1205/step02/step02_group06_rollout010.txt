"""
TriMul (outgoing) forward pass.

Algorithm:
1. Apply LayerNorm over the channel dimension.
2. Project the normalized tensor to a hidden dimension (left/right) and compute three
   sigmoidal gates (left_gate, right_gate, out_gate) – all in float16.
3. Optionally apply a mask (broadcast over the hidden dimension) and multiply by the gates.
4. Core contraction: for each hidden head perform a batched matrix multiplication
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.  This is equivalent to a
   batched GEMM `L @ Rᵀ` where L and R have shape (B·H, N, N).  The GEMM is executed
   with `torch.bmm`, which uses cuBLAS and is highly optimized.
5. Convert the result to float32, apply a second LayerNorm, and multiply element‑wise
   by `out_gate`.  The element‑wise scaling is performed by a tiny Triton kernel.
6. Final linear projection back to the original channel dimension.

All heavy compute is done in FP16 (except the final LayerNorm and output projection,
which run in FP32).  A small Triton kernel is used to satisfy the requirement of
including a custom kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: element‑wise product (used for the final out_gate scaling)
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(x_ptr, y_ptr, out_ptr, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(x_ptr + offs, mask=mask)
    b = tl.load(y_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Entry point – the custom Triton‑accelerated TriMul kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or float)
            - weights_dict: mapping from parameter name to torch.Tensor
            - config_dict: must contain keys 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # If nomask is True, we ignore the mask; otherwise we apply it.
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(
        x_fp16, weights["left_proj.weight"].to(torch.float16)
    )  # shape: [B, N, N, hidden_dim]
    right = F.linear(
        x_fp16, weights["right_proj.weight"].to(torch.float16)
    )  # shape: [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [B, N, N, 1] and fp16
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5. Gating (sigmoid after linear) – still in fp16
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_fp16, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = F.linear(
        x_fp16, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = F.linear(
        x_fp16, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    # Apply the element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Core contraction: batched GEMM L @ Rᵀ
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # Move hidden dimension to the batch axis for a simple batched bmm
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Collapse batch and head dimensions: (B*H, N, N)
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Batched matrix multiplication: L @ Rᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # Reshape back to (B, N, N, H) and convert to float32 for the next LayerNorm
    out = (
        out_flat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )  # shape: [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 7. Output LayerNorm (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 8. Element‑wise multiplication by out_gate (float32) via Triton
    # ------------------------------------------------------------------
    out_gate_fp32 = out_gate.to(torch.float32)

    # Flatten tensors for the kernel
    total_elems = out.numel()
    BLOCK = 1024  # convenient block size
    grid = (triton.cdiv(total_elems, BLOCK),)

    # Triton kernel launches work on contiguous tensors
    out_contig = out.contiguous()
    out_scaled = torch.empty_like(out_contig)

    elementwise_mul_kernel[grid](
        out_contig,
        out_gate_fp32,
        out_scaled,
        total_elems,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 9. Final linear projection back to `dim` (float32)
    # ------------------------------------------------------------------
    out_final = F.linear(
        out_scaled, weights["to_out.weight"].to(torch.float32)
    )  # shape: [B, N, N, dim]

    return out_final
```
