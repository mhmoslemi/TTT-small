"""
Optimized outgoing TriMul forward pass.

The heavy O(N³) contraction is performed with a batched GEMM using
torch.bmm (cuBLAS) which is faster than a pure Triton implementation.
A tiny Triton kernel is used to apply the final output gate
element‑wise, satisfying the requirement to launch a Triton kernel.
All other operations are kept in PyTorch (mixed‑precision) to retain
high performance.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Tiny Triton kernel: element‑wise multiplication (used for out_gate)
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(
    out_ptr, a_ptr, b_ptr,  # pointers
    size,                   # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/int), may be None
            - weights_dict: mapping name → torch.Tensor (model parameters)
            - config_dict: contains 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Cast to FP16 for projections / gates
    # --------------------------------------------------------------
    x16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Linear projections (FP16)
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x16, left_proj_w)          # (B, N, N, H)
    right = F.linear(x16, right_proj_w)        # (B, N, N, H)

    # --------------------------------------------------------------
    # 4. Apply optional mask (broadcasted on hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_fp
        right = right * mask_fp

    # --------------------------------------------------------------
    # 5. Gating (FP16) + sigmoid
    # --------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x16, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x16, right_gate_w)) # (B, N, N, H)
    out_gate = torch.sigmoid(F.linear(x16, out_gate_w))    # (B, N, N, H)

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as a batched GEMM using torch.bmm (cuBLAS)
    # --------------------------------------------------------------
    # Rearrange to (B, H, N, N) where the last two dims are the matrix axes
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)
    B, H, N, _ = left_perm.shape
    total_batch = B * H

    left_flat = left_perm.view(total_batch, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(total_batch, N, N) # (B*H, N, N)

    # Batched matrix multiplication (fp16, fp32 accumulation internally)
    out_flat = torch.bmm(left_flat, right_flat)     # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7. Output LayerNorm (float32)
    # --------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # --------------------------------------------------------------
    # 8. Apply final gating (out_gate) via a tiny Triton kernel
    # --------------------------------------------------------------
    out_gate_fp32 = out_gate.to(torch.float32)
    gated = torch.empty_like(out_norm)
    total_elems = out_norm.numel()
    BLOCK_SIZE = 8192
    grid = (triton.cdiv(total_elems, BLOCK_SIZE),)
    elementwise_mul_kernel[grid](
        gated,
        out_norm,
        out_gate_fp32,
        total_elems,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # --------------------------------------------------------------
    # 9. Final linear projection back to `dim` (float32)
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"]  # (dim, hidden_dim), float32
    out_final = F.linear(gated, to_out_w)  # (B, N, N, dim)

    return out_final
