"""
TriMul (outgoing) forward pass.

Algorithm:
1. Layer‑norm over the input channel dimension.
2. FP16 linear projections for left/right and their gates.
3. Optional mask applied (broadcast over the hidden dimension) – performed with
   a tiny Triton kernel when a mask is present.
4. Multiply projections by their respective gates.
5. Core contraction: for each hidden channel compute a matrix product
   left[b,i,:,h] @ right[b,j,:,h]ᵀ.  Implemented as a batched GEMM using
   torch.bmm (cuBLAS) on FP16 tensors.
6. Output layer‑norm, sigmoid gate, and final linear projection back to the
   original channel dimension.
The heavy computation (batched GEMM) uses cuBLAS, which is faster than a
generic Triton implementation for the problem sizes.  A small Triton kernel
is kept for mask handling to satisfy the “use Triton” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _apply_mask_kernel(
    left_ptr, right_ptr, mask_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_H: tl.constexpr
):
    """
    In‑place multiplication of `left` and `right` by a broadcast mask.
    left/right: [B, N, N, H] (contiguous, fp16)
    mask      : [B, N, N]   (contiguous, fp16)
    """
    pid = tl.program_id(0)          # flattened index over (b,i,j)
    pid_h = tl.program_id(1)        # hidden‑channel block

    # decode (b,i,j) from pid
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # load scalar mask value
    mask_offset = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    mask_val = tl.load(mask_ptr + mask_offset, mask=True, other=0.0)

    # hidden‑channel offsets for this block
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # left
    left_base = left_ptr + b * stride_left_b + i * stride_left_i + j * stride_left_j
    left_ptrs = left_base + offs_h * stride_left_h
    left = tl.load(left_ptrs, mask=mask_h, other=0.0)
    left = left * mask_val
    tl.store(left_ptrs, left, mask=mask_h)

    # right
    right_base = right_ptr + b * stride_right_b + i * stride_right_i + j * stride_right_j
    right_ptrs = right_base + offs_h * stride_right_h
    right = tl.load(right_ptrs, mask=mask_h, other=0.0)
    right = right * mask_val
    tl.store(right_ptrs, right, mask=mask_h)


def _apply_mask_triton(left: torch.Tensor, right: torch.Tensor, mask: torch.Tensor):
    """
    Launch the Triton kernel that multiplies `left` and `right` in‑place by `mask`.

    Arguments
    ---------
    left, right : torch.Tensor, shape [B, N, N, H], dtype torch.float16, contiguous
    mask        : torch.Tensor, shape [B, N, N],    dtype torch.float16, contiguous
    """
    B, N, N2, H = left.shape
    assert N == N2 and mask.shape == (B, N, N)

    # strides
    s_lb, s_li, s_lj, s_lh = left.stride()
    s_rb, s_ri, s_rj, s_rh = right.stride()
    s_mb, s_mi, s_mj = mask.stride()

    BLOCK_H = 64  # tile along hidden dimension
    total = B * N * N
    grid = (
        total,
        triton.cdiv(H, BLOCK_H),
    )
    _apply_mask_kernel[grid](
        left,
        right,
        mask,
        B,
        N,
        H,
        s_lb,
        s_li,
        s_lj,
        s_lh,
        s_rb,
        s_ri,
        s_rj,
        s_rh,
        s_mb,
        s_mi,
        s_mj,
        BLOCK_H=BLOCK_H,
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
        - mask         : torch.Tensor [B, N, N]     (bool/float, optional)
        - weights      : dict of torch.Tensor parameters
        - config       : dict with keys ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # float32

    # ------------------------------------------------------------------
    # 2. FP16 linear projections (no bias)
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # Ensure contiguous layout for the mask kernel
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension) – Triton fused kernel
    # ------------------------------------------------------------------
    if not nomask:
        # Convert mask to fp16 and make contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        _apply_mask_triton(left, right, mask_fp16)

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: batched GEMM using cuBLAS (torch.bmm)
    #    Compute out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    # ------------------------------------------------------------------
    # permute to (B, H, N, N) layout
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape

    # flatten batch*hidden for torch.bmm
    A = left_perm.view(B * H, N, N)          # (B*H, N, N)  fp16
    B_mat = right_perm.view(B * H, N, N)    # (B*H, N, N)  fp16

    out_flat = torch.bmm(A, B_mat)          # (B*H, N, N)  fp16

    # reshape back to (B, N, N, H) and cast to float32
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
