@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ah,
    stride_bk, stride_bn, stride_bh,
    stride_cm, stride_cn, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Compute base pointers for this batch
    a_batch_ptr = a_ptr + batch * stride_ah
    b_batch_ptr = b_ptr + batch * stride_bh
    c_batch_ptr = c_ptr + batch * stride_ch

    # Initialize accumulator

"""
Outgoing TriMul forward pass (AlphaFold3) with Triton‑accelerated contraction.

The heavy O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is computed as a batched GEMM for each hidden dimension d:
    left_perm  : (B, H, N, N)   where left_perm[b,d,i,k] = left[b,i,k,d]
    right_perm : (B, H, N, N)   where right_perm[b,d,k,j] = right[b,j,k,d]
    out_perm   = batched_gemm(left_perm, right_perm)  # (B,H,N,N)
All other operations (LayerNorm, linear projections, gating, final projection)
are performed with PyTorch in mixed precision (FP32 → FP16 → FP32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton batched GEMM kernel (C = A @ B)
# --------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ah,
    stride_bk, stride_bn, stride_bh,
    stride_cm, stride_cn, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute C = A @ B for a batch of independent matrices.

    A: (batch, M, K)
    B: (batch, K, N)
    C: (batch, M, N)
    All tensors are in FP16, accumulation is in FP32.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Tile offsets within M and N
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the current batch
    a_batch = a_ptr + batch_id * stride_ah
    b_batch = b_ptr + batch_id * stride_bh
    c_batch = c_ptr + batch_id * stride_ch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        # Load B tile (K x N)
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate matrix product
        acc += tl.dot(a, b)

    # Write back result in FP16
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def triton_batched_gemm(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B using Triton.

    A: (batch, M, K), FP16, contiguous
    B: (batch, K, N), FP16, contiguous
    Returns C: (batch, M, N), FP16.
    """
    assert A.is_cuda and B.is_cuda
    assert A.dtype == B.dtype == torch.float16
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides in number of elements
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bk, stride_bn = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    # Tunable tile sizes (work well for N up to 1024)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_gemm_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_am,
        stride_ak,
        stride_ah,
        stride_bk,
        stride_bn,
        stride_bh,
        stride_cm,
        stride_cn,
        stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# --------------------------------------------------------------
# Entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/int) – may be ignored if config["nomask"]=True
            - weights_dict: mapping from parameter name to torch.Tensor
            - config_dict: contains keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # shape [B,N,N,dim], float32

    # Convert to FP16 for subsequent matmul‑heavy work
    x_fp16 = x.half()  # B,N,N,dim  (fp16)

    # ----------------------------------------------------------
    # 2. Linear projections (FP16)
    # ----------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()

    left = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)  # (B,N,N,hidden_dim)

    # ----------------------------------------------------------
    # 3. Mask (optional)
    # ----------------------------------------------------------
    if not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_fp
        right = right * mask_fp

    # ----------------------------------------------------------
    # 4. Gates (FP16)
    # ----------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()

    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden_dim)

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------
    # 5. Core O(N³) contraction via Triton batched GEMM
    # ----------------------------------------------------------
    # Permute to (B, H, N, N) where H = hidden_dim.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) = (B, H, K, J)

    B, H, N, _ = left_perm.shape
    total_batch = B * H

    # Collapse batch and hidden dimensions for the batched GEMM
    left_flat = left_perm.view(total_batch, N, N)
    right_flat = right_perm.view(total_batch, N, N)

    # Perform batched matrix multiplication: (total_batch, N, N) @ (total_batch, N, N)
    out_flat = triton_batched_gemm(left_flat, right_flat)  # FP16

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.float()  # convert back to FP32 for the remaining layers

    # ----------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection
    # ----------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # Apply out_gate (originally FP16) – cast to match dtype
    out = out * out_gate.to(out.dtype)

    # Final linear projection back to original dimension
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out
