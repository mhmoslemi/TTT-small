@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    a_batch_ptr = a_ptr + batch_id * stride_a_batch
    b_batch_ptr = b_ptr + batch_id * stride_b_batch

"""
Triton-accelerated outgoing triangle multiplicative update (TriMul).

The implementation follows the PyTorch reference:

1. LayerNorm over the channel dimension.
2. Linear projections (left/right) and gating.
3. Optional mask applied to the projected tensors.
4. Core contraction: for each hidden channel d,
   out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d].
   This is a batched matrix multiplication left @ rightᵀ, realized
   by a custom Triton kernel that processes all (batch * hidden) GEMMs
   simultaneously.
5. LayerNorm on the hidden dimension, out‑gate modulation,
   and final linear projection back to the original dimension.

All heavy arithmetic is performed in FP16 with FP32 accumulation; the
final result is returned as FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Computes C = A @ B for a batch of matrices.
    A : (batch, M, K)   fp16
    B : (batch, K, N)   fp16   (already transposed for the TriMul contraction)
    C : (batch, M, N)   fp16
    """
    batch_id = tl.program_id(0)   # batch * hidden index
    pid_m = tl.program_id(1)      # tile row
    pid_n = tl.program_id(2)      # tile column

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for this batch
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A block (M x K)
        a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        # Load B block (K x N) – B is already transposed (K,N)
        b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        # Block‑wise matmul
        acc += tl.dot(a, b)

    # Store the result (cast back to fp16)
    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

    BLOCK_N = 128
    BLOCK_K = 32

        stride_b_k,

        stride_c_batch,

    right = right * right_gate

    # 5. Core contraction: batched GEMM left @ rightᵀ
    # ------------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N)   for left

"""
Triton-accelerated outgoing triangle multiplicative update (TriMul).

The algorithm follows the PyTorch reference:

1. LayerNorm over the channel dimension (float32).
2. Linear projections (left/right) to a hidden dimension (fp16, no bias).
3. Optional mask multiplication (broadcast over the hidden dim).
4. Gating values (sigmoid of three linear projections) applied element‑wise.
5. Core contraction:
   out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d].
   This is a batched matrix multiplication left @ rightᵀ, realized by a
   custom Triton kernel that processes all (batch × hidden) GEMMs in
   fp16 with fp32 accumulation.
6. LayerNorm on the hidden dimension, multiplication by the out‑gate,
   and a final linear projection back to the original dimension.
All heavy arithmetic is performed in fp16; the final output is fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    C = A @ B  (batched GEMM).
    A : (batch, M, K)   fp16
    B : (batch, K, N)   fp16  (already transposed for TriMul)
    C : (batch, M, N)   fp16
    """
    batch_id = tl.program_id(0)   # batch × hidden index
    pid_m = tl.program_id(1)      # tile‑row index
    pid_n = tl.program_id(2)      # tile‑col index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)          # fp16·fp16 → fp32 accumulator

    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])
