def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', True)

    # Step 1: LayerNorm on input (float32) → half
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights['norm.weight'],
        bias=weights['norm.bias']
    )
    x_h = x.to(torch.float16)

    # Flatten
    B, N, _, _ = x_h.shape
    total_positions = B * N * N
    x_flat = x_h.view(total_positions, dim)  # shape (total, dim)

    # Concatenate weight matrices
    # We want shape (5*hidden_dim, dim), but already per weight: (hidden_dim, dim)
    left_proj_w = weights['left_proj.weight']
    right_proj_w = weights['right_proj.weight']
    left_gate_w = weights['left_gate.weight']
    right_gate_w = weights['right_gate.weight']
    out_gate_w = weights['out_gate.weight']

    # Ensure half
    cat_weight = torch.cat([left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w], dim=0).to(torch.float16)  # shape (5*hidden_dim, dim)

    # Linear pass (no bias)
    proj = torch.nn.functional.linear(x_flat, cat_weight)  # shape (total, 5*hidden_dim) half

    # Split
    splits = torch.split(proj, hidden_dim, dim=1)  # yields 5 tensors
    left_proj, right_proj, left_gate_logit, right_gate_logit, out_gate_logit = splits

    # Gates
    left_gate = left_gate_logit.sigmoid()
    right_gate = right_gate_logit.sigmoid()
    out_gate = out_gate_logit.sigmoid()

    # Apply mask+gates via Triton or torch.
    # Flatten left/right and gates to 1D for kernel.

    left = left_proj  # will be overwritten with masked product
    right = right_proj
    # shape is (total, hidden_dim)
    # flatten to 1D for kernel
    left_flat = left.reshape(-1)
    right_flat = right.reshape(-1)
    left_gate_flat = left_gate.reshape(-1)
    right_gate_flat = right_gate.reshape(-1)

    if not nomask:
        # mask shape (B, N, N)
        mask_h = mask.to(torch.float16).contiguous()
        mask_flat = mask_h.view(-1)  # (total_positions)
        # Launch kernel with mask
        total_elements = left_flat.numel()
        BLOCK = 65536  # 64K threads per block
        grid = (triton.cdiv(total_elements, BLOCK),)
        HAS_MASK = 1
        gate_mask_apply_kernel[grid](
            left_flat, right_flat,
            left_gate_flat, right_gate_flat,
            mask_flat,
            total_elements,
            hidden_dim,
            HAS_MASK=HAS_MASK,
            BLOCK=BLOCK,
        )
    else:
        # No mask: just multiply left*gate and right*gate
        left = left * left_gate
        right = right * right_gate

    # After kernel, left and right hold the masked and gated values.
    # If kernel used, values are stored back to left_flat/right_flat
    # need to reshape back
    if not nomask:
        left = left_flat.view(B, N, N, hidden_dim)
        right = right_flat.view(B, N, N, hidden_dim)

    # Now core contraction
    # Permute to (B, hidden_dim, N, N) and transposed right
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden_dim, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, hidden_dim, N, N) (transpose)
    B_, H_, N_, _ = left_perm.shape
    # flatten batch*head dims
    left_mat = left_perm.view(B_ * H_, N_, N_)
    right_mat = right_perm.view(B_ * H_, N_, N_)
    # batched matmul
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N), half

    # reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B_, H_, N_, N_).permute(0, 2, 3, 1).contiguous()

    # to_out_norm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights['to_out_norm.weight'].to(torch.float16),
        bias=weights['to_out_norm.bias'].to(torch.float16),
    )
    # out_gate shape (B,N,N,hidden_dim) half
    out = out * out_gate

    # final projection
    out = torch.nn.functional.linear(out, weights['to_out.weight'].to(torch.float16))

    # return float32
    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass with a fused projection+gate+mask kernel.

Algorithm
---------
1. Layer‑normalize the input tensor x ∈ [B,N,N,dim].
2. Flatten x → (B·N·N, dim) and apply a *single* linear layer whose weight
   concatenates the five weight matrices (left_proj, right_proj,
   left_gate, right_gate, out_gate).  This yields a tensor of shape
   (B·N·N, 5·hidden_dim).
3. Split the concatenated result into the five components and compute the
   sigmoid gates for left/right/out.
4. Fuse element‑wise mask (if present) and the left/right gates with a small
   Triton kernel (in‑place multiplication).  The kernel processes the flattened
   tensors with one thread per element and broadcasts the mask over the hidden
   dimension.
5. Compute the core contraction
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]   (· = dot over k)
   using a single batched GEMM: left and right are reshaped to (B·hidden_dim,
   N, N) and multiplied with torch.bmm.
6. Apply the output LayerNorm, multiply by the out‑gate, and a final linear
   projection back to dim.
7. Return the result in float32.

Only the mask‑gate fusion step is performed in Triton; all heavy linear
operations use cuBLAS via torch.nn.functional.linear / torch.bmm for maximal
performance.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,              # half*
    right_ptr,             # half*
    left_gate_ptr,         # half*
    right_gate_ptr,        # half*
    mask_ptr,              # half*  (broadcast over hidden_dim)
    total_elements: tl.int32,
    hidden_dim: tl.int32,
    HAS_MASK: tl.constexpr,   # 1 if a mask is present, else 0
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    # load
    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    # apply gates
    left = left * left_gate
    right = right * right_gate

    if HAS_MASK:
        # each mask entry corresponds to hidden_dim consecutive elements
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)
        left = left * mask_val
        right = right * mask_val

    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or float) – may be ignored if nomask=True
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 → half)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    B, N, _, _ = x_h.shape
    total_positions = B * N * N

    # ------------------------------------------------------------------
    # 2. Fused projection & gate logits (single GEMM)
    # ------------------------------------------------------------------
    x_flat = x_h.reshape(total_positions, dim)                     # (B·N·N, dim)

    # concatenate the five weight matrices (all shape [hidden_dim, dim])
    cat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                            # (5·hidden_dim, dim)

    # single linear projection (no bias)
    proj = F.linear(x_flat, cat_weight)                             # (B·N·N, 5·hidden_dim)

    # split back into the five components
    left_proj, right_proj, left_gate_logit, right_gate_logit, out_gate_logit = proj.split(
        hidden_dim, dim=1
    )

    # gates with sigmoid
    left_gate = left_gate_logit.sigmoid()
    right_gate = right_gate_logit.sigmoid()
    out_gate = out_gate_logit.sigmoid()  # shape (B·N·N, hidden_dim)

    # ------------------------------------------------------------------
    # 3. Fuse mask (if any) and the left/right gates via Triton
    # ------------------------------------------------------------------
    # Ensure tensors are contiguous before flattening
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    left_flat = left_proj.view(-1)      # (total_positions·hidden_dim,)
    right_flat = right_proj.view(-1)
    left_gate_flat = left_gate.view(-1)
    right_gate_flat = right_gate.view(-1)

    if not nomask:
        # mask: [B, N, N] → half and broadcast over hidden_dim
        mask_h = mask.to(torch.float16).contiguous()
        mask_flat = mask_h.view(-1)                     # (total_positions,)

        total_elements = left_flat.numel()
        BLOCK = 131072  # 128 Ki elements per program
        grid = (triton.cdiv(total_elements, BLOCK),)

        _gate_mask_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total_elements,
            hidden_dim,
            HAS_MASK=1,
            BLOCK=BLOCK,
        )
        # reshape back
        left = left_flat.view(B, N, N, hidden_dim)
        right = right_flat.view(B, N, N, hidden_dim)
    else:
        # No mask – just apply the gates directly
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as a batched GEMM (cuBLAS) over (B·hidden_dim) batches.
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden_dim, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, hidden_dim, N, N)

    B_, H_, N_, _ = left_perm.shape
    left_mat = left_perm.view(B_ * H_, N_, N_)      # (B·H, N, N)
    right_mat = right_perm.view(B_ * H_, N_, N_)    # (B·H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)       # (B·H, N, N), half
    out = out_mat.view(B_, H_, N_, N_).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    # out_gate has shape (B·N·N, hidden_dim) → reshape to (B,N,N,hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)
    out = out * out_gate

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
    )

    # Return in float32 as required
    return out.to(torch.float32)
