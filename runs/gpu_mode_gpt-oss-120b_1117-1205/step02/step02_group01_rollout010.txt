def apply_gate_mask_kernel(out_ptr, src_ptr, gate_ptr, mask_ptr, total_elems, hidden_dim, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask_cond = offs < total_elems
    src = tl.load(src_ptr + offs, mask=mask_cond)
    gate = tl.load(gate_ptr + offs, mask=mask_cond)
    # compute mask index:
    mask_idx = offs // hidden_dim
    mask = tl.load(mask_ptr + mask_idx, mask=mask_cond)
    out = src * gate * mask
    tl.store(out_ptr + offs, out, mask=mask_cond)

"""
High‑performance TriMul (outgoing) forward pass.

Algorithm
---------
1) Input LayerNorm (FP16) using model weights.
2) Linear projections (left/right) → hidden dimension.
3) Compute three gates (left, right, out) with sigmoid.
4) Fuse element‑wise multiplication of projection, gate and mask via a
   Triton kernel (one launch per branch).
5) Contract over the sequence dimension k:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (torch.bmm) after a cheap transpose.
6) LayerNorm on the hidden dimension, multiply by the output gate,
   and final linear projection back to the original dimension.
7) Cast to float32 for the returned tensor.

All heavy tensor work stays in FP16 to maximise bandwidth and compute
throughput; only the final result is upcast to FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def apply_gate_mask_kernel(
    out_ptr,      # *float16 output pointer (flattened)
    src_ptr,      # *float16 source tensor pointer (flattened)
    gate_ptr,     # *float16 gate tensor pointer (flattened)
    mask_ptr,     # *float16 mask pointer (flattened, broadcast over H)
    total,        # int total number of elements (B*N*N*H)
    BLOCK_SIZE: tl.constexpr,
    H: tl.constexpr,  # hidden dimension (size of the innermost axis)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    a = tl.load(src_ptr + offs, mask=mask)
    g = tl.load(gate_ptr + offs, mask=mask)

    # Broadcast mask over hidden dimension
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    tl.store(out_ptr + offs, a * g * m, mask=mask)


def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor: torch.Tensor [B, N, N, dim] (float32)
            mask_tensor : torch.Tensor [B, N, N]   (float32) – may be all‑ones
            weights_dict: model weights as described in the prompt
            config_dict : {"dim": int, "hidden_dim": int}
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # Unpack inputs
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------
    # 1. LayerNorm on input (FP16)
    # ------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float16)
    norm_b = weights["norm.bias"].to(device).to(torch.float16)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b).to(torch.float16)

    # ------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim], fp16
    right = F.linear(x, right_proj_w)  # [B, N, N, hidden_dim], fp16

    # ------------------------------------------------------------
    # 3. Gates (sigmoid)
    # ------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------
    # 4. Fuse projection * gate * mask using Triton
    # ------------------------------------------------------------
    if mask_tensor is None:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(torch.float16)

    # Flatten tensors for the kernel
    total = B * N * N * hidden_dim
    left_flat = left.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_flat = right.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)
    mask_flat = mask_fp16.contiguous().view(-1)

    left_out_flat = torch.empty_like(left_flat)
    right_out_flat = torch.empty_like(right_flat)

    BLOCK = 8192  # reasonable block size
    grid = (triton.cdiv(total, BLOCK),)

    apply_gate_mask_kernel[grid](
        left_out_flat,
        left_flat,
        left_gate_flat,
        mask_flat,
        total,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
    )
    apply_gate_mask_kernel[grid](
        right_out_flat,
        right_flat,
        right_gate_flat,
        mask_flat,
        total,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
    )

    left = left_out_flat.view(B, N, N, hidden_dim)
    right = right_out_flat.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------
    # 5. Contract over k using batched GEMM (torch.bmm)
    # ------------------------------------------------------------
    # left: (B, N, N, H) -> (B*H, N, N), rows=i, cols=k
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    # right: (B, N, N, H) -> (B*H, N, N), rows=k, cols=j
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat)                     # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------
    # 6. Post‑processing: LayerNorm, out‑gate, final linear
    # ------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)

    out_norm = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply output gate (element‑wise)
    out_norm = out_norm * out_gate

    # Final projection back to original dimension
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)  # (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)  # [B, N, N, dim], fp16

    # Cast to float32 for the final result
    return output.to(torch.float32)
