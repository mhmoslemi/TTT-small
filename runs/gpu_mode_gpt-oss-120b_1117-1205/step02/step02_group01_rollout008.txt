Triton kernel code:

@triton.jit
def mul_gate_mask_kernel(out_ptr, a_ptr, gate_ptr, mask_ptr, total, BLOCK_SIZE: tl.constexpr, H: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total
    a = tl.load(a_ptr + offs, mask=mask)
    g = tl.load(gate_ptr + offs, mask=mask)
    # compute mask index: each hidden dim entry repeats mask value
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

"""
TriMul (outgoing) forward pass with Triton‑fused mask‑gate multiplication.
The heavy cubic contraction is performed as a batched matrix multiplication
(using torch.bmm) which leverages cuBLAS.  The elementwise
`projection * gate * mask` for the left and right branches is fused into a
single Triton kernel to reduce memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def mul_gate_mask_kernel(
    out_ptr,                # flattened output pointer (float16)
    a_ptr,                  # flattened input projection pointer (float16)
    gate_ptr,               # flattened gate pointer (float16)
    mask_ptr,               # flattened mask pointer (float16), broadcast over H
    total,                  # total number of elements (int)
    BLOCK_SIZE: tl.constexpr,
    H: tl.constexpr,        # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    a = tl.load(a_ptr + offs, mask=mask)
    g = tl.load(gate_ptr + offs, mask=mask)

    # each mask element corresponds to H consecutive values in the flattened tensor
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    tl.store(out_ptr + offs, a * g * m, mask=mask)


def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor: torch.Tensor [B, N, N, dim] (float32)
            mask_tensor : torch.Tensor [B, N, N] (float32) or None
            weights_dict : dict with the following keys (all tensors on CPU)
                - "norm.weight", "norm.bias"
                - "left_proj.weight", "right_proj.weight"
                - "left_gate.weight", "right_gate.weight", "out_gate.weight"
                - "to_out_norm.weight", "to_out_norm.bias"
                - "to_out.weight"
            config_dict : {"dim": int, "hidden_dim": int}
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # --------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_b = weights["norm.bias"].to(device, dtype=torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # [B,N,N,dim]

    # --------------------------------------------------------------------
    # 2. Cast to FP16 for the heavy work
    # --------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------------
    # 3. Linear projections (no bias)
    # --------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device, dtype=torch.float16)   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(device, dtype=torch.float16)

    left = F.linear(x_fp16, left_proj_w)    # [B,N,N,hidden_dim]
    right = F.linear(x_fp16, right_proj_w)  # [B,N,N,hidden_dim]

    # --------------------------------------------------------------------
    # 4. Gating (sigmoid) + optional mask (fused via Triton)
    # --------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device, dtype=torch.float16)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))    # [B,N,N,hidden_dim]

    # Prepare mask (broadcast over hidden_dim)
    if mask_tensor is None:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(device).to(torch.float16)

    # Flatten tensors for the kernel
    left_flat = left.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_flat = right.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)
    mask_flat = mask_fp16.contiguous().view(-1)

    # Allocate output buffers
    left_out_flat = torch.empty_like(left_flat)
    right_out_flat = torch.empty_like(right_flat)

    # Triton launch configuration
    BLOCK = 1024
    total = left_flat.numel()
    grid = (triton.cdiv(total, BLOCK),)

    # Fuse projection * gate * mask for the left branch
    mul_gate_mask_kernel[grid](
        left_out_flat,
        left_flat,
        left_gate_flat,
        mask_flat,
        total,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
    )
    # Fuse projection * gate * mask for the right branch
    mul_gate_mask_kernel[grid](
        right_out_flat,
        right_flat,
        right_gate_flat,
        mask_flat,
        total,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
    )

    # Reshape back to the original 4‑D layout
    left = left_out_flat.view(B, N, N, hidden_dim)
    right = right_out_flat.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------------
    # 5. Cubic contraction via batched GEMM (torch.bmm = cuBLAS)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------------
    # Rearrange to (batch*hidden_dim, N, N) where the K‑dimension is the 3rd axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden_dim, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, hidden_dim, N, N) with k as rows
    LH = B * hidden_dim
    left_mat = left_perm.view(LH, N, N)   # (B*hidden_dim, N, N)
    right_mat = right_perm.view(LH, N, N) # (B*hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat)  # (B*hidden_dim, N, N)

    # Restore original layout [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------------
    # 6. Post‑processing (LayerNorm on hidden dim, final gate, output proj)
    # --------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype=torch.float32)
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply the final gating
    out_norm = out_norm * out_gate.to(torch.float32)

    # Final linear projection back to the original channel dimension
    to_out_w = weights["to_out.weight"].to(device, dtype=torch.float32)  # (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)  # [B, N, N, dim]

    return output
