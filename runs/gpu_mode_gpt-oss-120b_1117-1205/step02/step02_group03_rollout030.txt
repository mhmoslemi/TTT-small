@triton.jit
def apply_gate_mask_kernel(
    left_ptr, left_gate_ptr, mask_ptr,
    out_left_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    BLOCK_H: tl.constexpr, NOMASK: tl.constexpr
):
    b = tl.program_id(0)
    i = tl.program_id(1)
    j = tl.program_id(2)

    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # Compute base offsets for this (b,i,j)
    left_base = left_ptr + b*stride_left_b + i*stride_left_i + j*stride_left_k
    left_gate_base = left_gate_ptr + b*stride_gate_b + i*stride_gate_i + j*stride_gate_k
    out_left_base = out_left_ptr + b*stride_left_b + i*stride_left_i + j*stride_left_k

    # Load mask scalar (broadcast across H)
    mask_val_ptr = mask_ptr + b*stride_mask_b + i*stride_mask_i + j*stride_mask_k

    # Compute

    if not NOMASK:

    # Store result
    tl.store(out_left_base + offs_h * stride_left_h, out_left, mask=mask_h)

"""
Optimized outgoing TriMul (AlphaFold3) forward pass.

1. LayerNorm on the input (float32).
2. FP16 linear projections for left/right and their gates.
3. Triton kernel fuses element‑wise mask (if present) and gate multiplication:
      left  = left  * left_gate  * mask
      right = right * right_gate * mask
4. Contract over the sequence dimension using a highly‑optimized batched GEMM
   (torch.bmm) on FP16 tensors—this leverages NVIDIA Tensor‑cores.
5. Output LayerNorm, output gate, and final FP32 linear projection.

Only the element‑wise mask & gate fusion is performed in Triton; the
matrix multiplication uses PyTorch’s cuBLAS implementation which already
achieves near‑peak throughput on the B200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def apply_gate_mask_kernel(
    a_ptr,          # fp16 tensor [B,N,N,H] (in‑place)
    gate_ptr,       # fp16 tensor [B,N,N,H]
    mask_ptr,       # fp16 tensor [B,N,N] (broadcasted on H)
    B, N, H,
    stride_a_b, stride_a_i, stride_a_k, stride_a_h,
    stride_g_b, stride_g_i, stride_g_k, stride_g_h,
    stride_m_b, stride_m_i, stride_m_k,
    BLOCK_H: tl.constexpr, NOMASK: tl.constexpr
):
    """Element‑wise: a = a * gate * (mask if present)"""
    b = tl.program_id(0)
    i = tl.program_id(1)
    k = tl.program_id(2)

    # base pointers for this (b,i,k) slice
    a_base   = a_ptr   + b*stride_a_b + i*stride_a_i + k*stride_a_k
    g_base   = gate_ptr+ b*stride_g_b + i*stride_g_i + k*stride_g_k
    m_base   = mask_ptr+ b*stride_m_b + i*stride_m_i + k*stride_m_k

    # Load mask once (scalar) if it is used
    if NOMASK == 0:
        mask_val = tl.load(m_base)
    else:
        mask_val = 1.0

    # Process the hidden dimension in tiles of BLOCK_H
    for h_offset in range(0, H, BLOCK_H):
        offs_h = h_offset + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        a_ptrs = a_base + offs_h * stride_a_h
        g_ptrs = g_base + offs_h * stride_g_h

        a = tl.load(a_ptrs, mask=mask_h, other=0.0)
        g = tl.load(g_ptrs, mask=mask_h, other=0.0)

        out = a * g * mask_val
        tl.store(a_ptrs, out, mask=mask_h)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool / numeric)
            - weights      : dict of model parameters
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim] float32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the heavy path
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (fp16, no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 4. Gate values (fp16)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate   = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 5. Fuse mask and gate multiplication with a Triton kernel
    # ------------------------------------------------------------------
    # Ensure contiguity (required for correct stride handling)
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_fp16 = mask.to(torch.float16).contiguous()

    B, N, _, _ = left.shape
    BLOCK_H = 128  # tile size over the hidden dimension

    # launch kernel for left
    grid = (B, N, N)
    apply_gate_mask_kernel[grid](
        left,
        left_gate,
        mask_fp16,
        B, N, hidden_dim,
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
        BLOCK_H=BLOCK_H,
        NOMASK=int(nomask),
    )
    # launch kernel for right
    apply_gate_mask_kernel[grid](
        right,
        right_gate,
        mask_fp16,
        B, N, hidden_dim,
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
        BLOCK_H=BLOCK_H,
        NOMASK=int(nomask),
    )

    # ------------------------------------------------------------------
    # 6. Core contraction: batched GEMM using torch.bmm (fp16 -> fp32 acc)
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) where the last two dims are (i,k) / (k,j)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N) : i,k
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) : k,j

    B, H, N, _ = left_perm.shape
    left_flat  = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N)  # (B*H, N, N)

    # Batched matrix multiplication; uses Tensor‑cores internally
    out_flat = torch.bmm(left_flat, right_flat)   # (B*H, N, N), FP16 input → FP32 acc

    # Restore [B, N, N, H] layout and convert back to FP32 for the rest
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, gating, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
