@triton.jit
def triton_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ah,    # A strides: batch (H), M, K? Actually A shape: (batch, M, K) -> strides: stride_ah (batch), stride_am (M) and stride_ak (K)
    stride_bk, stride_bn, stride_bh,
    stride_cm, stride_cn, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # The kernel processes a single batch idx = program_id(0). This is total_batch dimension (B*H)
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Compute base pointers for this batch
    a_batch = a_ptr + batch_id * stride_ah
    b_batch = b_ptr + batch_id * stride_bh
    c_batch = c_ptr + batch_id * stride_ch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0) # shape (BM, BK)

        # Load B tile - note B shape is (K,N)
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0) # shape (BK, BN)

        # Compute dot product
        # tl.dot expects two tensors of same shape? Actually tl.dot(A, B) where A.shape = (M, K), B.shape = (K, N) returns (M, N) accumulate.
        # In triton language, tl.dot expects two 2-D matrices with inner dimension equal. So we can do acc += tl.dot(a, b)
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def triton_batched_gemm(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    # Ensure contiguous and dtype = torch.float16
    assert A.dtype == torch.float16 and B.dtype == torch.float16, "GEMM inputs must be half"
    assert A.is_contiguous() and B.is_contiguous(), "Inputs must be contiguous"
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides (in elements)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bk, stride_bn = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (batch, triton.cdiv(M, BLOCK_M), triton.cdiv(N, BLOCK_N))
    triton_gemm_kernel[grid](
        A, B, C,
        M, N, K,
        stride_am, stride_ak, stride_ah,
        stride_bk, stride_bn, stride_bh,
        stride_cm, stride_cn, stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The heavy O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched GEMM using Triton.
We treat each hidden dimension `d` as a separate batch element, i.e.
    left → (B, H, N, N)  with axes (i,k)
    right → (B, H, N, N)  with axes (k,j)
and compute a batched matrix multiplication for each (b, d) pair.
All other operations (LayerNorms, linear projections, gating) are performed
in PyTorch (mostly in float16) and the final result is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ B   (A: (batch, M, K), B: (batch, K, N))
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ah,   # A: (batch, M, K)
    stride_bk, stride_bn, stride_bh,   # B: (batch, K, N)
    stride_cm, stride_cn, stride_ch,   # C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + batch_id * stride_ah
    b_batch = b_ptr + batch_id * stride_bh
    c_batch = c_ptr + batch_id * stride_ch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load B tile (K x N)
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # Write the result (FP16 output)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def triton_batched_gemm(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B using a Triton kernel.
    A: (batch, M, K)    fp16, contiguous
    B: (batch, K, N)    fp16, contiguous
    Returns C of shape (batch, M, N) fp16.
    """
    assert A.is_contiguous() and B.is_contiguous()
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides in element units
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bk, stride_bn = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    # Tile sizes (tuned for N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_gemm_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_am,
        stride_ak,
        stride_ah,
        stride_bk,
        stride_bn,
        stride_bh,
        stride_cm,
        stride_cn,
        stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (bool/int) – may be ignored.
            - weights_dict: mapping from parameter name to torch.Tensor.
            - config_dict: contains 'dim', 'hidden_dim', and optional 'nomask'.

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Cast to fp16 for the bulk of the work
    x_fp16 = x.to(torch.float16)

    # ----------------------------------------------------------
    # 2. Linear projections (fp16)
    # ----------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # (B, N, N, hidden_dim)
    right = F.linear(x_fp16, right_proj_w)  # (B, N, N, hidden_dim)

    # ----------------------------------------------------------
    # 3. Optional mask (fp16)
    # ----------------------------------------------------------
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)  # (B, N, N, 1)
        left = left * mask_fp
        right = right * mask_fp

    # ----------------------------------------------------------
    # 4. Gates (fp16)
    # ----------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()   # (B, N, N, hidden_dim)

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------
    # 5. Core contraction via Triton batched GEMM
    # ----------------------------------------------------------
    # Rearrange to (B, hidden_dim, N, N) where inner dims are (i,k) and (k,j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) = (B, H, K, J)

    B, H, N, _ = left_perm.shape
    total_batch = B * H

    # Collapse the batch and hidden dimensions for the kernel
    left_flat = left_perm.view(total_batch, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(total_batch, N, N) # (B*H, N, N)

    # Perform the batched matrix multiplication
    out_flat = triton_batched_gemm(left_flat, right_flat)  # fp16

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)   # convert to fp32 for the remaining ops

    # ----------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection
    # ----------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # Apply the output gate (cast to fp32)
    out = out * out_gate.to(out.dtype)

    # Final linear projection back to dim
    out = F.linear(out, weights["to_out.weight"])  # (B, N, N, dim)

    return out
