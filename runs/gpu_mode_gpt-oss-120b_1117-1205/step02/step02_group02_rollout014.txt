import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, n_elements, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < n_elements
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(a_ptr + offs, a * b, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. Input LayerNorm (float32)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim], float32

    # Cast to fp16 for heavy ops
    x_fp16 = x.to(torch.float16)

    # 2. Linear projections
    left = F.linear(
        x_fp16,
        weights["left_proj.weight"].to(torch.float16)
    )  # [B, N, N, hidden_dim]
    right = F.linear(
        x_fp16,
        weights["right_proj.weight"].to(torch.float16)
    )

    # 3. Optional mask
    if not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # 4. Gating
    left_gate = F.linear(
        x_fp16,
        weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = F.linear(
        x_fp16,
        weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = F.linear(
        x_fp16,
        weights["out_gate.weight"].to(torch.float16)

    right = right * right_gate

    # 5. Core contraction via batched GEMM (torch.bmm)
    # permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # transpose last two dims

    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    out_flat = torch.bmm(left_flat, right_flat)  # fp16 result
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden_dim]

    out = out.to(torch.float32)

import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, n_elements, BLOCK: tl.constexpr):
    """In‑place elementwise multiplication a *= b."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < n_elements
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(a_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold3).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool or numeric)
        - weights_dict : mapping parameter name -> torch.Tensor
        - config_dict  : must contain 'dim', 'hidden_dim', optional 'nomask'

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] float32

    # Cast to fp16 for the heavy portion
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (fp16, bias=False)
    # ------------------------------------------------------------------
    left = F.linear(
        x_fp16, weights["left_proj.weight"].to(torch.float16)
    )  # [B, N, N, hidden_dim]
    right = F.linear(
        x_fp16, weights["right_proj.weight"].to(torch.float16)
    )  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4. Gating (fp16)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_fp16, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()  # [B, N, N, hidden_dim]
    right_gate = F.linear(
        x_fp16, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = F.linear(
        x_fp16, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    )

    # ------------------------------------------------------------------
    # 7. Multiply by out_gate using a tiny Triton kernel
    # ------------------------------------------------------------------
    out_gate_f32 = out_gate.to(torch.float32)
    total_elements = out.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_elements, BLOCK),)
    elementwise_mul_kernel[grid](
        out.view(-1), out_gate_f32.view(-1), total_elements, BLOCK=BLOCK
    )  # out is now scaled in‑place

    # ------------------------------------------------------------------
    # 8. Final linear projection back to dim (float32)
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```
