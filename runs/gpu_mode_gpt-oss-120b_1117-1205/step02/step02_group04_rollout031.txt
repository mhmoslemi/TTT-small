"""
Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The heavy O(N³·H) contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is implemented as a single Triton kernel that performs a batched
matrix multiply for all heads (hidden_dim H) without any extra
permute/copy of the tensors.  All other operations (LayerNorm,
Linear projections, gating, final projection) stay in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def out_tri_mul_kernel(
    left_ptr,          # const float16 *
    right_ptr,         # const float16 *
    out_ptr,           # float32 *
    B, N, H,           # int32
    # strides for left  (B, i, k, h)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    # strides for right (B, i, j, h)
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    # strides for out   (B, i, j, h)
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out = left @ right^T   (batched over batch*H)
    left  : (batch, N, N)   -> accessed as [i, k]
    right : (batch, N, N)   -> accessed as [k, j]
    out   : (batch, N, N)   -> [i, j]
    All tensors are laid out as (B, N, N, H) in memory.
    """
    batch_id = tl.program_id(0)   # combined (b * H + h)
    pid_m    = tl.program_id(1)   # i‑tile
    pid_n    = tl.program_id(2)   # j‑tile

    # recover original batch and head indices
    b = batch_id // H
    h = batch_id - b * H

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for the current (b,h) slice
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = kk * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile:   (M, K)  -> rows=i, cols=k
        a_ptrs = left_base + offs_m[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # right tile:  (K, N)  -> rows=k, cols=j
        # we need right[b, j, k, h];  row stride = stride_r_j, col stride = stride_r_i
        b_ptrs = right_base + offs_k[:, None] * stride_r_j + offs_n[None, :] * stride_r_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # FP16 multiply, FP32 accumulate

    # store result
    c_ptrs = out_base + offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using Triton for the
    O(N³) contraction.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor : torch.Tensor of shape [B, N, N] (bool/int) – may be ignored
            - weights_dict: mapping from weight name to torch.Tensor (float32)
            - config_dict : contains keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # -----------------------------------------------------------------
    # 2. Cast to fp16 for the heavy path
    # -----------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections (fp16)
    # -----------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))   # (B,N,N,H)
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16)) # (B,N,N,H)

    # -----------------------------------------------------------------
    # 4. Optional mask (fp16)
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_fp
        right = right * mask_fp

    # -----------------------------------------------------------------
    # 5. Gating (fp16, sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16))
    )
    right_gate = torch.sigmoid(
        F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16))
    )
    out_gate = torch.sigmoid(
        F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16))
    )

    # fuse mask and gate in a single multiplication (in‑place to save memory)
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6. Core O(N³·H) contraction via Triton
    # -----------------------------------------------------------------
    B, N, _, _ = input_tensor.shape  # input shape is [B, N, N, dim]
    out = torch.empty((B, N, N, hidden_dim),
                      dtype=torch.float32,
                      device=device)

    # strides (in elements)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_j, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # tiling parameters – chosen to give good occupancy on B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    total_batch = B * hidden_dim
    grid = (
        total_batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    out_tri_mul_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_j, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 7. Output LayerNorm + gating + final linear (float32)
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(out.dtype)          # out_gate was fp16, up‑cast to fp32
    out = F.linear(out, weights["to_out.weight"])  # (B,N,N,dim)

    return out
