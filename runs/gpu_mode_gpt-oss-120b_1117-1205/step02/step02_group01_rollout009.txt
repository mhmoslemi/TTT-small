"""
TriMul (outgoing) forward pass – Triton‑accelerated elementwise mask‑gate fusion.

Algorithm
---------
1. Layer‑norm the input (FP16).
2. Fuse the two linear projections per side (proj + gate) by concatenating the
   weights, then split the result and apply the sigmoid to the gate part.
3. Fuse elementwise multiplication with the (optional) mask and gate using a
   small Triton kernel for the left and right branches.
4. Perform the cubic contraction via a batched GEMM:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is done with a single `torch.bmm` on tensors of shape
   (B*hidden, N, N).
5. Apply a second layer‑norm, an output gate, and the final linear projection.
6. Return the result in float32.

The Triton kernel only handles the elementwise product
    out = tensor * gate * mask
which reduces Python launch overhead and keeps the heavy compute on
high‑throughput cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def mul_gate_mask_kernel(
    out_ptr,    # ptr to output (flattened)
    a_ptr,      # ptr to tensor (flattened)
    gate_ptr,   # ptr to gate (flattened)
    mask_ptr,   # ptr to mask (flattened, broadcast over hidden dim)
    total,      # total number of elements (int64)
    BLOCK_SIZE: tl.constexpr,   # block size
    H: tl.constexpr,            # hidden dimension (size of innermost axis)
):
    """
    out = a * gate * mask
    mask is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    m = offs < total

    a = tl.load(a_ptr + offs, mask=m)
    g = tl.load(gate_ptr + offs, mask=m)

    # Compute mask index (broadcast over hidden dim)
    mask_idx = offs // H
    mask_val = tl.load(mask_ptr + mask_idx, mask=m)

    tl.store(out_ptr + offs, a * g * mask_val, mask=m)


def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor: torch.Tensor [B, N, N, dim] (float32)
            mask_tensor : torch.Tensor [B, N, N] (float32) or None
            weights_dict: model weights (see problem statement)
            config_dict : {"dim": int, "hidden_dim": int}
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------------
    # 1. Input layer‑norm (FP16)
    # --------------------------------------------------------------------
    x = input_tensor.to(torch.float16)

    norm_w = weights["norm.weight"].to(device).to(torch.float16)
    norm_b = weights["norm.bias"].to(device).to(torch.float16)
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------------
    # 2. Fused linear projections + gates for left and right branches
    # --------------------------------------------------------------------
    # Concatenate proj and gate weights (out_features first)
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device).to(torch.float16)          # (2*hidden, dim)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device).to(torch.float16)         # (2*hidden, dim)

    # Linear passes (output shape: [B,N,N,2*hidden])
    left_comb = F.linear(x, left_comb_w)    # no bias
    right_comb = F.linear(x, right_comb_w)

    # Split into proj and gate, apply sigmoid to gates
    left_proj = left_comb[..., :hidden_dim]
    left_gate = torch.sigmoid(left_comb[..., hidden_dim:])

    right_proj = right_comb[..., :hidden_dim]
    right_gate = torch.sigmoid(right_comb[..., hidden_dim:])

    # Release temporary tensors early
    del left_comb, right_comb

    # --------------------------------------------------------------------
    # 3. Out‑gate (independent of mask)
    # --------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------------
    # 4. Prepare mask (broadcast over hidden dim)
    # --------------------------------------------------------------------
    if mask_tensor is None:
        mask = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask = mask_tensor.to(device).to(torch.float16)

    # --------------------------------------------------------------------
    # 5. Triton fusion: tensor * gate * mask  (in‑place on proj tensors)
    # --------------------------------------------------------------------
    total_elems = left_proj.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_elems, BLOCK),)

    # left branch
    mul_gate_mask_kernel[grid](
        left_proj,            # out (in‑place)
        left_proj,            # a
        left_gate,            # gate
        mask,                 # mask
        total_elems,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
    )
    # right branch
    mul_gate_mask_kernel[grid](
        right_proj,
        right_proj,
        right_gate,
        mask,
        total_elems,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
    )

    # Gates no longer needed
    del left_gate, right_gate, mask

    # --------------------------------------------------------------------
    # 6. Cubic contraction via batched GEMM (torch.bmm)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------------
    # Permute to bring hidden dim to batch axis
    left_perm = left_proj.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_proj.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)

    batch_h = B * hidden_dim
    left_mat = left_perm.view(batch_h, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(batch_h, N, N) # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N), FP16

    # Restore original layout [B,N,N,hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------
    # 7. Post‑processing: layer‑norm, output gate, final linear
    # --------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate (FP16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)  # (dim, hidden_dim)
    out = F.linear(out, to_out_w)      # [B, N, N, dim] in FP16

    # --------------------------------------------------------------------
    # 8. Cast back to FP32 (as required by the interface)
    # --------------------------------------------------------------------
    return out.to(torch.float32)
