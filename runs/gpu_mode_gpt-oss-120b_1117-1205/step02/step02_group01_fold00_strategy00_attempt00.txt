        x_float = x.float()
    else:
        x_float = x.to(torch.float32, copy=True)
    # Perform normalization using nn.functional layer norm
    x = F.layer_norm(x_float, (dim,), weight=norm_w, bias=norm_b).to(torch.float32, copy=True)
    
    # Projection onto hidden space
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)
    
    # Two separate projections: one to hidden_dim
    left = F.linear(x, left_proj_weight)
    right = F.linear(x, right_proj_weight)
    
    left = left.permute(0, 2, 1, 3)
    right = right.permute(0, 2, 3, 1)
    
    # Transpose the right tensor for matrix multiplication
    right = right.contiguous().view(bs * seq_len, seq_len, hidden_dim)
    right = right.permute(0, 2, 1)

    # Define grid and launch the kernel
    BLOCK_SIZE = 128
    num_blocks = triton.cdiv(seq_len * seq_len * hidden_dim, BLOCK_SIZE)
    grid = [num_blocks]
    
    # We assume the output tensor to be [B, seq_len, seq_len, hidden_dim] for the contraction
    out = torch.zeros(bs, seq_len, seq_len, hidden_dim, dtype=torch.float32, device=device)
    out_ptr = out.view(-1).contiguous()
    left = left.permute(0, 2, 1, 3).contiguous().view(-1, hidden_dim)
    right = right.view(-1, hidden_dim).contiguous()
    
    triangle_multiplicative_update_kernel[grid](
        left_ptr, right_ptr, out_ptr, 
        seq_len=seq_len, hidden_dim=hidden_dim, 
        BLOCK_SIZE=BLOCK_SIZE
    )

    # Move to the correct layout: [B, N, N, hidden_dim]
    out = out.permute(0, 2, 1, 3).contiguous()

    # Postprocessing: LayerNorm on hidden_dim and final projection
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    to_out_weight = weights["to_out.weight"].to(torch.float16)

    # LayerNorm on the hidden_dim
    out_norm = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    
    # Apply the gating
    out = out * out_norm.gate(1.0)

    # Final projection back to input dimensions
    output_weight = weights["to_out.weight"].to(torch.float16)
    output_proj = F.linear(out, output_weight)

    return output_proj
