def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Input layernorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Convert to fp16 for projections
    x_fp16 = x.to(torch.float16)

    # Linear projections
    left = torch.nn.functional.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = torch.nn.functional.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # Apply mask if present
    if not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp
        right = right * mask_fp

    # Gates
    left_gate = torch.nn.functional.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Core contraction via Triton batched GEMM
    # permute so that we can treat hidden dim as batch dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) = (B, H, K, J)

    B_, H_, N_, _ = left_perm.shape
    total_batch = B_ * H_
    left_flat = left_perm.view(total_batch, N_, N_)
    right_flat = right_perm.view(total_batch, N_, N_)

    # Run kernel
    out_flat = triton_batched_gemm(left_flat, right_flat)

    # Reshape back
    out = out_flat.view(B_, H_, N_, N_).permute(0, 2, 3, 1).contiguous()
    del left, right, left_perm, right_perm, left_flat, right_flat

    # Output LayerNorm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # Apply out_gate
    out = out * out_gate

    # Final projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```

    batch, M, K = A.shape
    _, K2, N = B.shape

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    stride_bh, stride_bk, stride_bn = B.stride()

"""
Highly‑optimized Triton implementation of the outgoing Triangle Multiplicative
Update (TriMul) forward pass.

The costly contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is realized as a batched matrix multiplication
    (left) @ (right)ᵀ
for every hidden dimension `d`.  The hidden dimension is flattened into the
batch axis (B·H) so a single Triton kernel can compute all independent GEMMs.
All other operations (LayerNorm, linear projections, gating, final projection)
are performed with PyTorch in FP16/FP32 mixed precision.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ B   (A: (B, M, K), B: (B, K, N))
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix dims
    stride_am, stride_ak, stride_ah,   # strides for A
    stride_bk, stride_bn, stride_bh,   # strides for B
    stride_cm, stride_cn, stride_ch,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ B with FP16 inputs, FP32 accumulation."""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + batch_id * stride_ah
    b_batch = b_ptr + batch_id * stride_bh
    c_batch = c_ptr + batch_id * stride_ch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k < K),
            other=0.0,
        )

        # Load B tile (K x N)
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(
            b_ptrs,
            mask=(offs_k < K)[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Accumulate
        acc += tl.dot(a, b)

    # Store result (output dtype FP16)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def triton_batched_gemm(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B using Triton.

    A: (batch, M, K)  – fp16, contiguous
    B: (batch, K, N)  – fp16, contiguous
    Returns C: (batch, M, N) – fp16
    """
    assert A.is_contiguous() and B.is_contiguous()
    assert A.dtype == B.dtype == torch.float16

    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides in elements
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bk, stride_bn = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    # Tile sizes – chosen to fit in registers for N up to 1024
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_gemm_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_am,
        stride_ak,
        stride_ah,
        stride_bk,
        stride_bn,
        stride_bh,
        stride_cm,
        stride_cn,
        stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/int) – may be ignored if nomask=True
            - weights_dict: mapping from parameter names to tensors
            - config_dict: contains 'dim', 'hidden_dim', and optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the heavy work
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 4. Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 5. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Core contraction via Triton (batched GEMM)
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    # Bring hidden dimension into the batch axis.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) == (B, H, K, J)

    B_, H_, N_, _ = left_perm.shape
    total_batch = B_ * H_

    left_flat = left_perm.view(total_batch, N_, N_)
    right_flat = right_perm.view(total_batch, N_, N_)

    out_flat = triton_batched_gemm(left_flat, right_flat)   # (total_batch, N, N), fp16

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B_, H_, N_, N_).permute(0, 2, 3, 1).contiguous()
    del left, right, left_perm, right_perm, left_flat, right_flat, out_flat  # free memory

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # out_gate is fp16; cast to match out (float32)
    out = out * out_gate.to(out.dtype)

    # Final linear projection back to original dimension
    out = F.linear(out, weights["to_out.weight"])

    return out
