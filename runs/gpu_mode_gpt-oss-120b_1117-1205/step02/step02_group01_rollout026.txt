def fuse_gate_mask_kernel(
    left_out_ptr,
    right_out_ptr,
    left_ptr,
    right_ptr,
    left_gate_ptr,
    right_gate_ptr,
    mask_ptr,
    total_elements,
    hidden_dim: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    right_res = r * rg * m

    tl.store(left_out_ptr + offs, left_res, mask=mask)

"""
Optimized TriMul (outgoing) forward pass.

Algorithm
---------
1. Input LayerNorm (FP32) → x  [B, N, N, dim]
2. Cast to FP16 for heavy work.
3. Linear projections (left/right) → [B,N,N,hidden]
4. Compute gating (sigmoid(linear(x))) for left/right/out.
5. Fuse element‑wise mask, left‑gate, right‑gate multiplication
   in a single Triton kernel, producing left_out and right_out.
6. Cubic contraction:
   out[b,i,j,d] = Σ_k left_out[b,i,k,d] * right_out[b,j,k,d]
   Implemented as batched GEMM (torch.bmm) on shape (B*hidden, N, N).
7. Post‑processing: LayerNorm over hidden dim, multiply by out‑gate,
   final linear projection back to dim.
All heavy linear / GEMM work remains in cuBLAS (highly optimized);
the only Triton kernel fuses three element‑wise operations, reducing
kernel launch overhead and memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def fuse_gate_mask_kernel(
    left_out_ptr,          # [total] fp16
    right_out_ptr,         # [total] fp16
    left_ptr,              # [total] fp16
    right_ptr,             # [total] fp16
    left_gate_ptr,         # [total] fp16
    right_gate_ptr,        # [total] fp16
    mask_ptr,              # [B*N*N] fp16
    total_elements,        # i64 scalar
    hidden_dim: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load inputs
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Compute mask index (broadcast over hidden dim)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Fuse multiplications
    left_res = l * lg * m
    right_res = r * rg * m

    # Store results
    tl.store(left_out_ptr + offs, left_res, mask=mask)
    tl.store(right_out_ptr + offs, right_res, mask=mask)


def custom_kernel(data):
    """
    TriMul (outgoing) forward pass with a Triton‑accelerated
    elementwise mask‑gate fusion.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor: torch.Tensor [B, N, N, dim] (float32)
            mask_tensor : torch.Tensor [B, N, N] (float32) or None
            weights_dict : mapping of weight names to tensors (float32)
            config_dict  : {"dim": int, "hidden_dim": int}

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # [B,N,N,dim] FP32

    # --------------------------------------------------------------------
    # 2. Cast to FP16 for the heavy path
    # --------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------------
    # 3. Linear projections (no bias) → left/right (FP16)
    # --------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # [B,N,N,hidden]
    right = F.linear(x_fp16, right_proj_w) # [B,N,N,hidden]

    # --------------------------------------------------------------------
    # 4. Gating (sigmoid) (FP16)
    # --------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w)) # [B,N,N,hidden]
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))    # [B,N,N,hidden]

    # --------------------------------------------------------------------
    # 5. Prepare mask (FP16). If None → ones.
    # --------------------------------------------------------------------
    if mask_tensor is None:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(torch.float16).to(device)

    # --------------------------------------------------------------------
    # 6. Fuse mask & gate multiplications via Triton (single kernel)
    # --------------------------------------------------------------------
    total = left.numel()  # B * N * N * hidden
    left_flat = left.contiguous().view(-1)
    right_flat = right.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)
    mask_flat = mask_fp16.contiguous().view(-1)  # length B*N*N

    left_out_flat = torch.empty_like(left_flat)
    right_out_flat = torch.empty_like(right_flat)

    BLOCK = 16384  # number of elements processed per program
    grid = (triton.cdiv(total, BLOCK),)

    fuse_gate_mask_kernel[grid](
        left_out_flat,
        right_out_flat,
        left_flat,
        right_flat,
        left_gate_flat,
        right_gate_flat,
        mask_flat,
        total,
        hidden_dim=hidden_dim,
        BLOCK_SIZE=BLOCK,
    )

    # Reshape back to [B, N, N, hidden]
    left = left_out_flat.view(B, N, N, hidden_dim)
    right = right_out_flat.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------------
    # 7. Cubic contraction using batched GEMM (FP16)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------------
    # Rearrange tensors to (B*hidden, N, N) for bmm
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, hidden, N, N) (k as rows)

    LH = B * hidden_dim
    left_mat = left_perm.view(LH, N, N)   # (B*hidden, N, N)
    right_mat = right_perm.view(LH, N, N) # (B*hidden, N, N)

    out_mat = torch.bmm(left_mat, right_mat)   # (B*hidden, N, N) FP16

    # Restore [B, N, N, hidden]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # FP16

    # --------------------------------------------------------------------
    # 8. Post‑processing: LayerNorm over hidden dim, final gate, linear proj
    # --------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply out_gate (converted to FP32)
    out_norm = out_norm * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"].to(device)  # (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)  # [B,N,N,dim] FP32

    return output
