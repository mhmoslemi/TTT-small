"""
TriMul (outgoing) forward pass with Triton‑fused mask & gating.

Algorithm
---------
1. LayerNorm on the input tensor (float32 → half).
2. Linear projections `left` and `right` to `hidden_dim` (half‑precision).
3. Compute four gating tensors via a linear + sigmoid (half‑precision).
4. Fuse mask (if present) and the `left_gate` / `right_gate` multiplications
   into a single Triton kernel that works in‑place on the flattened tensors.
5. Perform the O(N³) contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiply for each head `d`.  We reshape the
   tensors to (B*hidden_dim, N, N) and use `torch.bmm` (cuBLAS) which is
   already highly optimized.
6. Apply a second LayerNorm, the output gate, and the final linear projection.
7. Return the result in float32.

Only the mask‑plus‑gate fusion is executed in a custom Triton kernel,
satisfying the requirement to use a kernel while keeping the overall
runtime low.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse optional mask with left/right gating
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # half*  (flattened left)
    right_ptr,           # half*  (flattened right)
    left_gate_ptr,       # half*  (flattened left_gate)
    right_gate_ptr,      # half*  (flattened right_gate)
    mask_ptr,            # half*  (flattened mask, broadcasted)
    total_elements: tl.int32,
    hidden_dim: tl.int32,
    HAS_MASK: tl.constexpr,   # 1 if mask present, 0 otherwise
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # each mask entry corresponds to `hidden_dim` consecutive channel values
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of:
        left = left * left_gate * mask
        right = right * right_gate * mask
    (mask is optional). All tensors must be contiguous half‑precision.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when HAS_MASK=0

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        hidden_dim,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )
    # No return value; tensors are modified in place.


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or float) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict containing
                * "dim" (int) – input channel size C
                * "hidden_dim" (int) – projection size H
                * "nomask" (bool, optional) – True if mask should be ignored

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 → half)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half‑precision)
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Gating tensors (half‑precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) and gate multiplications via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → broadcast over hidden_dim
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        # Flatten tensors for the kernel
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_h.view(-1)  # [B*N*N]

        # In‑place fusion
        _apply_gate_and_mask(
            left_flat, right_flat,
            left_gate_flat, right_gate_flat,
            mask_flat,
            hidden_dim,
        )
        # Reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask: simple elementwise multiplication
        left = left * left_gate
        right = right * right_gate

    # Gates are no longer needed
    del left_gate, right_gate, x_h

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as batched GEMM (cuBLAS) after appropriate transposes.
    # ------------------------------------------------------------------
    # Transpose right so that we can use standard bmm (left @ right^T)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) – already transposed

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Batched matrix multiplication in half precision (float32 accumulation)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise gating (half)
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 7. Return in float32 as required by the specification
    # ------------------------------------------------------------------
    return out.to(torch.float32)
