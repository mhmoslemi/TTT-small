def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    
    # LayerNorm (float32)
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    
    # Cast to fp16
    x_fp16 = x.to(torch.float16)
    
    # Projection weights to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    
    # Linear projections
    left = F.linear(x_fp16, left_proj_w)
    right = F.linear(x_fp16, right_proj_w)
    
    if not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp
        right = right * mask_fp
    
    # Gating
    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()
    
    left = left * left_gate
    right = right * right_gate
    
    # Permute for batched GEMM
    # shape: (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()
    
    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)
    
    # Choose block sizes
    if N <= 256:
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 32
    else:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 32
    
    out_flat = triton_matmul(left_flat, right_flat,
                             BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K)
    
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)
    
    # Output layernorm and final projection
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])
    
    out = out * out_gate.to(torch.float32)
    
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))
    
    return out

def triton_matmul(a, b, BLOCK_M=128, BLOCK_N=128, BLOCK_K=32):
    assert a.shape[0] == b.shape[0], "batch dimension mismatch"
    B_total, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2
    c = torch.empty((B_total, M, N), dtype=torch.float16, device=a.device)
    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_k, stride_b_n = b.stride()
    stride_c_batch, stride_c_m, stride_c_n = c.stride()
    
    grid = (B_total, triton.cdiv(M, BLOCK_M), triton.cdiv(N, BLOCK_N))
    batched_matmul_kernel[grid](
        a, b, c,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c

"""
TriMul (outgoing) forward pass with a fused Triton batched GEMM.

Algorithm
---------
1. Layer‑norm over the feature dimension (float32).
2. Cast the normalized tensor to fp16 and apply three linear projections:
   * left  = x @ left_proj    (hidden_dim)
   * right = x @ right_proj   (hidden_dim)
3. If a mask is provided, multiply left/right by mask (broadcast on hidden dim).
4. Compute three sigmoidal gates (left‑gate, right‑gate, out‑gate) from the
   original fp16 tensor and apply the first two gates element‑wise.
5. The core contraction is a batched matrix multiplication:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which can be seen as independent GEMM’s for each (batch, hidden) pair.
   We rearrange the tensors to shape (B·H, N, N) and launch a custom Triton
   kernel that computes C = A @ Bᵀ (fp16 inputs, fp32 accumulation, fp16 output).
6. Reshape the result back to [B, N, N, hidden_dim] and convert to fp32.
7. Apply output LayerNorm, multiply by out‑gate and a final linear projection
   back to the original feature dimension.

Only the heavy contraction (step 5) is implemented in Triton; all other
operations are performed with PyTorch functional calls.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched GEMM: C = A @ Bᵀ
    A : (batch, M, K)   fp16
    B : (batch, K, N)   fp16  (already transposed, i.e. original Bᵀ)
    C : (batch, M, N)   fp16 (fp32 accumulation)
    """
    batch_id = tl.program_id(0)               # batch dimension (B·H)
    pid_m = tl.program_id(1)                  # block row index
    pid_n = tl.program_id(2)                  # block column index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [BLOCK_N]
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the current batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        # Load B tile (K x N) – B is already transposed
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    # Store C tile
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def triton_matmul(a: torch.Tensor,
                  b: torch.Tensor,
                  *,
                  BLOCK_M: int = 128,
                  BLOCK_N: int = 128,
                  BLOCK_K: int = 32) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ Bᵀ using the Triton kernel above.
    Arguments:
        a : Tensor of shape (B, M, K), dtype float16, contiguous.
        b : Tensor of shape (B, K, N), dtype float16, contiguous.
    Returns:
        c : Tensor of shape (B, M, N), dtype float16, on the same device.
    """
    assert a.is_contiguous() and b.is_contiguous()
    assert a.dtype == b.dtype == torch.float16
    B_total, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    c = torch.empty((B_total, M, N), dtype=torch.float16, device=a.device)

    # Strides in element units
    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_k, stride_b_n = b.stride()
    stride_c_batch, stride_c_m, stride_c_n = c.stride()

    grid = (
        B_total,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        a, b, c,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool or numeric)
            - weights_dict : dict with the model parameters
            - config_dict  : must contain 'dim', 'hidden_dim', optional 'nomask'
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim], float32

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for the heavy linear ops
    # -----------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections (fp16)
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x_fp16, right_proj_w)  # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 4. Optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # -----------------------------------------------------------------
    # 5. Gating (fp16) and apply
    # -----------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6. Core contraction: batched GEMM via Triton
    #    Compute out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Equivalent to (B·H) independent matmuls of shape (N,N) @ (N,N)ᵀ.
    # -----------------------------------------------------------------
    # Rearrange to (batch, hidden, N, N) where the last two dims are (i,k) and (k,j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, k, j)

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)    # (B·H, N, K)  with K == N
    right_flat = right_perm.view(B * H, N, N)  # (B·H, K, N)

    # Choose block sizes based on sequence length to keep register usage reasonable
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32

    out_flat = triton_matmul(
        left_flat,
        right_flat,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )  # (B·H, N, N), fp16

    # Restore original layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)
    out = out.to(torch.float32)

    # -----------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final projection (float32)
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
