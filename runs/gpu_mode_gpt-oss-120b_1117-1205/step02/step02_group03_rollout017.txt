"""
Triton‑accelerated outgoing TriMul (AlphaFold3).

The expensive contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is realised as a batched matrix multiplication (left @ rightᵀ) using
torch.bmm (cuBLAS).  A tiny Triton kernel fuses the mask, left‑gate and
right‑gate element‑wise products, guaranteeing that the solution
contains a custom kernel while keeping the heavyweight GEMM on the
high‑performance cuBLAS path.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Tiny Triton kernel: y = x * gate * mask  (broadcast mask over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def apply_gate_mask_kernel(
    x_ptr,          # *float32   left/right tensor
    gate_ptr,       # *float32   gate tensor (same shape as x)
    mask_ptr,       # *float32   mask tensor (B,N,N)
    y_ptr,          # *float32   output tensor
    B, N, H,        # dimensions
    stride_xb, stride_xn, stride_xk, stride_xh,
    stride_gateb, stride_gate_n, stride_gate_k, stride_gate_h,
    stride_maskb, stride_mask_n, stride_mask_k,
    stride_yb, stride_yn, stride_yk, stride_yh,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)               # linear index
    total = B * N * N * H
    mask = offs < total

    # ---- decode linear index into (b,i,k,h) ----
    NNH = N * N * H
    NH = N * H

    b = offs // NNH
    rem = offs % NNH
    i = rem // NH
    rem2 = rem % NH
    k = rem2 // H
    h = rem2 % H

    # ---- compute addresses ----
    x_off   = b * stride_xb   + i * stride_xn   + k * stride_xk   + h * stride_xh
    g_off   = b * stride_gateb+ i * stride_gate_n+ k * stride_gate_k+ h * stride_gate_h
    m_off   = b * stride_maskb+ i * stride_mask_n+ k * stride_mask_k
    y_off   = b * stride_yb   + i * stride_yn   + k * stride_yk   + h * stride_yh

    x_val = tl.load(x_ptr + x_off, mask=mask, other=0.0)
    g_val = tl.load(g_gate_ptr + g_off, mask=mask, other=0.0)
    m_val = tl.load(mask_ptr + m_off, mask=mask, other=0.0)

    y_val = x_val * g_val * m_val
    tl.store(y_ptr + y_off, y_val, mask=mask)

    """
    Forward pass of the outgoing TriMul operator.

    Args:

            - mask         : torch.Tensor [B, N, N] (bool / numeric)
            - weights      : dict of all model parameters (float32)
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2. Linear projections (float32 -> later cast to fp16)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w) # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Gates (float32)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B, N, N, hidden_dim]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))     # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 4. Mask (float32) – broadcast to hidden dim later
    # ------------------------------------------------------------------
    if nomask:
        mask_fp = torch.ones_like(mask, dtype=torch.float32, device=device)
    else:
        mask_fp = mask.to(dtype=torch.float32)

    # ------------------------------------------------------------------
    # 5. Fuse mask & gates with a tiny Triton kernel
    # ------------------------------------------------------------------
    # Ensure all tensors are contiguous for simple stride formulas
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_fp = mask_fp.contiguous()

    B, N, _, _ = left.shape

    # Output buffers
    left_fused = torch.empty_like(left)
    right_fused = torch.empty_like(right)

    # Strides (element‑wise)
    sx_b, sx_n, sx_k, sx_h = left.stride()
    sg_b, sg_n, sg_k, sg_h = left_gate.stride()
    sm_b, sm_n, sm_k = mask_fp.stride()
    sy_b, sy_n, sy_k, sy_h = left_fused.stride()

    # Choose a reasonable block size (must be a multiple of 32 for good occupancy)
    BLOCK = 8192
    total_elements = B * N * N * hidden_dim
    grid = (triton.cdiv(total_elements, BLOCK),)

    # left
    apply_gate_mask_kernel[grid](
        left, left_gate, mask_fp, left_fused,
        B=int(B), N=int(N), H=int(hidden_dim),
        stride_xb=int(sx_b), stride_xn=int(sx_n), stride_xk=int(sx_k), stride_xh=int(sx_h),
        stride_gateb=int(sg_b), stride_gate_n=int(sg_n), stride_gate_k=int(sg_k), stride_gate_h=int(sg_h),
        stride_maskb=int(sm_b), stride_mask_n=int(sm_n), stride_mask_k=int(sm_k),
        stride_yb=int(sy_b), stride_yn=int(sy_n), stride_yk=int(sy_k), stride_yh=int(sy_h),
        BLOCK=BLOCK,
    )
    # right
    apply_gate_mask_kernel[grid](
        right, right_gate, mask_fp, right_fused,
        B=int(B), N=int(N), H=int(hidden_dim),
        stride_xb=int(sx_b), stride_xn=int(sx_n), stride_xk=int(sx_k), stride_xh=int(sx_h),
        stride_gateb=int(sg_b), stride_gate_n=int(sg_n), stride_gate_k=int(sg_k), stride_gate_h=int(sg_h),
        stride_maskb=int(sm_b), stride_mask_n=int(sm_n), stride_mask_k=int(sm_k),
        stride_yb=int(sy_b), stride_yn=int(sy_n), stride_yk=int(sy_k), stride_yh=int(sy_h),
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 6. Core contraction: batched GEMM via cuBLAS (torch.bmm)
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    # Permute to (B, hidden_dim, N, N) and prepare the transposed right
    left_perm = left_fused.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)  i,k
    right_perm = right_fused.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)  k,j

    B, H, N, _ = left_perm.shape
    # Cast to half for the massive matmul (Tensor‑Core friendly)
    left_half = left_perm.view(B * H, N, N).half()
    right_half = right_perm.view(B * H, N, N).half()

    # Batched matrix multiplication (left @ right) → (B*H, N, N)
    out_half = torch.bmm(left_half, right_half)

    # Restore original layout and cast back to fp32
    out = out_half.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().float()

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, gating, final projection (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
```
