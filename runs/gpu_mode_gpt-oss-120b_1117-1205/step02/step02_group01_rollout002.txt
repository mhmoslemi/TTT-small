def fuse_gate_mask_kernel(
    left_out_ptr, right_out_ptr,
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total,
    BLOCK_SIZE: tl.constexpr,
    H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total
    # Load
    left = tl.load(left_ptr + offs, mask=mask)
    right = tl.load(right_ptr + offs, mask=mask)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask)

    left_out = left * left_gate * mask_val
    right_out = right * right_gate * mask_val
    tl.store(left_out_ptr + offs, left_out, mask=mask)

All linear projections, layer‑norms and the final output projection are
executed in PyTorch (FP16 where appropriate).  The Triton kernel fuses
the three elementwise multiplications for both left and right branches
into a single launch, reducing kernel overhead and improving bandwidth
utilisation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def fuse_gate_mask_kernel(
    left_out_ptr,        # pointer to left output (flattened)
    right_out_ptr,       # pointer to right output (flattened)
    left_ptr,            # pointer to left tensor (flattened)
    right_ptr,           # pointer to right tensor (flattened)
    left_gate_ptr,       # pointer to left gate tensor (flattened)
    right_gate_ptr,      # pointer to right gate tensor (flattened)
    mask_ptr,            # pointer to mask tensor (flattened, shape B*N*N)
    total,               # total number of elements (B*N*N*H)
    BLOCK_SIZE: tl.constexpr,
    H: tl.constexpr,     # hidden dimension (size of the innermost axis)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    m = offs < total

    # Load tensors
    left = tl.load(left_ptr + offs, mask=m)
    right = tl.load(right_ptr + offs, mask=m)
    left_gate = tl.load(left_gate_ptr + offs, mask=m)
    right_gate = tl.load(right_gate_ptr + offs, mask=m)

    # Broadcast mask over the hidden dimension
    mask_idx = offs // H
    mask_val = tl.load(mask_ptr + mask_idx, mask=m)

    # Fuse elementwise multiplications
    left_out = left * left_gate * mask_val
    right_out = right * right_gate * mask_val

    # Store results
    tl.store(left_out_ptr + offs, left_out, mask=m)
    tl.store(right_out_ptr + offs, right_out, mask=m)


def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor: torch.Tensor [B, N, N, dim] (float32)
            mask_tensor : torch.Tensor [B, N, N]   (float32, may be None)
            weights_dict: dict containing model weights with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            config_dict : {"dim": int, "hidden_dim": int, "nomask": bool}
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # Unpack inputs
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # --------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # [B,N,N,dim] FP32

    # --------------------------------------------------------------------
    # 2. Cast to FP16 for heavy linear work
    # --------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------------
    # 3. Linear projections (no bias)
    # --------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)    # [B,N,N,hidden_dim] FP16
    right = F.linear(x_fp16, right_proj_w)  # [B,N,N,hidden_dim] FP16

    # --------------------------------------------------------------------
    # 4. Gating (sigmoid) and mask handling
    # --------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # [B,N,N,hidden_dim] FP16
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w)) # [B,N,N,hidden_dim] FP16
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))    # [B,N,N,hidden_dim] FP16

    if nomask or mask_tensor is None:
        # No mask => all ones
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(device).to(torch.float16)   # [B,N,N]

    # --------------------------------------------------------------------
    # 5. Fuse mask * gate * projection using Triton
    # --------------------------------------------------------------------
    # Flatten tensors for 1‑D kernel launch
    left_flat = left.contiguous().view(-1)            # (B*N*N*H,)
    right_flat = right.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)
    mask_flat = mask_fp16.contiguous().view(-1)       # (B*N*N,)

    total = left_flat.numel()                         # = B*N*N*H

    left_out_flat = torch.empty_like(left_flat)
    right_out_flat = torch.empty_like(right_flat)

    # Triton launch configuration
    BLOCK = 8192                                     # threads per program
    grid = (triton.cdiv(total, BLOCK),)

    fuse_gate_mask_kernel[grid](
        left_out_flat,
        right_out_flat,
        left_flat,
        right_flat,
        left_gate_flat,
        right_gate_flat,
        mask_flat,
        total,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
    )

    # Reshape back to [B,N,N,hidden_dim]
    left = left_out_flat.view(B, N, N, hidden_dim)
    right = right_out_flat.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------------
    # 6. Cubic contraction via batched GEMM (torch.bmm, cuBLAS)
    # --------------------------------------------------------------------
    # Prepare tensors for batched matmul: (B*H, N, N) matrices
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) - k as rows, j as cols

    LH = B * hidden_dim
    left_mat = left_perm.view(LH, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(LH, N, N) # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N), FP16

    # Restore original layout: [B,N,N,hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # FP16

    # --------------------------------------------------------------------
    # 7. Post‑processing: LayerNorm on hidden dim, final gating, output proj
    # --------------------------------------------------------------------
    # Convert to FP32 for layer norm (better numeric stability)
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float32)
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply final gating (out_gate is FP16, upcast to FP32)
    out_norm = out_norm * out_gate.to(torch.float32)

    # Final linear projection back to original dim
    to_out_w = weights["to_out.weight"].to(device).to(torch.float32)  # (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)  # [B, N, N, dim], FP32

    return output
