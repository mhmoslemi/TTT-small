```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def mul_gate_mask_kernel(out_ptr, a_ptr, gate_ptr, mask_ptr, total,
                         BLOCK_SIZE: tl.constexpr, H: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    a = tl.load(a_ptr + offs, mask=mask)
    g = tl.load(gate_ptr + offs, mask=mask)

    # mask index: flatten over hidden dim H
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    tl.store(out_ptr + offs, a * g * m, mask=mask)
```

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Cast weights to fp16
    # ... retrieve and cast
    # Norm weight bias to fp16
    # Input layernorm:
    norm_w = weights["norm.weight"].to(device).to(torch.float16)
    norm_b = weights["norm.bias"].to(device).to(torch.float16)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # fp16

    x_fp16 = x.to(torch.float16)  # maybe already fp16
    # Projections
    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)

    left = F.linear(x_fp16, left_proj_weight)   # [B,N,N,H] fp16
    right = F.linear(x_fp16, right_proj_weight)

    # Gates
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))

    # Mask
    if mask_tensor is None:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(dtype=torch.float16, device=device)

    # Flatten tensors for Triton kernel
    total = B * N * N * hidden_dim
    left_flat = left.view(-1)
    left_gate_flat = left_gate.view(-1)
    right_flat = right.view(-1)
    right_gate_flat = right_gate.view(-1)
    mask_flat = mask_fp16.view(-1)  # shape B*N*N

    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    # In-place kernel for left
    mul_gate_mask_kernel[grid](
        left_flat,
        left_flat,
        left_gate_flat,
        mask_flat,
        total,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
    )
    # In-place kernel for right
    mul_gate_mask_kernel[grid](
        right_flat,
        right_flat,
        right_gate_flat,
        mask_flat,
        total,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
    )
    # Reshape after scaling (still same shape)
    left = left_flat.view(B, N, N, hidden_dim)
    right = right_flat.view(B, N, N, hidden_dim)

    # Cubic contraction (batched GEMM) using einsum
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)

    # Output layernorm
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out_norm = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply final gate
    out_norm = out_norm * out_gate

    # Final linear projection
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    output = F.linear(out_norm, to_out_w)  # shape [B,N,N,dim]

    return output.to(torch.float32)

"""
Optimized TriMul (outgoing) forward pass.

The implementation follows the reference PyTorch TriMul but accelerates the
element‑wise mask & gate multiplication with a tiny Triton kernel and performs
the cubic contraction with a single `torch.einsum`, which avoids the costly
`permute`+`contiguous` copies required for a naïve `torch.bmm` approach.
All heavy linear work is done in FP16, while the final output is returned in
FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def mul_gate_mask_kernel(
    out_ptr,            # pointer to output (flattened)
    a_ptr,              # pointer to tensor to be gated (flattened)
    gate_ptr,           # pointer to gate tensor (flattened)
    mask_ptr,           # pointer to mask tensor (flattened, broadcast over H)
    total,              # total number of elements in the tensor
    BLOCK_SIZE: tl.constexpr,
    H: tl.constexpr,    # hidden dimension (size of the innermost axis)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    a = tl.load(a_ptr + offs, mask=mask)
    g = tl.load(gate_ptr + offs, mask=mask)

    # mask_idx = floor(offs / H) gives the index in the mask tensor (B,N,N)
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    tl.store(out_ptr + offs, a * g * m, mask=mask)


def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            mask_tensor : torch.Tensor of shape [B, N, N] (float32) or None
            weights_dict: mapping of weight names to tensors
            config_dict : {"dim": int, "hidden_dim": int}
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------------
    # 1. Input LayerNorm (FP16)
    # --------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float16)
    norm_b = weights["norm.bias"].to(device).to(torch.float16)
    # Cast input to fp16 first – LayerNorm works in fp16
    x = input_tensor.to(torch.float16)
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)  # [B,N,N,dim]

    # --------------------------------------------------------------------
    # 2. Linear projections (no bias) in FP16
    # --------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)  # [B,N,N,hidden_dim]

    # --------------------------------------------------------------------
    # 3. Gating (Linear + sigmoid) in FP16
    # --------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------------
    # 4. Fuse mask & gate with Triton (in‑place)
    # --------------------------------------------------------------------
    if mask_tensor is None:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(dtype=torch.float16, device=device)

    # Ensure tensors are contiguous before flattening for Triton
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_fp16 = mask_fp16.contiguous()

    left_flat = left.view(-1)
    right_flat = right.view(-1)
    left_gate_flat = left_gate.view(-1)
    right_gate_flat = right_gate.view(-1)
    mask_flat = mask_fp16.view(-1)  # size B*N*N

    total = left_flat.numel()                     # B * N * N * hidden_dim
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    # In‑place mask & gate fusion
    mul_gate_mask_kernel[grid](
        left_flat,               # out_ptr
        left_flat,               # a_ptr (in‑place)
        left_gate_flat,
        mask_flat,
        total,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
    )
    mul_gate_mask_kernel[grid](
        right_flat,
        right_flat,
        right_gate_flat,
        mask_flat,
        total,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
    )

    # Restore 4‑D view after scaling
    left = left_flat.view(B, N, N, hidden_dim)
    right = right_flat.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------------
    # 5. Cubic contraction via a single einsum (batched GEMM)
    # --------------------------------------------------------------------
    # out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)

    # --------------------------------------------------------------------
    # 6. Output LayerNorm, final gate and linear projection (all FP16)
    # --------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # final gating

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    output = F.linear(out, to_out_w)  # [B, N, N, dim] in fp16

    return output.to(torch.float32)
