"""
Implementation of the outgoing TriMul operator with Triton‑fused
elementwise mask‑gate multiplication.  The heavy linear projections and
the N‑by‑N matrix‑multiply contraction are performed with PyTorch/CUDA
BLAS; only the elementwise fusion of projection, gate and mask is
offloaded to a Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def mul_gate_mask_kernel(
    out_ptr,      # *mut fp16
    a_ptr,        # *const fp16
    gate_ptr,     # *const fp16
    mask_ptr,     # *const fp16 (broadcasted over hidden dim)
    total_elems: tl.int64,     # total number of elements in a/gate (B*N*N*H)
    hidden_dim: tl.int64,      # size of innermost dimension (H)
    BLOCK_SIZE: tl.constexpr,  # number of elements processed per program
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load the projection and its gate
    a = tl.load(a_ptr + offs, mask=mask)
    g = tl.load(gate_ptr + offs, mask=mask)

    # Index the mask (shape B*N*N) by collapsing the hidden dimension
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    tl.store(out_ptr + offs, a * g * m, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor : torch.Tensor of shape [B, N, N] or None
            - weights_dict: dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict: {"dim": int, "hidden_dim": int, "nomask": bool (optional)}
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack inputs
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # --------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # [B,N,N,dim] FP32

    # --------------------------------------------------------------------
    # 2. Cast to FP16 for the heavy work
    # --------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------------
    # 3. Linear projections (no bias) – still use PyTorch for speed
    # --------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # [B,N,N,hidden_dim], fp16
    right = F.linear(x_fp16, right_proj_w)  # [B,N,N,hidden_dim], fp16

    # --------------------------------------------------------------------
    # 4. Gating (sigmoid) – still in PyTorch
    # --------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w)) # [B,N,N,hidden_dim]
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))    # [B,N,N,hidden_dim]

    # --------------------------------------------------------------------
    # 5. Optional mask – broadcasted over hidden dimension
    # --------------------------------------------------------------------
    if mask_tensor is None or config.get("nomask", False):
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(device).to(torch.float16)

    # --------------------------------------------------------------------
    # 6. Fuse projection * gate * mask with a Triton kernel
    # --------------------------------------------------------------------
    left_flat = left.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_flat = right.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)
    mask_flat = mask_fp16.contiguous().view(-1)  # length B*N*N

    total_elems = left_flat.numel()  # B*N*N*hidden_dim
    BLOCK = 1024  # reasonable block size for elementwise work
    grid = (triton.cdiv(total_elems, BLOCK),)

    # Allocate output buffers
    left_out_flat = torch.empty_like(left_flat)
    right_out_flat = torch.empty_like(right_flat)

    # Launch for left branch
    mul_gate_mask_kernel[grid](
        left_out_flat,
        left_flat,
        left_gate_flat,
        mask_flat,
        total_elems,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
    )
    # Launch for right branch
    mul_gate_mask_kernel[grid](
        right_out_flat,
        right_flat,
        right_gate_flat,
        mask_flat,
        total_elems,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
    )

    # Reshape back to 4‑D tensors
    left = left_out_flat.view(B, N, N, hidden_dim)
    right = right_out_flat.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------------
    # 7. Cubic contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    This is a batched matrix‑multiply per hidden dimension.
    # --------------------------------------------------------------------
    # Rearrange to (B, hidden_dim, N, N) and transpose right on the k‑axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) – k as inner dim

    LH = B * hidden_dim
    left_mat = left_perm.view(LH, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(LH, N, N) # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N), fp16
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # --------------------------------------------------------------------
    # 8. Post‑processing: LayerNorm on hidden dimension, final gate, linear
    # --------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float32)

    out_fp32 = out.to(torch.float32)
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out_norm = out_norm * out_gate.to(torch.float32)  # apply final gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float32)  # (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)  # [B, N, N, dim], float32

    return output
