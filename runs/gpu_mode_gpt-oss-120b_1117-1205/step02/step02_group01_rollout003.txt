def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # layernorm
    norm_w = weights["norm.weight"].to(device).float()
    norm_b = weights["norm.bias"].to(device).float()
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    # cast to fp16
    x_fp16 = x.half()
    # linear projections
    left_proj_w = weights["left_proj.weight"].to(device).half()
    right_proj_w = weights["right_proj.weight"].to(device).half()
    left = F.linear(x_fp16, left_proj_w) # shape (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)
    # gating
    left_gate_w = weights["left_gate.weight"].to(device).half()

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))

@triton.jit
def mul_gate_mask_fused_kernel(
    left_out_ptr, right_out_ptr,
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total,
    BLOCK_SIZE: tl.constexpr,
    H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total
    # load
    left = tl.load(left_ptr + offs, mask=mask)
    right = tl.load(right_ptr + offs, mask=mask)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask)
    # compute mask index: each group of H elements shares same mask
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)
    left_out = left * left_gate * m
    right_out = right * right_gate * m
    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)

"""
TriMul (outgoing) forward pass with Triton‑accelerated elementwise
mask‑gate fusion.  The heavy linear projections, layer‑norms and the
cubic contraction (a batched matrix multiply) are performed with
PyTorch (using FP16 where beneficial).  Only the elementwise
multiplication `tensor * gate * mask` for the left and right branches
is fused into a single Triton kernel, reducing kernel launch overhead
and memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def fused_gate_mask_kernel(
    left_out_ptr,          # pointer to left output (flattened)
    right_out_ptr,         # pointer to right output (flattened)
    left_ptr,              # pointer to left input (flattened)
    right_ptr,             # pointer to right input (flattened)
    left_gate_ptr,         # pointer to left gate (flattened)
    right_gate_ptr,        # pointer to right gate (flattened)
    mask_ptr,              # pointer to mask (flattened, broadcast over H)
    total,                 # total number of elements (B*N*N*H)
    BLOCK_SIZE: tl.constexpr,
    H: tl.constexpr,       # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # load inputs
    left = tl.load(left_ptr + offs, mask=mask)
    right = tl.load(right_ptr + offs, mask=mask)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask)

    # broadcast mask over hidden dimension
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # fused multiply
    left_out = left * left_gate * m
    right_out = right * right_gate * m

    # store results
    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor : torch.Tensor of shape [B, N, N] (float32) or None
            - weights_dict: dict with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict: {"dim": int, "hidden_dim": int}
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).float()
    norm_b = weights["norm.bias"].to(device).float()
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for the heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.half()

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).half()
    right_proj_w = weights["right_proj.weight"].to(device).half()
    left = F.linear(x_fp16, left_proj_w)   # [B, N, N, hidden_dim], fp16
    right = F.linear(x_fp16, right_proj_w)  # [B, N, N, hidden_dim], fp16

    # ------------------------------------------------------------------
    # 4. Gating tensors (sigmoid, FP16)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).half()
    right_gate_w = weights["right_gate.weight"].to(device).half()
    out_gate_w = weights["out_gate.weight"].to(device).half()

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w)) # [B, N, N, hidden_dim]
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))    # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 5. Prepare mask (broadcasted over hidden dimension)
    # ------------------------------------------------------------------
    if mask_tensor is None:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 6. Fuse elementwise mask & gate multiplies via Triton kernel
    # ------------------------------------------------------------------
    # flatten tensors for 1‑D kernel launch
    left_flat = left.contiguous().view(-1)
    right_flat = right.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)
    mask_flat = mask_fp16.contiguous().view(-1)

    # allocate output buffers
    left_out_flat = torch.empty_like(left_flat)
    right_out_flat = torch.empty_like(right_flat)

    total_elements = left_flat.numel()
    BLOCK = 1024  # threads per program (must be <= 1024 for current hardware)
    grid = (triton.cdiv(total_elements, BLOCK),)

    fused_gate_mask_kernel[grid](
        left_out_flat,
        right_out_flat,
        left_flat,
        right_flat,
        left_gate_flat,
        right_gate_flat,
        mask_flat,
        total_elements,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
    )

    # reshape back to [B, N, N, hidden_dim]
    left = left_out_flat.view(B, N, N, hidden_dim)
    right = right_out_flat.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 7. Cubic contraction via batched matrix multiplication (FP16)
    #    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Rearrange tensors to (B, H, N, N) where the inner two dims are matrices
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) with k as rows

    # Batched matmul: broadcasting over B and H
    out_perm = torch.matmul(left_perm, right_perm)      # (B, H, N, N), fp16

    # Restore original layout [B, N, N, H]
    out = out_perm.permute(0, 2, 3, 1).contiguous()    # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 8. Post‑processing: LayerNorm on hidden dim, final gating, output proj
    # ------------------------------------------------------------------
    out_fp32 = out.float()

    to_out_norm_w = weights["to_out_norm.weight"].to(device).float()
    to_out_norm_b = weights["to_out_norm.bias"].to(device).float()
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # final gating (convert out_gate to FP32)
    out_norm = out_norm * out_gate.float()

    # final linear projection back to dim
    to_out_w = weights["to_out.weight"].to(device).float()  # shape (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)  # [B, N, N, dim]

    return output
