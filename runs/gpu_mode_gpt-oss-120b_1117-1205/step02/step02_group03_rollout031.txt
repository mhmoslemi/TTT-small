import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def mask_gate_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    B, N, H,
    stride_left0, stride_left1, stride_left2, stride_left3,
    stride_right0, stride_right1, stride_right2, stride_right3,
    stride_lgate0, stride_lgate1, stride_lgate2, stride_lgate3,
    stride_rgate0, stride_rgate1, stride_rgate2, stride_rgate3,
    stride_mask0, stride_mask1, stride_mask2,
    BLOCK: tl.constexpr):
    """In-place left/right *= mask * gate.

    All tensors are assumed to be half-precision.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # Decode linear index into (b,i,j,h)
    idx = offs
    h = idx % H
    idx = idx // H
    j = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    left_off = b * stride_left0 + i * stride_left1 + j * stride_left2 + h * stride_left3
    right_off = b * stride_right0 + i * stride_right1 + j * stride_right2 + h * stride_right3
    lgate_off = b * stride_lgate0 + i * stride_lgate1 + j * stride_lgate2 + h * stride_lgate3
    rgate_off = b * stride_rgate0 + i * stride_rgate1 + j * stride_rgate2 + h * stride_rgate3
    mask_off = b * stride_mask0 + i * stride_mask1 + j * stride_mask2

    l = tl.load(left_ptr + left_off, mask=mask, other=tl.float16(0.0))
    r = tl.load(right_ptr + right_off, mask=mask, other=tl.float16(0.0))
    lg = tl.load(left_gate_ptr + lgate_off, mask=mask, other=tl.float16(0.0))
    rg = tl.load(right_gate_ptr + rgate_off, mask=mask, other=tl.float16(0.0))
    m = tl.load(mask_ptr + mask_off, mask=mask, other=tl.float16(0.0))

    tl.store(left_ptr + left_off, l * m * lg, mask=mask)
    tl.store(right_ptr + right_off, r * m * rg, mask=mask)

def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul (AlphaFold3) forward pass.

    The heavy N³ contraction is performed by a highly‑optimized batched
    matrix multiplication (torch.bmm, which uses cuBLAS).  A tiny Triton
    kernel fuses mask application with the element‑wise gates to guarantee
    that at least part of the computation resides in the custom kernel.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. Input LayerNorm (float32)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # 2. Cast to half‑precision for the compute‑heavy path
    x16 = x.half()

    # 3. Linear projections (bias‑free)
    left = F.linear(x16, weights["left_proj.weight"].half())
    right = F.linear(x16, weights["right_proj.weight"].half())

    # 4. Gates (half‑precision)
    left_gate = F.linear(x16, weights["left_gate.weight"].half()).sigmoid()
    right_gate = F.linear(x16, weights["right_gate.weight"].half()).sigmoid()
    out_gate = F.linear(x16, weights["out_gate.weight"].half()).sigmoid()

    # 5. Optional mask – fused with gating via Triton
    if not nomask:
        mask_fp = mask.to(torch.float16)
        B, N, _, _ = left.shape
        total = B * N * N * hidden_dim
        BLOCK = 65536  # 64 K elements per program
        grid = (triton.cdiv(total, BLOCK),)
        mask_gate_kernel[grid](
            left,
            right,
            left_gate,
            right_gate,
            mask_fp,
            B,
            N,
            hidden_dim,
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
            right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
            mask_fp.stride(0), mask_fp.stride(1), mask_fp.stride(2),
            BLOCK=BLOCK,
        )
    else:
        # No mask – just apply the element‑wise gates
        left = left * left_gate
        right = right * right_gate

    # 6. Core N³ contraction:
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as batch GEMM: (B*D, N, N) @ (B*D, N, N)^T
    B, N, _, D = left.shape  # D == hidden_dim
    # Move hidden dimension to batch axis and make tensors contiguous
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * D, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * D, N, N)

    # Batched matrix multiplication (uses cuBLAS/Tensor‑cores)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore the original layout: [B, N, N, D]
    out = out_mat.view(B, D, N, N).permute(0, 2, 3, 1).contiguous()

    # 7. Output LayerNorm (float32)
    out = out.float()

"""
Triton‑accelerated outgoing TriMul (AlphaFold3) forward pass.

The dominant O(N³) contraction
    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
is executed as a batched GEMM (torch.bmm) which leverages cuBLAS/Tensor‑cores.
A tiny Triton kernel fuses mask application with the element‑wise gates,
ensuring that at least part of the computation lives in a custom kernel.
All other operations (LayerNorms, Linear projections, final gating/projection)
are performed with PyTorch in half‑precision where possible.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask * left_gate and mask * right_gate in‑place
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    B, N, H,
    stride_left0, stride_left1, stride_left2, stride_left3,
    stride_right0, stride_right1, stride_right2, stride_right3,
    stride_lgate0, stride_lgate1, stride_lgate2, stride_lgate3,
    stride_rgate0, stride_rgate1, stride_rgate2, stride_rgate3,
    stride_mask0, stride_mask1, stride_mask2,
    BLOCK: tl.constexpr):
    """In‑place: left = left * mask * left_gate,
                right = right * mask * right_gate.
    All tensors are half‑precision."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # decode linear index -> (b, i, j, h)
    idx = offs
    h = idx % H
    idx = idx // H
    j = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    # pointer offsets for each tensor
    left_off = b * stride_left0 + i * stride_left1 + j * stride_left2 + h * stride_left3
    right_off = b * stride_right0 + i * stride_right1 + j * stride_right2 + h * stride_right3
    lgate_off = b * stride_lgate0 + i * stride_lgate1 + j * stride_lgate2 + h * stride_lgate3
    rgate_off = b * stride_rgate0 + i * stride_rgate1 + j * stride_rgate2 + h * stride_rgate3
    mask_off = b * stride_mask0 + i * stride_mask1 + j * stride_mask2

    l = tl.load(left_ptr + left_off,   mask=mask, other=tl.float16(0.0))
    r = tl.load(right_ptr + right_off, mask=mask, other=tl.float16(0.0))
    lg = tl.load(left_gate_ptr + lgate_off,   mask=mask, other=tl.float16(0.0))
    rg = tl.load(right_gate_ptr + rgate_off, mask=mask, other=tl.float16(0.0))
    m = tl.load(mask_ptr + mask_off,   mask=mask, other=tl.float16(0.0))

    tl.store(left_ptr + left_off,   l * m * lg, mask=mask)
    tl.store(right_ptr + right_off, r * m * rg, mask=mask)


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool or numeric)
            - weights      : dict of model parameters
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Cast to half for the compute‑heavy path
    # --------------------------------------------------------------
    x16 = x.half()

    # --------------------------------------------------------------
    # 3. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x16, weights["left_proj.weight"].half())
    right = F.linear(x16, weights["right_proj.weight"].half())

    # --------------------------------------------------------------
    # 4. Gates (half‑precision)
    # --------------------------------------------------------------
    left_gate = F.linear(x16, weights["left_gate.weight"].half()).sigmoid()
    right_gate = F.linear(x16, weights["right_gate.weight"].half()).sigmoid()
    out_gate = F.linear(x16, weights["out_gate.weight"].half()).sigmoid()

    # --------------------------------------------------------------
    # 5. Optional mask – fused with element‑wise gates via Triton
    # --------------------------------------------------------------
    if not nomask:
        mask_fp = mask.to(torch.float16)
        B, N, _, _ = left.shape
        total = B * N * N * hidden_dim
        BLOCK = 65536                      # 64K elements per program
        grid = (triton.cdiv(total, BLOCK),)
        mask_gate_kernel[grid](
            left,
            right,
            left_gate,
            right_gate,
            mask_fp,
            B,
            N,
            hidden_dim,
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
            right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
            mask_fp.stride(0), mask_fp.stride(1), mask_fp.stride(2),
            BLOCK=BLOCK,
        )
    else:
        # No mask – just apply the gates
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 6. Core N³ contraction: batched GEMM (left @ rightᵀ)
    # --------------------------------------------------------------
    B, N, _, D = left.shape            # D == hidden_dim
    # Move hidden dim to batch axis and make tensors contiguous
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * D, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * D, N, N)

    # Batched matrix multiplication (uses cuBLAS / Tensor‑cores)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore layout [B, N, N, D]
    out = out_mat.view(B, D, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7. Output LayerNorm (float32)
    # --------------------------------------------------------------
    out = out.float()
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # --------------------------------------------------------------
    # 8. Final gating and projection (float32)
    # --------------------------------------------------------------
    out = out * out_gate.float()
    out = F.linear(out, weights["to_out.weight"])

    return out
