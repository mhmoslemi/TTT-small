"""
TriMul (outgoing) forward pass with a Triton‑accelerated elementwise
gate‑mask fusion.

The heavy linear projections, layer‑norms and the cubic contraction are
performed with PyTorch (FP16 where possible).  The only Triton kernel fuses
the three elementwise operations

    out = value * gate * mask

for the left and right branches, reducing memory traffic and launch
overhead.  The cubic contraction is expressed as a batched matrix‑multiply
using cuBLAS (torch.bmm), which is far faster than a naïve Triton
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def mul_gate_mask_kernel(
    out_ptr,    # pointer to output (flattened)
    val_ptr,    # pointer to value tensor (flattened)
    gate_ptr,   # pointer to gate tensor (flattened)
    mask_ptr,   # pointer to mask tensor (flattened, broadcast over hidden dim)
    total,      # total number of elements in the value/gate tensors
    BLOCK_SIZE: tl.constexpr,
    H: tl.constexpr,   # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # load value and gate
    val = tl.load(val_ptr + offs, mask=mask)
    g   = tl.load(gate_ptr + offs, mask=mask)

    # broadcast mask over the hidden dimension
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # fused multiply
    tl.store(out_ptr + offs, val * g * m, mask=mask)


def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            mask_tensor : torch.Tensor of shape [B, N, N] (float32) or None
            weights_dict: dictionary with the following keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            config_dict : {"dim": int, "hidden_dim": int, "nomask": bool}
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # [B,N,N,dim]

    # --------------------------------------------------------------------
    # 2. Cast to FP16 for the heavy work
    # --------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------------
    # 3. Linear projections (no bias)
    # --------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # [B,N,N,hidden_dim]
    right = F.linear(x_fp16, right_proj_w)  # [B,N,N,hidden_dim]

    # --------------------------------------------------------------------
    # 4. Gating (linear + sigmoid) and optional mask (fused via Triton)
    # --------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w  = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w)) # [B,N,N,hidden_dim]
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))    # [B,N,N,hidden_dim]

    # Prepare mask (broadcast over hidden dimension)
    if nomask or mask_tensor is None:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(dtype=torch.float16, device=device)

    # Flatten tensors for the Triton kernel
    left_flat      = left.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_flat      = right.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)
    mask_flat      = mask_fp16.contiguous().view(-1)

    total_elems = left_flat.numel()          # B * N * N * hidden_dim
    BLOCK = 65536                            # large block reduces launch overhead
    grid = (triton.cdiv(total_elems, BLOCK),)

    left_out  = torch.empty_like(left_flat)
    right_out = torch.empty_like(right_flat)

    # Fuse value * gate * mask for left and right branches
    mul_gate_mask_kernel[grid](
        left_out,
        left_flat,
        left_gate_flat,
        mask_flat,
        total_elems,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
    )
    mul_gate_mask_kernel[grid](
        right_out,
        right_flat,
        right_gate_flat,
        mask_flat,
        total_elems,
        BLOCK_SIZE=BLOCK,
        H=hidden_dim,
    )

    # Reshape back to [B,N,N,hidden_dim]
    left = left_out.view(B, N, N, hidden_dim)
    right = right_out.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------------
    # 5. Cubic contraction via a batched GEMM (torch.bmm)
    # --------------------------------------------------------------------
    # left: (B, N, N, H)   -> (B, H, N, N)
    # right: (B, N, N, H) -> (B, H, N, N) and transpose K <-> J
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) with K as rows, J as cols

    LH = B * hidden_dim
    left_mat  = left_perm.view(LH, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(LH, N, N)  # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N), FP16

    # Restore original layout [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # FP16

    # --------------------------------------------------------------------
    # 6. Post‑processing (LayerNorm on hidden dim, final gate, output proj)
    # --------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float32)
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply final gating
    out_norm = out_norm * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float32)  # (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)  # [B, N, N, dim], float32

    return output
```
