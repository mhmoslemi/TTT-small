"""
Outgoing TriMul (Triangle Multiplicative Update) forward pass.

The heavy O(N³) computation:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` are linear projections of the input tensor.
We fuse the O(N³) contraction into a single Triton kernel that performs a
batched GEMM (B*H independent N×N matrix multiplications).  The kernel
is autotuned over a few tile sizes to achieve high performance on the
B200 GPU.  All other operations (LayerNorm, gating, final projection) are
executed in PyTorch with mixed‑precision for speed while preserving
numerical fidelity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton batched GEMM kernel (C = A @ B) with autotuning
# --------------------------------------------------------------
@triton.autotune(
    configs=[
        # Good default for many shapes
        triton.Config(dict(BLOCK_M=64, BLOCK_N=64, BLOCK_K=64), num_stages=3, num_warps=4),
        # Larger M/N tiles for bigger N
        triton.Config(dict(BLOCK_M=128, BLOCK_N=128, BLOCK_K=32), num_stages=3, num_warps=8),
        triton.Config(dict(BLOCK_M=128, BLOCK_N=64, BLOCK_K=64), num_stages=3, num_warps=8),
        triton.Config(dict(BLOCK_M=64, BLOCK_N=128, BLOCK_K=64), num_stages=3, num_warps=8),
    ],
    key=["M", "N", "K"],
)
@triton.jit
def batched_gemm_kernel(
    a_ptr,                # *Pointer* to matrix A
    b_ptr,                # *Pointer* to matrix B
    c_ptr,                # *Pointer* to output matrix C
    M, N, K,              # matrix dimensions
    stride_am, stride_ak, stride_ah,   # strides for A (M, K, batch)
    stride_bk, stride_bn, stride_bh,   # strides for B (K, N, batch)
    stride_cm, stride_cn, stride_ch,   # strides for C (M, N, batch)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM kernel: C = A @ B for many independent batches."""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + batch_id * stride_ah
    b_batch = b_ptr + batch_id * stride_bh
    c_batch = c_ptr + batch_id * stride_ch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load B tile (K x N)
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # Accumulate
        acc += tl.dot(a, b)

    # Write back results (store as FP16 to save bandwidth)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def triton_batched_gemm(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B where
        A: (batch, M, K)  FP16, contiguous
        B: (batch, K, N)  FP16, contiguous
    Returns:
        C: (batch, M, N)  FP16, contiguous
    """
    assert A.is_cuda and B.is_cuda
    assert A.dtype == B.dtype == torch.float16
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides in elements
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bk, stride_bn = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    # Launch configuration – Triton will pick the best autotuned block size
    grid = (
        batch,
        triton.cdiv(M, 64),   # lower bound; actual BLOCK_M chosen by autotune
        triton.cdiv(N, 64),
    )
    batched_gemm_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_am,
        stride_ak,
        stride_ah,
        stride_bk,
        stride_bn,
        stride_bh,
        stride_cm,
        stride_cn,
        stride_ch,
        # BLOCK_* are filled by the autotuner
    )
    return C


# --------------------------------------------------------------
# Entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/int) or None
            - weights: dict mapping parameter names to tensors
            - config: dict containing 'dim', 'hidden_dim', and optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # shape [B,N,N,dim], float32

    # ----------------------------------------------------------
    # 2. Projections (FP16)
    # ----------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # [B,N,N,hidden_dim], fp16
    right = F.linear(x_fp16, right_proj_w)  # [B,N,N,hidden_dim], fp16

    # ----------------------------------------------------------
    # 3. Optional mask
    # ----------------------------------------------------------
    if not nomask:
        # mask is assumed to be broadcastable (bool/int)
        mask_fp = mask.unsqueeze(-1).to(torch.float16)  # [B,N,N,1]
        left = left * mask_fp
        right = right * mask_fp

    # ----------------------------------------------------------
    # 4. Gates (computed in float32 for better precision, then cast)
    # ----------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid().to(torch.float16)
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid().to(torch.float16)
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid().to(torch.float16)

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------
    # 5. Core O(N³) contraction via Triton batched GEMM
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ----------------------------------------------------------
    # Rearrange tensors to (B, hidden_dim, N, N) where the last two axes are
    # the matrix dimensions for the GEMM.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) = (B, H, K, J)

    B, H, N, _ = left_perm.shape
    total_batch = B * H

    left_flat = left_perm.view(total_batch, N, N)
    right_flat = right_perm.view(total_batch, N, N)

    # Batched GEMM: (total_batch, N, N) @ (total_batch, N, N)ᵀ → (total_batch, N, N)
    out_flat = triton_batched_gemm(left_flat, right_flat)

    # Reshape back to [B, N, N, hidden_dim] (FP16)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # subsequent layers work in FP32

    # ----------------------------------------------------------
    # 6. Output LayerNorm, output gate, final linear projection
    # ----------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(out.dtype)  # apply output gate
    out = F.linear(out, weights["to_out.weight"])  # final projection to dim (float32)

    return out
