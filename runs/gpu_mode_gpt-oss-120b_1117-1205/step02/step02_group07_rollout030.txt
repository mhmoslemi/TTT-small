def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)

    # 1. norm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # cast fp16
    x_fp16 = x.to(torch.float16)

    # projections
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # mask if needed
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_fp
        right = right * mask_fp

    # gates
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # apply gates
    left = left * left_gate
    right = right * right_gate

    B, N, _, _ = left.shape # shape (B,N,N,hidden_dim)
    # allocate output
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # kernel launch
    # get strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    # grid
    grid = (B * hidden_dim, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    tri_mul_batch_kernel[grid](
        left, right, out,
        N, N, N,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, HEADS=hidden_dim, num_warps=4
    )
    # convert out to float32
    out_fp32 = out.to(torch.float32)

    # output layernorm
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    # scale by out_gate
    out_scaled = out_norm * out_gate.to(torch.float32)

    # final linear
    out_final = F.linear(out_scaled, weights["to_out.weight"].to(torch.float32))

    return out_final

"""
TriMul (outgoing) forward pass with a fused Triton batched GEMM.

The core contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is implemented as a batched matrix multiplication where each
(head, batch) pair is a separate GEMM of shape (N,N)·(N,N)^T.
The kernel works directly on the original (B,N,N,H) layout
to avoid the costly permute+contiguous copies used in a naïve
implementation.  All heavy linear/gating work is done in fp16,
accumulation is fp32, and the final output is promoted to fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_batch_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,                          # all equal to seq_len
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HEADS: tl.constexpr
):
    """
    Batched GEMM for TriMul.

    Arguments:
        left_ptr, right_ptr, out_ptr : pointers to fp16 tensors
        M, N, K                     : dimensions (seq_len)
        stride_...                  : element‑wise strides of the tensors
        BLOCK_*                     : tile sizes
        HEADS                       : hidden dimension (used to split batch)
    """
    # program ids
    batch_head = tl.program_id(0)      # combined (batch * head)
    pid_m = tl.program_id(1)           # output tile row
    pid_n = tl.program_id(2)           # output tile col

    # split batch and head
    batch_id = batch_head // HEADS
    head_id = batch_head - batch_id * HEADS

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this (batch, head)
    left_base = left_ptr + batch_id * stride_left_b + head_id * stride_left_h
    right_base = right_ptr + batch_id * stride_right_b + head_id * stride_right_h
    out_base = out_ptr + batch_id * stride_out_b + head_id * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile: (M x K)
        a_ptrs = left_base + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # right tile: (K x N)   (k is row, j is column)
        b_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    # store result
    c_ptrs = out_base + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool or numeric)
            - weights_dict : mapping of parameter names to torch.Tensor
            - config_dict  : must contain 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim], float32

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for projections/gates
    # -----------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections (fp16) -> shape [B, N, N, hidden_dim]
    # -----------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # -----------------------------------------------------------------
    # 4. Optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)   # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # -----------------------------------------------------------------
    # 5. Gating (fp16) and apply
    # -----------------------------------------------------------------
    left_gate = F.linear(
        x_fp16, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = F.linear(
        x_fp16, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = F.linear(
        x_fp16, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6. Core contraction via Triton batched GEMM
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape  # left.shape == (B, N, N, hidden_dim)

    # allocate output tensor in fp16
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # strides (in elements)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # block sizes (tuned)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,               # batch * head dimension
        triton.cdiv(N, BLOCK_M),     # rows
        triton.cdiv(N, BLOCK_N)      # cols
    )

    tri_mul_batch_kernel[grid](
        left, right, out,
        N, N, N,                     # M, N, K
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HEADS=hidden_dim,
        num_warps=4
    )

    # convert to fp32 for the remaining layers
    out_fp32 = out.to(torch.float32)

    # -----------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final projection (float32)
    # -----------------------------------------------------------------
    out_norm = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out_scaled = out_norm * out_gate.to(torch.float32)
    out_final = F.linear(out_scaled, weights["to_out.weight"].to(torch.float32))

    return out_final
