import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        a = tl.load(a_ptrs,

def custom_kernel(data):
    """
    TriMul (outgoing) forward pass:
    1. LayerNorm on input.
    2. Linear projections left/right and gating (all in fp16).
    3. Optional mask applied to left/right.
    4. Elementwise gating.
    5. Core contraction computed as batched matrix multiplies via a Triton kernel.
    6. Output LayerNorm, out_gate, and final linear projection back to dim.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. Input LayerNorm (float32)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    x_fp16 = x.to(torch.float16)

    # 2. Linear projections (no bias) in fp16
    left = F.linear(
        x_fp16,
        weights["left_proj.weight"].to(torch.float16)
    )
    right = F.linear(
        x_fp16,
        weights["right_proj.weight"].to(torch.float16)
    )

    # 3. Apply mask if present
    if not nomask:
        # broadcast mask over hidden dimension
        mask_fp = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp
        right = right * mask_fp

    # 4. Gating (sigmoid)
    left_gate = F.linear(
        x_fp16,
        weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = F.linear(
        x_fp16,
        weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = F.linear(
        x_fp16,
        weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # 5. Core contraction via batched matmul
    #   left_perm: (B, H, N, N)
    #   right_perm: (B, H, N, N) with second N dimension transposed (j,k)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right must be transposed on the last two dims (k,j) -> (j,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape
    # flatten batch and hidden dim
    A_flat = left_perm.view(B * H, N, N)
    B_flat = right_perm.view(B * H, N, N)

    # batched matmul (fp16 in, fp16 out)
    out_flat = _triton_batched_matmul(A_flat, B_flat)  # (B*H, N, N)

    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)
    out = out.to(torch.float32)

    # 6. Output LayerNorm, out_gate, final projection
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward operator with a fused Triton batched GEMM.

Algorithm
---------
1. LayerNorm over the channel dimension of the input tensor (float32).
2. Linear projections (left, right) and three gating projections are computed
   in fp16 for speed.
3. An optional mask (broadcast over the hidden dimension) is applied to the
   projected tensors.
4. Sigmoid gates are applied element‑wise.
5. Core contraction:
       out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   is realized as a batched matrix multiplication
   C_{b,h} = left_{b,h} @ right_{b,h}ᵀ  where
   left_{b,h}  ∈ ℝ^{N×N},  right_{b,h} ∈ ℝ^{N×N}.
   The multiplication is performed by a custom Triton kernel that
   loads fp16 values, accumulates in fp32 and writes the result as fp32.
6. The result is reshaped back to [B, N, N, hidden_dim], LayerNormed,
   multiplied by the output gate and finally linearly projected back to
   the original channel dimension.

The kernel handles arbitrary batch sizes B ∈ {1,2}, sequence lengths N
up to 1024 and hidden dimensions up to 128 efficiently.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ B
#   A : (batch, M, K)   (fp16)
#   B : (batch, K, N)   (fp16)
#   C : (batch, M, N)   (fp32)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)          # batch dimension (B * hidden_dim)
    pid_m = tl.program_id(1)            # block row
    pid_n = tl.program_id(2)            # block column

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the current batch
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        # Load B tile (K x N)
        b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # dot automatically promotes fp16 → fp32

    # Store result as fp32
    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


def _triton_batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.
    A : (batch, M, K)  fp16
    B : (batch, K, N)  fp16
    Returns C : (batch, M, N) fp32
    """
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # Strides (in element units)
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tunable tile sizes (chosen to fit typical SM shared‑memory limits)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32, on CUDA)
            - mask_tensor:  torch.Tensor of shape [B, N, N] (bool/float) – may be ignored
            - weights_dict: mapping from parameter name to torch.Tensor (all on the same device)
            - config_dict: must contain keys 'dim' and 'hidden_dim'; optional 'nomask' (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) in fp16
    # ------------------------------------------------------------------
    left = F.linear(
        x_fp16,
        weights["left_proj.weight"].to(torch.float16)
    )
    right = F.linear(
        x_fp16,
        weights["right_proj.weight"].to(torch.float16)
    )

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) in fp16
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_fp16,
        weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = F.linear(
        x_fp16,
        weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = F.linear(
        x_fp16,
        weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via batched GEMM (Triton)
    #    left_perm: (B, H, N, N)
    #    right_perm: (B, H, N, N) with the last two axes transposed (k, j)
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) = (B, H, K, J)

    B, H, N, _ = left_perm.shape

    # Flatten batch and hidden dimensions for the kernel
    A_flat = left_perm.view(B * H, N, N)   # (B*H, M=N, K=N)
    B_flat = right_perm.view(B * H, N, N)  # (B*H, K=N, N=N)

    # Batched matmul: result is (B*H, N, N) in fp32
    out_flat = _triton_batched_matmul(A_flat, B_flat)

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, output gate, and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"])

    return out
