def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # cast to fp16 for speed
    x_fp16 = x.to(torch.float16)

    # Projections
    left = torch.nn.functional.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = torch.nn.functional.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    if not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_fp
        right = right * mask_fp

    # Gates
    left_gate = torch.nn.functional.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # permutation to bring hidden dim before N,N
    # left shape: B,N,N,H => B,H,N,N after permute
    left_p = left.permute(0, 3, 1, 2).contiguous()
    right_p = right.permute(0, 3, 1, 2).contiguous()  # Right will be transposed later

    B, H, N, _ = left_p.shape

    # Reshape to (B*H, N, N)
    left_flat = left_p.view(B*H, N, N)
    right_flat = right_p.view(B*H, N, N)

    # Compute batched GEMM: left @ right^T
    # Use torch.bmm after transposing right to get K dimension.
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Reshape back: (B, H, N, N) then permute to (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out_fp32 = out.to(torch.float32)

    # Output LayerNorm
    out_norm = torch.nn.functional.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out_norm = out_norm * out_gate.to(out_norm.dtype)
    # Final linear (dim)
    out = torch.nn.functional.linear(out_norm, weights["to_out.weight"])
    return out
```

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def apply_mask_kernel(x_ptr, mask_ptr, out_ptr, total_elements, hidden_dim, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    out = x * m
    tl.store(out_ptr + offs, out, mask=mask)

def custom_kernel(data):
    """
    Outgoing TriMul forward pass.
    Performs LayerNorm, linear projections, optional mask via a small Triton kernel,
    gating, and the heavy O(N³) contraction using a batched torch.bmm (cuBLAS).
    Returns tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. Input LayerNorm (float32)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16 for speed
    x_fp16 = x.to(torch.float16)

    # 2. Linear projections (fp16)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # (B, N, N, hidden_dim)
    right = F.linear(x_fp16, right_proj_w) # (B, N, N, hidden_dim)

    # 3. Optional mask via Triton (broadcast over hidden_dim)
    if not nomask:
        # flatten tensors
        left = left.contiguous()
        right = right.contiguous()
        mask_fp16 = mask.to(torch.float16).contiguous().view(-1)  # (B*N*N,)
        total = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)
        # in-place mask for left and right
        apply_mask_kernel[grid](
            left,
            mask_fp16,
            left,
            total,
            hidden_dim,
            BLOCK=BLOCK,
        )
        apply_mask_kernel[grid](
            right,
            mask_fp16,
            right,
            total,
            hidden_dim,
            BLOCK=BLOCK,
        )

    # 4. Gating (fp16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # 5. Core contraction: batched matmul left @ right^T per hidden channel
    # reshape to (B*hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # batched matmul using cuBLAS (torch.bmm)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # restore shape (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    # to float32 for following layer norm
    out = out.to(torch.float32)

    # 6. Output LayerNorm, gate, final linear
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(out.dtype)
    out = F.linear(out, weights["to_out.weight"])
    return out

"""
Outgoing TriMul forward pass.

The heavy O(N³) contraction is performed as a batched matrix multiplication
using cuBLAS (torch.bmm) for maximum throughput.  An optional mask is applied
by a tiny Triton kernel that broadcasts a [B, N, N] mask over the hidden
dimension.  All other operations (LayerNorm, linear projections, gating) are
implemented with PyTorch in mixed‑precision (FP16 where possible, FP32 for
final layers).  The result matches the reference PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def apply_mask_kernel(x_ptr, mask_ptr, out_ptr, total_elements, hidden_dim, BLOCK: tl.constexpr):
    """
    In‑place multiplication of a tensor x of shape (B, N, N, hidden_dim)
    with a broadcast mask of shape (B, N, N).  The mask is expanded over the
    hidden dimension inside the kernel.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    # Load element from x
    x_val = tl.load(x_ptr + offs, mask=valid, other=0.0)

    # Compute corresponding mask index (mask repeats every hidden_dim entries)
    mask_idx = offs // hidden_dim
    m_val = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    # Apply mask
    out_val = x_val * m_val
    tl.store(out_ptr + offs, out_val, mask=valid)


def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward pass.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/int/float), may be ignored
            - weights_dict: mapping of parameter names to torch.Tensor
            - config_dict: contains 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections (fp16)
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)          # (B, N, N, hidden_dim)
    right = F.linear(x_fp16, right_proj_w)        # (B, N, N, hidden_dim)

    # -----------------------------------------------------------------
    # 3. Optional mask via a tiny Triton kernel
    # -----------------------------------------------------------------
    if not nomask:
        # Ensure contiguity for pointer arithmetic
        left = left.contiguous()
        right = right.contiguous()
        mask_fp16 = mask.to(torch.float16).contiguous().view(-1)   # (B*N*N,)

        total = left.numel()                     # B * N * N * hidden_dim
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)

        # In‑place mask application (broadcast over hidden_dim)
        apply_mask_kernel[grid](
            left,
            mask_fp16,
            left,
            total,
            hidden_dim,
            BLOCK=BLOCK,
        )
        apply_mask_kernel[grid](
            right,
            mask_fp16,
            right,
            total,
            hidden_dim,
            BLOCK=BLOCK,
        )

    # -----------------------------------------------------------------
    # 4. Gating (fp16)
    # -----------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: batched matmul left @ right^T per hidden channel
    #    left: (B, N, N, H) -> (B, H, N, K)
    #    right: (B, N, N, H) -> (B, H, K, N)
    #    Compute out = left @ right   -> (B, H, N, N)
    # -----------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, K, N)

    B, H, N, K = left_perm.shape
    left_flat = left_perm.view(B * H, N, K)            # (B*H, N, K)
    right_flat = right_perm.view(B * H, K, N)          # (B*H, K, N)

    # Batched GEMM using cuBLAS (torch.bmm) – fast FP16 matmul with FP32 acc.
    out_flat = torch.bmm(left_flat, right_flat)        # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)                        # to float32 for final layers

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, outer gate, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(out.dtype)
    out = F.linear(out, weights["to_out.weight"])
    return out
