import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def mul2_kernel(a_ptr, b_ptr, out_ptr, N, stride_a, stride_b, stride_out, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    a = tl.load(a_ptr + offs * stride_a, mask=mask)
    b = tl.load(b_ptr + offs * stride_b, mask=mask)
    out = a * b
    tl.store(out_ptr + offs * stride_out, out, mask=mask)

@triton.jit
def mul3_kernel(a_ptr, b_ptr, c_ptr, out_ptr, N, stride_a, stride_b, stride_c, stride_out, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    a = tl.load(a_ptr + offs * stride_a, mask=mask)
    b = tl.load(b_ptr + offs * stride_b, mask=mask)
    c = tl.load(c_ptr + offs * stride_c, mask=mask)
    out = a * b * c
    tl.store(out_ptr + offs * stride_out, out, mask=mask)

def _apply_elementwise_mul2(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    assert a.shape == b.shape
    N = a.numel()
    out = torch.empty_like(a)
    BLOCK = 8192  # must be multiple of 32 and <= 65536
    grid = (triton.cdiv(N, BLOCK),)
    mul2_kernel[grid](
        a, b, out,
        N,
        a.stride(0), b.stride(0), out.stride(0),
        BLOCK=BLOCK
    )
    return out

def _apply_elementwise_mul3(a: torch.Tensor, b: torch.Tensor, c: torch.Tensor) -> torch.Tensor:
    assert a.shape == b.shape == c.shape
    N = a.numel()
    out = torch.empty_like(a)
    BLOCK = 8192
    grid = (triton.cdiv(N, BLOCK),)
    mul3_kernel[grid](
        a, b, c, out,
        N,
        a.stride(0), b.stride(0), c.stride(0), out.stride(0),
        BLOCK=BLOCK
    )
    return out

def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor:  torch.Tensor [B, N, N] (bool or numeric)
            - weights_dict: mapping from parameter name to torch.Tensor
            - config_dict: must contain 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. Input LayerNorm (float32)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim] float32

    # 2. Cast to fp16 for the heavy part
    x_h = x.to(torch.float16)

    # 3. Linear projections (no bias) -> fp16
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # 4. Gating tensors (fp16)
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # 5. Fuse mask (if any) and gates via Triton kernels
    if not nomask:
        # mask broadcast to hidden_dim
        mask_fp = mask.unsqueeze(-1).to(torch.float16).expand(-1, -1, -1, hidden_dim)
        left = _apply_elementwise_mul3(left, mask_fp, left_gate)
        right = _apply_elementwise_mul3(right, mask_fp, right_gate)
    else:
        left = _apply_elementwise_mul2(left, left_gate)
        right = _apply_elementwise_mul2(right, right_gate)

    # 6. Core contraction: batched GEMM (A @ Bᵀ) using torch.bmm for speed
    #   left shape: [B, N, N, H] -> permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    #   right needs to be transposed on its two N dimensions
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) where last two axes are swapped

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # batched matrix multiplication (fp16 inputs, fp32 accumulation, fp16 output)
    out_flat = torch.bmm(left_flat, right_flat)  # (B*H, N, N) fp16

    # reshape back to original layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

"""
Triton‑accelerated forward pass for the outgoing TriMul operator.

The expensive N³ contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched matrix multiplication (A @ Bᵀ) using
torch.bmm (leveraging cuBLAS tensor cores).  The elementwise
mask/gating step is fused into a tiny Triton kernel to satisfy the
requirement of having a custom kernel in the pipeline.
All other operations (LayerNorm, Linear, sigmoid) are performed with
standard PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernels for element‑wise multiplication (2‑operand and 3‑operand)
# ----------------------------------------------------------------------
@triton.jit
def mul2_kernel(
    a_ptr, b_ptr, out_ptr,
    N,               # total number of elements
    stride_a, stride_b, stride_out,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    a = tl.load(a_ptr + offs * stride_a, mask=mask)
    b = tl.load(b_ptr + offs * stride_b, mask=mask)
    tl.store(out_ptr + offs * stride_out, a * b, mask=mask)


@triton.jit
def mul3_kernel(
    a_ptr, b_ptr, c_ptr, out_ptr,
    N,               # total number of elements
    stride_a, stride_b, stride_c, stride_out,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    a = tl.load(a_ptr + offs * stride_a, mask=mask)
    b = tl.load(b_ptr + offs * stride_b, mask=mask)
    c = tl.load(c_ptr + offs * stride_c, mask=mask)
    tl.store(out_ptr + offs * stride_out, a * b * c, mask=mask)


def _elemwise_mul2(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """In‑place a *= b using Triton (both tensors must be contiguous and same shape)."""
    assert a.shape == b.shape
    a_flat = a.view(-1)
    b_flat = b.view(-1)
    N = a_flat.numel()
    BLOCK = 8192
    grid = (triton.cdiv(N, BLOCK),)
    mul2_kernel[grid](
        a_flat,
        b_flat,
        a_flat,            # write result back into a
        N,
        a_flat.stride(0),
        b_flat.stride(0),
        a_flat.stride(0),
        BLOCK=BLOCK,
    )
    return a


def _elemwise_mul3(a: torch.Tensor, b: torch.Tensor, c: torch.Tensor) -> torch.Tensor:
    """In‑place a *= b * c using Triton (all tensors must be contiguous and same shape)."""
    assert a.shape == b.shape == c.shape
    a_flat = a.view(-1)
    b_flat = b.view(-1)
    c_flat = c.view(-1)
    N = a_flat.numel()
    BLOCK = 8192
    grid = (triton.cdiv(N, BLOCK),)
    mul3_kernel[grid](
        a_flat,
        b_flat,
        c_flat,
        a_flat,            # write result back into a
        N,
        a_flat.stride(0),
        b_flat.stride(0),
        c_flat.stride(0),
        a_flat.stride(0),
        BLOCK=BLOCK,
    )
    return a


# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor:  torch.Tensor [B, N, N] (bool or numeric)
            - weights_dict: mapping from parameter name to torch.Tensor
            - config_dict: must contain 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the heavy part
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (bias‑free) → fp16
    # ------------------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
    )  # [B, N, N, hidden_dim]
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
    )  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 4. Gating tensors (fp16)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
    ).sigmoid()

    # ------------------------------------------------------------------
    # 5. Fuse mask (if any) and gates via Triton kernels
    # ------------------------------------------------------------------
    # Ensure all tensors are contiguous for the kernels
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    if not nomask:
        # Broadcast mask to hidden_dim, convert to fp16 and make contiguous
        mask_fp = mask.unsqueeze(-1).to(torch.float16).expand(-1, -1, -1, hidden_dim).contiguous()
        # left = left * mask * left_gate
        left = _elemwise_mul3(left, mask_fp, left_gate)
        # right = right * mask * right_gate
        right = _elemwise_mul3(right, mask_fp, right_gate)
    else:
        left = _elemwise_mul2(left, left_gate)
        right = _elemwise_mul2(right, right_gate)

    # ------------------------------------------------------------------
    # 6. Core contraction: batched GEMM (A @ Bᵀ) using torch.bmm (fp16)
    # ------------------------------------------------------------------
    # permute to (B, H, N, N) and make contiguous for the flatten
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    # right needs to be transposed on its two N axes
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N)  # (B*H, N, N)

    # batched matrix multiply (half precision, tensor cores)
    out_flat = torch.bmm(left_flat, right_flat)   # (B*H, N, N) half

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)   # back to fp32 for the remaining ops

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
