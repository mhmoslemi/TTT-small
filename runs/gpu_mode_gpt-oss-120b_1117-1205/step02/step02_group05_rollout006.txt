"""
Triton‐accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. LayerNorm over the feature dimension (float32).
2. Linear projection to hidden dimension (fp16, no bias).
3. Optional mask application (broadcast over hidden dim).
4. Gating values via linear + sigmoid (fp16) and element‑wise gating.
5. Core contraction:
   For each batch and hidden channel d compute
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched GEMM `left @ right.T`.  Implemented with a
   tiled Triton kernel (FP16 inputs, FP32 accumulation).
6. LayerNorm on the hidden dimension, out‑gate multiplication,
   final linear projection back to the original dimension (float32).

The heavy O(N³) contraction is performed by the Triton kernel, while
all other ops stay in PyTorch for simplicity and correctness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ B
#   A : (batch, M, K)   fp16
#   B : (batch, K, N)   fp16   (B is already transposed for the TriMul)
#   C : (batch, M, N)   fp16
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)          # which (batch * hidden) GEMM
    pid_m = tl.program_id(1)             # tile row index
    pid_n = tl.program_id(2)             # tile column index

    # Compute offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles of size BLOCK_K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        # Load B tile (K x N)
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        # Compute tile multiplication, accumulate in FP32
        acc += tl.dot(a, b)

    # Write result back (cast to fp16)
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def triton_batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using Triton: C = A @ B   (B is already transposed for TriMul)

    Args:
        A: (batch, M, K)   fp16, contiguous
        B: (batch, K, N)   fp16, contiguous

    Returns:
        C: (batch, M, N)   fp16
    """
    assert A.dtype == B.dtype == torch.float16
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides in element units
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tile sizes (tuned for A100 / B200 fp16)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool or numeric) or None
            - weights_dict : dict of model parameters
            - config_dict  : dict with keys 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # Unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. LayerNorm over feature dimension (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # Cast heavy arithmetic to float16
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(
        x_fp16,
        weights["left_proj.weight"].to(torch.float16)
    )
    right = F.linear(
        x_fp16,
        weights["right_proj.weight"].to(torch.float16)
    )

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if (not nomask) and (mask is not None):
        # mask shape -> [B, N, N, 1] for broadcasting
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 4. Gating values (sigmoid) and application
    # --------------------------------------------------------------
    left_gate = F.linear(
        x_fp16,
        weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = F.linear(
        x_fp16,
        weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = F.linear(
        x_fp16,
        weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction: batched GEMM left @ right.T via Triton
    # --------------------------------------------------------------
    # Rearrange to (batch, hidden, N, N) layout
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    # right must be transposed on the two sequence axes: (B, H, K, J)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) where last two dims are (k, j)

    B, H, N, _ = left_perm.shape
    # Merge batch and hidden into a single GEMM batch dimension
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)  # already B_T (K, N)

    # Perform batched matrix multiplication: (B*H, N, N) @ (B*H, N, N) -> (B*H, N, N)
    out_flat = triton_batched_matmul(left_flat, right_flat)  # fp16 result

    # Restore original layout (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)   # back to fp32 for final layers

    # --------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
