"""
Triton‑accelerated outgoing TriMul forward pass.

Algorithm:
1. Layer‑norm the input over the channel dimension.
2. Project the tensor to a hidden dimension with two linear layers, optionally
   applying a mask with a tiny Triton kernel.
3. Compute three sigmoid gates.
4. Multiply the projections by their corresponding gates.
5. Core contraction is a batched matrix multiplication:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
   We reshape the tensors to (B*H, N, N) and use torch.bmm, which leverages
   cuBLAS tensor‑core kernels.
6. Apply a second layer‑norm, the output gate and a final linear projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _apply_mask_kernel(
    x_ptr,      # pointer to the tensor to be masked (contiguous)
    mask_ptr,   # pointer to the mask tensor (contiguous, shape [B,N,N])
    total,      # total number of elements in x (int)
    hidden_dim, # hidden dimension H (int)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # load data
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)

    # compute the index of the corresponding mask element (broadcast over H)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # apply
    x = x * m

    # store back
    tl.store(x_ptr + offs, x, mask=mask)

def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored
            - weights_dict: name → torch.Tensor (already on device)
            - config_dict: must contain 'dim' and 'hidden_dim',
                           optional 'nomask' (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, N2, dim = input_tensor.shape
    assert N == N2, "input must be square in the 2nd and 3rd dims"
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. Input layer‑norm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    x_fp16 = x.to(torch.float16)

    # 2. Linear projections (bias‑free)
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # 3. Optional mask – fused Triton kernel
    if not nomask:
        # ensure contiguous layout
        left = left.contiguous()
        right = right.contiguous()
        mask = mask.to(torch.float16).contiguous()
        # left
        total = left.numel()
        BLOCK = 128
        grid = (triton.cdiv(total, BLOCK),)
        _apply_mask_kernel[grid](
            left,
            mask,
            total,
            hidden_dim,
            BLOCK=BLOCK,
        )
        # right
        total = right.numel()
        grid = (triton.cdiv(total, BLOCK),)
        _apply_mask_kernel[grid](
            right,
            mask,
            total,
            hidden_dim,
            BLOCK=BLOCK,
        )

    # 4. Gates (sigmoid after linear)
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # 5. Core contraction – batched matmul using cuBLAS (torch.bmm)
    # reshape to (B, H, N, N) then merge batch & head
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)

    # flatten batch * head
    A = left_perm.view(B * hidden_dim, N, N)
    B_mat = right_perm.view(B * hidden_dim, N, N)

    # batched matrix multiplication; torch uses tensor cores for fp16
    out_flat = torch.bmm(A, B_mat)   # (B*H, N, N), fp16

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # 6. Output layer‑norm, output gate and final projection
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _apply_mask_kernel(
    x_ptr,      # pointer to the tensor to be masked (contiguous, fp16)
    mask_ptr,   # pointer to the mask tensor (contiguous, fp16) shape [B,N,N]
    total,      # total number of elements in x (int64)
    hidden_dim, # hidden dimension H (int64)
    BLOCK: tl.constexpr,
):
    """
    In‑place element‑wise multiplication of a 4‑D tensor `x` (flattened) by a
    broadcast mask of shape [B,N,N].  The innermost dimension of `x` is the
    hidden dimension and is broadcast over.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total

    # Load the fp16 data
    x = tl.load(x_ptr + offs, mask=m, other=0.0)

    # Index of the corresponding mask element (broadcast over hidden_dim)
    mask_index = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_index, mask=m, other=1.0)

    # Apply mask
    x = x * mask_val

    # Store back in‑place
    tl.store(x_ptr + offs, x, mask=m)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or float) – may be ignored
            - weights_dict: mapping from parameter name to torch.Tensor
            - config_dict: must contain 'dim' and 'hidden_dim';
                           optional 'nomask' (bool) – if True the mask is ignored.

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, N2, dim = input_tensor.shape
    assert N == N2, "input must be square on the last two dimensions"
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free) in fp16
    # ------------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Optional mask – Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        # Ensure contiguous layout for the kernel
        left = left.contiguous()
        right = right.contiguous()
        mask_fp16 = mask.to(torch.float16).contiguous()

        # Apply mask to `left`
        total = left.numel()
        BLOCK = 128
        grid = (triton.cdiv(total, BLOCK),)
        _apply_mask_kernel[grid](
            left,
            mask_fp16,
            total,
            hidden_dim,
            BLOCK=BLOCK,
        )

        # Apply mask to `right`
        total = right.numel()
        grid = (triton.cdiv(total, BLOCK),)
        _apply_mask_kernel[grid](
            right,
            mask_fp16,
            total,
            hidden_dim,
            BLOCK=BLOCK,
        )

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction – batched matrix multiplication
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)

    # Merge batch and head dimensions
    A = left_perm.view(B * hidden_dim, N, N)   # (B*H, N, N)
    B_mat = right_perm.view(B * hidden_dim, N, N)  # (B*H, N, N)

    # Batched GEMM in fp16 (torch.bmm uses tensor‑core kernels)
    out_flat = torch.bmm(A, B_mat)   # (B*H, N, N)

    # Reshape back to (B, N, N, H) and cast to fp32 for the remaining ops
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, output gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
