@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

We need to check that strides are in elements (not bytes). In Triton, pointer arithmetic uses element offset.

Thus we set strides accordingly.

```
def triton_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    # A: (batch, M, K), B: (batch, K, N), dtype float16, contiguous
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2
    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)
    # strides in elements
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32  # could use 32 for better warp occupancy
    grid = (batch,
            triton.cdiv(M, BLOCK_M),
            triton.cdiv(N, BLOCK_N))
    batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C
```

"""
TriMul (outgoing) forward pass with a fused linear projection and a
Triton‑accelerated batched GEMM.

Algorithm
---------
1. Layer‑norm the input (float32) over the channel dimension *dim*.
2. Fuse the five linear layers (left_proj, right_proj, left_gate,
   right_gate, out_gate) into a single matmul with a stacked weight
   matrix (shape (5*hidden_dim, dim)).  Split the result into the
   respective tensors.
3. Apply the optional mask only to the left/right tensors.
4. Compute the gating sigmoids and multiply them into left/right.
5. Perform the heavy N³ contraction:
       out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   which is a batched matrix multiplication A @ Bᵀ for every hidden
   dimension *d*.  The batched GEMM is implemented in Triton (half‑
   precision inputs, float32 accumulation, float32 output).
6. Layer‑norm the contraction result over *hidden_dim*, apply the
   out‑gate, and a final linear projection back to *dim* (all in
   float32).

The kernel only handles the contraction; the surrounding linear
projections, normalisations and gating are performed with PyTorch
operations for simplicity and high performance.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for tensors of shape
# (batch, M, K) and (batch, K, N)  (inputs fp16, output fp32)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program identifiers
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A tile (M x K) – fp16
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        # Load B tile (K x N) – fp16 (already transposed layout)
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate
        acc += tl.dot(a, b)   # fp16*fp16 -> fp32

    # Store result as FP32
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def triton_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM C = A @ Bᵀ where A, B are (batch, M, K) and (batch, K, N)
    with dtype torch.float16. Returns C of shape (batch, M, N) with dtype
    torch.float32.
    """
    assert A.dtype == B.dtype == torch.float16
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # strides in elements
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tile sizes (tuned for Ampere B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Entrypoint expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor:  torch.Tensor [B, N, N] (bool or numeric)
            - weights_dict: mapping from parameter name to torch.Tensor
            - config_dict: must contain keys 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0. unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  float32
    x_fp16 = x.to(torch.float16)  # fp16 for the heavy ops

    # ------------------------------------------------------------------
    # 2. Fuse the five linear projections (no bias)
    # ------------------------------------------------------------------
    # Stack weights: (5*hidden_dim, dim)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    # Single matmul: result shape [B, N, N, 5*hidden_dim]
    proj = F.linear(x_fp16, combined_weight)  # fp16

    # Split into the five tensors
    left, right, left_gate_pre, right_gate_pre, out_gate_pre = torch.split(
        proj, hidden_dim, dim=-1
    )

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and apply
    # ------------------------------------------------------------------
    left_gate = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()
    out_gate = out_gate_pre.sigmoid()           # [B, N, N, hidden_dim] fp16

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton (batched GEMM)
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    # Permute to (B, H, N, N) where rows = i/k and rows = k/j respectively
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    B_, H_, N_, _ = left_perm.shape
    left_flat = left_perm.view(B_ * H_, N_, N_)   # (B*H, M=N, K=N)
    right_flat = right_perm.view(B_ * H_, N_, N_)  # (B*H, K=N, N=N)

    # Batched GEMM → result in float32
    out_flat = triton_matmul(left_flat, right_flat)   # (B*H, N, N) fp32

    # Reshape back to (B, N, N, H)
    out = (
        out_flat.view(B_, H_, N_, N_)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, hidden_dim] float32

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # float32

    out = out * out_gate.to(torch.float32)

    out = F.linear(out, weights["to_out.weight"])  # float32, shape [B, N, N, dim]

    return out
