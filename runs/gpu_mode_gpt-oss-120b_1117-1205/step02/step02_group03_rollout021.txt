"""
Triton‑accelerated outgoing TriMul (AlphaFold3).

Core contraction:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

The kernel treats each (batch, hidden) pair as a separate batch and
performs a tiled batched GEMM with the right operand transposed, thus
avoiding any costly permute+contiguous copies. All other operations
(LayerNorms, Linear projections, gates) stay in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Batched GEMM kernel for the TriMul contraction.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    where the right operand is implicitly transposed (k↔j).

    Arguments
    ----------
    * left_ptr, right_ptr, out_ptr : pointers to the tensors.
    * B, N, H                     : batch size, sequence length, hidden dim.
    * stride_*                     : tensor strides (in elements).
    * BLOCK_M, BLOCK_N, BLOCK_K   : tiling parameters.
    """
    pid_h = tl.program_id(0)  # combined batch*hidden dimension
    pid_i = tl.program_id(1)  # tile over i (rows)
    pid_j = tl.program_id(2)  # tile over j (cols)

    # decode batch and hidden indices
    hid = pid_h % H
    batch = pid_h // H

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (batch, hidden) pair
    left_base  = left_ptr  + batch * stride_left_b  + hid * stride_left_h
    right_base = right_ptr + batch * stride_right_b + hid * stride_right_h
    out_base   = out_ptr   + batch * stride_out_b   + hid * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension (the summed index)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left:  (M, K)  where K dimension is original “k”
        a_ptrs = left_base \
                 + offs_i[:, None] * stride_left_i \
                 + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # right (transposed): (K, N) where rows are “k”, columns are “j”
        b_ptrs = right_base \
                 + offs_k[:, None] * stride_right_k \
                 + offs_j[None, :] * stride_right_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # accumulate using tensor‑core‑friendly dot (fp16 → fp32)
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # store result
    c_ptrs = out_base \
             + offs_i[:, None] * stride_out_i \
             + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host‑side helper for the batched GEMM
# ----------------------------------------------------------------------
def tri_mul_contraction(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched contraction out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    left/right : [B, N, N, H]   (fp16, contiguous)
    returns    : [B, N, N, H]   (fp16)
    """
    assert left.is_contiguous() and right.is_contiguous()
    assert left.dtype == torch.float16 and right.dtype == torch.float16
    B, N, _, H = left.shape

    out = torch.empty_like(left)

    # strides (in element counts)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # tiling
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,                     # combined batch*hidden
        triton.cdiv(N, BLOCK_M),  # i‑tiles
        triton.cdiv(N, BLOCK_N),  # j‑tiles
    )

    tri_mul_outgoing_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_ri, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # reasonable default for B200
    )
    return out


# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool or numeric)
            - weights      : dict of model parameters
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)  # fp16 for the heavy path

    # --------------------------------------------------------------
    # 2. Linear projections (no bias, fp16)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left  = left * mask_fp
        right = right * mask_fp

    # --------------------------------------------------------------
    # 4. Gates (fp16)
    # --------------------------------------------------------------
    left_gate  = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate   = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction via Triton (fp16)
    # --------------------------------------------------------------
    # left/right have shape [B, N, N, hidden_dim] (fp16)
    out_fp16 = tri_mul_contraction(left, right)   # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection (float32)
    # --------------------------------------------------------------
    out = out_fp16.to(torch.float32)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
