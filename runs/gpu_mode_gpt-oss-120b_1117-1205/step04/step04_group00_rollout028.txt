def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Convert weights to half
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias = weights["norm.bias"].to(torch.float16)
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    to_out_weight = weights["to_out.weight"].to(torch.float16)

    # LayerNorm (half)
    x = input_tensor.to(torch.float16)
    # In PyTorch, dimension is last, so we compute layer_norm on dim
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    left = torch.nn.functional.linear(x, left_proj_weight)   # shape [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # Apply mask if needed
    if not nomask:
        # mask may be bool/int; convert to half
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Core contraction via batched GEMM kernel
    B, N, _, _ = left.shape  # B, N, N, H
    # Permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    total_batch = B * hidden_dim

    left_mat = left_perm.view(total_batch, N, N)
    right_mat = right_perm.view(total_batch, N, N)

    # Output tensor
    out_mat = torch.empty_like(left_mat)

    # Strides
    stride_a_batch, stride_a_m, stride_a_k = left_mat.stride()
    stride_b_batch, stride_b_row, stride_b_col = right_mat.stride()
    # For transposed B: use stride_b_k = stride_b_col, stride_b_n = stride_b_row
    stride_b_k = stride_b_col
    stride_b_n = stride_b_row
    stride_c_batch, stride_c_m, stride_c_n = out_mat.stride()

    # Block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (total_batch, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Reshape output back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Output layer norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Apply out_gate
    out = out * out_gate

    # Final linear
    out = torch.nn.functional.linear(out, to_out_weight)

    # Return float32
    return out.to(torch.float32)

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute base offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offs = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offs < K

        # Compute pointers
        a_ptrs = a_ptr + batch * stride_a_batch + offs_m[:, None] * stride_a_m + k_offs[None, :] * stride_a_k
        b_ptrs = b_ptr + batch * stride_b_batch + k_offs[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot accum
        acc += tl.dot(a, b, out_type=tl.float32)

    # Write back
    c_ptrs = c_ptr + batch * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass with a fused Triton batched GEMM kernel.
The kernel computes, for each batch and hidden channel, the matrix product
    out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]
which implements the einsum('... i k d, ... j k d -> ... i j d') in the
reference PyTorch implementation. All other operations (LayerNorm,
Linear projections, gating, masking) are performed in half precision with
PyTorch; the heavy N³ contraction is accelerated by the custom kernel.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    a_ptr,                 # half*   (B*H, M, K)
    b_ptr,                 # half*   (B*H, K, N)  (interpreted as transposed)
    c_ptr,                 # half*   (B*H, M, N)
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B^T  for each batch."""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        k_offs = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offs < K

        # Pointers for A and B (B is accessed as transposed)
        a_ptrs = a_ptr + batch_id * stride_a_batch \
                 + offs_m[:, None] * stride_a_m \
                 + k_offs[None, :] * stride_a_k
        b_ptrs = b_ptr + batch_id * stride_b_batch \
                 + k_offs[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b, out_type=tl.float32)

    # Write back
    c_ptrs = c_ptr + batch_id * stride_c_batch \
             + offs_m[:, None] * stride_c_m \
             + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---- 1️⃣ LayerNorm (half precision) ----
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)
    x = input_tensor.to(torch.float16)
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # ---- 2️⃣ Linear projections (to hidden_dim) ----
    w_left = weights["left_proj.weight"].to(torch.float16)
    w_right = weights["right_proj.weight"].to(torch.float16)

    left = torch.nn.functional.linear(x, w_left)   # [B, N, N, H]
    right = torch.nn.functional.linear(x, w_right) # [B, N, N, H]

    # ---- 3️⃣ Gates (sigmoid) ----
    w_lgate = weights["left_gate.weight"].to(torch.float16)
    w_rgate = weights["right_gate.weight"].to(torch.float16)
    w_ogate = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.nn.functional.linear(x, w_lgate).sigmoid()
    right_gate = torch.nn.functional.linear(x, w_rgate).sigmoid()
    out_gate = torch.nn.functional.linear(x, w_ogate).sigmoid()

    # ---- 4️⃣ Optional mask (broadcasted over hidden_dim) ----
    if not nomask:
        # mask shape: [B, N, N] -> [B, N, N, 1] then broadcast
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ---- 5️⃣ Core contraction via batched GEMM kernel ----
    B, N, _, _ = left.shape  # B, N, N, H
    # Rearrange to (B, H, N, N) for easy flattening
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    total_batch = B * hidden_dim
    left_mat = left_perm.view(total_batch, N, N)   # [B*H, N, N]
    right_mat = right_perm.view(total_batch, N, N) # [B*H, N, N]

    out_mat = torch.empty_like(left_mat)  # allocate output in half

    # Strides for A
    stride_a_batch, stride_a_m, stride_a_k = left_mat.stride()
    # Strides for B (will be accessed transposed)
    stride_b_batch, stride_b_i, stride_b_j = right_mat.stride()
    stride_b_k = stride_b_j          # column stride becomes K‑stride
    stride_b_n = stride_b_i          # row stride becomes N‑stride
    # Strides for C
    stride_c_batch, stride_c_m, stride_c_n = out_mat.stride()

    # Block sizes (tuned for typical Hopper/B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        total_batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout: [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ---- 6️⃣ Output LayerNorm, out‑gate scaling, final linear ----
    w_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_out_norm = weights["to_out_norm.bias"].to(torch.float16)

    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=w_out_norm, bias=b_out_norm
    )
    out = out * out_gate

    w_to_out = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, w_to_out)  # [B, N, N, dim]

    return out.to(torch.float32)
