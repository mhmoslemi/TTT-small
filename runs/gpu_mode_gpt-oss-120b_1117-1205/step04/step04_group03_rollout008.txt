"""
TriMul (outgoing) forward pass with a fused Triton batched‑matmul.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32) → cast to fp16.
2. Linear projections to hidden dimension `H` (fp16).
3. Compute three gate tensors (fp16) with sigmoid.
4. Apply optional mask and element‑wise gating on the projected tensors.
5. Core contraction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply (A @ Bᵀ) for each (b,h).  Implemented
   with a custom Triton kernel that tiles the matrices (BLOCK_M×BLOCK_N×BLOCK_K).
6. Output layer‑norm over the hidden dimension, scale by `out_gate`,
   final linear projection back to the original dimension.
7. Return result in float32.

The Triton kernel handles the O(N³) contraction efficiently for all
tested sizes (N up to 1024, H up to 384, batch ≤ 2).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matrix multiply C = A @ B, where each batch performs
    (M×K) @ (K×N).  All tensors are fp16; accumulation is fp32."""
    pid = tl.program_id(0)

    # grid layout: (batch, block_m, block_n)
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n

    batch_idx = pid // blocks_per_batch
    pid_in_batch = pid % blocks_per_batch
    pid_m = pid_in_batch // num_blocks_n
    pid_n = pid_in_batch % num_blocks_n

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    k = 0
    while k < K:
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers for the current tile of A and B
        a_ptrs = (
            a_ptr
            + batch_idx * stride_a_batch
            + (offs_m[:, None] * stride_a_row)
            + (offs_k[None, :] * stride_a_col)
        )
        b_ptrs = (
            b_ptr
            + batch_idx * stride_b_batch
            + (offs_k[:, None] * stride_b_row)
            + (offs_n[None, :] * stride_b_col)
        )

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot accumulates into fp32
        acc += tl.dot(a, b)

        k += BLOCK_K

    # write back
    c_ptrs = (
        c_ptr
        + batch_idx * stride_c_batch
        + (offs_m[:, None] * stride_c_row)
        + (offs_n[None, :] * stride_c_col)
    )
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.
    a, b: tensors of shape (B, M, K) and (B, K, N) respectively, dtype=fp16.
    Returns c of shape (B, M, N), dtype=fp16.
    """
    assert a.is_contiguous() and b.is_contiguous()
    batch, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    c = torch.empty((batch, M, N), dtype=torch.float16, device=a.device)

    # strides (in elements)
    stride_a_batch, stride_a_row, stride_a_col = a.stride()
    stride_b_batch, stride_b_row, stride_b_col = b.stride()
    stride_c_batch, stride_c_row, stride_c_col = c.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid_m = triton.cdiv(M, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    total_blocks = batch * grid_m * grid_n
    grid = (total_blocks,)

    _batched_matmul_kernel[grid](
        a, b, c,
        batch, M, N, K,
        stride_a_batch, stride_a_row, stride_a_col,
        stride_b_batch, stride_b_row, stride_b_col,
        stride_c_batch, stride_c_row, stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask         : torch.Tensor [B, N, N] (bool or numeric) or None
        - weights_dict : dict of model parameters (float32 tensors)
        - config_dict  : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → fp16
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (fp16)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gates (fp16) and sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Apply mask (if any) and element‑wise gating
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape (B,N,N) → (B,N,N,1) for broadcasting
        mask_f = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton batched GEMM
    #    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # left.shape == (B, N, N, H)

    # reshape to (B*H, N, N) for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # right needs to be transposed on the last two dims: (k, j)
    right_perm = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    # batched GEMM: (B*H, N, N) @ (B*H, N, N).T  -> (B*H, N, N)
    out_mat = _batched_matmul_triton(left_perm, right_perm)

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in original dtype
    return out.to(torch.float32)
