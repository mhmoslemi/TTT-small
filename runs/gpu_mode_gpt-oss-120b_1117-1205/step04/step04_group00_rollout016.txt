@triton.jit
def triton_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_k, stride_right_j,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids:
    pid_m = tl.program_id(0)  # i tile index
    pid_n = tl.program_id(1)  # j tile index
    pid_bh = tl.program_id(2)  # batch * head linear index

    # compute actual batch and head indices:
    b = pid_bh // H
    h = pid_bh % H

    # compute start offsets:
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for within N
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointer offsets:
    left_ptrs = left_ptr + b*stride_left_b + h*stride_left_h + offs_m[:,None]*stride_left_i + tl.arange(0, BLOCK_K)[None,:]*stride_left_k
    # Actually need build left and right for k loop.
    # For each k tile we load a block of left (M x K) and right (N x K) (since right is transposed for multiplication, we need right block as K x N?). We have right shape [B,H,N,N] and we need to multiply left (M x K) with right (N x K) but we need right transposed: right_{j,k} meaning we need right values at (k, n). So we can load right block as K x N by loading at offset (k axis, n axis). That requires stride_right_k and stride_right_j.

    # Right block ptr: location for K dimension (k) and j dimension (n):
    right_ptrs = right_ptr + b*stride_right_b + h*stride_right_h + tl.arange(0, BLOCK_K)[:,None]*stride_right_k + offs_n[None,:]*stride_right_j

    # accumulation:
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        # compute k indices:
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        left = tl.load(
            left_ptr + b*stride_left_b + h*stride_left_h + offs_m[:,None]*stride_left_i + offs_k[None,:]*stride_left_k,
            mask=mask_m[:,None] & mask_k[None,:],
            other=0.0
        )
        right = tl.load(
            right_ptr + b*stride_right_b + h*stride_right_h + offs_k[:,None]*stride_right_k + offs_n[None,:]*stride_right_j,
            mask=mask_k[:,None] & mask_n[None,:],
            other=0.0
        )
        # left shape M x K, right shape K x N
        acc += tl.dot(left, right)

    # Write the result
    out_ptrs = out_ptr + b*stride_out_b + h*stride_out_h + offs_m[:,None]*stride_out_i + offs_n[None,:]*stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:,None] & mask_n[None,:])

"""
Highly optimized forward pass for the outgoing TriMul operator.

The implementation fuses:
1) Gate + optional mask multiplication (Triton kernel).
2) The N³ contraction `out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]`
   implemented as a batched GEMM using a custom Triton kernel
   (left * rightᵀ for each head).

All other operations (LayerNorms, linear projections, final LayerNorm &
output projection) are performed with PyTorch in half precision for speed.
The final result is cast back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -----------------------------------------------------------
# Triton kernel: fuse gate multiplication with optional mask
# -----------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half*  (flattened left tensor)
    right_ptr,         # half*  (flattened right tensor)
    left_gate_ptr,     # half*  (flattened left gate)
    right_gate_ptr,    # half*  (flattened right gate)
    mask_ptr,          # half*  (flattened mask, size B*N*N)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 -> mask present, 0 -> no mask
    H: tl.constexpr,          # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # each hidden vector shares a mask entry -> mask index = offset // H
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store back in‑place
    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


# -----------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction
# -----------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    left_ptr,          # half*   [B, H, N, N]
    right_ptr,         # half*   [B, H, N, N] (will be read transposed)
    out_ptr,           # half*   [B, H, N, N]
    B: tl.int32,
    H: tl.int32,
    N: tl.int32,
    # strides for left
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    # strides for right (note: we need k and j strides)
    stride_right_b, stride_right_h, stride_right_k, stride_right_j,
    # strides for out
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile row (i)
    pid_n = tl.program_id(1)   # tile col (j)
    pid_bh = tl.program_id(2)  # combined batch*head index

    # Recover batch and head indices
    b = pid_bh // H
    h = pid_bh - b * H

    # Offsets for the tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this (b, h) slice
    base_left = left_ptr + b * stride_left_b + h * stride_left_h
    base_right = right_ptr + b * stride_right_b + h * stride_right_h
    base_out = out_ptr + b * stride_out_b + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block [M, K]
        left = tl.load(
            base_left
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load right block transposed -> shape [K, N]
        right = tl.load(
            base_right
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_j,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        # Accumulate
        acc += tl.dot(left, right)

    # Write back result
    out_ptrs = base_out + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# -----------------------------------------------------------
# Helper to apply gate & mask fusion
# -----------------------------------------------------------
def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    `mask` may be None.
    All tensors must be contiguous torch.float16.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when no mask

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )


# -----------------------------------------------------------
# Main entry point
# -----------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored if nomask=True
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------------
    # 1. Input LayerNorm (float32) -> half
    # -------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # -------------------------------------------------------
    # 2. Linear projections to hidden dimension (half)
    # -------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # -------------------------------------------------------
    # 3. Gates (half)
    # -------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # -------------------------------------------------------
    # 4. Fuse mask with gates via Triton (if mask present)
    # -------------------------------------------------------
    if not nomask:
        # Flatten tensors for the kernel (contiguous)
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask.to(torch.float16).contiguous().view(-1)  # [B*N*N]

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            hidden_dim,
        )
        # Reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # -------------------------------------------------------
    # 5. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as a batched GEMM (left * rightᵀ) via Triton.
    # -------------------------------------------------------
    # Rearrange to [B, H, N, N] for GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape

    out_perm = torch.empty_like(left_perm)  # (B, H, N, N), half

    # Strides (in elements, not bytes)
    s_lb = left_perm.stride(0)
    s_lh = left_perm.stride(1)
    s_li = left_perm.stride(2)
    s_lk = left_perm.stride(3)

    s_rb = right_perm.stride(0)
    s_rh = right_perm.stride(1)
    s_rk = right_perm.stride(3)   # k is last dimension
    s_rj = right_perm.stride(2)   # j is the other N dimension

    s_ob = out_perm.stride(0)
    s_oh = out_perm.stride(1)
    s_oi = out_perm.stride(2)
    s_oj = out_perm.stride(3)

    # Block sizes – tuned for typical B200 memory hierarchy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )
    _trimul_gemm_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        B,
        H,
        N,
        s_lb,
        s_lh,
        s_li,
        s_lk,
        s_rb,
        s_rh,
        s_rk,
        s_rj,
        s_ob,
        s_oh,
        s_oi,
        s_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Rearrange back to original layout [B, N, N, H]
    out = out_perm.permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # -------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # -------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise scaling (half)

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Cast back to float32 for the final output
    return out.to(torch.float32)
