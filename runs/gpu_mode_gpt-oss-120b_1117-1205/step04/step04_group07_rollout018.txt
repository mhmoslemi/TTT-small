@triton.jit
def mask_gate_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements, HIDDEN_DIM,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elements

    left = tl.load(left_ptr + offs, mask=m)
    right = tl.load(right_ptr + offs, mask=m)
    left_gate = tl.load(left_gate_ptr + offs, mask=m)
    right_gate = tl.load(right_gate_ptr + offs, mask=m)

    if HAS_MASK:
        idx_mask = offs // HIDDEN_DIM
        mask_val = tl.load(mask_ptr + idx_mask, mask=m)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=m)
    tl.store(right_ptr + offs, right, mask=m)
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    # Cast weights to half
    w_norm_weight = weights["norm.weight"].to(device, dtype=torch.float16)
    w_norm_bias = weights["norm.bias"].to(device, dtype=torch.float16)
    w_left_proj = weights["left_proj.weight"].to(device, dtype=torch.float16)
    w_right_proj = weights["right_proj.weight"].to(device, dtype=torch.float16)
    w_left_gate = weights["left_gate.weight"].to(device, dtype=torch.float16)
    w_right_gate = weights["right_gate.weight"].to(device, dtype=torch.float16)
    w_out_gate = weights["out_gate.weight"].to(device, dtype=torch.float16)
    w_to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype=torch.float16)
    w_to_out_norm_bias = weights["to_out_norm.bias"].to(device, dtype=torch.float16)
    w_to_out = weights["to_out.weight"].to(device, dtype=torch.float16)

    # 1. LayerNorm (half)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=w_norm_weight,
        bias=w_norm_bias,
    )
    x = x.to(torch.float16)

    # 2. Linear projections (half)
    left = torch.nn.functional.linear(x, w_left_proj)    # bias None
    right = torch.nn.functional.linear(x, w_right_proj)

    # 3. Gates (half)
    left_gate = torch.nn.functional.linear(x, w_left_gate).sigmoid()
    right_gate = torch.nn.functional.linear(x, w_right_gate).sigmoid()
    out_gate = torch.nn.functional.linear(x, w_out_gate).sigmoid()

        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        # fuse
        apply_gate_mask_fuse(left, right, left_gate, right_gate, mask_flat, hidden_dim)
    else:
        left = left * left_gate
        right = right * right_gate

    left_mat = left.view(B * H, N, N)

    HAS_MASK: tl.constexpr,

        right = right * right_gate

        torch.Tensor of shape [B, N, N, dim] (float32)
    """

    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Cast weights to half‑precision once.
    # ------------------------------------------------------------------
    w_norm_weight = weights["norm.weight"].to(device, dtype=torch.float16)
    w_norm_bias = weights["norm.bias"].to(device, dtype=torch.float16)

    w_left_proj = weights["left_proj.weight"].to(device, dtype=torch.float16)
    w_right_proj = weights["right_proj.weight"].to(device, dtype=torch.float16)

    w_left_gate = weights["left_gate.weight"].to(device, dtype=torch.float16)
    w_right_gate = weights["right_gate.weight"].to(device, dtype=torch.float16)
    w_out_gate = weights["out_gate.weight"].to(device, dtype=torch.float16)

    w_to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype=torch.float16)
    w_to_out_norm_bias = weights["to_out_norm.bias"].to(device, dtype=torch.float16)

    w_to_out = weights["to_out.weight"].to(device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 2. Input LayerNorm (half precision)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=w_norm_weight,
        bias=w_norm_bias,
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections to hidden dimension (half)
    # ------------------------------------------------------------------
    left = F.linear(x, w_left_proj)   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------

    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

        bias=w_to_out_norm_bias,
    )
    out = out * out_gate
    out = F.linear(out, w_to_out)

    # 8. Return float‑32 tensor
    # ------------------------------------------------------------------
    return out.to(torch.float32)

    w_to_out = weights["to_out.weight"].to(device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 2. Input LayerNorm (FP16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=w_norm_weight,
        bias=w_norm_bias,
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections to hidden dimension (FP16)
    # ------------------------------------------------------------------
    left = F.linear(x, w_left_proj)   # (B, N, N, hidden_dim)
    right = F.linear(x, w_right_proj)

    # ------------------------------------------------------------------
    # 4. Compute sigmoid gates (FP16)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, w_left_gate).sigmoid()
    right_gate = F.linear(x, w_right_gate).sigmoid()
    out_gate = F.linear(x, w_out_gate).sigmoid()

    # ------------------------------------------------------------------
    # 5. Fuse mask (if present) with gates via Triton; otherwise simple mul
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask.to(torch.float16).contiguous()
        mask_flat = mask_h.view(-1)                     # [B*N*N]
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        _apply_mask_gate_fuse(left, right, left_gate, right_gate, mask_flat, hidden_dim)
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Core contraction (batched GEMM in FP16)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # reshape to (B, hidden_dim, N, N) → (B*hidden_dim, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left.shape
    left_mat = left.view(B * H, N, N)          # (B*H, N, N)
    right_mat = right.view(B * H, N, N)        # (B*H, N, N)

    # Batched matrix multiplication: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate scaling and final linear projection (FP16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm_weight,
        bias=w_to_out_norm_bias,
    )
    out = out * out_gate
    out = F.linear(out, w_to_out)

    # ------------------------------------------------------------------
    # 8. Cast back to FP32 for the returned tensor
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
