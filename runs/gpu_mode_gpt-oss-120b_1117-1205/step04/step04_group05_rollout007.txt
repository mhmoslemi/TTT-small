def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Cast input to FP32? We'll keep as FP32 for layernorm then half.
    # Normalize
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # Convert to half for subsequent ops
    x_h = x.to(torch.float16)

    # Linear projections
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16), bias=None)

    # Gates
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # Fuse mask and gates using Triton kernel if needed
    if not nomask and mask is not None:
        # Convert mask to half and flatten
        mask_h = mask.to(torch.float16).contiguous().view(-1)  # shape B*N*N
        # Flatten tensors
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(left_f, right_f, left_gate_f, right_gate_f, mask_h, hidden_dim)

        # Reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # Free gates to release memory
    del left_gate, right_gate, x_h

    # Core contraction: batched matmul
    B, N, _, _ = left.shape  # left shape (B,N,N,hidden_dim)
    # reshape for batched GEMM
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H,N,N)
    # reshape back to (B,N,N,H)
    out = out_mat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Output layernorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16)
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    # Cast back to float32
    return out.to(torch.float32)

"""
Optimized outgoing TriMul forward pass.

Algorithm
---------
1. Layer‑norm the input tensor (float32 → half).
2. Linear projections to hidden dimension (half‑precision).
3. Compute sigmoid gates (half‑precision).
4. Fuse optional mask with left/right gates using a small Triton kernel
   (in‑place elementwise multiplication, no extra memory copies).
5. Core O(N³) contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   implemented as a batched GEMM (torch.bmm) on the (B·H, N, N) matrix
   view of the tensors, fully exploiting Tensor‑cores.
6. Output layer‑norm, out‑gate scaling and final linear projection back
   to the original dimension.  Result is cast back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel: fuse mask (if any) with left/right gates.
# -------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # half* left tensor (flattened)
    right_ptr,              # half* right tensor (flattened)
    left_gate_ptr,          # half* left‑gate tensor (flattened)
    right_gate_ptr,         # half* right‑gate tensor (flattened)
    mask_ptr,               # half* mask tensor (flattened, length = total/H)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present, else 0
    H: tl.constexpr,          # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elements

    left = tl.load(left_ptr + offs, mask=m, other=0.0)
    right = tl.load(right_ptr + offs, mask=m, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=m, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=m, other=0.0)

    if HAS_MASK:
        # one mask value per (batch, i, k) group of H hidden values
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=m, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=m)
    tl.store(right_ptr + offs, right, mask=m)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of mask (if present) and sigmoid gates using Triton.
    All tensors must be contiguous half‑precision.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when mask absent

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # No return; tensors are modified in‑place.


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor[B, N, N, dim] (float32)
            - mask: torch.Tensor[B, N, N] (bool/float) – may be ignored
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor[B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half‑precision for the rest
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half)
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16), bias=None)

    # ------------------------------------------------------------------
    # 3. Gating (half)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) with gates using Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).contiguous().view(-1)          # [B*N*N]
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(left_f, right_f, left_gate_f, right_gate_f, mask_h, hidden_dim)

        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # Release temporary tensors
    del left_gate, right_gate, x_h

    # ------------------------------------------------------------------
    # 5. Core contraction: batched GEMM over the sequence dimension
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # left shape: (B, N, N, H)
    # Convert to (B*H, N, N) without extra copies
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N) # (B*H, N, N)

    # Batched matrix multiplication with transpose on the second operand
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_mat.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    # Cast back to float32 for the final output
    return out.to(torch.float32)
