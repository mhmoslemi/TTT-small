"""
tri_mul_outgoing.py

Implementation of the "outgoing" Triangle Multiplicative Update (TriMul)
forward pass as described in AlphaFold3. The heavy mask‑gate fusion is
performed with a custom Triton kernel, while the O(N³) contraction is
executed as a batched half‑precision GEMM (torch.bmm), which leverages
cuBLAS for maximum performance.

The kernel:
    _gate_mask_fuse_kernel
fuses
    left = left * left_gate * mask
    right = right * right_gate * mask
(in the absence of a mask the multiplication with ``mask`` is omitted).

All other operations are carried out in PyTorch using half precision,
except the final output which is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half* (flattened left tensor)
    right_ptr,         # half* (flattened right tensor)
    left_gate_ptr,     # half* (flattened left gate)
    right_gate_ptr,    # half* (flattened right gate)
    mask_ptr,          # half* (flattened mask, broadcasted over hidden dim)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if a mask is present, 0 otherwise
    H: tl.constexpr,          # hidden dimension (channels)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load data (half precision)
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # mask is broadcasted over the hidden dimension, thus each
        # hidden entry shares the same mask entry.
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(
    left: torch.Tensor,
    right: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor | None,
    hidden_dim: int,
) -> None:
    """
    In‑place fusion of ``left = left * left_gate * mask`` and
    ``right = right * right_gate * mask`` using a Triton kernel.
    All tensors must be contiguous ``torch.float16`` tensors.
    """
    total = left.numel()
    BLOCK = 1024  # reasonable block size for 1‑D kernel
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    # If mask is absent we still need to pass a pointer; we give the left
    # tensor (its contents will never be read because HAS_MASK=0).
    mask_ptr = mask if mask is not None else left

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # No return – tensors are modified in place.


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or float) – may be ignored if config["nomask"] is True
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32) -> half precision
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections to hidden dimension (half)
    # -----------------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # -----------------------------------------------------------------
    # 3. Gates (half)
    # -----------------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # -----------------------------------------------------------------
    # 4. Fuse mask and gates via Triton (if mask present)
    # -----------------------------------------------------------------
    if not nomask:
        # Ensure contiguous half‑precision tensors for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        # mask broadcasted over hidden_dim, shape [B, N, N] -> [B*N*N]
        mask_f = mask.to(torch.float16).contiguous().view(-1)

        _apply_gate_and_mask(
            left_f, right_f, left_gate_f, right_gate_f, mask_f, hidden_dim
        )

        # Reshape back to original layout
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # Simple element‑wise multiplication when no mask is used
        left = left * left_gate
        right = right * right_gate

    # Gates no longer needed
    del left_gate, right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction:
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as a batched GEMM per (b, d) pair.
    # -----------------------------------------------------------------
    # Permute to (B, H, N, N) where H = hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    # Right needs the k‑dimension as the last axis for the transpose
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) where last two axes are swapped

    B, H, N, _ = left_perm.shape
    # Collapse batch and head dimensions for a single batched matmul call
    left_mat = left_perm.view(B * H, N, N)    # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)  # (B*H, N, N)

    # Batched matrix multiplication (half‑precision)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise gate (half)

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in original dtype (float32)
    return out.to(torch.float32)
