def custom_kernel(data):
    input_tensor, mask, weights, config = data

    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])
    # cast to fp16
    x_h = x.to(torch.float16)

    B, N, N2, _ = x_h.shape
    assert N == N2
    M = B * N * N

    # flatten
    x_flat = x_h.view(M, dim)

    # combine weights
    # extract raw weight tensors
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    # all have shape (hidden_dim, dim)
    combined_w = torch.cat([left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w], dim=0).to(torch.float16)

    # linear
    combined = torch.nn.functional.linear(x_flat, combined_w)  # (M, 5*hidden_dim)
    # free combined weight maybe
    del combined_w

    # split
    left_proj, left_gate_raw, right_proj, right_gate_raw, out_gate_raw = torch.split(combined, hidden_dim, dim=1)
    del combined

    # sigmoid gates using torch (fast)
    left_gate = left_gate_raw.sigmoid()
    right_gate = right_gate_raw.sigmoid()
    out_gate = out_gate_raw.sigmoid()
    del left_gate_raw, right_gate_raw, out_gate_raw

    # fuse gating and mask into left and right
    # we will reuse left_proj and right_proj buffers to store the final left/right
    left = left_proj
    right = right_proj

    # mask handling
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).contiguous().view(-1)  # shape (M,)
    else:
        # dummy mask
        mask_h = left  # just pass left pointer; not used when HAS_MASK=0
    # call kernel
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    _apply_gate_mask_kernel[grid](
        left, left_gate,
        right, right_gate,
        mask_h,
        total, H=hidden_dim,
        BLOCK=BLOCK,
        HAS_MASK=0 if nomask else 1
    )
    # after kernel, left and right hold gated (and masked) values.
    # no need to keep left_gate/right_gate
    del left_gate, right_gate

    # reshape left/right
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

    # contraction
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B_, H_, N_, _ = left_perm.shape
    left_mat = left_perm.view(B_ * H_, N_, N_)
    right_mat = right_perm.view(B_ * H_, N_, N_)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(B_, H_, N_, N_).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # Output layernorm
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=weights["to_out_norm.weight"].to(torch.float16),
                       bias=weights["to_out_norm.bias"].to(torch.float16))
    # multiply by out_gate
    out = out * out_gate.view(B, N, N, hidden_dim)
    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass – highly optimized version.

Algorithm
---------
1. Layer‑norm the input (float16).
2. Flatten to (M, dim) where M = B·N·N.
3. Fuse **all** linear projections (left_proj, left_gate, right_proj,
   right_gate, out_gate) into a single half‑precision GEMM by concatenating
   the weight matrices → (M, 5·hidden_dim).
4. Split the result, apply sigmoid to the three gate tensors.
5. Fuse left/right gating and optional mask in a custom Triton kernel
   (in‑place, eliminates separate element‑wise passes).
6. Reshape left/right to (B, N, N, hidden_dim) and compute the
   O(N³) contraction as a batched GEMM:
        out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
7. Apply LayerNorm on the hidden dimension, scale by out_gate,
   and a final linear projection back to dim.
8. Return the result in float32.

Only the gating‑mask fusion is performed in Triton; the remaining
operations use torch’s highly tuned kernels (layer‑norm, GEMM, etc.).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,                # flatten mask (M,) – ignored if HAS_MASK==0
    total_elements: tl.int32,
    H: tl.constexpr,         # hidden_dim
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,  # 1 → apply mask, 0 → skip
):
    """In‑place: left = left * sigmoid(left_gate) * mask,
                 right = right * sigmoid(right_gate) * mask"""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    # load values
    left = tl.load(left_ptr + offs, mask=active, other=tl.float16(0))
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=tl.float16(0))
    right = tl.load(right_ptr + offs, mask=active, other=tl.float16(0))
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=tl.float16(0))

    # sigmoid (1/(1+exp(-x))) in half precision
    left_sig = 1.0 / (1.0 + tl.exp(-left_gate))
    right_sig = 1.0 / (1.0 + tl.exp(-right_gate))

    if HAS_MASK:
        # broadcast mask across hidden_dim
        row = offs // H
        mask_val = tl.load(mask_ptr + row, mask=active, other=tl.float16(1.0))
        left = left * left_sig * mask_val
        right = right * right_sig * mask_val
    else:
        left = left * left_sig
        right = right * right_sig

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool / numeric) – may be ignored
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------------------
    # 1. Input LayerNorm (float16 for speed)
    # -------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)  # shape [B, N, N, dim]

    B, N, N2, _ = x.shape
    assert N == N2
    M = B * N * N                # number of (b,i,j) positions

    # -------------------------------------------------------------
    # 2. Flatten and perform **one** large GEMM that computes all
    #    linear projections and gates in a single pass.
    # -------------------------------------------------------------
    x_flat = x.view(M, dim)                     # (M, dim)
    # Concatenate all five weight matrices (left_proj, left_gate,
    # right_proj, right_gate, out_gate) → shape (5*hidden_dim, dim)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    combined = F.linear(x_flat, combined_weight)  # (M, 5*hidden_dim)
    del combined_weight, x_flat, x  # free memory early

    # -------------------------------------------------------------
    # 3. Split combined result and apply sigmoid to gate tensors.
    # -------------------------------------------------------------
    left_proj, left_gate_raw, right_proj, right_gate_raw, out_gate_raw = torch.split(
        combined, hidden_dim, dim=1
    )
    del combined  # free

    left_gate = left_gate_raw.sigmoid()
    right_gate = right_gate_raw.sigmoid()
    out_gate = out_gate_raw.sigmoid()          # (M, hidden_dim)
    del left_gate_raw, right_gate_raw, out_gate_raw

    # -------------------------------------------------------------
    # 4. Fuse gating (and optional mask) for left/right using Triton.
    #    The operation is performed in‑place on left_proj / right_proj.
    # -------------------------------------------------------------
    total = left_proj.numel()                  # M * hidden_dim
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).contiguous().view(-1)  # (M,)
        has_mask = 1
    else:
        # dummy pointer – never read because HAS_MASK == 0
        mask_h = left_proj
        has_mask = 0

    _gate_mask_fuse_kernel[grid](
        left_proj,
        left_gate,
        right_proj,
        right_gate,
        mask_h,
        total,
        H=hidden_dim,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
    )
    # left_proj and right_proj now hold the gated (and masked) values
    del left_gate, right_gate, mask_h

    # -------------------------------------------------------------
    # 5. Reshape for the O(N³) contraction.
    # -------------------------------------------------------------
    left = left_proj.view(B, N, N, hidden_dim)
    right = right_proj.view(B, N, N, hidden_dim)
    del left_proj, right_proj

    # Permute to (B, hidden_dim, N, N) for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    del left, right

    B_, H_, N_, _ = left_perm.shape
    left_mat = left_perm.view(B_ * H_, N_, N_)
    right_mat = right_perm.view(B_ * H_, N_, N_)

    # Batched matrix multiplication: (B*H, N, N) x (B*H, N, N)^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    del left_mat, right_mat, left_perm, right_perm

    out = (
        out_mat.view(B_, H_, N_, N_)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B, N, N, hidden_dim)

    # -------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling, final linear projection.
    # -------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out_gate = out_gate.view(B, N, N, hidden_dim)
    out = out * out_gate

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )
    return out.to(torch.float32)
