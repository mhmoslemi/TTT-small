"""
TriMul “outgoing” forward pass with Triton‑fused mask & gate.

Algorithm:
1. Layer‑norm the input tensor.
2. Linear projections → left, right (half precision).
3. Compute gate values (half) and optional mask.
4. Fuse mask, left‑gate and right‑gate with a tiny Triton kernel
   (in‑place multiplication on left/right).
5. Core contraction:
   For each hidden channel d compute
       out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]
   This is a batched matrix multiplication:
       reshape left/right to (B*H, N, N) and call torch.bmm.
6. Output LayerNorm, out‑gate, final linear projection.
7. Return float32 tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half *
    right_ptr,         # half *
    left_gate_ptr,     # half *
    right_gate_ptr,    # half *
    mask_ptr,          # half *
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present, 0 otherwise
    H: tl.constexpr,          # hidden dimension
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elems

    # load tensors
    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    if HAS_MASK:
        # each mask element corresponds to H consecutive channel entries
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def _fuse_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """In‑place fusion of mask & gates via Triton."""
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when no mask

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # left and right are updated in‑place


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (torch.Tensor)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> half)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )  # [B, N, N, hidden_dim]

    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gate values (half)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) and gates via Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] -> broadcast over hidden_dim
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        _fuse_gate_and_mask(left, right, left_gate, right_gate, mask_h, hidden_dim)
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction using batched GEMM
    #    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_reshape = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_reshape = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # batched matrix multiplication (right transposed on the second axis)
    out = torch.bmm(left_reshape, right_reshape.transpose(1, 2))
    # reshape back to [B, N, N, H]
    out = out.reshape(B, H, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 7. Return in float32
    # ------------------------------------------------------------------
    return out.to(torch.float32)
