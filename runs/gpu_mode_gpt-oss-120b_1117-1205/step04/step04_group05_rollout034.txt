@triton.jit
def trmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid0 = tl.program_id(0)  # batch*hidden index
    pid1 = tl.program_id(1)  # block row i
    pid2 = tl.program_id(2)  # block col j

    batch_h = pid0
    batch = batch_h // H
    h = batch_h % H

    i0 = pid1 * BLOCK_M
    j0 = pid2 * BLOCK_N

    # offsets for rows and cols
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    # masks for out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_tile * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile [M, K]
        left_ptrs = left_ptr + batch*stride_left_b + offs_i[:,None]*stride_left_i + offs_k[None,:]*stride_left_k + h*stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:,None] & mask_k[None,:], other=0.0).to(tl.float32)

        # load right tile transposed [K, N] (k as first dim, j as second)
        # right shape: [B,N,N,H] but we need right[batch, j, k, h] -> we get transposed by swapping i and k offsets.
        # To load right as (K, N):
        right_ptrs = right_ptr + batch*stride_right_b + offs_k[:,None]*stride_right_i + offs_j[None,:]*stride_right_k + h*stride_right_h
        right = tl.load(right_ptrs, mask=mask_k[:,None] & mask_j[None,:], other=0.0).to(tl.float32)

        # Compute dot
        acc += tl.dot(left, right)

    # Write output
    out_ptrs = out_ptr + batch*stride_out_b + offs_i[:,None]*stride_out_i + offs_j[None,:]*stride_out_j + h*stride_out_h
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:,None] & mask_j[None,:])

@triton.jit
def gate_fuse_kernel(
   left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr,
   total: tl.int32,
   hidden_dim: tl.constexpr,
   HAS_MASK: tl.constexpr,
   BLOCK: tl.constexpr
):
   pid = tl.program_id(0)
   offs = pid * BLOCK + tl.arange(0, BLOCK)
   mask = offs < total
   # load
   left = tl.load(left_ptr + offs, mask=mask, other=0.0)
   right = tl.load(right_ptr + offs, mask=mask, other=0.0)
   left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
   right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
   if HAS_MASK:
       # each group of hidden_dim elements shares mask value: mask shape (B,N,N)
       idx = offs // hidden_dim
       mask_val = tl.load(mask_ptr + idx, mask=mask, other=0.0)
       left = left * left_gate * mask_val
       right = right * right_gate * mask_val
   else:
       left = left * left_gate
       right = right * right_gate
   tl.store(left_ptr + offs, left, mask=mask)
   tl.store(right_ptr + offs, right, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', True)

    #1 layernorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'])
    x = x.to(torch.float16)

    #2 projection
    left = torch.nn.functional.linear(x, weights['left_proj.weight'].to(torch.float16))
    right = torch.nn.functional.linear(x, weights['right_proj.weight'].to(torch.float16))

    #3 gates
    left_gate = torch.nn.functional.linear(x, weights['left_gate.weight'].to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights['right_gate.weight'].to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights['out_gate.weight'].to(torch.float16)).sigmoid()

    #4 fuse masking+gating
    # Ensure contiguous
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    total = left.numel()
    BLOCK = 1024  # maybe 1024 elements per block
    # mask handling
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).contiguous()  # shape [B,N,N]
    else:
        # dummy mask all ones (no op), create a tensor of ones of shape [B,N,N] to pass to kernel
        mask_h = torch.ones_like(input_tensor[...,0], dtype=torch.float16, device=device)  # shape B,N,N

    # Flatten
    left_flat = left.view(-1)
    right_flat = right.view(-1)
    left_gate_flat = left_gate.view(-1)
    right_gate_flat = right_gate.view(-1)
    mask_flat = mask_h.view(-1)

    # launch kernel
    grid = (triton.cdiv(total, BLOCK),)
    _gate_fuse_kernel[grid](
        left_flat,
        right_flat,
        left_gate_flat,
        right_gate_flat,
        mask_flat,
        total,
        hidden_dim=hidden_dim,
        HAS_MASK=int(not nomask and mask is not None),
        BLOCK=BLOCK,
    )
    # reshape after kernel
    left = left_flat.view_as(left)
    right = right_flat.view_as(right)

    #5 core contraction
    # allocate output tensor
    out = torch.empty_like(left)  # shape B,N,N,hidden_dim
    # get strides
    B, N, _, H = left.shape
    # compute strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()
    # define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # compute grid dimensions
    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    trmul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    #6 output normalization + out_gate scaling + final linear
    # out shape: B,N,N,hidden_dim
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights['to_out_norm.weight'].to(torch.float16), bias=weights['to_out_norm.bias'].to(torch.float16))
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights['to_out.weight'].to(torch.float16))
    # cast back to float32
    return out.to(torch.float32)

"""
TriMul outgoing kernel (AlphaFold3) – fast forward pass.

The implementation follows the paper:
1. Layer‑norm the input tensor.
2. Linear projections to a hidden dimension (no bias).
3. Compute sigmoid gating tensors.
4. Fuse mask (if present) and the two gates into the projected tensors
   with a tiny Triton kernel (in‑place, half‑precision).
5. Core O(N³) contraction:
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM in Triton (one tile per (batch,channel)
   and a three‑dimensional grid for the (i,j) tiles).  Accumulation is
   performed in float32 and the result is stored as float16.
6. Output layer‑norm, out‑gate scaling and a final linear projection back
   to the original channel dimension.

All heavy arithmetic is performed in float16 (except the accumulation)
to reduce memory bandwidth while keeping numerical errors well within
the tolerance of the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Mask‑&‑gate fusion kernel (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _gate_fuse_kernel(
    left_ptr,               # half*  – flattened left tensor
    right_ptr,              # half*  – flattened right tensor
    left_gate_ptr,          # half*  – flattened left‑gate tensor
    right_gate_ptr,         # half*  – flattened right‑gate tensor
    mask_ptr,               # half*  – flattened mask (B×N×N) – used only if HAS_MASK=1
    total_elems,            # tl.int32 – total number of elements in left/right
    hidden_dim: tl.constexpr,
    HAS_MASK: tl.constexpr,  # 1 → use mask, 0 → skip mask
    BLOCK: tl.constexpr,     # thread‑block size (elements)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # one mask value per (b,i,k) – each group of hidden_dim values shares it
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


# ----------------------------------------------------------------------
# 2️⃣  Core O(N³) contraction – batched GEMM over the hidden dimension
# ----------------------------------------------------------------------
@triton.jit
def _trmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    NUM_K_TILES: tl.constexpr,   # ceil(N / BLOCK_K)
):
    pid0 = tl.program_id(0)          # batch × hidden
    pid1 = tl.program_id(1)          # tile row (i)
    pid2 = tl.program_id(2)          # tile col (j)

    batch_h = pid0
    batch = batch_h // H
    h = batch_h % H

    i0 = pid1 * BLOCK_M
    j0 = pid2 * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_tile in range(NUM_K_TILES):
        k0 = k_tile * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- left tile: (M, K) ------------------------------------
        left_ptrs = (
            left_ptr
            + batch * stride_left_b
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
            + h * stride_left_h
        )
        left_tile = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # ---- right tile transposed: (K, N) -----------------------
        # we load right[j, k] as (K, N) by swapping the i/k strides
        right_ptrs = (
            right_ptr
            + batch * stride_right_b
            + offs_k[:, None] * stride_right_k   # rows → k dimension
            + offs_j[None, :] * stride_right_i   # cols → j dimension
            + h * stride_right_h
        )
        right_tile = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        acc += tl.dot(left_tile, right_tile)

    # ---- store result ------------------------------------------------
    out_ptrs = (
        out_ptr
        + batch * stride_out_b
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
        + h * stride_out_h
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# 3️⃣  Entrypoint – combines all steps
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) – may be None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm  (float32 → half)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections  (half precision)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16), bias=None)

    # --------------------------------------------------------------
    # 3. Gates (half precision)
    # --------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # --------------------------------------------------------------
    # 4. Fuse mask + gates (in‑place) using Triton
    # --------------------------------------------------------------
    # make everything contiguous for the kernel
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    total = left.numel()                     # B * N * N * hidden_dim
    BLOCK = 1024                             # kernel block size (elements)
    has_mask = 0
    if not nomask and mask is not None:
        has_mask = 1
        mask_tensor = mask.to(torch.float16).contiguous()
    else:
        # dummy mask of all ones – never read when HAS_MASK=0
        mask_tensor = torch.ones_like(input_tensor[..., 0], dtype=torch.float16, device=device)

    # flatten for 1‑D kernel launch
    left_flat = left.view(-1)
    right_flat = right.view(-1)
    left_gate_flat = left_gate.view(-1)
    right_gate_flat = right_gate.view(-1)
    mask_flat = mask_tensor.view(-1)

    grid = (triton.cdiv(total, BLOCK),)
    _gate_fuse_kernel[grid](
        left_flat,
        right_flat,
        left_gate_flat,
        right_gate_flat,
        mask_flat,
        total,
        hidden_dim=hidden_dim,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )
    # tensors have been updated in‑place
    left = left_flat.view_as(left)
    right = right_flat.view_as(right)

    # --------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # same dtype (half) and shape

    # strides for the (B, N, N, H) layout
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # tile sizes – chosen to give good occupancy on B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    NUM_K_TILES = (N + BLOCK_K - 1) // BLOCK_K

    grid = (
        B * H,                         # batch × hidden
        triton.cdiv(N, BLOCK_M),      # i‑tiles
        triton.cdiv(N, BLOCK_N),      # j‑tiles
    )
    _trmul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        NUM_K_TILES=NUM_K_TILES,
    )

    # --------------------------------------------------------------
    # 6. Output layer‑norm, out‑gate scaling and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    # back to float32 as required by the interface
    return out.to(torch.float32)
```
