Define kernel:

@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    B: tl.int32,
    M: tl.int32, N: tl.int32, K: tl.int32,
    stride_ah: tl.int32, stride_ai: tl.int32, stride_ak: tl.int32,
    stride_bh: tl.int32, stride_bk: tl.int32, stride_bj: tl.int32,
    stride_ch: tl.int32, stride_ci: tl.int32, stride_cj: tl.int32,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets:
    off_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Compute mask for out-of-bounds:
    mask_i = off_i < M
    mask_j = off_j < N

    # Pointers to output tile:
    c_ptrs = c_ptr + batch_id * stride_ch \
                + off_i[:, None] * stride_ci \
                + off_j[None, :] * stride_cj

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)  # compute in fp32

    # Loop over k dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Offsets for k
        off_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = off_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + batch_id * stride_ah \
                   + off_i[:, None] * stride_ai \
                   + off_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B tile: shape (BLOCK_K, BLOCK_N) (since B_T)
        b_ptrs = b_ptr + batch_id * stride_bh \
                   + off_k[:, None] * stride_bk \
                   + off_j[None, :] * stride_bj
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Compute dot product
        acc += tl.dot(a, b)

    # Cast back to half and write
    c = acc.to(tl.float16)

@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH: tl.constexpr, H: tl.constexpr,
    N: tl.constexpr,
    stride_ab: tl.constexpr,  # batch stride for A
    stride_ah: tl.constexpr,  # hidden stride for A
    stride_ai: tl.constexpr,  # row stride (i) for A
    stride_ak: tl.constexpr,  # col stride (k) for A
    stride_bb: tl.constexpr,  # batch stride for B
    stride_bh: tl.constexpr,  # hidden stride for B
    stride_bk: tl.constexpr,  # row stride for B_T (k)
    stride_bj: tl.constexpr,  # col stride for B_T (j)
    stride_cb: tl.constexpr,  # batch stride for C
    stride_ch: tl.constexpr,  # hidden stride for C
    stride_ci: tl.constexpr,  # row stride for C (i)
    stride_cj: tl.constexpr,  # col stride for C (j)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,

    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    b_idx = batch_id // H
    h_idx = batch_id % H

    j_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH: tl.constexpr, H: tl.constexpr, N: tl.constexpr,
    stride_ab: tl.constexpr, stride_ah: tl.constexpr,
    stride_ai: tl.constexpr, stride_ak: tl.constexpr,
    stride_bb: tl.constexpr, stride_bh: tl.constexpr,
    stride_bk: tl.constexpr, stride_bj: tl.constexpr,
    stride_cb: tl.constexpr, stride_ch: tl.constexpr,
    stride_ci: tl.constexpr, stride_cj: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute batch and hidden indices
    b_idx = batch_id // H
    h_idx = batch_id % H

    # computed i/j offsets
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    i_mask = i < N
    j_mask = j < N

    # base pointers for this batch+hidden
    a_base = a_ptr + b_idx * stride_ab + h_idx * stride_ah
    b_base = b_ptr + b_idx * stride_bb + h_idx * stride_bh
    c_base = c_ptr + b_idx * stride_cb + h_idx * stride_ch

    # pointer to output tile
    c_ptrs = c_base + i[:, None] * stride_ci + j[None, :] * stride_cj

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k < N

        # Load A tile
        a_ptrs = a_base + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)

        # Load B tile (transposed)
        b_ptrs = b_base + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(b_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Store result
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=i_mask[:, None] & j_mask[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Step 0: LayerNorm (float16)
    # Use torch.nn.functional.layer_norm with weight and bias cast to half.
    # Cast input to half for subsequent ops.

    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # Linear projections: left, right (half)
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16), bias=None)

    # Gates (half)
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # Apply mask and gating
    if not nomask:
        # mask broadcast and fuse
        mask_h = mask.to(torch.float16).contiguous().view(-1)  # flatten
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(left_f, right_f, left_gate_f, right_gate_f, mask_h, hidden_dim)
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    del left_gate, right_gate, x  # free memory

    # Now compute contraction using Triton kernel.

    # output tensor allocate
    out = torch.empty_like(left)  # same shape

    # Kernel launch
    B = left.shape[0]
    N = left.shape[1]  # seq_len
    H = hidden_dim  # left.shape[3] also

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute strides (int)
    stride_a_batch = left.stride(0)
    stride_a_hidden = left.stride(3)
    stride_a_i = left.stride(1)
    stride_a_k = left.stride(2)

    stride_b_batch = right.stride(0)
    stride_b_hidden = right.stride(3)
    stride_b_k = right.stride(2)   # stride for k dimension in right (third dim)
    stride_b_j = right.stride(1)   # stride for j dimension

    stride_c_batch = out.stride(0)
    stride_c_hidden = out.stride(3)
    stride_c_i = out.stride(1)
    stride_c_j = out.stride(2)

    total_batches = B * H
    grid = (total_batches, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        H,
        N,
        stride_a_batch,
        stride_a_hidden,
        stride_a_i,
        stride_a_k,
        stride_b_batch,
        stride_b_hidden,
        stride_b_k,
        stride_b_j,
        stride_c_batch,
        stride_c_hidden,
        stride_c_i,
        stride_c_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Now apply output layernorm, out_gate, final linear.

    # Output layernorm:
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    return out.to(torch.float32)

"""
Optimized outgoing TriMul kernel.

- LayerNorm and linear projections are performed in half precision with
  torch.nn.functional.
- A tiny Triton kernel fuses the left/right gates with the optional mask.
- The O(N³) contraction is implemented as a custom batched GEMM in Triton,
  operating directly on the (B, N, N, H) layout to avoid expensive
  permute+contiguous copies.
- Output LayerNorm, out‑gate scaling and final linear projection complete
  the forward pass.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Gate + optional mask fusion kernel
# -------------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )

# -------------------------------------------------------------------------
# Core O(N³) contraction kernel (batched GEMM)
# -------------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH: tl.constexpr, H: tl.constexpr, N: tl.constexpr,
    stride_ab: tl.constexpr, stride_ah: tl.constexpr,
    stride_ai: tl.constexpr, stride_ak: tl.constexpr,
    stride_bb: tl.constexpr, stride_bh: tl.constexpr,
    stride_bk: tl.constexpr, stride_bj: tl.constexpr,
    stride_cb: tl.constexpr, stride_ch: tl.constexpr,
    stride_ci: tl.constexpr, stride_cj: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Decompose batch_id into (batch, hidden)
    b_idx = batch_id // H
    h_idx = batch_id % H

    # Offsets for the current tile
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    i_mask = i < N
    j_mask = j < N

    # Base pointers for this (b, h) slice
    a_base = a_ptr + b_idx * stride_ab + h_idx * stride_ah
    b_base = b_ptr + b_idx * stride_bb + h_idx * stride_bh
    c_base = c_ptr + b_idx * stride_cb + h_idx * stride_ch

    # Output pointer for the tile
    c_ptrs = c_base + i[:, None] * stride_ci + j[None, :] * stride_cj

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k < N

        # Load a tile from A (shape BLOCK_M x BLOCK_K)
        a_ptrs = a_base + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)

        # Load a tile from B transposed (shape BLOCK_K x BLOCK_N)
        b_ptrs = b_base + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(b_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0)

        acc += tl.dot(a, b, out_dtype=tl.float32)

    # Write the accumulated tile
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=i_mask[:, None] & j_mask[None, :])


# -------------------------------------------------------------------------
# Public entry point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) – may be None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 → half)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gates (half) with sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) with gates via Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask.to(torch.float16).contiguous().view(-1)  # [B*N*N]
        # flatten tensors for the fusion kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_h,
            hidden_dim,
        )
        # reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # free intermediate tensors we no longer need
    del left_gate, right_gate, x

    # ------------------------------------------------------------------
    # 5. Core O(N³) contraction using a custom Triton kernel
    # ------------------------------------------------------------------
    out = torch.empty_like(left)  # same shape, dtype float16

    B = left.shape[0]
    N = left.shape[1]      # seq_len
    H = hidden_dim

    # Strides for left (A)
    stride_ab = left.stride(0)   # batch stride
    stride_ah = left.stride(3)   # hidden stride (==1)
    stride_ai = left.stride(1)   # row (i) stride
    stride_ak = left.stride(2)   # col (k) stride

    # Strides for right (B)
    stride_bb = right.stride(0)
    stride_bh = right.stride(3)
    # For B_T we need stride for k (row) and j (col)
    stride_bk = right.stride(2)   # stride along k dimension
    stride_bj = right.stride(1)   # stride along j dimension

    # Strides for output (C)
    stride_cb = out.stride(0)
    stride_ch = out.stride(3)
    stride_ci = out.stride(1)
    stride_cj = out.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    total_batches = B * H
    grid = (
        total_batches,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        H,
        N,
        stride_ab,
        stride_ah,
        stride_ai,
        stride_ak,
        stride_bb,
        stride_bh,
        stride_bk,
        stride_bj,
        stride_cb,
        stride_ch,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in float32 as required
    return out.to(torch.float32)
