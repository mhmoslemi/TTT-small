"""
Outgoing TriMul (AlphaFold3) forward pass.

The implementation follows the reference PyTorch code but moves the mask &
gate fusion into a Triton kernel. The heavy O(N³) contraction is performed
with a batched half‑precision GEMM (torch.bmm), which is already highly
optimized on NVIDIA GPUs. The overall flow:

1. Layer‑norm the input (float32 → float16).
2. Linear projections to the hidden dimension (no bias).
3. Compute three gate tensors with a sigmoid.
4. Fuse optional mask with the left/right gate multiplications in a
   Triton kernel (in‑place).
5. Core contraction: for each hidden slice perform a batched matrix
   multiplication `out = left @ rightᵀ`.
6. Output LayerNorm, element‑wise gating with `out_gate` and a final
   linear projection back to the original feature dimension.
7. Return the result as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask (if present) and gate multiplications in-place
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # *half  [total]
    right_ptr,         # *half  [total]
    left_gate_ptr,     # *half  [total]
    right_gate_ptr,    # *half  [total]
    mask_ptr,          # *half  [B*N*N]  (broadcasted over hidden dim)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,   # hidden dimension (compile‑time constant)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load values
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    lgate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # each (b,i,j) spans H consecutive hidden entries
        mask_idx = offs // H
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * lgate * m
        right = right * rgate * m
    else:
        left = left * lgate
        right = right * rgate

    # Store back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_mask_and_gate(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    Fuse mask and gates into left/right tensors in‑place using Triton.
    All tensors must be flat (1‑D) half‑precision views.
    """
    total = left.numel()
    BLOCK = 65536  # large enough to reduce launch overhead
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    # If mask is not used, we can safely pass any pointer (here left)
    mask_ptr = mask if mask is not None else left

    _mask_gate_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (inp, mask, weights, config)
            inp   : torch.Tensor [B, N, N, dim] (float32)
            mask  : torch.Tensor [B, N, N] (bool/float) or None
            weights: dict with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight",
                "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            config: dict with at least "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    inp, mask, weights, config = data
    device = inp.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> half)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (no bias)
    # ------------------------------------------------------------------
    left = F.linear(
        x, weights["left_proj.weight"].to(torch.float16)
    )  # [B, N, N, H]
    right = F.linear(
        x, weights["right_proj.weight"].to(torch.float16)
    )  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Gate tensors (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = F.linear(
        x, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = F.linear(
        x, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) and gates via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> broadcast over hidden dim
        mask_broadcast = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]

        # Flatten tensors for the kernel
        left_f = left.view(-1)
        right_f = right.view(-1)
        lg_f = left_gate.view(-1)
        rg_f = right_gate.view(-1)
        mask_f = mask_broadcast.view(-1)  # length B*N*N

        _apply_mask_and_gate(left_f, right_f, lg_f, rg_f, mask_f, hidden_dim)

        # Reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # Simple element‑wise multiplication when no mask is required
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: batched GEMM per hidden slice
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # left is (B, N, N, H)

    # Rearrange to (B, H, N, N) and make contiguous for cuBLAS
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # Merge batch and hidden dimensions
    left_mat = left_t.view(B * hidden_dim, N, N)   # (B*H, N, N)
    right_mat = right_t.view(B * hidden_dim, N, N)  # (B*H, N, N)

    # Batched matrix multiply: (i,k) × (j,k)ᵀ → (i,j)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore original layout
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return in float32 as required
    return out.to(torch.float32)
```
