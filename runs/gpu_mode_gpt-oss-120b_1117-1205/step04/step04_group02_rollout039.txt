    stride_r_b, stride_r_j, stride_r_k, stride_r_h,

    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,

    pid_batch = tl.program_id(1)

def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HIDDEN_DIM: tl.constexpr
):
    pid = tl.program_id(0)
    pid_b = tl.program_id(1)
    
    # compute tile indices
    pid_m = pid // tl.cdiv(N, BLOCK_N)
    pid_n = pid % tl.cdiv(N, BLOCK_N)
    # Actually compute sizes:
    num_n_tiles = tl.cdiv(N, BLOCK_N)
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # offsets
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    offs_h = tl.arange(0, HIDDEN_DIM)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, HIDDEN_DIM), dtype=tl.float16)

    for k in range(0, N, BLOCK_K):
        # compute actual size of k block
        cur_k = k
        cur_k_size = tl.minimum(BLOCK_K, N - cur_k)
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # compute masks
        mask_k = offs_k < N

        # load left tile: shape (BLOCK_M, cur_k_size, H)
        left_ptrs = left_ptr + pid_b * stride_l_b + offs_i[:, None, None] * stride_l_i + offs_k[None, :, None] * stride_l_k + offs_h[None, None, :] * stride_l_h
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None], other=tl.float16(0.0))
        # shape (BLOCK_M, BLOCK_K, H)

        # load right tile: shape (cur_k_size, BLOCK_N, H)
        right_ptrs = right_ptr + pid_b * stride_r_b + offs_j[None, :, None] * stride_r_j + offs_k[:, None, None] * stride_r_k + offs_h[None, None, :] * stride_r_h
        right = tl.load(right_ptrs, mask=mask_j[None, :, None] & mask_k[:, None, None], other=tl.float16(0.0))
        # shape (BLOCK_K, BLOCK_N, H)

        # compute outer product across k axis
        # Loop over kk within block
        for kk in range(0, BLOCK_K):
            # conf: mask for kk
            if kk >= cur_k_size:
                continue
            left_kk = left[:, kk, :]    # (BLOCK_M, H)
            right_kk = right[kk, :, :]   # (BLOCK_N, H)
            # outer product and accumulate
            # broadcasting left_kk[:, None, :] * right_kk[None, :, :]
            acc += left_kk[:, None, :] * right_kk[None, :, :]
        # end inner kk
    # end k

    # write out
    out_ptrs = out_ptr + pid_b * stride_o_b + offs_i[:, None, None] * stride_o_i + offs_j[None, :, None] * stride_o_j + offs_h[None, None, :] * stride_o_h
    tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None])
```

"""
Outgoing TriMul (AlphaFold3) forward pass.

Algorithm
---------
1. Layer‑norm the input and cast to float16.
2. Two linear projections (left/right) and three sigmoid gates (left/right/out).
3. Fuse gates (and optional mask) with element‑wise multiplication (torch ops).
4. Core O(N³) contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a Triton kernel that works on the native [B,N,N,H] layout,
   avoiding the expensive transpose+contiguous copies required for a batched GEMM.
5. Output layer‑norm, out‑gate scaling and final linear projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # * float16  [B,N,N,H]
    right_ptr,         # * float16  [B,N,N,H]
    out_ptr,           # * float16  [B,N,N,H]
    B, N, H,           # runtime sizes
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):
    # program ids: (tile index over i/j, batch)
    pid = tl.program_id(0)
    batch = tl.program_id(1)

    # number of tiles
    num_tiles_i = tl.cdiv(N, BLOCK_M)
    num_tiles_j = tl.cdiv(N, BLOCK_N)

    pid_i = pid // num_tiles_j
    pid_j = pid % num_tiles_j

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # offsets for the tile
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_h = tl.arange(0, HIDDEN_DIM)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator for the tile (M, N, H)
    acc = tl.zeros((BLOCK_M, BLOCK_N, HIDDEN_DIM), dtype=tl.float16)

    # loop over the reduction dimension k
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k_blocks):
        k0 = kb * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: (M, K, H)
        left_ptrs = (
            left_ptr
            + batch * stride_l_b
            + offs_i[:, None, None] * stride_l_i
            + offs_k[None, :, None] * stride_l_k
            + offs_h[None, None, :] * stride_l_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None],
            other=tl.float16(0.0),
        )  # (M, K, H)

        # right tile: (K, N, H)
        right_ptrs = (
            right_ptr
            + batch * stride_l_b
            + offs_j[None, :, None] * stride_l_i   # j uses same stride as i
            + offs_k[:, None, None] * stride_l_k
            + offs_h[None, None, :] * stride_l_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_j[None, :, None] & mask_k[:, None, None],
            other=tl.float16(0.0),
        )  # (K, N, H)

        # outer‑product accumulation for each k in the block
        for kk in range(0, BLOCK_K):
            # left[:, kk, :]  -> (M, H)
            # right[kk, :, :] -> (N, H)
            l = left[:, kk, :]        # (M, H)
            r = right[kk, :, :]       # (N, H)
            # broadcast to (M, N, H) and accumulate
            acc += l[:, None, :] * r[None, :, :]
        # end kk

    # store the result
    out_ptrs = (
        out_ptr
        + batch * stride_o_b
        + offs_i[:, None, None] * stride_o_i
        + offs_j[None, :, None] * stride_o_j
        + offs_h[None, None, :] * stride_o_h
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None],
    )
    

def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool / float) or None
        - weights      : dict of torch.Tensors (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gates (half)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Apply gates and optional mask (element‑wise, torch ops)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask = mask.to(torch.float16).unsqueeze(-1)        # [B,N,N,1]
        left = left * left_gate * mask
        right = right * right_gate * mask
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented in Triton without transposes.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # left.shape == [B, N, N, hidden_dim]
    # ensure contiguous layout for the kernel
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)  # half precision output

    # strides for tensors (in elements)
    stride_l = left.stride()   # (B, i, k, h)
    stride_o = out.stride()

    # Triton launch parameters
    BLOCK_M = 4
    BLOCK_N = 4
    BLOCK_K = 4

    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    grid = (grid_m * grid_n, B)  # first axis tiles, second axis batch

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_l[0], stride_l[1], stride_l[2], stride_l[3],   # left strides
        stride_o[0], stride_o[1], stride_o[2], stride_o[3],   # out strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HIDDEN_DIM=H,
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in original dtype
    return out.to(torch.float32)
