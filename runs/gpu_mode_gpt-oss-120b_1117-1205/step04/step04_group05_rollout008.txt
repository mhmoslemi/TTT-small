"""
TriMul outgoing kernel (AlphaFold3) – fused mask+gate (Triton) + batched GEMM (Triton).

Algorithm
---------
1. Layer‑norm the input (float32) → cast to float16.
2. Linear projections to hidden dimension (no bias).
3. Compute sigmoid gates.
4. Fuse mask (if present) and gates into the projected tensors with a
   custom Triton kernel (_gate_mask_fuse_kernel).  This writes the gated
   tensors back in‑place.
5. Core O(N³) contraction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched half‑precision matrix‑multiply using a
   Triton GEMM kernel (tri_matmul_kernel).  The batch dimension is
   B·H (batch‑size × hidden‑dim).
6. Output layer‑norm, out‑gate scaling and final linear projection back
   to the original dimension.
7. Return float32 tensor.

The implementation uses only half‑precision arithmetic for the heavy
operations and leaves the cheap layer‑norms to PyTorch.  Both kernels
are launched with sensible block sizes for the tested sequence lengths.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: fuse mask + left/right gates (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # half*  flattened left tensor
    right_ptr,           # half*  flattened right tensor
    left_gate_ptr,       # half*  flattened left‑gate tensor
    right_gate_ptr,      # half*  flattened right‑gate tensor
    mask_ptr,            # half*  flattened mask (broadcast over hidden dim)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present, else 0
    H: tl.constexpr,          # hidden dimension
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elements

    left = tl.load(left_ptr + offs, mask=m, other=0.0)
    right = tl.load(right_ptr + offs, mask=m, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=m, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=m, other=0.0)

    if HAS_MASK:
        # one mask value per (batch, i, k) group of H hidden values
        mask_offs = offs // H
        mask_val = tl.load(mask_ptr + mask_offs, mask=m, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=m)
    tl.store(right_ptr + offs, right, mask=m)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """In‑place fusion of mask + gates using Triton."""
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when no mask

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # tensors are modified in‑place


# ----------------------------------------------------------------------
# Kernel 2: batched half‑precision GEMM for the O(N³) contraction
# ----------------------------------------------------------------------
@triton.jit
def tri_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to A, B, C  (half*)
    batch: tl.constexpr, N: tl.constexpr,
    stride_ah, stride_ai, stride_ak,    # strides for A (batch, i, k)
    stride_bh, stride_bk, stride_bj,    # strides for B (batch, k, j)
    stride_ch, stride_ci, stride_cj,    # strides for C (batch, i, j)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid0 = tl.program_id(0)   # encodes batch * block_i
    pid1 = tl.program_id(1)   # block_j

    num_block_m = tl.cdiv(N, BLOCK_M)

    batch_idx = pid0 // num_block_m
    block_i = pid0 % num_block_m
    block_j = pid1

    # coordinate ranges for this block
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows/cols
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers to the start of the current batch
    a_batch_ptr = a_ptr + batch_idx * stride_ah
    b_batch_ptr = b_ptr + batch_idx * stride_bh
    c_batch_ptr = c_ptr + batch_idx * stride_ch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute A block pointer and load
        a_ptrs = a_batch_ptr + (offs_i[:, None] * stride_ai) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # compute B block pointer and load (B is transposed: shape (batch, k, j))
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk) + (offs_j[None, :] * stride_bj)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_batch_ptr + (offs_i[:, None] * stride_ci) + (offs_j[None, :] * stride_cj)
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be None if config["nomask"]=True
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half precision)
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16), bias=None)

    # ------------------------------------------------------------------
    # 3. Gates (half precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask + gates (Triton)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).contiguous().view(-1)      # [B*N*N]
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(left_f, right_f, left_gate_f, right_gate_f, mask_h, hidden_dim)

        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # Simple element‑wise gating when no mask
        left = left * left_gate
        right = right * right_gate

    # free unused tensors
    del left_gate, right_gate, x_h

    # ------------------------------------------------------------------
    # 5. Core contraction: batch GEMM via Triton
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)
    # transpose the last two dimensions of right for matmul
    right_T = right_perm.transpose(-2, -1).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    batch = B * H

    # views as (batch, N, N)
    A = left_perm.view(batch, N, N)
    B_mat = right_T.view(batch, N, N)

    # allocate output
    out_mat = torch.empty_like(A)

    # stride arguments (row‑major)
    stride_ah, stride_ai, stride_ak = A.stride()
    stride_bh, stride_bk, stride_bj = B_mat.stride()
    stride_ch, stride_ci, stride_cj = out_mat.stride()

    # choose block sizes
    if N <= 64:
        BLOCK_M = BLOCK_N = 32
    elif N <= 128:
        BLOCK_M = BLOCK_N = 64
    else:
        BLOCK_M = BLOCK_N = 128
    BLOCK_K = 64

    grid = (triton.cdiv(batch * tl.cdiv(N, BLOCK_M), 1), tl.cdiv(N, BLOCK_N))
    # launch the kernel (2‑D grid)
    tri_matmul_kernel[
        (triton.cdiv(batch, 1) * triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    ](
        A, B_mat, out_mat,
        batch, N,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bk, stride_bj,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    torch.cuda.synchronize()

    # reshape output to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    # cast back to float32
    return out.to(torch.float32)
