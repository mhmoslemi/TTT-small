"""
Optimized outgoing TriMul (Triangular Multiplicative Update) forward pass.

Algorithm
---------
1. LayerNorm on the input tensor (float32) → cast to float16.
2. Fuse the left and right linear projections with their respective gates by
   concatenating the projection and gate weights and performing a single
   matmul per side.  The concatenated result is split into the raw projection
   and raw gate tensors.
3. A small Triton kernel applies the sigmoid gate and (optional) mask in-place:
       proj = proj * sigmoid(gate) * mask
   The kernel processes `left` and `right` separately.
4. Compute the out‑gate with a separate linear + sigmoid.
5. Contraction: for each batch and hidden channel perform a batched GEMM
       out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   This is implemented with a reshape + torch.bmm (highly tuned cuBLAS).
6. Apply LayerNorm on the hidden dimension, scale by the out‑gate and project
   back to the original channel dimension.
7. Return the result in float32.

A Triton kernel is used for the element‑wise gate‑mask fusion, all other
operations rely on Torch’s fast CUDA kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: apply sigmoid gate and optional mask in-place
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_fuse_kernel(
    proj_ptr,               # fp16*
    gate_ptr,               # fp16*
    mask_ptr,               # fp16* (broadcasted, may be dummy)
    total_elements: tl.int32,
    H: tl.constexpr,        # hidden dimension (size of innermost channel)
    HAS_MASK: tl.constexpr, # 1 if mask is present, else 0
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load values
    proj = tl.load(proj_ptr + offs, mask=mask)
    gate_raw = tl.load(gate_ptr + offs, mask=mask)

    # Sigmoid: 1 / (1 + exp(-x))
    gate = 1.0 / (1.0 + tl.exp(-gate_raw))

    if HAS_MASK:
        # mask is stored without the hidden dimension; each hidden entry
        # shares the same mask element.
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask)
        proj = proj * gate * mask_val
    else:
        proj = proj * gate

    tl.store(proj_ptr + offs, proj, mask=mask)


def _apply_proj_gate_mask(proj: torch.Tensor,
                          gate: torch.Tensor,
                          mask: torch.Tensor | None,
                          hidden_dim: int):
    """
    In‑place fusion of proj = proj * sigmoid(gate) * mask (if mask is provided).
    All tensors must be contiguous and of dtype float16.
    """
    total = proj.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else proj  # dummy when mask not used

    _proj_gate_mask_fuse_kernel[grid](
        proj,
        gate,
        mask_ptr,
        total,
        H=hidden_dim,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple containing
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (any numeric type) – may be ignored
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) -> cast to float16
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Concatenated linear projections + raw gates (half precision)
    # ------------------------------------------------------------------
    left_weight_cat = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)
    right_weight_cat = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)

    # Shape: [B, N, N, 2*hidden_dim]
    left_cat = F.linear(x, left_weight_cat, bias=None)
    right_cat = F.linear(x, right_weight_cat, bias=None)

    # Split into projection and raw gate halves
    left_proj = left_cat[..., :hidden_dim].contiguous()
    left_gate_raw = left_cat[..., hidden_dim:].contiguous()
    right_proj = right_cat[..., :hidden_dim].contiguous()
    right_gate_raw = right_cat[..., hidden_dim:].contiguous()

    # ------------------------------------------------------------------
    # 3. Fuse sigmoid gate and optional mask (Triton kernel)
    # ------------------------------------------------------------------
    if not nomask:
        mask_half = mask.to(torch.float16, non_blocking=True).contiguous()
    else:
        mask_half = None

    _apply_proj_gate_mask(left_proj, left_gate_raw, mask_half, hidden_dim)
    _apply_proj_gate_mask(right_proj, right_gate_raw, mask_half, hidden_dim)

    # After the kernel, left_proj and right_proj hold the final left/right tensors
    left = left_proj
    right = right_proj

    # ------------------------------------------------------------------
    # 4. Compute out‑gate (sigmoid of linear)
    # ------------------------------------------------------------------
    out_gate_raw = F.linear(
        x, weights["out_gate.weight"].to(torch.float16, non_blocking=True), bias=None
    )
    out_gate = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 5. Core contraction via batched GEMM (bmm)
    #    out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) and make contiguous for cuBLAS
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # Batch matrix multiplication with inner dimension transposed
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling, final linear projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16, non_blocking=True),
        bias=weights["to_out_norm.bias"].to(torch.float16, non_blocking=True),
    )
    out_scaled = out_norm * out_gate  # element‑wise scaling

    out_final = F.linear(
        out_scaled,
        weights["to_out.weight"].to(torch.float16, non_blocking=True),
        bias=None,
    )

    # Return in float32 as required
    return out_final.to(torch.float32)
