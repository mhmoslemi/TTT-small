custom_kernel(data):
    input_tensor, mask, weights, config = data

    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. LayerNorm
    # Input: shape (B,N,N,dim). Use F.layer_norm.
    # weight and bias should be same dtype as input (float16), cast.
    norm_weight = weights["norm.weight"].to(device).to(torch.float16)
    norm_bias = weights["norm.bias"].to(device).to(torch.float16)

    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-6,
    )
    # Cast to half for subsequent operations
    x = x.to(torch.float16)

    # 2. Projections
    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left = F.linear(x, left_proj_weight, bias=None)
    right = F.linear(x, right_proj_weight, bias=None)

    # 3. Gates
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)
    left_gate = F.linear(x, left_gate_weight, bias=None).sigmoid()
    right_gate = F.linear(x, right_gate_weight, bias=None).sigmoid()
    out_gate = F.linear(x, out_gate_weight, bias=None).sigmoid()

    # 4. Apply mask and gates
    if not nomask:
        # mask shape [B,N,N], broadcast last dim
        mask_h = mask.to(device).to(torch.float16).contiguous().view(-1)
        # flatten left/right and gates
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        _apply_gate_and_mask(left_f, right_f, left_gate_f, right_gate_f, mask_h, hidden_dim)
        left = left_f.view_as(left)
        right = right_f.view_as(right)
        # Clean up
        del left_f, right_f, left_gate_f, right_gate_f, mask_h
    else:
        left = left * left_gate
        right = right * right_gate
    # free
    del left_gate, right_gate

    # 5. Contraction
    # Reshape for batched matmul.
    # Permute to (B, hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_perm.shape
    # reshape to (B*H, N, N)
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)
    # bmm with transposed right: (B*H, N, N) x (B*H, N, N)^T? Actually right transpose on last two dims: we need right_mat.transpose(1,2)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # reshape back
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 6. Output layernorm
    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-6,
    )
    # apply out_gate (reshape)
    out = out * out_gate

    # 7. Final linear
    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_weight, bias=None)

    # 8. Return float32
    return out.to(torch.float32)

    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    has_mask = 1
    _gate_mask_fuse_kernel[grid](
        left_f,
        right_f,
        left_gate_f,
        right_gate_f,
        mask_f,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,

    )

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # offset for mask: integer division by H
        mask_idx = offs // H
        # each mask value broadcast across H hidden dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)

def _apply_gate_and_mask(left_f, right_f, left_gate_f, right_gate_f, mask_f, hidden_dim):
    total = left_f.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    _gate_mask_fuse_kernel[grid](
        left_f,
        right_f,
        left_gate_f,
        right_gate_f,
        mask_f,
        total,
        BLOCK=BLOCK,
        HAS_MASK=1,
        H=hidden_dim,
    )

    This implementation:

    - Fuses mask and gate multiplication with a tiny Triton kernel.
    - Computes the O(N³) contraction via a batched half‑precision GEMM

    - Applies the output LayerNorm, out‑gate, and final projection.
    - Returns the result in float32.
    """

    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. Input LayerNorm (float16)
    # -----------------------------------------------------------------

    norm_bias = weights["norm.bias"].to(device).to(torch.float16)

        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-6,
    )

    # 2. Linear projections (float16)
    # -----------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(device).to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(device).to(torch.float16),
        bias=None,
    )

    # 3. Gates (float16 + sigmoid)
    # -----------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(device).to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(device).to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(device).to(torch.float16),
        bias=None,
    ).sigmoid()

    # 4. Fuse mask (if any) with gates using Triton
    # -----------------------------------------------------------------
    if not nomask:

        mask_f = mask.to(device).to(torch.float16).contiguous().view(-1)

        # Flatten tensors for Triton kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(left_f, right_f, left_gate_f, right_gate_f, mask_f, hidden_dim)

        # Reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)

        # Clean up temporary tensors
        del left_f, right_f, left_gate_f, right_gate_f, mask_f
    else:
        left = left * left_gate
        right = right * right_gate

    # free gates that are no longer needed
    del left_gate, right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM over (batch * hidden)
    # -----------------------------------------------------------------
    # Rearrange to (B, hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # -----------------------------------------------------------------

        (hidden_dim,),

    )

        out,

    )

    return out.to(torch.float32)

    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # mask is broadcast over the hidden dimension
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)

def _apply_gate_and_mask(left_f, right_f, left_gate_f, right_gate_f, mask_f, hidden_dim):
    """Launch the Triton kernel that fuses mask with the two gates."""
    total = left_f.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    _gate_mask_fuse_kernel[grid](
        left_f,
        right_f,
        left_gate_f,
        right_gate_f,
        mask_f,
        total,
        BLOCK=BLOCK,
        HAS_MASK=1,
        H=hidden_dim,
    )

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold3).

    1. LayerNorm on the input.
    2. Linear projections to the hidden dimension.
    3. Sigmoid gates.
    4. Optional mask‑gate fusion via a tiny Triton kernel.
    5. O(N³) contraction performed as a batched half‑precision GEMM.
    6. Output LayerNorm, out‑gate scaling and final projection.
    Returns a tensor of shape [B, N, N, dim] in float32.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float16)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device).to(torch.float16)
    norm_bias = weights["norm.bias"].to(device).to(torch.float16)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-6,
    )
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections (float16)
    # -----------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(device).to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(device).to(torch.float16),
        bias=None,
    )

    # -----------------------------------------------------------------
    # 3. Gates (float16 + sigmoid)
    # -----------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(device).to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(device).to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(device).to(torch.float16),
        bias=None,
    ).sigmoid()

    # -----------------------------------------------------------------
    # 4. Fuse mask (if any) with gates using Triton
    # -----------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(device).to(torch.float16).contiguous().view(-1)

        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(left_f, right_f, left_gate_f, right_gate_f, mask_f, hidden_dim)

        left = left_f.view_as(left)
        right = right_f.view_as(right)

        del left_f, right_f, left_gate_f, right_gate_f, mask_f
    else:
        left = left * left_gate
        right = right * right_gate

    del left_gate, right_gate  # free memory

    # -----------------------------------------------------------------
    # 5. Core contraction via batched half‑precision GEMM
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # -----------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N) # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device).to(torch.float16),
        bias=weights["to_out_norm.bias"].to(device).to(torch.float16),
        eps=1e-6,
    )
    out = out * out_gate

    out = F.linear(
        out,
        weights["to_out.weight"].to(device).to(torch.float16),
        bias=None,
    )

    # Return in float32 as required by the interface
    return out.to(torch.float32)
