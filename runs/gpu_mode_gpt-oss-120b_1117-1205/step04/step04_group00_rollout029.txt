"""
Implementation of the outgoing TriMul operator using a fused Triton kernel for mask
and gate multiplication and a batched GEMM for the N³ contraction.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(left_ptr, right_ptr, left_gate_ptr, right_gate_ptr,
                           mask_ptr, total_elements,
                           BLOCK: tl.constexpr, HAS_MASK: tl.constexpr, H: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each hidden element shares the same mask entry
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store back
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In-place fusion of left/right multiplication with gates and optional mask.
    All tensors must be torch.float16 and contiguous.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # No return, operation performed in-place

def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored if config["nomask"] is True
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----- Load weights -----
    # Input LayerNorm (float32)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # Linear weights (float32 -> will be cast to fp16 where needed)
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ----- Step 1: Input LayerNorm (float32) -----
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # shape [B, N, N, dim], float32

    # ----- Step 2: Cast to half for most heavy ops -----
    x_h = x.to(torch.float16)

    # ----- Step 3: Linear projections to hidden dimension (half) -----
    left = F.linear(x_h, left_proj_weight.to(torch.float16), bias=None)   # [B,N,N,hidden]
    right = F.linear(x_h, right_proj_weight.to(torch.float16), bias=None)

    # ----- Step 4: Compute gates (half) -----
    left_gate = F.linear(x_h, left_gate_weight.to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x_h, right_gate_weight.to(torch.float16), bias=None).sigmoid()

    # Out gate stays in float32 to keep final precision
    out_gate = F.linear(x, out_gate_weight, bias=None).sigmoid()   # [B,N,N,hidden] float32

    # ----- Step 5: Apply mask and gate fusion -----
    if not nomask:
        # Ensure tensors are contiguous before flattening
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        # mask to half and flatten
        mask_f = mask.to(torch.float16).contiguous().view(-1)  # [B*N*N]
        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            hidden_dim,
        )
        # Reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask: simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # Gates no longer needed
    del left_gate, right_gate

    # ----- Step 6: Core N³ contraction via batched GEMM (half) -----
    # Rearrange to (B, hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    # Collapse batch and head dimensions
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Batched matrix multiplication: (i,k) * (j,k)ᵀ  => sum over k
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore shape (B, N, N, hidden)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, hidden)

    # ----- Step 7: Output LayerNorm (float32) -----
    out_f32 = out.to(torch.float32)
    out_norm = F.layer_norm(
        out_f32,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # ----- Step 8: Apply out gate (float32) -----
    out_scaled = out_norm * out_gate

    # ----- Step 9: Final linear projection back to original dim (float32) -----
    final = F.linear(out_scaled, to_out_weight, bias=None)  # [B,N,N,dim]

    return final
```

"""
Outgoing TriMul operator.

The implementation follows the reference PyTorch version but moves the
expensive mask‑gate fusion to a tiny Triton kernel and performs the
N³ contraction as a batched GEMM (half‑precision).  The overall flow is:

1. Input LayerNorm (float32) → cast to fp16.
2. Linear projections to the hidden dimension (fp16).
3. Gate projections (fp16) and the output gate (float32).
4. Fuse mask (if present) with the per‑position gates via a Triton kernel.
5. Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] using
   torch.bmm on the reshaped tensors (fp16).
6. Cast the result back to float32, apply the output LayerNorm,
   multiply by the output gate and a final linear projection back to the
   original dimension.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # fp16*  (flattened left tensor)
    right_ptr,         # fp16*  (flattened right tensor)
    left_gate_ptr,     # fp16*  (flattened left gate)
    right_gate_ptr,    # fp16*  (flattened right gate)
    mask_ptr,          # fp16*  (flattened mask, broadcasted over hidden dim)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 → mask present, 0 → no mask
    H: tl.constexpr,          # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    # Load data
    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    if HAS_MASK:
        # each hidden element shares a mask entry:
        # mask index = (flattened offset) // H
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store the results in‑place
    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place application of:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    (mask may be None). All tensors must be contiguous fp16 tensors.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when no mask

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # No return – operation performed in‑place.


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Load weights
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 2. Input LayerNorm (float32) → half precision
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim], float32
    x_h = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections to hidden dimension (fp16)
    # -----------------------------------------------------------------
    left = F.linear(x_h, left_proj_weight.to(torch.float16), bias=None)   # [B,N,N,hidden]
    right = F.linear(x_h, right_proj_weight.to(torch.float16), bias=None)

    # -----------------------------------------------------------------
    # 4. Gate projections (fp16) and output gate (float32)
    # -----------------------------------------------------------------
    left_gate = F.linear(x_h, left_gate_weight.to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x_h, right_gate_weight.to(torch.float16), bias=None).sigmoid()
    out_gate = F.linear(x, out_gate_weight, bias=None).sigmoid()   # float32

    # -----------------------------------------------------------------
    # 5. Fuse mask with gates (if needed) using Triton
    # -----------------------------------------------------------------
    if not nomask:
        # Flatten tensors for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask.to(torch.float16).contiguous().view(-1)  # [B*N*N]

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            hidden_dim,
        )
        # Reshape back to original layout
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # Simple element‑wise multiplication when no mask
        left = left * left_gate
        right = right * right_gate

    # Gates are no longer needed
    del left_gate, right_gate

    # -----------------------------------------------------------------
    # 6. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as a batched GEMM on half‑precision tensors.
    # -----------------------------------------------------------------
    # Transform to (B, hidden, N, N) layout
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)    # (B*hidden, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Batched matrix multiply: (i,k) @ (j,k)^T → sum over k
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*hidden, N, N)

    # Restore original layout (B, N, N, hidden)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 7. Output LayerNorm (float32), out‑gate scaling and final linear
    # -----------------------------------------------------------------
    out_f32 = out.to(torch.float32)
    out_norm = F.layer_norm(
        out_f32,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out_scaled = out_norm * out_gate
    final = F.linear(out_scaled, to_out_weight, bias=None)   # (B, N, N, dim)

    return final
