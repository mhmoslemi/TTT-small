def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. layer norm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(torch.float16),
        bias=weights["norm.bias"].to(torch.float16),
    ).to(torch.float16)

    # 2. linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = torch.nn.functional.linear(x, weights["right_proj.weight"].to(torch.float16))

    # 3. gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # 4. fuse mask and gate if needed
    if not nomask and mask is not None:
        mask_half = mask.to(torch.float16)
        # make contiguous and flatten
        mask_half = mask_half.contiguous()
        _fuse_gate_and_mask(left, right, left_gate, right_gate, mask_half, hidden_dim)
    else:
        left = left * left_gate
        right = right * right_gate

    # 5. core contraction using batched matmul
    # permute to bring hidden_dim to batch
    left_f = left.permute(0, 3, 1, 2)  # B, H, N, N
    right_f = right.permute(0, 3, 1, 2)  # B, H, N, N
    # compute out_f = left_f @ right_f^T over last two dims
    out_f = torch.matmul(left_f, right_f.transpose(-2, -1))  # B, H, N, N
    out = out_f.permute(0, 2, 3, 1)  # B, N, N, H

    # 6. to_out_norm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    # 7. final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16))
    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass – optimized with a fused Triton kernel for the
mask‑and‑gate multiplication and a batched GEMM for the O(N³) contraction.

Algorithm
---------
1. Layer‑norm the input tensor (float16).
2. Linear projections to the hidden dimension (half‑precision).
3. Compute left/right/out gates (sigmoid of linear projections).
4. If a mask is present, fuse ``mask * gate * tensor`` for *left* and *right*
   using a Triton kernel (in‑place).  Otherwise perform simple elementwise
   multiplication in PyTorch.
5. Core contraction:
   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is equivalent to a batch of matrix multiplications
   L_d @ R_dᵀ for every hidden channel d.
   We reshape to (B, H, N, N) and call ``torch.matmul`` which uses cuBLAS.
6. Apply a second layer‑norm over the hidden dimension, multiply by the out‑gate,
   and project back to the original dimension.
7. Return the result in float32.

At least one operation (mask‑gate fusion) is performed in Triton, satisfying the
requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask (if present) with the left/right gates
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,        # half *
    right_ptr,       # half *
    left_gate_ptr,   # half *
    right_gate_ptr,  # half *
    mask_ptr,        # half *
    total_elems: tl.int32,
    H: tl.constexpr,         # hidden dimension
    HAS_MASK: tl.constexpr,  # 1 if mask present else 0
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # Load tensors
    left      = tl.load(left_ptr + offs,      mask=mask, other=tl.float16(0.0))
    right     = tl.load(right_ptr + offs,     mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate= tl.load(right_gate_ptr + offs,mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # mask is broadcast over the hidden dimension
        mask_idx = offs // H                      # index into [B, N, N] mask
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left  = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left * left_gate
        right = right * right_gate

    # Write results back in‑place
    tl.store(left_ptr  + offs, left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _fuse_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """In‑place fusion of mask and gates for left/right tensors."""
    # Ensure contiguous half tensors

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or numeric) or None
            - weights: dict mapping parameter names to tensors
            - config: dict with keys ``dim`` and ``hidden_dim`` and optional
                      ``nomask`` (bool, default True)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float16)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(torch.float16),
        bias=weights["norm.bias"].to(torch.float16),
    ).to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections (to hidden_dim, half‑precision)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 3. Gate values (half‑precision)
    # --------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # --------------------------------------------------------------
    # 4. Fuse mask (if any) and gates via Triton
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask broadcasted over hidden_dim
        mask_half = mask.to(torch.float16)
        _fuse_gate_and_mask(left, right, left_gate, right_gate, mask_half, hidden_dim)
    else:
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    This is a batch of GEMMs across the hidden dimension.
    # --------------------------------------------------------------
    # reshape to (B, H, N, N) for batched matmul
    left_f = left.permute(0, 3, 1, 2)   # [B, H, N, N]
    right_f = right.permute(0, 3, 1, 2) # [B, H, N, N]

    # batch matmul: (N,N) @ (N,N)^T  -> (N,N)
    out_f = torch.matmul(left_f, right_f.transpose(-2, -1))  # [B, H, N, N]

    # reshape back to original layout
    out = out_f.permute(0, 2, 3, 1)   # [B, N, N, H]

    # --------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return in float32 as requested
    return out.to(torch.float32)
