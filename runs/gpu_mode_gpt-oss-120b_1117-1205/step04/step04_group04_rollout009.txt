```python
@triton.jit
def fuse_gate_mask_kernel(left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr, total_elems, hidden_dim, BLOCK: tl.constexpr, HAS_MASK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems
    left = tl.load(left_ptr + offs, mask=mask)
    right = tl.load(right_ptr + offs, mask=mask)
    left_g = tl.load(left_gate_ptr + offs, mask=mask)
    right_g = tl.load(right_gate_ptr + offs, mask=mask)
    if HAS_MASK:
        # mask index = offs // hidden_dim (since mask broadcast across hidden dimension)
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * left_g * mask_val
        right = right * right_g * mask_val
    else:
        left = left * left_g
        right = right * right_g
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype = torch.float16

    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x = x.to(dtype)

    # Linear projections
    left = F.linear(x, weights["left_proj.weight"].to(dtype))
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # Gates
    left_gate = F.linear(x, weights["left_gate.weight"].to(dtype)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(dtype)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(dtype)).sigmoid()

    # Fuse mask and gates
    if not nomask:
        # mask shape [B,N,N]; broadcast over hidden dim
        mask_h = mask.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        # flatten all tensors for kernel
        total = left.numel()
        BLOCK = 65536
        grid = (triton.cdiv(total, BLOCK),)
        has_mask = 1
        # mask pointer flatten to shape [B*N*N]
        mask_flat = mask_h.view(-1)
        left_flat = left.view(-1)
        right_flat = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)

        fuse_gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            BLOCK=BLOCK,
            HAS_MASK=has_mask,
        )
        # reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # core contraction using batched GEMM
    B, N, _, _ = left.shape
    # reshape as (B*hidden_dim, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # compute out_mat = left_mat @ right_mat.transpose(1,2)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape to [B,N,N,hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # output LayerNorm, gating, linear projection
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"].to(dtype), bias=weights["to_out_norm.bias"].to(dtype))
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    # cast back to float32
    return out.to(torch.float32)
```

"""
Outgoing TriMul (AlphaFold3) forward pass.

Algorithm
---------
1. Layer‑norm the input (float32 → float16).
2. Linear projections to the hidden dimension (no bias).
3. Compute three gate tensors with a sigmoid.
4. Fuse optional mask with left/right gate multiplications in a Triton kernel
   (in‑place, half‑precision).
5. Core O(N³) contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM (B×H independent matrix multiplies) using
   torch.bmm on contiguous half‑precision tensors.
6. Output LayerNorm, element‑wise gating with out_gate and a final linear
   projection back to the original feature dimension.
7. Return the result in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _fuse_mask_gate_kernel(
    left_ptr,          # *half  (flattened left tensor)
    right_ptr,         # *half  (flattened right tensor)
    left_gate_ptr,     # *half  (flattened left gate)
    right_gate_ptr,    # *half  (flattened right gate)
    mask_ptr,          # *half  (flattened mask, broadcasted over hidden dim)
    total_elems: tl.int32,
    hidden_dim: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """
    In‑place fusion:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    The mask is broadcasted over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elems

    left      = tl.load(left_ptr      + offs, mask=m, other=tl.float16(0.0))
    right     = tl.load(right_ptr     + offs, mask=m, other=tl.float16(0.0))
    lgate     = tl.load(left_gate_ptr + offs, mask=m, other=tl.float16(0.0))
    rgate     = tl.load(right_gate_ptr+ offs, mask=m, other=tl.float16(0.0))

    if HAS_MASK:
        # each hidden slice shares the same mask entry
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=m, other=tl.float16(0.0))
        left  = left  * lgate * mask_val
        right = right * rgate * mask_val
    else:
        left  = left  * lgate
        right = right * rgate

    tl.store(left_ptr  + offs, left,  mask=m)
    tl.store(right_ptr + offs, right, mask=m)


def _apply_mask_and_gate(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    Launch the Triton kernel that fuses mask and gates in‑place.
    All inputs are flat half‑precision tensors.
    """
    total = left.numel()
    BLOCK = 65536  # large block to keep launch overhead low
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy ptr when no mask

    _fuse_mask_gate_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        hidden_dim,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
    )
    # In‑place mutation; nothing to return


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict mapping parameter names to float32 tensors
            - config: dict containing at least "dim" and "hidden_dim",
                      optionally "nomask" (bool)

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    inp, mask, weights, config = data
    device = inp.device
    dtype = torch.float16

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 → half)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(dtype)

    # -----------------------------------------------------------------
    # 2. Linear projections to hidden dimension
    # -----------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"].to(dtype))   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # -----------------------------------------------------------------
    # 3. Gate tensors (sigmoid)
    # -----------------------------------------------------------------
    left_gate  = F.linear(x, weights["left_gate.weight"].to(dtype)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(dtype)).sigmoid()
    out_gate   = F.linear(x, weights["out_gate.weight"].to(dtype)).sigmoid()

    # -----------------------------------------------------------------
    # 4. Fuse mask with gates (if required) via Triton
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B,N,N] → broadcast over hidden dimension
        mask_h = mask.to(dtype).unsqueeze(-1)                 # [B,N,N,1]

        # Flatten tensors for the kernel
        left_f      = left.view(-1)
        right_f     = right.view(-1)
        left_gate_f = left_gate.view(-1)
        right_gate_f = right_gate.view(-1)
        mask_f      = mask_h.view(-1)                         # length B*N*N

        _apply_mask_and_gate(
            left_f, right_f, left_gate_f, right_gate_f, mask_f, hidden_dim
        )

        # Reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask – simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: batched matrix multiplication per hidden slice
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape   # left shape = (B, N, N, H)
    # Rearrange to (B*H, N, N) for a single batched GEMM
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched GEMM: (i,k) × (j,k)ᵀ → (i,j)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    # Return in float32 as specified
    return out.to(torch.float32)
