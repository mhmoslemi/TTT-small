```
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements: tl.int32,
    H: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        mask_idx = offs // H
        m_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * m_val
        right = right * right_gate * m_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm on last dim (dim)
    x = F.layer_norm(input_tensor, (dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])

    # Cast to half
    x_h = x.to(torch.float16)

    # Weights casted to half
    w_left = weights["left_proj.weight"].to(torch.float16)
    w_right = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)
    w_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    w_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    w_out = weights["to_out.weight"].to(torch.float16)

    # Linear projections
    # x_h: [B, N, N, dim]
    # we can flatten the first three dims for linear: view (-1, dim)
    B, N, _, _ = x_h.shape
    x_flat = x_h.reshape(-1, dim)  # (B*N*N, dim)

    left = torch.nn.functional.linear(x_flat, w_left)  # (B*N*N, hidden_dim)
    right = torch.nn.functional.linear(x_flat, w_right)

    # Gates
    left_gate = torch.nn.functional.linear(x_flat, w_left_gate).sigmoid()
    right_gate = torch.nn.functional.linear(x_flat, w_right_gate).sigmoid()
    out_gate = torch.nn.functional.linear(x_flat, w_out_gate).sigmoid()

    # reshape back
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)
    left_gate = left_gate.view(B, N, N, hidden_dim)
    right_gate = right_gate.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # fuse mask+gate
    if not nomask:
        # mask shape [B,N,N] broadcast to hidden_dim
        mask_f = mask.to(torch.float16).view(B, N, N, 1)
        # fuse using kernel
        total = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)
        HAS_MASK = 1
        gate_mask_fuse_kernel[grid](
            left.view(-1), right.view(-1),
            left_gate.view(-1), right_gate.view(-1),
            mask_f.view(-1),
            total,
            hidden_dim,
            HAS_MASK,
            BLOCK=BLOCK
        )
    else:
        # no mask: simple elementwise multiplication (we may also use kernel but it's fine)
        left = left * left_gate
        right = right * right_gate

    # Now left and right already fused (if gating applied)
    if nomask:
        # if no mask, we must multiply left*left_gate etc already done
        left = left * left_gate  # Wait we already did not multiply; we need this if no mask path.
        right = right * right_gate
    # Delete gates
    del left_gate, right_gate

    # Compute contraction via batched GEMM
    # Permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # reshape batch*head
    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B*H, N, N)
    right_mat = right_perm.view(B*H, N, N)

    # bmm left_mat (B*H,N,N) * right_mat^T (B*H,N,N) -> (B*H,N,N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output layernorm
    out = F.layer_norm(out, (hidden_dim,), weight=w_out_norm_w, bias=w_out_norm_b)

    # Scale by out_gate
    out = out * out_gate

    # Final projection
    out = torch.nn.functional.linear(out, w_out)

    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

The kernel fuses the element‑wise multiplication of the projected tensors
with their gates and an optional mask:
    left  = (x @ W_left )  * sigmoid(x @ W_left_gate)  * mask
    right = (x @ W_right)  * sigmoid(x @ W_right_gate) * mask
All tensors are kept in float16 to minimise memory traffic.
The cubic contraction is performed as a single strided‑batched GEMM
using torch.bmm, which internally maps to cuBLAS.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # half* (flattened left tensor)
    right_ptr,              # half* (flattened right tensor)
    left_gate_ptr,          # half* (flattened left gate)
    right_gate_ptr,         # half* (flattened right gate)
    mask_ptr,               # half* (flattened mask, broadcast over hidden dim)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,    # threads per program
    HAS_MASK: tl.constexpr, # 1 -> mask present, 0 -> no mask
    H: tl.constexpr,        # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask_offs = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask_offs, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask_offs, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask_offs, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask_offs, other=0.0)

    if HAS_MASK:
        # each mask entry corresponds to H consecutive hidden values
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask_offs, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask_offs)
    tl.store(right_ptr + offs, right, mask=mask_offs)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim, nomask):
    """
    In‑place multiplication:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    If `nomask` is True the mask is omitted.
    All tensors must be contiguous float16.
    """
    total = left.numel()
    BLOCK = 1024                         # max threads per block on B200
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 0 if nomask else 1
    # When mask is not used we can pass any pointer; we reuse left to keep the signature
    mask_ptr = mask if not nomask else left

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # The operation is performed in‑place; nothing to return


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored if config['nomask']=True
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision for the remaining ops
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half) – flatten the (B,N,N) axes
    # ------------------------------------------------------------------
    B, N, _, _ = x_h.shape
    x_flat = x_h.reshape(-1, dim)               # (B*N*N, dim)

    w_left = weights["left_proj.weight"].to(torch.float16)
    w_right = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    left = F.linear(x_flat, w_left)             # (B*N*N, hidden_dim)
    right = F.linear(x_flat, w_right)

    left_gate = F.linear(x_flat, w_left_gate).sigmoid()
    right_gate = F.linear(x_flat, w_right_gate).sigmoid()
    out_gate = F.linear(x_flat, w_out_gate).sigmoid()

    # ------------------------------------------------------------------
    # 3. Reshape to [B, N, N, hidden_dim] (contiguous)
    # ------------------------------------------------------------------
    left = left.view(B, N, N, hidden_dim).contiguous()
    right = right.view(B, N, N, hidden_dim).contiguous()
    left_gate = left_gate.view(B, N, N, hidden_dim).contiguous()
    right_gate = right_gate.view(B, N, N, hidden_dim).contiguous()
    out_gate = out_gate.view(B, N, N, hidden_dim).contiguous()

    # ------------------------------------------------------------------
    # 4. Fuse gates (and mask if present) via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # mask broadcast to hidden_dim, shape [B, N, N, 1] → flatten
        mask_h = mask.to(torch.float16).contiguous().view(-1)
    else:
        mask_h = None  # dummy, not used

    _apply_gate_and_mask(
        left, right,
        left_gate, right_gate,
        mask_h,
        hidden_dim,
        nomask,
    )
    # Gates are no longer needed
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 5. Cubic contraction: out_{i,j,d} = Σ_k left_{i,k,d} * right_{j,k,d}
    #    Implemented as a strided‑batched GEMM (half precision)
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden_dim, N, N) for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)               # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)             # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final projection
    # ------------------------------------------------------------------
    w_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    w_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w_out_norm_w,
        bias=w_out_norm_b,
    )
    out = out * out_gate                                # element‑wise scaling
    w_out = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, w_out)                          # (B, N, N, dim)

    # Return in the original (float32) dtype
    return out.to(torch.float32)
