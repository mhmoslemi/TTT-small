@triton.jit
def gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total: tl.int32, H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    if HAS_MASK:
        # compute mask index
        pos = offs // H   # integer division
        mask_val = tl.load(mask_ptr + pos, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

Note: mask_val default when mask absent should be 1. Use other=1.0.

"""
TriMul (outgoing) forward pass optimized with Triton.

Algorithm:
1. Half‑precision LayerNorm on the input.
2. Fuse the five linear projections (left_proj, right_proj, left_gate,
   right_gate, out_gate) into a single matrix multiplication by concatenating
   their weight matrices.
3. Split the result, apply sigmoid to the three gate tensors.
4. Fuse mask (if present) and gate multiplication into an in‑place Triton
   kernel that updates `left` and `right`.
5. Compute the O(N³) contraction as a batched GEMM:
   out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   which equals left @ rightᵀ for each hidden channel.
6. Apply LayerNorm over the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to the original `dim`.
All heavy arithmetic is performed in float16; the final result is cast to
float32 as required by the interface.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse left/right gating and optional mask
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,           # float16 *
    right_ptr,          # float16 *
    left_gate_ptr,      # float16 *
    right_gate_ptr,     # float16 *
    mask_ptr,           # float16 * (optional, broadcasted)
    total_elements: tl.int32,    # total number of elements = B*N*N*H
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,      # 1 if mask present, 0 otherwise
    H: tl.constexpr,             # hidden dimension size
):
    """In‑place: left = left * left_gate * mask; right = right * right_gate * mask."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load data
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # Compute position index for mask (mask is broadcast over hidden dim)
        pos = offs // H
        mask_val = tl.load(mask_ptr + pos, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store results back in place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) – may be None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (half precision)
    # ------------------------------------------------------------------
    # Convert to FP16 early.
    x = input_tensor.to(torch.float16)
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"].to(torch.float16),
        bias=weights["norm.bias"].to(torch.float16),
    )  # [B, N, N, dim] FP16

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 2. Fuse all linear projections (proj + gates + out_gate) into one GEMM
    # ------------------------------------------------------------------
    # Flatten for matrix multiplication: (B*N*N, dim)
    x_flat = x.view(-1, dim)  # shape (B*N*N, dim)

    # Concatenate weight matrices along the output dimension.
    # Each weight has shape (hidden, dim); out_gate also uses hidden.
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # shape (5*hidden, dim)

    # Single linear pass
    out_concat = F.linear(x_flat, w_concat)  # (B*N*N, 5*hidden)
    out_concat = out_concat.view(B, N, N, 5 * hidden)  # (B, N, N, 5*hidden)

    # Split the concatenated output
    left = out_concat[..., 0:hidden]                     # proj left
    right = out_concat[..., hidden:2 * hidden]           # proj right
    left_gate = out_concat[..., 2 * hidden:3 * hidden].sigmoid()
    right_gate = out_concat[..., 3 * hidden:4 * hidden].sigmoid()
    out_gate = out_concat[..., 4 * hidden:5 * hidden].sigmoid()

    # ------------------------------------------------------------------
    # 3. Fuse mask (if any) and gate multiplication via Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> broadcast over hidden dim
        mask_h = mask.to(torch.float16).unsqueeze(-1).contiguous()  # (B, N, N, 1)
        mask_flat = mask_h.view(-1)  # (B*N*N,)

        # Ensure tensors are contiguous for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        total = left_f.numel()  # B * N * N * hidden
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)

        _gate_mask_fuse_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_flat,
            total,
            BLOCK=BLOCK,
            HAS_MASK=1,
            H=hidden,
        )

        # Reshape back (views share storage)
        left = left_f.view_as(left)
        right = right_f.view_as(right)

        # Cleanup temporary tensors
        del left_f, right_f, left_gate_f, right_gate_f, mask_flat, mask_h
    else:
        # Simple element‑wise multiplication when no mask
        left = left * left_gate
        right = right * right_gate

    # Gates no longer needed
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 4. Core contraction: batched GEMM over the hidden dimension
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden, N, N) for efficient batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, hidden, N, N)

    B2, H2, N2, _ = left_perm.shape
    left_mat = left_perm.view(B2 * H2, N2, N2)    # (B*hidden, N, N)
    right_mat = right_perm.view(B2 * H2, N2, N2)  # (B*hidden, N, N)

    # Batched matrix multiplication with transposed right operand
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*hidden, N, N)

    # Restore original layout: (B, N, N, hidden)
    out = out_mat.view(B2, H2, N2, N2).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, out‑gate scaling, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise gating (half)

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )  # (B, N, N, dim) in FP16

    # Return in float32 as required by the API
    return out.to(torch.float32)
