```
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(left_ptr, right_ptr,
                           left_gate_ptr, right_gate_ptr,
                           mask_ptr,
                           total_elems: tl.int32,
                           BLOCK: tl.constexpr,
                           HAS_MASK: tl.constexpr,
                           H: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # compute mask index broadcasting across H
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)
```

Potential changes: Use tl.float16 for other.

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # layernorm of input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Move to appropriate dtype
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias)
    x = x.to(torch.float16)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x, left_proj_weight, bias=None)   # shape [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight, bias=None)

    # Gate values
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x, left_gate_weight, bias=None).sigmoid()
    right_gate = F.linear(x, right_gate_weight, bias=None).sigmoid()
    out_gate = F.linear(x, out_gate_weight, bias=None).sigmoid()

    # Apply mask and gates
    if not nomask and mask is not None:
        # mask is [B,N,N]; cast to float16, keep dimension
        mask_h = mask.to(torch.float16)
        # Ensure contiguous flatten for kernel
        mask_h = mask_h.contiguous()
        # Pass to kernel
        _apply_gate_and_mask(left, right, left_gate, right_gate, mask_h, hidden_dim)
    else:
        left = left * left_gate
        right = right * right_gate

    # Core contraction: batched GEMM
    B, N, N2, H = left.shape
    assert N == N2
    # Permute to [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # reshape to (B*H, N, N)
    batch = B * H
    left_mat = left_perm.view(batch, N, N)
    right_mat = right_perm.view(batch, N, N)
    # Compute out

"""
TriMul (outgoing) forward pass with Triton acceleration.

Algorithm
---------
1. Layer‑norm the input tensor (float32 → float16).
2. Linear projections `left`/`right` and gate projections (`left_gate`,
   `right_gate`, `out_gate`) using half‑precision weights.
3. Fuse mask (if present) with gate multiplications via a Triton kernel.
4. Core contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batch of H independent N×N matrix multiplications.
   We reshape to (B*H, N, N) and use a single torch.bmm call (cuBLAS).
5. Layer‑norm over the hidden dimension, apply `out_gate`, and final
   linear projection to the original dimension.
6. Return result in float32.

The Triton kernel only handles the element‑wise mask+gate fusion; the
O(N³) contraction is performed with a highly‑optimized batched GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # half *
    right_ptr,           # half *
    left_gate_ptr,       # half *
    right_gate_ptr,      # half *
    mask_ptr,            # half *
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present else 0
    H: tl.constexpr,          # hidden dimension (size of innermost channel)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elems

    # Load tensors
    left = tl.load(left_ptr + offs,      mask=valid, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs,    mask=valid, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs,   mask=valid, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=tl.float16(0.0))

    if HAS_MASK:
        # mask is broadcast over the hidden dimension H
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Write back in‑place
    tl.store(left_ptr + offs, left,  mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """In‑place multiplication:
       left  = left * left_gate * mask
       right = right * right_gate * mask
       (mask may be None → no mask applied)."""
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when no mask

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # tensors modified in place


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or numeric) or None
            - weights: dict mapping parameter names to tensors
            - config: dict with keys ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> half)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half precision)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )  # [B, N, N, hidden_dim]

    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gate values (half precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) with gates via Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → half, broadcast over hidden_dim
        mask_h = mask.to(torch.float16).contiguous()
        _apply_gate_and_mask(left, right, left_gate, right_gate, mask_h, hidden_dim)
    else:
        # No mask: simple elementwise multiplication
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction (batched GEMM)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    assert H == hidden_dim
    # Move hidden dim to batch dimension for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    batch = B * H
    left_mat = left_perm.view(batch, N, N)              # [B*H, N, N]
    right_mat = right_perm.view(batch, N, N)            # [B*H, N, N]

    # Batched matrix multiplication: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N]

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in float32 as required
    return out.to(torch.float32)
