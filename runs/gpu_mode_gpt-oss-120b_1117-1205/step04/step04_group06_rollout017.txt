    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # Cast weight tensors to appropriate dtype
    # Use half for speed

    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias = weights["norm.bias"].to(torch.float16)
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    to_out_weight = weights["to_out.weight"].to(torch.float16)

    # Input LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    x_h = x.to(torch.float16)

    # Projections
    left = F.linear(x_h, left_proj_weight, bias=None)
    right = F.linear(x_h, right_proj_weight, bias=None)

    # Gates
    left_gate = F.linear(x_h, left_gate_weight, bias=None).sigmoid()
    right_gate = F.linear(x_h, right_gate_weight, bias=None).sigmoid()
    out_gate = F.linear(x_h, out_gate_weight, bias=None).sigmoid() # shape [B,N,N,hidden]

    if not nomask:

        right_f = right.contiguous().view(-1)

            mask_f,
            total,
            BLOCK=BLOCK,
            HAS_MASK=1,
            H=hidden_dim,
        )
        # Reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
        # No need to reshape gates; they already used.

        right = right * right_gate

    # Clear gating tensors to free memory optionally
    del left_gate, right_gate, x_h

    left_perm = left.permute(0, 3, 1, 2).contiguous()

    out_final = F.linear(out_scaled, to_out_weight, bias=None)

"""
TriMul (outgoing) forward pass optimized for GPU.

Algorithm
---------
1. Layer‑normalize the input tensor and cast to fp16.
2. Compute left / right projections (Linear) and three gates (Linear + sigmoid).
3. Fuse element‑wise mask (if present) with the left/right gates in a single Triton
   kernel, updating left and right in‑place.
4. Perform the O(N³) contraction
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   as a batched GEMM:
      - reshape left/right to (B*H, N, N)
      - torch.bmm(left, rightᵀ)
   and restore the original layout.
5. Apply output LayerNorm, scale by the out‑gate, and a final linear projection.
6. Return the result in float32.

Only the mask‑gate fusion is implemented in Triton; the heavy matrix multiply
uses cuBLAS via torch.bmm, which is already highly optimized.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # fp16*
    right_ptr,         # fp16*
    left_gate_ptr,     # fp16*
    right_gate_ptr,    # fp16*
    mask_ptr,          # fp16* (flattened mask, length = B*N*N)
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if a mask is supplied, 0 otherwise
    H: tl.constexpr,          # hidden dimension (size of innermost channel)
):
    """In‑place: left = left * left_gate * mask; right = right * right_gate * mask."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elems

    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # Each hidden entry shares the same mask entry.
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def _fuse_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """Fuse gates (and optional mask) into left/right tensors in‑place using Triton."""
    total = left.numel()
    BLOCK = 1024  # threads per program; adjust if needed
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy ptr when HAS_MASK==0

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # In‑place update; nothing is returned.


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple containing (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------- 1. LayerNorm --------------------
    norm_weight = weights["norm.weight"].to(torch.float16, device=device)
    norm_bias = weights["norm.bias"].to(torch.float16, device=device)

    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    x_h = x.to(torch.float16)

    # -------------------- 2. Projections --------------------
    left_proj_weight = weights["left_proj.weight"].to(torch.float16, device=device)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16, device=device)

    left = F.linear(x_h, left_proj_weight, bias=None)   # [B, N, N, H]
    right = F.linear(x_h, right_proj_weight, bias=None)  # [B, N, N, H]

    # -------------------- 3. Gates --------------------
    left_gate_weight = weights["left_gate.weight"].to(torch.float16, device=device)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16, device=device)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16, device=device)

    left_gate = F.linear(x_h, left_gate_weight, bias=None).sigmoid()
    right_gate = F.linear(x_h, right_gate_weight, bias=None).sigmoid()
    out_gate = F.linear(x_h, out_gate_weight, bias=None).sigmoid()   # used later

    # -------------------- 4. Fuse mask and gates --------------------
    if not nomask:
        # mask: [B, N, N] -> half, flattened to [B*N*N]
        mask_h = mask.to(torch.float16, device=device).contiguous().view(-1)

        # Ensure all tensors are contiguous for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _fuse_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_h,
            hidden_dim,
        )

        # Reshape back to original layout (views share storage)
        left = left_f.view_as(left)
        right = right_f.view_as(right)
        # left_gate/right_gate are no longer needed
        del left_gate_f, right_gate_f, mask_h, left_f, right_f
    else:
        # Simple element‑wise multiplication when no mask is present
        left = left * left_gate
        right = right * right_gate

    # Gates no longer needed
    del left_gate, right_gate, x_h

    # -------------------- 5. Core contraction (batched GEMM) --------------------
    # Bring hidden dimension to the batch dimension: (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_perm.shape

    left_mat = left_perm.view(B * H, N, N)          # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)        # (B*H, N, N)

    # out_mat[b*h, i, j] = Σ_k left_mat[b*h, i, k] * right_mat[b*h, j, k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # -------------------- 6. Output LayerNorm, out‑gate, final projection --------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16, device=device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16, device=device)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    out = out * out_gate  # element‑wise gating

    to_out_weight = weights["to_out.weight"].to(torch.float16, device=device)
    out = F.linear(out, to_out_weight, bias=None)   # (B, N, N, dim)

    # Return in float32 as required by the interface
    return out.to(torch.float32)
