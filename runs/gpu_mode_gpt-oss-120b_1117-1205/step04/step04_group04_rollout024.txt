"""
Outgoing TriMul (AlphaFold3) forward pass implemented with Triton.

Algorithm
---------
1. Layer‑norm the input (float32 → float16).
2. Linear projections to the hidden dimension (no bias).
3. Compute three gate tensors and apply sigmoid.
4. Fuse optional mask with the left/right gate multiplications in a
   Triton kernel (in‑place).
5. Core O(N³·H) contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (A @ Bᵀ) where the hidden dimension is
   folded into the batch dimension and executed by a custom Triton kernel.
6. Output LayerNorm, element‑wise gating with out_gate and a final linear
   projection back to the original feature dimension.
7. Return the result as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: fuse mask and gates (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # *half
    right_ptr,         # *half
    left_gate_ptr,     # *half
    right_gate_ptr,    # *half
    mask_ptr,          # *half (broadcasted over hidden dim)
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,   # hidden dimension
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elems

    left = tl.load(left_ptr + offs, mask=m, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=m, other=tl.float16(0.0))
    lgate = tl.load(left_gate_ptr + offs, mask=m, other=tl.float16(0.0))
    rgate = tl.load(right_gate_ptr + offs, mask=m, other=tl.float16(0.0))

    if HAS_MASK:
        # one mask value per (b,i,j) -> repeated over hidden dim
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=m, other=tl.float16(0.0))
        left = left * lgate * mask_val
        right = right * rgate * mask_val
    else:
        left = left * lgate
        right = right * rgate

    tl.store(left_ptr + offs, left, mask=m)
    tl.store(right_ptr + offs, right, mask=m)


# ----------------------------------------------------------------------
# Kernel 2: batched GEMM (A @ Bᵀ) for the O(N³) contraction
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ab, stride_am, stride_ak,          # A: batch, row, col
    stride_bb, stride_bk, stride_bn,          # Bᵀ: batch, row(k), col(j)
    stride_cb, stride_cm, stride_cn,          # C: batch, row, col
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)               # batch index (B*H)
    pid_m = tl.program_id(0)               # block row
    pid_n = tl.program_id(1)               # block col

    # Offsets in the M and N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointer to the output tile
    c = c_ptr + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a = a_ptr + pid_b * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a_tile = tl.load(a, mask=mask_m[:, None] & mask_k[None, :], other=tl.float16(0.0)).to(tl.float32)

        # Load Bᵀ tile (K x N) – note B is accessed transposed
        b = b_ptr + pid_b * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b_tile = tl.load(b, mask=mask_k[:, None] & mask_n[None, :], other=tl.float16(0.0)).to(tl.float32)

        # Accumulate
        acc += tl.dot(a_tile, b_tile)

    # Write back
    c_tile = acc.to(tl.float16)
    tl.store(c, c_tile, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, D] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict mapping parameter names to float32 tensors
            - config: dict containing at least "dim" and "hidden_dim",
                      optionally "nomask" (bool)

    Returns:
        torch.Tensor [B, N, N, D] (float32)
    """
    inp, mask, weights, config = data
    device = inp.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> half)
    # --------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections to hidden dim
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 3. Gates (sigmoid)
    # --------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # --------------------------------------------------------------
    # 4. Fuse mask with gates (if required) via Triton
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape: [B,N,N] -> broadcast over hidden dim
        mask_h = mask.to(torch.float16).unsqueeze(-1)          # [B,N,N,1]

        # flatten tensors for 1‑D kernel launch
        left_f = left.view(-1).contiguous()
        right_f = right.view(-1).contiguous()
        lg_f = left_gate.view(-1).contiguous()
        rg_f = right_gate.view(-1).contiguous()
        mask_f = mask_h.view(-1).contiguous()                 # length B*N*N

        total = left_f.numel()
        BLOCK = 16384                                          # < 65536 threads per program
        grid = (triton.cdiv(total, BLOCK),)

        _mask_gate_fuse_kernel[grid](
            left_f,
            right_f,
            lg_f,
            rg_f,
            mask_f,
            total,
            BLOCK=BLOCK,
            HAS_MASK=1,
            H=hidden_dim,
        )

        # reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # no mask – simple elementwise multiplication
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction: batched GEMM (A @ Bᵀ)
    # --------------------------------------------------------------
    B, N, _, _ = left.shape                     # left shape (B,N,N,H)
    # Fold hidden dimension into the batch dimension
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous()
    left_mat = left_t.view(B * hidden_dim, N, N)     # (B*H, N, N)
    right_mat = right_t.view(B * hidden_dim, N, N)

    # Allocate output
    out_mat = torch.empty_like(left_mat)            # (B*H, N, N), fp16

    # Strides (in elements)
    stride_ab = left_mat.stride(0)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)

    # For Bᵀ we need row‑stride = original col‑stride, col‑stride = original row‑stride
    stride_bb = right_mat.stride(0)
    stride_bk = right_mat.stride(2)   # k dimension (original col)
    stride_bn = right_mat.stride(1)   # j dimension (original row)

    stride_cb = out_mat.stride(0)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)

    # Block sizes – tuned for typical B200-like GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_blocks_m = triton.cdiv(N, BLOCK_M)
    num_blocks_n = triton.cdiv(N, BLOCK_N)

    grid = (num_blocks_m, num_blocks_n, B * hidden_dim)

    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_ab, stride_am, stride_ak,
        stride_bb, stride_bk, stride_bn,
        stride_cb, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return in float32 as required
    return out.to(torch.float32)
