```
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Batch id
    batch_id = tl.program_id(0)   # 0..total_batches-1
    # Compute block ids for i and j
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # derive batch and hidden idx
    b = batch_id // H
    h = batch_id % H

    # pointers offset for this batch and hidden slice
    base_left = left_ptr + b * stride_left_b + h * stride_left_h
    base_right = right_ptr + b * stride_left_b + h * stride_left_h   # same stride for right; assume same layout
    base_out = out_ptr + b * stride_out_b + h * stride_out_h

    # offsets
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for i/j dimensions
    i_mask = i_off < N
    j_mask = j_off < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # Load left tile (i,k)
        left_ptrs = base_left + i_off[:, None] * stride_left_i + k_off[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0).to(tl.float32)

        # Load right tile (k,j) -> need to index as (k, j)
        right_ptrs = base_right + k_off[:, None] * stride_left_k + j_off[None, :] * stride_left_i
        # Wait right stride for k is stride_left_k (since k is third dimension). Row dimension j uses stride_i.
        right = tl.load(right_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0).to(tl.float32)

        # Compute outer product accumulation across k dimension
        # Using tl.dot which does matrix multiplication (M,K) * (K,N) -> (M,N)
        acc += tl.dot(left, right)

    # Write back output
    out_ptrs = base_out + i_off[:, None] * stride_out_i + j_off[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=i_mask[:, None] & j_mask[None, :])
```

    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

"""
TriMul outgoing implementation (AlphaFold3 style).

The heavy O(N³·H) contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is performed by a custom Triton kernel that treats the hidden dimension `h`
as an extra batch dimension.  The kernel executes a blocked GEMM (M×K @ K×N)
for each (b, h) pair, using fp16 inputs and fp32 accumulation.  All other
operations (LayerNorm, Linear, gating, mask handling) are done in PyTorch
with half precision; the final result is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: blocked GEMM for the TriMul contraction.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,       # * half, shape [B, N, N, H]
    right_ptr,      # * half, shape [B, N, N, H]
    out_ptr,        # * half, shape [B, N, N, H]
    B: tl.int32,
    N: tl.int32,
    H: tl.int32,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: for each (b, h) compute out[b,:, :, h] = left @ right^T."""
    # --- batch (b, h) -------------------------------------------------
    batch_id = tl.program_id(0)                 # 0 .. B*H-1
    pid_i = tl.program_id(1)                    # block row index
    pid_j = tl.program_id(2)                    # block col index

    b = batch_id // H
    h = batch_id % H

    # base pointers for this (b, h) slice
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_left_b  + h * stride_left_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # --- tile offsets -------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --- loop over k --------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: [BLOCK_M, BLOCK_K]  (i, k)
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right tile (transposed): [BLOCK_K, BLOCK_N]  (k, j)
        b_ptrs = right_base + offs_k[:, None] * stride_left_k + offs_j[None, :] * stride_left_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # accumulate block result
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # --- write back ----------------------------------------------------
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Helper: perform the O(N³·H) contraction via Triton
# ----------------------------------------------------------------------
def triton_tri_mul_contraction(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """Compute out = Σ_k left[...,k,:] * right[...,k,:] using Triton.
    left, right: [B, N, N, H] (float16)
    returns out: same shape (float16)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides (int64 -> int32 for Triton)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # block sizes (must satisfy BLOCK_M * BLOCK_N <= 1024)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    total_batches = B * H
    grid = (
        total_batches,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict mapping parameter names to tensors (float32)
            - config: dict containing at least keys 'dim' and 'hidden_dim',
                      optional 'nomask' (bool)

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> half)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device),
        bias=weights["norm.bias"].to(device),
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half precision)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16, device=device),
        bias=None,
    )  # [B, N, N, hidden_dim]

    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16, device=device),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gates (half precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16, device=device),
        bias=None,
    ).sigmoid()

    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16, device=device),
        bias=None,
    ).sigmoid()

    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16, device=device),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Apply mask (if present) and gates
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape [B, N, N] -> broadcast over hidden_dim
        mask_h = mask.unsqueeze(-1).to(torch.float16)
        left = left * left_gate * mask_h
        right = right * right_gate * mask_h
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core triangular multiplication (Triton)
    # ------------------------------------------------------------------
    out = triton_tri_mul_contraction(left, right)   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16, device=device),
        bias=weights["to_out_norm.bias"].to(torch.float16, device=device),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16, device=device),
        bias=None,
    )

    # Return in the required precision
    return out.to(torch.float32)
