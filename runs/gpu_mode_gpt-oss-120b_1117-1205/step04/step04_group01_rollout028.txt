def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. LayerNorm (float32 -> half)
    # Use torch.nn.functional.layer_norm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    x = x.to(torch.float16)

    # 2. Linear projections and gates (half)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = torch.nn.functional.linear(x, weights["right_proj.weight"].to(torch.float16))
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # 3. Fuse mask and gates
    if (not nomask) and mask is not None:
        # mask shape: [B,N,N], expand to last dim
        mask_h = mask.unsqueeze(-1).to(torch.float16)
        # ensure contiguous
        left = left.contiguous()

"""
TriMul “outgoing” forward pass with Triton acceleration.

- LayerNorm and linear projections are performed in PyTorch (FP16).
- A small Triton kernel fuses mask, left/right gates, and optional mask.
- The O(N³) contraction is implemented as a batched matrix multiplication
  kernel: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
  This kernel tiles the (i, j) output plane and iterates over k in blocks,
  accumulating in FP32 and storing FP16.  It handles arbitrary N up to 1024
  and any batch/hidden dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: fuse mask (if any) and gate multiplication in‑place
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,               # half *
    left_gate_ptr, right_gate_ptr,     # half *
    mask_ptr,                          # half * (optional)
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,            # 1 if mask exists, else 0
    H: tl.constexpr,                   # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elems

    left      = tl.load(left_ptr + offs,      mask=valid, other=0.0)
    right     = tl.load(right_ptr + offs,     mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate= tl.load(right_gate_ptr + offs,mask=valid, other=0.0)

    if HAS_MASK:
        # each mask element corresponds to H consecutive hidden entries
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=0.0)
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    tl.store(left_ptr  + offs, left,  mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def _fuse_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """In‑place fusion of left/right with their gates and an optional mask."""
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when no mask

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )


# ----------------------------------------------------------------------
# Kernel 2: batched matmul for out = left @ right^T
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    B, N, H,                              # batch, seq_len, hidden_dim
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)   # flattened batch*hidden index
    pid_m = tl.program_id(1)   # tile row (i)
    pid_n = tl.program_id(2)   # tile col (j)

    # decode batch and hidden indices
    b = pid_z // H
    h = pid_z % H

    # base pointers for this (b, h) matrix
    left_base  = b * stride_left_b  + h * stride_left_h
    right_base = b * stride_right_b + h * stride_right_h
    out_base   = b * stride_out_b   + h * stride_out_h

    # start offsets for this tile
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # range vectors
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds rows/cols
    mask_i = i < N
    mask_j = j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k_tiles):
        k0 = kk * BLOCK_K
        k_idx = k0 + k
        mask_k = k_idx < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + (i[:, None] * stride_left_i
                             + k_idx[None, :] * stride_left_k
                             + left_base)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # right tile transposed (k as rows, j as cols): shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_ptr + (k_idx[:, None] * stride_right_k
                              + j[None, :] * stride_right_i
                              + right_base)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # accumulate
        acc += tl.dot(a, b)

    # write result
    out_ptrs = out_ptr + (i[:, None] * stride_out_i
                          + j[None, :] * stride_out_j
                          + out_base)
    out_val = acc.to(tl.float16)
    tl.store(out_ptrs, out_val, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or numeric) or None
            - weights: dictionary of model parameters
            - config: dict containing ``dim`` and ``hidden_dim`` (and optional ``nomask``)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm + cast to FP16
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (FP16)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16), bias=None)

    # ------------------------------------------------------------------
    # 3. Gate values (FP16)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate   = F.linear(x, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) with gates via Triton
    # ------------------------------------------------------------------
    if (not nomask) and mask is not None:
        # mask: [B, N, N] -> [B, N, N, 1] -> broadcast over hidden_dim
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        _fuse_gate_and_mask(
            left, right,
            left_gate, right_gate,
            mask_h,
            hidden_dim,
        )
    else:
        left  = left  * left_gate
        right = right * right_gate

    # Ensure contiguous layout for the matmul kernel
    left  = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 5. Core contraction: out = left @ right^T  (batched GEMM)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # left shape: [B, N, N, hidden_dim]
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    groups = B * hidden_dim
    grid = (groups,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    # stride tuples (in elements, not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    _batched_matmul_kernel[grid](
        left, right, out,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        B, N, hidden_dim,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    # return in float32 as required
    return out.to(torch.float32)
