def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. LayerNorm
    x_norm = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x_h = x_norm.to(torch.float16)

    # 2. Projections
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # 3. Gates
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # 4. Apply mask and gates
    if not nomask and mask is not None:
        # Convert mask to half, broadcast
        mask_h = mask.to(torch.float16)
        # Flatten tensors
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask_h.unsqueeze(-1).contiguous().view(-1) # shape B*N*N
        _apply_gate_and_mask(left_f, right_f, left_gate_f, right_gate_f, mask_f, hidden_dim)
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # 5. Core contraction using Triton kernel
    B, N, _, _ = left.shape  # B,N,N,H

    # Permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Collapse batch and hidden dim
    batch_H = B * hidden_dim
    left_mat = left_perm.view(batch_H, N, N)
    right_mat = right_perm.view(batch_H, N, N)

    # Output mat
    out_mat = torch.empty_like(left_mat)

    # Compute batch stride (elements per matrix)
    batch_stride = N * N

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (batch_H, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        batch_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # 6. Output layernorm, gate, linear
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Convert to float32
    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass with Triton fusion.

Algorithm
---------
1. Layer‑norm the input tensor (float32 → float16).
2. Compute left/right linear projections and three sigmoid gates (float16).
3. Fuse left/right gates and the optional mask into the projected tensors
   using a small Triton kernel (in‑place, half‑precision).
4. Core O(N³) contraction: for each hidden channel h,
   out[...,i,j,h] = Σ_k left[...,i,k,h] * right[...,j,k,h].
   Implemented as a batched GEMM C = A @ Bᵀ with a custom Triton kernel
   that processes each (batch * hidden) matrix independently.
5. Output layer‑norm, out‑gate scaling and final linear projection back
   to the original channel dimension.
All heavy arithmetic is performed in float16 with float32 accumulation.
The final result is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: fuse mask (if present) and sigmoid gates into left/right tensors
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # half* left tensor (flattened)
    right_ptr,           # half* right tensor (flattened)
    left_gate_ptr,       # half* left‑gate tensor (flattened)
    right_gate_ptr,      # half* right‑gate tensor (flattened)
    mask_ptr,            # half* mask tensor (flattened, broadcast over hidden dim)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask is present, else 0
    H: tl.constexpr,          # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # one mask value per (batch, i, k) group of H hidden values
        mask_offs = offs // H
        mask_val = tl.load(mask_ptr + mask_offs, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of gating and optional masking using Triton.
    All tensors must be contiguous half‑precision.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when mask absent

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # No return – tensors are modified in place


# ----------------------------------------------------------------------
# Kernel 2: batched matrix multiplication C = A @ Bᵀ
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr,            # half* pointer to A (batch, N, N)
    b_ptr,            # half* pointer to B (batch, N, N)
    c_ptr,            # half* pointer to C (batch, N, N)
    N: tl.int32,      # sequence length
    batch_stride: tl.int32,   # N * N (elements per matrix)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute start offsets of this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for the current batch
    a_batch = a_ptr + batch_id * batch_stride
    b_batch = b_ptr + batch_id * batch_stride
    c_batch = c_ptr + batch_id * batch_stride

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds
        a_mask = (offs_m[:, None] < N) & (offs_k[None, :] < N)
        b_mask = (offs_n[:, None] < N) & (offs_k[None, :] < N)

        # Pointers to tiles (row‑major layout)
        a_ptrs = a_batch + (offs_m[:, None] * N + offs_k[None, :])
        b_ptrs = b_batch + (offs_n[:, None] * N + offs_k[None, :])

        a_tile = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b_tile = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # Transpose B tile to obtain (K, N) layout for dot product
        b_T = tl.trans(b_tile, (1, 0))

        # Accumulate
        acc += tl.dot(a_tile, b_T)

    # Write back result
    c_ptrs = c_batch + (offs_m[:, None] * N + offs_n[None, :])
    c_mask = (offs_m[:, None] < N) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> half)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections (half)
    # --------------------------------------------------------------
    left = F.linear(
        x_h, weights["left_proj.weight"].to(torch.float16), bias=None
    )
    right = F.linear(
        x_h, weights["right_proj.weight"].to(torch.float16), bias=None
    )

    # --------------------------------------------------------------
    # 3. Gates (half)
    # --------------------------------------------------------------
    left_gate = F.linear(
        x_h, weights["left_gate.weight"].to(torch.float16), bias=None
    ).sigmoid()
    right_gate = F.linear(
        x_h, weights["right_gate.weight"].to(torch.float16), bias=None
    ).sigmoid()
    out_gate = F.linear(
        x_h, weights["out_gate.weight"].to(torch.float16), bias=None
    ).sigmoid()

    # --------------------------------------------------------------
    # 4. Fuse mask (if present) with gates via Triton
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # Ensure contiguous half tensors
        left_c = left.contiguous()
        right_c = right.contiguous()
        left_gate_c = left_gate.contiguous()
        right_gate_c = right_gate.contiguous()
        mask_c = mask.to(torch.float16).contiguous()

        # Flatten all tensors
        left_f = left_c.view(-1)
        right_f = right_c.view(-1)
        left_gate_f = left_gate_c.view(-1)
        right_gate_f = right_gate_c.view(-1)
        mask_f = mask_c.view(-1)  # shape B*N*N

        _apply_gate_and_mask(
            left_f, right_f, left_gate_f, right_gate_f, mask_f, hidden_dim
        )

        # Restore original shapes
        left = left_f.view_as(left_c)
        right = right_f.view_as(right_c)
    else:
        # No mask ⇒ simple element‑wise gating
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM C = A @ Bᵀ with a Triton kernel.
    # --------------------------------------------------------------
    B, N, _, _ = left.shape  # left shape [B, N, N, hidden_dim]

    # Move hidden dimension to batch axis for the GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    batch_H = B * hidden_dim
    left_mat = left_perm.view(batch_H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(batch_H, N, N) # (B*H, N, N)

    out_mat = torch.empty_like(left_mat)  # half precision buffer

    # Launch Triton GEMM kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        batch_H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        N * N,               # batch_stride = N * N elements per matrix
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out, weights["to_out.weight"].to(torch.float16), bias=None
    )

    # Return in float32 as required
    return out.to(torch.float32)
