        stride_l_bh, stride_l_i, stride_l_k,
        stride_r_bh, stride_r_k, stride_r_j,

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x = x.to(torch.float16)

    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    if not nomask:
        # mask shape [B, N, N]
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # broadcast
        left = left * left_gate * mask_h
        right = right * right_gate * mask_h
    else:
        left = left * left_gate
        right = right * right_gate

    # Permute to [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_perm.shape

    # reshape to (B*H, N, N)
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # allocate output
    out_mat = torch.empty_like(left_mat)

    # kernel launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (B * H, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    gemm_trimul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(2), right_mat.stride(1), # transposed stride order
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # output layernorm
    out = F.layer_norm(out, (hidden_dim,),
                       weight=weights["to_out_norm.weight"].to(torch.float16),
                       bias=weights["to_out_norm.bias"].to(torch.float16))
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

"""
TriMul (outgoing) forward pass – highly fused implementation.

Algorithm
---------
1. Layer‑norm the input tensor (float32 → float16).
2. Linear projections to a hidden dimension `H` (half‑precision).
3. Compute three gates (left, right, out) with a sigmoid.
4. Apply mask (if provided) and element‑wise gating.
5. Core contraction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM over the `k` dimension.
   Implemented with a custom Triton kernel that computes
   left @ rightᵀ for every (batch, head) pair.
6. Output layer‑norm, out‑gate scaling and final linear projection.
7. Cast result back to float32.

The heavy O(N³) contraction is executed entirely on the GPU via Triton,
while the surrounding linear, layer‑norm and gating steps use fast
torch.nn.functional calls.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gemm_trimul_kernel(
    left_ptr,          # half*   [BH, N, N]   (left)
    right_ptr,         # half*   [BH, N, N]   (right)
    out_ptr,           # half*   [BH, N, N]   (output)
    stride_l_bh, stride_l_i, stride_l_k,
    stride_r_bh, stride_r_k, stride_r_j,
    stride_o_bh, stride_o_i, stride_o_j,
    N,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: out = left @ rightᵀ  (both tensors are [BH, N, N])"""
    pid_bh = tl.program_id(0)      # batch * hidden index
    pid_m  = tl.program_id(1)      # tile row
    pid_n  = tl.program_id(2)      # tile col

    #--- tile coordinates -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this (batch, head) pair
    base_l = pid_bh * stride_l_bh
    base_r = pid_bh * stride_r_bh
    base_o = pid_bh * stride_o_bh

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: (BLOCK_M, BLOCK_K)
        ptr_l = left_ptr + base_l + offs_m[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left = tl.load(ptr_l, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # right tile transposed: we need rightᵀ → right_T[k, j] = right[j, k]
        # original right strides: (bh, i, k) → (stride_r_bh, stride_r_j, stride_r_k)
        # we load as (BLOCK_K, BLOCK_N) where rows are k, cols are j
        ptr_r = right_ptr + base_r + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_j
        right = tl.load(ptr_r, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(left, right)

    # store result
    ptr_o = out_ptr + base_o + offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j
    tl.store(ptr_o, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C]    (float32)
        - mask         : torch.Tensor [B, N, N]      (bool/float, may be ignored)
        - weights_dict : dict with all model weights (float32)
        - config_dict  : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half precision)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gates (half) and sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Apply mask (if present) and element‑wise gating
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] → broadcast over hidden dimension
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * left_gate * mask_h
        right = right * right_gate * mask_h
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM (left @ rightᵀ) via Triton.
    # ------------------------------------------------------------------
    # Permute to (B, H, N, N) → treat each head as a separate batch.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    # Collapse batch and head dimensions for the kernel.
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Allocate output buffer.
    out_mat = torch.empty_like(left_mat)

    # Kernel launch configuration.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        B * H,                     # batch * hidden
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
    )
    # Strides for left, right (transposed) and output tensors.
    # left/right are contiguous: stride order (BH, N, N) → (stride_bh, stride_i, stride_k)
    stride_l_bh, stride_l_i, stride_l_k = left_mat.stride()
    stride_r_bh, stride_r_i, stride_r_k = right_mat.stride()
    # For the transposed access of right we swap the last two strides.
    # right_T uses (k, j) ordering → stride_r_k (original k dim) as row stride,
    # stride_r_i (original j dim) as column stride.
    stride_r_j = stride_r_i

    stride_o_bh, stride_o_i, stride_o_j = out_mat.stride()

    _gemm_trimul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_l_bh, stride_l_i, stride_l_k,
        stride_r_bh, stride_r_k, stride_r_j,
        stride_o_bh, stride_o_i, stride_o_j,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in original dtype
    return out.to(torch.float32)
