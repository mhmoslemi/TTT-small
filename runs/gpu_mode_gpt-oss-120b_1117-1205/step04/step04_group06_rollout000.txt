"""
High‑performance outgoing TriMul forward pass (AlphaFold3).

Key ideas
---------
* Input layer‑norm and all linear maps are performed in float16 for speed.
* The expensive O(N³) contraction is realised as a batched GEMM:
      out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
  which maps to a single torch.bmm on (B·H, N, N) matrices.
* A tiny Triton kernel fuses the left/right gates with the optional mask,
  performing the element‑wise multiplications in‑place and avoiding an
  extra memory pass.
* All subsequent layer‑norm, gating and final projection stay in half‑precision.
* The function returns a float32 tensor as required by the interface.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # half*
    right_ptr,           # half*
    left_gate_ptr,       # half*
    right_gate_ptr,      # half*
    mask_ptr,            # half* (broadcasted)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present
    H: tl.constexpr,          # hidden_dim
):
    """In‑place: left = left * left_gate * mask; right = right * right_gate * mask."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # mask broadcast over hidden dimension:
        # each hidden entry shares the same mask entry.
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    Fuse left/right with their gates and optional mask in‑place via Triton.
    All tensors must be contiguous and dtype torch.float16.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy if no mask

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # operation is performed in‑place; nothing to return.


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> half)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gate values (half) with sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse gates (and mask if present) in‑place
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] -> half, flattened
        mask_f = mask.to(torch.float16).contiguous().view(-1)  # [B*N*N]

        # Ensure tensors are contiguous before flattening
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            hidden_dim,
        )
        # Reshape back (views share storage)
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask: simple element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate

    # Release intermediate tensors not needed any more
    del left_gate, right_gate, x

    # ------------------------------------------------------------------
    # 5. Core O(N³) contraction via batched GEMM
    #    out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    # ------------------------------------------------------------------
    # Bring hidden dimension to the second axis for efficient matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)          # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)        # (B*H, N, N)

    # Batched matrix multiplication (right matrix transposed on the k dim)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise gating (half)

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in float32 as required by the interface
    return out.to(torch.float32)
