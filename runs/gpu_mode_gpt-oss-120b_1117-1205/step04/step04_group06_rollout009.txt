def fuse_gate_mask_kernel(
    left_ptr,
    right_ptr,
    left_gate_ptr,
    right_gate_ptr,
    mask_ptr,
    total_elements,
    hidden_dim: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):

    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

        right = right * right_gate
    tl.store(left_ptr + offs, left, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to half
    x_h = x.to(torch.float16)

    # Projections
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16), bias=None)

    # Gates
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # Fuse mask and gates via Triton
    if not nomask:
        # ensure mask is float16 and contiguous
        mask_h = mask.to(torch.float16).contiguous().view(-1)  # shape B*N*N
        # flatten left/right/gates
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        total_elements = left_f.numel()
        BLOCK = 1024  # can be tuned
        grid = (triton.cdiv(total_elements, BLOCK),)
        fuse_gate_mask_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_h,
            total_elements,
            hidden_dim=hidden_dim,
            HAS_MASK=1,
            BLOCK=BLOCK,
        )

        # reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
        # left_gate,right_gate no longer needed
        del left_gate, right_gate, left_gate_f, right_gate_f, mask_h, left_f, right_f
    else:
        # No mask case: just multiply by gates
        left = left * left_gate
        right = right * right_gate
        # free gates
        del left_gate, right_gate

    # Core contraction: batched GEMM
    # Permute to (B, hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # Reshape to [B,N,N,H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output LayerNorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # elementwise

    # Final linear
    out = F.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    # Back to fp32
    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass with Triton fusion.

Algorithm
---------
1. LayerNorm the input [B,N,N,dim] (float32 → float16).
2. Linear projections `left` and `right` to `hidden_dim` (float16).
3. Compute three gating vectors via linear + sigmoid:
   - left_gate, right_gate (used to modulate left/right)
   - out_gate   (applied after the core contraction)
4. Fuse mask (if present) and the left/right gates in a single Triton kernel,
   rewriting `left = left * left_gate * mask` and `right = right * right_gate * mask`.
   The kernel works on flattened tensors; each mask element is broadcast over the
   hidden dimension.
5. Core contraction: for each batch and hidden channel compute
   `out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}`.
   This is a batched matrix multiplication:
   `out = (left_mat @ right_matᵀ)` where
   `left_mat = left.permute(0,3,1,2).view(B*H,N,N)` and similarly for right.
6. Apply a second LayerNorm over the hidden dimension, multiply by `out_gate`,
   and project back to the original `dim` with a final linear layer.
7. Return the result in float32.

Only the mask‑gate fusion is implemented in Triton; all other ops use
highly‑optimized PyTorch/CuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # float16*
    right_ptr,         # float16*
    left_gate_ptr,     # float16*
    right_gate_ptr,    # float16*
    mask_ptr,          # float16* (flattened mask, length B*N*N)
    total_elements,    # int32
    hidden_dim: tl.constexpr,   # hidden dimension (size of innermost channel)
    HAS_MASK: tl.constexpr,     # 1 if mask present, 0 otherwise
    BLOCK: tl.constexpr,        # thread block size
):
    """In‑place fusion:
       left  = left  * left_gate  * (mask if present)
       right = right * right_gate * (mask if present)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=active, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=active, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=tl.float16(0.0))

    if HAS_MASK:
        # each hidden block shares the same mask entry
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store results back
    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict: mapping str → torch.Tensor (float32)
            - config_dict: { "dim": int, "hidden_dim": int, "nomask": bool (optional) }

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # Unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half)
    # ------------------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Compute gates (half) with sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) with left/right gates via Triton kernel
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → half, flattened
        mask_h = mask.to(torch.float16).contiguous().view(-1)  # (B*N*N)

        # Flatten tensors for kernel (must be contiguous)
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        total = left_f.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)

        _gate_mask_fuse_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_h,
            total,
            hidden_dim=hidden_dim,
            HAS_MASK=1,
            BLOCK=BLOCK,
        )
        # Reshape back (views share storage with the original tensors)
        left = left_f.view_as(left)
        right = right_f.view_as(right)

        # Cleanup temporaries
        del left_f, right_f, left_gate_f, right_gate_f, mask_h
    else:
        # No mask: just apply gates
        left = left * left_gate
        right = right * right_gate

    # Gates no longer needed
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via batched GEMM (B * hidden_dim independent matmuls)
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden_dim, N, N) for easy view
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # Batched matrix multiply: left @ right^T  → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore shape to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise gating (half)

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in float32 as required by the interface
    return out.to(torch.float32)
```
