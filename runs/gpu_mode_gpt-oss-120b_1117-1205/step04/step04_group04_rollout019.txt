import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr,
    mask_ptr, total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    lgate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # Each mask element repeats H times across hidden dim.
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * lgate * mask_val
        right = right * rgate * mask_val
    else:
        left = left * lgate
        right = right * rgate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def custom_kernel(data):
    """
    Outgoing TriMul forward pass.

    The implementation follows the reference PyTorch module but
    accelerates the element‑wise mask‑gate fusion with a Triton kernel
    and performs the O(N³) contraction via a half‑precision batched GEMM.
    """
    inp, mask, weights, config = data
    device = inp.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. LayerNorm on input (float32 -> half)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float16, device=device)
    norm_bias = weights["norm.bias"].to(torch.float16, device=device)
    x = F.layer_norm(inp, (dim,), weight=norm_weight, bias=norm_bias)
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections to hidden dimension (no bias)
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16, device=device)
    right_proj_w = weights["right_proj.weight"].to(torch.float16, device=device)
    left = F.linear(x, left_proj_w)   # shape [B,N,N,hidden_dim]

"""
Outgoing TriMul (AlphaFold3) forward pass.

The implementation follows the reference PyTorch module but accelerates the
element‑wise mask‑gate fusion with a small Triton kernel and performs the
core O(N³) contraction via a half‑precision batched matrix multiplication
(torch.matmul). All heavy arithmetic is carried out in fp16; the final
output is cast back to fp32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # *half  (flattened left tensor)
    right_ptr,         # *half  (flattened right tensor)
    left_gate_ptr,     # *half
    right_gate_ptr,    # *half
    mask_ptr,          # *half  (flattened mask, broadcast over hidden dim)
    total_elems: tl.int32,
    BLOCK: tl.constexpr,      # number of elements processed per program
    HAS_MASK: tl.constexpr,   # 1 => mask present, 0 => no mask
    H: tl.constexpr,          # hidden dimension (repeat factor for mask)
):
    """In‑place: left = left * left_gate * mask, right = right * right_gate * mask."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    lgate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # each mask value repeats H times across the hidden dimension
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * lgate * mask_val
        right = right * rgate * mask_val
    else:
        left = left * lgate
        right = right * rgate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict mapping parameter names to float32 tensors
            - config: dict containing at least "dim" and "hidden_dim",
                      optionally "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    inp, mask, weights, config = data
    device = inp.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Layer‑norm on the input (float32 → fp16)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(torch.float16, device=device)
    norm_b = weights["norm.bias"].to(torch.float16, device=device)
    x = F.layer_norm(inp, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections to the hidden dimension (no bias)
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16, device=device)
    right_proj_w = weights["right_proj.weight"].to(torch.float16, device=device)

    left = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)

    # -----------------------------------------------------------------
    # 3. Gate tensors (sigmoid)
    # -----------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16, device=device)
    right_gate_w = weights["right_gate.weight"].to(torch.float16, device=device)
    out_gate_w = weights["out_gate.weight"].to(torch.float16, device=device)

    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # -----------------------------------------------------------------
    # 4. Fuse mask with gates (if a mask is provided) via Triton
    # -----------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → broadcast over hidden dimension
        mask_h = mask.to(torch.float16, device=device).unsqueeze(-1)  # [B, N, N, 1]

        # Ensure contiguous layout before flattening
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_h = mask_h.contiguous()

        total = left.numel()                     # B * N * N * hidden_dim
        BLOCK = 1024                              # threads per program (≤ 1024)
        grid = (triton.cdiv(total, BLOCK),)

        _mask_gate_fuse_kernel[grid](
            left.view(-1),
            right.view(-1),
            left_gate.view(-1),
            right_gate.view(-1),
            mask_h.view(-1),
            total,
            BLOCK=BLOCK,
            HAS_MASK=1,
            H=hidden_dim,
        )
        # left/right now contain the masked‑gated values
    else:
        # Simple element‑wise fusion when no mask is needed
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core O(N³) contraction via a batched half‑precision GEMM
    #    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
    # -----------------------------------------------------------------
    # Permute to (B, hidden_dim, N, N) and perform a batched matmul
    left_perm = left.permute(0, 3, 1, 2)          # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1)        # (B, H, N, N) = rightᵀ over k
    out_perm = torch.matmul(left_perm, right_perm)  # (B, H, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating, and final linear projection back to dim
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16, device=device)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16, device=device)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate                       # element‑wise gating
    to_out_w = weights["to_out.weight"].to(torch.float16, device=device)
    out = F.linear(out, to_out_w)

    # Return as float32
    return out.to(torch.float32)
