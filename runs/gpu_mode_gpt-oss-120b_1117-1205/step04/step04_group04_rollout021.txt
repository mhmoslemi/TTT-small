    input_tensor, mask, weights, config = data
    device = input_tensor.device
    # dims
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)

    # 1. LayerNorm
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=weights["norm.weight"].to(torch.float16),
                     bias=weights["norm.bias"].to(torch.float16))
    x = x.to(torch.float16)

    # 2. Linear proj left/right
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

        mask_h = mask.to(torch.float16).unsqueeze(-1)  # shape [B,N,N,1]

        left_f = left.view(-1)
        right_f = right.view(-1)

        right_gate_f = right_gate.view(-1)
        mask_f = mask_h.view(-1)

        total = left_f.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)
        has_mask = 1

            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,

            HAS_MASK=has_mask,
            H=hidden_dim,

        left = left_f.view_as(left)

        right = right * right_gate

"""
Outgoing TriMul (AlphaFold3) forward pass.

This implementation follows the reference PyTorch version but fuses the mask
application and the three per‑element gates (left, right, out) into a single
Triton kernel.  The expensive O(N³) contraction is performed with a batched
matrix‑multiply (torch.bmm) after reshaping the tensors so that each hidden
channel becomes a separate batch item.  All computations are carried out in
float16 except the final output which is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # *fp16  left tensor (flattened)
    right_ptr,         # *fp16  right tensor (flattened)
    left_gate_ptr,     # *fp16  left gate tensor (flattened)
    right_gate_ptr,    # *fp16  right gate tensor (flattened)
    mask_ptr,          # *fp16  mask tensor (flattened, length = B*N*N)
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,   # hidden dimension
):
    """
    In‑place: left  = left  * left_gate  * mask
              right = right * right_gate * mask
    The mask is broadcasted over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elems

    left       = tl.load(left_ptr + offs,       mask=m, other=tl.float16(0.0))
    right      = tl.load(right_ptr + offs,      mask=m, other=tl.float16(0.0))
    lgate      = tl.load(left_gate_ptr + offs,  mask=m, other=tl.float16(0.0))
    rgate      = tl.load(right_gate_ptr + offs, mask=m, other=tl.float16(0.0))

    if HAS_MASK:
        # one mask entry per (b,i,j) -> stride H in the flattened view
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=m, other=tl.float16(0.0))
        left  = left  * lgate * mask_val
        right = right * rgate * mask_val
    else:
        left  = left  * lgate
        right = right * rgate

    tl.store(left_ptr + offs,  left,  mask=m)
    tl.store(right_ptr + offs, right, mask=m)


def _apply_mask_and_gate(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    Launch the Triton kernel that fuses mask (if present) and the left/right gates
    in‑place.  All tensors must be flattenable to 1‑D views of the same ordering.
    """
    total = left.numel()
    BLOCK = 1024                      # ≤ max threads per block
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    # Triton will not read mask_ptr when HAS_MASK = 0, so any pointer works.
    mask_ptr = mask if mask is not None else left

    _mask_gate_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # In‑place modification; nothing to return.


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/float) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with at least "dim" and "hidden_dim",
                             optionally "nomask" (bool)

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    inp, mask, weights, config = data
    device = inp.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> float16)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"].to(torch.float16, device=device),
        bias=weights["norm.bias"].to(torch.float16, device=device),
    )
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections to hidden dimension (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16, device=device))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16, device=device))

    # -----------------------------------------------------------------
    # 3. Compute gates (including the output gate)
    # -----------------------------------------------------------------
    left_gate = F.linear(
        x, weights["left_gate.weight"].to(torch.float16, device=device)
    ).sigmoid()
    right_gate = F.linear(
        x, weights["right_gate.weight"].to(torch.float16, device=device)
    ).sigmoid()
    out_gate = F.linear(
        x, weights["out_gate.weight"].to(torch.float16, device=device)
    ).sigmoid()

    # -----------------------------------------------------------------
    # 4. Fuse mask (if present) and the left/right gates in‑place via Triton
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B,N,N] -> broadcast over hidden dim
        mask_h = mask.to(torch.float16, device=device).unsqueeze(-1)  # [B,N,N,1]

        # Flatten all tensors for the kernel
        left_f = left.view(-1)
        right_f = right.view(-1)
        left_gate_f = left_gate.view(-1)
        right_gate_f = right_gate.view(-1)
        mask_f = mask_h.view(-1)   # length B*N*N

        _apply_mask_and_gate(
            left_f, right_f, left_gate_f, right_gate_f, mask_f, hidden_dim
        )

        # Reshape back to their original layout
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask: simply apply the left/right gates
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core O(N³) contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched GEMM over the hidden dimension.
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape  # left shape = (B, N, N, hidden_dim)

    # Move hidden dim to the batch axis and make tensors contiguous for cuBLAS
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden_dim, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous() # (B, hidden_dim, N, N)

    left_mat = left_t.view(B * hidden_dim, N, N)   # (B*hidden_dim, N, N)
    right_mat = right_t.view(B * hidden_dim, N, N) # (B*hidden_dim, N, N)

    # Batched matrix multiply: (i,k) × (j,k)ᵀ → (i,j)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*hidden_dim, N, N)

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection back to dim
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16, device=device),
        bias=weights["to_out_norm.bias"].to(torch.float16, device=device),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16, device=device))

    # Return as float32 as required
    return out.to(torch.float32)
