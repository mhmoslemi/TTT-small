@triton.jit
def triton_matmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    N: tl.int32,
    stride_left_batch: tl.int32, stride_left_i: tl.int32, stride_left_k: tl.int32,
    stride_right_batch: tl.int32, stride_right_i: tl.int32, stride_right_k: tl.int32,
    stride_out_batch: tl.int32, stride_out_i: tl.int32, stride_out_j: tl.int32,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for M,N
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        current_k = k_base + offs_k
        mask_k = current_k < N

        # load A block (left)
        a_ptr = left_ptr + pid_bh * stride_left_batch + offs_m[:, None] * stride_left_i + (current_k[None, :] ) * stride_left_k
        a = tl.load(a_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load B block transposed (right): need shape (BLOCK_K, BLOCK_N)
        b_ptr = right_ptr + pid_bh * stride_right_batch + (current_k[:, None]) * stride_right_k + offs_n[None, :] * stride_right_i
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # compute dot product
        acc += tl.dot(a, b)

    # write output
    out_ptrs = out_ptr + pid_bh * stride_out_batch + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass with a fused Triton kernel for the O(N³) contraction.

Algorithm
---------
1. Layer‑norm the 4‑D input (B,N,N,C) using the provided LN parameters (float32).
2. Cast to float16 and apply two linear projections (left/right) to the hidden
   dimension H.
3. Compute three gating linear layers + sigmoid (left_gate, right_gate, out_gate).
4. Optionally multiply by a mask (shape B,N,N) and gate tensors.
5. Core contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM (left @ rightᵀ) for each head h.
   Implemented with a custom Triton kernel that processes all (B*H) matrices
   in parallel, using block‑wise loads and `tl.dot`. Accumulation is performed
   in float32 and the result is stored as float16.
6. Convert the result back to float32, apply a second LayerNorm over the
   hidden dimension, scale by out_gate, and project back to the input dimension.
7. Return the final tensor in float32.

Only the heavy O(N³) part is written in Triton; all other ops use
torch.nn.functional for simplicity and numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction.
# ----------------------------------------------------------------------
@triton.jit
def _triton_trimul_matmul_kernel(
    left_ptr,               # half*   (B*H, N, N)
    right_ptr,              # half*   (B*H, N, N)
    out_ptr,                # half*   (B*H, N, N)
    N: tl.int32,            # sequence length
    stride_l_batch: tl.int32,
    stride_l_i: tl.int32,
    stride_l_k: tl.int32,
    stride_r_batch: tl.int32,
    stride_r_i: tl.int32,
    stride_r_k: tl.int32,
    stride_o_batch: tl.int32,
    stride_o_i: tl.int32,
    stride_o_j: tl.int32,
    BLOCK_M: tl.constexpr,  # tile size in output rows
    BLOCK_N: tl.constexpr,  # tile size in output cols
    BLOCK_K: tl.constexpr,  # reduction tile size
):
    # program IDs
    pid_bh = tl.program_id(0)               # combined batch*head dimension
    pid_m = tl.program_id(1)                # output row block
    pid_n = tl.program_id(2)                # output col block

    # offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------- load A = left (shape: M x K) -------
        a_ptr = (left_ptr
                 + pid_bh * stride_l_batch
                 + offs_m[:, None] * stride_l_i
                 + offs_k[None, :] * stride_l_k)
        a = tl.load(a_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # ------- load Bᵀ = right transposed (shape: K x N) -------
        # right is stored as (i, k). We need (k, j) where j corresponds to column.
        b_ptr = (right_ptr
                 + pid_bh * stride_r_batch
                 + offs_k[:, None] * stride_r_k
                 + offs_n[None, :] * stride_r_i)
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float32)

        # dot product
        acc += tl.dot(a, b)

    # ------- store output tile -------
    c_ptr = (out_ptr
             + pid_bh * stride_o_batch
             + offs_m[:, None] * stride_o_i
             + offs_n[None, :] * stride_o_j)
    tl.store(c_ptr, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights_dict, config_dict)
        input_tensor : torch.Tensor [B, N, N, C] (float32)
        mask         : torch.Tensor [B, N, N] (bool/int) or None
        weights_dict : dict of model parameters (float32 tensors)
        config_dict  : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision for heavy ops
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half precision)
    # ------------------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gates (half) and sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Apply mask (if any) and gates
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape: [B, N, N] → broadcast over hidden dim
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * left_gate * mask_h
        right = right * right_gate * mask_h
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton batched GEMM
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) where the last two dims are the matrices
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    # flatten batch & head dimension for the kernel
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # output placeholder
    out_flat = torch.empty_like(left_flat)

    # strides for the flattened tensors (they are contiguous)
    stride_l_batch = left_flat.stride(0)
    stride_l_i = left_flat.stride(1)
    stride_l_k = left_flat.stride(2)

    stride_r_batch = right_flat.stride(0)
    stride_r_i = right_flat.stride(1)
    stride_r_k = right_flat.stride(2)

    stride_o_batch = out_flat.stride(0)
    stride_o_i = out_flat.stride(1)
    stride_o_j = out_flat.stride(2)

    # Tunable tile sizes (chosen to fit in registers for fp16)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _triton_trimul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_l_batch,
        stride_l_i,
        stride_l_k,
        stride_r_batch,
        stride_r_i,
        stride_r_k,
        stride_o_batch,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final projection
    # ------------------------------------------------------------------
    # Cast to float32 for the final LN + linear for numerical stability
    out_fp32 = out.to(torch.float32)

    out_fp32 = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_fp32 = out_fp32 * out_gate.to(torch.float32)

    out_fp32 = F.linear(
        out_fp32,
        weights["to_out.weight"],
        bias=None,
    )

    # Return float32 result
    return out_fp32
