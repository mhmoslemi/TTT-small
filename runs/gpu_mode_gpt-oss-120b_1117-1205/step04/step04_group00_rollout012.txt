@triton.jit
def gate_mask_fusion_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total: tl.int32,
    H: tl.int32,  # hidden dim
    HAS_MASK: tl.constexpr,  # 1 or 0
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    left = tl.load(left_ptr + offs, mask=mask)
    right = tl.load(right_ptr + offs, mask=mask)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask)

    left = left * left_gate
    right = right * right_gate

    if HAS_MASK:
        idx = offs // H
        mask_val = tl.load(mask_ptr + idx, mask=mask)
        left = left * mask_val
        right = right * mask_val

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype_fp16 = torch.float16

    # Cast weights to fp16
    norm_weight = weights["norm.weight"].to(dtype_fp16)
    norm_bias = weights["norm.bias"].to(dtype_fp16)
    left_proj_w = weights["left_proj.weight"].to(dtype_fp16)  # shape (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(dtype_fp16)
    left_gate_w = weights["left_gate.weight"].to(dtype_fp16)
    right_gate_w = weights["right_gate.weight"].to(dtype_fp16)
    out_gate_w = weights["out_gate.weight"].to(dtype_fp16)
    to_out_norm_weight = weights["to_out_norm.weight"].to(dtype_fp16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(dtype_fp16)
    to_out_w = weights["to_out.weight"].to(dtype_fp16)

    # Step 1: LayerNorm
    # Using torch.nn.functional.layer_norm with half precision
    x = torch.nn.functional.layer_norm(
        input_tensor.to(dtype_fp16),
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B,N,N,dim] half

    # Step 2: Linear projections (hidden_dim)
    # Linear expects weight shape (out_features, in_features)
    left = torch.nn.functional.linear(x, left_proj_w)  # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_w)

    # Step 3: Gates (sigmoid)
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # Step 4: Fuse mask and gates via kernel if mask present
    nomask = config.get("nomask", True)
    if not nomask:
        # Ensure mask is half and reshape
        mask_fp16 = mask.to(dtype_fp16)
        # flatten for kernel
        left_f = left.reshape(-1)
        right_f = right.reshape(-1)
        left_gate_f = left_gate.reshape(-1)
        right_gate_f = right_gate.reshape(-1)

        BLOCK = 1024
        total = left_f.numel()
        grid = (triton.cdiv(total, BLOCK),)
        has_mask = 1
        gate_mask_fusion_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_fp16,
            total,
            hidden_dim,
            HAS_MASK=has_mask,
            BLOCK=BLOCK,
        )
        # Reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # no mask: just multiply by gates
        left = left * left_gate
        right = right * right_gate

    # If mask is present, we also need to multiply the gates themselves? Wait gating used above after multiplication. The kernel multiplies left = left * left_gate * mask and right = right * right_gate * mask, but we also need to keep gate values for out_gate later. The left_gate and right_gate after multiplication not needed after this; we can discard.

    # Step 5: Core contraction: out_{b,i,j,d} = sum_k left[b,i,k,d] * right[b,j,k,d]
    # Permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_perm.shape

    # Reshape for batched matmul
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

        out,
        normalized_shape=(hidden_dim,),

        bias=to_out_norm_bias,

    # Final linear projection to dim
    out = torch.nn.functional.linear(out, to_out_w)  # shape (B,N,N,dim)

    # Cast back to float32
    return out.to(torch.float32)

"""TriMul (outgoing) forward pass.

The implementation follows the reference PyTorch code but:
* Casts everything to float16 early to exploit Tensor‑Core speed.
* Fuses the mask‑and‑gate multiplication for the left/right tensors with a
  small Triton kernel (in‑place, no extra temporaries).
* Performs the O(N³) contraction as a batched half‑precision GEMM
  (torch.bmm) on the reshaped tensors, which is the most efficient
  primitive on current GPUs.
* Applies the remaining LayerNorms and linear projections in half‑precision
  and finally returns a float32 tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half*  (flattened left tensor)
    right_ptr,         # half*  (flattened right tensor)
    left_gate_ptr,     # half*  (flattened left gate)
    right_gate_ptr,    # half*  (flattened right gate)
    mask_ptr,          # half*  (flattened mask, broadcasted over hidden dim)
    total: tl.int32,   # total number of elements in the tensors
    H: tl.int32,       # hidden dimension size
    HAS_MASK: tl.constexpr,  # 1 if mask present else 0
    BLOCK: tl.constexpr,     # block size (threads per program)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total

    # Load tensors
    left = tl.load(left_ptr + offs, mask=m)
    right = tl.load(right_ptr + offs, mask=m)
    left_gate = tl.load(left_gate_ptr + offs, mask=m)
    right_gate = tl.load(right_gate_ptr + offs, mask=m)

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Optional mask (broadcasted over hidden dim)
    if HAS_MASK:
        mask_idx = offs // H                      # index into [B,N,N] mask
        mask_val = tl.load(mask_ptr + mask_idx, mask=m)
        left = left * mask_val
        right = right * mask_val

    # Store results back in‑place
    tl.store(left_ptr + offs, left, mask=m)
    tl.store(right_ptr + offs, right, mask=m)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) – may be ignored
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Cast model parameters to fp16 once
    # ------------------------------------------------------------------
    w_norm = weights["norm.weight"].to(torch.float16, device=device)
    b_norm = weights["norm.bias"].to(torch.float16, device=device)

    w_left_proj = weights["left_proj.weight"].to(torch.float16, device=device)
    w_right_proj = weights["right_proj.weight"].to(torch.float16, device=device)

    w_left_gate = weights["left_gate.weight"].to(torch.float16, device=device)
    w_right_gate = weights["right_gate.weight"].to(torch.float16, device=device)
    w_out_gate = weights["out_gate.weight"].to(torch.float16, device=device)

    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16, device=device)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16, device=device)

    w_to_out = weights["to_out.weight"].to(torch.float16, device=device)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (half precision)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor.to(torch.float16, device=device),
        normalized_shape=(dim,),
        weight=w_norm,
        bias=b_norm,
        eps=1e-5,
    )  # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (fp16)
    # ------------------------------------------------------------------
    left = F.linear(x, w_left_proj)   # [B, N, N, hidden_dim]
    right = F.linear(x, w_right_proj)

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid, fp16)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, w_left_gate).sigmoid()
    right_gate = F.linear(x, w_right_gate).sigmoid()
    out_gate = F.linear(x, w_out_gate).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask with gates (in‑place Triton kernel)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16, device=device).contiguous()
        # Flatten the tensors for the kernel
        left_f = left.reshape(-1).contiguous()
        right_f = right.reshape(-1).contiguous()
        left_gate_f = left_gate.reshape(-1).contiguous()
        right_gate_f = right_gate.reshape(-1).contiguous()

        total_elem = left_f.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total_elem, BLOCK),)
        _gate_mask_fuse_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_fp16,
            total_elem,
            hidden_dim,
            HAS_MASK=1,
            BLOCK=BLOCK,
        )
        # Reshape back to original layout
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask – simply apply the gates
        left = left * left_gate
        right = right * right_gate

    # Gates no longer needed
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as a batched GEMM on half‑precision tensors.
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden_dim, N, N) for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_perm.shape

    # Collapse batch & head dimensions
    left_mat = left_perm.view(B * H, N, N)          # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)        # (B*H, N, N)

    # Batched matrix multiplication: (B*H, N, N) @ (B*H, N, N)^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore layout: (B, N, N, hidden_dim)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, w_to_out)   # (B, N, N, dim) fp16

    # Return in original dtype (float32)
    return out.to(torch.float32)
