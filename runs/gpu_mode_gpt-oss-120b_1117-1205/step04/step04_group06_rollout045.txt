"""
TriMul (outgoing) forward implementation with a fused Triton kernel for
gate+mask multiplication. The heavy O(N³) contraction is performed as a
batched GEMM (left @ rightᵀ) using torch.bmm on half‑precision tensors.
All other operations (LayerNorm, Linear projections, sigmoids) use
torch.nn.functional for speed. The final result is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse gate multiplication and optional mask
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # float16*
    right_ptr,         # float16*
    left_gate_ptr,     # float16*
    right_gate_ptr,    # float16*
    mask_ptr,          # float16* (broadcasted over hidden dim)
    total_elements,    # int32
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask is present, else 0
    H: tl.constexpr,          # hidden_dim
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    # Load operands
    left      = tl.load(left_ptr + offs,      mask=active, other=tl.float16(0))
    right     = tl.load(right_ptr + offs,     mask=active, other=tl.float16(0))
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=tl.float16(0))
    right_gate= tl.load(right_gate_ptr + offs,mask=active, other=tl.float16(0))

    if HAS_MASK:
        # Each hidden channel shares the same mask entry
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=tl.float16(0))
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    # Store results back in‑place
    tl.store(left_ptr + offs,  left,  mask=active)
    tl.store(right_ptr + offs, right, mask=active)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) – may be None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16), bias=None)

    # --------------------------------------------------------------
    # 3. Gate projections + sigmoid (half)
    # --------------------------------------------------------------
    left_gate  = F.linear(x, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate   = F.linear(x, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # --------------------------------------------------------------
    # 4. Fuse gates (and optional mask) with left/right in‑place via Triton
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask → half and flatten
        mask_f = mask.to(torch.float16).contiguous().view(-1)        # [B*N*N]
        # flatten tensors (contiguous view, no copy)
        left_f      = left.contiguous().view(-1)
        right_f     = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f= right_gate.contiguous().view(-1)

        total = left_f.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)

        _gate_mask_fuse_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            total,
            BLOCK=BLOCK,
            HAS_MASK=1,
            H=hidden_dim,
        )
        # reshape back (views share storage)
        left  = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask: simple element‑wise multiplication with gates
        left  = left * left_gate
        right = right * right_gate

    # Free unused tensors early
    del left_gate, right_gate, x

    # --------------------------------------------------------------
    # 5. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as a batched GEMM (left @ rightᵀ)
    # --------------------------------------------------------------
    # Move hidden_dim to the second axis for efficient batched GEMM
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat  = left_perm.view(B * H, N, N)          # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)         # (B*H, N, N)

    # Batched matrix multiplication with transposition on the second operand
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate        # element‑wise gating (half)

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in float32 as required by the interface
    return out.to(torch.float32)
