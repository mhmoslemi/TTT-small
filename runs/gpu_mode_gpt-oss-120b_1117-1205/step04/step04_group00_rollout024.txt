@triton.jit
def gate_mask_fuse_kernel(
    left_ptr,
    right_ptr,
    left_gate_ptr,
    right_gate_ptr,
    mask_ptr,
    total_elements,
    H: tl.constexpr,
    has_mask: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    # apply gates
    left = left * left_gate
    right = right * right_gate

    if has_mask:
        # mask offset index is element index // H
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * mask_val
        right = right * mask_val

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)
```

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input (float16).
2. Fuse all five linear projections (left_proj, right_proj, left_gate,
   right_gate, out_gate) into a single GEMM:
       combined = x_flat @ W_combined   # (M, 5*H)
   where W_combined is the concatenation of the five weight matrices.
   Split the result into pre‑activations for left/right and their gates,
   and the out‑gate.
3. Apply sigmoid to the gate pre‑activations and fuse the element‑wise
   multiplication (left = left_pre * sigmoid(left_gate_pre), same for right)
   together with the optional mask using a Triton kernel.
4. Compute the pairwise contraction:
       out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   performed as a batched GEMM on half‑precision tensors.
5. Apply output LayerNorm, multiply by the out‑gate and a final linear
   projection back to the original dimension.
6. Return float32 output.

Only the gating+mask multiplication is performed in Triton; all large
matrix multiplications use Torch's highly‑optimized cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half*  (flattened left pre‑activation)
    right_ptr,         # half*  (flattened right pre‑activation)
    left_gate_ptr,     # half*  (flattened left gate pre‑activation)
    right_gate_ptr,    # half*  (flattened right gate pre‑activation)
    mask_ptr,          # half*  (flattened mask, broadcasted over hidden dim)
    total_elements,    # int
    H: tl.constexpr,   # hidden dimension size
    HAS_MASK: tl.constexpr,   # 1 -> mask present, 0 -> no mask
    BLOCK: tl.constexpr,      # block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load inputs
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate_pre = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate_pre = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    # Sigmoid gates
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_pre))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_pre))

    if HAS_MASK:
        # each hidden channel shares the same mask entry
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store back in place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float16).to(device)
    norm_bias = weights["norm.bias"].to(torch.float16).to(device)

    x = input_tensor.to(torch.float16)
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2. Fuse all linear projections into a single GEMM
    # ------------------------------------------------------------------
    M = B * N * N  # total pairwise positions
    x_flat = x.view(M, dim)  # (M, dim)

    # Concatenate the five weight matrices (out_features, in_features)
    # Order: left_proj, left_gate, right_proj, right_gate, out_gate
    w_left = weights["left_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right = weights["right_proj.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]
    w_combined = torch.cat(
        [w_left, w_left_gate, w_right, w_right_gate, w_out_gate],
        dim=0,
    ).to(torch.float16).to(device)  # (5*H, dim)

    combined = F.linear(x_flat, w_combined)  # (M, 5*H)

    # Split
    left_pre = combined[:, 0:H]
    left_gate_pre = combined[:, H:2 * H]
    right_pre = combined[:, 2 * H:3 * H]
    right_gate_pre = combined[:, 3 * H:4 * H]
    out_gate_pre = combined[:, 4 * H:5 * H]

    # Reshape back to [B, N, N, H]
    left_pre = left_pre.view(B, N, N, H)
    left_gate_pre = left_gate_pre.view(B, N, N, H)
    right_pre = right_pre.view(B, N, N, H)
    right_gate_pre = right_gate_pre.view(B, N, N, H)
    out_gate = torch.sigmoid(out_gate_pre.view(B, N, N, H))

    # ------------------------------------------------------------------
    # 3. Fuse gating, optional mask multiplication via Triton
    # ------------------------------------------------------------------
    left_pre_f = left_pre.contiguous().view(-1)
    right_pre_f = right_pre.contiguous().view(-1)
    left_gate_pre_f = left_gate_pre.contiguous().view(-1)
    right_gate_pre_f = right_gate_pre.contiguous().view(-1)

    if nomask:
        mask_f = left_pre_f  # dummy pointer (won't be read)
        has_mask = 0
    else:
        # mask shape [B, N, N] -> flatten
        mask_f = mask.to(torch.float16).to(device).contiguous().view(-1)
        has_mask = 1

    total_elems = left_pre_f.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_elems, BLOCK),)

    _gate_mask_fuse_kernel[grid](
        left_pre_f,
        right_pre_f,
        left_gate_pre_f,
        right_gate_pre_f,
        mask_f,
        total_elems,
        H=H,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )

    # Retrieve gated tensors
    left = left_pre_f.view_as(left_pre)
    right = right_pre_f.view_as(right_pre)

    # ------------------------------------------------------------------
    # 4. Core contraction (batched GEMM)
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    # Move hidden dimension to batch axis for efficient bmm
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B_, H_, N_, _ = left_perm.shape
    left_mat = left_perm.view(B_ * H_, N_, N_)            # (B*H, N, N)
    right_mat = right_perm.view(B_ * H_, N_, N_)          # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    out = out_mat.view(B_, H_, N_, N_).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, out‑gate scaling and final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16).to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16).to(device)
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # element‑wise scaling

    to_out_w = weights["to_out.weight"].to(torch.float16).to(device)
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # Return in original dtype
    return out.to(torch.float32)
