def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Cast weights
    norm_weight = weights["norm.weight"].to(device).to(torch.float16)
    norm_bias = weights["norm.bias"].to(device).to(torch.float16)
    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)

    # 1. Input LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    x = x.to(torch.float16)

    # 2. Projections
    left = F.linear(x, left_proj_weight, bias=None)   # shape [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight, bias=None)

    # 3. Gates
    left_gate = F.linear(x, left_gate_weight, bias=None).sigmoid()
    right_gate = F.linear(x, right_gate_weight, bias=None).sigmoid()
    out_gate = F.linear(x, out_gate_weight, bias=None).sigmoid()

    # 4. Fuse mask and gates
    if not nomask and mask is not None:
        # mask: [B,N,N], expand to hidden_dim dim
        mask_h = mask.to(torch.float16)  # shape [B,N,N]
        # broadcast for kernel: the kernel expects flattened mask of size B*N*N
        # mask must be contiguous
        mask_h = mask_h.contiguous()
        _fuse_gate_and_mask(left, right, left_gate, right_gate, mask_h, hidden_dim)
    else:
        left = left * left_gate
        right = right * right_gate

    # 5. Core contraction using batched GEMM
    B, N, _, _ = left.shape  # left shape [B,N,N,hidden_dim]
    # Permute to bring hidden_dim next to batch
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, hidden_dim, N, N]

    # reshape to (B*hidden_dim, N, N)
    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)

    # matrix multiplication: left_mat @ right_mat^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # 6. Output layer norm
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # 7. Apply out gate
    out = out * out_gate

    # 8. Final projection to dim
    out = F.linear(out, to_out_weight, bias=None)

    # Return float32
    return out.to(torch.float32)

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # Load
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # Each mask entry corresponds to H hidden elements
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

"""
TriMul “outgoing” forward pass (no gradients).

The implementation follows the reference PyTorch module but
* utilizes a small Triton kernel to fuse the mask (if present) with the
  left/right gates in a single pass,
* computes the O(N³) contraction with a batched GEMM (torch.bmm) on
  fp16 tensors – this is equivalent to the einsum
  `out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]`,
* keeps the rest of the ops (LayerNorms, linear layers, sigmoids) in
  PyTorch for simplicity and performance.

All heavy arithmetic is performed in fp16; the final result is returned
as fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask (if present) with left/right gates
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half *
    right_ptr,         # half *
    left_gate_ptr,     # half *
    right_gate_ptr,    # half *
    mask_ptr,          # half *
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask is present, else 0
    H: tl.constexpr,          # hidden dimension (size of innermost channel)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # Load tensors
    left      = tl.load(left_ptr + offs,      mask=mask, other=0.0)
    right     = tl.load(right_ptr + offs,     mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr + offs,mask=mask, other=0.0)

    if HAS_MASK:
        # One mask element corresponds to H consecutive channel entries.
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left  = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left * left_gate
        right = right * right_gate

    # Store back in‑place
    tl.store(left_ptr  + offs, left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _fuse_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of left/right with their gates and an optional mask.
    All tensors must be half‑precision and have the same layout
    [B, N, N, hidden_dim].
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when HAS_MASK=0

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # left and right are mutated in place


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or float) or None
            - weights: dict mapping parameter names to tensors (float32)
            - config: dict containing at least ``dim`` and ``hidden_dim``.
                      May also contain ``nomask`` (bool).

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Cast weights to fp16 on the correct device
    # ------------------------------------------------------------------
    def to_fp16(t):
        return t.to(device, dtype=torch.float16)

    norm_weight = to_fp16(weights["norm.weight"])
    norm_bias   = to_fp16(weights["norm.bias"])

    left_proj_weight  = to_fp16(weights["left_proj.weight"])
    right_proj_weight = to_fp16(weights["right_proj.weight"])

    left_gate_weight  = to_fp16(weights["left_gate.weight"])
    right_gate_weight = to_fp16(weights["right_gate.weight"])
    out_gate_weight   = to_fp16(weights["out_gate.weight"])

    to_out_norm_weight = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_bias   = to_fp16(weights["to_out_norm.bias"])

    to_out_weight = to_fp16(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> fp16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (fp16)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight, bias=None)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight, bias=None)

    # ------------------------------------------------------------------
    # 3. Gate values (fp16)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight, bias=None).sigmoid()
    right_gate = F.linear(x, right_gate_weight, bias=None).sigmoid()
    out_gate = F.linear(x, out_gate_weight, bias=None).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) and gates via Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape [B, N, N] -> fp16, contiguous
        mask_h = mask.to(device, dtype=torch.float16).contiguous()
        _fuse_gate_and_mask(left, right, left_gate, right_gate, mask_h, hidden_dim)
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction using batched GEMM
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    This is equivalent to left_d @ right_d^T for each hidden channel d.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # Bring hidden_dim next to batch for easy reshaping
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, hidden_dim, N, N]

    left_mat = left_perm.view(B * hidden_dim, N, N)    # [B*H, N, N]
    right_mat = right_perm.view(B * hidden_dim, N, N)  # [B*H, N, N]

    # Batched matrix multiplication: left_mat @ right_mat^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N]

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # ------------------------------------------------------------------
    # 7. Apply out‑gate and final projection
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_weight, bias=None)

    # ------------------------------------------------------------------
    # Return in the requested precision (float32)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
