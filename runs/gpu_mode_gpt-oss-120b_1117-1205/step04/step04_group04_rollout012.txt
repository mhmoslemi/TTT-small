def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. layernorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # Flatten
    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)   # (B*N*N, dim)

    # 2. fused left_proj and left_gate linear
    left_weight_concat = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(torch.float16)
    left_res = torch.nn.functional.linear(x_flat, left_weight_concat)  # (B*N*N, 2*hidden_dim)
    left_proj = left_res[:, :hidden_dim].view(B, N, N, hidden_dim)
    left_gate = left_res[:, hidden_dim:].view(B, N, N, hidden_dim).sigmoid()

    # 3. fused right_proj and right_gate
    right_weight_concat = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(torch.float16)
    right_res = torch.nn.functional.linear(x_flat, right_weight_concat)
    right_proj = right_res[:, :hidden_dim].view(B, N, N, hidden_dim)
    right_gate = right_res[:, hidden_dim:].view(B, N, N, hidden_dim).sigmoid()

    # 4. out_gate
    out_gate = torch.nn.functional.linear(x_flat, weights["out_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # 5. Apply mask and gates (optionally using Triton)
    left = left_proj
    right = right_proj

    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        # Use Triton kernel to apply left = left * left_gate * mask, etc.
        # Flatten for kernel
        left_f = left.view(-1)
        right_f = right.view(-1)
        left_gate_f = left_gate.view(-1)
        right_gate_f = right_gate.view(-1)
        mask_f = mask_h.view(-1)  # shape B*N*N

        # Launch kernel
        total = left_f.numel()
        BLOCK = 65536
        grid = (triton.cdiv(total, BLOCK),)
        _mask_gate_fuse_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            total,
            BLOCK=BLOCK,
            HAS_MASK=1,
            H=hidden_dim,
        )
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # 6. Core contraction via batched gemm
    # Permute to (B, hidden_dim, N, N)
    left_t = left.permute(0, 3, 1, 2).contiguous()
    right_t = right.permute(0, 3, 1, 2).contiguous()
    left_mat = left_t.view(B * hidden_dim, N, N)
    right_mat = right_t.view(B * hidden_dim, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))  # (B*H, N, N)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden_dim)

    # 7. Output LayerNorm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    # 8. Gate and final linear
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16))

    # 9. Cast back to float32
    return out.to(torch.float32)

@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,
    right_ptr,
    left_gate_ptr,
    right_gate_ptr,
    mask_ptr,
    total_elems,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    lgate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    if HAS_MASK:
        mask_idx = offs // H
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * lgate * m
        right = right * rgate * m
    else:
        left = left * lgate
        right = right * rgate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)
```

"""
Outgoing TriMul (AlphaFold3) forward pass, optimized.

Algorithm
---------
1.  LayerNorm the input (float32 → float16).
2.  Fuse the left/right linear projections with their gates:
        [proj, gate] = Linear(x) with concatenated weights,
        then split and apply sigmoid to gate.
3.  Compute out_gate (Linear + sigmoid).
4.  Fuse mask (if any) and gate multiplications into a Triton kernel
    (in‑place on left/right).
5.  Core O(N³) contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    Implemented as a batched GEMM by reshaping to (B*H, N, N) and
    using torch.bmm (cuBLAS) on half‑precision tensors.
6.  Output LayerNorm, element‑wise gating with out_gate,
    and final Linear back to the original dimension.
7.  Return float32 output.

The Triton kernel only handles the mask‑gate fusion; all heavy linear
operations use cuBLAS for maximal performance.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: in‑place mask & gate fusion
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # *half
    right_ptr,         # *half
    left_gate_ptr,     # *half
    right_gate_ptr,    # *half
    mask_ptr,          # *half (broadcasted over hidden dim)
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,   # hidden dimension
):
    """
    For each element of left/right (flattened):
        left  = left  * left_gate  * (mask if present)
        right = right * right_gate * (mask if present)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # Load values
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    lgate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # Each H consecutive elements share the same mask entry.
        mask_idx = offs // H
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * lgate * m
        right = right * rgate * m
    else:
        left = left * lgate
        right = right * rgate

    # Store results back in place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (float/bool) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict containing "dim" and "hidden_dim" (int),
                      optionally "nomask" (bool)

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # unpack
    inp, mask, weights, config = data
    device = inp.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 → float16)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)          # [B, N, N, dim]

    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)         # (B*N*N, dim)

    # -----------------------------------------------------------------
    # 2. Fuse left_proj + left_gate linear ops
    # -----------------------------------------------------------------
    left_weight_concat = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)               # (2*H, dim)
    left_res = F.linear(x_flat, left_weight_concat)   # (B*N*N, 2*H)
    left_proj = left_res[:, :hidden_dim].view(B, N, N, hidden_dim)
    left_gate = left_res[:, hidden_dim:].view(B, N, N, hidden_dim).sigmoid()

    # -----------------------------------------------------------------
    # 3. Fuse right_proj + right_gate linear ops
    # -----------------------------------------------------------------
    right_weight_concat = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)               # (2*H, dim)
    right_res = F.linear(x_flat, right_weight_concat)
    right_proj = right_res[:, :hidden_dim].view(B, N, N, hidden_dim)
    right_gate = right_res[:, hidden_dim:].view(B, N, N, hidden_dim).sigmoid()

    # -----------------------------------------------------------------
    # 4. out_gate (Linear + sigmoid)
    # -----------------------------------------------------------------
    out_gate = F.linear(
        x_flat, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid().view(B, N, N, hidden_dim)   # (B, N, N, H)

    # -----------------------------------------------------------------
    # 5. Apply mask and gates (mask fusion via Triton)
    # -----------------------------------------------------------------
    left = left_proj
    right = right_proj

    if not nomask and mask is not None:
        # mask: [B,N,N] → broadcast over hidden dim
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)

        # Flatten tensors for the kernel
        left_f = left.view(-1)
        right_f = right.view(-1)
        left_gate_f = left_gate.view(-1)
        right_gate_f = right_gate.view(-1)
        mask_f = mask_h.view(-1)                       # length B*N*N

        total = left_f.numel()
        BLOCK = 65536
        grid = (triton.cdiv(total, BLOCK),)

        _mask_gate_fuse_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            total,
            BLOCK=BLOCK,
            HAS_MASK=1,
            H=hidden_dim,
        )
        # Reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask: simple element‑wise gating
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 6. Core contraction: batched GEMM (A * Bᵀ) over the sequence dimension
    # -----------------------------------------------------------------
    # reshape to (B, H, N, N) for batched matmul
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_mat = left_t.view(B * hidden_dim, N, N)     # (B*H, N, N)
    right_mat = right_t.view(B * hidden_dim, N, N)   # (B*H, N, N)

    # out_mat[b*h] = left_mat[b*h] @ right_mat[b*h].T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B, N, N, H)

    # -----------------------------------------------------------------
    # 7. Output LayerNorm, gating, final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # -----------------------------------------------------------------
    # 8. Cast back to float32
    # -----------------------------------------------------------------
    return out.to(torch.float32)
