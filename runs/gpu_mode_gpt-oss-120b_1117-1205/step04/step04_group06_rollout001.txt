"""
Optimized outgoing TriMul forward pass.

Algorithm:
1. Layer‑norm the input tensor (float32 → float16).
2. Linear projections to hidden_dim for left/right and gates (all float16).
3. Fuse element‑wise multiplication of left/right with their gates and
   optional mask using a tiny Triton kernel (in‑place).
4. Compute the O(N³) contraction via a batched GEMM:
       out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   which is equivalent to left @ rightᵀ for each hidden channel.
5. Apply output LayerNorm, out‑gate scaling and final linear projection.
6. Return result in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_fuse_kernel(
    left_ptr,               # float16*
    right_ptr,              # float16*
    left_gate_ptr,          # float16*
    right_gate_ptr,         # float16*
    mask_ptr,               # float16* (flattened mask) – used only when HAS_MASK=1
    total_elements,         # int32
    hidden_dim,             # int32
    HAS_MASK: tl.constexpr, # 1 if mask is present, else 0
    BLOCK: tl.constexpr,    # compile‑time block size
):
    """In‑place: left = left * left_gate * mask?, right = right * right_gate * mask?."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # each hidden entry shares the same mask entry -> map linear index to mask index
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool / numeric) – may be ignored
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gates (half) with sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) with gates via Triton kernel
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    total = left.numel()                     # B * N * N * hidden_dim
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    # Flatten tensors for the kernel
    left_flat = left.contiguous().view(-1)
    right_flat = right.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)

    if not nomask:
        mask_flat = mask.to(torch.float16).contiguous().view(-1)  # B*N*N
        HAS_MASK = 1
    else:
        # dummy mask – never read because HAS_MASK=0
        mask_flat = left_flat
        HAS_MASK = 0

    _gate_fuse_kernel[grid](
        left_flat,
        right_flat,
        left_gate_flat,
        right_gate_flat,
        mask_flat,
        total,
        hidden_dim,
        HAS_MASK=HAS_MASK,
        BLOCK=BLOCK,
    )

    # Restore shapes
    left = left_flat.view(B, N, N, hidden_dim)
    right = right_flat.view(B, N, N, hidden_dim)

    # Free intermediate gate tensors
    del left_gate_flat, right_gate_flat, left_gate, right_gate, x

    # ------------------------------------------------------------------
    # 5. Core contraction using batched GEMM:
    #    out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    #    -> left @ rightᵀ per (b, h)
    # ------------------------------------------------------------------
    # Move hidden dimension to the second axis for efficient matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # Batched matrix multiplication (left @ rightᵀ)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise gating (half)

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in float32 as required by the interface
    return out.to(torch.float32)
