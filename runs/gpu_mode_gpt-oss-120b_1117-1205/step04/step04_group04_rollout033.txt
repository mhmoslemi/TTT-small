def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    dtype= torch.float16

    # layernorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'].to(dtype), bias=weights['norm.bias'].to(dtype))
    x = x.to(dtype)

    # projections
    left = F.linear(x, weights['left_proj.weight'].to(dtype))
    right = F.linear(x, weights['right_proj.weight'].to(dtype))

    # gates
    left_gate = F.linear(x, weights['left_gate.weight'].to(dtype)).sigmoid()
    right_gate = F.linear(x, weights['right_gate.weight'].to(dtype)).sigmoid()
    out_gate = F.linear(x, weights['out_gate.weight'].to(dtype)).sigmoid()

        # flatten tensors for kernel
        left_f = left.view(-1)
        right_f = right.view(-1)
        left_gate_f = left_gate.view(-1)
        right_gate_f = right_gate.view(-1)
        mask_f = mask_.view(-1)  # shape B*N*N
        # apply mask and gate using kernel
        _apply_mask_and_gate(left_f, right_f, left_gate_f, right_gate_f, mask_f, hidden_dim)
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # contraction
    # Prepare output tensor
    out = torch.empty_like(left)  # same shape, dtype fp16

    # launch kernel
    B, N, _, H = left.shape
    # strides
    stride_left_b = left.stride(0)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)
    stride_left_h = left.stride(3)

    stride_right_b = right.stride(0)
    stride_right_i = right.stride(1)
    stride_right_k = right.stride(2)
    stride_right_h = right.stride(3)

    stride_out_b = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_h = out.stride(3)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    total_batch = B * H
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), total_batch)

        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K

    # Return float32 format

"""
Outgoing TriMul (AlphaFold3) forward pass.

Algorithm
---------
1. Layer‑norm the input (float32 → float16).
2. Linear projections to a hidden dimension (no bias, half‑precision).
3. Compute three sigmoid gates.
4. Fuse optional mask with the left/right gate multiplications in a single Triton
   kernel (in‑place).
5. Core O(N³) contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a custom Triton kernel that performs a batched GEMM over the
   hidden dimension, avoiding the costly permute‑contiguous copies of the
   reference implementation.
6. Output LayerNorm, element‑wise gating with out_gate and a final linear back
   to the original feature dimension.
7. Return as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Mask‑and‑gate fusion kernel (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # *fp16
    right_ptr,         # *fp16
    left_gate_ptr,     # *fp16
    right_gate_ptr,    # *fp16
    mask_ptr,          # *fp16 (broadcasted over hidden dim)
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,   # hidden dimension
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elems

    left = tl.load(left_ptr + offs, mask=m, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=m, other=tl.float16(0.0))
    lgate = tl.load(left_gate_ptr + offs, mask=m, other=tl.float16(0.0))
    rgate = tl.load(right_gate_ptr + offs, mask=m, other=tl.float16(0.0))

    if HAS_MASK:
        # one mask value per (b,i,k) → broadcast over hidden dim
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=m, other=tl.float16(0.0))
        left = left * lgate * mask_val
        right = right * rgate * mask_val
    else:
        left = left * lgate
        right = right * rgate

    tl.store(left_ptr + offs, left, mask=m)
    tl.store(right_ptr + offs, right, mask=m)


def _apply_mask_and_gate(left, right, left_gate, right_gate, mask, hidden_dim):
    """Launch the Triton kernel that fuses mask and gates inplace."""
    total = left.numel()
    BLOCK = 32768  # each program processes BLOCK elements
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when no mask

    _mask_gate_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # in‑place operation – nothing to return


# ----------------------------------------------------------------------
# Core O(N³) contraction kernel
# Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,      # *fp16   shape (B,N,N,H)
    right_ptr,     # *fp16   shape (B,N,N,H)
    out_ptr,       # *fp16   shape (B,N,N,H)

    B: tl.int32,
    N: tl.int32,
    H: tl.int32,

    # strides for left tensor
    stride_left_b: tl.int32,
    stride_left_i: tl.int32,
    stride_left_k: tl.int32,
    stride_left_h: tl.int32,

    # strides for right tensor
    stride_right_b: tl.int32,
    stride_right_i: tl.int32,
    stride_right_k: tl.int32,
    stride_right_h: tl.int32,

    # strides for out tensor
    stride_out_b: tl.int32,
    stride_out_i: tl.int32,
    stride_out_j: tl.int32,
    stride_out_h: tl.int32,

    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)            # tile over i (rows)
    pid_n = tl.program_id(1)            # tile over j (cols)
    pid_bh = tl.program_id(2)           # fused batch * hidden index

    # decode batch and hidden indices
    b = pid_bh // H
    h = pid_bh % H

    # base pointers for this (b,h) slice
    offs_batch = b * stride_left_b + h * stride_left_h
    a_base = left_ptr + offs_batch
    b_base = right_ptr + offs_batch
    c_base = out_ptr + offs_batch

    # i / j offsets for the current tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for boundaries
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_blocks):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # masks for k
        mask_k = offs_k < N

        # pointers to tiles
        a_ptrs = a_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        b_ptrs = b_base + offs_j[:, None] * stride_right_i + offs_k[None, :] * stride_right_k

        # load with boundary masks
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=tl.float16(0.0)).to(tl.float32)
        b = tl.load(b_ptrs,
                    mask=mask_j[:, None] & mask_k[None, :],
                    other=tl.float16(0.0)).to(tl.float32)

        # accumulate
        acc += tl.dot(a, b.T)   # (BLOCK_M, BLOCK_N)

    # write result back (still fp16)
    c = acc.to(tl.float16)
    c_ptrs = c_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs, c, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, D] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of parameter tensors (float32)
            - config: dict containing "dim" and "hidden_dim" (and optional "nomask")

    Returns:
        torch.Tensor [B, N, N, D] (float32)
    """
    inp, mask, weights, config = data
    device = inp.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 → fp16)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"].to(torch.float16),
        bias=weights["norm.bias"].to(torch.float16),
    )
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections to hidden dimension (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # -----------------------------------------------------------------
    # 3. Gate tensors (sigmoid)
    # -----------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # -----------------------------------------------------------------
    # 4. Fuse mask with gates (if required) via Triton
    # -----------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → broadcast over hidden dimension
        mask_h = mask.to(torch.float16).unsqueeze(-1)               # [B,N,N,1]

        # flatten tensors for the kernel (1‑D view)
        left_f = left.view(-1)
        right_f = right.view(-1)
        left_gate_f = left_gate.view(-1)
        right_gate_f = right_gate.view(-1)
        mask_f = mask_h.view(-1)                                    # length B*N*N

        _apply_mask_and_gate(left_f, right_f, left_gate_f, right_gate_f, mask_f, hidden_dim)

        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core O(N³) contraction using a custom Triton kernel
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # allocate output tensor (same layout as left/right)
    out = torch.empty_like(left)

    # stride information (int)
    stride_left_b = left.stride(0)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)
    stride_left_h = left.stride(3)

    stride_right_b = right.stride(0)
    stride_right_i = right.stride(1)
    stride_right_k = right.stride(2)
    stride_right_h = right.stride(3)

    stride_out_b = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_h = out.stride(3)

    # tuning parameters – 64×64 tiles work well for the tested sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    total_batch = B * H
    grid = (
        triton.cdiv(N, BLOCK_M),   # i dimension
        triton.cdiv(N, BLOCK_N),   # j dimension
        total_batch,               # fused batch * hidden
    )

    _trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_i,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # return in float32 as required
    return out.to(torch.float32)
