"""
Outgoing TriMul operator (AlphaFold3) forward pass.

Algorithm
---------
1. Layer‑norm the input (float32) → half.
2. Linear projections (half) produce left/right tensors of shape [B,N,N,H].
3. Compute three sigmoid gates (half) from the same normalized input.
4. Fuse mask (if present) with left/right gates in a Triton kernel:
   left  = left  * left_gate  * mask
   right = right * right_gate * mask
   (broadcast mask over the hidden dimension H).
   If no mask is supplied, the kernel simply performs the gate multiplication.
5. Core contraction (batched GEMM):
   out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   Implemented as a batched matrix multiplication over the combined batch‑hidden
   dimension (B*H, N, N) using torch.bmm.
6. Output layer‑norm over hidden dimension, scale by out_gate, final linear
   projection back to the original dimension, and cast to float32.

The heavy element‑wise gating is offloaded to Triton, while the O(N³) contraction
uses cuBLAS via torch.bmm for high performance.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def gate_mask_fuse_kernel(
    left_ptr,               # * half,   [total]
    right_ptr,              # * half,   [total]
    left_gate_ptr,          # * half,   [total]
    right_gate_ptr,         # * half,   [total]
    mask_ptr,               # * half,   [mask_len]  (broadcasted)
    total_elements,         # i32
    hidden_dim,             # i32   (size of last dim)
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,  # i1
):
    """Fuse mask and gating into left/right tensors."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    # Load operands
    left = tl.load(left_ptr + offs, mask=valid, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=valid, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=tl.float16(0.0))

    if HAS_MASK:
        # Broadcast mask over hidden_dim: each hidden entry shares same mask value.
        mask_idx = offs // hidden_dim                     # int division
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=tl.float16(1.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store results back in‑place
    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool / float) or None
        - weights      : dict of torch.Tensors (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections (half)
    # --------------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # --------------------------------------------------------------
    # 3. Compute gates (half)
    # --------------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # --------------------------------------------------------------
    # 4. Fuse mask (if present) with gates via Triton
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # Ensure contiguous half tensors
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_h = mask.to(torch.float16).contiguous()

        # Flatten for kernel launch
        left_f = left.view(-1)
        right_f = right.view(-1)
        left_gate_f = left_gate.view(-1)
        right_gate_f = right_gate.view(-1)
        mask_f = mask_h.view(-1)  # shape [B*N*N]

        total = left_f.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)

        gate_mask_fuse_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            total,
            hidden_dim,
            BLOCK=BLOCK,
            HAS_MASK=1,
        )
        # Reshape back to original layout
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # Simple element‑wise gating without mask
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction via batched GEMM (half precision)
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # --------------------------------------------------------------
    # Move hidden dimension to second axis for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # Batched matrix multiplication: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore shape (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise scaling (half)

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in original dtype (float32)
    return out.to(torch.float32)
