@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements,
    HIDDEN_DIM: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    if HAS_MASK:
        mask_idx = offs // HIDDEN_DIM
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

    # Output layer norm

        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    # Apply output gate
    out = out * out_gate

    # Final linear projection
    out = F.linear(out, to_out_weight, bias=None)

    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Project to a hidden dimension with two linear layers (left/right).
3. Compute three sigmoid gates (left, right, out).
4. Fuse mask (if present) and element‑wise gating for `left` and `right`
   via a small Triton kernel (in‑place).
5. Contract over the sequence dimension:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched GEMM (left @ rightᵀ) performed in fp16.
6. Apply a second layer‑norm, multiply by the out‑gate, and project back
   to the original dimension.
7. Return the result in float32.

Only the mask‑gate fusion is implemented in Triton; the heavy O(N³) contraction
uses cuBLAS via torch.bmm for maximum throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # * half
    right_ptr,              # * half
    left_gate_ptr,          # * half
    right_gate_ptr,         # * half
    mask_ptr,               # * half (broadcasted)
    total_elements: tl.int32,
    HIDDEN_DIM: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # mask index repeats for each hidden dimension
        mask_idx = offs // HIDDEN_DIM
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_mask_fuse(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place Triton fusion of mask (if any) and gating for `left` and `right`.
    All tensors must be contiguous half‑precision.
    """
    total = left.numel()
    BLOCK = 32768  # must be multiple of 32
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when no mask

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        HIDDEN_DIM=hidden_dim,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict containing at least "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # ------------------------------------------------------------------
    # Cast weights to half precision on the correct device
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device).to(torch.float16)
    norm_bias = weights["norm.bias"].to(device).to(torch.float16)
    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> half)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension
    # ------------------------------------------------------------------
    left = F.linear(x_h, left_proj_weight, bias=None)      # [B,N,N,H]
    right = F.linear(x_h, right_proj_weight, bias=None)    # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3. Compute gates (half precision, sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, left_gate_weight, bias=None).sigmoid()
    right_gate = F.linear(x_h, right_gate_weight, bias=None).sigmoid()
    out_gate = F.linear(x_h, out_gate_weight, bias=None).sigmoid()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) with gates via Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).contiguous()
        # Flatten tensors for 1‑D kernel view
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask_h.view(-1)

        _apply_gate_mask_fuse(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            hidden_dim,
        )
        # Reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # Gates no longer needed
    del left_gate, right_gate, x_h, x_norm

    # ------------------------------------------------------------------
    # 5. Core contraction: einsum('... i k d, ... j k d -> ... i j d')
    #    Realized as batched GEMM (left @ rightᵀ) in half precision.
    # ------------------------------------------------------------------
    # Bring hidden dimension to the second axis for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)          # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)        # (B*H, N, N)

    # out_mat[b*h,i,j] = Σ_k left[i,k] * right[j,k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight, bias=None)    # (B, N, N, dim)

    # Return in float32 as required
    return out.to(torch.float32)
