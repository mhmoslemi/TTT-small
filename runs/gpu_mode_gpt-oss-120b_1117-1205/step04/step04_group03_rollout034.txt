@triton.jit
def _gate_mask_sigmoid_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,   # hidden_dim
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total
    # load
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate_raw = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate_raw = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    # sigmoid
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_raw))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))
    if HAS_MASK:
        # mask index per hidden channel group: since mask shape is (B,N,N), each element of hidden dimension shares same mask.
        # We need to map offs to mask index = offs // H
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left_gate = left_gate * mask_val
        right_gate = right_gate * mask_val
    # Multiply
    left = left * left_gate
    right = right * right_gate
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def custom_kernel(data):
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Convert weights to half (if not already)
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias = weights["norm.bias"].to(torch.float16)
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    to_out_weight = weights["to_out.weight"].to(torch.float16)

    # 1. LayerNorm & cast to half
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight.float(),  # layernorm in float32
        bias=norm_bias.float(),
    )
    x_h = x.to(torch.float16)

    # 2. Combined linear projections and gates
    # Combine proj weights
    proj_weight_combined = torch.cat([left_proj_weight, right_proj_weight], dim=0)  # shape (2*hidden_dim, dim)
    # Linear
    proj_out = F.linear(x_h, proj_weight_combined, bias=None)  # shape (B,N,N,2*hidden_dim)
    # Split
    left_proj_raw = proj_out[..., :hidden_dim]
    right_proj_raw = proj_out[..., hidden_dim:]

    # Combined gates weights
    gate_weight_combined = torch.cat([left_gate_weight, right_gate_weight, out_gate_weight], dim=0)  # shape (3*hidden_dim, dim)
    gate_out = F.linear(x_h, gate_weight_combined, bias=None)  # shape (B,N,N,3*hidden_dim)
    left_gate_raw = gate_out[..., :hidden_dim]
    right_gate_raw = gate_out[..., hidden_dim: 2*hidden_dim]
    out_gate_raw = gate_out[..., 2*hidden_dim:]

    # 3. Apply mask and sigmoid gates to left and right via Triton kernel
    total = left_proj_raw.numel()  # B*N*N*hidden_dim

    # Ensure contiguity
    left_proj = left_proj_raw.contiguous()
    right_proj = right_proj_raw.contiguous()
    left_gate = left_gate_raw.contiguous()
    right_gate = right_gate_raw.contiguous()

    # Prepare mask pointer
    if not nomask:
        # mask shape (B, N, N)
        mask_h = mask.to(torch.float16).contiguous()
        mask_flat = mask_h.view(-1)  # B*N*N
    else:
        # dummy mask pointer can be left_proj
        mask_flat = left_proj  # will not be used when HAS_MASK=0
    # Launch kernel
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    HAS_MASK = 0 if nomask else 1
    _gate_mask_sigmoid_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask_flat,
        total,
        BLOCK=BLOCK,
        HAS_MASK=HAS_MASK,
        H=hidden_dim,
    )
    # after kernel, left_proj and right_proj now contain gated (and masked) values

    # 4. Compute out_gate sigmoid (no mask)
    out_gate = torch.sigmoid(out_gate_raw)  # half dtype? out_gate_raw is half, resulting half

    # 5. Core contraction: batched GEMM per hidden_dim
    # left_proj shape (B,N,N,H)
    # right_proj shape (B,N,N,H)
    # reshape to (B*H, N, N)
    left_perm = left_proj.permute(0, 3, 1, 2).contiguous()
    right_perm = right_proj.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H,N,N)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # 6. Output layernorm (hidden_dim)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # 7. Multiply by out_gate
    out = out * out_gate

    # 8. Final linear projection to original dim
    out = F.linear(out, to_out_weight, bias=None)

    # Cast to float32
    return out.to(torch.float32)

    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,

"""
TriMul outgoing forward pass (AlphaFold3 style).

Algorithm
---------
1. Layer‑norm the 4‑D input (B,N,N,C) in float32, then cast to float16.
2. Fuse the two linear projections (left/right) into one GEMM, and
   fuse the three gate linear layers (left_gate, right_gate, out_gate)
   into a second GEMM.
3. A tiny Triton kernel processes the projected tensors element‑wise:
   * apply the optional binary mask (broadcast over the hidden dimension),
   * compute sigmoid(gate) = 1/(1+exp(-x)),
   * multiply the mask, the gate and the projection → final left/right tensors.
   The kernel is always launched (mask‑branch compiled out when not needed).
4. Batched matrix multiplication implements the core contraction:
   out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h].
   Implemented as a large torch.bmm over (B·H) matrices of shape (N,N).
5. Apply a second LayerNorm over the hidden dimension, scale by the
   out‑gate (sigmoid of the third gate), and a final linear projection
   back to the original channel dimension.
6. Cast the result back to float32.

All heavy linear work uses FP16 GEMMs; only the element‑wise gate/mask
fusion uses a custom Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_sigmoid_kernel(
    left_ptr,         # fp16*  (left projection, will be overwritten)
    right_ptr,        # fp16*  (right projection, will be overwritten)
    left_gate_ptr,    # fp16*  (raw left‑gate pre‑sigmoid)
    right_gate_ptr,   # fp16*  (raw right‑gate pre‑sigmoid)
    mask_ptr,         # fp16*  (flattened mask, length B·N·N)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 → mask is applied, 0 → no mask
    H: tl.constexpr,          # hidden dimension (size of each gate block)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    within = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=within, other=0.0)
    right = tl.load(right_ptr + offs, mask=within, other=0.0)
    left_gate_raw = tl.load(left_gate_ptr + offs, mask=within, other=0.0)
    right_gate_raw = tl.load(right_gate_ptr + offs, mask=within, other=0.0)

    # Apply optional mask (same mask entry for all hidden channels)
    if HAS_MASK:
        mask_idx = offs // H                # map each hidden element to its mask entry
        mask_val = tl.load(mask_ptr + mask_idx, mask=within, other=0.0)
        left = left * mask_val
        right = right * mask_val

    # Sigmoid on gate values
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_raw))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))

    # Gating
    left = left * left_gate
    right = right * right_gate

    # Write results back in‑place
    tl.store(left_ptr + offs, left, mask=within)
    tl.store(right_ptr + offs, right, mask=within)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : torch.Tensor[bs, seq_len, seq_len, dim]      (float32)
        mask_tensor  : torch.Tensor[bs, seq_len, seq_len]          (bool or numeric)
        weights_dict : dict of model parameters (float32)
        config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor[bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 0. Cast all weights to FP16 on the correct device
    # ------------------------------------------------------------------
    def to_fp16(t):
        return t.to(device=device, dtype=torch.float16)

    norm_w = to_fp16(weights["norm.weight"])
    norm_b = to_fp16(weights["norm.bias"])

    left_proj_w = to_fp16(weights["left_proj.weight"])
    right_proj_w = to_fp16(weights["right_proj.weight"])

    left_gate_w = to_fp16(weights["left_gate.weight"])
    right_gate_w = to_fp16(weights["right_gate.weight"])
    out_gate_w = to_fp16(weights["out_gate.weight"])

    to_out_norm_w = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_b = to_fp16(weights["to_out_norm.bias"])
    to_out_w = to_fp16(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → FP16
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w.float(),
        bias=norm_b.float(),
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Combined projection (left & right) and combined gates (3×hidden)
    # ------------------------------------------------------------------
    # 2a. Projection
    proj_w = torch.cat([left_proj_w, right_proj_w], dim=0)            # (2*H, dim)
    proj_out = F.linear(x_h, proj_w, bias=None)                      # (B,N,N,2*H)
    left_proj = proj_out[..., :hidden_dim].contiguous()
    right_proj = proj_out[..., hidden_dim:].contiguous()

    # 2b. Gates
    gate_w = torch.cat([left_gate_w, right_gate_w, out_gate_w], dim=0)  # (3*H, dim)
    gate_out = F.linear(x_h, gate_w, bias=None)                        # (B,N,N,3*H)
    left_gate_raw = gate_out[..., :hidden_dim].contiguous()
    right_gate_raw = gate_out[..., hidden_dim:2 * hidden_dim].contiguous()
    out_gate_raw = gate_out[..., 2 * hidden_dim:].contiguous()

    # ------------------------------------------------------------------
    # 3. Triton kernel: apply mask (if any), sigmoid gates, and fuse them
    # ------------------------------------------------------------------
    total = left_proj.numel()                     # B * N * N * H
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    if not nomask:
        mask_h = mask.to(torch.float16, device=device).contiguous()
        mask_flat = mask_h.view(-1)               # (B*N*N,)
        HAS_MASK = 1
    else:
        mask_flat = left_proj                      # dummy pointer, not read when HAS_MASK=0
        HAS_MASK = 0

    _gate_mask_sigmoid_kernel[grid](
        left_proj,
        right_proj,
        left_gate_raw,
        right_gate_raw,
        mask_flat,
        total,
        BLOCK=BLOCK,
        HAS_MASK=HAS_MASK,
        H=hidden_dim,
    )
    # After the kernel, `left_proj` and `right_proj` hold the masked‑gated tensors.

    # ------------------------------------------------------------------
    # 4. out_gate (sigmoid, no mask)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_raw)       # (B,N,N,H), fp16

    # ------------------------------------------------------------------
    # 5. Core contraction: batched GEMM over the hidden dimension
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) for batched matmul
    left_perm = left_proj.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_proj.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)       # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)     # (B*H, N, N)

    # left @ rightᵀ  (each slice corresponds to a hidden channel)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gate scaling and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w, bias=None)

    # ------------------------------------------------------------------
    # 7. Return float32 tensor
    # ------------------------------------------------------------------
    return out.to(torch.float32)
