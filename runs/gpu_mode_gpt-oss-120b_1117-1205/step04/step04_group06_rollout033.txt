"""
Optimized TriMul (outgoing) forward pass.

- Input is normalized with half-precision LayerNorm.
- All five linear projections (left, right, left_gate, right_gate, out_gate) are
  fused into a single matmul to reduce memory traffic.
- Mask and gate fusion for left/right is performed by a tiny Triton kernel.
- The O(N³) contraction is realized as a batched GEMM (B*H matrix‑multiply)
  using torch.bmm which maps to cuBLAS’s highly tuned kernel.
- Output LayerNorm, out‑gate scaling and final linear projection are executed
  in half‑precision; the result is returned as float32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr,
    mask_ptr, total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    # Load data
    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # mask index corresponds to flattened (b, i, k)
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)

def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    Fuse left/right with gates and optional mask in-place using Triton.
    All arguments must be contiguous float16 tensors.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # No return; operation is in-place.

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) – may be None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm on input (half precision)
    # ------------------------------------------------------------------
    # Cast to half early to stay in fp16 throughout
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(torch.float16),
        bias=weights["norm.bias"].to(torch.float16),
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Fused linear projections for left, right and gates
    # ------------------------------------------------------------------
    # Concatenate projection weights: each (hidden_dim, dim)
    proj_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # shape (5*hidden_dim, dim)

    # Perform a single matmul: (B*N*N, dim) x (5*hidden_dim, dim)^T
    # torch.nn.functional.linear expects weight of shape (out_features, in_features)
    all_proj = F.linear(x, proj_weight, bias=None)  # (B, N, N, 5*hidden_dim)
    # ------------------------------------------------------------------
    # 3. Slice projections and apply sigmoid to gates
    # ------------------------------------------------------------------
    H = hidden_dim
    left = all_proj[..., 0:H]
    right = all_proj[..., H:2*H]
    left_gate = all_proj[..., 2*H:3*H]
    right_gate = all_proj[..., 3*H:4*H]
    out_gate = all_proj[..., 4*H:5*H]

    # Sigmoid gates (half precision)
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    # Ensure tensors are contiguous before Triton kernel (if needed)
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) with gates using Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] -> half flattened
        mask_h = mask.to(torch.float16).contiguous().view(-1)  # (B*N*N)
        _apply_gate_and_mask(left, right, left_gate, right_gate, mask_h, hidden_dim)
        # No need to keep gate tensors afterwards
        del left_gate, right_gate
    else:
        # Simple element‑wise multiplication when no mask
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³) contraction via batched GEMM (left @ right^T)
    # ------------------------------------------------------------------
    # Permute to (B, H, N, N) for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # Batched matrix multiplication with transpose on second operand
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore layout (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    # Apply out gate (half precision)
    out = out * out_gate

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in float32 as required
    return out.to(torch.float32)

import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,         # half*
    right_ptr,        # half*
    left_gate_ptr,    # half*
    right_gate_ptr,   # half*
    mask_ptr,         # half* (broadcasted, may be dummy)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,
):
    """In‑place: left = left * left_gate * mask; right = right * right_gate * mask."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # mask is shared across the hidden dimension
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    Fuse left/right with their gates and optional mask in‑place via Triton.
    All arguments must be contiguous half‑precision tensors.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when HAS_MASK==0

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # operation is in‑place; nothing is returned


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) – may be None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm in half precision
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(torch.float16),
        bias=weights["norm.bias"].to(torch.float16),
    )
    x = x.to(torch.float16)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2. Fused linear projections for left, right and all gates
    # ------------------------------------------------------------------
    # Concatenate projection weights: each is (hidden_dim, dim)
    proj_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # (5*hidden_dim, dim)

    # Single matmul creates all required projections at once
    all_proj = F.linear(x, proj_weight, bias=None)   # (B, N, N, 5*hidden_dim)

    # ------------------------------------------------------------------
    # 3. Slice projections and apply sigmoid to gating tensors
    # ------------------------------------------------------------------
    H = hidden_dim
    left = all_proj[..., 0:H]
    right = all_proj[..., H:2 * H]
    left_gate = all_proj[..., 2 * H:3 * H]
    right_gate = all_proj[..., 3 * H:4 * H]
    out_gate = all_proj[..., 4 * H:5 * H]

    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    # Ensure contiguous layout before Triton kernel
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) with gates using a Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask.to(torch.float16).contiguous().view(-1)  # (B*N*N)
        _apply_gate_and_mask(left, right, left_gate, right_gate, mask_h, hidden_dim)
        # Gates are no longer needed
        del left_gate, right_gate
    else:
        # Simple element‑wise multiplication when there is no mask
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # Batched matrix multiplication with transpose on second operand
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore original layout (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise gating (half precision)

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in float32 as required by the interface
    return out.to(torch.float32)
