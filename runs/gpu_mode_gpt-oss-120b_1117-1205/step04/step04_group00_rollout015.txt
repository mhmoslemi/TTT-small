"""
TriMul (outgoing) forward pass with a fused Triton kernel for the
N‑dimensional contraction.  The heavy N³ pairwise operation is expressed as
a batched GEMM: for each head d we compute
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to left[b,:, :, d] @ right[b,:, :, d].T.
The kernel performs a batched half‑precision GEMM (accumulating in fp32)
and the rest of the pipeline (LayerNorms, linear projections, gates) is
handled in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,          # A: (M, K)
    stride_bk, stride_bn,          # B: (K, N) – note swapped strides for transposed access
    stride_cm, stride_cn,          # C: (M, N)
    stride_ab, stride_bb, stride_cb,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: C = A @ B   with A: (batch, M, K), B: (batch, K, N)
    All tensors are in row‑major order. B is accessed with swapped strides
    to implement the required transpose (right matrix is used as Bᵀ).
    """
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block column
    pid_b = tl.program_id(2)  # batch index

    # Offsets for the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Starting pointers for this batch
    a_batch = a_ptr + pid_b * stride_ab
    b_batch = b_ptr + pid_b * stride_bb
    c_batch = c_ptr + pid_b * stride_cb

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak),
            mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
            other=0.0,
        )
        # Load B tile (K x N) – note swapped strides for transposition
        b = tl.load(
            b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
        )
        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_N)

    # Write back result in FP16
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),
    )


def _batched_gemm(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Wrapper for the Triton batched GEMM.
    A: (batch, M, K)  FP16
    B: (batch, K, N)  FP16 (will be accessed as transposed)
    Returns C: (batch, M, N) FP16
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # Block sizes (tuned for typical N up to 1024)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch,
    )

    _batched_gemm_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        A.stride(1),          # stride_am
        A.stride(2),          # stride_ak
        B.stride(2),          # stride_bk  (col stride – swapped for transpose)
        B.stride(1),          # stride_bn  (row stride)
        C.stride(1),          # stride_cm
        C.stride(2),          # stride_cn
        A.stride(0),          # batch stride for A
        B.stride(0),          # batch stride for B
        C.stride(0),          # batch stride for C
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half)
    # -----------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # -----------------------------------------------------------------
    # 3. Gate computations (half)
    # -----------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # -----------------------------------------------------------------
    # 4. Apply mask & gates
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape: [B, N, N] → broadcast over hidden dim
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f * left_gate
        right = right * mask_f * right_gate
    else:
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction via batched GEMM (Triton)
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # -----------------------------------------------------------------
    # Rearrange to (B, H, N, N) where H = hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape

    # Collapse batch and head dimensions -> (B*H, N, N)
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Perform batched GEMM: left_mat @ right_mat^T using our Triton kernel
    # right_mat is accessed with swapped strides inside the kernel to achieve the transpose.
    out_mat = _batched_gemm(left_mat, right_mat)  # (B*H, N, N), fp16

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise scaling (half)

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in original dtype (float32)
    return out.to(torch.float32)
