def mask_gate_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements: tl.int32,
    hidden_dim: tl.constexpr, # used for dividing offs
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=0.)
    right = tl.load(right_ptr + offs, mask=mask, other=0.)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.)

    if HAS_MASK:
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

"""
TriMul (outgoing) forward pass optimized with Triton.

Algorithm
---------
1. LayerNorm the input (half precision).
2. Linear projections to hidden_dim (left/right) and gate values
   (left_gate, right_gate, out_gate) – all half precision.
3. Fuse mask (if present) and gate multiplication into a single Triton
   kernel that writes the results back in‑place.
   * left = left * left_gate * mask
   * right = right * right_gate * mask
4. Core contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
   This is a batch of H independent N×N matrix multiplications.
   Implemented via a single `torch.bmm` on reshaped tensors (highly
   optimized cuBLAS GEMM).
5. LayerNorm over hidden_dim, element‑wise out‑gate, final linear
   projection back to dim.
6. Return float32 tensor.

The kernel only handles the mask‑gate fusion; the heavy N³ GEMM is
performed by cuBLAS through `torch.bmm`, which is far faster than a
naïve Python loop or `torch.einsum`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask (if present) with left/right gate multiplication
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,        # half *
    right_ptr,       # half *
    left_gate_ptr,   # half *
    right_gate_ptr,  # half *
    mask_ptr,        # half *
    total_elems: tl.int32,          # total number of elements in left/right
    BLOCK: tl.constexpr,
    H: tl.constexpr,                # hidden dimension size
    HAS_MASK: tl.constexpr,         # 1 if mask present else 0
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # Load operands
    left      = tl.load(left_ptr + offs,      mask=mask, other=0.0)
    right     = tl.load(right_ptr + offs,     mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr + offs,mask=mask, other=0.0)

    if HAS_MASK:
        # Each mask element corresponds to H consecutive channel entries
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left  = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left * left_gate
        right = right * right_gate

    # Write back in‑place
    tl.store(left_ptr  + offs, left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _fuse_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    (mask may be None → only gate multiplication).
    All tensors are expected to be contiguous half‑precision.
    """
    total = left.numel()
    BLOCK = 1024  # max 1024 threads per block
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask.view(-1) if mask is not None else left  # dummy when no mask

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        H=hidden_dim,
        HAS_MASK=has_mask,
    )
    # left and right are modified in‑place


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict mapping parameter names to tensors
            - config: dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float16  # half precision for the heavy path

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (in half precision)
    # ------------------------------------------------------------------
    # Cast input to half first to keep everything in half precision
    x = input_tensor.to(dtype)
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (half precision)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(dtype),
        bias=None,
    )  # [B, N, N, hidden_dim]
    right = F.linear(
        x,
        weights["right_proj.weight"].to(dtype),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gate values (half precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(dtype),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(dtype),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(dtype),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) and gates via Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → broadcast over hidden_dim
        mask_h = mask.to(dtype)
        _fuse_gate_and_mask(
            left, right,
            left_gate, right_gate,
            mask_h,
            hidden_dim,
        )
    else:
        # No mask – simple element‑wise gate multiplication
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction (batch of independent GEMMs)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N) for batched GEMM
    l = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    r = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # batched matmul: left @ right^T
    out = torch.bmm(l, r.transpose(1, 2))
    # reshape back to (B, N, N, H)
    out = out.view(B, H, N, N).permute(0, 2, 3, 1)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(dtype),
        bias=None,
    )  # [B, N, N, dim]

    # Return float32 as required by the interface
    return out.to(torch.float32)
