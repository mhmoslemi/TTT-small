@triton.jit
def batched_gemm_kernel(
    a_ptr,  # left: [batch, M, K] float16
    b_ptr,  # right_T: [batch, K, N] float16  (since right_T shape is [batch, K, N] where K=N)
    c_ptr,  # output: [batch, M, N] float16
    batch,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets for M & N
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Create a mask for boundary
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute offset in K dimension
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A and B
        # A: (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + pid_batch * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        # B: we need B submatrix of shape (BLOCK_K, BLOCK_N). Since b_ptr is shaped [batch, K, N], its strides: stride_b_batch, stride_b_k (for rows), stride_b_n (for columns)
        b_ptrs = b_ptr + pid_batch * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        # Load with masks to avoid out-of-bounds
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & mask_n[None, :], other=0.0)

        # Dot: compute block MxN update
        acc += tl.dot(a, b)  # a: (BM, BK), b: (BK, BN) -> (BM, BN)

    # Write back
    c_ptrs = c_ptr + pid_batch * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    # Cast accumulator to half
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

    dim = config['dim']
    hidden_dim = config['hidden_dim']

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor,
                                       normalized_shape=(dim,),
                                       weight=weights['norm.weight'],
                                       bias=weights['norm.bias'])
    # Cast to half for projection
    x_h = x.to(torch.float16)

    # Linear projections
    left = torch.nn.functional.linear(x_h,
                                      weights['left_proj.weight'].to(torch.float16),
                                      bias=None)
    right = torch.nn.functional.linear(x_h,
                                      weights['right_proj.weight'].to(torch.float16),
                                      bias=None)

    left_gate = torch.nn.functional.linear(x_h,
                                           weights['left_gate.weight'].to(torch.float16),
                                           bias=None).sigmoid()
    right_gate = torch.nn.functional.linear(x_h,
                                            weights['right_gate.weight'].to(torch.float16),

    out_gate = torch.nn.functional.linear(x_h,

                                          bias=None).sigmoid()

"""
Outgoing TriMul implementation with Triton.

The operator computes:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

The pipeline:
1. LayerNorm + linear projections (float32).
2. Compute sigmoid gates and (optional) mask, apply them in float32.
3. Cast projected tensors to float16 and execute the O(N³) contraction with a
   custom batched GEMM Triton kernel (half‑precision arithmetic, float32
   accumulation).
4. Apply output LayerNorm, out‑gate scaling and final linear projection
   (float32).

Only the contraction is executed on Triton; all other ops use PyTorch for
clarity and numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr,                      # left  [B, M, K]   (half)
    b_ptr,                      # right [B, K, N]   (half) – already transposed
    c_ptr,                      # out   [B, M, N]   (half)
    BATCH: tl.constexpr,
    M: tl.constexpr,
    N: tl.constexpr,
    K: tl.constexpr,
    stride_a_batch, stride_a_m, stride_a_k,   # strides for a
    stride_b_batch, stride_b_k, stride_b_n,   # strides for b
    stride_c_batch, stride_c_m, stride_c_n,   # strides for c
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute start indices of the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask to avoid OOB accesses
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # pointers to the current sub‑matrices
        a_ptrs = (
            a_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        b_ptrs = (
            b_ptr
            + pid_batch * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n
        )

        # masks for the loads
        a_mask = mask_m[:, None] & (offs_k < K)[None, :]
        b_mask = (offs_k < K)[:, None] & mask_n[None, :]

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        acc += tl.dot(a, b)      # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # write the result back
    c_ptrs = (
        c_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    c = acc.to(tl.float16)
    c_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, c, mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict containing at least "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (float32)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"],
        bias=None,
    )  # [B, N, N, hidden_dim]
    right = F.linear(
        x,
        weights["right_proj.weight"],
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Compute gates (float32) and apply mask if present
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"],
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"],
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"],
        bias=None,
    ).sigmoid()

    if not nomask and mask is not None:
        # mask shape [B, N, N]; broadcast over hidden_dim
        mask_f = mask.to(torch.float32).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f * left_gate
        right = right * mask_f * right_gate
    else:
        left = left * left_gate
        right = right * right_gate

    # release unused tensors
    del left_gate, right_gate, x

    # ------------------------------------------------------------------
    # 4. Cast to half‑precision and prepare for GEMM
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    del left, right

    # Rearrange tensors to (B, H, N, N) -> (B*H, N, N)
    B, N, _, _ = input_tensor.shape
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # transpose the second pair of dimensions of right for multiplication
    right_T = right_perm.permute(0, 1, 3, 2).contiguous()   # (B, H, N, N) where axes 2/3 are swapped

    batch = B * hidden_dim
    left_mat = left_perm.view(batch, N, N)      # (batch, M=N, K=N)
    right_mat = right_T.view(batch, N, N)       # (batch, K=N, N)

    # allocate output tensor
    out_mat = torch.empty_like(left_mat)        # (batch, N, N), half precision

    # ------------------------------------------------------------------
    # 5. Batched GEMM via Triton (out = left @ right_T)
    # ------------------------------------------------------------------
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    stride_a_batch, stride_a_m, stride_a_k = left_mat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_mat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_mat.stride()

    grid = (
        batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)
    del out_mat, left_mat, right_mat, left_perm, right_perm, right_T

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate scaling and final projection
    # ------------------------------------------------------------------
    out_f32 = out.to(torch.float32)

    out_f32 = F.layer_norm(
        out_f32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    out_f32 = out_f32 * out_gate.to(torch.float32)

    out_f32 = F.linear(
        out_f32,
        weights["to_out.weight"],
        bias=None,
    )  # final shape (B, N, N, dim)

    return out_f32
