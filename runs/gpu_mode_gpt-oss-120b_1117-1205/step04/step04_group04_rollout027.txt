    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()

    # 4. Fuse mask with gates using Triton
    if not nomask:

        # flatten all
        left_f = left.view(-1)
        right_f = right.view(-1)

        right_gate_f = right_gate.view(-1)

        right = right * right_gate

@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    N,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)

        # Compute masks for a and b loads
        a_mask = (offs_m < N)[:, None] & (offs_k < N)[None, :]
        b_mask = (offs_k < N)[:, None] & (offs_n < N)[None, :]

        # Compute pointers
        a_ptrs = a_ptr + pid_batch * stride_a_batch + offs_m[:, None] * stride_a_row + offs_k[None, :] * stride_a_col
        b_ptrs = b_ptr + pid_batch * stride_b_batch + offs_k[:, None] * stride_b_col + offs_n[None, :] * stride_b_row
        # Note: b_ptr uses stride_b_col for k dimension (original column stride) and stride_b_row for n dimension (since transposed)

        a = tl.load(a_ptrs, mask=a_mask, other=tl.float16(0.0))
        b = tl.load(b_ptrs, mask=b_mask, other=tl.float16(0.0))

        # Dot product
        acc += tl.dot(a, b)

    # Write back
    c_mask = (offs_m < N)[:, None] & (offs_n < N)[None, :]
    c_ptrs = c_ptr + pid_batch * stride_c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)

"""
Outgoing TriMul (AlphaFold3) forward pass with Triton acceleration.

Algorithm
---------
1. Layer‑norm the input and cast to FP16.
2. Linear projections to hidden dimension (no bias).
3. Compute three gate tensors (sigmoid).
4. Fuse optional mask with left/right gates in‑place via a Triton kernel.
5. Core O(N³) contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM (left @ rightᵀ) where the hidden
   dimension is fused into the batch axis and executed by a custom Triton
   matmul kernel (FP16 inputs, FP32 accumulation, Tensor‑Core friendly).
6. Output LayerNorm, element‑wise gating with out_gate and final linear
   projection back to the original feature dimension.
7. Return FP32 tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # *fp16
    right_ptr,         # *fp16
    left_gate_ptr,     # *fp16
    right_gate_ptr,    # *fp16
    mask_ptr,          # *fp16 (broadcasted over hidden dim)
    total_elems,       # i32
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,   # hidden dimension
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    lgate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # one mask value per (b,i,j) → divide by hidden dim to get mask index
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * lgate * mask_val
        right = right * rgate * mask_val
    else:
        left = left * lgate
        right = right * rgate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

    # tensors are updated in‑place


# ----------------------------------------------------------------------
# Triton kernel: batched matmul (A @ Bᵀ) for shape (B*H, N, N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    N,                                   # matrix dimension (square)
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds loads
        a_mask = (offs_m < N)[:, None] & (offs_k < N)[None, :]
        b_mask = (offs_k < N)[:, None] & (offs_n < N)[None, :]

        # Pointers respecting the transposition of B (we need Bᵀ)
        a_ptrs = (
            a_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_row
            + offs_k[None, :] * stride_a_col
        )
        b_ptrs = (
            b_ptr
            + pid_batch * stride_b_batch
            + offs_k[:, None] * stride_b_col   # k dimension uses original column stride
            + offs_n[None, :] * stride_b_row   # n dimension uses original row stride
        )

        a = tl.load(a_ptrs, mask=a_mask, other=tl.float16(0.0))
        b = tl.load(b_ptrs, mask=b_mask, other=tl.float16(0.0))

        # Tensor‑core dot (FP16 inputs, FP32 accumulation)
        acc += tl.dot(a, b)

    # Write result back (cast to FP16)
    c_mask = (offs_m < N)[:, None] & (offs_n < N)[None, :]
    c_ptrs = (
        c_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_row
        + offs_n[None, :] * stride_c_col
    )
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (float32)
            - config: contains "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    inp, mask, weights, config = data
    device = inp.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32 → FP16)
    # --------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    ).to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections to hidden dimension
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 3. Gate tensors (sigmoid)
    # --------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # --------------------------------------------------------------
    # 4. Fuse mask with gates (if required) via Triton
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → broadcast over hidden dimension
        mask_h = mask.to(torch.float16).unsqueeze(-1)               # [B, N, N, 1]

        # Flatten everything for the 1‑D kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        lg_f = left_gate.contiguous().view(-1)
        rg_f = right_gate.contiguous().view(-1)
        mask_f = mask_h.contiguous().view(-1)                      # length B*N*N

        _apply_mask_and_gate(left_f, right_f, lg_f, rg_f, mask_f, hidden_dim)

        # Reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction: batched GEMM (left @ rightᵀ)
    # --------------------------------------------------------------
    # Rearrange to (B, H, N, N) and make contiguous
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_t.shape
    left_mat = left_t.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_t.view(B * H, N, N) # (B*H, N, N)

    out_mat = torch.empty_like(left_mat)  # FP16 destination

    # Kernel launch parameters (tuned for Tensor Cores)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                         # batch dimension
        triton.cdiv(N, BLOCK_M),      # rows
        triton.cdiv(N, BLOCK_N),      # cols
    )

    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        left_mat.stride(0),
        left_mat.stride(1),
        left_mat.stride(2),
        right_mat.stride(0),
        right_mat.stride(1),
        right_mat.stride(2),
        out_mat.stride(0),
        out_mat.stride(1),
        out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 7. Return FP32 tensor
    # --------------------------------------------------------------
    return out.to(torch.float32)
