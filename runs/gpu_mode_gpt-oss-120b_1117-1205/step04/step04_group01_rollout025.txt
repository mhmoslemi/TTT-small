"""Implementation of the outgoing TriMul operator using Triton for mask&gate fusion and
batched GEMM for the O(N³) contraction.

The kernel fuses:  left = left * left_gate * mask
                right = right * right_gate * mask
(when a mask is provided).  The main contraction is rewritten as a batch of
matrix multiplications per hidden channel:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to
    out_d[b] = left_d[b] @ right_d[b].T
and is executed efficiently with `torch.bmm` on FP16 tensors.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ---------------------------------------------------------------
# Triton kernel: fuse mask and gates (optional)
# ---------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half *
    right_ptr,         # half *
    left_gate_ptr,     # half *
    right_gate_ptr,    # half *
    mask_ptr,          # half *
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    H: tl.constexpr,           # hidden dimension
    HAS_MASK: tl.constexpr,    # 1 if mask present, else 0
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    # Load the four tensors
    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    # Apply mask if present
    mask_val = 1.0
    if HAS_MASK:
        # each mask element is broadcast over H channel entries
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=0.0)

    left = left * left_gate * mask_val
    right = right * right_gate * mask_val

    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def _fuse_gate_and_mask(left: torch.Tensor,
                        right: torch.Tensor,
                        left_gate: torch.Tensor,
                        right_gate: torch.Tensor,
                        mask: torch.Tensor,
                        hidden_dim: int):
    """In‑place elementwise fused multiplication:
    left = left * left_gate * mask
    right = right * right_gate * mask
    (mask may be None – then only gates are applied)."""
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer for kernel

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        H=hidden_dim,
        HAS_MASK=has_mask,
    )


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weight_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/int/float) or None
            - weights: dict of model parameters
            - config: dict with keys ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP16 for speed)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    ).to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (FP16)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gate values (FP16)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) with gates using Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask broadcasting: [B,N,N] -> [B,N,N,1] -> flatten for kernel
        mask_h = mask.to(torch.float16).contiguous()
        # Ensure contiguous layout for the kernel
        _fuse_gate_and_mask(
            left,
            right,
            left_gate,
            right_gate,
            mask_h,
            hidden_dim,
        )
    else:
        # No mask – simple elementwise multiplication
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³) contraction using batched GEMM (FP16)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Bring hidden dimension to batch dimension
    # left: [B, N, N, H] -> [B, H, N, N] (i,k)
    # right: [B, N, N, H] -> [B, H, N, N] (k,j) after transposition of last two dims
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_mat = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (k,j)

    B, H, N, _ = left_mat.shape
    left_flat = left_mat.view(B * H, N, N)   # (B*H, N, N)
    right_flat = right_mat.view(B * H, N, N) # (B*H, N, N)

    # Batched matrix multiplication (FP16)
    out_flat = torch.bmm(left_flat, right_flat)  # (B*H, N, N)

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output layer norm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in float32 as required by the API
    return out.float()
