"""
TriMul (outgoing) forward pass with fused gate & mask multiplication via Triton.

Algorithm:
1. Layer‑norm the input (float32) → half precision.
2. Linear projections to `hidden_dim` for left/right tensors.
3. Compute element‑wise gates (sigmoid of linear projections of the same input).
4. Fuse `left = left * left_gate * mask` and `right = right * right_gate * mask`
   (or just the gates when no mask) with a custom Triton kernel.
5. Core contraction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM:
     left_perm (B,H,N,N) and right_perm (B,H,N,N) → view as (B*H, N, N)
     out_mat = bmm(left_mat, right_mat^T) → reshape back to (B,N,N,H).
6. Apply output LayerNorm, element‑wise scaling by `out_gate`,
   and final linear projection back to `dim`.
7. Return result in float32.

Only the gate+mask multiplication is performed in Triton; the heavy
contraction uses torch.bmm (cuBLAS) which is already highly optimized.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half* (flattened left tensor)
    right_ptr,         # half* (flattened right tensor)
    left_gate_ptr,     # half* (flattened left gate)
    right_gate_ptr,    # half* (flattened right gate)
    mask_ptr,          # half* (flattened mask, broadcast over hidden dim)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present else 0
    H: tl.constexpr,          # hidden dimension
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load data
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # Each hidden element shares the same mask entry.
        # mask_idx = floor(offs / H)
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] float32
    x_h = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half precision)
    # -----------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_weight, bias=None)   # [B, N, N, H]
    right = F.linear(x_h, right_proj_weight, bias=None)  # [B, N, N, H]

    # -----------------------------------------------------------------
    # 3. Gates (half precision)
    # -----------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_h, left_gate_weight, bias=None).sigmoid()   # [B, N, N, H]
    right_gate = F.linear(x_h, right_gate_weight, bias=None).sigmoid() # [B, N, N, H]
    out_gate = F.linear(x_h, out_gate_weight, bias=None).sigmoid()    # [B, N, N, H]

    # -----------------------------------------------------------------
    # 4. Fuse mask and gates with Triton (if mask needed)
    # -----------------------------------------------------------------
    if not nomask:
        # Ensure contiguous half tensors
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_h = mask.to(torch.float16).contiguous()  # [B, N, N]

        # Flatten tensors for the kernel
        left_f = left.view(-1)
        right_f = right.view(-1)
        left_gate_f = left_gate.view(-1)
        right_gate_f = right_gate.view(-1)
        # mask is broadcast over hidden dimension; flatten without hidden dim
        mask_f = mask_h.view(-1)

        total = left_f.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)

        _gate_mask_fuse_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            total,
            BLOCK=BLOCK,
            HAS_MASK=1,
            H=hidden_dim,
        )
        # Reshape back to original layout
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # Simple element‑wise multiplication (no mask)
        left = left * left_gate
        right = right * right_gate

    # Gates are no longer needed
    del left_gate, right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    #    Implemented as batch GEMM via torch.bmm.
    # -----------------------------------------------------------------
    # Permute to (B, H, N, N) for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)          # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)        # (B*H, N, N)

    # Batched matrix multiplication: (B*H, N, N) x (B*H, N, N)^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore original layout (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final projection
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )  # half precision

    # Apply out_gate (already half)
    out = out * out_gate

    # Final linear projection back to dim
    to_out_weight = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_weight, bias=None)  # [B, N, N, dim] half

    # Return in float32 as required by the interface
    return out.to(torch.float32)
```
