"""
Outgoing TriMul implementation (AlphaFold3).

Algorithm
---------
1. Layer‑norm the input tensor (float32 → float16).
2. Linear projections `left` and `right` to `hidden_dim`.
3. Compute three gating vectors (sigmoid of linear projections):
   `left_gate`, `right_gate`, `out_gate`.
4. Fuse gating and optional binary mask in a Triton kernel:
   left  = left  * left_gate  * mask
   right = right * right_gate * mask
5. Core contraction:
   For each batch and hidden channel `h`,
       out_{i,j,h} = Σ_k left_{i,k,h} * right_{j,k,h}
   This is a batched GEMM `L @ Rᵀ` using torch.bmm.
6. Layer‑norm over `hidden_dim`, apply `out_gate`, and final linear
   projection back to `dim`.
All heavy element‑wise work (step 4) is done in Triton; the
large matrix multiplication uses cuBLAS via torch.bmm.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # float16*
    right_ptr,           # float16*
    left_gate_ptr,       # float16*
    right_gate_ptr,      # float16*
    mask_ptr,            # float16* (broadcasted, length = total_mask)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask is present
    H: tl.constexpr,          # hidden_dim (size of innermost channel)
):
    """In‑place fusion:
    left  = left  * left_gate  * (mask if present)
    right = right * right_gate * (mask if present)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elements

    left = tl.load(left_ptr + offs, mask=m, other=0.0)
    right = tl.load(right_ptr + offs, mask=m, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=m, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=m, other=0.0)

    if HAS_MASK:
        # each mask entry corresponds to H consecutive hidden entries
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=m, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=m)
    tl.store(right_ptr + offs, right, mask=m)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    Fuse left/right with their respective gates and (optionally) a mask
    using the Triton kernel above. All tensors must be contiguous and
    of dtype torch.float16.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    # Dummy pointer when mask is not used (kernel will ignore it)
    mask_ptr = mask if mask is not None else left

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # In‑place operation; nothing returned.


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple containing
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) – may be None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision for speed
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half precision)
    # ------------------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gating vectors (half) with sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) with gates via Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → half, flattened
        mask_f = mask.to(torch.float16).contiguous().view(-1)  # [B*N*N]

        # Flatten tensors for the kernel (contiguous view)
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            hidden_dim,
        )
        # Reshape back (views share storage)
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask: simple element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate

    # Release memory that is no longer needed
    del left_gate, right_gate, x_h

    # ------------------------------------------------------------------
    # 5. Core contraction using batched GEMM:
    #    out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) for efficient bmm
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # Batched matrix multiplication: L @ Rᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore layout (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise gating (half)

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in float32 as required by the interface
    return out.to(torch.float32)
```
