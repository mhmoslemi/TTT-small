"""
Outgoing TriMul (AlphaFold3) forward pass, optimized with Triton.
The kernel fuses the O(N³) contraction (batched matrix‑multiply) and
a separate Triton kernel fuses mask & gating element‑wise.
All heavy arithmetic is performed in FP16 with FP32 accumulation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernel: fuse mask (if present) and gating into the left/right tensors
# -------------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # *fp16
    right_ptr,         # *fp16
    left_gate_ptr,     # *fp16
    right_gate_ptr,    # *fp16
    mask_ptr,          # *fp16  (broadcasted over hidden dim)
    total_elems: tl.int32,
    H: tl.constexpr,   # hidden dimension (fastest stride)
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elems

    # load the four tensors
    left = tl.load(left_ptr + offs, mask=m, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=m, other=tl.float16(0.0))
    lgate = tl.load(left_gate_ptr + offs, mask=m, other=tl.float16(0.0))
    rgate = tl.load(right_gate_ptr + offs, mask=m, other=tl.float16(0.0))

    if HAS_MASK:
        # one mask value per (b, i, j); repeats H times consecutively
        mask_idx = offs // H
        mask = tl.load(mask_ptr + mask_idx, mask=m, other=tl.float16(0.0))
        left = left * lgate * mask
        right = right * rgate * mask
    else:
        left = left * lgate
        right = right * rgate

    tl.store(left_ptr + offs, left, mask=m)
    tl.store(right_ptr + offs, right, mask=m)

        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        hidden_dim,

    )

# -------------------------------------------------------------------------
# Triton kernel: batched GEMM for the core O(N³) contraction
#   C[b,i,j] = Σ_k A[b,i,k] * B[b,j,k]
# where A = left, B = right (not transposed, we load B^T inside the kernel)
# -------------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr,               # *fp16  (left)
    B_ptr,               # *fp16  (right)
    C_ptr,               # *fp16  (output)
    batch: tl.int32,     # total number of independent GEMMs = B * hidden_dim
    M: tl.int32,         # seq_len (rows of A)
    N: tl.int32,         # seq_len (cols of B)
    K: tl.int32,         # seq_len (reduction dim)
    stride_Ab, stride_Ai, stride_Ak,
    stride_Bb, stride_Bj, stride_Bk,
    stride_Cb, stride_Ci, stride_Cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # boundary masks for rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch entry
    a_batch_off = batch_idx * stride_Ab
    b_batch_off = batch_idx * stride_Bb
    c_batch_off = batch_idx * stride_Cb

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (M, K)
        a_ptrs = A_ptr + a_batch_off + offs_m[:, None] * stride_Ai + offs_k[None, :] * stride_Ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=tl.float16(0.0))

        # B tile transposed: we need (K, N) = B[j,k] accessed as B_T[k,j]
        b_ptrs = B_ptr + b_batch_off + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bj
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=tl.float16(0.0))

        # dot: (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(a, b)

    # write result back
    c_ptrs = C_ptr + c_batch_off + offs_m[:, None] * stride_Ci + offs_n[None, :] * stride_Cj
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# -------------------------------------------------------------------------
# Entrypoint
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor_or_None, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    inp, mask, weights, config = data
    device = inp.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. LayerNorm + cast to FP16
    # -----------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections to hidden_dim (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # -----------------------------------------------------------------
    # 3. Compute gate tensors (sigmoid)
    # -----------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # -----------------------------------------------------------------
    # 4. Apply mask & gates (fused Triton kernel if a mask is present)
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] -> broadcast over hidden dimension
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        # flatten everything
        left_f = left.view(-1)
        right_f = right.view(-1)
        lgate_f = left_gate.view(-1)
        rgate_f = right_gate.view(-1)
        mask_f = mask_h.view(-1)                     # length B*N*N

        _apply_mask_and_gate(left_f, right_f, lgate_f, rgate_f, mask_f, hidden_dim)

        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # simple element‑wise multiplication (in‑place to save memory)
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core O(N³) contraction via a batched GEMM Triton kernel
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape  # left shape = (B, N, N, hidden_dim)

    # bring batch & hidden dim together and make the matrices contiguous
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    batch = B * hidden_dim
    left_flat = left_mat.view(batch, N, N)   # (B*H, N, N)
    right_flat = right_mat.view(batch, N, N)

    # allocate output
    out_flat = torch.empty((batch, N, N), dtype=torch.float16, device=device)

    # strides (row‑major)
    stride_Ab, stride_Ai, stride_Ak = left_flat.stride()
    stride_Bb, stride_Bj, stride_Bk = right_flat.stride()
    stride_Cb, stride_Ci, stride_Cj = out_flat.stride()

    # block sizes – tuned for the B200 GPU
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch,
        N,
        N,
        N,
        stride_Ab,
        stride_Ai,
        stride_Ak,
        stride_Bb,
        stride_Bj,
        stride_Bk,
        stride_Cb,
        stride_Ci,
        stride_Cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating and final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # return in float32 as required
    return out.to(torch.float32)
