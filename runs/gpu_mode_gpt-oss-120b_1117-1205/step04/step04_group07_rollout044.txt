def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)

    # Layer Norm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # Cast to half
    x_h = x.to(torch.float16)

    # Linear projections:
    left = torch.nn.functional.linear(x_h, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = torch.nn.functional.linear(x_h, weights["right_proj.weight"].to(torch.float16), bias=None)

    left_gate = torch.nn.functional.linear(x_h, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = torch.nn.functional.linear(x_h, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = torch.nn.functional.linear(x_h, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # Apply mask and gating
    if not nomask:
        # mask shape [B,N,N]; expand and flatten
        mask_f = mask_tensor.to(torch.float16).contiguous().view(-1)  # shape B*N*N
        # flatten left etc
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)

        # Launch kernel
        total = left_flat.numel()
        BLOCK = 32768  # maybe 16384? Ensure multiple of 32
        grid = (triton.cdiv(total, BLOCK),)
        HAS_MASK = 1
        apply_mask_gate_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_f,
            total,
            BLOCK=BLOCK,
            HAS_MASK=HAS_MASK,
            HIDDEN_DIM=hidden_dim,
        )
        # reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # Core contraction: left shape [B,N,N,H]; need to compute out = left @ right^T per head.
    # We'll permute to [B,H,N,N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # B, H, N, N
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_perm.shape

    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)
    # batch matrix multiply left @ right^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))
    # reshape back: (B,H,N,N)
    out = out_mat.view(B, H, N, N).permute(0,2,3,1).contiguous()  # B,N,N,H

    # to_out_norm: normalize across hidden_dim (H)

                                         (hidden_dim,),
                                         weight=weights["to_out_norm.weight"].to(torch.float16),
                                         bias=weights["to_out_norm.bias"].to(torch.float16))

    # Apply out_gate (elementwise)
    out = out * out_gate

    # Final linear projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    # Return float32
    return out.to(torch.float32)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel to fuse mask and gating for left/right tensors
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    # Load values
    left = tl.load(left_ptr + offs, mask=active, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=active, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=tl.float16(0.0))

    if HAS_MASK:
        # Each element repeats HIDDEN_DIM times across the hidden dimension.
        mask_idx = offs // HIDDEN_DIM
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=tl.float16(1.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold3).

    The implementation follows the PyTorch reference but fuses the
    mask‑and‑gate multiplication of the ``left`` and ``right`` tensors
    in a single Triton kernel.  All heavy linear algebra (the N³
    contraction) is performed with a batched GEMM (torch.bmm), which
    maps to cuBLAS.  Computation is carried out in float16 for speed;
    the final result is returned in float32.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float16)
    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half precision)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16), bias=None)

    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # ------------------------------------------------------------------
    # 3. Fuse mask and gating (Triton)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] -> flatten
        mask_flat = mask_tensor.to(torch.float16).contiguous().view(-1)

        # flatten all tensors for kernel launch
        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)

        total_elems = left_flat.numel()
        BLOCK = 16384  # works well for half precision

        grid = (triton.cdiv(total_elems, BLOCK),)
        _mask_gate_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total_elems,
            BLOCK=BLOCK,
            HAS_MASK=1,
            HIDDEN_DIM=hidden_dim,
        )
        # Reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask – just apply gates
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Core contraction: Left @ Right^T per head (batched GEMM)
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    # Collapse batch and head dimensions
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # Matrix multiply left @ right^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore shape (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, out‑gate scaling, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    # Return in float32 as required by the reference implementation
    return out.to(torch.float32)

"""
Outgoing TriMul (AlphaFold3) forward pass.

- Input is LayerNormed, projected to a hidden dimension, masked and gated.
- The mask‑and‑gate fusion for the left/right tensors is done in a single
  Triton kernel (half precision).
- The N³ contraction is expressed as a batched GEMM (torch.bmm) which maps
  to cuBLAS and is highly efficient.
- Output is LayerNormed, scaled by an output gate and projected back to the
  original channel size.  The final tensor is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,            # * half
    right_ptr,           # * half
    left_gate_ptr,       # * half
    right_gate_ptr,      # * half
    mask_ptr,            # * half (flattened B*N*N)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):
    """In‑place fuse of mask and sigmoid gates.

    left, right      : [B,N,N,H] flattened
    left_gate, right_gate : same shape
    mask (if present): [B,N,N] flattened and broadcast over H
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=active, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=active, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=tl.float16(0.0))

    if HAS_MASK:
        # Each mask element repeats HIDDEN_DIM times along the hidden axis.
        mask_idx = offs // HIDDEN_DIM
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=tl.float16(1.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32, CUDA)
            - mask_tensor : torch.Tensor [B, N, N] (bool/float, CUDA) or None
            - weights     : dict with the following keys (all torch.Tensors on CUDA):
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config      : dict with at least:
                "dim": input channel dimension,
                "hidden_dim": hidden dimension,
                "nomask" (optional): True/False

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm of the input (float16 for downstream ops)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half precision)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16), bias=None)

    left_gate = (
        F.linear(x, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    )
    right_gate = (
        F.linear(x, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    )
    out_gate = (
        F.linear(x, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()
    )

    # ------------------------------------------------------------------
    # 3. Fuse mask (if any) and gating with Triton
    # ------------------------------------------------------------------
    if not nomask:
        # Flatten mask to [B*N*N] and left/right tensors to 1‑D.
        mask_flat = mask_tensor.to(torch.float16).contiguous().view(-1)

        left_flat = left.contiguous().view(-1)
        right_flat = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)

        total = left_flat.numel()
        BLOCK = 16384  # must be a multiple of 32
        grid = (triton.cdiv(total, BLOCK),)

        _mask_gate_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            BLOCK=BLOCK,
            HAS_MASK=1,
            HIDDEN_DIM=hidden_dim,
        )

        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask – simple element‑wise gate multiplication
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Core N³ contraction via batched GEMM (left @ rightᵀ per head)
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)  # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, out‑gate scaling, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    # Return in float32 as required by the reference implementation
    return out.to(torch.float32)
