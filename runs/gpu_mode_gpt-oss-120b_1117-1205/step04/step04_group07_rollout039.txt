        bias=None

    block_i = tl.program_id(1)

    """
    Optimized forward pass of the outgoing TriMul operator using a fused Triton kernel.
    The kernel computes the gated and masked bilinear contraction
        out_{b,i,j,d} = Σ_k (left_{b,i,k,d} * left_gate_{b,i,k,d} * mask_{b,i,k}) *
                       (right_{b,j,k,d} * right_gate_{b,j,k,d} * mask_{b,j,k})
    for all batches b and hidden dimensions d.
    Subsequent layer‑norm, out‑gate scaling and final linear projection are done in PyTorch.
    """
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. LayerNorm over last dimension
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    x_h = x.to(torch.float16)

    # 2. Linear projections to hidden_dim
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16), bias=None)   # [B,N,N,H]
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16), bias=None)

    # 3. Gate projections (sigmoid)
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # 4. Rearrange to [B, H, N, N] for kernel
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_gate_perm = left_gate.permute(0, 3, 1, 2).contiguous()
    right_gate_perm = right_gate.permute(0, 3, 1, 2).contiguous()
    # out_gate stays in original layout [B,N,N,H] for later multiplication

    B, H, N, _ = left_perm.shape
    assert H == hidden_dim, "hidden_dim mismatch"

    # flatten batch*hidden dimension
    left_view = left_perm.view(B * H, N, N)
    right_view = right_perm.view(B * H, N, N)
    left_gate_view = left_gate_perm.view(B * H, N, N)
    right_gate_view = right_gate_perm.view(B * H, N, N)

    out_view = torch.empty_like(left_view)  # half

    # Mask handling
    if nomask:
        mask_ptr = left_view  # dummy pointer
        has_mask = 0
        stride_mask_batch = left_view.stride(0)
        stride_mask_i = left_view.stride(1)
        stride_mask_j = left_view.stride(2)
    else:
        mask = mask.to(torch.float16).contiguous()
        mask_ptr = mask
        has_mask = 1
        stride_mask_batch = mask.stride(0)
        stride_mask_i = mask.stride(1)
        stride_mask_j = mask.stride(2)

    # Strides for left
    stride_left_batch = left_view.stride(0)
    stride_left_m = left_view.stride(1)
    stride_left_k = left_view.stride(2)

    # Strides for right (transposed view)
    stride_right_batch = right_view.stride(0)
    stride_right_m = right_view.stride(2)   # row stride for j (original col)
    stride_right_k = right_view.stride(1)   # col stride for k (original row)

    # Strides for left gate (same layout as left)
    stride_left_gate_batch = left_gate_view.stride(0)
    stride_left_gate_m = left_gate_view.stride(1)
    stride_left_gate_k = left_gate_view.stride(2)

    # Strides for right gate (transposed)
    stride_right_gate_batch = right_gate_view.stride(0)
    stride_right_gate_m = right_gate_view.stride(2)  # row stride for j
    stride_right_gate_k = right_gate_view.stride(1)  # col stride for k

    # Strides for output
    stride_out_batch = out_view.stride(0)
    stride_out_m = out_view.stride(1)
    stride_out_n = out_view.stride(2)

    # Kernel launch parameters
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (B * H, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    triton_trimul_kernel[grid](
        left_view,
        right_view,
        left_gate_view,
        right_gate_view,
        mask_ptr,
        out_view,
        N, N, N,
        stride_left_batch, stride_left_m, stride_left_k,
        stride_right_batch, stride_right_m, stride_right_k,
        stride_left_gate_batch, stride_left_gate_m, stride_left_gate_k,
        stride_right_gate_batch, stride_right_gate_m, stride_right_gate_k,
        stride_out_batch, stride_out_m, stride_out_n,
        stride_mask_batch, stride_mask_i, stride_mask_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        H=hidden_dim,
        HAS_MASK=has_mask,
    )

    # Reshape back to [B,N,N,H]
    out = out_view.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Apply out_gate
    out = out * out_gate

    # Output layer norm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16)
    )

    # Final linear projection back to dim
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None
    )

    return out.to(torch.float32)

"""
Optimized outgoing TriMul forward pass.

The core O(N³) contraction is performed by a custom Triton kernel that
fuses the left/right linear projections, sigmoid gates and optional mask
into the matrix multiplication:
    out[b, i, j, d] = Σₖ (left[b,i,k,d] * left_gate[b,i,k,d] * mask[b,i,k]) *
                         (right[b,j,k,d] * right_gate[b,j,k,d] * mask[b,j,k])
The remaining operations (LayerNorms, out‑gate, final linear) are handled
in PyTorch using half‑precision tensors, and the final result is returned
as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def triton_trimul_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    out_ptr,
    M, N, K,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_m, stride_right_k,
    stride_left_gate_batch, stride_left_gate_m, stride_left_gate_k,
    stride_right_gate_batch, stride_right_gate_m, stride_right_gate_k,
    stride_out_batch, stride_out_m, stride_out_n,
    stride_mask_batch, stride_mask_i, stride_mask_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    H: tl.constexpr, HAS_MASK: tl.constexpr
):
    batch = tl.program_id(0)          # 0 .. B*H-1
    block_i = tl.program_id(1)        # tile row index
    block_j = tl.program_id(2)        # tile col index

    i_start = block_i * BLOCK_M
    j_start = block_j * BLOCK_N

    # original batch index for mask (mask does not depend on hidden dim)
    b_idx = batch // H

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K

        # ---- left tile (A) -------------------------------------------------
        a_i = i_start + tl.arange(0, BLOCK_M)
        a_k = k_start + tl.arange(0, BLOCK_K)
        a_i_valid = a_i < M
        a_k_valid = a_k < K
        a_mask = a_i_valid[:, None] & a_k_valid[None, :]

        left = tl.load(
            left_ptr + batch * stride_left_batch
                     + a_i[:, None] * stride_left_m
                     + a_k[None, :] * stride_left_k,
            mask=a_mask, other=0.0)

        left_gate = tl.load(
            left_gate_ptr + batch * stride_left_gate_batch
                         + a_i[:, None] * stride_left_gate_m
                         + a_k[None, :] * stride_left_gate_k,
            mask=a_mask, other=0.0)

        if HAS_MASK:
            mask_val = tl.load(
                mask_ptr + b_idx * stride_mask_batch
                         + a_i[:, None] * stride_mask_i
                         + a_k[None, :] * stride_mask_j,
                mask=a_mask, other=1.0)
            left = left * left_gate * mask_val
        else:
            left = left * left_gate

        # ---- right tile (Bᵀ) ----------------------------------------------
        b_j = j_start + tl.arange(0, BLOCK_N)
        b_k = k_start + tl.arange(0, BLOCK_K)
        b_j_valid = b_j < N
        b_k_valid = b_k < K
        b_mask = b_j_valid[:, None] & b_k_valid[None, :]

        # right is accessed as right[k, j] => transposed view
        right = tl.load(
            right_ptr + batch * stride_right_batch
                      + b_k[:, None] * stride_right_m
                      + b_j[None, :] * stride_right_k,
            mask=b_mask, other=0.0)

        right_gate = tl.load(
            right_gate_ptr + batch * stride_right_gate_batch
                           + b_k[:, None] * stride_right_gate_m
                           + b_j[None, :] * stride_right_gate_k,
            mask=b_mask, other=0.0)

        if HAS_MASK:
            mask_val_r = tl.load(
                mask_ptr + b_idx * stride_mask_batch
                         + b_j[:, None] * stride_mask_i
                         + b_k[None, :] * stride_mask_j,
                mask=b_mask, other=1.0)
            right = right * right_gate * mask_val_r
        else:
            right = right * right_gate

        # ---- accumulate ----------------------------------------------------
        acc = acc + tl.dot(left, right)   # (BLOCK_M, BLOCK_N)

    # ---- store result ----------------------------------------------------
    c_i = i_start + tl.arange(0, BLOCK_M)
    c_j = j_start + tl.arange(0, BLOCK_N)
    c_i_valid = c_i < M
    c_j_valid = c_j < N
    c_mask = c_i_valid[:, None] & c_j_valid[None, :]

    tl.store(
        out_ptr + batch * stride_out_batch
                 + c_i[:, None] * stride_out_m
                 + c_j[None, :] * stride_out_n,
        acc.to(tl.float16),
        mask=c_mask
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor of shape [B, N, N] (float16 or float32)
            - weights: dict with layer‑norm, linear and gate weights (float32)
            - config: dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half precision)
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16), bias=None)

    # ------------------------------------------------------------------
    # 3. Gate values (half, sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # ------------------------------------------------------------------
    # 4. Rearrange tensors to [B, H, N, N] and flatten batch*H
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_gate_perm = left_gate.permute(0, 3, 1, 2).contiguous()
    right_gate_perm = right_gate.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    assert H == hidden_dim, "hidden_dim mismatch"

    left_view = left_perm.view(B * H, N, N)
    right_view = right_perm.view(B * H, N, N)
    left_gate_view = left_gate_perm.view(B * H, N, N)
    right_gate_view = right_gate_perm.view(B * H, N, N)

    out_view = torch.empty_like(left_view)   # half‑precision output

    # ------------------------------------------------------------------
    # 5. Mask handling (broadcast across hidden dimension)
    # ------------------------------------------------------------------
    if nomask:
        mask_ptr = left_view          # dummy, never accessed
        has_mask = 0
        stride_mask_batch = left_view.stride(0)
        stride_mask_i = left_view.stride(1)
        stride_mask_j = left_view.stride(2)
    else:
        mask = mask.to(torch.float16).contiguous()
        mask_ptr = mask
        has_mask = 1
        stride_mask_batch = mask.stride(0)
        stride_mask_i = mask.stride(1)
        stride_mask_j = mask.stride(2)

    # ------------------------------------------------------------------
    # 6. Strides for all tensors
    # ------------------------------------------------------------------
    # left (non‑transposed)
    stride_left_batch = left_view.stride(0)
    stride_left_m = left_view.stride(1)
    stride_left_k = left_view.stride(2)

    # right (will be accessed transposed)
    stride_right_batch = right_view.stride(0)
    stride_right_m = right_view.stride(2)   # row stride for j
    stride_right_k = right_view.stride(1)   # col stride for k

    # left gate (same layout as left)
    stride_left_gate_batch = left_gate_view.stride(0)
    stride_left_gate_m = left_gate_view.stride(1)
    stride_left_gate_k = left_gate_view.stride(2)

    # right gate (transposed layout)
    stride_right_gate_batch = right_gate_view.stride(0)
    stride_right_gate_m = right_gate_view.stride(2)
    stride_right_gate_k = right_gate_view.stride(1)

    # output
    stride_out_batch = out_view.stride(0)
    stride_out_m = out_view.stride(1)
    stride_out_n = out_view.stride(2)

    # ------------------------------------------------------------------
    # 7. Triton kernel launch (batched GEMM with fused gating & mask)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (B * H, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    triton_trimul_kernel[grid](
        left_view,
        right_view,
        left_gate_view,
        right_gate_view,
        mask_ptr,
        out_view,
        N, N, N,                                    # M, N, K
        stride_left_batch, stride_left_m, stride_left_k,
        stride_right_batch, stride_right_m, stride_right_k,
        stride_left_gate_batch, stride_left_gate_m, stride_left_gate_k,
        stride_right_gate_batch, stride_right_gate_m, stride_right_gate_k,
        stride_out_batch, stride_out_m, stride_out_n,
        stride_mask_batch, stride_mask_i, stride_mask_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        H=hidden_dim,
        HAS_MASK=has_mask,
    )

    # ------------------------------------------------------------------
    # 8. Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_view.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9. Apply out‑gate, output LayerNorm, and final projection
    # ------------------------------------------------------------------
    out = out * out_gate

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16)
    )
    out = F.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    # Return in float32 as specified
    return out.to(torch.float32)
