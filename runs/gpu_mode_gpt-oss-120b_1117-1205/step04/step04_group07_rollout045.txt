```python
@triton.jit
def fuse_gate_mask_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr,
    mask_ptr,   # mask broadcasted along hidden_dim
    total_elems, hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # Compute mask index (one mask per hidden_dim block)
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(1.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)
```

```python
        input_tensor.to(torch.float16),
        normalized_shape=(dim,),
        weight=w_norm,
        bias=b_norm,
    )
    # Linear projections
    left = F.linear(x, w_left_proj)   # (B,N,N,hidden_dim)
    right = F.linear(x, w_right_proj)
    # Gating
    left_gate = torch.sigmoid(F.linear(x, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x, w_right_gate))
    out_gate = torch.sigmoid(F.linear(x, w_out_gate))

    # Fuse gates and optional mask using Triton kernel
    if not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        # ensure contiguous
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp16 = mask_fp16.contiguous()
        total = left.numel()
        BLOCK = 65536
        grid = (triton.cdiv(total, BLOCK),)
        HAS_MASK = 1
        fuse_gate_mask_kernel[grid](
            left,
            right,
            left_gate,
            right_gate,
            mask_fp16,
            total,
            hidden_dim,
            HAS_MASK=HAS_MASK,
            BLOCK=BLOCK,
        )
    else:
        # No mask: still fuse with gates via kernel (HAS_MASK=0)
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        total = left.numel()
        BLOCK = 65536
        grid = (triton.cdiv(total, BLOCK),)
        HAS_MASK = 0
        # dummy mask pointer (reuse left)
        fuse_gate_mask_kernel[grid](
            left,
            right,
            left_gate,
            right_gate,
            left,  # dummy
            total,
            hidden_dim,
            HAS_MASK=HAS_MASK,
            BLOCK=BLOCK,
        )

    # Release gating tensors
    del left_gate, right_gate

    # Core contraction: batched matmul per hidden dim
    # Permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2)
    right_perm = right.permute(0, 3, 1, 2)

    # Perform batch matmul: left @ right^T over K dimension
    out_perm = torch.matmul(left_perm, right_perm.transpose(-2, -1))

    # Permute back to (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1)

    # Output LayerNorm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
    )
    # Apply out_gate scaling
    out = out * out_gate

    # Final linear projection back to dim
    out = F.linear(out, w_to_out)

    # Cast back to float32
    return out.to(torch.float32)
```

"""
TriMul (outgoing) forward pass with Triton acceleration.

Algorithm
---------
1. Layer‑normalize the input tensor and cast to fp16.
2. Linear projections produce `left` and `right` tensors (shape [B,N,N,H]).
3. Compute three gate tensors (`left_gate`, `right_gate`, `out_gate`) with a
   sigmoid activation.
4. Fuse element‑wise multiplication of `left` with `left_gate`,
   `right` with `right_gate`, and optional masking in a single Triton
   kernel (the only required Triton component).
5. Perform the O(N³) contraction as a batched matrix multiplication:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
   This is realised by `torch.matmul(left_perm, right_perm.T)`.
6. Apply a second LayerNorm, scale by `out_gate`, and project back to the
   original `dim` with a final linear layer.
7. Return the result in float32.

All heavy arithmetic is performed in fp16; only the final output is cast
back to fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # fp16 *
    right_ptr,         # fp16 *
    left_gate_ptr,     # fp16 *
    right_gate_ptr,    # fp16 *
    mask_ptr,          # fp16 *
    total_elems,       # int64
    hidden_dim,        # int64
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elems

    # Load tensors
    left = tl.load(left_ptr + offs, mask=active, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=active, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=tl.float16(0.0))

    if HAS_MASK:
        # One mask value per (b,i,j) → repeat hidden_dim times
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=tl.float16(1.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (cuda, float32)
            - mask_tensor:  torch.Tensor [B, N, N] (bool/float) or None
            - weights: dictionary containing all linear/LayerNorm weights
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Cast weights to fp16 (single conversion per call)
    # ------------------------------------------------------------------
    w_norm = weights["norm.weight"].to(device=device, dtype=torch.float16)
    b_norm = weights["norm.bias"].to(device=device, dtype=torch.float16)

    w_left_proj = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    w_right_proj = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    w_left_gate = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    w_right_gate = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    w_out_gate = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    w_to_out_norm = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)

    w_to_out = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 2. Input LayerNorm (fp16) and linear projections
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor.to(torch.float16),
        normalized_shape=(dim,),
        weight=w_norm,
        bias=b_norm,
    )  # (B, N, N, dim), fp16

    left = F.linear(x, w_left_proj)    # (B, N, N, hidden_dim)
    right = F.linear(x, w_right_proj)  # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3. Compute gate tensors (sigmoid, fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, w_left_gate))   # (B, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, w_right_gate)) # (B, N, N, hidden_dim)
    out_gate = torch.sigmoid(F.linear(x, w_out_gate))    # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4. Fuse gates (and optional mask) with a Triton kernel
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    total = left.numel()
    BLOCK = 65536  # 64K threads per launch (must be multiple of 32)
    grid = (triton.cdiv(total, BLOCK),)

    if not nomask and mask is not None:
        # mask: [B, N, N] → fp16 and broadcast over hidden_dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1).contiguous()
        HAS_MASK = 1
        _gate_mask_fuse_kernel[grid](
            left,
            right,
            left_gate,
            right_gate,
            mask_fp16,
            total,
            hidden_dim,
            HAS_MASK=HAS_MASK,
            BLOCK=BLOCK,
        )
    else:
        # No mask – still use the kernel to fuse the two gates
        HAS_MASK = 0
        # dummy mask pointer (won't be read because HAS_MASK==0)
        _gate_mask_fuse_kernel[grid](
            left,
            right,
            left_gate,
            right_gate,
            left,            # dummy
            total,
            hidden_dim,
            HAS_MASK=HAS_MASK,
            BLOCK=BLOCK,
        )

    # Gates are no longer needed
    del left_gate, right_gate, x

    # ------------------------------------------------------------------
    # 5. Core O(N³) contraction via batched matmul
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Permute to bring hidden_dim into batch dimension
    left_perm = left.permute(0, 3, 1, 2)   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2) # (B, H, N, N)

    # Batched matrix multiplication: (B*H, N, N) @ (B*H, N, N)^T
    out_perm = torch.matmul(left_perm, right_perm.transpose(-2, -1))  # (B, H, N, N)

    # Transpose back to original layout
    out = out_perm.permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
    )
    out = out * out_gate
    out = F.linear(out, w_to_out)   # (B, N, N, dim), fp16

    # ------------------------------------------------------------------
    # 7. Cast back to float32 for the final result
    # ------------------------------------------------------------------
    return out.to(torch.float32)
