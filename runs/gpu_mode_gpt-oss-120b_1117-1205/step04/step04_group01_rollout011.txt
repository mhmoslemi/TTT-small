@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask)
    right = tl.load(right_ptr + offs, mask=mask)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask)

    if HAS_MASK:
        # mask index corresponds to element // H (since mask broadcast over hidden dimension)
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    # Input layernorm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Convert to half
    x = x.to(torch.float16)
    # Compute left/right
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))
    # Compute gates
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()
    # Fuse mask and gates
    if mask is not None and not nomask:
        mask_h = mask.to(torch.float16).unsqueeze(-1)
        _apply_gates_and_mask(left, right, left_gate, right_gate, mask_h, hidden_dim)
    else:
        left = left * left_gate
        right = right * right_gate
    # Contract
    B, N, N2, H = left.shape
    assert N == N2
    left_ = left.permute(0, 3, 1, 2).contiguous()
    right_ = right.permute(0, 3, 2, 1).contiguous()
    left_flat = left_.view(B * H, N, N)
    right_flat = right_.view(B * H, N, N)
    out = torch.bmm(left_flat, right_flat)
    out = out.view(B, H, N, N).permute(0, 2, 3, 1)  # shape [B,N,N,H]
    # output layernorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

"""
TriMul “outgoing” forward pass with Triton fusion.

The kernel fuses the element‑wise multiplication of the two projected
tensors with their sigmoid gates and (optionally) a mask.  The heavy
O(N³·H) contraction is performed as a batched matrix multiplication
using torch.bmm on half‑precision data, which is much faster than the
reference einsum.  The rest of the operations (LayerNorms, linear
projections and final gating) are done in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask (if present) with the left/right gates.
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half *
    right_ptr,         # half *
    left_gate_ptr,     # half *
    right_gate_ptr,    # half *
    mask_ptr,          # half *
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask is present, else 0
    H: tl.constexpr,          # hidden dimension (size of innermost channel)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elems

    # Load operands
    left      = tl.load(left_ptr + offs,      mask=valid, other=0.0)
    right     = tl.load(right_ptr + offs,     mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate= tl.load(right_gate_ptr + offs,mask=valid, other=0.0)

    if HAS_MASK:
        # One mask value corresponds to H consecutive channel entries.
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=0.0)
        left  = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left * left_gate
        right = right * right_gate

    # Store the fused results back in‑place
    tl.store(left_ptr  + offs, left,  mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """In‑place fusion of left/right with their gates and optional mask."""
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when no mask

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # left and right are modified in‑place


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask: torch.Tensor [B, N, N] (bool or float) or None
            - weights: dict of parameter tensors
            - config: dict with keys ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns:
        torch.Tensor [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> half)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half‑precision)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )  # [B, N, N, hidden_dim]
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gate values (half‑precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) and gates via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape: [B, N, N] → broadcast over hidden_dim
        mask_h = mask.to(torch.float16).unsqueeze(-1).contiguous()
        _apply_gate_and_mask(left, right, left_gate, right_gate, mask_h, hidden_dim)
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM (B*H separate N×N matmuls).
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # left: [B,N,N,H] → [B*H, N, N]
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # right: [B,N,N,H] → [B*H, N, N] after transposing k↔j
    right_ = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    out = torch.bmm(left_, right_)                # [B*H, N, N]
    out = out.view(B, H, N, N).permute(0, 2, 3, 1)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in the requested precision
    return out.to(torch.float32)
