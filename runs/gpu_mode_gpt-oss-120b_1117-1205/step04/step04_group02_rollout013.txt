"""
Optimized forward pass for the outgoing TriMul operator.

Algorithm:
1. Input LayerNorm (float32) → half.
2. Linear projections to hidden_dim (half).
3. Compute three sigmoid gates (half).
4. Fuse mask (if present) and gates with a Triton kernel (in‑place element‑wise).
5. Core O(N³) contraction: for each hidden channel perform a batched matrix
   multiplication L @ Rᵀ via torch.bmm (leverages cuBLAS).
6. Output LayerNorm, out‑gate scaling, final linear projection.
   Result returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel for mask+gate fusion
@triton.jit
def gate_mask_fuse_kernel(
    left_ptr,          # * half
    right_ptr,         # * half
    left_gate_ptr,     # * half
    right_gate_ptr,    # * half
    mask_ptr,          # * half (broadcasted)
    total_elements,    # i32
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=valid, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=valid, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=tl.float16(0.0))

    if HAS_MASK:
        # Broadcast mask over hidden dimension.
        mask_idx = offs // HIDDEN_DIM
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=tl.float16(1.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store results (in‑place)
    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of left/right with their corresponding gates and
    optionally a mask. All tensors must be contiguous and of dtype float16.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    # Triton requires a pointer even when HAS_MASK==0; a dummy pointer is fine.
    mask_ptr = mask if mask is not None else left

    gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        HIDDEN_DIM=hidden_dim,
    )
    # No return – operation is performed in‑place.


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool / float) or None
        - weights      : dict of torch.Tensors (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Cast weight tensors to half‑precision once (avoid repeated .to calls)
    # ------------------------------------------------------------------
    # Linear projection weights
    left_proj_w = weights["left_proj.weight"].to(torch.float16, non_blocking=True)
    right_proj_w = weights["right_proj.weight"].to(torch.float16, non_blocking=True)

    # Gate weights
    left_gate_w = weights["left_gate.weight"].to(torch.float16, non_blocking=True)
    right_gate_w = weights["right_gate.weight"].to(torch.float16, non_blocking=True)
    out_gate_w = weights["out_gate.weight"].to(torch.float16, non_blocking=True)

    # Output layer‑norm parameters
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16, non_blocking=True)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16, non_blocking=True)

    # Final linear weight
    to_out_w = weights["to_out.weight"].to(torch.float16, non_blocking=True)

    # Input layer‑norm parameters (float32; keep in original dtype)
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half)
    # ------------------------------------------------------------------
    left = F.linear(x_h, left_proj_w, bias=None)    # [B, N, N, hidden_dim]
    right = F.linear(x_h, right_proj_w, bias=None)

    # ------------------------------------------------------------------
    # 3. Compute gates (half) – sigmoid applied element‑wise
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, left_gate_w, bias=None).sigmoid()
    right_gate = F.linear(x_h, right_gate_w, bias=None).sigmoid()
    out_gate = F.linear(x_h, out_gate_w, bias=None).sigmoid()  # [B, N, N, hidden_dim]

    # 4. Fuse mask (if present) with gates via Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:

        mask_h = mask.to(torch.float16, non_blocking=True).contiguous().view(-1)

        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)

        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_h,
            hidden_dim,
        )
        # Reshape back to original dimensions
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask – simple in‑place gating
        left = left * left_gate
        right = right * right_gate

    # Discard gate tensors that are no longer needed
    del left_gate, right_gate, x_h

    # ------------------------------------------------------------------
    # 5. Core O(N³) contraction via batched GEMM (half precision)
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden_dim, N, N) for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    # Collapse batch and hidden dimensions to feed torch.bmm
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # Batched matrix multiplication: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore original layout (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate  # element‑wise scaling (half)
    out = F.linear(out, to_out_w, bias=None)  # back to dim

    # Return result in float32 as expected by the rest of the model
    return out.to(torch.float32)

"""
Optimized forward pass for the outgoing TriMul operator (AlphaFold3).

Algorithm
---------
1. Input LayerNorm (float32) → half precision.
2. Linear projections to `hidden_dim` (half).
3. Compute three sigmoid gates (half).
4. Fuse mask (if present) and the per‑position gates with a small Triton kernel.
   The kernel performs the in‑place element‑wise multiplication:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
   (when no mask is supplied only the gate multiplication is done).
5. Core O(N³) contraction:
   out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   Implemented as a batched GEMM (torch.bmm) on half tensors:
   for each hidden channel d we compute L_d @ R_dᵀ.
6. Output LayerNorm, multiply by `out_gate`, final linear projection.
   Result is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: gate + optional mask fusion (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_fuse_kernel(
    left_ptr,          # * half
    right_ptr,         # * half
    left_gate_ptr,     # * half
    right_gate_ptr,    # * half
    mask_ptr,          # * half (broadcasted)
    total_elements,    # i32
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    left = tl.load(left_ptr + offs, mask=valid, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=valid, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=tl.float16(0.0))

    if HAS_MASK:
        # Broadcast mask over the hidden dimension.
        mask_idx = offs // HIDDEN_DIM
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=tl.float16(1.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of left/right with their corresponding gates and,
    optionally, a mask. All tensors must be contiguous and of dtype float16.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    # Triton requires a pointer even when HAS_MASK==0; a dummy pointer works.
    mask_ptr = mask if mask is not None else left

    gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        HIDDEN_DIM=hidden_dim,
    )
    # No return – the operation is performed in‑place.

# Entry point

    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
