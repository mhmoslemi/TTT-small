"""
TriMul outgoing kernel (AlphaFold3)

Implements the forward pass of the Triangle Multiplicative Update (outgoing)
using mixed‑precision Torch + Triton.  The heavy element‑wise mask & gate
fusion is performed by a small Triton kernel; the remaining arithmetic
(including the O(N³) contraction) is done with half‑precision batched GEMM
using torch.bmm.  The final result is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # half* flattened left tensor
    right_ptr,           # half* flattened right tensor
    left_gate_ptr,       # half* flattened left‑gate tensor
    right_gate_ptr,      # half* flattened right‑gate tensor
    mask_ptr,            # half* flattened mask (broadcast over hidden dim)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present, else 0
    H: tl.constexpr,          # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # one mask value per (batch, i, k) group spanning H hidden values
        mask_offs = offs // H
        mask_val = tl.load(mask_ptr + mask_offs, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place Triton‑fused gating (and optional masking).
    All tensors must be contiguous torch.float16.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when no mask

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # tensors are modified in place


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32, on CUDA)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (float32, may be on CPU)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32, on same device as input)
    """
    # Unpack arguments (mask may be omitted)
    if len(data) == 4:
        input_tensor, mask, weights, config = data
    else:
        input_tensor, weights, config = data
        mask = None

    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half precision, no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"].to(device, dtype=torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device, dtype=torch.float16)

    left = F.linear(x_h, left_proj_weight, bias=None)   # [B, N, N, H]
    right = F.linear(x_h, right_proj_weight, bias=None)

    # ------------------------------------------------------------------
    # 3. Gates (half precision, sigmoid)
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"].to(device, dtype=torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device, dtype=torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device, dtype=torch.float16)

    left_gate = F.linear(x_h, left_gate_weight, bias=None).sigmoid()
    right_gate = F.linear(x_h, right_gate_weight, bias=None).sigmoid()
    out_gate = F.linear(x_h, out_gate_weight, bias=None).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) with gates via Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).contiguous().view(-1)                # [B*N*N]
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(left_f, right_f, left_gate_f, right_gate_f, mask_h, hidden_dim)

        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask: simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched GEMM (torch.bmm) over the hidden dimension.
    # ------------------------------------------------------------------
    # Bring hidden dimension to the second axis for batched matmul:
    # left: (B, N, N, H) → (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)               # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)             # (B*H, N, N)

    # Batched matrix multiplication with transpose on the second operand
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype=torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device, dtype=torch.float16)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    to_out_weight = weights["to_out.weight"].to(device, dtype=torch.float16)
    out = F.linear(out, to_out_weight, bias=None)

    # Cast back to float32 for the final output
    return out.to(torch.float32)
```
