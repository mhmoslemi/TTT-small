def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    batch_size, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # Cast to half
    x = x.to(torch.float16)

    # Linear projections
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16), bias=None)

    # Gates
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # Fuse mask and gate using Triton
    if not nomask:
        # flatten for kernel
        left_f = left.view(-1)
        right_f = right.view(-1)
        left_gate_f = left_gate.view(-1)
        right_gate_f = right_gate.view(-1)
        mask_f = mask.to(torch.float16).view(-1)  # shape (B N N)
        # expand mask to match hidden_dim
        # Since mask_f repeats over hidden_dim in the kernel using division.
        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            hidden_dim,
        )
        # reshape
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # At this point left, right are half tensors shape [B,N,N,hidden_dim]

    # Rearrange for batched GEMM
    # Permute to [B, hidden_dim, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    batch_sz = batch_size * hidden_dim

    # Flatten batch
    left_mat = left_perm.view(batch_sz, N, N)
    right_mat_T = right_perm.permute(0, 1, 3, 2).contiguous().view(batch_sz, N, N)

    # Output buffer
    out_mat = torch.empty_like(left_mat)

    # Strides
    stride_am = left_mat.stride(1)  # M stride (N)
    stride_ak = left_mat.stride(2)  # K stride (1)
    stride_ab = left_mat.stride(0)  # batch stride
    stride_bk = right_mat_T.stride(1)  # K stride (N)
    stride_bn = right_mat_T.stride(2)  # N stride (1)
    stride_bb = right_mat_T.stride(0)  # batch stride
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)
    stride_cb = out_mat.stride(0)

    # Grid for Triton GEMM
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid_x = triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N)
    grid = (grid_x, batch_sz)
    batched_gemm_kernel[grid](
        left_mat,
        right_mat_T,
        out_mat,
        N, N, N,
        stride_am,
        stride_ak,
        stride_ab,
        stride_bk,
        stride_bn,
        stride_bb,
        stride_cm,
        stride_cn,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_stages=3,
        num_warps=8,
    )

    # Reshape back
    out = out_mat.view(batch_size, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # To_out_norm (LayerNorm over hidden_dim)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # elementwise

    # Final linear projection to dim
    out = F.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    # Return float32
    return out.to(torch.float32)

```python
@triton.jit
def batched_gemm_kernel(
    A,
    B,
    C,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    NUM_STAGES: tl.constexpr,
    NUM_WARPS: tl.constexpr,
):
    pid = tl.program_id(0)
    batch = tl.program_id(1)

    # compute block coordinates
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # offsets within block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Create a mask for valid rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute offsets for K block
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A block
        a_ptrs = A + batch * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B block (B is already transposed: shape (batch, K, N))
        b_ptrs = B + batch * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Compute
        acc += tl.dot(a, b)  # both a,b are (BLOCK_M, BLOCK_K) and (BLOCK_K, BLOCK_N) respectively

    # Write back results
    c_ptrs = C + batch * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    # Cast acc to half for storing
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])
```

"""
TriMul (outgoing) forward pass with fused Triton kernels.

Algorithm:
1. LayerNorm on the input (float32 → half).
2. Linear projections to hidden_dim (half).
3. Compute left/right gates (sigmoid) and optionally fuse a mask.
   A Triton kernel fuses left/right projection, gate, and mask
   (in‑place multiplication).
4. Core contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
   This is a batched GEMM on the (i,k) and (k,j) matrices for each
   (b,d) pair.  A custom Triton batched‑gemm kernel is used.
5. LayerNorm over hidden_dim, out‑gate scaling, final linear projection
   back to dim.
All heavy compute is performed in FP16; the final output is cast to FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: fuse gate and (optional) mask multiplication.
# --------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half*  (flattened left tensor)
    right_ptr,         # half*  (flattened right tensor)
    left_gate_ptr,     # half*  (flattened left gate)
    right_gate_ptr,    # half*  (flattened right gate)
    mask_ptr,          # half*  (flattened mask, broadcasted over hidden dim)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 -> mask present, 0 -> no mask
    H: tl.constexpr,          # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each hidden channel shares the same mask entry
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # store results back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """In‑place fusion of left/right * gate * (optional) mask."""
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy ptr when unused

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # No Python return – operation is performed in‑place


# --------------------------------------------------------------
# Triton kernel: batched matrix multiplication (A @ Bᵀ) for many
# batches (batch = B * hidden_dim).  A shape (batch, M, K), B shape
# (batch, K, N) (already transposed), output (batch, M, N).
# --------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A, B, C,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    NUM_STAGES: tl.constexpr,
):
    pid = tl.program_id(0)            # tile id for (M,N)
    batch = tl.program_id(1)          # which batch (B * hidden_dim)

    # block coordinates
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A block: (M, K)
        a_ptrs = A + batch * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B block: (K, N) – already transposed
        b_ptrs = B + batch * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K) • (BLOCK_K, BLOCK_N)

    # write results
    c_ptrs = C + batch * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------
# Entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float16

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(dtype)

    # --------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half)
    # --------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(dtype),
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(dtype),
        bias=None,
    )

    # --------------------------------------------------------------
    # 3. Gates (half)
    # --------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(dtype),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(dtype),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(dtype),
        bias=None,
    ).sigmoid()

    # --------------------------------------------------------------
    # 4. Fuse mask with gates using Triton (if needed)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # flatten tensors for the kernel
        left_f = left.view(-1).contiguous()
        right_f = right.view(-1).contiguous()
        left_gate_f = left_gate.view(-1).contiguous()
        right_gate_f = right_gate.view(-1).contiguous()
        mask_f = mask.to(dtype).contiguous().view(-1)   # [B, N, N]

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            hidden_dim,
        )
        # reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # simple element‑wise multiplication (no mask)
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as a batched GEMM (A @ Bᵀ) using Triton.
    # --------------------------------------------------------------
    # permute to (B, hidden_dim, N, N) -> (B*H, N, N) for GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    batch_sz = B * H

    # reshape to (batch_sz, N, N)
    A_mat = left_perm.view(batch_sz, N, N)
    B_mat_T = right_perm.permute(0, 1, 3, 2).contiguous().view(batch_sz, N, N)  # transposed on last two axes

    # output buffer
    C_mat = torch.empty_like(A_mat)

    # strides (all tensors are contiguous)
    stride_am = A_mat.stride(1)      # = K
    stride_ak = A_mat.stride(2)      # = 1
    stride_ab = A_mat.stride(0)      # = M*K

    stride_bk = B_mat_T.stride(1)    # = N (since shape (batch, K, N))
    stride_bn = B_mat_T.stride(2)    # = 1
    stride_bb = B_mat_T.stride(0)    # = K*N

    stride_cm = C_mat.stride(1)      # = N
    stride_cn = C_mat.stride(2)      # = 1
    stride_cb = C_mat.stride(0)      # = M*N

    # launch parameters
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid_x = triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N)
    grid = (grid_x, batch_sz)

    _batched_gemm_kernel[grid](
        A_mat,
        B_mat_T,
        C_mat,
        N, N, N,
        stride_am,
        stride_ak,
        stride_ab,
        stride_bk,
        stride_bn,
        stride_bb,
        stride_cm,
        stride_cn,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        NUM_STAGES=3,
        num_warps=8,
    )

    # reshape back to (B, N, N, hidden_dim)
    out = C_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate

    out = F.linear(
        out,
        weights["to_out.weight"].to(dtype),
        bias=None,
    )

    # return in original dtype (float32)
    return out.to(torch.float32)
