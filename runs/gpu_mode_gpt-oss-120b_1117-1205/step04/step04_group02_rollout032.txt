@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    seq_len, hidden_dim,
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_j, stride_right_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # compute batch idx
    pid = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    hidden = hidden_dim  # constant argument? Actually hidden_dim is passed, not constexpr
    # Compute b, h:
    b = pid // hidden
    h = pid % hidden

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < seq_len
    mask_n = offs_n < seq_len

    # base pointer offset for this (b,h)
    off_batch_left = b * stride_left_b + h * stride_left_h
    off_batch_right = b * stride_right_b + h * stride_right_h
    off_batch_out = b * stride_out_b + h * stride_out_h

    # initialization
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(seq_len, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < seq_len

        # ptrs for left block
        left_ptrs = left_ptr + off_batch_left + \
            offs_m[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k

        # ptrs for right block (k, j)
        right_ptrs = right_ptr + off_batch_right + \
            offs_k[:, None] * stride_right_k + \
            offs_n[None, :] * stride_right_j

        a = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=tl.float16(0.0))
        b_mat = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=tl.float16(0.0))

        # cast to float32
        a_fp32 = tl.cast(a, tl.float32)
        b_fp32 = tl.cast(b_mat, tl.float32)

        acc += tl.dot(a_fp32, b_fp32)

    # store result
    out_ptrs = out_ptr + off_batch_out + \
        offs_m[:, None] * stride_out_i + \
        offs_n[None, :] * stride_out_j

    tl.store(out_ptrs, tl.cast(acc, tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
Outgoing TriMul (AlphaFold3) – Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32) and cast to float16.
2. Linear projections → left/right tensors (half‑precision).
3. Compute three sigmoid gates (half‑precision).
4. Fuse mask (if present) with the left/right gates in‑place using a small
   Triton kernel (element‑wise multiplication, broadcast mask over the hidden
   channel dimension).
5. Core O(N³) contraction:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a custom batched‑GEMM Triton kernel that processes each
   (batch, hidden) pair as an independent matrix multiplication.
6. Output LayerNorm (half), scale by out_gate and final linear projection.
7. Return the result in float32.

The kernel in step 4 is required (ensures “part of the operations in a kernel”).
Step 5 is also in Triton for better performance on large N.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 4. Gate + (optional) mask fusion kernel
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_fuse_kernel(
    left_ptr,          # * half
    right_ptr,         # * half
    left_gate_ptr,     # * half
    right_gate_ptr,    # * half
    mask_ptr,          # * half (broadcasted)
    total_elems,       # int32
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elems

    # load
    left = tl.load(left_ptr + offs, mask=valid, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=valid, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=tl.float16(0.0))

    if HAS_MASK:
        # each mask element repeats HIDDEN_DIM times in the flattened tensor
        mask_idx = offs // HIDDEN_DIM
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=tl.float16(1.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # store back in‑place
    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    `mask` may be None (no masking). All tensors must be contiguous and half‑precision.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when HAS_MASK==0

    gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        HIDDEN_DIM=hidden_dim,
    )
    # operation performed in‑place


# ----------------------------------------------------------------------
# 5. Core contraction kernel (batched GEMM per hidden channel)
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    seq_len,                     # N
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_j, stride_right_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):
    pid = tl.program_id(0)   # combined batch * hidden index
    pid_m = tl.program_id(1) # i‑tile
    pid_n = tl.program_id(2) # j‑tile

    # recover batch and hidden indices
    b = pid // HIDDEN_DIM
    h = pid % HIDDEN_DIM

    # offsets within the i/j tiles
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < seq_len
    mask_j = offs_j < seq_len

    # base pointer for this (b, h) pair
    off_batch_left = b * stride_left_b + h * stride_left_h
    off_batch_right = b * stride_right_b + h * stride_right_h
    off_batch_out = b * stride_out_b + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension in tiles
    for k in range(0, tl.cdiv(seq_len, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < seq_len

        # left block: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + off_batch_left + \
            offs_i[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k

        # right block (transposed): shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + off_batch_right + \
            offs_k[:, None] * stride_right_k + \
            offs_j[None, :] * stride_right_j

        a = tl.load(left_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=tl.float16(0.0))
        b_mat = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=tl.float16(0.0))

        # accumulate in float32 for better precision
        acc += tl.dot(tl.cast(a, tl.float32), tl.cast(b_mat, tl.float32))

    # store result (cast back to half)
    out_ptrs = out_ptr + off_batch_out + \
        offs_i[:, None] * stride_out_i + \
        offs_j[None, :] * stride_out_j

    tl.store(out_ptrs,
             tl.cast(acc, tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of torch.Tensors (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half)
    # ------------------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gates (half)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) with gates via Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).contiguous().view(-1)  # [B*N*N]
        # flatten tensors for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_h,
            hidden_dim,
        )
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # simple element‑wise gating when no mask
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via custom Triton batched GEMM
    #    out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B, H, N, N) for convenient batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # allocate output tensor
    out_perm = torch.empty_like(left_perm)

    # strides (in elements)
    sL_b, sL_h, sL_i, sL_k = left_perm.stride()
    sR_b, sR_h, sR_j, sR_k = right_perm.stride()
    sO_b, sO_h, sO_i, sO_j = out_perm.stride()

    # Triton block sizes – tuned for B200
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    trimul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        N,
        sL_b, sL_h, sL_i, sL_k,
        sR_b, sR_h, sR_j, sR_k,
        sO_b, sO_h, sO_i, sO_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HIDDEN_DIM=hidden_dim,
    )

    # bring back to original layout [B, N, N, H]
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # return in original dtype
    return out.to(torch.float32)
