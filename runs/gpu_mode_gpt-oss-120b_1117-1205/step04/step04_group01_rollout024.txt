@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute starting indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load a block (M x K)
        a_ptrs = a_ptr + batch_id * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0)

        # Load b block (K x N)
        b_ptrs = b_ptr + batch_id * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Write out
    c_ptrs = c_ptr + batch_id * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))
```

    left = F.linear(x, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16), bias=None)

    # Gates
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    if not nomask and mask is not None:
        mask = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask * left_gate
        right = right * mask * right_gate
    else:
        left = left * left_gate
        right = right * right_gate

    # Prepare for batched matmul
    B, N, _, _ = left.shape
    # Permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()
    # Reshape to (B*H, N, N)
    left_reshaped = left_perm.view(B * hidden_dim, N, N)
    right_reshaped = right_perm.view(B * hidden_dim, N, N)  # treat as (B*H, K, N)

    # Allocate output
    out_reshaped = torch.empty_like(left_reshaped)  # same dtype half

    # Get strides
    stride_a_batch, stride_a_m, stride_a_k = left_reshaped.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_reshaped.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_reshaped.stride()

    # Define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_matmul_kernel[grid](
        left_reshaped,
        right_reshaped,
        out_reshaped,
        N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back
    out = out_reshaped.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B,N,N,H)

    # Output layer norm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate

    # Final linear projection
    out = F.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    # Return float32
    return out.to(torch.float32)

def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_aba, stride_ama, stride_ak,
    stride_bba, stride_bka, stride_bna,
    stride_cca, stride_cma, stride_cna,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)

        # Load A
        a_ptrs = a_ptr + batch_id * stride_aba + offs_m[:, None] * stride_ama + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
                    other=0.0)

        # Load B
        b_ptrs = b_ptr + batch_id * stride_bba + offs_k[:, None] * stride_bka + offs_n[None, :] * stride_bna
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0)

        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_ptr + batch_id * stride_cca + offs_m[:, None] * stride_cma + offs_n[None, :] * stride_cna
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c,
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

"""
TriMul "outgoing" forward pass with a fused Triton batched matmul.

Algorithm
---------
1. Layer‑norm the input tensor (float32 → float16).
2. Compute left/right projections and three gates with linear layers
   (all in float16).
3. Apply optional mask and element‑wise gate multiplication.
4. Rearrange tensors to (B·H, N, N) where H = hidden_dim and launch a
   Triton kernel that performs a batched matrix multiplication:
        out[b,i,j] = Σ_k left[b,i,k] * right[b,k,j]
   The kernel tiles the M×N output using BLOCK_M×BLOCK_N blocks and
   accumulates the inner‑product over K in float32.
5. Reshape the result back to (B, N, N, H).
6. Apply a second LayerNorm, out‑gate scaling and the final linear
   projection to the original dimension (float16 → float32).

Only the O(N³) contraction is implemented in Triton; all other
operations use PyTorch (highly optimized and negligible compared to
the matrix multiplication).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # problem size
    stride_a_batch, stride_a_m, stride_a_k,   # strides for A
    stride_b_batch, stride_b_k, stride_b_n,   # strides for B
    stride_c_batch, stride_c_m, stride_c_n,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds loads/stores
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K sized chunks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A: [M x K]
        a_ptrs = (
            a_ptr
            + batch_id * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B: [K x N]
        b_ptrs = (
            b_ptr
            + batch_id * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Write back result in FP16
    c_ptrs = (
        c_ptr
        + batch_id * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (float32/bool) or None
            - weights: dict of parameter tensors
            - config: dict with keys "dim" and "hidden_dim" and optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> float16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half-precision)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )  # [B, N, N, hidden_dim]
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gate values (half-precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Apply mask (if any) and gates
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] -> broadcast over hidden_dim
        mask_h = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_h * left_gate
        right = right * mask_h * right_gate
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³) contraction via Triton batched matmul
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # Bring hidden dimension to the batch axis for efficient matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()        # [B, H, N, N] (j/k swapped)

    # Merge batch & hidden dim
    left_reshaped = left_perm.view(B * hidden_dim, N, N)      # [B*H, M, K]
    right_reshaped = right_perm.view(B * hidden_dim, N, N)    # [B*H, K, N]
    out_reshaped = torch.empty_like(left_reshaped)            # allocate output

    # Strides (in elements)
    stride_a_batch, stride_a_m, stride_a_k = left_reshaped.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_reshaped.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_reshaped.stride()

    # Tiling configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _batched_matmul_kernel[grid](
        left_reshaped,
        right_reshaped,
        out_reshaped,
        N, N, N,                     # M, N, K
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = out_reshaped.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in the requested precision
    return out.to(torch.float32)
