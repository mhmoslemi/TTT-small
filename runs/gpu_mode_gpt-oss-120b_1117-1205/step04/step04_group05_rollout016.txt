    pid_m = tl.program_id(1)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)

"""
TriMul outgoing kernel (AlphaFold3) – fast Triton implementation.

Algorithm
---------
1. Layer‑normalize the input tensor (float32) and cast to float16.
2. Linear projections to hidden_dim (no bias) → left, right.
3. Compute three sigmoid gates (left_gate, right_gate, out_gate).
4. Apply mask (if present) and element‑wise gating:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
   (mask is broadcast over the hidden dimension).
5. Core contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] .
   This is a batched GEMM: for each (b,d) we compute
        out_{b,d} = left_{b,d} @ right_{b,d}.T .
   Implemented with a custom Triton kernel that fuses the
   batch dimension, uses float32 accumulation and stores the
   result in float16.
6. Output LayerNorm over the hidden dimension, scaling by out_gate,
   final linear projection back to the original dimension,
   and cast the result back to float32.

The kernel only covers the O(N³) contraction; the remaining
operations stay in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    batch, M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: C = A @ B.T   (A,B shape = [batch, M, K] , [batch, K, N])"""
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets within the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_ptr = A_ptr + pid_batch * stride_Ab
    b_ptr = B_ptr + pid_batch * stride_Bb
    c_ptr = C_ptr + pid_batch * stride_Cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr + (offs_m[:, None] * stride_Am) + (offs_k[None, :] * stride_Ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B tile: (BLOCK_K, BLOCK_N) – B already transposed (K x N)
        b = tl.load(
            b_ptr + (offs_k[:, None] * stride_Bk) + (offs_n[None, :] * stride_Bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

    # write back
    c = acc.to(tl.float16)
    tl.store(
        c_ptr + (offs_m[:, None] * stride_Cm) + (offs_n[None, :] * stride_Cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )

    ).sigmoid()
    out_gate = F.linear(
        x, weights["out_gate.weight"].to(torch.float16), bias=None
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Apply mask (if any) and gating
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # broadcast mask over hidden dimension
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_h * left_gate
        right = right * mask_h * right_gate
    else:
        left = left * left_gate
        right = right * right_gate

    # free intermediate tensors that are no longer needed
    del left_gate, right_gate, x

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton batched GEMM
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # N = seq_len

    # bring hidden dimension to the second axis and make matrices contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # transpose right for the GEMM (K × N)
    right_T = right_perm.transpose(2, 3).contiguous()   # (B, H, N, N) where last two axes are swapped

    # merge batch and hidden dimension
    batch = B * hidden_dim
    left_mat = left_perm.view(batch, N, N)   # (B*H, N, N)
    right_mat = right_T.view(batch, N, N)    # (B*H, N, N)

    # allocate output
    out_mat = torch.empty_like(left_mat)  # half precision

    # strides (batch, row, col)
    stride_Ab, stride_Am, stride_Ak = left_mat.stride()
    stride_Bb, stride_Bk, stride_Bn = right_mat.stride()
    stride_Cb, stride_Cm, stride_Cn = out_mat.stride()

    # tile sizes – 64 is a good trade‑off for the tested seq lengths
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch,
        N,
        N,
        N,
        stride_Ab,
        stride_Am,
        stride_Ak,
        stride_Bb,
        stride_Bk,
        stride_Bn,
        stride_Cb,
        stride_Cm,
        stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to the original 4‑D layout
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gate scaling and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out, weights["to_out.weight"].to(torch.float16), bias=None
    )

    # cast back to float32 for the final output
    return out.to(torch.float32)
