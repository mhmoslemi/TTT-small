import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. Input layer norm (float32)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to half for subsequent ops
    x_h = x.to(torch.float16)

    # 2. Linear projections
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left = F.linear(x_h, left_proj_w, bias=None)
    right = F.linear(x_h, right_proj_w, bias=None)

    # 3. Gates
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_h, left_gate_w, bias=None).sigmoid()
    right_gate = F.linear(x_h, right_gate_w, bias=None).sigmoid()
    out_gate = F.linear(x_h, out_gate_w, bias=None).sigmoid()

    # 4. Fuse gates and mask via Triton kernel
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).contiguous()
        # flatten mask to 1D (B,N,N)
        mask_f = mask_h.view(-1)
        # flatten tensors
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _fuse_gate_mask(left_f, right_f, left_gate_f, right_gate_f, mask_f, hidden_dim)
        # reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask: elementwise multiplication
        left = left * left_gate
        right = right * right_gate

    # 5. Core contraction (batched GEMM)
    # Permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)
    # compute out = left_mat @ right_mat^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 6. Output LayerNorm (half)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # 7. Apply out_gate
    out = out * out_gate

    # 8. Final linear projection to dim
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w, bias=None)

    # Return float32
    return out.to(torch.float32)

"""
custom_kernel: outgoing TriMul (AlphaFold3) forward pass.

Algorithm
---------
1. LayerNorm over the input channels (dim) – float32 for stability.
2. Cast to float16 and project to hidden_dim with two linear layers:
   left = x·W_left , right = x·W_right.
3. Compute three sigmoid gates (left, right, out) from the same input.
4. Fuse element‑wise mask (if present) with left/right gates in a single
   Triton kernel:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
   (mask is broadcast over the hidden dimension).
5. Core contraction:
        out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   This is a batched GEMM for each head h:
        left_{b,h} (N×N) @ right_{b,h}ᵀ (N×N) → out_{b,h} (N×N).
6. LayerNorm over hidden_dim, scale by out_gate, and final linear
   projection back to dim.
7. Return result in float32.

The heavy element‑wise mask & gate fusion is executed on GPU via Triton,
while the N³ contraction uses cuBLAS‑accelerated batched matrix multiplication.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # float16 *
    right_ptr,         # float16 *
    left_gate_ptr,     # float16 *
    right_gate_ptr,    # float16 *
    mask_ptr,          # float16 * (broadcasted)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 = mask present, 0 = no mask
    H: tl.constexpr,          # hidden dimension size
):
    """In‑place fusion of mask and gating:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # mask is indexed by (b,i,k) ⇒ flatten offset // H
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def _fuse_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):

    Fuse left/right with their gates and (optionally) a mask in-place using Triton.
    All tensors must be contiguous, dtype torch.float16.

        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # No return – operation performed in-place

    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to half for the rest of the pipeline
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half precision)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w, bias=None)   # [B, N, N, H]
    right = F.linear(x_h, right_proj_w, bias=None)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Gates (half) with sigmoid
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_h, left_gate_w, bias=None).sigmoid()
    right_gate = F.linear(x_h, right_gate_w, bias=None).sigmoid()
    out_gate = F.linear(x_h, out_gate_w, bias=None).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) with gates via Triton kernel
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → half, contiguous, flattened
        mask_h = mask.to(torch.float16).contiguous()
        mask_f = mask_h.view(-1)  # (B*N*N,)

        # Flatten tensors for the kernel (they share storage with originals)
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _fuse_gate_and_mask(left_f, right_f, left_gate_f, right_gate_f, mask_f, hidden_dim)

        # Reshape back (views keep modified data)
        left = left_f.view_as(left)
        right = right_f.view_as(right)
        # Gates are no longer needed
        del left_gate_f, right_gate_f, mask_f, left_f, right_f, mask_h
    else:
        # No mask – simple element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate

    # Gates no longer needed
    del left_gate, right_gate, x_h

    # ------------------------------------------------------------------
    # 5. Core contraction via batched GEMM
    #    out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) for easy batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)    # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)  # (B*H, N, N)

    # Batched matrix multiplication: left * rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore original layout [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out_gate scaling, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate  # element‑wise scaling (half)

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w, bias=None)   # back to [B, N, N, dim]

    # Return in float32 as required by the interface
    return out.to(torch.float32)
