def trimul_outgoing_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B: tl.constexpr, N: tl.constexpr, H: tl.constexpr,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    NUM_K: tl.constexpr):
    # compute block indices
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_batch = tl.program_id(2)
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    batch_idx = pid_batch
    # decompose batch_idx to b and h
    b = batch_idx // H
    h = batch_idx % H
    # compute base pointers
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

import triton
import triton.language as tl

@triton.jit
def trimul_gemm_kernel(
    left, right, out,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # compute start indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # compute batch and hidden idx
    b = pid_bh // H
    h = pid_bh % H

    # compute base ptrs
    left_ptr = left + b * stride_left_b + h * stride_left_h
    right_ptr = right + b * stride_right_b + h * stride_right_h
    out_ptr = out + b * stride_out_b + h * stride_out_h

    # offsets
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left block (M,K)
        left_ptrs = left_ptr + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left_val = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=tl.float16(0.0))

        # load right block transposed (K,N)
        right_ptrs = right_ptr + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        right_val = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=tl.float16(0.0))

        # accumulate
        acc += tl.dot(left_val, right_val)  # (M,N) result

    # store result
    out_ptrs = out_ptr + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
Outgoing TriMul (AlphaFold3) – fast forward implementation.

Algorithm
---------
1. LayerNorm the input (float32 → float16).
2. Linear projections (half) → left, right.
3. Compute three sigmoid gates (half) → left_gate, right_gate, out_gate.
4. Fuse mask (if present) with left/right and their gates in a single Triton kernel.
5. Core O(N³) contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM in Triton (no layout copies).
6. Output LayerNorm → multiply by out_gate → final linear projection.
7. Cast back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# 1. Kernel to fuse mask and gates (in‑place)
# --------------------------------------------------------------
@triton.jit
def gate_mask_fuse_kernel(
    left_ptr,            # * half
    right_ptr,           # * half
    left_gate_ptr,       # * half
    right_gate_ptr,      # * half
    mask_ptr,            # * half (broadcasted)
    total_elements,      # i32
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # broadcast mask over hidden dimension
        mask_idx = offs // HIDDEN_DIM
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(1.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    `mask` may be None.
    All tensors must be contiguous and dtype float16.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when no mask

    gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        HIDDEN_DIM=hidden_dim,
    )


# --------------------------------------------------------------
# 2. Core batched GEMM kernel (TriMul contraction)
# --------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    left,               # * half  (B,N,N,H)
    right,              # * half  (B,N,N,H)
    out,                # * half  (B,N,N,H)
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)       # i‑tile
    pid_j = tl.program_id(1)       # j‑tile
    pid_bh = tl.program_id(2)      # batch*hidden

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # decode batch and hidden index
    b = pid_bh // H
    h = pid_bh % H

    # base pointers for this (b,h)
    left_base  = left  + b * stride_left_b  + h * stride_left_h
    right_base = right + b * stride_right_b + h * stride_right_h
    out_base   = out   + b * stride_out_b   + h * stride_out_h

    # offsets within the tile
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i,k]
        left_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(left_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=tl.float16(0.0))

        # right[j,k]  --> we need right transposed: (k,j)
        right_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        b_mat = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=tl.float16(0.0))

        acc += tl.dot(a, b_mat)   # (M,N) float32 accumulation

    # write back
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------
# 3. Public entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask         : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights      : dict of tensors (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"
    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim], dtype float32
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 → half)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half precision)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16), bias=None)

    # ------------------------------------------------------------------
    # 3. Gates (half precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask with gates (in‑place) – Triton kernel
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).contiguous().view(-1)          # [B*N*N]
        # flatten everything to 1‑D for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(
            left_f, right_f, left_gate_f, right_gate_f, mask_h, hidden_dim,
        )
        # reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # no mask → simple element‑wise gating
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³) contraction via a fused Triton GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # B, N, N, hidden_dim
    out = torch.empty_like(left)  # (B,N,N,H), half

    # strides (in element units)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Tile sizes – good trade‑off for typical B200 resources
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )
    trimul_gemm_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_i,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    # ------------------------------------------------------------------
    # 7. Cast back to float32
    # ------------------------------------------------------------------
    return out.to(torch.float32)
