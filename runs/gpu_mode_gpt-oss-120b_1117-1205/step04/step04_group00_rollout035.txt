@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_b, stride_i, stride_k, stride_d,
    out_stride_b, out_stride_i, out_stride_j, out_stride_d,
    B, N, D,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_bd = tl.program_id(0)  # batch*D
    pid_i = tl.program_id(1)   # block i
    pid_j = tl.program_id(2)   # block j

    b = pid_bd // D
    d = pid_bd % D

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # compute offsets for i and j
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    # mask for valid i/j
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, N, BLOCK_K):
        cur_K = tl.minimum(BLOCK_K, N - k)
        # offsets for k
        offs_k = k + tl.arange(0, BLOCK_K)

        # Compute left block offsets: (i,k)
        left_ptrs = left_ptr + b*stride_b + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k + d
        left = tl.load(left_ptrs, mask=mask_i[:, None] & (offs_k < N), other=0.0)

        # Compute right block as transposed: shape (K,N)
        # right_T offset: (k, j)
        right_ptrs = right_ptr + b*stride_b + offs_k[:, None] * stride_k + offs_j[None, :] * stride_i + d
        right_T = tl.load(right_ptrs, mask=(offs_k < N)[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(left, right_T)

    # compute output offsets
    out_ptrs = out_ptr + b*out_stride_b + offs_i[:, None] * out_stride_i + offs_j[None, :] * out_stride_j + d
    # mask output
    mask_out = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_out)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Load weights as half
    # Norm weight/bias
    norm_weight = weights["norm.weight"].to(device).to(torch.float16)
    norm_bias = weights["norm.bias"].to(device).to(torch.float16)

    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)

    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)

    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)

    # 1. Input LayerNorm (float32 -> half)
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x = x.to(torch.float16)

    # 2. Linear projections
    left = F.linear(x, left_proj_weight, bias=None)   # shape (B,N,N,hidden)
    right = F.linear(x, right_proj_weight, bias=None)

    # 3. Gates
    left_gate = F.linear(x, left_gate_weight, bias=None).sigmoid()
    right_gate = F.linear(x, right_gate_weight, bias=None).sigmoid()
    out_gate = F.linear(x, out_gate_weight, bias=None).sigmoid()

    # 4. Fuse mask and gates
    if not nomask and mask is not None:
        # flatten tensors
        total = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)
        has_mask = 1
        # Ensure all tensors are contiguous
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask.to(torch.float16).contiguous().view(-1)  # shape B*N*N

        _gate_mask_fuse_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            total,
            BLOCK=BLOCK,
            HAS_MASK=has_mask,
            H=hidden_dim,
        )
        # reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask: simple elementwise multiplication
        left = left * left_gate
        right = right * right_gate

    # free gates not needed
    del left_gate, right_gate

    # 5. Core contraction using Triton
    # Prepare output tensor
    out = torch.empty((batch_size, seq_len, seq_len, hidden_dim), dtype=torch.float16, device=device)

    # Strides
    stride_b, stride_i, stride_k, stride_d = left.stride()
    # For out: stride same as left
    out_stride_b, out_stride_i, out_stride_j, out_stride_d = out.stride()

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(batch_size * hidden_dim, 1),   # pid0 dimension
        triton.cdiv(seq_len, BLOCK_M),            # pid1 i dimension
        triton.cdiv(seq_len, BLOCK_N),            # pid2 j dimension
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        stride_b, stride_i, stride_k, stride_d,
        out_stride_b, out_stride_i, out_stride_j, out_stride_d,
        batch_size, seq_len, hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # 6. Output layernorm, gate, final linear
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight, bias=None)

    return out.to(torch.float32)

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_b, stride_i, stride_k, stride_d,
    out_stride_b, out_stride_i, out_stride_j, out_stride_d,
    B, N, D,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_bd = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    b = pid_bd // D
    d = pid_bd % D

    # tile start indices
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # offset vectors
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k_start in range(0, N, BLOCK_K):
        # compute K range
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + b*stride_b + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k + d
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right block transposed: shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + b*stride_b + offs_k[:, None] * stride_k + offs_j[None, :] * stride_i + d
        right_T = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # matrix multiplication
        acc += tl.dot(left, right_T)

    # store results
    out_ptrs = out_ptr + b*out_stride_b + offs_i[:, None] * out_stride_i + offs_j[None, :] * out_stride_j + d
    mask_out = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_out)

"""
TriMul (outgoing) forward pass optimized with Triton.

Steps:
1. LayerNorm on the input (float32) → cast to float16.
2. Linear projection to the hidden dimension (half precision).
3. Compute three sigmoid gates.
4. Fuse mask (if present) and the left/right gates in‑place using a small Triton kernel.
5. Contract left and right tensors:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a blocked GEMM in Triton, avoiding the costly
   permute/contiguous copies required by a PyTorch batched matmul.
6. Apply a second LayerNorm, the output gate, and the final linear projection.
   The result is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half*  (flattened left tensor)
    right_ptr,         # half*  (flattened right tensor)
    left_gate_ptr,     # half*  (flattened left gate)
    right_gate_ptr,    # half*  (flattened right gate)
    mask_ptr,          # half*  (flattened mask, broadcasted over hidden dim)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 -> mask present, 0 -> no mask
    H: tl.constexpr,          # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        mask_idx = offs // H               # each mask entry repeats H times
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_b, stride_i, stride_k, stride_d,
    out_stride_b, out_stride_i, out_stride_j, out_stride_d,
    B, N, D,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_bd = tl.program_id(0)   # combined batch & hidden dimension
    pid_i = tl.program_id(1)    # block over the first sequence axis
    pid_j = tl.program_id(2)    # block over the second sequence axis

    b = pid_bd // D            # batch index
    d = pid_bd % D             # hidden channel index

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # tiled reduction over k
    for k_start in range(0, N, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + b*stride_b + offs_i[:, None]*stride_i + offs_k[None, :]*stride_k + d
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right block transposed (K, N): shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + b*stride_b + offs_k[:, None]*stride_k + offs_j[None, :]*stride_i + d
        right_T = tl.load(right_ptrs,
                          mask=mask_k[:, None] & mask_j[None, :],
                          other=0.0)

        # block matrix multiplication
        acc += tl.dot(left, right_T)

    # store the result
    out_ptrs = out_ptr + b*out_stride_b + offs_i[:, None]*out_stride_i + offs_j[None, :]*out_stride_j + d
    mask_out = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_out)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.
    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (optional)
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections to hidden dimension (half)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16), bias=None)

    # -----------------------------------------------------------------
    # 3. Compute gates (half)
    # -----------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # -----------------------------------------------------------------
    # 4. Fuse mask (if any) and gates in‑place via Triton
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        total = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)

        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask.to(torch.float16).contiguous().view(-1)   # [B, N, N]

        _gate_mask_fuse_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            total,
            BLOCK=BLOCK,
            HAS_MASK=1,
            H=hidden_dim,
        )
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # Simple element‑wise multiplication when no mask is used
        left = left * left_gate
        right = right * right_gate

    # Gates no longer needed
    del left_gate, right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a blocked GEMM in Triton.
    # -----------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    stride_b, stride_i, stride_k, stride_d = left.stride()
    out_stride_b, out_stride_i, out_stride_j, out_stride_d = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(B * hidden_dim, 1),          # PID 0 : batch * hidden
        triton.cdiv(N, BLOCK_M),                 # PID 1 : i‑blocks
        triton.cdiv(N, BLOCK_N),                 # PID 2 : j‑blocks
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        stride_b, stride_i, stride_k, stride_d,
        out_stride_b, out_stride_i, out_stride_j, out_stride_d,
        B, N, hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    return out.to(torch.float32)
