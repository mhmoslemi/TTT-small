import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def mask_gate_fuse_kernel(left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr,
                         total_elems, BLOCK: tl.constexpr,
                         HAS_MASK: tl.constexpr,
                         H: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    lgate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # Each group of H elements share same mask index.
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * lgate * mask_val
        right = right * rgate * mask_val
    else:
        left = left * lgate
        right = right * rgate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)
```

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _mask_gate_fuse_kernel(left_ptr, right_ptr,
                           left_gate_ptr, right_gate_ptr,
                           mask_ptr,
                           total_elems,
                           BLOCK: tl.constexpr,
                           HAS_MASK: tl.constexpr,
                           H: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    l = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    r = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # each hidden slice shares the same mask element
        mask_idx = offs // H
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        l = l * lg * m
        r = r * rg * m
    else:
        l = l * lg
        r = r * rg

    tl.store(left_ptr + offs, l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)
```

def custom_kernel(data):
    inp, mask, weights, config = data
    device = inp.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Cast weights to half on device
    norm_weight = weights["norm.weight"].to(device=device, dtype=torch.float16)
    norm_bias = weights["norm.bias"].to(device=device, dtype=torch.float16)
    left_proj_weight = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_weight = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # 1. LayerNorm
    x = F.layer_norm(inp, (dim,), weight=norm_weight, bias=norm_bias)
    x = x.to(torch.float16)

    # 2. Projections
    left = F.linear(x, left_proj_weight)   # (B,N,N,H)
    right = F.linear(x, right_proj_weight)

    # 3. Gates
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    # 4. Mask + gate fusion
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        # flatten tensors
        left_f = left.view(-1).contiguous()
        right_f = right.view(-1).contiguous()
        lg_f = left_gate.view(-1).contiguous()
        rg_f = right_gate.view(-1).contiguous()
        mask_f = mask_h.view(-1).contiguous()
        _apply_mask_and_gate(left_f, right_f, lg_f, rg_f, mask_f, hidden_dim)
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # 5. Core contraction via batched GEMM
    B, N, _, _ = left.shape
    # move hidden dim to batch axis and make contiguous
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous()

    left_mat = left_t.view(B * hidden_dim, N, N)
    right_mat = right_t.view(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # 6. Output layernorm, out_gate, final proj

"""
Outgoing TriMul (AlphaFold3) forward pass.

The kernel fuses the element‑wise mask and gate multiplications for the
left/right projected tensors.  All other heavy lifting (LayerNorm,
linear projections, batched GEMM) is performed with PyTorch’s highly‑
optimised kernels.  The computation steps are:

1. Layer‑norm the input (float32 → float16).
2. Linear projections to the hidden dimension.
3. Compute three gate tensors with a sigmoid.
4. If a mask is present, launch a Triton kernel that multiplies
   left/right by their respective gates **and** the mask in‑place.
   Otherwise perform the two element‑wise products with PyTorch.
5. Core O(N³) contraction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM (B*H independent matrix‑multiplications)
   using torch.bmm which maps to cuBLAS.
6. Output LayerNorm, element‑wise gating with out_gate and a final linear
   projection back to the original feature dimension.
7. Return the result as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # *half
    right_ptr,         # *half
    left_gate_ptr,     # *half
    right_gate_ptr,    # *half
    mask_ptr,          # *half (broadcasted over hidden dim)
    total_elems,       # i32
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,   # hidden dimension
):
    """In‑place: left = left * left_gate * mask, right = right * right_gate * mask."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    l = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    r = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # One mask value per (b,i,k) group – same for every hidden slice.
        mask_idx = offs // H
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        l = l * lg * m
        r = r * rg * m
    else:
        l = l * lg
        r = r * rg

    tl.store(left_ptr + offs, l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)


def _apply_mask_and_gate(left_f, right_f, left_gate_f, right_gate_f, mask_f, hidden_dim):
    """Launch the Triton kernel that fuses mask and gate multiplications in‑place."""
    total = left_f.numel()
    BLOCK = 1024  # must not exceed the hardware limit (max 1024 threads per block)
    grid = (triton.cdiv(total, BLOCK),)
    _mask_gate_fuse_kernel[grid](
        left_f,
        right_f,
        left_gate_f,
        right_gate_f,
        mask_f,
        total,
        BLOCK=BLOCK,
        HAS_MASK=1,
        H=hidden_dim,
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict mapping parameter names to float32 tensors
            - config: dict containing at least "dim" and "hidden_dim",
                      optionally "nomask" (bool)

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    inp, mask, weights, config = data
    device = inp.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # Cast weights to half precision on the target device
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device=device, dtype=torch.float16)
    norm_bias = weights["norm.bias"].to(device=device, dtype=torch.float16)

    left_proj_weight = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    left_gate_weight = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)

    to_out_weight = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 → float16)
    # -----------------------------------------------------------------
    x = F.layer_norm(inp, (dim,), weight=norm_weight, bias=norm_bias)
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections to hidden dimension
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight)

    # -----------------------------------------------------------------
    # 3. Gate tensors (sigmoid)
    # -----------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    # -----------------------------------------------------------------
    # 4. Fuse mask with gates (if required) via Triton
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → broadcast over hidden dimension
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]

        # Flatten tensors for the kernel
        left_f = left.view(-1).contiguous()
        right_f = right.view(-1).contiguous()
        lg_f = left_gate.view(-1).contiguous()
        rg_f = right_gate.view(-1).contiguous()
        mask_f = mask_h.view(-1).contiguous()  # length B*N*N

        _apply_mask_and_gate(left_f, right_f, lg_f, rg_f, mask_f, hidden_dim)

        # Restore original shapes
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask – simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: batched matrix multiplication per hidden slice
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape  # left shape = (B, N, N, H)

    # Move hidden dimension to the batch axis
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_mat = left_t.view(B * hidden_dim, N, N)     # (B*H, N, N)
    right_mat = right_t.view(B * hidden_dim, N, N)   # (B*H, N, N)

    # Batched GEMM: (i,k) × (j,k)ᵀ → (i,j)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)

    # Return as float32 as required
    return out.to(torch.float32)
