@triton.jit
def gate_mask_fuse_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,
):
   pid = tl.program_id(0)
   offs = pid * BLOCK + tl.arange(0, BLOCK)
   mask = offs < total_elements
   left = tl.load(left_ptr + offs, mask=mask)
   right = tl.load(right_ptr + offs, mask=mask)
   left_gate = tl.load(left_gate_ptr + offs, mask=mask)
   right_gate = tl.load(right_gate_ptr + offs, mask=mask)
   if HAS_MASK:
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
   else:
        left = left * left_gate
        right = right * right_gate
   tl.store(left_ptr + offs, left, mask=mask)
   tl.store(right_ptr + offs, right, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Convert weights to half once.
    norm_weight_h = weights["norm.weight"].to(torch.float16)
    norm_bias_h = weights["norm.bias"].to(torch.float16)
    left_proj_w_h = weights["left_proj.weight"].to(torch.float16)
    right_proj_w_h = weights["right_proj.weight"].to(torch.float16)
    left_gate_w_h = weights["left_gate.weight"].to(torch.float16)
    right_gate_w_h = weights["right_gate.weight"].to(torch.float16)
    out_gate_w_h = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w_h = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b_h = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w_h = weights["to_out.weight"].to(torch.float16)

    # 1 LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight_h, bias=norm_bias_h, eps=1e-5)
    x = x.to(torch.float16)

    # 2 Projections
    left = F.linear(x, left_proj_w_h, bias=None)  # shape B,N,N,hidden
    right = F.linear(x, right_proj_w_h, bias=None)

    # 3 Gates
    left_gate = F.linear(x, left_gate_w_h, bias=None).sigmoid()
    right_gate = F.linear(x, right_gate_w_h, bias=None).sigmoid()
    out_gate = F.linear(x, out_gate_w_h, bias=None).sigmoid()

    # 4 Fuse gates and mask
    if not nomask:
        # mask shape [B,N,N]; convert to half
        mask_h = mask.to(torch.float16).contiguous()
        # flatten slices for kernel
        left_f = left.view(-1)
        right_f = right.view(-1)
        left_gate_f = left_gate.view(-1)
        right_gate_f = right_gate.view(-1)
        mask_f = mask_h.view(-1)
        total = left_f.numel()
        BLOCK = 16384  # can be large for efficiency (max 65536)
        grid = (triton.cdiv(total, BLOCK),)
        _gate_mask_fuse_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            total,
            BLOCK=BLOCK,
            HAS_MASK=1,
            H=hidden_dim,
        )
        left = left_f.view_as(left)
        right = right_f.view_as(right)
        # free gates
        del left_gate_f, right_gate_f, mask_f, left_gate, right_gate
    else:
        left = left * left_gate
        right = right * right_gate
        # free gates
        del left_gate, right_gate

    # 5 Contraction
    # shape: B,N,N,H --> permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2)
    right_perm = right.permute(0, 3, 1, 2)
    # B*H, N,N.
    B, H, N, _ = left_perm.shape
    left_mat = left_perm.reshape(B*H, N, N)
    right_mat = right_perm.reshape(B*H, N, N)
    # bmm left_mat @ right_mat.transpose(1,2)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))
    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 6 Output norm and gating
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w_h, bias=to_out_norm_b_h, eps=1e-5)
    out = out * out_gate

    # 7 final linear
    out = F.linear(out, to_out_w_h, bias=None)

    # Return float32
    return out.to(torch.float32)

"""
Fast outgoing TriMul (AlphaFold3) forward pass.

The implementation follows the reference PyTorch module but keeps all heavy
computations in half precision and uses a tiny Triton kernel to fuse the
element‑wise mask and gate multiplications.  The O(N³) contraction is performed
as a batched half‑precision GEMM (`torch.bmm`) after reshaping the
[batch, seq, seq, hidden] tensors to (B*hidden, N, N).  The final result is
returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # float16*
    right_ptr,         # float16*
    left_gate_ptr,     # float16*
    right_gate_ptr,    # float16*
    mask_ptr,          # float16* (broadcasted)
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask is present
    H: tl.constexpr,          # hidden dimension size
):
    """
    In‑place: left = left * left_gate * mask;
             right = right * right_gate * mask
    mask is shared across the hidden dimension, i.e. each (b,i,k) entry
    multiplies all H hidden channels.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each hidden entry shares the same mask entry
        mask_idx = offs // H
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * m
        right = right * right_gate * m
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    Fuse gate and (optional) mask into `left` and `right` in‑place using Triton.
    All tensors must be contiguous and dtype float16.
    """
    total = left.numel()
    BLOCK = 16384  # large enough for good occupancy
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when no mask

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) – may be ignored
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Convert all weights to half precision once
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(torch.float16, device=device)
    norm_b = weights["norm.bias"].to(torch.float16, device=device)

    left_proj_w = weights["left_proj.weight"].to(torch.float16, device=device)
    right_proj_w = weights["right_proj.weight"].to(torch.float16, device=device)

    left_gate_w = weights["left_gate.weight"].to(torch.float16, device=device)
    right_gate_w = weights["right_gate.weight"].to(torch.float16, device=device)
    out_gate_w = weights["out_gate.weight"].to(torch.float16, device=device)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16, device=device)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16, device=device)

    to_out_w = weights["to_out.weight"].to(torch.float16, device=device)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (half precision)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w, bias=None)   # [B,N,N,H]
    right = F.linear(x, right_proj_w, bias=None)  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3. Gates (half) with sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_w, bias=None).sigmoid()
    right_gate = F.linear(x, right_gate_w, bias=None).sigmoid()
    out_gate = F.linear(x, out_gate_w, bias=None).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse element‑wise mask and gates (in‑place Triton kernel)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → half and flatten
        mask_h = mask.to(torch.float16, device=device).contiguous().view(-1)

        # flatten tensors for kernel launch
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_h,
            hidden_dim,
        )
        # reshape back (views share storage)
        left = left_f.view_as(left)
        right = right_f.view_as(right)
        # clean up temporaries
        del left_f, right_f, left_gate_f, right_gate_f, mask_h
    else:
        # No mask: simple element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate

    # gates are no longer needed
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³) contraction via batched GEMM
    #    out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) and make contiguous for view
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)          # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)        # (B*H, N, N)

    # Batched matrix multiplication: left @ right^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w, bias=None)

    # ------------------------------------------------------------------
    # 7. Return in float32 as required by the interface
    # ------------------------------------------------------------------
    return out.to(torch.float32)
