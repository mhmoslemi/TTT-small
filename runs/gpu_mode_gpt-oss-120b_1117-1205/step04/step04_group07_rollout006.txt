"""
TriMul (outgoing) forward pass optimized with Triton.

Algorithm
---------
1. Layer‑normalize the input tensor (float32 → float16).
2. Linear projections to a hidden dimension for “left” and “right”.
3. Compute three sigmoid gates (left, right, out) from the same
   normalized input.
4. Fuse mask (if present) and the left/right gates in a single Triton
   kernel – this performs the elementwise
   `left = left * left_gate * mask` and
   `right = right * right_gate * mask` (or skips the mask).
5. The core O(N³) contraction
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is realized as a batched matrix multiplication:
      (B·H, N, N) @ (B·H, N, N)ᵀ  via `torch.bmm`.
6. Apply a LayerNorm on the hidden dimension, scale by the out‑gate,
   and project back to the original dimension.
7. Return the result in float32.

Only the mask‑gate fusion is performed in Triton; the heavy‑weight
contraction uses cuBLAS through `torch.bmm`, which is already highly
optimized on NVIDIA GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # * half
    right_ptr,         # * half
    left_gate_ptr,     # * half
    right_gate_ptr,    # * half
    mask_ptr,          # * half (flattened mask, broadcasted)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load data
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # Each mask element repeats HIDDEN_DIM times in the flattened tensor
        mask_idx = offs // HIDDEN_DIM
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store results back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (bool/float) – ignored if
                           config["nomask"] is True.
            - weights: dict with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict containing at least "dim", "hidden_dim",
                      optionally "nomask".
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (half‑precision)
    # ------------------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Compute sigmoid gates (half‑precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) and gating with Triton
    # ------------------------------------------------------------------
    # Flatten tensors for the kernel
    left_f = left.contiguous().view(-1)
    right_f = right.contiguous().view(-1)
    left_gate_f = left_gate.contiguous().view(-1)
    right_gate_f = right_gate.contiguous().view(-1)

    if not nomask:
        mask_f = mask.to(torch.float16).contiguous().view(-1)  # [B*N*N]
        has_mask = 1
    else:
        mask_f = left_f  # dummy pointer, never read
        has_mask = 0

    total = left_f.numel()
    BLOCK = 32768  # should be a multiple of 32
    grid = (triton.cdiv(total, BLOCK),)

    _gate_mask_fuse_kernel[grid](
        left_f,
        right_f,
        left_gate_f,
        right_gate_f,
        mask_f,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        HIDDEN_DIM=hidden_dim,
    )

    # Restore original shapes
    left = left_f.view_as(left)
    right = right_f.view_as(right)

    # ------------------------------------------------------------------
    # 5. Core O(N³) contraction via batched GEMM
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Move hidden_dim to the second axis for efficient bmm
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)    # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)  # (B*H, N, N)

    # Matrix multiplication with the transpose of right
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Reshape back to [B, N, N, H]
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in float32 as required
    return out.to(torch.float32)
