"""
Outgoing TriMul (AlphaFold3) forward pass.

The implementation follows the reference PyTorch module but moves the expensive
element‑wise mask‑gate fusion into a Triton kernel.  The heavy O(N³) contraction
is performed with a batched half‑precision GEMM (torch.bmm), which is much faster
than a naïve Python loop and comparable to a custom Triton kernel while keeping
the code simple.

Algorithm
---------
1. Layer‑norm the input tensor (float32) → half precision.
2. Linear projections (half) produce `left` and `right`.
3. Compute three sigmoid gates (half).
4. Fuse mask (if present) and the left/right gates with a Triton kernel
   (in‑place element‑wise multiplication, broadcasting the mask over the hidden
   dimension).
5. Core contraction:
   out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   Implemented as a batched GEMM for each hidden channel.
6. Output layer‑norm, out‑gate scaling and final linear projection.
7. Return result in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def gate_mask_fuse_kernel(
    left_ptr,          # * half  (flattened left tensor)
    right_ptr,         # * half  (flattened right tensor)
    left_gate_ptr,     # * half
    right_gate_ptr,    # * half
    mask_ptr,          # * half (broadcasted mask, length = B*N*N)
    total_elements,    # i32  total number of elements in left/right
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):
    """In‑place fusion:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
    where mask is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # each mask element corresponds to HIDDEN_DIM consecutive entries
        mask_idx = offs // HIDDEN_DIM
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(1.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # store in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _gate_mask_fuse(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    Apply mask and gates in‑place using the Triton kernel.
    All tensors must be contiguous and have dtype torch.float16.
    `mask` may be None (no mask).
    """

        right_gate,

        BLOCK=BLOCK,
        HAS_MASK=has_mask,

    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype torch.float32
        - mask         : torch.Tensor, shape [B, N, N] (float or bool) or None
        - weights      : dict mapping weight names to torch.Tensor (float32)
        - config       : dict with keys:
            * "dim"        : input channel dimension
            * "hidden_dim" : hidden projection dimension
            * "nomask"     : bool, True if no mask is applied

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim], dtype torch.float32
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half precision)
    # ------------------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gates (half precision, sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) with gates via Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # Convert mask to half and flatten
        mask_h = mask.to(torch.float16).contiguous().view(-1)          # [B*N*N]
        # Flatten tensors for kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _gate_mask_fuse(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_h,
            hidden_dim,
        )
        # Reshape back to original layout
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask – simple elementwise gating in PyTorch
        left = left * left_gate
        right = right * right_gate

    # Gates no longer needed
    del left_gate, right_gate, x_h

    # ------------------------------------------------------------------
    # 5. Core O(N³) contraction via batched GEMM (half precision)
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden_dim, N, N) for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # Batched GEMM: left @ rightᵀ  (each hidden channel independent)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore (B, N, N, H) layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate                     # element‑wise scaling (half)
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in original dtype (float32)
    return out.to(torch.float32)
```
