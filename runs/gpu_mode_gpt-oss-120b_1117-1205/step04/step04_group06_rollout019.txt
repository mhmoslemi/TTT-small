def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device
    # LayerNorm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # (x is float32)
    # Convert to float16
    x_h = x.to(torch.float16)
    # Compute linear projections
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )
    # Compute gates
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    # Apply mask and gate fusion using Triton
    if not nomask:
        # ensure mask is torch.float16 and broadcast shape
        mask_h = mask.to(torch.float16).contiguous()
        # mask shape [B, N, N]; flatten
        mask_f = mask_h.view(-1)
        # flatten left/right/gates
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        total = left_f.numel()
        BLOCK = 1024  # or 2048
        grid = (triton.cdiv(total, BLOCK),)
        has_mask = 1
        _gate_mask_fuse_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            total,
            BLOCK=BLOCK,
            H=hidden_dim,
            HAS_MASK=has_mask,
        )
        # reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
        # Release extra tensors
        del left_f, right_f, left_gate_f, right_gate_f, mask_f
    else:
        left = left * left_gate
        right = right * right_gate
    # Free gates and x_h
    del left_gate, right_gate, x_h
    # Contraction: out = sum_k left[...,k,:] * right[...,k,:]
    # Permute to bring hidden dim to second axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)
    # Perform batched matmul: (B*H, N, N) x (B*H, N, N)^T
    # This yields (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # Reshape back
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    # Output layernorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16)
    )
    # Apply out_gate
    out = out * out_gate
    # Final linear
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )
    # Return float32
    return out.to(torch.float32)
```

"""
Outgoing TriMul implementation (AlphaFold3).

The kernel fuses the mask and gating operations:
    left  = left  * left_gate  * mask
    right = right * right_gate * mask
All other steps (LayerNorm, linear projections, the O(N³) contraction,
output LayerNorm, gating, final linear) are performed with PyTorch
operations (torch.nn.functional). The heavy contraction is realized as a
batched matrix multiplication:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
implemented via torch.bmm on tensors of shape (B*H, N, N). The final
output is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,                # float16*
    right_ptr,               # float16*
    left_gate_ptr,           # float16*
    right_gate_ptr,          # float16*
    mask_ptr,                # float16* (flattened mask, broadcasted)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    H: tl.constexpr,         # hidden_dim
    HAS_MASK: tl.constexpr,  # 1 if mask is present else 0
):
    """In‑place fusion of gating and optional masking."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each mask element corresponds to H consecutive entries
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool / numeric) – may be ignored
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision for subsequent ops
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (half)
    # ------------------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gate values (half) with sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) with gates via Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → half, flattened
        mask_h = mask.to(torch.float16).contiguous()
        mask_f = mask_h.view(-1)                      # shape (B*N*N,)

        # Flatten tensors for the kernel (contiguous layout required)
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        total = left_f.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)

        _gate_mask_fuse_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            total,
            BLOCK=BLOCK,
            H=hidden_dim,
            HAS_MASK=1,
        )

        # Reshape back to original layout
        left = left_f.view_as(left)
        right = right_f.view_as(right)

        # Release temporaries
        del left_f, right_f, left_gate_f, right_gate_f, mask_f, mask_h
    else:
        # No mask: simple element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate

    # Gates and high‑precision input are no longer needed
    del left_gate, right_gate, x_h

    # ------------------------------------------------------------------
    # 5. Core contraction using a batched GEMM
    #    out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    # ------------------------------------------------------------------
    # Move hidden dimension next to batch for efficient matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)    # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)  # (B*H, N, N)

    # Batched matrix multiplication: (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise gating (half)

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in float32 as required by the interface
    return out.to(torch.float32)
