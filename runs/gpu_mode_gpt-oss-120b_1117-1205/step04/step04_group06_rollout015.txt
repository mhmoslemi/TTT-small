def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # shapes and values
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    # Convert to half
    device = input_tensor.device
    # LayerNorm
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])
    # Cast to half
    x_h = x.to(torch.float16)
    # Linear proj
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16), bias=None)
    # Gates:
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()
    # Fuse mask and gates
    if not nomask:
        mask_f = mask.to(torch.float16).contiguous().view(-1)   # [B*N*N]
        # Flatten tensors to 1d
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        _apply_gate_and_mask(left_f, right_f, left_gate_f, right_gate_f, mask_f, hidden_dim)
        # now left and right have gate+mask applied
        left = left_f.view_as(left)
        right = right_f.view_as(right)
        # free intermediates: left_gate_f etc.
        del left_gate_f, right_gate_f, mask_f, left_f, right_f
        del left_gate, right_gate
    else:
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate
    # contraction
    # Permute to B, H, N, N
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()
    B, H, N, _ = left_perm.shape
    # reshape
    left_mat = left_perm.view(B*H, N, N)  # (B*H, N, N)
    right_mat = right_perm.view(B*H, N, N)  # (B*H, N, N)
    # Matrix multiplication, need R^T: right_mat.transpose(1,2)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))
    out = out_mat.view(B, H, N, N).permute(0,2,3,1).contiguous()
    # Output layernorm
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=weights["to_out_norm.weight"].to(torch.float16),
                       bias=weights["to_out_norm.bias"].to(torch.float16))
    # out gate scaling
    out = out * out_gate
    # final linear
    out = F.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)
    # Cast to float32
    return out.to(torch.float32)

"""
TriMul forward (outgoing) – fast Triton‑fused gate & mask.

Algorithm
---------
1. LayerNorm over the input channel dimension (float32 → float16).
2. Linear projections `left`, `right` to `hidden_dim` (half precision).
3. Compute three gate tensors (`left_gate`, `right_gate`, `out_gate`) via
   linear + sigmoid (half precision).
4. Fuse mask (if present) with the two gates in-place using a Triton kernel:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
   The kernel processes the flattened tensors of shape
   (B·N·N·hidden_dim,) and broadcasts the scalar mask over the hidden dimension.
5. Perform the core contraction as a batched GEMM:
       out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   which is equivalent to a batched matrix multiply
       (B·hidden_dim, N, N) @ (B·hidden_dim, N, N)ᵀ .
6. Apply a second LayerNorm over `hidden_dim`, multiply by `out_gate`,
   and a final linear projection back to `dim`.
7. Return the result in float32.

Only the gate‑mask fusion is Triton‑implemented; the heavy O(N³) contraction
uses cuBLAS via torch.bmm for maximal throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton kernel: fuse gate tensors with optional mask (in‑place)
# -------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # float16*  (B·N·N·H)
    right_ptr,              # float16*
    left_gate_ptr,          # float16*
    right_gate_ptr,         # float16*
    mask_ptr,               # float16* (B·N·N) – dummy if HAS_MASK==0
    total_elements,         # int32   – number of elements in left/right
    BLOCK: tl.constexpr,    # compile‑time block size
    HAS_MASK: tl.constexpr, # 1 if mask is present
    HIDDEN_DIM: tl.constexpr # hidden dimension (size of innermost channel)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    # Load data
    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    if HAS_MASK:
        # Each hidden entry shares the same mask entry.
        mask_idx = offs // HIDDEN_DIM
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store in‑place
    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of mask and gate tensors for `left` and `right`.
    All tensors must be contiguous, 1‑D, dtype float16.
    """
    total = left.numel()
    BLOCK = 8192  # empirically good for large tensors
    grid = (triton.cdiv(total, BLOCK),)

    # Launch kernel – HAS_MASK=1 (mask is guaranteed present)
    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask,
        total,
        BLOCK=BLOCK,
        HAS_MASK=1,
        HIDDEN_DIM=hidden_dim,
    )


# -------------------------------------------------------------
# Entry point required by the test harness
# -------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul forward (outgoing) implementation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------------------------------------------------------
    # 1. Input LayerNorm (float32 -> half)
    # ---------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ---------------------------------------------------------
    # 2. Linear projections to hidden_dim (half precision)
    # ---------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ---------------------------------------------------------
    # 3. Gates (half) + sigmoid
    # ---------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ---------------------------------------------------------
    # 4. Fuse mask (if present) with gates via Triton kernel
    # ---------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> float16, flattened
        mask_f = mask.to(torch.float16).contiguous().view(-1)  # (B*N*N)

        # Flatten tensors for kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(
            left_f, right_f,
            left_gate_f, right_gate_f,
            mask_f,
            hidden_dim,
        )

        # Reshape back to original layout
        left = left_f.view_as(left)
        right = right_f.view_as(right)

        # Release temporaries
        del left_f, right_f, left_gate_f, right_gate_f, mask_f
        del left_gate, right_gate
    else:
        # Simple element‑wise gate application (no mask)
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ---------------------------------------------------------
    # 5. Core contraction: batched GEMM over (i,k) and (j,k)
    # ---------------------------------------------------------
    # Rearrange to (B, H, N, N) for efficient batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # Batched matmul with transposition on the second operand
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling, final linear
    # ---------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise gating (half)

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in float32 as required by the interface
    return out.to(torch.float32)
