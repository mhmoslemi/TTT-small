def triton_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch, N, hidden_dim,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_n, stride_right_k,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0) # batch index
    pid_m = tl.program_id(1) # tile row i
    pid_n = tl.program_id(2) # tile col j

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask
    mask_m = offs_m < N
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile
        a_ptrs = left_ptr + pid_b * stride_left_batch + offs_m[:, None] * stride_left_m + offs_k[None, :] * stride_left_k

import torch
import torch.nn.functional as F
import triton

@triton.jit
def triton_trimul_kernel(...):
    # Implementation

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # layernorm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    ).to(torch.float16)

    # projection
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # mask
    if mask is not None and not nomask:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    # gate
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    B, N, _, _ = left.shape

    # reshape for triton gemm
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # flatten batch
    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)

    # allocate output
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # get strides
    stride_l_batch = left_mat.stride(0)
    stride_l_m = left_mat.stride(1)
    stride_l_k = left_mat.stride(2)

    stride_r_batch = right_mat.stride(0)
    stride_r_n = right_mat.stride(1)  # j dimension
    stride_r_k = right_mat.stride(2)

    stride_o_batch = out_mat.stride(0)
    stride_o_m = out_mat.stride(1)
    stride_o_n = out_mat.stride(2)

    # launch triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * hidden_dim, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    triton_trimul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        stride_l_batch, stride_l_m, stride_l_k,
        stride_r_batch, stride_r_n, stride_r_k,
        stride_o_batch, stride_o_m, stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # output layernorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass implemented with Triton.

The algorithm:
1. Layer‑norm the input tensor (float32 → float16).
2. Linear projections to a hidden dimension (no bias).
3. Optional mask multiplication (broadcast over the hidden dim).
4. Gating: sigmoid on three linear projections (left, right, out).
5. Core contraction: for each batch & hidden slice compute
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   implemented as a custom batched GEMM (A @ Bᵀ) in Triton.
6. Output LayerNorm, element‑wise gating with `out_gate`,
   and final linear projection back to the original feature dimension.
7. Return the result in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_gemm_kernel(
    left_ptr,               # *half, shape (B*H, N, N)
    right_ptr,              # *half, shape (B*H, N, N)
    out_ptr,                # *half, shape (B*H, N, N)
    N: tl.int32,            # sequence length
    # strides (in elements)
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_j, stride_r_k,
    stride_o_batch, stride_o_i, stride_o_j,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_b = tl.program_id(0)   # batch‑hidden index
    pid_m = tl.program_id(1)   # tile row (i)
    pid_n = tl.program_id(2)   # tile col (j)

    # offsets within the N×N matrices
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction axis k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: A[i, k]
        a_ptrs = (
            left_ptr
            + pid_b * stride_l_batch
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=tl.float16(0.0),
        )

        # right tile transposed: Bᵀ[k, j]  (original right is [j, k])
        b_ptrs = (
            right_ptr
            + pid_b * stride_r_batch
            + offs_k[:, None] * stride_r_k
            + offs_j[None, :] * stride_r_j
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=tl.float16(0.0),
        )

        # accumulate mat‑mul
        acc += tl.dot(a, b)

    # write result back in fp16
    c_ptrs = (
        out_ptr
        + pid_b * stride_o_batch
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
    )
    tl.store(
        c_ptrs,
        tl.cast(acc, tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict with all linear / layer‑norm parameters
            - config: dict with at least "dim", "hidden_dim" and optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    inp, mask, weights, config = data
    device = inp.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 → half)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    ).to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections to hidden dimension
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # -----------------------------------------------------------------
    # 4. Gate tensors (sigmoid)
    # -----------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    B, N, _, _ = left.shape  # B, seq_len, _, hidden_dim

    # -----------------------------------------------------------------
    # 5. Core contraction via Triton (batched GEMM)
    # -----------------------------------------------------------------
    # Move hidden dimension to batch axis and make tensors contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Collapse batch & hidden into a single batch axis
    left_mat = left_perm.view(B * hidden_dim, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N) # (B*H, N, N)

    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # strides in elements
    stride_l_batch = left_mat.stride(0)
    stride_l_i = left_mat.stride(1)
    stride_l_k = left_mat.stride(2)

    stride_r_batch = right_mat.stride(0)
    stride_r_j = right_mat.stride(1)   # original j dimension
    stride_r_k = right_mat.stride(2)   # k dimension

    stride_o_batch = out_mat.stride(0)
    stride_o_i = out_mat.stride(1)
    stride_o_j = out_mat.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _trimul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        stride_l_batch,
        stride_l_i,
        stride_l_k,
        stride_r_batch,
        stride_r_j,
        stride_r_k,
        stride_o_batch,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return in float32 as specified
    return out.to(torch.float32)
