import triton
import triton.language as tl

@triton.jit
def batched_gemm_kernel(a_ptr, b_ptr, c_ptr,
                        M, N, K,
                        stride_am, stride_ak, stride_ab,
                        stride_bm, stride_bk, stride_bb,
                        stride_cm, stride_cn, stride_cb,
                        BLOCK_M: tl.constexpr,
                        BLOCK_N: tl.constexpr,
                        BLOCK_K: tl.constexpr):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    off_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = off_m < M
    mask_n = off_n < N

    # pointer offsets
    a_batch_offset = batch * stride_ab
    b_batch_offset = batch * stride_bb
    c_batch_offset = batch * stride_cb

    # acc
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # k loop
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)

        mask_k = k_offset < K

        # Load A
        a = tl.load(a_ptr + a_batch_offset
                      + off_m[:, None] * stride_am
                      + k_offset[None, :] * stride_ak,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B (as K x N)
        b = tl.load(b_ptr + b_batch_offset
                      + k_offset[:, None] * stride_bk
                      + off_n[None, :] * stride_bm,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_K) dot (BLOCK_K, BLOCK_N)

    # Store
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + c_batch_offset + off_m[:, None] * stride_cm + off_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

"""
Outgoing TriMul (AlphaFold3) forward pass.

The implementation follows the PyTorch reference but replaces the
O(N³) contraction with a custom batched GEMM written in Triton.
All heavy arithmetic is performed in float16 while accumulation is
done in float32 for numerical stability.

Steps
-----
1. LayerNorm on the input (float32) → float16.
2. Linear projections (left/right) to hidden_dim (float16).
3. Compute three sigmoid gates (float16).
4. Optional mask broadcasting (float16).
5. Fuse gates with left/right (element‑wise, PyTorch).
6. Core contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM (B*hidden_dim independent
   matrix multiplications) via a Triton kernel.
7. Output LayerNorm, out_gate scaling and final linear projection.
   Result is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Batched GEMM kernel: computes C = A @ Bᵀ for many independent batches.
# A : (batch, M, K)   – row‑major
# B : (batch, N, K)   – row‑major (we read it as K×N inside the kernel)
# C : (batch, M, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr,        # * half
    b_ptr,        # * half
    c_ptr,        # * half
    M, N, K,      # int32
    # strides for A
    stride_am, stride_ak, stride_ab,
    # strides for B
    stride_bm, stride_bk, stride_bb,
    # strides for C
    stride_cm, stride_cn, stride_cb,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the current batch
    # ------------------------------------------------------------------
    a_batch = a_ptr + batch_id * stride_ab
    b_batch = b_ptr + batch_id * stride_bb
    c_batch = c_ptr + batch_id * stride_cb

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for kb in range(0, num_k_blocks):
        k_offset = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_batch
            + offs_m[:, None] * stride_am
            + k_offset[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=tl.float16(0.0),
        )

        # Load B tile as (BLOCK_K, BLOCK_N) – we treat B as K×N
        b = tl.load(
            b_batch
            + k_offset[:, None] * stride_bk
            + offs_n[None, :] * stride_bm,
            mask=mask_k[:, None] & mask_n[None, :],
            other=tl.float16(0.0),
        )

        # Accumulate dot product
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ------------------------------------------------------------------
    # Write out
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask        : torch.Tensor [B, N, N] (bool/float) or None
            - weights     : dict of torch.Tensors (float32)
            - config      : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> float16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gates (half)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Optional mask and gate fusion (pure PyTorch – cheap)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: [B, N, N] -> broadcast over hidden_dim
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via custom batched GEMM (Triton)
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    # Permute to (B, hidden_dim, N, N) and flatten batch & hidden_dim
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left.shape
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out_mat = torch.empty_like(left_mat)   # half

    # Strides (in elements)
    stride_a0, stride_a1, stride_a2 = left_mat.stride()   # (batch, N, 1)
    stride_b0, stride_b1, stride_b2 = right_mat.stride()
    stride_c0, stride_c1, stride_c2 = out_mat.stride()

    # GEMM tile sizes – tuned for RTX/A100‑class GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        left_mat.shape[0],                # batch * hidden_dim
        triton.cdiv(N, BLOCK_M),          # tiles over M
        triton.cdiv(N, BLOCK_N),          # tiles over N
    )
    batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_a1, stride_a2, stride_a0,   # A: M stride, K stride, batch stride
        stride_b1, stride_b2, stride_b0,   # B: N stride, K stride, batch stride
        stride_c1, stride_c2, stride_c0,   # C: M stride, N stride, batch stride
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output processing
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate          # scale by out_gate (half)
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Back to float32 for the API contract
    return out.to(torch.float32)
