"""
TriMul outgoing forward pass (AlphaFold3 style).

Algorithm
---------
1. Layer‑norm the input tensor (float32 → float16).
2. Linear projections `left` and `right` from dim → hidden_dim (half‑precision).
3. Compute sigmoid gates (`left_gate`, `right_gate`, `out_gate`) also in half.
4. Fuse elementwise multiplication of `left`/`right` with their gates and
   optionally with the mask using a tiny Triton kernel (in‑place).
5. Core contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication:
   out = (left_permuted) @ (right_permuted).T  where the batch dimension is B*H.
6. Layer‑norm over hidden_dim, multiply by `out_gate`, final linear projection
   back to `dim`.
7. Return the result in float32.

The only Triton‑accelerated part is the gate‑and‑mask fusion; the heavy
O(N³) contraction uses cuBLAS via torch.matmul, which is faster than a
pure‑Python/Triton implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask (if present) with left/right gates
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,       # half *
    right_ptr,      # half *
    left_gate_ptr,  # half *
    right_gate_ptr, # half *
    mask_ptr,       # half *
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask is present, else 0
    H: tl.constexpr,          # hidden dimension (size of innermost channel)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # Load data
    left      = tl.load(left_ptr + offs,      mask=mask, other=0.0)
    right     = tl.load(right_ptr + offs,     mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr + offs,mask=mask, other=0.0)

    if HAS_MASK:
        # mask index = element index // hidden_dim
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left  = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left * left_gate
        right = right * right_gate

    # Write back in‑place
    tl.store(left_ptr  + offs, left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _fuse_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """In‑place fusion of gates (and mask if present) using Triton."""
    total = left.numel()
    BLOCK = 1024                     # must be ≤ max threads per block
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when no mask

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # left and right are modified in‑place
    return left, right


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or numeric) or None
            - weights: dict mapping parameter names to tensors
            - config: dict with keys ``dim`` and ``hidden_dim``,
                      optional ``nomask`` (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) -> half
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half‑precision)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )   # (B, N, N, hidden_dim)

    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gate values (half‑precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse gate (and mask if present) via Triton kernel
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → float16, broadcast over hidden_dim
        mask_h = mask.to(torch.float16).contiguous()
    else:
        mask_h = None

    _fuse_gate_and_mask(left, right, left_gate, right_gate, mask_h, hidden_dim)

    # ------------------------------------------------------------------
    # 5. Core contraction: batched matmul (left @ right^T)
    #    Shapes -> (B, hidden_dim, N, N)
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    # flatten batch*head dimension for torch.bmm
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # batched matrix multiplication with transpose on the second operand
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in requested precision
    return out.to(torch.float32)
