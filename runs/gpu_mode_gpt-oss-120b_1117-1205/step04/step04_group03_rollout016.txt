@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_head, N,
    stride_left_batch, stride_left_row, stride_left_col,
    stride_right_batch, stride_right_row, stride_right_col,
    stride_out_batch, stride_out_row, stride_out_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    total_tiles_m = tl.cdiv(N, BLOCK_M)
    total_tiles_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch_head = total_tiles_m * total_tiles_n

    bh = pid // tiles_per_batch_head
    tile_id = pid % tiles_per_batch_head
    tile_i = tile_id // total_tiles_n
    tile_j = tile_id % total_tiles_n

    i0 = tile_i * BLOCK_M
    j0 = tile_j * BLOCK_N

    # Offsets for row & col within tile
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, N, BLOCK_K):
        k0 = k
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile (M, K)
        left_ptrs = left_ptr + bh * stride_left_batch + (offs_i[:, None] * stride_left_row) + (offs_k[None, :] * stride_left_col)
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load right tile (K, N) - transposed
        right_ptrs = right_ptr + bh * stride_right_batch + (offs_k[:, None] * stride_right_row) + (offs_j[None, :] * stride_right_col)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(left, right)

    # store output
    out_ptrs = out_ptr + bh * stride_out_batch + (offs_i[:, None] * stride_out_row) + (offs_j[None, :] * stride_out_col)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', True)

    # 1 LayerNorm
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=weights['norm.weight'],
                     bias=weights['norm.bias'])
    x = x.to(torch.float16)

    # 2 Projections
    left = F.linear(x, weights['left_proj.weight'].to(torch.float16), bias=None)
    right = F.linear(x, weights['right_proj.weight'].to(torch.float16), bias=None)

    # 3 Gates
    left_gate = F.linear(x, weights['left_gate.weight'].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x, weights['right_gate.weight'].to(torch.float16), bias=None).sigmoid()
    out_gate = F.linear(x, weights['out_gate.weight'].to(torch.float16), bias=None).sigmoid()

    if not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f * left_gate
        right = right * mask_f * right_gate
    else:

        right = right * right_gate

    right_perm = right.permute(0, 3, 1, 2).contiguous()

@triton.jit
def trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_left_batch, stride_left_row, stride_left_col,
    stride_right_batch, stride_right_row, stride_right_col,
    stride_out_batch, stride_out_row, stride_out_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # Compute tile indices
    tiles_m = tl.cdiv(N, BLOCK_M)
    tiles_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = tiles_m * tiles_n

    batch_head = pid // tiles_per_batch
    tile_id = pid % tiles_per_batch
    tile_i = tile_id // tiles_n
    tile_j = tile_id % tiles_n

    i0 = tile_i * BLOCK_M
    j0 = tile_j * BLOCK_N

    # Offsets for rows and cols within tile
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, N, BLOCK_K):
        k0 = k
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (M, K)
        left_ptrs = left_ptr + batch_head * stride_left_batch + (offs_i[:, None] * stride_left_row) + (offs_k[None, :] * stride_left_col)
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

"""
TriMul (outgoing) forward pass – Triton‑accelerated contraction.

The implementation follows the reference PyTorch module:
  1. Layer‑norm the input (float32 → half).
  2. Linear projections to a hidden dimension (half precision).
  3. Compute three sigmoid gates (half).
  4. Apply optional mask and element‑wise gating.
  5. Core O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
     Implemented as a batched GEMM (left @ rightᵀ) using a custom Triton kernel.
  6. Output layer‑norm, out‑gate scaling and final linear projection back to dim.
  7. Return result in float32.

A single Triton kernel performs the batched matrix multiplication for all
(batch * hidden) pairs, using fp32 accumulation for numerical stability.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for left @ rightᵀ.
# left, right, out: [B*H, N, N] in row‑major (contiguous) layout, half precision.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    left_ptr,          # half*   (B*H, N, N)
    right_ptr,         # half*   (B*H, N, N)
    out_ptr,           # half*   (B*H, N, N)
    N: tl.int32,       # sequence length
    # strides for left
    stride_l_batch,    # stride between different (B*H) entries
    stride_l_row,      # stride for the row index i
    stride_l_col,      # stride for the column index k
    # strides for right (used as transposed)
    stride_r_batch,
    stride_r_row,      # stride for the row index k (original row)
    stride_r_col,      # stride for the column index j
    # strides for out
    stride_o_batch,
    stride_o_row,
    stride_o_col,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # tile indexing
    # ------------------------------------------------------------------
    tiles_m = tl.cdiv(N, BLOCK_M)
    tiles_n = tl.cdiv(N, BLOCK_N)
    tiles_per_pair = tiles_m * tiles_n

    pair_id = pid // tiles_per_pair               # which (batch*head) we are processing
    tile_id = pid % tiles_per_pair

    tile_i = tile_id // tiles_n
    tile_j = tile_id % tiles_n

    i0 = tile_i * BLOCK_M
    j0 = tile_j * BLOCK_N

    # row / column offsets inside the tile
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # k‑loop
    # ------------------------------------------------------------------
    for k in range(0, N, BLOCK_K):
        k0 = k
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i, k]   → shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + \
            pair_id * stride_l_batch + \
            (offs_i[:, None] * stride_l_row) + \
            (offs_k[None, :] * stride_l_col)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right_T[j, k] = right[k, j]   → shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + \
            pair_id * stride_r_batch + \
            (offs_k[:, None] * stride_r_row) + \
            (offs_j[None, :] * stride_r_col)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        acc += tl.dot(left, right)

    # ------------------------------------------------------------------
    # store result
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + \
        pair_id * stride_o_batch + \
        (offs_i[:, None] * stride_o_row) + \
        (offs_j[None, :] * stride_o_col)
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask: torch.Tensor [B, N, N] (bool / numeric) – may be ignored if config["nomask"]=True
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> half)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half precision)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gates (half) + sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Apply optional mask and element‑wise gating
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] for broadcasting
        mask_f = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_f * left_gate
        right = right * mask_f * right_gate
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM (left @ rightᵀ) with a Triton kernel.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B, H, N, N) and flatten batch*head dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left_perm.view(-1, N, N).contiguous()   # (B*H, N, N)
    right_flat = right_perm.view(-1, N, N).contiguous()
    out_flat = torch.empty_like(left_flat)

    # strides for the flattened tensors
    stride_lb = left_flat.stride(0)
    stride_li = left_flat.stride(1)
    stride_lk = left_flat.stride(2)

    stride_rb = right_flat.stride(0)
    stride_rk = right_flat.stride(1)   # row = k
    stride_rj = right_flat.stride(2)   # col = j

    stride_ob = out_flat.stride(0)
    stride_oi = out_flat.stride(1)
    stride_oj = out_flat.stride(2)

    # tile sizes – 64 is a good default for fp16 on modern GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    total_pairs = left_flat.shape[0]                # B * hidden_dim
    tiles_m = triton.cdiv(N, BLOCK_M)
    tiles_n = triton.cdiv(N, BLOCK_N)
    grid = (total_pairs * tiles_m * tiles_n,)

    _trimul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_lb,
        stride_li,
        stride_lk,
        stride_rb,
        stride_rk,
        stride_rj,
        stride_ob,
        stride_oi,
        stride_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in original dtype
    return out.to(torch.float32)
