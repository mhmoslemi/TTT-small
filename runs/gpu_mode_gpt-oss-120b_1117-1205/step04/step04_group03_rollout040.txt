"""TriMul outgoing kernel using Triton for mask & gating fusion.

Algorithm:
1. LayerNorm (float32) -> half.
2. Linear projections left/right (half).
3. Compute gate tensors (half) with sigmoid.
4. Fuse mask (if present) with left/right gating via a tiny Triton kernel
   that works in‑place on flattened tensors.
5. Compute the core contraction using torch.einsum, which maps to a batched
   GEMM over the hidden dimension.
6. LayerNorm over hidden dim, multiply with out_gate and project back to
   original dim.
7. Return float32 tensor.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """In‑place fuse mask (if provided) and gating into left/right."""
    total = left.numel()
    BLOCK = 1024  # max threads per block
    grid = (triton.cdiv(total, BLOCK),)
    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer
    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )

def custom_kernel(data):
    """
    Forward pass for outgoing TriMul.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or numeric)
            - weights_dict: dict of model parameters (float32)
            - config_dict: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. LayerNorm over dim (float32) -> half
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # 2. Projections to hidden_dim (half precision, no bias)
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16), bias=None)

    # 3. Gates (half) + sigmoid
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # 4. Fuse mask (if present) with gating via Triton
    if not nomask:
        # Convert mask to half (0.0 or 1.0)
        mask_h = mask.to(torch.float16).contiguous()
        # Flatten tensors for Triton kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask_h.view(-1)

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            hidden_dim,
        )
        # Reshape back to original layout
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask: simple elementwise gating
        left = left * left_gate
        right = right * right_gate

    # 5. Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # Use torch.einsum which maps to batched GEMM internally.
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)

    # 6. Output LayerNorm over hidden dimension, scale by out_gate, project back
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    # 7. Return float32
    return out.to(torch.float32)
```

"""TriMul outgoing kernel using Triton for mask & gating fusion.

Algorithm
---------
1. LayerNorm over the input channel dimension (float32) and cast to float16.
2. Linear projections to the hidden dimension (half‑precision, no bias).
3. Compute three gate tensors (half) and apply sigmoid.
4. Fuse the optional mask with the left/right gating using a tiny Triton
   kernel that works in‑place on the flattened tensors.
5. Core contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented with a single torch.einsum (batched GEMM).
6. Apply a second LayerNorm over the hidden dimension, scale by out_gate,
   and linearly project back to the original channel dimension.
7. Cast the result back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask (if present) with gating.
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # half*  (flattened left tensor)
    right_ptr,              # half*  (flattened right tensor)
    left_gate_ptr,          # half*  (flattened left_gate)
    right_gate_ptr,         # half*  (flattened right_gate)
    mask_ptr,               # half*  (flattened mask, length B*N*N)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if a mask is supplied, else 0
    H: tl.constexpr,          # hidden dimension
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # One mask entry per (b,i,k) -> shared across H hidden channels.
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """In‑place fuse mask (if provided) and gating into `left` and `right`."""
    total = left.numel()
    BLOCK = 1024  # max threads per block on most GPUs
    grid = (triton.cdiv(total, BLOCK),)
    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when no mask
    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args
    ----
    data : tuple (input_tensor, mask, weights_dict, config_dict)
        input_tensor : torch.Tensor [B, N, N, C] (float32)
        mask         : torch.Tensor [B, N, N] (bool or numeric)
        weights_dict : dict of model parameters (float32)
        config_dict  : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16), bias=None)

    # ------------------------------------------------------------------
    # 3. Gates (half) + sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x, weights["left_gate.weight"].to(torch.float16), bias=None
    ).sigmoid()
    right_gate = F.linear(
        x, weights["right_gate.weight"].to(torch.float16), bias=None
    ).sigmoid()
    out_gate = F.linear(
        x, weights["out_gate.weight"].to(torch.float16), bias=None
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) with gates via Triton
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask.to(torch.float16).contiguous()
        # Flatten everything for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask_h.view(-1)

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            hidden_dim,
        )
        # Reshape back to original layout
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask: simple element‑wise gating in PyTorch
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Using torch.einsum which maps to a batched GEMM.
    # ------------------------------------------------------------------
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    # ------------------------------------------------------------------
    # 7. Cast back to float32
    # ------------------------------------------------------------------
    return out.to(torch.float32)
