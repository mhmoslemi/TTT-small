def custom_kernel(data):
    inp, mask, weights, config = data

    device = inp.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm
    x = F.layer_norm(inp, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x = x.to(torch.float16)

    # Linear projections
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # Gates
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # Apply mask and gates
    if not nomask:
        # mask shape [B,N,N] broadcast to hidden_dim
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        # Ensure contiguity
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_h = mask_h.contiguous()
        # Apply kernel
        total = left.numel()
        BLOCK = 65536
        grid = (triton.cdiv(total, BLOCK),)
        HAS_MASK = 1
        mask_ptr = mask_h.view(-1)  # flatten
        mask_gate_fuse_kernel[grid](
            left.view(-1),
            right.view(-1),
            left_gate.view(-1),
            right_gate.view(-1),
            mask_ptr,
            total,
            hidden_dim,
            HAS_MASK,
            BLOCK=BLOCK,
        )
        # After kernel, left/right already multiplied; out_gate remains unchanged
    else:
        left = left * left_gate
        right = right * right_gate

    # Core contraction
    B, N, _, _ = left.shape  # B,N,N,H
    left_t = left.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous()
    left_mat = left_t.view(B * hidden_dim, N, N)
    right_mat = right_t.view(B * hidden_dim, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Output norm
    out = F.layer_norm(

        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    return out.to(torch.float32)

"""
Outgoing TriMul (AlphaFold3) forward pass.

Algorithm
---------
1. Layer‑norm the input (float32 → float16).
2. Linear projections to hidden dimension (no bias).
3. Compute three gate tensors (sigmoid) and fuse optional mask with the
   left/right projections in a single Triton kernel (in‑place).
4. Core contraction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM (left @ rightᵀ) per hidden slice using
   torch.bmm which is cuBLAS‑accelerated.
5. Output LayerNorm, element‑wise gating with out_gate and a final linear
   projection back to the original feature dimension.
6. Return float32 tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,            # *float16
    right_ptr,           # *float16
    left_gate_ptr,       # *float16
    right_gate_ptr,      # *float16
    mask_ptr,            # *float16 (broadcasted over hidden dim)
    total_elems: tl.int64,
    hidden_dim: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """In‑place: left = left * left_gate * mask?, right = right * right_gate * mask?."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask)
    right = tl.load(right_ptr + offs, mask=mask)
    lgate = tl.load(left_gate_ptr + offs, mask=mask)
    rgate = tl.load(right_gate_ptr + offs, mask=mask)

    if HAS_MASK:
        # one mask value per (b,i,j) → divide by hidden_dim to collapse that axis
        mask_idx = offs // hidden_dim
        mval = tl.load(mask_ptr + mask_idx, mask=mask)
        left = left * lgate * mval
        right = right * rgate * mval
    else:
        left = left * lgate
        right = right * rgate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_mask_and_gate(left, right, left_gate, right_gate, mask, hidden_dim):
    """Launch Triton kernel that fuses mask and gate multiplication in‑place."""
    # All tensors must be contiguous half‑precision and flattened.
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask = mask.contiguous()

    total = left.numel()
    BLOCK = 65536  # reasonable large block for 1‑D launch
    grid = (triton.cdiv(total, BLOCK),)

    HAS_MASK = 1

    _mask_gate_fuse_kernel[grid](
        left.view(-1),
        right.view(-1),
        left_gate.view(-1),
        right_gate.view(-1),
        mask.view(-1),
        total,
        hidden_dim,
        HAS_MASK=HAS_MASK,
        BLOCK=BLOCK,
    )
    # The operation is performed in‑place; we return the (now modified) tensors.
    return left, right

    """

    # -------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # -------------------------------------------------
    # 4. Fuse mask with gates (if required) via Triton
    # -------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B,N,N] → broadcast over hidden_dim
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left, right = _apply_mask_and_gate(
            left, right, left_gate, right_gate, mask_h, hidden_dim
        )
    else:
        # simple element‑wise multiplication when no mask
        left = left * left_gate
        right = right * right_gate

    # -------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # -------------------------------------------------
    B, N, _, _ = left.shape  # left shape is (B, N, N, H)

    # Rearrange to (B, H, N, N) for batched GEMM
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_mat = left_t.view(B * hidden_dim, N, N)     # (B*H, N, N)
    right_mat = right_t.view(B * hidden_dim, N, N)   # (B*H, N, N)

    # Batched GEMM (i,k) × (j,k)ᵀ → (i,j)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------
    # 6. Output LayerNorm, gating, final linear projection
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return float32 as required
    return out.to(torch.float32)
