```
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    x = x.to(torch.float16)

    # Projections
    left = torch.nn.functional.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None
    )
    right = torch.nn.functional.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None
    )

    # Gates (before sigmoid)
    left_gate = torch.nn.functional.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None
    ).sigmoid()

    # Fuse mask & gates
    if not nomask and mask is not None:
        # mask broadcast to hidden_dim
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        # ensure left etc contiguous
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)
        _gate_mask_fuse_kernel[grid](
            left, right,
            left_gate, right_gate,
            mask_h,
            total,
            BLOCK=BLOCK,
            HAS_MASK=1,
            H=hidden_dim,
        )
        # After kernel, left and right contain final values
    else:
        left = left * left_gate
        right = right * right_gate

    # Contraction via batched matmul
    # Permute to B, hidden_dim, N, N
    left_perm = left.permute(0, 3, 1, 2)       # B, H, N, N
    right_perm = right.permute(0, 3, 1, 2)     # B, H, N, N
    # Transpose last two dims of right for multiplication
    right_perm_t = right_perm.transpose(2, 3)  # B, H, N, N (swap N,N)
    B, H, N, _ = left_perm.shape
    # reshape to (B*H, N, N)
    left_mat = left_perm.reshape(B*H, N, N)
    right_mat = right_perm_t.reshape(B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat)  # shape (B*H, N, N) half
    # reshape back
    out = out_mat.reshape(B, H, N, N).permute(0, 2, 3, 1)  # B, N, N, H

    # Output LayerNorm over hidden_dim
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16)
    )
    # Multiply with out_gate
    out = out * out_gate
    # Final linear projection back to dim
    out = torch.nn.functional.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None
    )
    # Return float32
    return out.to(torch.float32)
```

```
"""
TriMul outgoing forward pass optimized with Triton for mask/gate fusion
and a batched half‑precision matrix multiplication for the O(N³) contraction.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,       # half *
    right_ptr,      # half *
    left_gate_ptr,  # half *
    right_gate_ptr, # half *
    mask_ptr,       # half *
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,           # hidden_dim
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # Load values
    left      = tl.load(left_ptr + offs,      mask=mask, other=tl.float16(0.0))
    right     = tl.load(right_ptr + offs,     mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate= tl.load(right_gate_ptr + offs,mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # mask is broadcast over the hidden dimension
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left  = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left * left_gate
        right = right * right_gate

    tl.store(left_ptr  + offs, left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def _fuse_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """In‑place fusion of left/right with their gates and optional mask."""
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left   # dummy pointer when no mask
    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )

def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or numeric) or None
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim", and optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half precision)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )   # [B, N, N, hidden_dim]

    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gate values (half precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) and gates via Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape [B, N, N] -> broadcast to hidden_dim
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        # Ensure tensors are contiguous for the kernel
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        _fuse_gate_and_mask(left, right, left_gate, right_gate, mask_h, hidden_dim)
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched half‑precision matrix multiplication.
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2)   # B, H, N, N
    right_perm = right.permute(0, 3, 1, 2) # B, H, N, N
    # Transpose the last two dimensions of right for multiplication
    right_perm_t = right_perm.transpose(2, 3)  # B, H, N, N

    B, H, N, _ = left_perm.shape
    # Flatten batch and channel dimensions for a single bmm call
    left_mat = left_perm.reshape(B * H, N, N)
    right_mat = right_perm_t.reshape(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)   # (B*H, N, N) half

    # Restore original layout [B, N, N, hidden_dim]
    out = out_mat.reshape(B, H, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )
    # Return float32 as required by the API
    return out.to(torch.float32)
```
