@triton.jit
def gate_fuse_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr,
    mask_ptr, total_elems, hidden_dim, HAS_MASK,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    # Multiply gates
    left = left * left_gate
    right = right * right_gate

    if HAS_MASK:
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * mask_val
        right = right * mask_val

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. Input layer norm
    # Use torch.nn.functional.layer_norm

    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # Cast to half for next ops
    x = x.to(torch.float16)

    # 2. Linear projections
    # we need to reshape weight to (hidden_dim, dim)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = torch.nn.functional.linear(x, weights["right_proj.weight"].to(torch.float16), bias=None)

    # 3. Gates: left_gate, right_gate, out_gate (float16)
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # 4. Fuse mask and gates
    if not nomask:
        # Ensure contiguous
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask = mask.to(torch.float16).contiguous()
        total = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)
        HAS_MASK = 1
        # Launch kernel
        gate_fuse_kernel[grid](
            left.data_ptr(),
            right.data_ptr(),
            left_gate.data_ptr(),
            right_gate.data_ptr(),
            mask.data_ptr(),
            total,
            BLOCK=BLOCK,
            HAS_MASK=HAS_MASK,
            H=hidden_dim,
        )
    else:
        # no mask just multiply by gates
        left = left * left_gate
        right = right * right_gate

    # 5. Core contraction: out = einsum('... i k d, ... j k d -> ... i j d', left, right)
    # Equivalent to matmul per hidden dim
    # Permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_perm.shape

    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Compute out_mat = left_mat @ right_mat.transpose(1, 2)
    # Using torch.bmm (batch matmul)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 6. Output layer norm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16)
    )
    # 7. Apply out gate
    out = out * out_gate

    # 8. Final linear projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    # Return to float32 (original dtype)
    return out.to(torch.float32)
```

def custom_kernel(data):
    # unpack
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)  # default true if not provided

    # LayerNorm
    x = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    x = x.to(torch.float16)

    # Linear projections
    left = torch.nn.functional.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = torch.nn.functional.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )
    # Gates
    left_gate = torch.nn.functional.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # Fuse gates and mask
    if not nomask:
        # flatten
        left_f = left.contiguous()
        right_f = right.contiguous()
        left_gate_f = left_gate.contiguous()
        right_gate_f = right_gate.contiguous()
        mask_f = mask.to(torch.float16).contiguous()
        total = left_f.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)
        HAS_MASK = 1
        _gate_fuse_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            total,
            BLOCK=BLOCK,
            HAS_MASK=HAS_MASK,
            H=hidden_dim,
        )
        # No need to reshape because we performed in-place
        left = left_f
        right = right_f
    else:
        left = left * left_gate
        right = right * right_gate

    # free gates
    del left_gate, right_gate

    # Core contraction via batched GEMM
    # left shape (B,N,N,hidden_dim)
    # reorder to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output LayerNorm and gating
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16)
    )
    out = out * out_gate

    # Final linear projection
    out = torch.nn.functional.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )
    # Cast back to float32
    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

Algorithm:
1. Layer‑norm the input tensor (float32 → half).
2. Linear projections to hidden dimension (half‑precision).
3. Compute left/right gates and output gate (half, sigmoid).
4. Fuse mask (if present) and gate multiplications into a single Triton
   kernel that updates `left` and `right` in‑place.
5. Core contraction: for each batch and hidden channel compute
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]  →  batched GEMM
   (implemented with `torch.bmm` on half tensors).
6. Output LayerNorm, apply output gate, final linear projection.
7. Return result in float32.

The kernel only handles the element‑wise gate/mask fusion; the heavy
N³ contraction is performed as a highly‑optimized batched matrix multiply.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_fuse_kernel(
    left_ptr,                # half*  (flattened left tensor)
    right_ptr,               # half*  (flattened right tensor)
    left_gate_ptr,           # half*  (flattened left gate)
    right_gate_ptr,          # half*  (flattened right gate)
    mask_ptr,                # half*  (flattened mask, length = B*N*N)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,  # 1 = mask present, 0 = no mask
    H: tl.constexpr,         # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    if HAS_MASK:
        # Broadcast mask over hidden dimension
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * mask_val
        right = right * mask_val

    # Store results back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections to hidden dimension (half)
    # -----------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # -----------------------------------------------------------------
    # 3. Gates (half precision)
    # -----------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # -----------------------------------------------------------------
    # 4. Fuse mask with gates via Triton (if mask present)
    # -----------------------------------------------------------------
    if not nomask:
        # Ensure contiguous tensors for the kernel
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask = mask.to(torch.float16).contiguous()  # shape [B, N, N]

        total = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)
        HAS_MASK = 1

        _gate_fuse_kernel[grid](
            left,
            right,
            left_gate,
            right_gate,
            mask,
            total,
            BLOCK=BLOCK,
            HAS_MASK=HAS_MASK,
            H=hidden_dim,
        )
        # Gates are no longer needed after fusion
    else:
        left = left * left_gate
        right = right * right_gate

    # Release gate tensors to free memory
    del left_gate, right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: batched GEMM over the sequence dimension
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # -----------------------------------------------------------------
    # Rearrange to (B, H, N, N) for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)                     # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)                   # (B*H, N, N)

    # Batched matrix multiplication with the second operand transposed
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore layout (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in original dtype (float32)
    return out.to(torch.float32)
