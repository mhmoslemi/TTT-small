@triton.jit
def triton_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr):

    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    # Initialize accumulator for each hidden dimension (vector length BLOCK_H)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # Loop over k blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K

        # Load left block: shape (BLOCK_I, BLOCK_K, BLOCK_H)
        left_offset = (pid_b * stride_left_b
                       + i_base * stride_left_i
                       + k_base * stride_left_k) # shape indexing further with hidden
        # left_ptr[i,i_idx,k_idx,h] etc.

        left = tl.load(
            left_ptr + left_offset + tl.arange(0, BLOCK_I)[:,None,None] * stride_left_i
                                   + tl.arange(0, BLOCK_K)[None,:,None] * stride_left_k
                                   + tl.arange(0, BLOCK_H)[None,None,:] * stride_left_h,
            mask= (i_base + tl.arange(0,BLOCK_I) < N)[:,None,None]
               & (k_base + tl.arange(0,BLOCK_K) < N)[None,:,None]
               & (tl.arange(0,BLOCK_H) < H)[None,None,:],
            other=0.0
        )
        # Load right block: shape (BLOCK_J, BLOCK_K, BLOCK_H)
        right_offset = (pid_b * stride_right_b
                        + j_base * stride_right_j
                        + k_base * stride_right_k)
        right = tl.load(
            right_ptr + right_offset + tl.arange(0, BLOCK_J)[:,None,None] * stride_right_j
                                   + tl.arange(0, BLOCK_K)[None,:,None] * stride_right_k
                                   + tl.arange(0, BLOCK_H)[None,None,:] * stride_right_h,
            mask= (j_base + tl.arange(0,BLOCK_J) < N)[:,None,None]
               & (k_base + tl.arange(0,BLOCK_K) < N)[None,:,None]
               & (tl.arange(0,BLOCK_H) < H)[None,None,:],
            other=0.0
        )
        # Multiply and sum over k
        # left shape (I,K,H), right (J,K,H). Want sum over K per H: out_{i,j,h} += sum_k left_{i,k,h} * right_{j,k,h}
        # We can compute with broadcasting: left_{i,k,h} * right_{j,k,h} -> shape (I,J,K,H) then sum over k.
        # To avoid large memory, we can do per k reduction:
        # For each k element, compute outer product across i,j for each h and accumulate.
        # Let's compute vectorized: left_{i,k,:} (I,H), right_{j,k,:} (J,H). For each h we need outer product across i and j.

        # Doing elementwise multiply across K dimension with reduce: we can compute using tl.dot (matrix multiply) per h separately.
        # But tl.dot only works on 2D matrices: we can compute per H slice: left_slice: (I, K) right_slice: (J, K) and then dot. However left and right have per-h values.

        # Since H is small (128), we could loop over h in small intervals (BLOCK_H). Use tl.dot to compute I x J reduction across K.

        # Actually we can cast left and right to float32 shape (I, K, H) and then compute: for each h, acc[i,j,h] += sum_k left[i,k,h] * right[j,k,h].

        # For efficiency, we can compute by matrix multiplication of left[:,:,h] (I,K) with right[:,:,h].transpose (J,K) using tl.dot. For block of H we can loop.

        # Since BLOCK_H maybe 32 or 64, we can handle per-h iteration.

        for h_offset in range(0, BLOCK_H, 1): # loop over each hidden dimension
            # compute dot product for this h
            # left_slice: (I,K) = left[...,h_offset]
            # right_slice: (J,K) = right[...,h_offset]
            # Dot: (I,J) = left_slice @ right_slice.T
            # Add to accumulator at [:,:,h_offset]
            # Use tl.dot
            left_h = left[:,:,h_offset]   # shape I x K
            right_h = right[:,:,h_offset] # shape J x K
            # need right transposed: right_h_T shape K x J, but tl.dot expects (I,K)*(K,J) => (I,J)
            # we can compute dot of left_h and right_h.T using tl.dot with right_h_transposed via stride trick.

            # Not straightforward; need to load transposed right block.
            # Alternative: compute elementwise multiply and sum over k directly:
            # left_h: (I,K), right_h: (J,K)
            # We can compute broadcasting: left_h[:,None,:] * right_h[None,:,:] -> (I,J,K) then sum over K.
            # This may be computationally heavy but okay for small BLOCK_K.

            # Use tl.sum after multiply
            prod = left_h[:,None,:] * right_h[None,:,:]  # shape (I,J,K)
            acc[:,:,h_offset] += tl.sum(prod, axis=2)

        # end h loop

    # Store output
    out_offset = (pid_b * stride_out_b
                  + i_base * stride_out_i
                  + j_base * stride_out_j)
    # Store acc
    tl.store(
        out_ptr + out_offset + tl.arange(0,BLOCK_I)[:,None,None] * stride_out_i
                            + tl.arange(0,BLOCK_J)[None,:,None] * stride_out_j
                            + tl.arange(0,BLOCK_H)[None,None,:] * stride_out_h,
        acc,
        mask= (i_base + tl.arange(0,BLOCK_I) < N)[:,None,None]
            & (j_base + tl.arange(0,BLOCK_J) < N)[None,:,None]
            & (tl.arange(0,BLOCK_H) < H)[None,None,:]
    )

"""
Optimized outgoing TriMul forward pass.

The implementation fuses the five linear projections (left, right, three
gates) into a single matrix multiplication, then uses a tiny Triton kernel
to fuse mask, left‑gate and right‑gate element‑wise multiplications.
The core O(N³) contraction is performed as a batched GEMM via
torch.bmm, which is highly tuned on the B200. All heavy arithmetic is done
in fp16 while the final result is returned in fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask (if any) and gating into left/right tensors.
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # * fp16
    right_ptr,         # * fp16
    left_gate_ptr,     # * fp16
    right_gate_ptr,    # * fp16
    mask_ptr,          # * fp16 (broadcasted per token)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load values (half‑precision)
    left = tl.load(left_ptr + offs, mask=mask, other=0.0, dtype=tl.float16)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0, dtype=tl.float16)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0, dtype=tl.float16)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0, dtype=tl.float16)

    if HAS_MASK:
        # one mask value per token (i.e. per HIDDEN_DIM entries)
        mask_idx = offs // HIDDEN_DIM
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0, dtype=tl.float16)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

            - input_tensor: torch.Tensor (float32) of shape [B, N, N, dim]
            - mask_tensor: torch.Tensor (float32/bool) of shape [B, N, N]
            - weights_dict: dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict: contains "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor (float32) of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape
    total_tokens = B * N * N

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → fp16
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # float32
    x_h = x.to(torch.float16)  # fp16

    # ------------------------------------------------------------------
    # 2. Fuse the five linear projections into a single matmul
    # ------------------------------------------------------------------
    # flatten to (total_tokens, dim)
    x_h_flat = x_h.view(total_tokens, dim)

    # concatenate all weight matrices (shape: (5*hidden_dim, dim))
    w_all = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    # single linear: (total_tokens, dim) @ (dim, 5*hidden_dim).T → (total_tokens, 5*hidden_dim)
    proj_all = F.linear(x_h_flat, w_all)  # fp16

    # split projections / gate logits
    h = hidden_dim
    left_proj = proj_all[:, 0:h]
    right_proj = proj_all[:, h : 2 * h]
    left_gate_logits = proj_all[:, 2 * h : 3 * h]
    right_gate_logits = proj_all[:, 3 * h : 4 * h]
    out_gate_logits = proj_all[:, 4 * h : 5 * h]

    # sigmoid for gates (still fp16)
    left_gate = left_gate_logits.sigmoid()
    right_gate = right_gate_logits.sigmoid()
    out_gate = out_gate_logits.sigmoid()

    # ------------------------------------------------------------------
    # 3. Fuse mask (if any) and gates into left/right via Triton
    # ------------------------------------------------------------------
    # ensure contiguity before flattening for the kernel
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    left_flat = left_proj.view(-1)   # (total_tokens * hidden_dim,)
    right_flat = right_proj.view(-1)
    left_gate_flat = left_gate.view(-1)
    right_gate_flat = right_gate.view(-1)

    if nomask:
        has_mask = 0
        mask_flat = torch.ones(total_tokens, dtype=torch.float16, device=device)
    else:
        has_mask = 1
        # mask shape [B,N,N] → flatten to [total_tokens]
        mask_flat = mask_tensor.to(torch.float16).view(total_tokens).contiguous()

    total_elements = left_flat.numel()  # = total_tokens * hidden_dim
    BLOCK = 65536  # tuneable; must be multiple of 32
    grid = (triton.cdiv(total_elements, BLOCK),)

    _gate_mask_fuse_kernel[grid](
        left_flat,
        right_flat,
        left_gate_flat,
        right_gate_flat,
        mask_flat,
        total_elements,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        HIDDEN_DIM=hidden_dim,
    )

    # reshape back to 4‑D tensors
    left = left_flat.view(B, N, N, hidden_dim)
    right = right_flat.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4. Core O(N³) contraction using batched GEMM (fp16)
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    # rearrange to (B, H, N, N) for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)          # (B*H, N, N)
    right_mat = right_perm.view(BH, N, N)        # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, out‑gate scaling and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )  # fp16

    # apply out_gate (reshape to broadcast)
    out_gate = out_gate.view(B, N, N, hidden_dim)
    out = out * out_gate

    # final linear to original dim
    out_flat = out.view(total_tokens, hidden_dim)
    out_final_flat = F.linear(out_flat, weights["to_out.weight"].to(torch.float16))
    out_final = out_final_flat.view(B, N, N, dim)

    # return in float32 as required
    return out_final.to(torch.float32)
