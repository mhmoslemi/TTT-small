def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    N: tl.int32,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_n = tl.program_id(0)  # column block (j)
    pid_m = tl.program_id(1)  # row block (i)
    pid_b = tl.program_id(2)  # batch*hidden

    # compute block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for within N
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base offset for this batch/h
    # offset = pid_b * N * N
    batch_offset = pid_b * N * N

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, N, BLOCK_K):
        # current k range
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # load left tile: shape (BLOCK_M, BLOCK_K)
        # pointer = left_ptr + batch_offset + offs_m[:,None] * N + cur_k[None,:]
        left_ptrs = left_ptr + batch_offset + offs_m[:, None] * N + cur_k[None, :]
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load right tile (transposed) shape (BLOCK_K, BLOCK_N)
        # pointer = right_ptr + batch_offset + cur_k[:, None] + offs_n[None, :] * N
        right_ptrs = right_ptr + batch_offset + cur_k[:, None] + offs_n[None, :] * N
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product
        acc += tl.dot(left, right)

    # store output
    out_ptrs = out_ptr + batch_offset + offs_m[:, None] * N + offs_n[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""Triton implementation of the outgoing Triangle Multiplicative Update (TriMul).

The forward pass consists of:
1. Layer‑norm the input tensor.
2. Linear projections to a hidden dimension (no bias).
3. Compute sigmoid gates (left, right, out) and optionally apply a mask.
4. Fuse the gating and mask (pure PyTorch) and perform the O(N³) contraction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   The contraction is expressed as a batched matrix‑multiply (left @ rightᵀ) and
   is executed entirely in a custom Triton kernel for maximum performance.
5. Output layer‑norm, out‑gate scaling and final linear projection back to the
   original dimension.
All heavy arithmetic is performed in float16; the final result is returned as
float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,          # *fp16, shape (batch_h, N, N)
    right_ptr,         # *fp16, shape (batch_h, N, N)
    out_ptr,           # *fp16, shape (batch_h, N, N)
    N: tl.int32,       # sequence length
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute out = left @ rightᵀ for each batch_h slice.
    left and right are indexed as (b, i, k) and (b, j, k) respectively.
    The result out[b, i, j] = Σ_k left[b, i, k] * right[b, j, k].
    """
    pid_n = tl.program_id(0)  # block column (j)
    pid_m = tl.program_id(1)  # block row    (i)
    pid_b = tl.program_id(2)  # batch * hidden channel

    # Offsets within the N×N matrices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base offset for the current batch_h slice
    batch_offset = pid_b * N * N

    # Accumulator in FP32 for better precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k_idx in range(num_k):
        cur_k = k_idx * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # ---- load left tile (M × K) -------------------------------------------------
        # offset = batch_offset + i*N + k
        left_ptrs = left_ptr + batch_offset + offs_m[:, None] * N + cur_k[None, :]
        left = tl.load(
            left_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # ---- load right tile (K × N) ------------------------------------------------
        # we need right[j, k] -> offset = batch_offset + k + j*N
        right_ptrs = right_ptr + batch_offset + cur_k[:, None] + offs_n[None, :] * N
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # ---- multiply‑accumulate ----------------------------------------------------
        acc += tl.dot(left, right)  # (M, K) @ (K, N) -> (M, N)

    # ---- store the result ---------------------------------------------------------
    out_ptrs = out_ptr + batch_offset + offs_m[:, None] * N + offs_n[None, :]
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, D] (float32)
            - mask_tensor:  torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dictionary containing model parameters (float32)
            - config: dictionary with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half precision)
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16), bias=None)

    # ------------------------------------------------------------------
    # 3. Gates (half precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_h, weights["left_gate.weight"].to(torch.float16), bias=None
    ).sigmoid()
    right_gate = F.linear(
        x_h, weights["right_gate.weight"].to(torch.float16), bias=None
    ).sigmoid()
    out_gate = F.linear(
        x_h, weights["out_gate.weight"].to(torch.float16), bias=None
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Apply optional mask and gates
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [B, N, N, 1] and cast to FP16
        mask_h = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_h * left_gate
        right = right * mask_h * right_gate
    else:
        left = left * left_gate
        right = right * right_gate

    # Free intermediate tensors we no longer need
    del left_gate, right_gate, x_h

    # ------------------------------------------------------------------
    # 5. Core O(N³) contraction: out = left @ rightᵀ for each hidden channel
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) and merge batch & hidden dimensions
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left.shape
    batch_h = B * H

    left_flat = left.view(batch_h, N, N).contiguous()
    right_flat = right.view(batch_h, N, N).contiguous()
    out_flat = torch.empty_like(left_flat)

    # Triton kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_N),   # columns (j)
        triton.cdiv(N, BLOCK_M),   # rows    (i)
        batch_h,                    # batch * hidden
    )
    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore the original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    # Return in float32 as required by the interface
    return out.to(torch.float32)
