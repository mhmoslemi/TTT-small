"""
TriMul (outgoing) forward implementation.

The computation follows the reference PyTorch module:
1. LayerNorm on the input.
2. Linear projections to a hidden dimension (left/right).
3. Compute three gating vectors (left, right, out) via linear + sigmoid.
4. Fuse left/right with their gates and an optional mask using a Triton kernel.
5. Perform the core O(N³) contraction, which is equivalent to a batched matrix
   multiplication per hidden channel:
       out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   Implemented with torch.bmm on reshaped tensors.
6. Apply a second LayerNorm, scale by the out‑gate, and a final linear
   projection back to the original channel dimension.
All heavy arithmetic after the first LayerNorm is done in float16; the final
output is cast to float32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse left/right with gates and optional mask (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # float16*
    right_ptr,              # float16*
    left_gate_ptr,          # float16*
    right_gate_ptr,         # float16*
    mask_ptr,               # float16* (broadcasted)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,    # threads per program
    HAS_MASK: tl.constexpr, # 1 → mask present, 0 → no mask
    H: tl.constexpr,        # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load data
    left       = tl.load(left_ptr + offs,       mask=mask, other=0.0)
    right      = tl.load(right_ptr + offs,      mask=mask, other=0.0)
    left_gate  = tl.load(left_gate_ptr + offs,  mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # mask is broadcast over the hidden dimension
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    # Store results back in‑place
    tl.store(left_ptr + offs,  left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) – may be ignored
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to half precision for the remaining ops
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gates (half) with sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()  # shape [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) with gates via Triton kernel
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    total = left.numel()  # B * N * N * hidden_dim

    # Flatten tensors for kernel launch
    left_f      = left.contiguous().view(-1)
    right_f     = right.contiguous().view(-1)
    left_gate_f = left_gate.contiguous().view(-1)
    right_gate_f = right_gate.contiguous().view(-1)

    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).contiguous().view(-1)  # [B*N*N]
        has_mask = 1
    else:
        # Dummy pointer – never dereferenced when HAS_MASK=0
        mask_f = left_f
        has_mask = 0

    BLOCK = 8192  # reasonable block size for elementwise work
    grid = (triton.cdiv(total, BLOCK),)

    _gate_mask_fuse_kernel[grid](
        left_f,
        right_f,
        left_gate_f,
        right_gate_f,
        mask_f,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )

    # Reshape back to original geometry
    left = left_f.view(B, N, N, hidden_dim)
    right = right_f.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5. Core contraction: batched GEMM per hidden channel
    #    out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    # ------------------------------------------------------------------
    # Move hidden dimension next to batch for efficient bmm
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)               # (B*H, N, N)
    right_mat = right_perm.view(BH, N, N)             # (B*H, N, N)

    # Batch matrix multiplication with transposition on the second operand
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore layout: (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise gating (half)

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in float32 as required
    return out.to(torch.float32)
```
