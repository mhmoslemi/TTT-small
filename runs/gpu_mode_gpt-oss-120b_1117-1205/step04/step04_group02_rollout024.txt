"""
Outgoing TriMul (AlphaFold3) forward pass.

Algorithm
---------
1. Layer‑norm the input tensor (float32) and cast to float16.
2. Linear projections (half‑precision) produce left and right tensors of
   shape [B, N, N, H].
3. Compute three sigmoid gates (half‑precision) and apply them to left/right.
   If a mask is provided, broadcast‑multiply it (also half‑precision).
4. Core contraction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (A @ Bᵀ) using a custom Triton kernel.
   The kernel treats each (b,h) pair as an independent matrix‑multiply,
   processing tiles of size BLOCK_M×BLOCK_N with a K‑loop of size BLOCK_K.
5. Output layer‑norm (float32), scale by out_gate, final linear projection,
   and cast back to float32.

Only the heavy O(N³) contraction is moved to Triton; the rest uses
torch.nn.functional for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batch_gemm_kernel(
    # pointers
    a_ptr, b_ptr, c_ptr,
    # matrix dimensions
    M, N, K,
    # strides (in elements)
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_a_batch, stride_b_batch, stride_c_batch,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ Bᵀ  where A,B have shape (batch, M, K) and (batch, N, K)."""
    pid_b = tl.program_id(2)          # batch * hidden channel
    pid_m = tl.program_id(0)          # tile over M dimension
    pid_n = tl.program_id(1)          # tile over N dimension

    # -------------------------------------------------------------
    #  Compute block offsets
    # -------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch_ptr = a_ptr + pid_b * stride_a_batch
    b_batch_ptr = b_ptr + pid_b * stride_b_batch
    c_batch_ptr = c_ptr + pid_b * stride_c_batch

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    #  K‑loop
    # -------------------------------------------------------------
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers for the current K‑tile
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

        # Load with masks
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=tl.float16(0.0)).to(tl.float32)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=tl.float16(0.0)).to(tl.float32)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # -------------------------------------------------------------
    #  Store result
    # -------------------------------------------------------------
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    mask_c = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of torch.Tensors (float32) containing all linear
                         and layer‑norm parameters.
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half)
    # ------------------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Compute gates (half) and apply them
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4. Core contraction via Triton‑implemented batched GEMM
    #    left:  [B, N, N, H]   -> permute to [B, H, N, N]
    #    right: [B, N, N, H]   -> permute to [B, H, N, N]
    #    Compute out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    batch_h = B * H

    left_mat = left_perm.view(batch_h, N, N)    # (B*H, N, N)
    right_mat = right_perm.view(batch_h, N, N)  # (B*H, N, N)

    # Allocate output buffer (half)
    out_mat = torch.empty_like(left_mat)

    # Strides (in elements)
    stride_a_batch = left_mat.stride(0)
    stride_am = left_mat.stride(1)   # row stride (M dimension)
    stride_ak = left_mat.stride(2)   # col stride (K dimension)

    stride_b_batch = right_mat.stride(0)
    # For Bᵀ we swap the last two strides
    stride_bk = right_mat.stride(2)  # original col stride -> stride over K after transpose
    stride_bn = right_mat.stride(1)  # original row stride -> stride over N after transpose

    stride_c_batch = out_mat.stride(0)
    stride_cm = out_mat.stride(1)    # row stride (M)
    stride_cn = out_mat.stride(2)    # col stride (N)

    # Tile sizes – tuned for typical B200 GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch_h,
    )
    batch_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,          # M = N
        N,          # N = N
        N,          # K = N
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, out‑gate scaling and final linear projection
    # ------------------------------------------------------------------
    # Convert to float32 for the layer‑norm (higher precision)
    out_f32 = out.to(torch.float32)

    out_f32 = F.layer_norm(
        out_f32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # Apply out_gate (cast to float32)
    out_gate_f32 = out_gate.to(torch.float32)
    out_f32 = out_f32 * out_gate_f32

    # Final projection (half‑precision for speed)
    out_h = out_f32.to(torch.float16)
    out_h = F.linear(
        out_h,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Cast back to float32 for the returned tensor
    return out_h.to(torch.float32)
