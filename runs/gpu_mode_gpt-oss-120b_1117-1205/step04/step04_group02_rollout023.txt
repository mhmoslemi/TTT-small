"""
Outgoing TriMul operator (AlphaFold3) forward pass.

The implementation follows the reference PyTorch code but accelerates the
element‑wise gating (with optional mask) and the O(N³) contraction using
custom Triton kernels.  All heavy arithmetic is performed in float16 with
fp32 accumulation; the final result is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse left/right gates and (optional) mask.
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_fuse_kernel(
    left_ptr,          # * fp16
    right_ptr,         # * fp16
    left_gate_ptr,     # * fp16
    right_gate_ptr,    # * fp16
    mask_ptr,          # * fp16 (broadcasted)
    total_elems,       # i32
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elems

    left = tl.load(left_ptr + offs, mask=valid, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=valid, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=tl.float16(0.0))

    if HAS_MASK:
        # mask is broadcast over the hidden dimension
        mask_idx = offs // HIDDEN_DIM
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=tl.float16(1.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place multiplication of left/right tensors with their gates and
    optional mask using a Triton kernel.

    All tensors must be contiguous and of dtype torch.float16.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when HAS_MASK==0

    gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        HIDDEN_DIM=hidden_dim,
    )

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction
#   out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    batch, M, N, K,
    stride_ah, stride_ai, stride_ak,   # A: (batch, M, K)
    stride_bh, stride_bk, stride_bn,   # B: (batch, K, N) – transposed view of right
    stride_ch, stride_ci, stride_cj,   # C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)           # batch index (B * hidden_dim)
    pid_m = tl.program_id(0)           # block row
    pid_n = tl.program_id(1)           # block column

    # Starting coordinates of the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Batch offsets
    batch_off_A = pid_b * stride_ah
    batch_off_B = pid_b * stride_bh
    batch_off_C = pid_b * stride_ch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    k_tiles = tl.cdiv(K, BLOCK_K)
    for tk in range(0, k_tiles):
        offs_k = tk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = A_ptr + batch_off_A + (offs_m[:, None] * stride_ai) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=tl.float16(0.0))

        # B tile (interpreted as transposed): (BLOCK_K, BLOCK_N)
        b_ptrs = B_ptr + batch_off_B + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=tl.float16(0.0))

        # Accumulate
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = C_ptr + batch_off_C + (offs_m[:, None] * stride_ci) + (offs_n[None, :] * stride_cj)
    tl.store(c_ptrs,

             mask=mask_m[:, None] & mask_n[None, :])

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return as float32
    return out.to(torch.float32)
