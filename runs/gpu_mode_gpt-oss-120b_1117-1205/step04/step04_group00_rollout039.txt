@triton.jit
def tri_mul_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    mask_ptr,
    out_ptr,
    B: tl.constexpr, N: tl.constexpr, H: tl.constexpr,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_rgate_b, stride_rgate_j, stride_rgate_k, stride_rgate_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0) # i block id
    pid_j = tl.program_id(1) # j block id
    pid_bh = tl.program_id(2) # batch*head id

    # Compute block offsets:
    i_start = pid * BLOCK_M
    j_start = pid_j * BLOCK_N
    bh = pid_bh
    b = bh // H
    h = bh % H

    # Compute base pointers
    left_base = left_proj_ptr + b * stride_left_b + h * stride_left_h
    left_gate_base = left_gate_ptr + b * stride_gate_b + h * stride_gate_h
    # right base
    right_base = right_proj_ptr + b * stride_right_b + h * stride_right_h
    right_gate_base = right_gate_ptr + b * stride_rgate_b + h * stride_rgate_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h
    # mask base
    mask_base = mask_ptr + b * stride_mask_b

    # Offsets for rows and columns
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N

    # Allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k tiles
    for k_start in range(0, N, BLOCK_K):
        # compute actual k offsets for this tile (global) => we need to compute offsets appropriately
        cur_k = k_start + offs_k
        # mask for k
        cur_mask_k = cur_k < N

        # left A block: shape (M,K)
        # A_proj: left_proj[i,k]
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + cur_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & cur_mask_k[None, :], other=0.0).to(tl.float32)
        # left gate
        ag_ptrs = left_gate_base + offs_i[:, None] * stride_gate_i + cur_k[None, :] * stride_gate_k
        ag = tl.load(ag_ptrs, mask=mask_i[:, None] & cur_mask_k[None, :], other=0.0).to(tl.float32)
        # sigmoid of gate
        ag_sig = 1.0 / (1.0 + tl.exp(-ag))

        # mask value left
        if HAS_MASK:
            mask_a_ptrs = mask_base + offs_i[:, None] * stride_mask_i + cur_k[None, :] * stride_mask_k
            mask_a = tl.load(mask_a_ptrs, mask=mask_i[:, None] & cur_mask_k[None, :], other=0.0).to(tl.float32)
        else:
            mask_a = 1.0

        a_scaled = a * ag_sig * mask_a

        # right B block: shape (K,N)
        # we need B[k,j]
        b_ptrs = right_base + cur_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        b = tl.load(b_ptrs, mask=cur_mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # right gate
        bg_ptrs = right_gate_base + cur_k[:, None] * stride_rgate_k + offs_j[None, :] * stride_rgate_j
        bg = tl.load(bg_ptrs, mask=cur_mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)
        bg_sig = 1.0 / (1.0 + tl.exp(-bg))

        if HAS_MASK:
            mask_b_ptrs = mask_base + offs_j[:, None] * stride_mask_i + cur_k[None, :] * stride_mask_k
            # But mask_b indexing: mask[j,k] => row index j, col index k
            # offsets: first index i = j, second = k. So pointer = mask_base + j*stride_i + k*stride_k
            mask_b = tl.load(mask_b_ptrs, mask=mask_j[:, None] & cur_mask_k[None, :], other=0.0).to(tl.float32)
        else:
            mask_b = 1.0

        b_scaled = b * bg_sig * mask_b

        # Multiply A_scaled (M,K) with B_scaled (K,N)
        acc += tl.dot(a_scaled, b_scaled)

    # Apply mask for final output (if needed). But accumulator already zero where i/j out-of-range due to masks.

    # Store output
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # Convert to half
    # Layernorm on input:
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x_h = x.to(torch.float16)

    # Concatenated linear weights
    left_concat_weight = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0)
    right_concat_weight = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0)

    # Linear projection
    left_concat = F.linear(x_h, left_concat_weight.to(torch.float16), bias=None)
    right_concat = F.linear(x_h, right_concat_weight.to(torch.float16), bias=None)

    # split
    left_proj_raw = left_concat[..., :hidden_dim]
    left_gate_raw = left_concat[..., hidden_dim:]

    right_proj_raw = right_concat[..., :hidden_dim]
    right_gate_raw = right_concat[..., hidden_dim:]

    # Compute out_gate
    out_gate_raw = F.linear(x_h, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # Prepare mask
    if mask is not None:
        mask_f = mask.to(torch.float16).contiguous()
        has_mask = True
    else:
        # dummy mask (unused)
        mask_f = torch.empty(0, device=device, dtype=torch.float16)
        has_mask = False

    # Allocate output tensor
    out = torch.empty_like(left_proj_raw)  # shape (B,N,N,hidden_dim), half

    # Get strides
    B, N, _, H = left_proj_raw.shape  # B,N,N,H
    # left_proj_raw strides in elements
    stride_left = left_proj_raw.stride()
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = stride_left  # (b,i,k,h)
    # right same strides
    stride_right = right_proj_raw.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = stride_right
    # left_gate_strides:
    stride_lgate = left_gate_raw.stride()
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_h = stride_lgate
    # right_gate_strides:
    stride_rgate = right_gate_raw.stride()
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_h = stride_rgate
    # mask strides
    if has_mask:
        stride_mask = mask_f.stride()
        stride_mask_b, stride_mask_i, stride_mask_k = stride_mask  # shape (B,N,N)
    else:
        stride_mask_b = stride_mask_i = stride_mask_k = 0

    # out strides
    stride_out = out.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = stride_out

    BLOCK_N = 64
    BLOCK_K = 32

        right_gate_raw,

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

The heavy N³ contraction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
is implemented as a custom Triton matmul that directly works on the
original layout (B, N, N, H) without any permute+contiguous copies.
LayerNorm, linear projections and gating are performed in PyTorch using
half precision; the gate·mask multiplication is fused inside the Triton
kernel to minimise memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr, out_ptr,
    # compile‑time constants
    B: tl.constexpr, H: tl.constexpr, N: tl.constexpr,
    # strides (in elements)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    b = pid_bh // H
    h = pid_bh % H

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (b, h) slice
    left_base      = left_ptr      + b * stride_left_b  + h * stride_left_h
    left_gate_base = left_gate_ptr + b * stride_gate_b  + h * stride_gate_h
    right_base     = right_ptr     + b * stride_right_b + h * stride_right_h
    right_gate_base= right_gate_ptr+ b * stride_rgate_b + h * stride_rgate_h
    out_base       = out_ptr       + b * stride_out_b   + h * stride_out_h
    mask_base      = mask_ptr      + b * stride_mask_b

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    offs_k = tl.arange(0, BLOCK_K)

    # loop over reduction dimension k
    for k_start in range(0, N, BLOCK_K):
        cur_k = k_start + offs_k
        mask_k = cur_k < N

        # ----- left side (i, k) -----
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + cur_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        ag_ptrs = left_gate_base + offs_i[:, None] * stride_gate_i + cur_k[None, :] * stride_gate_k
        ag = tl.load(ag_ptrs,
                     mask=mask_i[:, None] & mask_k[None, :],
                     other=0.0).to(tl.float32)

        a_scaled = a * ag

        if HAS_MASK:
            mask_a_ptrs = mask_base + offs_i[:, None] * stride_mask_i + cur_k[None, :] * stride_mask_k
            mask_a = tl.load(mask_a_ptrs,
                             mask=mask_i[:, None] & mask_k[None, :],
                             other=0.0).to(tl.float32)
            a_scaled = a_scaled * mask_a

        # ----- right side (k, j) -----
        b_ptrs = right_base + cur_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        bg_ptrs = right_gate_base + cur_k[:, None] * stride_rgate_k + offs_j[None, :] * stride_rgate_i
        bg = tl.load(bg_ptrs,
                     mask=mask_k[:, None] & mask_j[None, :],
                     other=0.0).to(tl.float32)

        b_scaled = b * bg

        if HAS_MASK:
            mask_b_ptrs = mask_base + cur_k[:, None] * stride_mask_k + offs_j[None, :] * stride_mask_i
            mask_b = tl.load(mask_b_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0).to(tl.float32)
            b_scaled = b_scaled * mask_b

        # accumulate the block product
        acc += tl.dot(a_scaled, b_scaled)

    # write result
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_input, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Fuse projection & gate linear layers (concatenated weights)
    # --------------------------------------------------------------
    left_concat_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)
    right_concat_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)

    # (B, N, N, 2*hidden_dim)
    left_concat = F.linear(x_h, left_concat_weight, bias=None)
    right_concat = F.linear(x_h, right_concat_weight, bias=None)

    # Split projections and gates
    left_proj = left_concat[..., :hidden_dim]      # (B, N, N, H)
    left_gate_raw = left_concat[..., hidden_dim:]  # (B, N, N, H)
    right_proj = right_concat[..., :hidden_dim]
    right_gate_raw = right_concat[..., hidden_dim:]

    # Apply sigmoid to gates (half precision)
    left_gate = left_gate_raw.sigmoid()
    right_gate = right_gate_raw.sigmoid()

    # --------------------------------------------------------------
    # 3. Out‑gate (independent of the contraction)
    # --------------------------------------------------------------
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # --------------------------------------------------------------
    # 4. Prepare mask (half) – may be ignored
    # --------------------------------------------------------------
    has_mask = (not nomask) and (mask_input is not None)
    if has_mask:
        mask = mask_input.to(torch.float16).contiguous()
    else:
        # dummy pointer – will never be read because HAS_MASK=0
        mask = left_proj  # any tensor with same dtype and device

    B, N, _, _ = left_proj.shape  # B, N, N, H

    # --------------------------------------------------------------
    # 5. Allocate output tensor for the contraction
    # --------------------------------------------------------------
    out = torch.empty_like(left_proj)  # (B, N, N, H), half

    # --------------------------------------------------------------
    # 6. Extract strides (in elements) for all tensors
    # --------------------------------------------------------------
    s_left_b, s_left_i, s_left_k, s_left_h = left_proj.stride()
    s_lgate_b, s_lgate_i, s_lgate_k, s_lgate_h = left_gate.stride()
    s_right_b, s_right_i, s_right_k, s_right_h = right_proj.stride()
    s_rgate_b, s_rgate_i, s_rgate_k, s_rgate_h = right_gate.stride()
    s_mask_b, s_mask_i, s_mask_k = mask.stride() if has_mask else (0, 0, 0)
    s_out_b, s_out_i, s_out_j, s_out_h = out.stride()

    # --------------------------------------------------------------
    # 7. Launch the fused Triton kernel
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )
    tri_mul_kernel[grid](
        left_proj,
        left_gate,
        right_proj,
        right_gate,
        mask,
        out,
        B,
        hidden_dim,
        N,
        # strides for left projection
        s_left_b, s_left_i, s_left_k, s_left_h,
        # strides for left gate (sigmoid)
        s_lgate_b, s_lgate_i, s_lgate_k, s_lgate_h,
        # strides for right projection
        s_right_b, s_right_i, s_right_k, s_right_h,
        # strides for right gate (sigmoid)
        s_rgate_b, s_rgate_i, s_rgate_k, s_rgate_h,
        # mask strides
        s_mask_b, s_mask_i, s_mask_k,
        # output strides
        s_out_b, s_out_i, s_out_j, s_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HAS_MASK=1 if has_mask else 0,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 8. Post‑processing: LayerNorm → out‑gate scaling → final linear
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise scaling (half)

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in float32 as required
    return out.to(torch.float32)
