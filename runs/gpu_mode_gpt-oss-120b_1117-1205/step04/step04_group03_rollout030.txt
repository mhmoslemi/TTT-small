@triton.jit
def triton_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,  # Actually left shape (B, H, N, N)
    stride_right_b, stride_right_h, stride_right_j, stride_right_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,

    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ID's
    pid_bh = tl.program_id(0)  # index for (b,h) pair
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    
    # Compute offsets
    b = pid_bh // H  # integer division
    h = pid_bh % H
    
    # Compute i and j block offsets:
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # K offsets loop
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # mask for out-of-bounds for i, j, k
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        # load A and B
        # A: left[b, h, i, k] shape (i,k)
        a_ptrs = left_ptr + b * stride_left_b + h * stride_left_h + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # B_T: we want B_T = right[b, h, j, k] where original right shape (b,h,N,N)
        # Actually we need right_trans: right_T[j, k] = right_original[k, j].
        # So load from right with swapped axes: right_ptr + b*stride_right_b + h*stride_right_h + offs_k[:, None]*stride_right_k + offs_j[None, :]* stride_right_j
        # But we need row dimension = k, col = j (since transposed)
        b_ptrs = right_ptr + b * stride_right_b + h * stride_right_h + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # compute dot: A @ B_T = sum over k
        # Use tl.dot on rows
        acc += tl.dot(a, b)
        # increment pointers for next k block (will be recomputed each iteration)
    # store acc
    # apply scaling? Not needed
    # store to out
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def triton_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    # strides
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_i, stride_right_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    # compile-time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid0 = tl.program_id(0)  # index over B*H
    pid1 = tl.program_id(1)  # i tile
    pid2 = tl.program_id(2)  # j tile

    # compute b and h from pid0
    b = pid0 // H
    h = pid0 % H

    # compute base pointers for this (b,h)
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # offsets for i and j
    offs_i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load A tile: left[b,h,i,k]
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load B tile (transposed): right[b,h,k,j]
        b_ptrs = right_base + offs_k[:, None] * stride_right_i + offs_j[None, :] * stride_right_k
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot product
        acc += tl.dot(a, b)

    # write back
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # layernorm
    x = F.layer_norm(input_tensor, (dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])
    x = x.to(torch.float16)

    # linear proj
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left = F.linear(x, left_proj_w)   # shape [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)

    # gates
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # apply mask and gates
    if not nomask and mask is not None:
        mask = mask.to(torch.float16)
        mask = mask.unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask
        right = right * mask

    left = left * left_gate
    right = right * right_gate

    # prepare for triton contraction
    # shape (B,N,N,hidden_dim) -> (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_perm.shape

    # output tensor
    out_perm = torch.empty_like(left_perm)  # (B, H, N, N) half

    # strides
    s_lb = left_perm.stride(0)
    s_lh = left_perm.stride(1)
    s_li = left_perm.stride(2)
    s_lk = left_perm.stride(3)
    s_rb = right_perm.stride(0)
    s_rh = right_perm.stride(1)
    s_ri = right_perm.stride(2)  # i stride (original i)
    s_rk = right_perm.stride(3)  # k stride (original k)
    s_ob = out_perm.stride(0)
    s_oh = out_perm.stride(1)
    s_oi = out_perm.stride(2)
    s_oj = out_perm.stride(3)

    # launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (B * H, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    triton_trimul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        B,
        H,
        N,
        s_lb, s_lh, s_li, s_lk,
        s_rb, s_rh, s_ri, s_rk,
        s_ob, s_oh, s_oi, s_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B,N,N,H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # output layernorm
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # out_gate scaling
    out = out * out_gate

    # final linear
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)

    # cast back to float32
    return out.to(torch.float32)

"""
TriMul outgoing implementation (forward only).

Algorithm
---------
1. Layer‑norm the 4‑D input (float32) → half precision.
2. Linear projections to the hidden dimension (half) → tensors `left` and `right`.
3. Compute three gating tensors (`left_gate`, `right_gate`, `out_gate`) via linear
   layers + sigmoid (half).
4. Optional mask (bool/float) is applied element‑wise together with the gates.
5. Core contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a fused Triton kernel that performs a batched GEMM for all
   (batch, head) pairs simultaneously.  The kernel tiles the N×N matrices
   with BLOCK_M×BLOCK_N tiles and iterates over K in BLOCK_K steps, using
   `tl.dot` for the inner multiplication.
6. Output layer‑norm, element‑wise scaling by `out_gate`, and final linear
   projection back to the original dimension (half → float32).

Only the contraction (step 5) is performed in Triton; the surrounding
operations use PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _triton_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    # strides of left (B, H, N, N)
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    # strides of right (B, H, N, N)
    stride_right_b, stride_right_h, stride_right_i, stride_right_k,
    # strides of out (B, H, N, N)
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid0 = tl.program_id(0)          # index over B * H
    pid1 = tl.program_id(1)          # i‑tile
    pid2 = tl.program_id(2)          # j‑tile

    # recover batch and head indices
    b = pid0 // H
    h = pid0 % H

    # base pointers for this (b, h)
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # i / j offsets for the output tile
    offs_i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- load A tile: left[b, h, i, k] ----
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # ---- load B tile (transposed): right[b, h, k, j] ----
        b_ptrs = right_base + offs_k[:, None] * stride_right_i + offs_j[None, :] * stride_right_k
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # ---- dot product and accumulate ----
        acc += tl.dot(a, b)

    # write the result back
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : dict of model parameters (float32)
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gates (half) and sigmoid
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(
            x,
            weights["left_gate.weight"].to(torch.float16),
            bias=None,
        )
    )
    right_gate = torch.sigmoid(
        F.linear(
            x,
            weights["right_gate.weight"].to(torch.float16),
            bias=None,
        )
    )
    out_gate = torch.sigmoid(
        F.linear(
            x,
            weights["out_gate.weight"].to(torch.float16),
            bias=None,
        )
    )

    # ------------------------------------------------------------------
    # 4. Apply optional mask and element‑wise gating (torch)
    # ------------------------------------------------------------------
    if (not nomask) and (mask is not None):
        # mask shape [B, N, N] → expand to [B, N, N, 1]
        mask_h = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_h
        right = right * mask_h

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton: out = Σ_k left[...,k,:] * right[...,k,:]
    #    Operates on tensors of shape (B, H, N, N)
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) for a batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    out_perm = torch.empty_like(left_perm)  # (B, H, N, N), half

    # strides (int64)
    s_lb, s_lh, s_li, s_lk = left_perm.stride()
    s_rb, s_rh, s_ri, s_rk = right_perm.stride()
    s_ob, s_oh, s_oi, s_oj = out_perm.stride()

    # launch Triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (B * H, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    _triton_trimul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        B,
        H,
        N,
        s_lb, s_lh, s_li, s_lk,
        s_rb, s_rh, s_ri, s_rk,
        s_ob, s_oh, s_oi, s_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output layer‑norm, out‑gate scaling and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in the original dtype
    return out.to(torch.float32)
