"""
TriMul (outgoing) forward pass optimized with Triton.

Algorithm
---------
1. Layer‑norm the input tensor over the feature dimension `dim`.
2. Linear projections to the hidden dimension `hidden_dim` produce `left` and `right`.
3. Compute three gating vectors (left, right, output) via linear + sigmoid.
4. Fuse mask (if present), left‑gate and right‑gate in a single Triton kernel:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
   (When no mask is supplied the kernel is skipped and only the gate
    multiplication is performed.)
5. Core contraction:
        out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   This is a batched matrix multiply per hidden channel:
        out_{b,h} = left_{b,h} @ right_{b,h}^T
   Implemented with `torch.bmm` (cuBLAS).
6. Layer‑norm over the hidden dimension, scale by the output gate,
   and a final linear projection back to `dim`.
7. Return the result as float32.

A Triton kernel is used for step 4 to avoid separate
element‑wise passes over the large tensors.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # half*
    right_ptr,           # half*
    left_gate_ptr,       # half*
    right_gate_ptr,      # half*
    mask_ptr,            # half*  (broadcasted, length = B*N*N)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask is present, else 0
    H: tl.constexpr,          # hidden dimension size
):
    """In‑place fusion: left = left * left_gate * mask ; right = right * right_gate * mask."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # mask is broadcast over the hidden dimension
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=0.0)
    else:
        mask_val = 1.0

    left = left * left_gate * mask_val
    right = right * right_gate * mask_val

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    Fuse left/right tensors with their gates and an optional mask in-place using Triton.
    All tensors must be contiguous half tensors.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when HAS_MASK==0

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # In‑place modification; nothing to return.


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric)
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) -> half precision
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half)
    # ------------------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),  # (hidden_dim, dim)
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),  # (hidden_dim, dim)
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gates (half) with sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),  # (hidden_dim, dim)
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),  # (hidden_dim, dim)
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),   # (hidden_dim, dim)
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) with gates via Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> half, flattened
        mask_h = mask.to(torch.float16).contiguous().view(-1)  # shape (B*N*N,)

        # Flatten tensors for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_h,
            hidden_dim,
        )
        # Restore original shapes (views share storage)
        left = left_f.view_as(left)
        right = right_f.view_as(right)
        # Cleanup temporaries
        del left_f, right_f, left_gate_f, right_gate_f, mask_h
    else:
        # No mask: simple element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate

    # Gates have been fused; we no longer need gate tensors
    del left_gate, right_gate, x_h

    # ------------------------------------------------------------------
    # 5. Core contraction: out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    #    Implemented as batched GEMM per hidden channel.
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape

    # Collapse batch and hidden dimensions for torch.bmm
    left_mat = left_perm.view(B * H, N, N)            # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)          # (B*H, N, N)

    # Batched matrix multiplication with transposed right operand
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Release large intermediates early
    del left_perm, right_perm, left_mat, right_mat, out_mat, left, right

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise gating (half)

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),  # (dim, hidden_dim)
        bias=None,
    )

    # Return result in float32 as required by the interface
    return out.to(torch.float32)
