@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(2)  # batch*hidden index = b * H + h
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    b = pid_bh // H
    h = pid_bh % H

    # Offsets for output tile
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # scale offsets to memory addresses
    # compute base pointer for this (b,h)
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k offset within tile
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # mask for out-of-bounds in k dimension
        k_valid = k_offset < N

        # load left tile
        left_ptrs = left_base + (i_offset[:, None] * stride_left_i) + (k_offset[None, :] * stride_left_k)
        left = tl.load(left_ptrs, mask=(i_offset[:, None] < N) & k_valid[None, :], other=0.0)
        # left is half
        # load right tile
        right_ptrs = right_base + (j_offset[:, None] * stride_right_j) + (k_offset[None, :] * stride_right_k)
        right = tl.load(right_ptrs, mask=(j_offset[:, None] < N) & k_valid[None, :], other=0.0)
        # compute dot
        acc += tl.dot(left, right)

    # store output
    out_ptrs = out_base + (i_offset[:, None] * stride_out_i) + (j_offset[None, :] * stride_out_j)
    tl.store(out_ptrs, acc.to(tl.float16), mask=(i_offset[:, None] < N) & (j_offset[None, :] < N))

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,),
        weight=weights["norm.weight"], bias=weights["norm.bias"])
    # Convert to half
    x = x.to(torch.float16)

    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = torch.nn.functional.linear(x, weights["right_proj.weight"].to(torch.float16), bias=None)

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # Apply mask and gates
    if not nomask and mask is not None:
        mask_h = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_h * left_gate
        right = right * mask_h * right_gate
    else:
        left = left * left_gate
        right = right * right_gate

    # Ensure contiguous for strides
    left = left.contiguous()
    right = right.contiguous()

    # Prepare out tensor
    out = torch.empty_like(left)  # same shape half

    # Compute strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    B, N, _, H = left.shape

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * H)

    tri_mul_matmul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Now out is half [B,N,N,H]
    # Apply to_out_norm layernorm over hidden_dim (H)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16)
    )

    # Apply out_gate (broadcast)
    out = out * out_gate

    # Final linear projection to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    # Return float32
    return out.to(torch.float32)

def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    b = pid_bh // H
    h = pid_bh % H

    # Offsets for the output tile
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers base for this batch/channel
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)

        j_valid = j_offset < N

"""
TriMul outgoing forward pass (AlphaFold3).

The kernel fuses the O(N³) contraction:
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
using a tiled matrix‑multiply implementation in Triton.
All heavy arithmetic is performed in float16 with float32 accumulation.
Masking and gating are done in PyTorch; the contraction is the
performance‑critical part and runs entirely on the GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: tiled batched matrix multiplication over the hidden dim
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out = left @ right.T for each batch and hidden channel.

    left  : [B, N, N, H] (float16)
    right : [B, N, N, H] (float16)
    out   : [B, N, N, H] (float16)
    """
    pid_bh = tl.program_id(2)          # combined batch & hidden index
    pid_i = tl.program_id(0)           # tile row
    pid_j = tl.program_id(1)           # tile col

    # Decode batch and hidden index
    b = pid_bh // H
    h = pid_bh % H

    # Offsets for the current tile
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this (b, h)
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # Accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)

        i_valid = i_offset < N
        j_valid = j_offset < N
        k_valid = k_offset < N

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + i_offset[:, None] * stride_left_i + k_offset[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=i_valid[:, None] & k_valid[None, :],
                       other=0.0)

        # Load right tile: shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_base + j_offset[:, None] * stride_right_j + k_offset[None, :] * stride_right_k
        right = tl.load(right_ptrs,
                        mask=j_valid[:, None] & k_valid[None, :],
                        other=0.0)

        # Matrix multiply‑accumulate
        acc += tl.dot(left, right)

    # Store the result (cast back to float16)
    out_ptrs = out_base + i_offset[:, None] * stride_out_i + j_offset[None, :] * stride_out_j
    mask_out = i_valid[:, None] & j_valid[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_out)


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] or None
            - weights: dict of model parameters
            - config: dict with keys ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )  # [B, N, N, hidden_dim]

    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gate values (float16)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Apply mask (if any) and gates
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → broadcast over hidden_dim
        mask_h = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_h * left_gate
        right = right * mask_h * right_gate
    else:
        left = left * left_gate
        right = right * right_gate

    # Ensure contiguous memory for stride calculations
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 5. Core O(N³) contraction via Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # [B, N, N, hidden_dim] float16

    # Strides (int64) for the 4‑D tensors
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Tile sizes – chosen to fit within shared memory limits
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),      # i‑tiles
        triton.cdiv(N, BLOCK_N),      # j‑tiles
        B * H,                        # batch × hidden
    )

    tri_mul_matmul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in the requested float32 precision
    return out.to(torch.float32)
