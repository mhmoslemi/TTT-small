"""
TriMul (outgoing) forward pass with fused mask/gate Triton kernel.

Algorithm
---------
1. LayerNorm on the input tensor (float32) → half precision.
2. Linear projections to hidden dimension (half):
   left = x @ W_left, right = x @ W_right.
3. Compute gate values (half) from the same normalized x:
   left_gate, right_gate, out_gate = sigmoid(x @ W_gate).
4. Fuse mask (if present) with gating in a single Triton kernel:
   left = left * left_gate * mask
   right = right * right_gate * mask
   (mask is broadcast over the hidden dimension).
5. Core contraction:
   out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   This is a batched GEMM: for each hidden channel d,
   out[b,d] = left[b,d] @ right[b,d].T .
   Implemented via torch.bmm on reshaped tensors.
6. Output LayerNorm (over hidden dim), multiply by out_gate,
   final linear projection back to original dim.
7. Return result in float32.

Only the mask‑gate fusion is performed in Triton; all other ops use
high‑performance PyTorch kernels (including cuBLAS for the O(N³) reduction).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask (optional) with left/right gating
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # * half
    right_ptr,         # * half
    left_gate_ptr,     # * half
    right_gate_ptr,    # * half
    mask_ptr,          # * half (flattened, length = B*N*N)
    total_elements,   # i32
    hidden_dim,        # i32
    HAS_MASK: tl.constexpr,   # 0 or 1
    BLOCK: tl.constexpr,      # compile‑time block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask)
    right = tl.load(right_ptr + offs, mask=mask)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask)

    if HAS_MASK:
        # mask index = flatten index without the hidden dimension
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=mask)
        left = left * left_gate * m
        right = right * right_gate * m
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _fuse_mask_and_gates(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fuse mask and gates using Triton.
    All tensors must be contiguous half‑precision.
    `mask` may be None (no masking). When present, its shape is (B, N, N).
    """
    total = left.numel()
    BLOCK = 65536  # 2^16, a good trade‑off for most sizes
    grid = (triton.cdiv(total, BLOCK),)

    if mask is None:
        HAS_MASK = 0
        dummy_mask = left  # unused when HAS_MASK=0
    else:
        HAS_MASK = 1
        dummy_mask = mask  # shape (B*N*N,)

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        dummy_mask,
        total,
        hidden_dim,
        HAS_MASK=HAS_MASK,
        BLOCK=BLOCK,
    )
    # tensors are modified in‑place


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor of shape [B, N, N] (bool or float)
            - weights: dict with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict with at least "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    # ------------------------------------------------------------------
    # 1. LayerNorm on input (float32) → half
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (half)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device, dtype=torch.float16)

    left = F.linear(x_h, left_proj_w, bias=None)   # (B, N, N, H)
    right = F.linear(x_h, right_proj_w, bias=None)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 3. Compute gates (half) + sigmoid
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device, dtype=torch.float16)

    left_gate = F.linear(x_h, left_gate_w, bias=None).sigmoid()
    right_gate = F.linear(x_h, right_gate_w, bias=None).sigmoid()
    out_gate = F.linear(x_h, out_gate_w, bias=None).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) with gates using Triton
    # ------------------------------------------------------------------
    # Ensure tensors are contiguous for the kernel
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    if not nomask:
        # mask shape [B, N, N]; broadcast over hidden dimension
        mask_h = mask_tensor.to(device, dtype=torch.float16).contiguous()
        mask_flat = mask_h.view(-1)                 # (B*N*N,)
        _fuse_mask_and_gates(left, right, left_gate, right_gate, mask_flat, hidden_dim)
    else:
        # Simple element‑wise fusion without mask
        left = left * left_gate
        right = right * right_gate

    # Release intermediate gate tensors early
    del left_gate, right_gate, x_h, x

    # ------------------------------------------------------------------
    # 5. Core contraction: batched GEMM over the sequence dimension
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)            # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)          # (B*H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T  → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype=torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # Multiply by out_gate (broadcasted)
    out = out * out_gate

    # Final linear back to original dimension
    to_out_w = weights["to_out.weight"].to(device, dtype=torch.float16)
    out = F.linear(out, to_out_w, bias=None)   # (B, N, N, dim)

    # Return float32 as specified
    return out.to(torch.float32)
