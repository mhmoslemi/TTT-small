Now the kernel for gating and mask: Let's design to compute left and right in-place.

Define:

```
@triton.jit
def fuse_gate_mask_kernel(
    left_ptr,
    right_ptr,
    left_gate_ptr,
    right_gate_ptr,
    mask_ptr,
    H,
    total_elements,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # mask per (b,i,k) index; each hidden repeats H times
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)
```

"""
Optimized forward pass for the outgoing TriMul operator.

Algorithm
---------
1. Layer‑norm the input (fp16) and compute linear projections:
      left  = x @ W_left          (shape [B,N,N,H])
      right = x @ W_right
2. Compute three sigmoid gates from the same normalised input:
      left_gate , right_gate , out_gate  (shape [B,N,N,H])
3. Fuse left/right with their gates and an optional mask in a single
   Triton kernel (in‑place).  This avoids extra memory traffic.
4. Perform the heavy O(N³) contraction as a batched matrix multiplication:
      out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   which is equivalent to many independent GEMMs of shape (N,N) for each
   (batch, hidden) pair.  We reshape the tensors so that torch.bmm can
   execute all GEMMs in one call.
5. Apply a second LayerNorm, scale by out_gate, and a final linear
   projection back to the original dimension.
6. Return the result in float32 as required.

Only the mask‑gate fusion is implemented in Triton; the remaining
operations use highly‑optimized PyTorch kernels (including cuBLAS for the
batched GEMM).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,        # float16 *
    right_ptr,       # float16 *
    left_gate_ptr,   # float16 *
    right_gate_ptr,  # float16 *
    mask_ptr,        # float16 * (broadcasted)
    H: tl.int32,     # hidden dimension size
    total: tl.int32, # total number of elements = B*N*N*H
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present, else 0
):
    """In‑place: left = left * left_gate * mask; right = right * right_gate * mask."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear index
    mask = offs < total

    # Load data
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # mask is broadcast over the hidden dimension
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store results back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    Fuse left/right with their gates and an optional mask using a Triton kernel.
    All tensors must be contiguous fp16 tensors.
    """
    total = left.numel()
    BLOCK = 65536  # can be tuned; must be multiple of 32
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when HAS_MASK==0

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        hidden_dim,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
    )
    # No return – operation is in‑place


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) – may be ignored
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------- 1. Cast weights to fp16 -------------------
    norm_w = weights["norm.weight"].to(torch.float16, device=device)
    norm_b = weights["norm.bias"].to(torch.float16, device=device)

    left_proj_w = weights["left_proj.weight"].to(torch.float16, device=device)
    right_proj_w = weights["right_proj.weight"].to(torch.float16, device=device)

    left_gate_w = weights["left_gate.weight"].to(torch.float16, device=device)
    right_gate_w = weights["right_gate.weight"].to(torch.float16, device=device)
    out_gate_w = weights["out_gate.weight"].to(torch.float16, device=device)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16, device=device)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16, device=device)

    to_out_w = weights["to_out.weight"].to(torch.float16, device=device)

    # ------------------- 2. LayerNorm & projections -------------------
    x = input_tensor.to(torch.float16, device=device)
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # Linear projections to hidden dimension
    left = F.linear(x, left_proj_w, bias=None)      # [B, N, N, H]
    right = F.linear(x, right_proj_w, bias=None)    # [B, N, N, H]

    # Gates (sigmoid)
    left_gate = F.linear(x, left_gate_w, bias=None).sigmoid()
    right_gate = F.linear(x, right_gate_w, bias=None).sigmoid()
    out_gate = F.linear(x, out_gate_w, bias=None).sigmoid()

    # ------------------- 3. Fuse gates (and mask) -------------------
    if not nomask:
        # Prepare mask: shape [B, N, N] -> flat [B*N*N]
        mask_h = mask.to(torch.float16, device=device).contiguous()
        mask_flat = mask_h.view(-1)

        # Ensure contiguous tensors for kernel
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        # Flatten to 1-D for the kernel
        left_f = left.view(-1)
        right_f = right.view(-1)
        left_gate_f = left_gate.view(-1)
        right_gate_f = right_gate.view(-1)

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_flat,
            hidden_dim,
        )
        # Reshape back (views share storage)
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # Simple element‑wise fusion when no mask is present
        left = left * left_gate
        right = right * right_gate

    # Gates are no longer needed
    del left_gate, right_gate, x

    # ------------------- 4. Core O(N³) contraction -------------------
    # Reshape tensors to (B*H, N, N) for a batched GEMM.
    B, N, _, _ = left.shape  # left shape [B, N, N, H]
    # Permute to bring hidden dim to axis 1, then make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_mat = left_perm.view(B * hidden_dim, N, N)     # (B*H, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)   # (B*H, N, N)

    # Batched matrix multiply with transposed second operand
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------- 5. Output processing -------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate  # apply out‑gate (fp16)

    out = F.linear(out, to_out_w, bias=None)  # final projection back to dim

    # Return in float32 as required by the interface
    return out.to(torch.float32)
