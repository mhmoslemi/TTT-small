@triton.jit
def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          N, total_batch,
                          stride_a_batch, stride_a_i, stride_a_k,
                          stride_b_batch, stride_b_k, stride_b_n,
                          stride_c_batch, stride_c_i, stride_c_n,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute start offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile (M,K)
        a_ptrs = A_ptr + batch_idx * stride_a_batch + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B tile transposed (K,N)
        # B stored as [batch, row (k), col (j)] -> we need B_T = (K,N)
        b_ptrs = B_ptr + batch_idx * stride_b_batch + offs_k[:, None] * stride_b_k + offs_j[None, :] * stride_b_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Compute acc += a @ b
        acc += tl.dot(a, b)

    # Write back result
    c_ptrs = C_ptr + batch_idx * stride_c_batch + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) forward pass with Triton acceleration.

- LayerNorm + linear projections are performed in PyTorch (float16).
- A Triton kernel fuses mask (if present) with left/right gates.
- The core O(N³) contraction is implemented as a batched GEMM kernel
  (A @ Bᵀ) over the hidden dimension, using fp16 arithmetic and fp32
  accumulation.
- The result is post‑processed with a second LayerNorm, out‑gate scaling,
  and a final linear projection back to the original channel dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # half* flattened left tensor
    right_ptr,           # half* flattened right tensor
    left_gate_ptr,       # half* flattened left‑gate tensor
    right_gate_ptr,      # half* flattened right‑gate tensor
    mask_ptr,            # half* flattened mask (broadcast over hidden dim)
    total_elements: tl.int32,
    H: tl.constexpr,          # hidden dimension size
    HAS_MASK: tl.constexpr,   # 1 if mask present, else 0
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # One mask value per (batch, i, k) group of H hidden values
        mask_offs = offs // H
        mask_val = tl.load(mask_ptr + mask_offs, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of masking and gating using Triton.
    All tensors must be contiguous and of dtype torch.float16.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when absent
    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        H=hidden_dim,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )
    # tensors are modified in place


@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    N,                         # sequence length
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_i, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)          # index over (batch * hidden)
    pid_m = tl.program_id(1)              # block row
    pid_n = tl.program_id(2)              # block col

    # Offsets for output tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile (M, K)
        a_ptrs = A_ptr + batch_idx * stride_a_batch \
                 + offs_i[:, None] * stride_a_i \
                 + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # Load B tile transposed (K, N)
        # B is stored as [batch, row(k), col(j)] => we need B_T[k, j] = B[batch, k, j]
        b_ptrs = B_ptr + batch_idx * stride_b_batch \
                 + offs_k[:, None] * stride_b_k \
                 + offs_j[None, :] * stride_b_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        b = b.to(tl.float32)

        # Matrix multiply‑accumulate
        acc += tl.dot(a, b)

    # Write back C tile
    c_ptrs = C_ptr + batch_idx * stride_c_batch \
             + offs_i[:, None] * stride_c_i \
             + offs_j[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

        input_tensor,

        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(left_f, right_f, left_gate_f, right_gate_f, mask_h, hidden_dim)

        # Reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask – simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # Release intermediate tensors we no longer need
    del left_gate, right_gate, x_h

    # ------------------------------------------------------------------
    # 5. Core contraction: batched GEMM (A @ Bᵀ)
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) for easy flattening over (B*H)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    total_batch = B * H

    # View as (total_batch, N, N)
    left_view = left_perm.view(total_batch, N, N)
    right_view = right_perm.view(total_batch, N, N)

    # Allocate output
    out_view = torch.empty_like(left_view)

    # Strides for the flattened (batch, N, N) tensors
    stride_batch = N * N
    stride_i = N
    stride_k = 1
    # For the right tensor we need stride for the K (row) dimension = N, and for the N (col) dimension = 1
    stride_b_k = N
    stride_b_n = 1

    # Block sizes – tuned for typical N up to 1024
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (total_batch,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    _batched_gemm_kernel[grid](
        left_view,
        right_view,
        out_view,
        N,
        stride_batch, stride_i, stride_k,
        stride_batch, stride_b_k, stride_b_n,
        stride_batch, stride_i, stride_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_view.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    # Cast to float32 for the final output
    return out.to(torch.float32)
