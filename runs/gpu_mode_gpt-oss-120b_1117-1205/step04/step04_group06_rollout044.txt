    right_proj_weight = weights["right_proj.weight"].to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    to_out_weight = weights["to_out.weight"].to(torch.float16)

    # Input to half
    x = input_tensor.to(torch.float16)
    # Input LayerNorm
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)
    # Projections
    left = F.linear(x, left_proj_weight)  # bias None
    right = F.linear(x, right_proj_weight)
    # Gates
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()
    # Fuse mask and gates if needed
    if not nomask:
        # mask shape [B,N,N], broadcast to hidden dim
        mask_f = mask.to(torch.float16)
        # flatten everything
        total = left.numel()
        # Already contiguous? Ensure contiguous for flatten
        left_f = left.reshape(-1)
        right_f = right.reshape(-1)
        left_gate_f = left_gate.reshape(-1)
        right_gate_f = right_gate.reshape(-1)
        mask_f_flat = mask_f.reshape(-1)
        # kernel
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)
        _gate_mask_fuse_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f_flat,
            total,
            BLOCK=BLOCK,
            HAS_MASK=1,
            H=hidden_dim,
        )
        # reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # Free gates and x etc
    del left_gate, right_gate, x

    # Core contraction: compute out = sum_k left[...,k,:] * right[...,k,:] for each pair i,j
    # Use torch.einsum to avoid large permute
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)

    # Output LayerNorm
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    # apply out_gate
    out = out * out_gate
    # Final linear projection
    out = F.linear(out, to_out_weight)

    # Cast to float32 for return
    return out.to(torch.float32)

"""
Optimized outgoing TriMul forward pass (AlphaFold3).

The implementation:
- Performs all FP16 arithmetic (except final cast) to reduce memory bandwidth.
- Uses a tiny Triton kernel to fuse the mask, left/right gates, and element‑wise
  multiplication, avoiding separate Python loops and extra temporaries.
- Computes the O(N³) contraction with a single `torch.einsum`, which lets PyTorch
  flatten the batch and hidden dimensions and invoke cuBLAS’s highly tuned batched
  GEMM without the costly permute‑contiguous copies present in a naïve
  implementation.
- Keeps the rest of the pipeline (LayerNorms, Linear projections, final
  projection) in PyTorch for simplicity and to leverage existing optimizations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # float16 *
    right_ptr,           # float16 *
    left_gate_ptr,       # float16 *
    right_gate_ptr,      # float16 *
    mask_ptr,            # float16 *
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present else 0
    H: tl.constexpr,          # hidden dimension
):
    """In‑place: left = left * left_gate * mask; right = right * right_gate * mask."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    # Load the tensors
    left = tl.load(left_ptr + offs, mask=active, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=active, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=tl.float16(0.0))

    if HAS_MASK:
        # One mask element per (b,i,k) location; it is broadcast across the hidden dim.
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store back in‑place
    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (float32 or bool) – may be None when nomask=True
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Cast weights to FP16 once
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    to_out_w = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP16) -> x
    # ------------------------------------------------------------------
    x = input_tensor.to(torch.float16)
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (FP16)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)   # shape [B, N, N, H]
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Gates (FP16) + sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask, gates, and element‑wise multiplication via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # Ensure contiguous tensors for the kernel
        left_f = left.reshape(-1).contiguous()
        right_f = right.reshape(-1).contiguous()
        left_gate_f = left_gate.reshape(-1).contiguous()
        right_gate_f = right_gate.reshape(-1).contiguous()
        mask_f = mask.to(torch.float16).reshape(-1).contiguous()

        total_el = left_f.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total_el, BLOCK),)

        _gate_mask_fuse_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            total_el,
            BLOCK=BLOCK,
            HAS_MASK=1,
            H=hidden_dim,
        )

        # Reshape back to the original layout (views share storage)
        left = left_f.view_as(left)
        right = right_f.view_as(right)
        # Release temporaries
        del left_f, right_f, left_gate_f, right_gate_f, mask_f
    else:
        # Simple element‑wise gating when no mask is present
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³) contraction via einsum (batched GEMM under the hood)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)

    # Release memory no longer needed
    del left, right, left_gate, right_gate, x

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling, final projection (FP16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)

    # ------------------------------------------------------------------
    # 7. Cast back to FP32 for the API contract
    # ------------------------------------------------------------------
    return out.to(torch.float32)
