```python
import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)
```

## Triton launch convention

Every failure mode below has appeared in this search. Read this before writing.

A `@triton.jit` function is NOT a Python function. It is launched with a grid
subscript, and its `tl.constexpr` parameters exist only inside it:

```python
@triton.jit
def my_kernel(x_ptr, y_ptr, n, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < n
    tl.store(y_ptr + offs, tl.load(x_ptr + offs, mask=m, other=0.0), mask=m)

def custom_kernel(data):
    ...
    BLOCK = 128                                   # a host variable YOU define
    grid = (triton.cdiv(n, BLOCK),)               # host-side, use triton.cdiv
    my_kernel[grid](x, y, n, BLOCK=BLOCK)         # note the [grid] subscript
```

- `BLOCK`, `BLOCK_M`, `BLOCK_N` are not defined on the host unless you define

  raises NameError.
- `tl.cdiv` is device-side. On the host use `triton.cdiv`.

```python
Optimized TriMul forward (outgoing) for AlphaFold3.

- LayerNorm and linear projections are performed with torch.nn.functional for
  speed, using float16 where possible.
- A tiny Triton kernel fuses the mask (if present) with the left/right gates,
  computing `left = left * left_gate * mask` and `right = right * right_gate *
  mask` in a single pass.
- The expensive O(N³) contraction is expressed as a batched GEMM:
     out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
  which maps to cuBLAS’s highly tuned `torch.bmm`.
- Output LayerNorm, out‑gate scaling and the final linear projection restore
  the original `dim` channels. The final tensor is returned in float32.

All kernels are written for Triton 3.3.1 and assume a CUDA device (B200).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # float16*
    right_ptr,           # float16*
    left_gate_ptr,       # float16*
    right_gate_ptr,      # float16*
    mask_ptr,            # float16* (broadcasted)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask is present
    H: tl.constexpr,          # hidden_dim (size of innermost channel)
):
    """In‑place: left = left * left_gate * mask; right = right * right_gate * mask."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # mask is broadcast over the hidden dimension:
        # each hidden entry shares the same mask entry.
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):

    Fuse left/right with their gates and optional mask in‑place using Triton.
    All tensors must be contiguous and dtype float16.

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when HAS_MASK==0

        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) – may be ignored
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half)
    # ------------------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gates (half) with sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) with gates via Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → half, flattened
        mask_h = mask.to(torch.float16).contiguous().view(-1)  # [B*N*N]

        # Flatten tensors for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_h,
            hidden_dim,
        )
        # Reshape back (views share storage)
        left = left_f.view_as(left)
        right = right_f.view_as(right)
        # No longer need the flattened intermediates
        del left_f, right_f, left_gate_f, right_gate_f, mask_h
    else:
        # No mask: simple element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate

    # Gates have been fused; free memory
    del left_gate, right_gate, x_h

    # ------------------------------------------------------------------
    # 5. Core contraction using batched GEMM
    #    out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    # ------------------------------------------------------------------
    # Bring hidden dimension to the second axis for efficient matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # Batched matrix multiplication with transposition on the second operand
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise gating (half)

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in float32 as required by the interface
    return out.to(torch.float32)
```

This implementation adopts a hierarchical kernel fusion strategy using Triton:

- **Key Insight**: The TriMul operation is essentially a batched matrix multiplication with extra dimensions. We must compute `output[b,i,j] = (left[b,i] @ right[b,j]).t()` but with careful memory access patterns.
- **Manual Blocking**: We implement dynamic tiling of the sequences with explicit reduction over the masked sequence dimensions. This avoids excessive grid sizes while maximizing Tensor Core utilization.
- **Optimization Focus**: Main gains come from:
  - In-place fused operations ($*$ gate $*$ mask) to reduce memory overhead
  - Tensor Core-ready tiling in the `@` operation (contraction stage)
  - Minimal use of explicit loops at element-wise level

The primary benefit is reducing the O(N^3) complexity to a Tensor Core fast path, but noting that the entire implementation requires manual blocking.
