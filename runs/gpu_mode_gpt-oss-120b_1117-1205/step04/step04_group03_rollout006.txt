```
@triton.jit
def gate_mask_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements: tl.int32,
    hidden: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # Compute mask index per hidden channel group
        mask_idx = offs // hidden
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)
```

"""
TriMul (outgoing) forward pass with Triton fusion.

Algorithm
---------
1. Layer‑norm the input tensor (float32 → float16).
2. Linear projections to the hidden dimension (half precision).
3. Compute three gate tensors (half + sigmoid).
4. Fuse mask (if present) and element‑wise gating in a Triton kernel.
5. Batched matrix multiplication C = A @ Bᵀ for each (batch, hidden) pair
   using a custom Triton GEMM kernel (accumulate in fp32, store fp16).
6. Output layer‑norm, out‑gate scaling and final linear projection back to
   the original dimension.
7. Return result in float32.

Both the mask/gate fusion and the core N³ contraction are executed on the
GPU via Triton, satisfying the requirement to use a custom kernel while
keeping the implementation simple and correct.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Kernel 1: fuse mask (optional) with left/right gating.
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # half* (flattened left tensor)
    right_ptr,              # half* (flattened right tensor)
    left_gate_ptr,          # half* (flattened left_gate)
    right_gate_ptr,         # half* (flattened right_gate)
    mask_ptr,               # half* (flattened mask, length B*N*N) – ignored if HAS_MASK=0
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 → mask present, 0 → no mask
    H_DIM: tl.constexpr,      # hidden dimension (channels)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs,    mask=valid, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs,  mask=valid, other=tl.float16(0.0))
    lgate = tl.load(left_gate_ptr + offs,  mask=valid, other=tl.float16(0.0))
    rgate = tl.load(right_gate_ptr + offs, mask=valid, other=tl.float16(0.0))

    if HAS_MASK:
        # mask is broadcast over the hidden dimension
        mask_idx = offs // H_DIM                # integer division
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=tl.float16(0.0))
        left = left * lgate * mask_val
        right = right * rgate * mask_val
    else:
        left = left * lgate
        right = right * rgate

    tl.store(left_ptr + offs, left,  mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)

# ----------------------------------------------------------------------
# Kernel 2: batched matrix multiplication C = A @ Bᵀ
#   A, B, C have shape (batch_head, N, N) – row‑major contiguous.
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,                         # all equal to seq_len
    stride_ah, stride_am, stride_ak, # A: batch, row, col stride
    stride_bh, stride_bk, stride_bn, # B: batch, col, row stride (note transposed)
    stride_ch, stride_cm, stride_cn, # C: batch, row, col stride
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)          # batch_head dimension
    pid_m = tl.program_id(1)        # block row index
    pid_n = tl.program_id(2)        # block col index

    # Compute start offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers to the current batch_head matrix
    a_batch = a_ptr + pid * stride_ah
    b_batch = b_ptr + pid * stride_bh
    c_batch = c_ptr + pid * stride_ch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        cur_k_offs = cur_k + offs_k
        k_mask = cur_k_offs < K

        # Load A tile (M x K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_am + cur_k_offs[None, :] * stride_ak)
        a_mask = (offs_m < M)[:, None] & k_mask[None, :]
        a = tl.load(a_ptrs, mask=a_mask, other=tl.float16(0.0))

        # Load Bᵀ tile (K x N) → original B tile is (N x K)
        b_ptrs = b_batch + (cur_k_offs[:, None] * stride_bn + offs_n[None, :] * stride_bk)
        b_mask = k_mask[:, None] & (offs_n < N)[None, :]
        b = tl.load(b_ptrs, mask=b_mask, other=tl.float16(0.0))

        # Dot product (M x K) @ (K x N) → (M x N)
        acc += tl.dot(a, b)

    # Write back C tile
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    c_mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights_dict, config_dict)
        input_tensor : torch.Tensor [B, N, N, C] (float32)
        mask         : torch.Tensor [B, N, N] (bool or numeric) – may be ignored.
        weights_dict : dict of model parameters (float32 tensors)
        config_dict  : dict with keys
                       - "dim"          : input feature dimension C
                       - "hidden_dim"   : hidden dimension H
                       - "nomask" (optional) : True → no mask, False → use mask

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision for the heavy path
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half precision)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gates (half) and sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) with gates via Triton
    # ------------------------------------------------------------------
    # Flatten tensors to 1‑D contiguous buffers
    left_f = left.contiguous().view(-1)
    right_f = right.contiguous().view(-1)
    left_gate_f = left_gate.contiguous().view(-1)
    right_gate_f = right_gate.contiguous().view(-1)

    has_mask = 0 if nomask else 1
    if has_mask:
        mask_h = mask.to(torch.float16).contiguous().view(-1)  # [B*N*N]
        mask_ptr = mask_h
    else:
        # dummy pointer – will never be dereferenced because HAS_MASK=0
        mask_ptr = left_f

    total_elements = left_f.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_elements, BLOCK),)

    _gate_mask_fuse_kernel[grid](
        left_f,
        right_f,
        left_gate_f,
        right_gate_f,
        mask_ptr,
        total_elements,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H_DIM=hidden_dim,
    )

    # Reshape back to original 4‑D layout
    left = left_f.view_as(left)
    right = right_f.view_as(right)

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM (A @ Bᵀ) using Triton.
    # ------------------------------------------------------------------
    # Bring hidden dimension to the batch axis for easy tiling
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    batch_head = B * H

    left_mat = left_perm.view(batch_head, N, N)
    right_mat = right_perm.view(batch_head, N, N)
    out_mat = torch.empty_like(left_mat)  # will hold (B*H, N, N)

    # Strides (in elements) for the three tensors
    stride_ah = left_mat.stride(0)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)

    stride_bh = right_mat.stride(0)
    stride_bn = right_mat.stride(1)   # row stride
    stride_bk = right_mat.stride(2)   # col stride (==1)

    stride_ch = out_mat.stride(0)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_head,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _batched_matmul_transpose_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_ah, stride_am, stride_ak,
        stride_bh, stride_bk, stride_bn,
        stride_ch, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore the (B, N, N, H) layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in float32 as required by the interface
    return out.to(torch.float32)
