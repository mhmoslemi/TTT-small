```
@triton.jit
def matmul_kernel(
    A, B, C,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)  # block row
    pid_n = tl.program_id(1) # block col
    pid_batch = tl.program_id(2) # batch index

    # offsets for block in M dimension
    block_row = pid * BLOCK_M
    block_col = pid_n * BLOCK_N

    # pointer to each batch
    a_ptr = A + pid_batch * batch_stride_a
    b_ptr = B + pid_batch * batch_stride_b
    c_ptr = C + pid_batch * batch_stride_c

    # compute offsets for rows and cols
    offs_m = block_row + tl.arange(0, BLOCK_M)
    offs_n = block_col + tl.arange(0, BLOCK_N)

    # mask for valid rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulator as FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile: rows x BLOCK_K
        a = tl.load(a_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak),
                    mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        # load B tile: BLOCK_K x cols
        b = tl.load(b_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
                    mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # compute dot
        acc += tl.dot(a, b)

    # store C
    c = acc.to(tl.float16)
    tl.store(c_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
             c, mask=mask_m[:, None] & mask_n[None, :])
```

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # layernorm input
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x = x.to(torch.float16)

    left = F.linear(x, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16), bias=None)

    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()

    HAS_MASK = 0 if nomask else 1
    mask_fused = mask if not nomask else left  # placeholder

    left_flat = left.contiguous().view(-1)
    right_flat = right.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)

"""
Optimized Triton implementation of the outgoing TriMul operator (AlphaFold3).

The forward pass performs:
  1. LayerNorm on the input.
  2. Linear projections to a hidden dimension (left/right).
  3. Gating (sigmoid) and optional mask fusion (performed by a Triton kernel).
  4. Core contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
     implemented as a batched half‑precision GEMM on the GPU (Triton kernel).
  5. Output LayerNorm, out‑gate scaling and final linear projection.
All heavy work is done in fp16; the final result is returned as fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Kernel 1: fuse mask (if present) and gates into the left/right tensors.
# --------------------------------------------------------------------------- #
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half*  (flattened left tensor)
    right_ptr,         # half*  (flattened right tensor)
    left_gate_ptr,     # half*  (flattened left gate)
    right_gate_ptr,    # half*  (flattened right gate)
    mask_ptr,          # half*  (flattened mask, broadcasted over hidden dim)
    total_elements,    # int32
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 -> mask present, 0 -> no mask
    H: tl.constexpr,          # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # one mask value per (b,i,k) -> repeats H times
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store the results back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


# --------------------------------------------------------------------------- #
# Kernel 2: batched half‑precision GEMM (C = A @ B) where A,B have shape
# (batch, M, K) and (batch, K, N) respectively.
# --------------------------------------------------------------------------- #
@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak,          # A: (M stride, K stride)
    stride_bk, stride_bn,          # B: (K stride, N stride)
    stride_cm, stride_cn,          # C: (M stride, N stride)
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_b = tl.program_id(2)   # batch index

    # starting offsets of the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out-of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the current batch
    a_batch = A_ptr + pid_b * stride_batch_a
    b_batch = B_ptr + pid_b * stride_batch_b
    c_batch = C_ptr + pid_b * stride_batch_c

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B tile: (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # FP16 dot → FP32 accumulation
        acc += tl.dot(a, b).to(tl.float32)

    # write back
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# --------------------------------------------------------------------------- #
# Entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------------- #
    # 1. Input LayerNorm (float32 → half)
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # --------------------------------------------------------------------- #
    # 2. Linear projections (half)
    # --------------------------------------------------------------------- #
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # --------------------------------------------------------------------- #
    # 3. Gates (half) and optional mask fusion via Triton
    # --------------------------------------------------------------------- #
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # Fuse mask and gates in‑place
    # Flatten tensors so that the Triton kernel sees a 1‑D view.
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    left_f = left.contiguous().view(-1)
    right_f = right.contiguous().view(-1)
    left_gate_f = left_gate.contiguous().view(-1)
    right_gate_f = right_gate.contiguous().view(-1)

    if not nomask:
        mask_f = mask.to(torch.float16).contiguous().view(-1)  # [B*N*N]
    else:
        # dummy pointer – never dereferenced because HAS_MASK=0
        mask_f = left_f

    has_mask = 0 if nomask else 1
    _gate_mask_fuse_kernel[grid](
        left_f,
        right_f,
        left_gate_f,
        right_gate_f,
        mask_f,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # Reshape back
    left = left_f.view_as(left)
    right = right_f.view_as(right)

    # --------------------------------------------------------------------- #
    # 4. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as a batched GEMM (left @ rightᵀ) in half precision.
    # --------------------------------------------------------------------- #
    # Rearrange to (B, hidden, N, N) for convenient batching
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # Transpose the second operand (right) so that we can use a standard GEMM
    right_t = right_perm.transpose(-2, -1).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    batch_dim = B * H

    A = left_perm.view(batch_dim, N, N)   # (batch_dim, M=N, K=N)
    B_mat = right_t.view(batch_dim, N, N)  # (batch_dim, K=N, N)

    # Allocate output tensor
    C = torch.empty_like(A)  # (batch_dim, N, N), fp16

    # Strides
    stride_batch_A = A.stride(0)
    stride_m_A = A.stride(1)   # = K
    stride_k_A = A.stride(2)   # = 1

    stride_batch_B = B_mat.stride(0)
    stride_k_B = B_mat.stride(1)  # = N
    stride_n_B = B_mat.stride(2)  # = 1

    stride_batch_C = C.stride(0)
    stride_m_C = C.stride(1)   # = N
    stride_n_C = C.stride(2)   # = 1

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        batch_dim,                 # batch
    )

    _batched_gemm_kernel[grid](
        A,
        B_mat,
        C,
        N, N, N,
        stride_m_A, stride_k_A,
        stride_k_B, stride_n_B,
        stride_m_C, stride_n_C,
        stride_batch_A, stride_batch_B, stride_batch_C,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden)
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------- #
    # 5. Output LayerNorm, out‑gate scaling and final projection
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    return out.to(torch.float32)
