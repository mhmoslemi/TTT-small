@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Abatch, stride_Arow, stride_Acol,
    stride_Bbatch, stride_Brow, stride_Bcol,
    stride_Cbatch, stride_Crow, stride_Ccol,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    ACC_TYPE: tl.constexpr,
):
    pid = tl.program_id(0) # batch index
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    block_row = pid_m * BLOCK_M
    block_col = pid_n * BLOCK_N

    # offsets within each tile
    row_offs = block_row + tl.arange(0, BLOCK_M)
    col_offs = block_col + tl.arange(0, BLOCK_N)

    # mask for boundaries
    mask_m = row_offs < M
    mask_n = col_offs < N

    # pointer to start of each batch
    a_ptr = A_ptr + pid * stride_Abatch
    b_ptr = B_ptr + pid * stride_Bbatch
    c_ptr = C_ptr + pid * stride_Cbatch

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=ACC_TYPE)

    # iterate over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute offsets for K tile
        k_offs = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offs < K

        # load A tile (M x K)
        a = tl.load(a_ptr + (row_offs[:, None] * stride_Arow + k_offs[None, :] * stride_Acol), mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load B tile (N x K) (note we want B[j,k], so we load B with row index = col_offs, col index = k_offs)
        b = tl.load(b_ptr + (col_offs[:, None] * stride_Brow + k_offs[None, :] * stride_Bcol), mask=mask_n[:, None] & mask_k[None, :], other=0.0)

        # compute dot product
        # a: M x K, b: N x K, we need sum over k: a[i,k] * b[j,k]
        # use tl.dot
        acc += tl.dot(a, b)  # a (M x K), b (N x K) -> (M x N)

        # advance pointers for next K block (but we used offsets, no need)
    # store to C
    c = acc.to(tl.float16)  # convert to half
    tl.store(c_ptr + (row_offs[:, None] * stride_Crow + col_offs[None, :] * stride_Ccol), c, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass with a fused Triton kernel for the
core N³ contraction. The algorithm:

1. Layer‑norm the input tensor (float32) and cast to float16.
2. Linear projections to hidden_dim (half‑precision).
3. Compute three sigmoid gates (half‑precision).
4. Apply mask (if present) and element‑wise gate multiplication.
5. Compute the contraction
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   as a batched GEMM (left @ rightᵀ) using a custom Triton kernel.
6. Apply a second LayerNorm, scale by the out‑gate, and a final linear
   projection back to the original dimension.
All heavy arithmetic (step 5) runs on the GPU in half‑precision with
float‑32 accumulation for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Abatch, stride_Arow, stride_Acol,
    stride_Bbatch, stride_Brow, stride_Bcol,
    stride_Cbatch, stride_Crow, stride_Ccol,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    ACC_TYPE: tl.constexpr,
):
    # program ids
    pid_b = tl.program_id(0)   # batch * head index
    pid_m = tl.program_id(1)   # tile row index
    pid_n = tl.program_id(2)   # tile col index

    # tile origin
    offset_m = pid_m * BLOCK_M
    offset_n = pid_n * BLOCK_N

    # row / col offsets within the tile
    offs_m = offset_m + tl.arange(0, BLOCK_M)
    offs_n = offset_n + tl.arange(0, BLOCK_N)

    # boundary masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_ptr = A_ptr + pid_b * stride_Abatch
    b_ptr = B_ptr + pid_b * stride_Bbatch
    c_ptr = C_ptr + pid_b * stride_Cbatch

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=ACC_TYPE)

    # loop over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr + (offs_m[:, None] * stride_Arow + offs_k[None, :] * stride_Acol),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile (transposed access): shape (BLOCK_N, BLOCK_K)
        b = tl.load(
            b_ptr + (offs_n[:, None] * stride_Brow + offs_k[None, :] * stride_Bcol),
            mask=mask_n[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Accumulate dot product over K
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # Write out result
    c = acc.to(tl.float16)
    tl.store(
        c_ptr + (offs_m[:, None] * stride_Crow + offs_n[None, :] * stride_Ccol),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operation.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) -> half precision
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half precision)
    # ------------------------------------------------------------------
    left_proj = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right_proj = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gates (half precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Apply mask (if any) and gate multiplication (element‑wise)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] -> [B, N, N, 1] and cast to half
        mask_h = mask.to(torch.float16).unsqueeze(-1)
        left = left_proj * left_gate * mask_h
        right = right_proj * right_gate * mask_h
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton batched GEMM
    #    out = left @ rightᵀ   (per head)
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) so that each head can be processed independently
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    # Collapse batch and head dimensions
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)
    out_mat = torch.empty_like(left_mat)  # half precision

    # Strides for the three tensors (batch, row, col)
    stride_Ab = left_mat.stride(0)
    stride_Ar = left_mat.stride(1)
    stride_Ac = left_mat.stride(2)

    stride_Bb = right_mat.stride(0)
    stride_Br = right_mat.stride(1)
    stride_Bc = right_mat.stride(2)

    stride_Cb = out_mat.stride(0)
    stride_Cr = out_mat.stride(1)
    stride_Cc = out_mat.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_Ab, stride_Ar, stride_Ac,
        stride_Bb, stride_Br, stride_Bc,
        stride_Cb, stride_Cr, stride_Cc,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        ACC_TYPE=tl.float32,
    )
    torch.cuda.synchronize()

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise scaling (half)

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in float32 as required by the reference implementation
    return out.to(torch.float32)
