@triton.jit
def triton_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,
    stride_lm, stride_lk, stride_lb,
    stride_rk, stride_rn, stride_rb,
    stride_om, stride_on, stride_ob,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for block of output
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # Load A tile
        a_ptrs = left_ptr + pid_batch * stride_lb + offs_m[:, None] * stride_lm + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_lk
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < K), other=0.0)
        # Load B tile (right)
        b_ptrs = right_ptr + pid_batch * stride_rb + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_rk + offs_n[None, :] * stride_rn
        b = tl.load(b_ptrs, mask=(k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N), other=0.0)
        # Compute dot product
        acc += tl.dot(a, b)

    # Write back result
    out_ptrs = out_ptr + pid_batch * stride_ob + offs_m[:, None] * stride_om + offs_n[None, :] * stride_on
    tl.store(out_ptrs, acc.to(tl.float16), mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

def _gate_mask_fuse_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr,
    total, hidden_dim: tl.constexpr, HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total
    # Load data
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    if HAS_MASK:
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate
    # Store back
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

@triton.jit
def batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute block start offsets
    block_m_offset = pid_m * BLOCK_M
    block_n_offset = pid_n * BLOCK_N

    # Create range offsets
    offs_m = block_m_offset + tl.arange(0, BLOCK_M)
    offs_n = block_n_offset + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension (reduction)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        block_k_offset = k * BLOCK_K
        # Compute offsets for A and B tiles
        a_ptrs = a_ptr + pid_batch * stride_ab + offs_m[:, None] * stride_am + (block_k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        b_ptrs = b_ptr + pid_batch * stride_bb + (block_k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Load tiles
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (block_k_offset + tl.arange(0, BLOCK_K)[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(block_k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N), other=0.0)

        # Compute dot product
        acc += tl.dot(a, b)

    # Write result
    c_ptrs = c_ptr + pid_batch * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    hidden_dim = config["hidden_dim"]

    del left_gate, right_gate, xh

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr,
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # Compute corresponding mask index (per element)
        mask_idx = offs // HIDDEN_DIM
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

"""
Optimized outgoing TriMul forward pass.

The implementation follows the reference PyTorch module but moves the
most expensive parts to Triton:
  1) Fuse mask and sigmoid gates into the projected tensors (`left` and `right`)
     with a simple element‑wise kernel.
  2) Compute the O(N³) contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
     as a batched GEMM using a custom tiled Triton kernel that works on
     half‑precision data and accumulates in float32.
  3) The remaining LayerNorms, final gate and linear projection are kept in
     PyTorch (half precision) for simplicity.

All tensors are kept in half‑precision except the final output which is
cast back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
#  Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # * fp16
    right_ptr,         # * fp16
    left_gate_ptr,     # * fp16
    right_gate_ptr,    # * fp16
    mask_ptr,          # * fp16 (broadcasted)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    # load values
    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # each mask element corresponds to HIDDEN_DIM consecutive values
        mask_idx = offs // HIDDEN_DIM
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        start_k = k * BLOCK_K
        # A tile (M x K)
        a_ptrs = a_ptr + pid_batch * stride_ab + \
                 offs_m[:, None] * stride_am + \
                 (start_k + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & ((start_k + tl.arange(0, BLOCK_K)) < K),
                    other=0.0)

        # B tile (K x N)
        b_ptrs = b_ptr + pid_batch * stride_bb + \
                 (start_k + tl.arange(0, BLOCK_K))[:, None] * stride_bk + \
                 offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=((start_k + tl.arange(0, BLOCK_K))[:, None] < K) & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_ptr + pid_batch * stride_cb + \
             offs_m[:, None] * stride_cm + \
             offs_n[None, :] * stride_cn
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)


# ----------------------------------------------------------------------
#  Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (float32/bool) – may be ignored.
            - weights: dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict with at least "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half)
    # ------------------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gates (half + sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) and gating into left/right via Triton
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    total_elements = left.numel()
    BLOCK = 65536  # can be tuned
    grid = (triton.cdiv(total_elements, BLOCK),)
    HAS_MASK = 0 if nomask else 1

    if not nomask:
        mask_h = mask.to(torch.float16).contiguous()
        mask_flat = mask_h.view(-1)  # shape B*N*N
        mask_ptr = mask_flat
    else:
        # dummy pointer – never dereferenced because HAS_MASK==0
        mask_ptr = left

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total_elements,
        BLOCK=BLOCK,
        HAS_MASK=HAS_MASK,
        HIDDEN_DIM=hidden_dim,
    )
    # we no longer need the raw gate tensors
    del left_gate, right_gate, x_h

    # ------------------------------------------------------------------
    # 5. Batched GEMM for the O(N³) contraction
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape  # input shape [B,N,N,dim]

    # reshape to (B, H, N, N) and then batch the GEMMs
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) with inner dims transposed

    batch_dim = B * hidden_dim
    left_mat = left_perm.view(batch_dim, N, N)          # (B*H, N, N)   A : (M=N, K=N)
    right_mat = right_perm.view(batch_dim, N, N)        # (B*H, N, N)   B : (K=N, N)

    out_mat = torch.empty_like(left_mat)                # (B*H, N, N)

    # strides (in elements)
    stride_am, stride_ak, stride_ab = left_mat.stride(1), left_mat.stride(2), left_mat.stride(0)
    stride_bk, stride_bn, stride_bb = right_mat.stride(1), right_mat.stride(2), right_mat.stride(0)
    stride_cm, stride_cn, stride_cb = out_mat.stride(1), out_mat.stride(2), out_mat.stride(0)

    # launch tiled GEMM kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        batch_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_am, stride_ak, stride_ab,
        stride_bk, stride_bn, stride_bb,
        stride_cm, stride_cn, stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # apply out‑gate after norm, as in reference
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in float32 as required
    return out.to(torch.float32)
