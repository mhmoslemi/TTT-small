@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, H_: tl.constexpr
):
   # compute program IDs
   pid_i = tl.program_id(0)
   pid_j = tl.program_id(1)
   pid_bn = tl.program_id(2)  # combines batch and channel

   # compute batch and hidden channel
   b = pid_bn // H_
   h = pid_bn % H_

   # compute offsets
   i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
   j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
   # mask for i/j < N
   mask_i = i_offset < N
   mask_j = j_offset < N

   # base pointers for this batch and channel
   left_base = left_ptr + b * stride_left_b + h * stride_left_h
   right_base = right_ptr + b * stride_right_b + h * stride_right_h
   out_base = out_ptr + b * stride_out_b + h * stride_out_h

   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   # loop over k dimension
   for k in range(0, N, BLOCK_K):
       cur_k = k + tl.arange(0, BLOCK_K)
       mask_k = cur_k < N

       # load left tile
       left_ptrs = left_base + (i_offset[:, None] * stride_left_i) + (cur_k[None, :] * stride_left_k)
       left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
       left_tile = left_tile.to(tl.float32)

       # load right tile
       right_ptrs = right_base + (j_offset[:, None] * stride_right_i) + (cur_k[None, :] * stride_right_k)
       right_tile = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)
       right_tile = right_tile.to(tl.float32)

       # accumulate: left_tile (M,K) dot right_tile.T (K,N)
       acc += tl.dot(left_tile, tl.trans(right_tile))

   # store output
   out_ptrs = out_base + (i_offset[:, None] * stride_out_i) + (j_offset[None, :] * stride_out_j)
   tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul outgoing forward pass with Triton acceleration.

Algorithm:
1. Layer‑norm the input tensor (float32 → float16).
2. Linear projections to hidden_dim and gate values (all float16).
3. Fuse mask (if present) and gate multiplications in‑place via a small Triton kernel.
4. Core O(N³) contraction:
   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   Implemented as batched GEMM (left × rightᵀ) with a tiled Triton kernel.
5. Output layer‑norm, multiplication by out‑gate, final linear projection.
6. Return float32 output.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Fuse mask and gate multiplications (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,        # half *
    right_ptr,       # half *
    left_gate_ptr,   # half *
    right_gate_ptr,  # half *
    mask_ptr,        # half *
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,  # 1 if mask is present, else 0
    H: tl.constexpr,         # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elems

    # Load tensors
    left      = tl.load(left_ptr + offs,      mask=valid, other=0.0)
    right     = tl.load(right_ptr + offs,     mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate= tl.load(right_gate_ptr + offs,mask=valid, other=0.0)

    if HAS_MASK:
        # one mask value per H channel entries
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=0.0)
        left  = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left * left_gate
        right = right * right_gate

    tl.store(left_ptr  + offs, left,  mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def _fuse_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """In‑place fuse: left = left * left_gate * mask, right = right * right_gate * mask."""
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy ptr when no mask

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # left and right are modified in place


# ----------------------------------------------------------------------
# 2) Core O(N³) contraction kernel (batched GEMM with transposed RHS)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, H_: tl.constexpr,
):
    pid_i = tl.program_id(0)          # block row (i)
    pid_j = tl.program_id(1)          # block col (j)
    pid_bn = tl.program_id(2)         # batch * hidden channel

    # Decode batch and hidden channel
    b = pid_bn // H_
    h = pid_bn % H_

    # Offsets for i, j dimensions
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_offset < N
    mask_j = j_offset < N

    # Base pointers for this (b, h)
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, N, BLOCK_K):
        k_offset = k + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # Load left tile [M, K]
        left_ptrs = left_base + (i_offset[:, None] * stride_l_i) + (k_offset[None, :] * stride_l_k)
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)

        # Load right tile [N, K]
        right_ptrs = right_base + (j_offset[:, None] * stride_r_i) + (k_offset[None, :] * stride_r_k)
        right_tile = tl.load(right_ptrs,
                             mask=mask_j[:, None] & mask_k[None, :],
                             other=0.0).to(tl.float32)

        # GEMM: left_tile (M,K) * right_tileᵀ (K,N) → (M,N)
        acc += tl.dot(left_tile, tl.trans(right_tile))

    # Store result
    out_ptrs = out_base + (i_offset[:, None] * stride_o_i) + (j_offset[None, :] * stride_o_j)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)

    """

    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> float16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],

    # ------------------------------------------------------------------
    left_gate = F.linear(

        weights["left_gate.weight"].to(torch.float16),

    out_gate = F.linear(
        x,

    # ------------------------------------------------------------------
    if (not nomask) and (mask is not None):

    # ------------------------------------------------------------------

    BLOCK_K = 32

        B, N, hidden_dim,

        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
