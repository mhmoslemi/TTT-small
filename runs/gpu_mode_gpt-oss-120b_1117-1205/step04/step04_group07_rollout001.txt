"""
Outgoing TriMul forward pass (no gradient) optimized with Triton.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Linear projections to a hidden dimension (half‑precision).
3. Compute left/right/out gating values (sigmoid, half‑precision).
4. Fuse mask (if present) and gating into `left` and `right` with a tiny Triton kernel.
5. Perform the O(N³) contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched matrix multiply using cuBLAS (torch.bmm) on reshaped data.
6. Apply output LayerNorm, multiply by `out_gate`, and project back to the original
   dimension.
7. Return the result in float32.

The Triton kernel only handles the element‑wise mask‑gate fusion, which is the
most memory‑bound part of the computation. All other heavy work (the N³
contraction) is delegated to highly‑optimized cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # * half
    right_ptr,         # * half
    left_gate_ptr,     # * half
    right_gate_ptr,    # * half
    mask_ptr,          # * half (broadcasted)
    total_elements: tl.int32,
    HIDDEN_DIM: tl.constexpr,   # hidden dimension size
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    zero = tl.float16(0.0)
    one = tl.float16(1.0)

    left = tl.load(left_ptr + offs, mask=mask, other=zero)
    right = tl.load(right_ptr + offs, mask=mask, other=zero)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=zero)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=zero)

    if HAS_MASK:
        # broadcast mask across the hidden dimension
        mask_idx = offs // HIDDEN_DIM
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=one)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_mask_fuse(left, right, left_gate, right_gate, mask, hidden_dim):

    In‑place fus​e of mask and gates using Triton.
    All tensors must be contiguous 1‑D half‑precision views.
    """
    total = left.numel()

    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when no mask

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        HIDDEN_DIM=hidden_dim,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )
    # tensors are modified in‑place; nothing to return.


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (bool/float) – may be ignored.
            - weights: dict with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict with keys "dim", "hidden_dim", optional "nomask".
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1) Input LayerNorm (float32 -> half)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2) Linear projections to hidden_dim (half precision)
    # --------------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # --------------------------------------------------------------
    # 3) Gates (half precision, sigmoid)
    # --------------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # --------------------------------------------------------------
    # 4) Fuse mask and gates (Triton)
    # --------------------------------------------------------------
    if not nomask:
        # Ensure contiguous half‑precision tensors
        mask_h = mask.to(torch.float16).contiguous()
        mask_flat = mask_h.view(-1)                     # [B*N*N]

        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_mask_fuse(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_flat,
            hidden_dim,
        )

        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # Release intermediate tensors we no longer need
    del left_gate, right_gate, x_h

    # --------------------------------------------------------------
    # 5) Core O(N³) contraction via batched GEMM (cuBLAS)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    # reshape to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)    # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)  # (B*H, N, N)

    # batched matrix multiplication: (B*H, N, N) x (B*H, N, N)^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6) Output LayerNorm, out‑gate scaling and final linear projection
    # --------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out_norm = out_norm * out_gate
    out_final = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Convert back to float32 for the final output
    return out_final.to(torch.float32)
