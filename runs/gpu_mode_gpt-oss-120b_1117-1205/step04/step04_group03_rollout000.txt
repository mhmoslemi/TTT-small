"""
TriMul (outgoing) forward pass with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32) → half precision.
2. Linear projections to the hidden dimension (left/right) → half.
3. Compute three gate tensors (left_gate, right_gate, out_gate) → half + sigmoid.
4. Fuse mask (if present) with the two gates in a single Triton kernel:
   left  = left  * left_gate  * mask
   right = right * right_gate * mask
   (mask is broadcast across the hidden channel dimension.)
5. Core contraction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM (left @ rightᵀ) performed with cuBLAS via
   torch.bmm on half‑precision tensors.
6. Output layer‑norm over the hidden dimension, element‑wise scaling by
   out_gate, and final linear projection back to the original dimension.
7. Cast the result back to float32.

Only the mask‑gate fusion is done in Triton; the heavy O(N³) contraction
uses cuBLAS which is already optimal on the target hardware.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask (if present) with left/right gating.
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half*  (flattened left tensor)
    right_ptr,         # half*  (flattened right tensor)
    left_gate_ptr,     # half*  (flattened left_gate)
    right_gate_ptr,    # half*  (flattened right_gate)
    mask_ptr,          # half*  (flattened mask, length B*N*N) – ignored if HAS_MASK=0
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if a mask is supplied, else 0
    H: tl.constexpr,          # hidden dimension
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load operands
    left      = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    left_g    = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_g   = tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)

    if HAS_MASK:
        # Each (b,i,j) entry shares a single mask value across the H channels.
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left  = left  * left_g * mask_val
        right = right * right_g * mask_val
    else:
        left  = left  * left_g
        right = right * right_g

    tl.store(left_ptr  + offs, left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of mask and gates using Triton.

    Parameters
    ----------
    left, right, left_gate, right_gate : torch.float16 tensors
        Shape (B, N, N, hidden_dim), contiguous.
    mask : torch.float16 tensor or None
        Shape (B, N, N). If None, the mask is ignored.
    hidden_dim : int
    """
    total = left.numel()
    BLOCK = 1024  # reasonable for half‑precision access
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when unused

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        mask_tensor  : torch.Tensor of shape [B, N, N] (any dtype, may be unused)
        weights_dict : dict mapping parameter names to torch.Tensor (float32)
        config_dict  : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half precision)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gate computations (half) + sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) with gates via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # Ensure mask is half and contiguous, then flatten
        mask_h = mask_tensor.to(torch.float16).contiguous()
        # Flatten tensors for the kernel
        left_f  = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        lg_f    = left_gate.contiguous().view(-1)
        rg_f    = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(
            left_f,
            right_f,
            lg_f,
            rg_f,
            mask_h.view(-1),
            hidden_dim,
        )

        # Reshape back to original layout
        left  = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask: simple element‑wise gating
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM (left @ rightᵀ) with cuBLAS.
    # ------------------------------------------------------------------
    # Convert to (B, H, N, N) layout for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    # Merge batch and head dimensions
    left_mat = left_perm.view(B * H, N, N)          # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)        # (B*H, N, N)

    # Batched GEMM: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore shape (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise scaling

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 7. Cast back to float32
    # ------------------------------------------------------------------
    return out.to(torch.float32)
