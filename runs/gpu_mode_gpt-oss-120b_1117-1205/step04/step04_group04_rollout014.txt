@triton.jit
def _apply_mask_gate_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems
    left = tl.load(left_ptr + offs, mask=mask)
    right = tl.load(right_ptr + offs, mask=mask)
    lgate = tl.load(left_gate_ptr + offs, mask=mask)
    rgate = tl.load(right_gate_ptr + offs, mask=mask)
    if HAS_MASK:
        mask_idx = offs // H
        m_val = tl.load(mask_ptr + mask_idx, mask=mask)
        left = left * lgate * m_val
        right = right * rgate * m_val
    else:
        left = left * lgate
        right = right * rgate
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)
```

    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

        input_tensor,
        (dim,),

        bias=weights["norm.bias"]
    )

        grid = (triton.cdiv(total, BLOCK),)
        HAS_MASK = 1
        _mask_gate_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            BLOCK=BLOCK,
            HAS_MASK=HAS_MASK,
            H=hidden_dim,
        )
        # reshape back
        left = left_flat.view_as(left_proj)
        right = right_flat.view_as(right_proj)
    else:
        # No mask, just multiply gates
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Core contraction
    # left,right are half
    left_t = left.permute(0, 3, 1, 2).contiguous()
    right_t = right.permute(0, 3, 1, 2).contiguous()

    left_mat = left_t.view(B * hidden_dim, N, N)
    right_mat = right_t.view(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Output LayerNorm and gating
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16))

    return out.float()

"""
Outgoing TriMul (AlphaFold3) forward pass – Triton‑accelerated core.

Algorithm
---------
1. Layer‑norm the input tensor (float32 → float16).
2. Fuse five linear maps (left_proj, right_proj, left_gate, right_gate, out_gate)
   into a single half‑precision GEMM by concatenating the weight matrices.
3. Split the result, apply sigmoid to the three gate tensors.
4. Fuse optional mask with left/right gate multiplications in a Triton kernel
   (in‑place on the projected tensors).
5. Perform the heavy O(N³) contraction as a batched GEMM:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   using torch.bmm on half‑precision tensors (cuBLAS).
6. Output LayerNorm, element‑wise gating with out_gate, final linear projection.
7. Return the result in float32.

The kernel `_mask_gate_fuse_kernel` fuses mask, left‑gate and right‑gate
multiplications, reducing memory traffic. The rest of the computation uses
high‑performance PyTorch kernels (layer‑norm, linear, batched GEMM).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # *half
    right_ptr,         # *half
    left_gate_ptr,     # *half
    right_gate_ptr,    # *half
    mask_ptr,          # *half (broadcasted over hidden dim)
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,   # hidden dimension
):
    """In‑place: left = left * left_gate * mask, right = right * right_gate * mask."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    lgate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        mask_idx = offs // H                      # one mask entry per (b,i,j)
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * lgate * m
        right = right * rgate * m
    else:
        left = left * lgate
        right = right * rgate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict mapping parameter names to float32 tensors
            - config: dict containing at least "dim" and "hidden_dim",
                      optionally "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    inp, mask, weights, config = data
    device = inp.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 → half)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)                # [B, N, N, dim]

    B, N, _, _ = x.shape

    # -----------------------------------------------------------------
    # 2. Fuse five linear maps into a single GEMM
    # -----------------------------------------------------------------
    # Gather weight matrices and convert to half precision once
    left_proj_w = weights["left_proj.weight"].to(torch.float16)   # (H, dim)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    combined_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )  # (5*H, dim)

    # Linear layer on the flattened input
    x_flat = x.view(-1, dim)                     # (B*N*N, dim)
    proj = F.linear(x_flat, combined_weight)    # (B*N*N, 5*H)

    # Split the concatenated result
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj, hidden_dim, dim=1
    )  # each (B*N*N, H)

    # Sigmoid on gating tensors
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    # Reshape to [B, N, N, H]
    left_proj = left_proj.view(B, N, N, hidden_dim)
    right_proj = right_proj.view(B, N, N, hidden_dim)
    left_gate = left_gate.view(B, N, N, hidden_dim)
    right_gate = right_gate.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # -----------------------------------------------------------------
    # 3. Apply mask and gate multiplications (Triton kernel if mask needed)
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        # Prepare tensors for the kernel
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)      # [B, N, N, 1]
        left_flat = left_proj.view(-1)
        right_flat = right_proj.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)
        mask_flat = mask_fp16.view(-1)                       # length B*N*N

        total = left_flat.numel()
        BLOCK = 65536                                         # must be multiple of 32
        grid = (triton.cdiv(total, BLOCK),)

        _mask_gate_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            BLOCK=BLOCK,
            HAS_MASK=1,
            H=hidden_dim,
        )

        left = left_flat.view_as(left_proj)
        right = right_flat.view_as(right_proj)
    else:
        # No mask: simple element‑wise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 4. Core contraction: batched GEMM over the hidden dimension
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # -----------------------------------------------------------------
    # Rearrange to (B*H, N, N) for a batched matrix multiply
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_mat = left_t.view(B * hidden_dim, N, N)     # (B*H, N, N)
    right_mat = right_t.view(B * hidden_dim, N, N)   # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # -----------------------------------------------------------------
    # 5. Output LayerNorm, gating with out_gate, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return as float32 as required
    return out.to(torch.float32)
