"""
Optimized outgoing TriMul forward (AlphaFold3).

The kernel hierarchy:
1. LayerNorm + linear projections (FP16) – done in PyTorch for high‑throughput.
2. Optional mask & gate fusion – a tiny Triton kernel fuses `left/right *
   gate * mask` in‑place, avoiding an extra memory pass.
3. Core O(N³·H) contraction – expressed as a batched GEMM:
      out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   which is implemented by reshaping to (B·H, N, N) and calling `torch.bmm`.
4. Output LayerNorm, out‑gate scaling and final linear projection – also in
   FP16 via torch.nn.functional.

All heavy arithmetic stays in FP16; the final result is cast back to FP32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,                # fp16*
    right_ptr,               # fp16*
    left_gate_ptr,           # fp16*
    right_gate_ptr,          # fp16*
    mask_ptr,                # fp16* (broadcasted over hidden dim)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask is present, else 0
    H: tl.constexpr,          # hidden dimension size
):
    """In‑place: left = left * left_gate * mask ; right = right * right_gate * mask."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask_off = offs < total_elements

    # load
    left = tl.load(left_ptr + offs, mask=mask_off, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask_off, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask_off, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask_off, other=0.0)

    if HAS_MASK:
        # each hidden entry shares the same mask entry
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask_off, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask_off)
    tl.store(right_ptr + offs, right, mask=mask_off)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    Fuse left/right with their gates and optional mask in‑place using Triton.
    All tensors are expected to be contiguous FP16.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when HAS_MASK==0

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # in‑place – nothing to return


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
              - input_tensor: torch.Tensor [B, N, N, dim] (float32)
              - mask: torch.Tensor [B, N, N] (bool or numeric) – may be None
              - weights: dict of model parameters (float32)
              - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Cast to fp16 and LayerNorm the input
    # ------------------------------------------------------------------
    x_fp16 = input_tensor.to(torch.float16)
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)
    x = F.layer_norm(x_fp16, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2. Linear projections (to hidden_dim) – fp16
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x, left_proj_w, bias=None)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w, bias=None)

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid) – fp16
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x, left_gate_w, bias=None).sigmoid()
    right_gate = F.linear(x, right_gate_w, bias=None).sigmoid()
    out_gate = F.linear(x, out_gate_w, bias=None).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) with gates via Triton kernel
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask → fp16, flattened; broadcasted over hidden_dim
        mask_fp16 = mask.to(torch.float16).contiguous().view(-1)  # [B*N*N]

        # flatten tensors for kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_fp16,
            hidden_dim,
        )
        # Restore original shapes (views share storage)
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask – simple gating
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³·H) contraction via batched GEMM
    #    out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # Batched GEMM: left @ right^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate  # element‑wise gating (fp16)

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w, bias=None)

    # Cast back to FP32 for the API contract
    return out.to(torch.float32)
