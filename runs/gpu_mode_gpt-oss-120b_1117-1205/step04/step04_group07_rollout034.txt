@triton.jit
def gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr, total_elements: tl.int32,
    HIDDEN_DIM: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr):

    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements
    # load
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:

        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

"""
TriMul (outgoing) forward pass optimized for half precision.
- Merges linear projection + gate for left and right pathways (single matmul).
- Applies mask, gating and optional masking via a small Triton kernel.
- Uses batched GEMM (torch.matmul) for the O(N³) contraction.
- All heavy tensor arithmetic is performed in fp16; final output cast to fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,           # * fp16
    right_ptr,          # * fp16
    left_gate_ptr,      # * fp16
    right_gate_ptr,     # * fp16
    mask_ptr,           # * fp16 (broadcasted)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elements

    left = tl.load(left_ptr + offs, mask=m, other=0.0)
    right = tl.load(right_ptr + offs, mask=m, other=0.0)
    l_gate = tl.load(left_gate_ptr + offs, mask=m, other=0.0)
    r_gate = tl.load(right_gate_ptr + offs, mask=m, other=0.0)

    if HAS_MASK:
        mask_idx = offs // HIDDEN_DIM
        mask_val = tl.load(mask_ptr + mask_idx, mask=m, other=1.0)
        left = left * l_gate * mask_val
        right = right * r_gate * mask_val
    else:
        left = left * l_gate
        right = right * r_gate

    tl.store(left_ptr + offs, left, mask=m)
    tl.store(right_ptr + offs, right, mask=m)


def _apply_gate_mask_fuse(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    Fuses gating (sigmoid-ed) and optional mask into left/right tensors in-place
    using a Triton kernel. All tensors must be contiguous fp16.
    """
    total = left.numel()
    BLOCK = 131072  # large enough for good occupancy, multiple of 32
    grid = (triton.cdiv(total, BLOCK),)
    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when no mask

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        HIDDEN_DIM=hidden_dim,
    )
    # tensors are modified in‑place


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32 on GPU)
            - mask_tensor: torch.Tensor [B, N, N] (bool/float) – may be ignored
            - weights: dict with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight",
                "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict with at least "dim", "hidden_dim", optionally "nomask"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------- 1. LayerNorm (fp16) ----------
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)
    x = F.layer_norm(inp, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)  # now fp16

    B, N, _, _ = inp.shape
    M = B * N * N  # number of pairwise positions

    # flatten for linear layers
    x_flat = x.view(M, dim)  # (M, dim)

    # ---------- 2. Linear proj + gate (merged) ----------
    # concatenate proj and gate weights for left and right pathways
    left_proj_w = weights["left_proj.weight"].to(torch.float16)      # (hidden_dim, dim)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)      # (hidden_dim, dim)
    left_cat_w = torch.cat([left_proj_w, left_gate_w], dim=0)        # (2*hidden_dim, dim)

    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    right_cat_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    # merged matmuls
    left_cat = F.linear(x_flat, left_cat_w, bias=None)   # (M, 2*hidden_dim)
    right_cat = F.linear(x_flat, right_cat_w, bias=None)

    # split
    left_proj = left_cat[:, :hidden_dim]
    left_gate = torch.sigmoid(left_cat[:, hidden_dim:])   # (M, hidden_dim)

    right_proj = right_cat[:, :hidden_dim]
    right_gate = torch.sigmoid(right_cat[:, hidden_dim:])

    # reshape back to [B, N, N, hidden_dim]
    left_proj = left_proj.view(B, N, N, hidden_dim).contiguous()
    left_gate = left_gate.view(B, N, N, hidden_dim).contiguous()
    right_proj = right_proj.view(B, N, N, hidden_dim).contiguous()
    right_gate = right_gate.view(B, N, N, hidden_dim).contiguous()

    # ---------- 3. out_gate ----------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_w, bias=None))
    out_gate = out_gate.view(B, N, N, hidden_dim).contiguous()

    # ---------- 4. Apply mask & gating (fused kernel) ----------
    if not nomask:
        mask_half = mask.to(torch.float16).contiguous().view(-1)          # (M,)
        # flatten tensors for kernel
        left_flat = left_proj.view(-1)
        right_flat = right_proj.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)

        _apply_gate_mask_fuse(
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_half,
            hidden_dim,
        )
        # reshape back
        left = left_flat.view(B, N, N, hidden_dim)
        right = right_flat.view(B, N, N, hidden_dim)
    else:
        # simple elementwise multiply (no mask)
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ---------- 5. Core contraction (batched GEMM) ----------
    # shape to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # batched matmul: (B, H, N, N) @ (B, H, N, N)^T -> (B, H, N, N)
    out_perm = torch.matmul(left_perm, right_perm.transpose(-1, -2))

    # back to [B, N, N, hidden_dim]
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ---------- 6. Output LayerNorm ----------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # ---------- 7. Out‑gate scaling ----------
    out = out * out_gate

    # ---------- 8. Final linear ----------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out_flat = out.view(B * N * N, hidden_dim)
    out_flat = F.linear(out_flat, to_out_w, bias=None)
    out = out_flat.view(B, N, N, dim)

    # cast to float32 as required by the original module
    return out.to(torch.float32)
