@triton.jit
def matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_batch_a,
    stride_bk, stride_bn, stride_batch_b,
    stride_cm, stride_cn, stride_batch_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets in M and N
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Zero out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Compute pointer base for batch
    a_batch_ptr = a_ptr + pid_batch * stride_batch_a
    b_batch_ptr = b_ptr + pid_batch * stride_batch_b
    c_batch_ptr = c_ptr + pid_batch * stride_batch_c

    # Initialize accumulation
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A and B sub-blocks
        a = tl.load(a_batch_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak), mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_batch_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn), mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Write back C
    c = acc.to(tl.float16)
    tl.store(c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn), c, mask=mask_m[:, None] & mask_n[None, :])

The core O(N³) contraction out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched half‑precision GEMM (A @ Bᵀ) using a custom Triton
kernel. All other operations (LayerNorms, linear projections, gates, mask)
are performed in PyTorch with float16 precision. The final result is returned
as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr,                # *fp16  left  (batch, M, K)
    b_ptr,                # *fp16  right (batch, K, N) – accessed transposed
    c_ptr,                # *fp16  output (batch, M, N)

    stride_cm, stride_cn, stride_batch_c,   # C strides

    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_n = offs_n < N

    c_batch = c_ptr + batch_idx * stride_batch_c

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile (transposed): (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_N) accumulation in fp32

    # Write back the result in fp16
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask:        torch.Tensor [B, N, N] (bool/float) – may be None
            - weights:     dict of model parameters (float32)
            - config:      dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) -> half
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.half()

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (half)
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].half())
    right = F.linear(x_h, weights["right_proj.weight"].half())

    # ------------------------------------------------------------------
    # 3. Gates (half) and optional mask
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].half()).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].half()).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].half()).sigmoid()

    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * left_gate * mask_h
        right = right * right_gate * mask_h
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Core contraction via Triton (batched GEMM: A @ Bᵀ)
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape

    # Collapse batch & hidden dims -> (B*H, N, N)
    left_mat = left_perm.view(-1, N, N)
    right_mat = right_perm.view(-1, N, N)

    # Allocate output
    out_mat = torch.empty_like(left_mat)

    # Strides (in elements)
    stride_am = left_mat.stride(1)          # row stride = N
    stride_ak = left_mat.stride(2)          # col stride = 1
    stride_batch_a = left_mat.stride(0)     # batch stride = N * N

    # For B we need transposed access: swap row/col strides
    stride_bk = right_mat.stride(2)         # original col stride = 1
    stride_bn = right_mat.stride(1)         # original row stride = N
    stride_batch_b = right_mat.stride(0)    # batch stride = N * N

    stride_cm = out_mat.stride(1)           # row stride = N
    stride_cn = out_mat.stride(2)           # col stride = 1
    stride_batch_c = out_mat.stride(0)      # batch stride = N * N

    # Tiling parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        left_mat.shape[0],                     # batch dimension (B*H)
        triton.cdiv(N, BLOCK_M),               # M tiles
        triton.cdiv(N, BLOCK_N),               # N tiles
    )

    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,               # M
        N,               # N
        N,               # K
        stride_am, stride_ak, stride_batch_a,
        stride_bk, stride_bn, stride_batch_b,
        stride_cm, stride_cn, stride_batch_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore layout: (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, out‑gate scaling and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].half(),
        bias=None,
    )

    # Return in float32 as required
    return out.float()
