    import torch.nn.functional as F
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # layernorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x = x.to(torch.float16)

    # projections
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16), bias=None)

    # gates
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # gating + mask fusion
    if not nomask and mask is not None:
        # flatten
        total = left.numel()
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        mask_f = mask.to(torch.float16).contiguous().view(-1)  # shape bs*N*N

        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)
        HAS_MASK = 1
        gate_mask_fuse_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            total,
            BLOCK=BLOCK,
            HAS_MASK=HAS_MASK,
            HIDDEN_DIM=hidden_dim,
        )
        # reshape
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # core contraction

    left_perm = left.permute(0, 3, 1, 2).contiguous()

"""
Outgoing TriMul (AlphaFold3) forward pass.

The implementation follows the reference PyTorch code but offloads the
element‑wise mask/gate fusion and the per‑position layer‑normalisation (with
out‑gate scaling) to Triton kernels.  The O(N³) contraction is performed as a
batched half‑precision GEMM using torch.bmm, which leverages cuBLAS/Tensor
Cores.  All intermediate tensors are kept in float16; the final result is
returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------


@triton.jit
def gate_mask_fuse_kernel(
    left_ptr,          # * half
    right_ptr,         # * half
    left_gate_ptr,     # * half
    right_gate_ptr,    # * half
    mask_ptr,          # * half (broadcasted)
    total_elems: tl.constexpr,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):
    """
    In‑place element‑wise fusion:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    If HAS_MASK == 0 the mask load is omitted.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # broadcast mask over the hidden dimension
        mask_idx = offs // HIDDEN_DIM
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(1.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


@triton.jit
def outnorm_and_gate_kernel(
    out_ptr,           # * half   (B*N*N*H)
    out_gate_ptr,      # * half   (same shape as out)
    norm_weight_ptr,   # * half   (H)
    norm_bias_ptr,     # * half   (H)
    total_pos: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
    EPS: tl.constexpr = 1e-5,
):
    """
    For each (b,i,j) position:
        1) compute mean / variance over the hidden dimension
        2) apply LayerNorm weight & bias
        3) multiply by out_gate
    The result is written back into out_ptr (half precision).
    """
    pid = tl.program_id(0)
    if pid >= total_pos:
        return

    base = pid * HIDDEN_DIM
    idx = base + tl.arange(0, HIDDEN_DIM)

    # load data
    x = tl.load(out_ptr + idx)            # half
    g = tl.load(out_gate_ptr + idx)       # half
    w = tl.load(norm_weight_ptr + tl.arange(0, HIDDEN_DIM))  # half
    b = tl.load(norm_bias_ptr + tl.arange(0, HIDDEN_DIM))   # half

    # promote to float32 for statistics
    x_f = x.to(tl.float32)
    g_f = g.to(tl.float32)
    w_f = w.to(tl.float32)
    b_f = b.to(tl.float32)

    inv_h = 1.0 / HIDDEN_DIM
    sum_x = tl.sum(x_f)
    sum_x2 = tl.sum(x_f * x_f)
    mean = sum_x * inv_h
    var = sum_x2 * inv_h - mean * mean
    inv_std = tl.rsqrt(var + EPS)

    # normalize
    norm = (x_f - mean) * inv_std
    # apply LayerNorm weight & bias
    y = norm * w_f + b_f
    # apply out_gate
    y = y * g_f

    # store back as half
    tl.store(out_ptr + idx, y.to(tl.float16))

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, dim] (float32)
        - mask         : torch.Tensor[B, N, N] (bool/float) or None
        - weights      : dict of torch.Tensors (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) and cast to fp16
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (half precision)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Compute gating tensors (half precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) with the left/right gates via Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # flatten tensors for the kernel
        total_elements = left.numel()
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        mask_f = mask.to(torch.float16).contiguous().view(-1)  # [B*N*N]

        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)
        HAS_MASK = 1

        gate_mask_fuse_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            total_elements,
            BLOCK=BLOCK,
            HAS_MASK=HAS_MASK,
            HIDDEN_DIM=hidden_dim,
        )
        # reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # simple element‑wise gating when no mask is used
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³) contraction via batched GEMM (half precision)
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    # bring hidden dimension before the two spatial axes
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, _, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)    # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)  # (B*H, N, N)

    # batched matrix multiplication: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. LayerNorm over the hidden dimension + out_gate scaling (Triton)
    # ------------------------------------------------------------------
    total_positions = bs * N * N
    out_flat = out.view(total_positions, hidden_dim)          # (P, H)
    out_gate_flat = out_gate.contiguous().view(total_positions, hidden_dim)

    norm_weight = weights["to_out_norm.weight"].to(torch.float16).contiguous()
    norm_bias = weights["to_out_norm.bias"].to(torch.float16).contiguous()

    grid = (total_positions,)
    outnorm_and_gate_kernel[grid](
        out_flat,
        out_gate_flat,
        norm_weight,
        norm_bias,
        total_positions,
        HIDDEN_DIM=hidden_dim,
        EPS=1e-5,
    )

    # ------------------------------------------------------------------
    # 7. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # return in float32 as required
    return out.to(torch.float32)
