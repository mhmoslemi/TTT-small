"""
Outgoing TriMul (AlphaFold3) forward pass – Triton version.

Algorithm
---------
1. LayerNorm the input (float32 → float16).
2. Linear projections to the hidden dimension (no bias).
3. Compute three gates (sigmoid).
4. Apply optional mask and element‑wise gating (pure torch, cheap).
5. Core O(N³) contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM using a custom Triton kernel that
   processes all hidden slices (B·H) in a single launch.
6. Output LayerNorm, element‑wise out‑gate, final linear projection.
7. Cast back to float32.

The heavy O(N³) part is fully fused in Triton, achieving a large speed‑up
over the pure‑torch reference.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    left_ptr,          # *half  [BH, N, N]
    right_ptr,         # *half  [BH, N, N]
    out_ptr,           # *half  [BH, N, N]
    BH,                # total number of (batch * hidden) slices
    N,                 # sequence length
    stride_l_bh, stride_l_i, stride_l_k,
    stride_r_bh, stride_r_j, stride_r_k,
    stride_o_bh, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM for the TriMul contraction.

    Computes, for each bh ∈ [0, BH):
        out[bh, i, j] = Σ_k left[bh, i, k] * right[bh, j, k]
    (note the transposition of the second operand).
    """
    pid_bh = tl.program_id(0)                # index over BH
    pid_m  = tl.program_id(1)                # tile row
    pid_n  = tl.program_id(2)                # tile column

    # Offsets of the tile in the output matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in higher precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension in BLOCK_K sized chunks
    for k_offset in range(0, N, BLOCK_K):
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left tile (M x K) -----
        # left[bh, i, k]
        a_ptrs = (
            left_ptr
            + pid_bh * stride_l_bh
            + offs_m[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=tl.float16(0.0)
        )                                 # (BLOCK_M, BLOCK_K)

        # ----- load right tile (K x N) -----
        # right[bh, j, k] -> we need its transpose (k, j)
        b_ptrs = (
            right_ptr
            + pid_bh * stride_r_bh
            + offs_k[:, None] * stride_r_k   # k dimension becomes rows
            + offs_n[None, :] * stride_r_j   # j dimension becomes cols
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=tl.float16(0.0)
        )                                 # (BLOCK_K, BLOCK_N)

        # ---- dot product over K ----
        acc += tl.dot(a, b)               # (BLOCK_M, BLOCK_N)

    # ---- write back ----
    c_ptrs = (
        out_ptr
        + pid_bh * stride_o_bh
        + offs_m[:, None] * stride_o_i
        + offs_n[None, :] * stride_o_j
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :]
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict mapping parameter names to float32 tensors
            - config: dict with at least "dim" and "hidden_dim",
                      optionally "nomask" (bool)

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack arguments
    # -----------------------------------------------------------------
    inp, mask, weights, config = data
    device = inp.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 → float16)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections to hidden dimension (half‑precision)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # -----------------------------------------------------------------
    # 3. Gate tensors (sigmoid)
    # -----------------------------------------------------------------
    left_gate  = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate   = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # -----------------------------------------------------------------
    # 4. Optional mask + gating (element‑wise, cheap)
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → broadcast over hidden dim
        mask_h = mask.to(torch.float16).unsqueeze(-1)          # [B, N, N, 1]
        left = left * left_gate * mask_h
        right = right * right_gate * mask_h
    else:
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core O(N³) contraction via a Triton GEMM kernel
    # -----------------------------------------------------------------
    # Rearrange to (B, hidden, N, N) and flatten batch*hidden → BH
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    BH = B * H

    left_mat  = left_perm.view(BH, N, N)   # [BH, N, N]
    right_mat = right_perm.view(BH, N, N)  # [BH, N, N]
    out_mat   = torch.empty_like(left_mat) # [BH, N, N]

    # Tile sizes – chosen to fit comfortably in registers / shared memory
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Strides for the flattened tensors (they are contiguous)
    stride_l_bh, stride_l_i, stride_l_k = left_mat.stride()
    stride_r_bh, stride_r_j, stride_r_k = right_mat.stride()
    stride_o_bh, stride_o_i, stride_o_j = out_mat.stride()

    # Launch grid: (BH, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    grid = (
        BH,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        BH,
        N,
        stride_l_bh, stride_l_i, stride_l_k,
        stride_r_bh, stride_r_j, stride_r_k,
        stride_o_bh, stride_o_i, stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to the original 4‑D layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating with out_gate and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # -----------------------------------------------------------------
    # 7. Cast back to float32 as required by the spec
    # -----------------------------------------------------------------
    return out.to(torch.float32)
