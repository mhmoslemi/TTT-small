"""
Outgoing TriMul (AlphaFold3) forward pass.

Algorithm
---------
1. Layer‑norm the input tensor (float32 → float16).
2. Linear projections to the hidden dimension (no bias) producing `left` and `right`.
3. Compute three gate tensors (`left_gate`, `right_gate`, `out_gate`) with sigmoid.
4. Fuse optional mask with the left/right gates in a Triton kernel (in‑place).
5. Core O(N³) contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM over the hidden dimension using torch.bmm.
6. Output LayerNorm, element‑wise gating with `out_gate` and final linear
   projection back to the original feature dimension.
7. Return the result as float32.

All heavy element‑wise work (mask + gate fusion) is performed in Triton,
while the cubic contraction uses cuBLAS via torch.bmm for maximum throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # *half
    right_ptr,         # *half
    left_gate_ptr,     # *half
    right_gate_ptr,    # *half
    mask_ptr,          # *half (broadcasted over hidden dimension)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HIDDEN: tl.constexpr,   # hidden dimension size
    HAS_MASK: tl.constexpr, # 1 if mask present else 0
):
    """In‑place: left = left * left_gate * mask, right = right * right_gate * mask."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elements

    left = tl.load(left_ptr + offs, mask=m, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=m, other=tl.float16(0.0))
    lgate = tl.load(left_gate_ptr + offs, mask=m, other=tl.float16(0.0))
    rgate = tl.load(right_gate_ptr + offs, mask=m, other=tl.float16(0.0))

    if HAS_MASK:
        # each mask element corresponds to HIDDEN consecutive hidden entries
        mask_idx = offs // HIDDEN
        mask_val = tl.load(mask_ptr + mask_idx, mask=m, other=tl.float16(0.0))
        left = left * lgate * mask_val
        right = right * rgate * mask_val
    else:
        left = left * lgate
        right = right * rgate

    tl.store(left_ptr + offs, left, mask=m)
    tl.store(right_ptr + offs, right, mask=m)


def _apply_mask_and_gate(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    Fuse mask (if any) and gates into `left` and `right` tensors in‑place.
    All tensors must be contiguous, half‑precision and on the same device.
    """
    total = left.numel()
    # Choose a large block size to reduce launch overhead.
    BLOCK = 32768
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    # Triton requires a valid pointer even when mask is absent; we pass `left`
    # which will never be dereferenced inside the kernel because HAS_MASK=0.
    mask_ptr = mask if mask is not None else left

    _mask_gate_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HIDDEN=hidden_dim,
        HAS_MASK=has_mask,
    )
    # No return; operation is in‑place.


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (float/bool) or None
            - weights: dict mapping parameter names to float32 tensors
            - config: dict containing at least "dim" and "hidden_dim",
                      optionally "nomask" (bool)

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    inp, mask, weights, config = data
    device = inp.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> float16)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections to hidden dimension (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # -----------------------------------------------------------------
    # 3. Gate tensors (sigmoid)
    # -----------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # -----------------------------------------------------------------
    # 4. Fuse mask with gates via Triton (in‑place)
    # -----------------------------------------------------------------
    if not nomask:
        # Ensure mask is half‑precision and has shape [B,N,N]
        mask_h = mask.to(torch.float16)  # [B,N,N]
        _apply_mask_and_gate(left, right, left_gate, right_gate, mask_h, hidden_dim)
    else:
        # Simple element‑wise multiplication when mask is absent
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: batched GEMM over the hidden dimension
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape  # left shape: (B, N, N, H)

    # Rearrange to (B, H, N, N) and make contiguous for cuBLAS.
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Merge batch and hidden dims → (B*H, N, N)
    left_mat = left_t.view(B * hidden_dim, N, N)
    right_mat = right_t.view(B * hidden_dim, N, N)

    # Batched matrix multiplication: left @ right^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return in float32 as specified
    return out.to(torch.float32)
