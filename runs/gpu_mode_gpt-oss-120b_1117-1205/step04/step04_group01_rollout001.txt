"""
TriMul outgoing forward pass (AlphaFold3 style) with Triton fusion.

Algorithm
---------
1. Layer‑norm the input tensor (float32 → float16).
2. Linear projections to hidden dimension (`left` and `right`) using half‑precision.
3. Compute three gate tensors (left, right, out) via linear + sigmoid, also half.
4. Fuse mask (if present) and the left/right gates in a single Triton kernel,
   performing    left = left * mask * left_gate
   and          right = right * mask * right_gate    in‑place.
5. Core O(N³) contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication for each hidden channel:
   left[b,…,d] (N×N) @ right[b,…,d].T (N×N).
   Implemented with `torch.bmm`, which uses highly‑optimised cuBLAS.
6. Apply a second layer‑norm on the hidden tensor, multiply by the out‑gate,
   and project back to the original dimension.
7. Return the result in float32.

Only the mask‑gate fusion is performed in Triton; the heavy O(N³) contraction
uses cuBLAS via `torch.bmm`, which is far faster than the reference `einsum`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask (if present) with left/right gates
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,       # half *
    right_ptr,      # half *
    left_gate_ptr,  # half *
    right_gate_ptr, # half *
    mask_ptr,       # half *
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask is present, else 0
    H: tl.constexpr,          # hidden dimension (size of innermost channel)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elems

    # Load tensors
    left      = tl.load(left_ptr + offs,      mask=valid, other=0.0)
    right     = tl.load(right_ptr + offs,     mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate= tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    if HAS_MASK:
        # Each mask value is repeated H times across the hidden channel.
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=0.0)
        left  = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left * left_gate
        right = right * right_gate

    # Store results back in-place
    tl.store(left_ptr  + offs, left,  mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)

def _fuse_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """In‑place fusion of mask and left/right gates using Triton."""
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy ptr when no mask

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )

    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of parameter tensors
            - config: dict with keys 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device).to(torch.float16),
        bias=weights["norm.bias"].to(device).to(torch.float16),
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(device).to(torch.float16),
        bias=None,
    )  # [B, N, N, hidden_dim]
    right = F.linear(
        x,
        weights["right_proj.weight"].to(device).to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gate values (half)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(device).to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(device).to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(device).to(torch.float16),
        bias=None,

    # ------------------------------------------------------------------

            left, right,
            left_gate, right_gate,
            mask_h,
            hidden_dim,
        )

    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Rearrange to (B, hidden_dim, N, N) and reshape to (B*hidden_dim, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    left_batch = left.view(B * hidden_dim, N, N)            # (B*H, N, N)
    right_batch = right.view(B * hidden_dim, N, N)          # (B*H, N, N)

    # Matrix multiplication: A @ B^T
    out_batch = torch.bmm(left_batch, right_batch.transpose(1, 2))  # (B*H, N, N)

    # Reshape back to (B, N, N, hidden_dim)
    out = out_batch.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device).to(torch.float16),
        bias=weights["to_out_norm.bias"].to(device).to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(device).to(torch.float16),
        bias=None,
    )  # [B, N, N, dim]

    # Return float32 as required by the interface
    return out.to(torch.float32)
```
