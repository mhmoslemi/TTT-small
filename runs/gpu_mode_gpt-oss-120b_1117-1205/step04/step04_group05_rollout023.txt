def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    
    # 1. LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x = x.to(torch.float16)
    
    # 2. Combine weights
    left_comb_weight = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(torch.float16)
    right_comb_weight = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(torch.float16)
    
    # 3. Linear with combined weights
    left_comb = F.linear(x, left_comb_weight, bias=None) # shape [B,N,N,2*hidden]
    right_comb = F.linear(x, right_comb_weight, bias=None) # shape [B,N,N,2*hidden]
    
    # split
    left_proj, left_gate_pre = torch.split(left_comb, hidden_dim, dim=-1)
    right_proj, right_gate_pre = torch.split(right_comb, hidden_dim, dim=-1)

    left_gate = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()
    
    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # shape [B,N,N,1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f
    # else no mask

    left = left_proj * left_gate
    right = right_proj * right_gate

    # compute out_gate
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # 4. core contraction using Triton kernel
    # Permute to (B, hidden, N, N) and contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_perm.shape
    batch_total = B * H

    A = left_perm.view(batch_total, N, N)
    B_mat = right_perm.view(batch_total, N, N).transpose(1, 2)  # shape (batch_total, N, N)
    C = torch.empty_like(A)  # same dtype and shape

    # Determine strides
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B_mat.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # block sizes - tune
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (batch_total, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    batched_matmul_kernel[grid](
        A, B_mat, C,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        N, N, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # Result C shape (batch_total, N, N)
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # shape (B,N,N,hidden)

    # 5. output layer norm
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=weights["to_out_norm.weight"].to(torch.float16),
                       bias=weights["to_out_norm.bias"].to(torch.float16))
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    # Cast back to float32
    return out.to(torch.float32)

@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # initialize accumulation
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile
        a_ptrs = A_ptr + pid_batch * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)

        # Load B tile
        b_ptrs = B_ptr + pid_batch * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        acc += tl.dot(a, b)

    # Write back
    c_ptrs = C_ptr + pid_batch * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=c_mask)

"""
TriMul outgoing kernel (AlphaFold3) with Triton‑accelerated batched GEMM.

Algorithm
---------
1. Layer‑normalize the input tensor.
2. Fuse the two linear projections (proj + gate) for the left and right
   tensors by concatenating their weights, then split the result.
3. Apply the optional mask, sigmoid gates and element‑wise multiplication.
4. Compute the core O(N³) contraction:
      out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
   implemented as a batched matrix multiplication using a custom
   Triton kernel (one program per (batch*hidden) pair).
5. Apply output layer‑norm, out‑gate scaling and the final linear projection.
All heavy arithmetic is performed in float16 (accumulated in float32);
the final result is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                     # pointers
    stride_a_batch, stride_a_m, stride_a_k,   # strides for A (batch, M, K)
    stride_b_batch, stride_b_k, stride_b_n,   # strides for B (batch, K, N)
    stride_c_batch, stride_c_m, stride_c_n,   # strides for C (batch, M, N)
    M, N, K,                                  # matrix dimensions
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C[b] = A[b] @ B[b]   (A: M×K, B: K×N)"""
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulate in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile (M×K)
        a_ptrs = (A_ptr +
                  pid_batch * stride_a_batch +
                  offs_m[:, None] * stride_a_m +
                  offs_k[None, :] * stride_a_k)
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)

        # Load B tile (K×N)
        b_ptrs = (B_ptr +
                  pid_batch * stride_b_batch +
                  offs_k[:, None] * stride_b_k +
                  offs_n[None, :] * stride_b_n)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Write back result
    c_ptrs = (C_ptr +
              pid_batch * stride_c_batch +
              offs_m[:, None] * stride_c_m +
              offs_n[None, :] * stride_c_n)
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32/bool) – may be None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Fuse proj+gate for left and right sides
    # ------------------------------------------------------------------
    # concatenate weights: [proj; gate] → (2*hidden, dim)
    left_comb_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)
    right_comb_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)

    # combined linear (produces proj and gate pre‑sigmoid together)
    left_comb = F.linear(x, left_comb_weight, bias=None)   # [B,N,N,2*hidden]
    right_comb = F.linear(x, right_comb_weight, bias=None) # [B,N,N,2*hidden]

    # split back
    left_proj, left_gate_pre = torch.split(left_comb, hidden_dim, dim=-1)
    right_proj, right_gate_pre = torch.split(right_comb, hidden_dim, dim=-1)

    # free combined tensors early
    del left_comb, right_comb

    left_gate = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()

    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)      # [B,N,N,1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # Apply gates
    left = left_proj * left_gate          # [B,N,N,hidden]
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 3. Out‑gate (separate linear)
    # ------------------------------------------------------------------
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()                            # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 4. Core contraction via Triton batched GEMM
    # ------------------------------------------------------------------
    # Bring hidden dimension to batch axis for the GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    batch_total = B * H

    # reshape to (batch_total, N, N)
    A = left_perm.view(batch_total, N, N)
    B_mat = right_perm.view(batch_total, N, N).transpose(1, 2)  # K×N
    C = torch.empty_like(A)

    # strides
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B_mat.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # tiling parameters (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_total,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_matmul_kernel[grid](
        A,
        B_mat,
        C,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B,N,N,hidden)
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, out‑gate scaling and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Cast back to float32 for the final output
    return out.to(torch.float32)
