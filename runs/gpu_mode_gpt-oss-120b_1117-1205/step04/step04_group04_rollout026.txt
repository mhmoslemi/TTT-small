def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # layer norm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # Linear projections
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))   # shape B,N,N,H
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # Compute gates
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # Apply mask (if any) and gate
    if not nomask and mask is not None:
        # mask shape [B,N,N] broadcast to [B,N,N,1]
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * left_gate * mask_h
        right = right * right_gate * mask_h
    else:
        left = left * left_gate
        right = right * right_gate

    # Core O(N³) contraction using Triton
    B, N, _, H = left.shape

    # allocate output
    out = torch.empty((B, N, N, H), dtype=torch.float16, device=device)

    # strides
    # left: (B, i, k, H)
    stride_b_l, stride_i_l, stride_k_l, stride_h_l = left.stride()
    stride_b_r, stride_i_r, stride_k_r, stride_h_r = right.stride()
    stride_b_o, stride_i_o, stride_j_o, stride_h_o = out.stride()

    # kernel launch parameters
    # Choose block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # typical for FP16

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * H)

    batched_trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_b_left=stride_b_l,
        stride_i_left=stride_i_l,
        stride_k_left=stride_k_l,
        stride_h_left=stride_h_l,
        stride_b_right=stride_b_r,
        stride_i_right=stride_i_r,
        stride_k_right=stride_k_r,
        stride_h_right=stride_h_r,
        stride_b_out=stride_b_o,
        stride_i_out=stride_i_o,
        stride_j_out=stride_j_o,
        stride_h_out=stride_h_o,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),

    # Cast back to float32

@triton.jit
def batched_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b_left, stride_i_left, stride_k_left, stride_h_left,
    stride_b_right, stride_i_right, stride_k_right, stride_h_right,
    stride_b_out, stride_i_out, stride_j_out, stride_h_out,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # tile over i dimension
    pid_n = tl.program_id(1)  # tile over j dimension
    pid_bh = tl.program_id(2) # batch * hidden

    # Compute batch and hidden indices
    bh = pid_bh
    b = bh // H
    h = bh % H

    # Compute offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this batch and hidden
    # Compute batch & hidden offset in elements
    batch_offset_left = b * stride_b_left
    hidden_offset_left = h * stride_h_left
    batch_offset_right = b * stride_b_right
    hidden_offset_right = h * stride_h_right
    batch_offset_out = b * stride_b_out
    hidden_offset_out = h * stride_h_out

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointers for A (left)
        a_ptrs = left_ptr + batch_offset_left + hidden_offset_left + offs_i[:, None] * stride_i_left + offs_k[None, :] * stride_k_left
        # Pointers for B (right) – note B is indexed by j (rows) and k (cols)
        b_ptrs = right_ptr + batch_offset_right + hidden_offset_right + offs_j[:, None] * stride_i_right + offs_k[None, :] * stride_k_right

        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=tl.float16(0.0))
        b = tl.load(b_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=tl.float16(0.0))

        # Dot product over k dimension (BLOCK_K)
        acc += tl.dot(a, b)

    # Write back
    c = acc.to(tl.float16)
    c_ptrs = out_ptr + batch_offset_out + hidden_offset_out + offs_i[:, None] * stride_i_out + offs_j[None, :] * stride_j_out
    tl.store(c_ptrs, c, mask=mask_i[:, None] & mask_j[None, :])

"""
Outgoing TriMul (AlphaFold3) – fast Triton implementation.

Algorithm
---------
1. Layer‑norm the input tensor (float32 → float16 for speed).
2. Linear projections to the hidden dimension (no bias) → `left` and `right`.
3. Compute three gate tensors with a sigmoid.
4. Fuse mask (if present) with the left/right projections using a simple
   element‑wise PyTorch operation (broadcast is cheap).
5. Core O(N³) contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM using a Triton kernel.  The kernel treats
   each (batch, hidden) pair as an independent matrix‑multiply and streams
   the k‑dimension in tiles.
6. Output LayerNorm (float32), element‑wise gating with `out_gate`,
   final linear projection back to `dim`.
7. Return the result in float32.

Only the O(N³) contraction is executed in Triton; all other ops use
high‑performance PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    left_ptr,      # *half  [B, N, N, H]
    right_ptr,     # *half  [B, N, N, H]
    out_ptr,       # *half  [B, N, N, H]
    N,             # int32  sequence length
    H,             # int32  hidden dimension
    stride_b_l, stride_i_l, stride_k_l, stride_h_l,   # left strides
    stride_b_r, stride_i_r, stride_k_r, stride_h_r,   # right strides
    stride_b_o, stride_i_o, stride_j_o, stride_h_o,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids:
    pid_m = tl.program_id(0)          # i‑tile
    pid_n = tl.program_id(1)          # j‑tile
    pid_bh = tl.program_id(2)         # (batch, hidden) tile

    # decode batch and hidden indices
    bh = pid_bh
    b = bh // H
    h = bh % H

    # offsets within i and j dimensions
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute constant offsets for this (b, h) pair
    batch_off_l = b * stride_b_l
    hidden_off_l = h * stride_h_l
    batch_off_r = b * stride_b_r
    hidden_off_r = h * stride_h_r
    batch_off_o = b * stride_b_o
    hidden_off_o = h * stride_h_o

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension in tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (left_ptr + batch_off_l + hidden_off_l
                  + offs_i[:, None] * stride_i_l
                  + offs_k[None, :] * stride_k_l)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :],
                    other=tl.float16(0.0))

        # right: shape (BLOCK_N, BLOCK_K) – rows are j, cols are k
        b_ptrs = (right_ptr + batch_off_r + hidden_off_r
                  + offs_j[:, None] * stride_i_r
                  + offs_k[None, :] * stride_k_r)
        b = tl.load(b_ptrs, mask=mask_j[:, None] & mask_k[None, :],
                    other=tl.float16(0.0))

        # accumulate dot product over the k‑tile
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) × (BLOCK_N, BLOCK_K) → (BLOCK_M, BLOCK_N)

    # write back the result
    c = acc.to(tl.float16)
    c_ptrs = (out_ptr + batch_off_o + hidden_off_o
              + offs_i[:, None] * stride_i_o
              + offs_j[None, :] * stride_j_o)
    tl.store(c_ptrs, c, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    inp, mask, weights, config = data
    device = inp.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Gate tensors (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 4. Apply optional mask and gates (broadcasted element‑wise)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)          # [B,N,N,1]
        left = left * left_gate * mask_h
        right = right * right_gate * mask_h
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³) contraction via Triton batched GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left.shape               # left/ right shape = [B, N, N, H]
    out = torch.empty((B, N, N, H), dtype=torch.float16, device=device)

    # strides (in elements, not bytes)
    stride_b_l, stride_i_l, stride_k_l, stride_h_l = left.stride()
    stride_b_r, stride_i_r, stride_k_r, stride_h_r = right.stride()
    stride_b_o, stride_i_o, stride_j_o, stride_h_o = out.stride()

    # block sizes – tuned for typical A100 / B200 hardware
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # i‑tiles
        triton.cdiv(N, BLOCK_N),   # j‑tiles
        B * H                      # (batch, hidden) tiles
    )

    _trimul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        N=N,
        H=H,
        stride_b_l=stride_b_l,
        stride_i_l=stride_i_l,
        stride_k_l=stride_k_l,
        stride_h_l=stride_h_l,
        stride_b_r=stride_b_r,
        stride_i_r=stride_i_r,
        stride_k_r=stride_k_r,
        stride_h_r=stride_h_r,
        stride_b_o=stride_b_o,
        stride_i_o=stride_i_o,
        stride_j_o=stride_j_o,
        stride_h_o=stride_h_o,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = out.to(torch.float32)                     # cast back for higher‑precision norm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 7. Return float32 result
    # ------------------------------------------------------------------
    return out.to(torch.float32)
