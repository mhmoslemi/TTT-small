@triton.jit
def gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements
    left = tl.load(left_ptr + offs, mask=mask)
    right = tl.load(right_ptr + offs, mask=mask)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask)
    if HAS_MASK:
        # mask index per hidden dimension group
        # each hidden channel group of size H has same mask entry
        # Since mask shape is [B,N,N] flattened as same as left without hidden dim
        # Actually flatten left shape to B*N*N*H. So mask index = offs // H
        mask_idx = offs // H 
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    # Cast to half
    dtype = torch.float16
    # 1. LayerNorm (half)
    norm_weight = weights["norm.weight"].to(dtype)
    norm_bias = weights["norm.bias"].to(dtype)
    x = torch.nn.functional.layer_norm(
        input_tensor.to(dtype),
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    # 2. Linear projections (half)
    left_proj_weight = weights["left_proj.weight"].to(dtype)
    right_proj_weight = weights["right_proj.weight"].to(dtype)
    left = torch.nn.functional.linear(x, left_proj_weight, bias=None)
    right = torch.nn.functional.linear(x, right_proj_weight, bias=None)
    # 3. Gates (half)
    left_gate_weight = weights["left_gate.weight"].to(dtype)
    right_gate_weight = weights["right_gate.weight"].to(dtype)
    out_gate_weight = weights["out_gate.weight"].to(dtype)
    left_gate = torch.nn.functional.linear(x, left_gate_weight, bias=None).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight, bias=None).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight, bias=None).sigmoid()
    # 4. Apply mask and gates
    if not nomask and mask is not None:
        # Flatten for kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask.to(dtype).contiguous().view(-1)  # shape [B*N*N]
        total = left_f.numel()
        BLOCK = 16384  # maybe 16KB per block? Choose 32768? Not exceed max threads 65536?
        grid = (triton.cdiv(total, BLOCK),)
        HAS_MASK = 1
        H = hidden_dim
        gate_mask_fuse_kernel[grid](
            left_f, right_f,
            left_gate_f, right_gate_f,
            mask_f,
            total,
            BLOCK=BLOCK,
            HAS_MASK=HAS_MASK,
            H=H,
        )
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate
    # 5. Core contraction
    # Permute to [B, hidden_dim, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N) # (B*H, N, N)
    # GEMM
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))   # (B*H, N, N)
    # reshape back to [B,N,N,hidden_dim]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    # 6. Output LayerNorm
    to_out_norm_weight = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_bias = weights["to_out_norm.bias"].to(dtype)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    # 7. Scale by out_gate
    out = out * out_gate
    # 8. Final linear
    to_out_weight = weights["to_out.weight"].to(dtype)
    out = torch.nn.functional.linear(out, to_out_weight, bias=None)
    # Convert to float32
    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass with a fused Triton kernel for the
mask‑gate multiplication. All heavy linear operations are performed in
half precision with PyTorch’s highly‑optimized kernels; the contraction
is a batched GEMM (torch.bmm). The Triton kernel applies
(left * left_gate * mask) and (right * right_gate * mask) in‑place,
eliminating extra memory passes. The final output is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half*   (flattened left tensor)
    right_ptr,         # half*   (flattened right tensor)
    left_gate_ptr,     # half*   (flattened left gate)
    right_gate_ptr,    # half*   (flattened right gate)
    mask_ptr,          # half*   (flattened mask, broadcast over hidden dim)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 -> mask present, 0 -> no mask
    H: tl.constexpr,          # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load operands
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # Each hidden channel shares the same mask entry.
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store results back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask_triton(
    left: torch.Tensor,
    right: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    hidden_dim: int,
) -> None:
    """
    In‑place fuse:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    All tensors must be contiguous half‑precision.
    """
    total = left.numel()
    BLOCK = 16384  # #elements processed per program
    grid = (triton.cdiv(total, BLOCK),)

    # Triton expects a pointer for mask even when not used; we always have mask here.
    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask,
        total,
        BLOCK=BLOCK,
        HAS_MASK=1,
        H=hidden_dim,
    )


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights_dict, config_dict)
            input_tensor: torch.Tensor [B, N, N, dim] (float32)
            mask: torch.Tensor [B, N, N] (bool or float) – may be None
            weights: dict with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            config: dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float16

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32) -> half precision
    # --------------------------------------------------------------
    x_fp32 = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x_fp32.to(dtype)

    # --------------------------------------------------------------
    # 2. Linear projections (half)
    # --------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(dtype),
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(dtype),
        bias=None,
    )

    # --------------------------------------------------------------
    # 3. Gates (half)
    # --------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(dtype),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(dtype),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(dtype),
        bias=None,
    ).sigmoid()

    # --------------------------------------------------------------
    # 4. Fuse mask with gates via Triton (if mask present)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # Ensure contiguous half tensors
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask.to(dtype).contiguous().view(-1)  # shape [B*N*N]

        _apply_gate_and_mask_triton(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            hidden_dim,
        )
        # Reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as batched GEMM on half‑precision tensors.
    # --------------------------------------------------------------
    # Rearrange to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)          # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)        # (B*H, N, N)

    # Batched matrix multiplication: left @ right^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore (B, N, N, hidden_dim)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(dtype),
        bias=None,
    )

    # Return in original dtype (float32)
    return out.to(torch.float32)
