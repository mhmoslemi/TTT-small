"""
TriMul forward (outgoing) implementation.

Algorithm:
1. Layer‑norm the input over the `dim` channel.
2. Linear projections to `hidden_dim` → tensors `left` and `right`.
3. Compute three gating vectors (sigmoid of linear projections):
   `left_gate`, `right_gate`, `out_gate`.
4. Fuse mask (if present) and the two gates into `left` and `right` in‑place
   with a tiny Triton kernel, avoiding extra temporaries.
5. Core contraction:
   for each hidden channel d we need
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication of shape
   (B*hidden_dim, N, N) × (B*hidden_dim, N, N)ᵀ.
6. Output layer‑norm, out‑gate scaling and final linear projection back to `dim`.

All heavy element‑wise work (gate+mask) is done in Triton; the O(N³) contraction
uses cuBLAS via `torch.bmm`, which is extremely fast on modern GPUs.
The final tensor is returned in float32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask and gate into left/right tensors in‑place
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # float16*
    right_ptr,         # float16*
    left_gate_ptr,     # float16*
    right_gate_ptr,    # float16*
    mask_ptr,          # float16* (broadcasted, same layout as [B,N,N])
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask is present
    H: tl.constexpr,          # hidden_dim
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # Each hidden entry shares the same mask value.
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)

    All tensors must be contiguous and dtype float16.

        right,

        BLOCK=BLOCK,
        HAS_MASK=has_mask,

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) – may be None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float16, device=device)
    norm_bias = weights["norm.bias"].to(torch.float16, device=device)

    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"].to(torch.float16, device=device)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16, device=device)

    left = F.linear(x_h, left_proj_weight, bias=None)   # [B,N,N,hidden_dim]
    right = F.linear(x_h, right_proj_weight, bias=None)

    # ------------------------------------------------------------------
    # 3. Gates (half) with sigmoid
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"].to(torch.float16, device=device)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16, device=device)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16, device=device)

    left_gate = F.linear(x_h, left_gate_weight, bias=None).sigmoid()
    right_gate = F.linear(x_h, right_gate_weight, bias=None).sigmoid()
    out_gate = F.linear(x_h, out_gate_weight, bias=None).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) with gates via Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → half, flattened
        mask_h = mask.to(torch.float16, device=device).contiguous().view(-1)  # [B*N*N]
    else:
        mask_h = None

    # Ensure contiguous layout for kernel
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    # Flatten tensors for the kernel (same storage)
    left_f = left.view(-1)
    right_f = right.view(-1)
    left_gate_f = left_gate.view(-1)
    right_gate_f = right_gate.view(-1)

    _apply_gate_and_mask(
        left_f,
        right_f,
        left_gate_f,
        right_gate_f,
        mask_h,
        hidden_dim,
    )
    # Reshape back (views share storage, no copy)
    left = left_f.view_as(left)
    right = right_f.view_as(right)

    # Cleanup unused temporaries
    del left_gate, right_gate, left_gate_f, right_gate_f, left_f, right_f, mask_h

    # ------------------------------------------------------------------
    # 5. Core contraction using batched GEMM:
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden_dim, N, N) for batch GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # Batched matrix multiplication: left * rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore original layout (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16, device=device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16, device=device)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate  # element‑wise scaling

    to_out_weight = weights["to_out.weight"].to(torch.float16, device=device)
    out = F.linear(out, to_out_weight, bias=None)

    # Return as float32 (the interface expects float32)
    return out.to(torch.float32)
