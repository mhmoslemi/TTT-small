def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    BATCH: tl.constexpr,
    N: tl.constexpr,
    stride_left_batch, stride_left_row, stride_left_col,
    stride_right_batch, stride_right_row, stride_right_col,
    stride_out_batch, stride_out_row, stride_out_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):

Within kernel:

batch_id = tl.program_id(0)
pid_m = tl.program_id(1)
pid_n = tl.program_id(2)

offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

m_mask = offs_m < N
n_mask = offs_n < N

# Base pointers for this batch
left_batch_ptr = left_ptr + batch_id * stride_left_batch
right_batch_ptr = right_ptr + batch_id * stride_right_batch
out_batch_ptr = out_ptr + batch_id * stride_out_batch

# Create pointers for k loop
offs_k = tl.arange(0, BLOCK_K)

acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

num_blocks_k = tl.cdiv(N, BLOCK_K)
for k_iter in range(num_blocks_k):
    k_offset = k_iter * BLOCK_K
    # compute k indices
    k_valid = (k_offset + offs_k) < N
    # Load A tile [m,k] from left
    a_ptrs = left_batch_ptr + (offs_m[:, None] * stride_left_row) + ((k_offset + offs_k)[None, :] * stride_left_col)
    a = tl.load(a_ptrs, mask=m_mask[:, None] & k_valid[None, :], other=0.0).to(tl.float32)

    # Load B_T tile [k,n] from right (right_T)
    b_ptrs = right_batch_ptr + ((k_offset + offs_k)[:, None] * stride_right_row) + (offs_n[None, :] * stride_right_col)
    b = tl.load(b_ptrs, mask=k_valid[:, None] & n_mask[None, :], other=0.0).to(tl.float32)

    acc += tl.dot(a, b)

    # loop continues

# Store result
c_ptrs = out_batch_ptr + (offs_m[:, None] * stride_out_row) + (offs_n[None, :] * stride_out_col)
tl.store(c_ptrs, acc.to(tl.float16), mask=m_mask[:, None] & n_mask[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)  # if true no mask else mask present

    # Get weights and convert to half as needed
    # norm weights (dim,) are float32, to keep stable we can compute layernorm in fp32.
    # But we can also cast to float16 for speed; but output must be float32.
    # Let's compute layernorm in fp32 then cast to half.

    # Convert weights for linear layers to half
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    to_out_weight = weights["to_out.weight"].to(torch.float16)
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)

    # LayerNorm on input
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x_norm.to(torch.float16)

    # Linear projections
    left = F.linear(x_h, left_proj_weight, bias=None)
    right = F.linear(x_h, right_proj_weight, bias=None)

    # Gates
    left_gate = F.linear(x_h, left_gate_weight, bias=None).sigmoid()
    right_gate = F.linear(x_h, right_gate_weight, bias=None).sigmoid()
    out_gate = F.linear(x_h, out_gate_weight, bias=None).sigmoid()

    if not nomask and mask is not None:
        # Broadcast mask to hidden_dim: shape (B,N,N,1)
        mask = mask.to(torch.float16).unsqueeze(-1)  # shape (B,N,N,1)
        # We'll incorporate mask in kernel later.
    else:
        mask = None

    # Apply gating (and mask) elementwise. We'll compute left_gated and right_gated.
    # For memory usage & performance, compute as:
    left = left * left_gate
    right = right * right_gate
    if mask is not None:
        left = left * mask
        right = right * mask

    # Now we have left and right gated. Compute contraction using Triton kernel.

    # Permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2)   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2) # (B, H, N, N)
    # No need .contiguous() if we call .reshape; but better to ensure contiguous for simple stride.
    left_cont = left_perm.contiguous()
    right_cont = right_perm.contiguous()

    B, H, N, _ = left_cont.shape
    BH = B * H

    # Flatten batch and hidden
    left_mat = left_cont.view(BH, N, N)
    right_mat = right_cont.view(BH, N, N)

    # Allocate output
    out_mat = torch.empty_like(left_mat)  # shape (BH,N,N) half

    # Compute strides for flattened tensors. Assume contiguous row-major.
    stride_l_batch = left_mat.stride(0)  # N*N
    stride_l_row = left_mat.stride(1)    # N
    stride_l_col = left_mat.stride(2)    # 1
    stride_r_batch = right_mat.stride(0)
    stride_r_row = right_mat.stride(1)
    stride_r_col = right_mat.stride(2)
    stride_o_batch = out_mat.stride(0)
    stride_o_row = out_mat.stride(1)
    stride_o_col = out_mat.stride(2)

    # Choose block sizes. Let's try B_M=64, B_N=64, B_K=32.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Compute grid
    num_blocks_m = triton.cdiv(N, BLOCK_M)
    num_blocks_n = triton.cdiv(N, BLOCK_N)
    grid = (BH, num_blocks_m, num_blocks_n)

    # Launch kernel
    trimul_outgoing_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_l_batch,
        stride_l_row,
        stride_l_col,
        stride_r_batch,
        stride_r_row,
        stride_r_col,
        stride_o_batch,
        stride_o_row,
        stride_o_col,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to (B, H, N, N) then permute to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output layernorm (half)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # Apply out_gate scaling (out_gate shape is (B,N,N,hidden_dim) half)
    # but out_gate was computed earlier; we need to broadcast shape left.
    # out_gate is shape (B,N,N,hidden_dim) half (same dims)
    out = out * out_gate

    # Final linear projection to original dim
    out = F.linear(out, to_out_weight, bias=None)

    # Cast to float32 for output
    return out.to(torch.float32)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_l_batch, stride_l_row, stride_l_col,
    stride_r_batch, stride_r_row, stride_r_col,
    stride_o_batch, stride_o_row, stride_o_col,
    N: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base ptr for this batch
    left_batch_ptr = left_ptr + batch_id * stride_l_batch
    right_batch_ptr = right_ptr + batch_id * stride_r_batch
    out_batch_ptr = out_ptr + batch_id * stride_o_batch

    # Number of k-tiles
    num_k = tl.cdiv(N, BLOCK_K)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(num_k):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile (left)
        a_ptrs = left_batch_ptr + (offs_m[:, None] * stride_l_row) + (offs_k[None, :] * stride_l_col)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B_T tile (right transposed)
        b_ptrs = right_batch_ptr + (offs_k[:, None] * stride_r_row) + (offs_n[None, :] * stride_r_col)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Store result
    c_ptrs = out_batch_ptr + (offs_m[:, None] * stride_o_row) + (offs_n[None, :] * stride_o_col)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Outgoing Triangular Multiplicative Update (TriMul) forward pass.

    Implements the AlphaFold3 TriMul operator using half‑precision tensors and a
    Triton kernel for the core O(N³) contraction. The kernel computes, for each
    batch and hidden channel d:

        out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

    where left/right are linear projections of the normalized input, gated by
    sigmoid gates and an optional mask. The rest of the pipeline (layer‑norms,
    linear projections and final output projection) is performed with PyTorch
    fused operations.
    """
    import torch
    import torch.nn.functional as F
    import triton

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------- LayerNorm (FP32) ----------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ---------- Linear projections ----------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16), bias=None)

    # ---------- Gates ----------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # ---------- Apply gates and optional mask ----------
    # Elementwise multiplication in FP16
    left = left * left_gate
    right = right * right_gate
    if not nomask and mask is not None:
        mask = mask.to(torch.float16).unsqueeze(-1)  # shape (B,N,N,1)
        left = left * mask
        right = right * mask

    # ---------- Core O(N³) contraction via Triton ----------
    # Rearrange to (B, hidden_dim, N, N) and flatten batch & hidden dimensions
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_perm.shape
    BH = B * H

    left_mat = left_perm.view(BH, N, N)
    right_mat = right_perm.view(BH, N, N)

    out_mat = torch.empty_like(left_mat)  # (BH, N, N) half

    # Strides (in elements) for flattened tensors (row‑major)
    stride_l_batch = left_mat.stride(0)
    stride_l_row = left_mat.stride(1)
    stride_l_col = left_mat.stride(2)

    stride_r_batch = right_mat.stride(0)
    stride_r_row = right_mat.stride(1)
    stride_r_col = right_mat.stride(2)

    stride_o_batch = out_mat.stride(0)
    stride_o_row = out_mat.stride(1)
    stride_o_col = out_mat.stride(2)

    # Tile sizes – chosen to fit in shared memory and provide good occupancy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        BH,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    trimul_outgoing_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_l_batch,
        stride_l_row,
        stride_l_col,
        stride_r_batch,
        stride_r_row,
        stride_r_col,
        stride_o_batch,
        stride_o_row,
        stride_o_col,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ---------- Reshape back to (B, N, N, hidden_dim) ----------
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- Output LayerNorm ----------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # ---------- Out‑gate scaling ----------
    out = out * out_gate

    # ---------- Final linear projection ----------
    out = F.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    # Return as float32 as in reference implementation
    return out.to(torch.float32)

"""
Outgoing Triangular Multiplicative Update (TriMul) forward pass.

The implementation follows the AlphaFold3 specification:
1. Layer‑norm the input (FP32) and cast to FP16.
2. Linear projections to a hidden dimension and sigmoid gated masks.
3. Fuse optional mask with the gates (element‑wise FP16 multiplication).
4. Core O(N³) contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM using a custom Triton kernel that
   computes C = A @ Bᵀ for each hidden channel in FP16 (accumulating in FP32).
5. Output layer‑norm, out‑gate scaling and final linear projection back to the
   original dimension.
All heavy arithmetic is performed in half‑precision, the final result is
returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # * half  (BH, N, N)
    right_ptr,         # * half  (BH, N, N)
    out_ptr,           # * half  (BH, N, N)
    stride_l_batch, stride_l_row, stride_l_col,
    stride_r_batch, stride_r_row, stride_r_col,
    stride_o_batch, stride_o_row, stride_o_col,
    N: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication C = A @ Bᵀ where
    A = left  (BH, N, N), B = right (BH, N, N), C = out (BH, N, N).
    The kernel is tiled on M, N and K dimensions.
    """
    batch_id = tl.program_id(0)          # which (B*H) pair
    pid_m = tl.program_id(1)             # tile row
    pid_n = tl.program_id(2)             # tile col

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for the current batch
    left_batch_ptr = left_ptr + batch_id * stride_l_batch
    right_batch_ptr = right_ptr + batch_id * stride_r_batch
    out_batch_ptr = out_ptr + batch_id * stride_o_batch

    # Number of K‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(num_k):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile (left) : shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_batch_ptr + (offs_m[:, None] * stride_l_row) + (offs_k[None, :] * stride_l_col)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load Bᵀ tile (right transposed) : shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_batch_ptr + (offs_k[:, None] * stride_r_row) + (offs_n[None, :] * stride_r_col)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Write the tile back
    c_ptrs = out_batch_ptr + (offs_m[:, None] * stride_o_row) + (offs_n[None, :] * stride_o_col)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of torch.Tensors (float32) containing all linear
                         and layer‑norm parameters
        - config       : dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------- 1. Input LayerNorm (FP32) ----------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ---------- 2. Linear projections ----------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16), bias=None)

    # ---------- 3. Gates ----------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # ---------- 4. Apply gates and optional mask (element‑wise FP16) ----------
    left = left * left_gate
    right = right * right_gate
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_h
        right = right * mask_h

    # ---------- 5. Core O(N³) contraction using Triton ----------
    # Rearrange to (B, hidden_dim, N, N) and flatten batch & hidden dims
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    BH = B * H

    left_mat = left_perm.view(BH, N, N)   # (BH, N, N)
    right_mat = right_perm.view(BH, N, N) # (BH, N, N)

    out_mat = torch.empty_like(left_mat)  # (BH, N, N), half

    # Strides in elements for the flattened tensors (row‑major layout)
    stride_l_batch = left_mat.stride(0)
    stride_l_row = left_mat.stride(1)
    stride_l_col = left_mat.stride(2)

    stride_r_batch = right_mat.stride(0)
    stride_r_row = right_mat.stride(1)
    stride_r_col = right_mat.stride(2)

    stride_o_batch = out_mat.stride(0)
    stride_o_row = out_mat.stride(1)
    stride_o_col = out_mat.stride(2)

    # Tile sizes – chosen to respect shared‑memory limits and provide good occupancy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        BH,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    trimul_outgoing_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_l_batch,
        stride_l_row,
        stride_l_col,
        stride_r_batch,
        stride_r_row,
        stride_r_col,
        stride_o_batch,
        stride_o_row,
        stride_o_col,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ---------- 6. Reshape back to (B, N, N, hidden_dim) ----------
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- 7. Output LayerNorm ----------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # ---------- 8. Out‑gate scaling ----------
    out = out * out_gate

    # ---------- 9. Final linear projection ----------
    out = F.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    # Return in float32 as the reference model does
    return out.to(torch.float32)
