@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute row and col offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute starting pointers
    a_ptrs = a_ptr + batch_id * stride_a_batch + offs_m[:, None] * stride_a_m + 0
    b_ptrs = b_ptr + batch_id * stride_b_batch + 0 + offs_n[None, :] * stride_b_n

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_tile = tl.load(a_ptrs + offs_k[None, :] * stride_a_k,
                         mask=mask_m[:, None] & mask_k[None, :],
                         other=0.0)
        b_tile = tl.load(b_ptrs + offs_k[:, None] * stride_b_k,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)
        acc += tl.dot(a_tile, b_tile)

        # advance pointers
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # Write back
    c_ptrs = c_ptr + batch_id * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

"""Outgoing TriMul (AlphaFold3) forward pass using Triton for mask‑gate fusion.

Algorithm
---------
1. Layer‑norm the input tensor (float32 → float16).
2. Linear projections to the hidden dimension (no bias).
3. Compute three gate tensors with a sigmoid.
4. Fuse optional mask with left/right gate multiplications in a Triton kernel
   (in‑place, half‑precision).
5. Core O(N³) contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM (left @ rightᵀ) using torch.bmm on half‑precision
   tensors.
6. Output LayerNorm, element‑wise gating with out_gate and a final linear
   projection back to the original feature dimension.
7. Return the result in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,            # *half
    right_ptr,           # *half
    left_gate_ptr,       # *half
    right_gate_ptr,      # *half
    mask_ptr,            # *half (broadcasted over hidden dim)
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    H: tl.constexpr,     # hidden dimension
    HAS_MASK: tl.constexpr,
):
    """In‑place: left = left * left_gate * mask, right = right * right_gate * mask."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    lgate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # one mask value per (b,i,j) entry → repeat across hidden dimension
        mask_idx = offs // H
        mval = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * lgate * mval
        right = right * rgate * mval
    else:
        left = left * lgate
        right = right * rgate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_mask_and_gate(left, right, left_gate, right_gate, mask, hidden_dim):
    """Launch Triton kernel to fuse mask and gates in‑place (half precision)."""
    total = left.numel()
    BLOCK = 65536  # large enough to keep launch overhead low
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when mask absent

    _mask_gate_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        H=hidden_dim,
        HAS_MASK=has_mask,
    )
    # Kernel updates left/right in place; nothing to return


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict mapping parameter names to float32 tensors
            - config: dict with "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    inp, mask, weights, config = data
    device = inp.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. LayerNorm on input (float32 → float16)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(torch.float16, device=device)
    norm_b = weights["norm.bias"].to(torch.float16, device=device)
    x = F.layer_norm(inp, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections to hidden dimension (no bias)
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16, device=device)
    right_proj_w = weights["right_proj.weight"].to(torch.float16, device=device)

    left = F.linear(x, left_proj_w)   # [B, N, N, H]
    right = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 3. Gate tensors (sigmoid)
    # --------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16, device=device)
    right_gate_w = weights["right_gate.weight"].to(torch.float16, device=device)
    out_gate_w = weights["out_gate.weight"].to(torch.float16, device=device)

    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()   # used after output LayerNorm

    # --------------------------------------------------------------
    # 4. Fuse mask with gates (optional) via Triton
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → broadcast over hidden dim
        mask_h = mask.to(torch.float16, device=device).unsqueeze(-1)  # [B,N,N,1]

        # Flatten tensors for kernel launch
        left_f = left.view(-1)
        right_f = right.view(-1)
        lg_f = left_gate.view(-1)
        rg_f = right_gate.view(-1)
        mask_f = mask_h.view(-1)   # length B*N*N

        _apply_mask_and_gate(left_f, right_f, lg_f, rg_f, mask_f, hidden_dim)

        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # Simple element‑wise multiplication when mask is absent
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core O(N³) contraction via batched GEMM (half precision)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    B, N, _, _ = left.shape  # left shape = (B, N, N, H)

    # Rearrange to (B, H, N, N) and make contiguous for cuBLAS
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Collapse batch and hidden dimensions
    left_mat = left_t.view(B * hidden_dim, N, N)    # (B*H, N, N)
    right_mat = right_t.view(B * hidden_dim, N, N)  # (B*H, N, N)

    # Batched matrix multiplication: (N×N) × (N×N)ᵀ → (N×N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore original layout
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating, final linear projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16, device=device)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16, device=device)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    out = out * out_gate  # element‑wise gating

    to_out_w = weights["to_out.weight"].to(torch.float16, device=device)
    out = F.linear(out, to_out_w)  # back to original dim

    # --------------------------------------------------------------
    # 7. Return in float32 as specified
    # --------------------------------------------------------------
    return out.to(torch.float32)
