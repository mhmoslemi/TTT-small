"""Outgoing TriMul (AlphaFold3) forward pass.

The implementation follows the reference PyTorch code:
1. Layer‑norm the input (float32 → float16).
2. Linear projections to the hidden dimension (no bias).
3. Compute three gate tensors with a sigmoid.
4. Fuse optional mask with the left/right gate multiplications in a
   Triton kernel (in‑place, half‑precision).
5. Core O(N³) contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   implemented as a batched GEMM over the hidden dimension using
   ``torch.bmm`` on half‑precision tensors.
6. Output LayerNorm, element‑wise gating with ``out_gate`` and a final
   linear projection back to the original feature dimension.
7. Return the result as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # *half  (flattened left tensor)
    right_ptr,         # *half  (flattened right tensor)
    left_gate_ptr,     # *half  (flattened left‑gate tensor)
    right_gate_ptr,    # *half  (flattened right‑gate tensor)
    mask_ptr,          # *half  (flattened mask tensor, length = total/H)
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,   # hidden dimension
):
    """In‑place: left = left * left_gate * mask, right = right * right_gate * mask."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elems

    # Load values
    left = tl.load(left_ptr + offs, mask=m, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=m, other=tl.float16(0.0))
    lgate = tl.load(left_gate_ptr + offs, mask=m, other=tl.float16(0.0))
    rgate = tl.load(right_gate_ptr + offs, mask=m, other=tl.float16(0.0))

    if HAS_MASK:
        # one mask value per (b,i,k) → repeat over hidden dim
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=m, other=tl.float16(0.0))
        left = left * lgate * mask_val
        right = right * rgate * mask_val
    else:
        left = left * lgate
        right = right * rgate

    # Store back
    tl.store(left_ptr + offs, left, mask=m)
    tl.store(right_ptr + offs, right, mask=m)


def _apply_mask_and_gate(left, right, left_gate, right_gate, mask, hidden_dim):
    """Launch Triton kernel that fuses mask and gate multiplications (in‑place)."""
    total = left.numel()
    BLOCK = 1024                     # ≤ max threads per block
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy ptr when mask missing

    _mask_gate_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # In‑place modification; nothing to return


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of parameter tensors (float32)
            - config: dict containing at least "dim" and "hidden_dim",
                      optionally "nomask" (bool)

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    inp, mask, weights, config = data
    device = inp.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 → half)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections to hidden dimension
    # -----------------------------------------------------------------
    left = F.linear(
        x, weights["left_proj.weight"].to(torch.float16)
    )  # [B, N, N, H]
    right = F.linear(
        x, weights["right_proj.weight"].to(torch.float16)
    )

    # -----------------------------------------------------------------
    # 3. Gate tensors (sigmoid)
    # -----------------------------------------------------------------
    left_gate = F.linear(
        x, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = F.linear(
        x, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = F.linear(
        x, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    # -----------------------------------------------------------------
    # 4. Fuse mask with gates (if required) via Triton
    # -----------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → broadcast over hidden dimension
        mask_h = mask.to(torch.float16).unsqueeze(-1)          # [B,N,N,1]

        # Flatten tensors for the kernel
        left_f = left.view(-1)
        right_f = right.view(-1)
        left_gate_f = left_gate.view(-1)
        right_gate_f = right_gate.view(-1)
        mask_f = mask_h.view(-1)                               # length B*N*N

        _apply_mask_and_gate(
            left_f, right_f, left_gate_f, right_gate_f, mask_f, hidden_dim
        )

        # Restore original shapes
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: batched matrix multiplication per hidden slice
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape  # left shape: (B, N, N, H)

    # Move hidden dim to batch axis and make contiguous for cuBLAS
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_mat = left_t.view(B * hidden_dim, N, N)     # (B*H, N, N)
    right_mat = right_t.view(B * hidden_dim, N, N)   # (B*H, N, N)

    # Batched GEMM over the sequence dimension
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return as float32 as required
    return out.to(torch.float32)
