"""
Outgoing TriMul (AlphaFold3) forward pass.

Algorithm
---------
1. Layer‑norm the 4‑D input (half precision).
2. Linear projections → left / right (half precision).
3. Compute three sigmoid gates (half precision).
4. Fuse mask (if present) with left / right gates in a single Triton kernel.
5. Core contraction:
       out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   Implemented as a batched GEMM (torch.bmm) on half tensors.
6. Output layer‑norm, out‑gate scaling and final linear projection.
   All steps stay in half precision; the result is cast back to float32.

A Triton kernel is used for the element‑wise gating + optional mask fusion,
which eliminates extra Python‑level passes and keeps the heavy GEMM in cuBLAS.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def gate_mask_fuse_kernel(
    left_ptr,          # * half
    right_ptr,         # * half
    left_gate_ptr,     # * half
    right_gate_ptr,    # * half
    mask_ptr,          # * half (broadcasted)
    total_elements,    # i32
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    # Load data
    left = tl.load(left_ptr + offs, mask=valid, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=valid, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=tl.float16(0.0))

    if HAS_MASK:
        # mask is broadcast over the hidden dimension
        mask_idx = offs // HIDDEN_DIM
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=tl.float16(1.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store back in‑place
    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    ``mask`` may be None (no masking). All tensors must be contiguous and half.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when HAS_MASK=0

    gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        HIDDEN_DIM=hidden_dim,
    )
    # The operation is performed in‑place; nothing is returned.


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of torch.Tensors (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm on half‑precision tensors
    # ------------------------------------------------------------------
    # Cast everything to half once
    input_h = input_tensor.to(torch.float16)

    norm_weight_h = weights["norm.weight"].to(torch.float16)
    norm_bias_h = weights["norm.bias"].to(torch.float16)

    x = F.layer_norm(
        input_h,
        (dim,),
        weight=norm_weight_h,
        bias=norm_bias_h,
    )  # (B, N, N, dim), half

    # ------------------------------------------------------------------
    # 2. Linear projections (half)
    # ------------------------------------------------------------------
    left_proj_w_h = weights["left_proj.weight"].to(torch.float16)
    right_proj_w_h = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x, left_proj_w_h, bias=None)   # (B, N, N, hidden_dim)
    right = F.linear(x, right_proj_w_h, bias=None) # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3. Gates (half)
    # ------------------------------------------------------------------
    left_gate_w_h = weights["left_gate.weight"].to(torch.float16)
    right_gate_w_h = weights["right_gate.weight"].to(torch.float16)
    out_gate_w_h = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x, left_gate_w_h, bias=None).sigmoid()
    right_gate = F.linear(x, right_gate_w_h, bias=None).sigmoid()
    out_gate = F.linear(x, out_gate_w_h, bias=None).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) with gates via Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # Prepare mask: broadcast over hidden_dim
        mask_h = mask.to(torch.float16).contiguous().view(-1)  # (B*N*N,)

        # Ensure contiguous flattened tensors for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_h,
            hidden_dim,
        )
        # Reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # Simple element‑wise gating in PyTorch (fast when mask absent)
        left = left * left_gate
        right = right * right_gate

    # Release memory that is no longer needed
    del left_gate, right_gate, x

    # ------------------------------------------------------------------
    # 5. Core contraction: batched GEMM (half precision)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden_dim, N, N) and fuse batch & hidden dims
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # Batched GEMM: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore (B, N, N, hidden_dim) layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w_h = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b_h = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w_h,
        bias=to_out_norm_b_h,
    )
    out = out * out_gate  # element‑wise scaling (half)

    to_out_w_h = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w_h, bias=None)  # (B, N, N, dim), half

    # Return as float32 (as in the reference implementation)
    return out.to(torch.float32)
