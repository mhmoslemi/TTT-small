"""
Optimized TriMul (outgoing) forward pass.

- LayerNorm on input.
- Combined linear layers for left/right projections and gates (half precision).
- Optional mask & gate fusion via a tiny Triton kernel.
- Batched GEMM (left @ right^T) per hidden channel using torch.bmm.
- Output LayerNorm, out‑gate scaling, final linear projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half*  flattened left tensor
    right_ptr,         # half*  flattened right tensor
    left_gate_ptr,     # half*  flattened left_gate
    right_gate_ptr,    # half*  flattened right_gate
    mask_ptr,          # half*  flattened mask (B*N*N)
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,   # hidden dim
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # each hidden channel shares the same mask entry
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask: torch.Tensor [B, N, N] (float32 or bool)
            - weights_dict: dict with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict: dict containing "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) -> half precision
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Combined linear projections for left/right (half)
    # ------------------------------------------------------------------
    # flatten input to (B*N*N, dim)
    B, N, _, _ = input_tensor.shape
    flat_x = x.view(-1, dim)  # (B*N*N, dim)

    # combine left_proj and right_proj weights
    left_w = weights["left_proj.weight"]
    right_w = weights["right_proj.weight"]
    proj_w = torch.cat([left_w, right_w], dim=0).to(torch.float16).contiguous()  # (2*H, dim)

    proj_out = F.linear(flat_x, proj_w, bias=None)  # (B*N*N, 2*H)
    left = proj_out[:, :hidden_dim].view(B, N, N, hidden_dim)
    right = proj_out[:, hidden_dim:].view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3. Combined linear for gates (half) + sigmoid
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    gate_w = torch.cat([left_gate_w, right_gate_w, out_gate_w], dim=0).to(torch.float16).contiguous()  # (3*H, dim)

    gate_out = F.linear(flat_x, gate_w, bias=None)  # (B*N*N, 3*H)
    left_gate = gate_out[:, :hidden_dim].view(B, N, N, hidden_dim).sigmoid()
    right_gate = gate_out[:, hidden_dim:2*hidden_dim].view(B, N, N, hidden_dim).sigmoid()
    out_gate = gate_out[:, 2*hidden_dim:].view(B, N, N, hidden_dim).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) with gating via a tiny Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        # mask is broadcast over hidden dimension
        mask_h = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_h.view(-1)  # (B*N*N)

        # flatten tensors for kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        total = left_f.numel()
        BLOCK = 1024  # max threads per block
        grid = (triton.cdiv(total, BLOCK),)

        _gate_mask_fuse_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_flat,
            total,
            BLOCK=BLOCK,
            HAS_MASK=1,
            H=hidden_dim,
        )
        # reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # simple element‑wise gating without mask
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as batched GEMM (left @ rightᵀ) per hidden channel.
    # ------------------------------------------------------------------
    # shape to (B, H, N, N) for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N) # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w, bias=None)

    # Return in float32 as expected by downstream modules
    return out.to(torch.float32)

"""
Optimized outgoing TriMul forward pass.

Algorithm
---------
1. Layer‑norm the input tensor (float32) and cast to float16.
2. Combine linear projections:
   - One fused matmul computes both `left` and `right` projections.
   - A second fused matmul computes `left_gate`, `right_gate` and `out_gate`.
3. Apply sigmoid to the three gate tensors.
4. If a mask is present, fuse element‑wise mask & gate multiplication with a
   tiny Triton kernel (otherwise use a simple element‑wise multiply).
5. Core contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM (left @ rightᵀ) per hidden channel using
   `torch.bmm`.
6. Output LayerNorm, scale by `out_gate`, final linear projection back to the
   original feature dimension.
7. Return the result in float32.

A Triton kernel is used for the mask‑gate fusion; the heavy matrix
multiplications are performed with half‑precision `torch.nn.functional.linear`
and `torch.bmm`, which map to highly‑optimized cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half*  flattened left tensor
    right_ptr,         # half*  flattened right tensor
    left_gate_ptr,     # half*  flattened left_gate
    right_gate_ptr,    # half*  flattened right_gate
    mask_ptr,          # half*  flattened mask (B*N*N)
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,   # hidden dimension
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    left      = tl.load(left_ptr      + offs, mask=mask, other=tl.float16(0.0))
    right     = tl.load(right_ptr     + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate= tl.load(right_gate_ptr+ offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # each hidden channel shares the same mask entry
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    tl.store(left_ptr  + offs, left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask: torch.Tensor [B, N, N] (float32 or bool)
            - weights_dict: contains all learned parameters
            - config_dict: must contain "dim", "hidden_dim" and optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) -> half precision for the heavy part
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)                               # (B, N, N, dim)

    B, N, _, _ = input_tensor.shape
    flat_x = x.view(-1, dim)                              # (B*N*N, dim)

    # ------------------------------------------------------------------
    # 2. Combined linear projection for left/right (half)
    # ------------------------------------------------------------------
    proj_w = torch.cat(
        [weights["left_proj.weight"], weights["right_proj.weight"]], dim=0
    ).to(torch.float16).contiguous()                     # (2*H, dim)

    proj_out = F.linear(flat_x, proj_w, bias=None)        # (B*N*N, 2*H)
    left  = proj_out[:, :hidden_dim].view(B, N, N, hidden_dim)
    right = proj_out[:, hidden_dim:].view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3. Combined linear for gates (half) + sigmoid
    # ------------------------------------------------------------------
    gate_w = torch.cat(
        [
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16).contiguous()                    # (3*H, dim)

    gate_out = F.linear(flat_x, gate_w, bias=None)       # (B*N*N, 3*H)
    left_gate  = gate_out[:, :hidden_dim].view(B, N, N, hidden_dim).sigmoid()
    right_gate = gate_out[:, hidden_dim:2*hidden_dim].view(B, N, N, hidden_dim).sigmoid()
    out_gate   = gate_out[:, 2*hidden_dim:].view(B, N, N, hidden_dim).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) with gating via Triton
    # ------------------------------------------------------------------
    if not nomask:
        mask_h   = mask_tensor.to(torch.float16).contiguous()
        mask_f   = mask_h.view(-1)                        # (B*N*N)

        # flatten tensors for the kernel
        left_f       = left.contiguous().view(-1)
        right_f      = right.contiguous().view(-1)
        left_gate_f  = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        total = left_f.numel()
        BLOCK = 1024                                        # max threads per block
        grid  = (triton.cdiv(total, BLOCK),)

        _gate_mask_fuse_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            total,
            BLOCK=BLOCK,
            HAS_MASK=1,
            H=hidden_dim,
        )
        # reshape back to original layout
        left  = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # simple element‑wise gating without a mask
        left  = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as batched GEMM (left @ rightᵀ) per hidden channel.
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat   = left_perm.view(B * H, N, N)           # (B*H, N, N)
    right_mat  = right_perm.view(B * H, N, N)          # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 7. Cast back to float32 for downstream consumption
    # ------------------------------------------------------------------
    return out.to(torch.float32)
