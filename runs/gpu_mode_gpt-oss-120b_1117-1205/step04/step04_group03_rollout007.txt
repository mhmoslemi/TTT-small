"""
TriMul (outgoing) forward pass with Triton acceleration.

Algorithm
---------
1. Layer‑norm the input tensor (float32) → FP16.
2. Linear projections to a hidden dimension H (FP16).
3. Compute three gate tensors (FP16) with sigmoid.
4. Fuse mask (if present) and the left/right gate multiplications in a
   custom Triton kernel (`gate_mask_fuse_kernel`).
5. Compute the core contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   as a batched matrix multiplication (left @ rightᵀ) using a
   Triton GEMM kernel (`matmul_kernel`).  The tensors are reshaped to
   (B·H, N, N) and processed as independent GEMMs.
6. Output LayerNorm (FP16), element‑wise scaling by `out_gate`, and final
   linear projection back to the original dimension.
7. Cast the result back to float32.

Only the mask–gate fusion and the batched GEMM are performed on the GPU
via Triton; the remaining lightweight operations use PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def gate_mask_fuse_kernel(
    left_ptr,          # half*  (flattened left tensor)
    right_ptr,         # half*  (flattened right tensor)
    left_gate_ptr,     # half*  (flattened left_gate)
    right_gate_ptr,    # half*  (flattened right_gate)
    mask_ptr,          # half*  (flattened mask, length B*N*N) – dummy if not used
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask supplied, else 0
    H: tl.constexpr,          # hidden dimension
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each mask value is shared across the hidden dimension
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


@triton.jit
def matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to A, B, C (all half)
    stride_ah, stride_am, stride_ak,    # batch, row, col strides for A
    stride_bh, stride_bn, stride_bk,    # batch, row, col strides for B
    stride_ch, stride_cm, stride_cn,    # batch, row, col strides for C
    M, N, K,                            # matrix dimensions (all int32)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_b = tl.program_id(2)   # batch*head dimension
    pid_m = tl.program_id(0)   # row tile
    pid_n = tl.program_id(1)   # col tile

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + pid_b * stride_ah
    b_batch = b_ptr + pid_b * stride_bh
    c_batch = c_ptr + pid_b * stride_ch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B tile (need Bᵀ): (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # accumulate in FP32
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

    # write result back in FP16
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (optional, may be None)
            - weights_dict: dict containing all model weights
            - config_dict: {"dim": C, "hidden_dim": H, "nomask": bool (optional)}

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gates (half) and sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask and gate multiplications via Triton
    # ------------------------------------------------------------------
    # Flatten tensors for the kernel
    left_f = left.contiguous().view(-1)
    right_f = right.contiguous().view(-1)
    left_gate_f = left_gate.contiguous().view(-1)
    right_gate_f = right_gate.contiguous().view(-1)

    if not nomask:
        mask_f = mask.to(torch.float16).contiguous().view(-1)  # [B*N*N]
    else:
        # dummy placeholder (will be ignored by the kernel)
        mask_f = left_f

    total = left_f.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    HAS_MASK = 0 if nomask else 1
    gate_mask_fuse_kernel[grid](
        left_f,
        right_f,
        left_gate_f,
        right_gate_f,
        mask_f,
        total,
        BLOCK=BLOCK,
        HAS_MASK=HAS_MASK,
        H=hidden_dim,
    )

    # Reshape back to original layout
    left = left_f.view_as(left)
    right = right_f.view_as(right)

    # ------------------------------------------------------------------
    # 5. Core contraction: batched GEMM left @ rightᵀ per hidden head
    # ------------------------------------------------------------------
    # Prepare tensors of shape (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    batch_head = B * H

    # Collapse batch & head dimensions for the GEMM
    left_mat = left_perm.view(batch_head, N, N)
    right_mat = right_perm.view(batch_head, N, N)
    out_mat = torch.empty_like(left_mat)   # (batch_head, N, N), half

    # Block sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch_head,
    )
    matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore (B, N, N, H) layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gate scaling and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in original dtype (float32)
    return out.to(torch.float32)
