def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Move to half for performance
    # Use layernorm in float32 then convert to half
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = torch.nn.functional.linear(x, weights["right_proj.weight"].to(torch.float16), bias=None)

    # Gate values
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # Fuse gates and mask using Triton
    if not nomask:
        # mask shape [B,N,N]
        mask_h = mask.to(torch.float16).contiguous()
        mask_h = mask_h.unsqueeze(-1)  # shape [B,N,N,1]
    else:
        mask_h = None

    # Ensure contiguous
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    if mask_h is not None:
        mask_h = mask_h.contiguous()

    fuse_gates_and_mask(left, right, left_gate, right_gate, mask_h, hidden_dim)

    # Now left and right are ready
    del left_gate, right_gate # free memory

    # Compute out = einsum ... using batched GEMM
    # Permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    # Reshape to (B*H, N, N)
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Multiply left_mat (B*H, N, N) with right_mat.transpose(1,2) (B*H, N, N) -> (B*H, N, N)
    # Use torch.bmm which uses half precision
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output LayerNorm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    # Apply out_gate scaling
    out = out * out_gate

    # Final linear projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    # Return float32
    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input (float32) then cast to half.
2. Linear projections to `hidden_dim` for left/right tensors.
3. Compute gating values (sigmoid) for left, right and out gates.
4. Fuse element‑wise mask (if present) and gate multiplications for the
   left/right tensors using a Triton kernel (in‑place, half‑precision).
5. Perform the core O(N³) contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM (torch.bmm) after reshaping
   (hidden‑dim is treated as batch dimension).
6. Apply output LayerNorm, out‑gate scaling and final linear projection.
7. Return the result as float32.

The kernel only handles the element‑wise part; the heavy matrix multiply
uses cuBLAS via torch.bmm which is already highly optimized.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any, Optional


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # * half
    right_ptr,         # * half
    left_gate_ptr,     # * half
    right_gate_ptr,    # * half
    mask_ptr,          # * half (optional, broadcasted)
    total_elements: tl.int32,
    hidden_dim: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # Each hidden_dim elements share the same mask entry
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_mask_fuse(
    left: torch.Tensor,
    right: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: Optional[torch.Tensor],
    hidden_dim: int,
) -> None:
    """
    In‑place fuse of mask (if any) and gating for left/right tensors using Triton.
    All tensors must be contiguous half‑precision.
    """
    total = left.numel()
    BLOCK = 32768  # multiple of 32, fits well on modern GPUs
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when mask not used

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        hidden_dim,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )
    # No return; tensors are modified in‑place


def custom_kernel(data: Tuple[torch.Tensor, Optional[torch.Tensor], Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, dim] (float32)
            - mask_tensor:  [B, N, N] (bool/float) or None
            - weights: dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict with fields "dim", "hidden_dim", optional "nomask"
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) -> half
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gating values (half, sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) with gates via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> half, broadcast to hidden dim
        mask_h = mask.to(torch.float16).contiguous()
        mask_h = mask_h.unsqueeze(-1)  # [B, N, N, 1]
        # Ensure contiguity for all tensors
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_h = mask_h.contiguous()

        _apply_gate_mask_fuse(
            left,
            right,
            left_gate,
            right_gate,
            mask_h,
            hidden_dim,
        )
    else:
        # Simple element‑wise fusion without Triton
        left = left * left_gate
        right = right * right_gate

    # Release unused tensors early
    del left_gate, right_gate, x

    # ------------------------------------------------------------------
    # 5. Core contraction via batched GEMM (torch.bmm)
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    # Permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    # Shape (B*H, N, N)
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Batched matrix multiplication (half precision)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original layout: (B, N, N, hidden_dim)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in float32 as required by the reference implementation
    return out.to(torch.float32)
