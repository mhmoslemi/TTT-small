        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    left = F.linear(x, left_proj_weight, bias=None)   # (B,N,N,H)
    right = F.linear(x, right_proj_weight, bias=None)

    left_gate = F.linear(x, left_gate_weight, bias=None).sigmoid()
    right_gate = F.linear(x, right_gate_weight, bias=None).sigmoid()

    gate_val = tl.load(gate_ptr + offs, mask=mask)
    out_val = out_val * gate_val
    tl.store(out_ptr + offs, out_val, mask=mask)

            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor:  torch.Tensor [B, N, N] (bool/float) – may be ignored.
            - weights_dict: model weights with the keys used below.
            - config_dict: must contain "dim", "hidden_dim", optional "nomask".

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Cast weights to half precision on the correct device
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device=device, dtype=torch.float16)
    norm_bias = weights["norm.bias"].to(device=device, dtype=torch.float16)

    left_proj_weight = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    left_gate_weight = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)

    to_out_weight = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 2. Input LayerNorm (float16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    ).to(torch.float16)  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 3. Linear projections to the hidden dimension
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight, bias=None)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight, bias=None)

    # ------------------------------------------------------------------
    # 4. Gating values (sigmoid) – also half precision
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight, bias=None).sigmoid()   # [B, N, N, H]
    right_gate = F.linear(x, right_gate_weight, bias=None).sigmoid()
    out_gate = F.linear(x, out_gate_weight, bias=None).sigmoid()    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5. Optional mask (broadcasted over hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(device=device, dtype=torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 6. Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as a batched GEMM over the hidden dimension.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # Move hidden dimension to the batch axis for a single torch.bmm call
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # GEMM: (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm (half precision)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # ------------------------------------------------------------------
    # 9. Fuse out‑gate scaling via a tiny Triton kernel
    # ------------------------------------------------------------------
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    numel = out.numel()
    BLOCK = 65536  # multiple of 32
    grid = (triton.cdiv(numel, BLOCK),)
    _out_gate_scale_kernel[grid](
        out, out_gate, numel, BLOCK=BLOCK
    )

    # ------------------------------------------------------------------
    # 10. Final linear projection back to original dimensionality
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight, bias=None)   # [B, N, N, dim]

    # Return in float32 as required
    return out.to(torch.float32)
