def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float16

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )
    x = x.to(dtype)

    # Projections
    left = F.linear(x, weights["left_proj.weight"].to(dtype), bias=None)  # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"].to(dtype), bias=None)

    # Gate values
    left_gate = F.linear(x, weights["left_gate.weight"].to(dtype), bias=None).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(dtype), bias=None).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(dtype), bias=None).sigmoid()

    # Fuse mask and gates via Triton (in-place on left/right)
    if not nomask and mask is not None:
        # mask shape [B,N,N]
        mask_fp16 = mask.to(dtype).unsqueeze(-1).contiguous()
        _fuse_gate_and_mask(left, right, left_gate, right_gate, mask_fp16, hidden_dim)
    else:
        # no mask, just multiply gates
        left = left * left_gate
        right = right * right_gate

    # Core contraction using batched matmul per hidden dim
    # Reshape for batched gemm
    # Ensure contiguous for permute
    left = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right = right.permute(0, 3, 2, 1).contiguous() # [B,H,N,N] and transposed K and J

    B, H, N, _ = left.shape
    left_flat = left.reshape(B*H, N, N)
    right_flat = right.reshape(B*H, N, N)
    out_flat = torch.matmul(left_flat, right_flat)   # (B*H, N, N) with fp16
    out = out_flat.reshape(B, H, N, N).permute(0, 2, 3, 1)  # [B,N,N,H]

    # Output layernorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    # Apply out_gate
    out = out * out_gate

    # Final linear projection back to dim
    out = F.linear(out, weights["to_out.weight"].to(dtype), bias=None)

    return out.to(torch.float32)

def _gate_mask_fuse_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr,
    total_elems,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,

    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)

"""TriMul “outgoing” forward pass with Triton‑fused mask+gate and
batched GEMM for the N³ contraction.

Algorithm
---------
1. Layer‑norm the input (float16).
2. Linear projections → left, right (hidden_dim channels).
3. Compute left/right/out gates (sigmoid).
4. Fuse mask (if any) and left/right gates in‑place with a Triton kernel.
5. Perform the core contraction
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   by reshaping to (B*H, N, N) and using a batched torch.matmul
   (equivalent to left @ rightᵀ for each hidden slice).
6. Layer‑norm the result, apply the out‑gate, and a final linear projection.
7. Return the result in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask (if present) with left/right gates
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half *
    right_ptr,         # half *
    left_gate_ptr,     # half *
    right_gate_ptr,    # half *
    mask_ptr,          # half *
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present else 0
    H: tl.constexpr,          # hidden dimension (size of innermost channel)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    left       = tl.load(left_ptr + offs,       mask=mask, other=0.0)
    right      = tl.load(right_ptr + offs,      mask=mask, other=0.0)
    left_gate  = tl.load(left_gate_ptr + offs,  mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each mask element corresponds to H consecutive channel entries
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    tl.store(left_ptr  + offs, left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _fuse_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """In‑place fusion of gates (and optional mask) into left/right."""
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    # dummy pointer when mask is absent (won't be accessed)
    mask_ptr = mask if mask is not None else left

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )

    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] or None
            - weights: dict of model parameters
            - config: dict with keys 'dim' and 'hidden_dim' and optional 'nomask'

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """

    device = input_tensor.device
    dtype = torch.float16

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )
    x = x.to(dtype)

    # ------------------------------------------------------------------
    # 2. Linear projections (to hidden_dim)
    # ------------------------------------------------------------------

    right = F.linear(x, weights["right_proj.weight"].to(dtype), bias=None)

    # ------------------------------------------------------------------
    # 3. Gate values (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, weights["left_gate.weight"].to(dtype), bias=None).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(dtype), bias=None).sigmoid()
    out_gate   = F.linear(x, weights["out_gate.weight"].to(dtype), bias=None).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) and gates via Triton (in‑place)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape: [B,N,N] → broadcast over hidden_dim
        mask_fp16 = mask.to(dtype).unsqueeze(-1).contiguous()  # [B,N,N,1]
        _fuse_gate_and_mask(left, right, left_gate, right_gate, mask_fp16, hidden_dim)
    else:
        # No mask: simple elementwise multiplication
        left  = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core N³ contraction via batched GEMM
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Equivalent to (left_d) @ (right_d).T for each hidden slice.
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) for batched matmul
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (transpose K & J)

    B, H, N, _ = left.shape
    left_flat  = left.reshape(B * H, N, N)
    right_flat = right.reshape(B * H, N, N)

    # Batched matrix multiplication (uses Tensor Cores in fp16)
    out_flat = torch.matmul(left_flat, right_flat)   # (B*H, N, N)
    out = out_flat.reshape(B, H, N, N).permute(0, 2, 3, 1)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(dtype), bias=None)

    # Return in float32 as required by the API
    return out.to(torch.float32)
```
