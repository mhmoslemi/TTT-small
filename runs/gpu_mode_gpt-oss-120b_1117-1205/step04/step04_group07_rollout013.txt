@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    off_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    a_ptrs = a_ptr + batch_id * stride_a_batch + off_m[:, None] * stride_a_m + tl.arange(0, BLOCK_K)[None, :] * stride_a_k
    b_ptrs = b_ptr + batch_id * stride_b_batch + tl.arange(0, BLOCK_K)[:, None] * stride_b_k + off_n[None, :] * stride_b_n

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=off_m < M and tl.arange(0, BLOCK_K) < K, other=0.0)
        b = tl.load(b_ptrs, mask=tl.arange(0, BLOCK_K)[:, None] < K and off_n < N, other=0.0)
        acc += tl.dot(a, b)
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    c = acc.to(tl.float16)
    # store
    c_ptrs = c_ptr + batch_id * stride_c_batch + off_m[:, None] * stride_c_m + off_n[None, :] * stride_c_n
    tl.store(c_ptrs, c, mask=(off_m[:, None] < M) & (off_n[None, :] < N))

"""
Optimized outgoing TriMul forward pass.

Algorithm
---------
1. Layer‑norm the input (float16).
2. Linear projections to hidden dimension (half‑precision, no bias).
3. Optional mask (broadcast) applied element‑wise.
4. Compute three gating vectors (half + sigmoid) and fuse them with the
   projected tensors.
5. Fuse the O(N³) contraction using a custom Triton batched‑GEMM kernel:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
   The hidden dimension d is folded into the batch dimension, so the kernel
   processes (batch*hidden) independent matrix multiplications of size N×N.
6. Layer‑norm over the hidden dimension, multiply by the output gate and
   project back to the original dimension.
7. Return float32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    BATCH, M, N, K,               # sizes
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B  where A:[BATCH,M,K], B:[BATCH,K,N], C:[BATCH,M,N]"""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    #----- tile offsets -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers for the current tile
        a_ptrs = (
            a_ptr
            + batch_id * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        b_ptrs = (
            b_ptr
            + batch_id * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n
        )

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)  # fp16 dot accumulates into fp32

    # store result
    c_ptrs = (
        c_ptr
        + batch_id * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict with entries "dim", "hidden_dim", "nomask" (bool, optional)

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float16)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(torch.float16),
        bias=weights["norm.bias"].to(torch.float16),
    ).to(torch.float16)  # [B, N, N, dim] half

    # --------------------------------------------------------------
    # 2. Linear projections (half)
    # --------------------------------------------------------------
    left = F.linear(
        x, weights["left_proj.weight"].to(torch.float16), bias=None
    )  # [B, N, N, hidden_dim]
    right = F.linear(
        x, weights["right_proj.weight"].to(torch.float16), bias=None
    )  # same shape

    # --------------------------------------------------------------
    # 3. Optional mask (broadcasted over hidden_dim)
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] -> [B,N,N,1] -> half
        mask_h = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_h
        right = right * mask_h

    # --------------------------------------------------------------
    # 4. Gating vectors (half, sigmoid)
    # --------------------------------------------------------------
    left_gate = F.linear(
        x, weights["left_gate.weight"].to(torch.float16), bias=None
    ).sigmoid()
    right_gate = F.linear(
        x, weights["right_gate.weight"].to(torch.float16), bias=None
    ).sigmoid()
    out_gate = F.linear(
        x, weights["out_gate.weight"].to(torch.float16), bias=None
    ).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction via custom batched GEMM (Triton)
    # --------------------------------------------------------------
    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim
    # Rearrange to (B, hidden, N, N) for batching
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    batch = B * hidden_dim
    M = N
    K = N
    N_ = N

    # view as (batch, M, K) and (batch, K, N_)
    left_mat = left_perm.view(batch, M, K)   # (batch, N, N)
    right_mat = right_perm.view(batch, M, K)  # (batch, N, N)

    # We need B as (batch, K, N) i.e. right_mat transposed without copy
    right_mat_T = right_mat.permute(0, 2, 1)  # (batch, K, N) non‑contiguous

    # Allocate output
    out_mat = torch.empty((batch, M, N_), dtype=torch.float16, device=device)

    # Block sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N_, BLOCK_N),
    )

    _batched_gemm_kernel[
        grid
    ](
        left_mat,
        right_mat_T,
        out_mat,
        batch,
        M,
        N_,
        K,
        # strides (in elements)
        left_mat.stride(0),
        left_mat.stride(1),
        left_mat.stride(2),
        right_mat_T.stride(0),
        right_mat_T.stride(1),
        right_mat_T.stride(2),
        out_mat.stride(0),
        out_mat.stride(1),
        out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, hidden_dim] half

    # --------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # pointwise gating (half)

    out = F.linear(
        out, weights["to_out.weight"].to(torch.float16), bias=None
    )  # back to dim (half)

    # Return in float32 as required
    return out.to(torch.float32)
