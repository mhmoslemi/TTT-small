"""
TriMul (outgoing) forward pass with fused mask+gate Triton kernel.

Algorithm
---------
1. Layer‑normalize the input tensor (float32) and cast to float16.
2. Linear projections to hidden_dim → `left` and `right`.
3. Compute sigmoid gates (`left_gate`, `right_gate`, `out_gate`).
4. If a mask is present, a Triton kernel fuses
   `left = left * left_gate * mask` and
   `right = right * right_gate * mask` in‑place.
   Otherwise the element‑wise products are done in PyTorch.
5. Core contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication `left @ rightᵀ`
   for each (batch, hidden) pair.
6. Apply output LayerNorm, scale by `out_gate`, and a final linear
   projection back to the original dimensionality.
7. Cast the result back to float32.

Only the mask‑gate fusion is performed in Triton; the heavy
batched GEMM leverages cuBLAS through `torch.bmm`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # half*   left tensor  (flattened)
    right_ptr,           # half*   right tensor (flattened)
    left_gate_ptr,       # half*   left gate    (flattened)
    right_gate_ptr,      # half*   right gate   (flattened)
    mask_ptr,            # half*   mask tensor  (flattened, broadcast over hidden dim)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present, else 0
    H: tl.constexpr,          # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # One mask value per (batch, i, k) – same for every hidden channel
        mask_offs = offs // H
        mask_val = tl.load(mask_ptr + mask_offs, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of gating and (optional) masking using Triton.
    All tensors must be contiguous and of dtype torch.float16.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when no mask

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # No return – tensors are modified in place

            - mask: torch.Tensor [B, N, N] (bool/float) – may be None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half precision)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x_h, right_proj_w)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Gates (half precision)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_h, left_gate_w).sigmoid()
    right_gate = F.linear(x_h, right_gate_w).sigmoid()
    out_gate = F.linear(x_h, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) with gates via Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # Prepare mask: broadcast over hidden dimension
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]

        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask_h.contiguous().view(-1)  # length = B*N*N

        # Apply fused mask+gate
        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            hidden_dim,
        )

        # Restore original shapes
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask: simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # Release intermediate tensors we no longer need
    del left_gate, right_gate, x_h

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched GEMM (left @ rightᵀ) for each hidden channel.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # left shape = (B, N, N, hidden_dim)

    # Move hidden dim to batch dim for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left_mat = left_perm.view(B * hidden_dim, N, N)     # (B*H, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)   # (B*H, N, N)

    # Batched matrix multiplication with transpose on the second operand
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)

    # Cast back to float32 for the final output
    return out.to(torch.float32)
