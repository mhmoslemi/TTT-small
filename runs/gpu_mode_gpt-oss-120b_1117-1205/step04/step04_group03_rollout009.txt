@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_j,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    batch_heads = B * H
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    bh = pid // (num_i * num_j)
    block_idx = pid % (num_i * num_j)
    i0 = (block_idx // num_j) * BLOCK_M
    j0 = (block_idx % num_j) * BLOCK_N

    # Pointers to start of each batch head
    left_batch_ptr = left_ptr + bh * stride_left_batch
    right_batch_ptr = right_ptr + bh * stride_right_batch
    out_batch_ptr = out_ptr + bh * stride_out_batch

    # Offsets within left and out
    # For i block
    offs_i = i0 + tl.arange(0, BLOCK_M)
    # For j block
    offs_j = j0 + tl.arange(0, BLOCK_N)
    # For k block
    offs_k = tl.arange(0, BLOCK_K)

    # Create masks for boundary
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N

    # Pointers to rows
    # left: shape (i,k)
    left_ptrs = left_batch_ptr + (offs_i[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k)

    # right: we need B_T so shape (k,j)
    right_ptrs = right_batch_ptr + (offs_k[:, None] * stride_right_k) + (offs_j[None, :] * stride_right_j)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # Compute dynamic offsets for k block
        k_offset = k * BLOCK_K
        # Update offsets for k
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Compute dot (accumulate)
        acc += tl.dot(left, right, out_dtype=tl.float32)

        # Advance pointers
        left_ptrs += BLOCK_K * stride_left_k
        right_ptrs += BLOCK_K * stride_right_k

    # Write out
    # out_ptr shape (i,j)
    out_ptrs = out_batch_ptr + (offs_i[:, None] * stride_out_i) + (offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor (float32) and cast to float16.
2. Linear projections to the hidden dimension (left/right) and compute three
   gating vectors (left_gate, right_gate, out_gate) – all in float16.
3. If a mask is supplied, fuse it together with the two gate tensors in‑place
   using a tiny Triton kernel (`_gate_mask_fuse_kernel`).  This removes an
   extra PyTorch pass over the large tensors.
4. Core contraction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM (left @ rightᵀ) per hidden channel.  We
   reshape the tensors to (B*H, N, N) and let cuBLAS (via `torch.bmm`) compute
   the product – the fastest path on modern GPUs.
5. Output LayerNorm, element‑wise scaling by `out_gate`, and final linear
   projection back to the original dimension.
6. Return the result in float32.

Only the mask‑gate fusion is performed in Triton (the requirement “implement at
least part of the operations in a kernel”).  All heavy linear algebra is
delegated to cuBLAS for maximum throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask (if present) with left/right gate tensors.
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # half*  (flattened left tensor)
    right_ptr,           # half*  (flattened right tensor)
    left_gate_ptr,       # half*  (flattened left_gate)
    right_gate_ptr,      # half*  (flattened right_gate)
    mask_ptr,            # half*  (flattened mask, length = B*N*N) – dummy if not used
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if a mask is supplied, else 0
    H: tl.constexpr,          # hidden dimension (size of a gate channel group)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    if HAS_MASK:
        # Every hidden channel shares the same mask entry.
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask):
    """
    Fuse mask and gating into `left` and `right` tensors in‑place using Triton.
    All tensors must be contiguous float16 and have shape [B, N, N, H].
    `mask` may be None (no masking).
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    # If no mask, pass a dummy pointer (the kernel will ignore it).
    mask_ptr = mask if mask is not None else left

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=left.shape[-1],
    )


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        input_tensor : torch.Tensor [B, N, N, C]   (float32)
        mask         : torch.Tensor [B, N, N]    (float32/bool) – may be None
        weights_dict : dict of model parameters (float32)
        config_dict  : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half precision)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gates (half) and sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) with gates via Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # Convert mask to half and broadcast to hidden dimension
        mask_h = mask.to(torch.float16).unsqueeze(-1).contiguous()   # [B,N,N,1]
        # The `_apply_gate_and_mask` kernel expects flattened tensors.
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask_h.contiguous().view(-1)   # [B*N*N]

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
        )

        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask – just apply the gates.
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM (left @ rightᵀ) per hidden channel.
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) layout for batched matmul.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)          # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)        # (B*H, N, N)

    # Batched GEMM: left @ rightᵀ   (half → half, accumulate in float32)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in original dtype (float32)
    return out.to(torch.float32)
