    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_b, stride_h, stride_i, stride_k,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr

    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    h = pid_head % H

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    j_range = j_start + tl.arange(0, BLOCK_J)

"""
Optimized outgoing TriMul forward (AlphaFold3) using Triton.

Algorithm:
1. LayerNorm on the input tensor (float32 → float16).
2. Linear projections to hidden_dim for the left/right tensors.
3. Compute three gate tensors (sigmoid of linear projections).
4. Fuse gates (and optional mask) with left/right tensors using
   element‑wise multiplication in PyTorch.
5. Perform the heavy O(N³·hidden_dim) contraction
        out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   as a batched matrix multiplication implemented in a custom Triton
   kernel.  The tensors are reshaped to (B, H, N, N) and each (b,h)
   pair computes a GEMM of shape (N, N) × (N, N)^T using tiled
   loads and `tl.dot`, accumulating in FP32.
6. Apply the output LayerNorm, out‑gate scaling and the final linear
   projection back to the original `dim`.
7. Return the result in float32.

The Triton kernel processes one head (a specific batch and hidden_dim
index) per program and tiles the i- and j‑dimensions with BLOCK_I×BLOCK_J
tiles, looping over k in BLOCK_K sized chunks.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,          # fp16*
    right_ptr,         # fp16*
    out_ptr,           # fp16*
    B, H, N,           # int32 scalars
    stride_b, stride_h, stride_i, stride_k,   # int64 strides of the (B,H,N,N) tensors
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out[b,h,i,j] = sum_k left[b,h,i,k] * right[b,h,j,k]"""

    pid_head = tl.program_id(0)           # (b * H + h)
    pid_i    = tl.program_id(1)           # block index over i
    pid_j    = tl.program_id(2)           # block index over j

    # Resolve batch and head indices
    b = pid_head // H
    h = pid_head % H

    # Starting coordinates of this tile
    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    # Ranges within the tile
    i_range = i_start + tl.arange(0, BLOCK_I)
    j_range = j_start + tl.arange(0, BLOCK_J)

    # Masks for out‑of‑bounds rows/cols
    i_mask = i_range < N
    j_mask = j_range < N

    # Offset to the beginning of the (b,h) sub‑tensor
    head_offset = b * stride_b + h * stride_h

    # Accumulator in FP32 for better precision
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Loop over the reduction dimension k
    k = 0
    while k < N:
        k_range = k + tl.arange(0, BLOCK_K)
        k_mask = k_range < N

        # ---- load left tile (i,k) ----
        # Shape: (BLOCK_I, BLOCK_K)
        left_ptrs = left_ptr + head_offset \
                    + i_range[:, None] * stride_i \
                    + k_range[None, :] * stride_k
        left_mask = i_mask[:, None] & k_mask[None, :]
        left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0)

        # ---- load right tile transposed (k,j) ----
        # Desired shape for dot: (BLOCK_K, BLOCK_J)
        # Right layout: (j,k) in memory, so we load with k as row, j as column.
        right_ptrs = right_ptr + head_offset \
                     + k_range[:, None] * stride_k \
                     + j_range[None, :] * stride_i
        right_mask = k_mask[:, None] & j_mask[None, :]
        right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0)

        # Accumulate dot product: (BLOCK_I, BLOCK_K) @ (BLOCK_K, BLOCK_J)
        acc += tl.dot(left_tile, right_tile)

        k += BLOCK_K

    # ---- store the result ----
    out_ptrs = out_ptr + head_offset \
               + i_range[:, None] * stride_i \
               + j_range[None, :] * stride_k
    out_mask = i_mask[:, None] & j_mask[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=out_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) – may be ignored
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half)
    # ------------------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gates (half) with sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse gates (and optional mask) with left/right tensors
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → broadcast to hidden_dim
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * left_gate * mask_h
        right = right * right_gate * mask_h
    else:
        left = left * left_gate
        right = right * right_gate

    # Release unused tensors early
    del left_gate, right_gate, x_h

    # ------------------------------------------------------------------
    # 5. Core contraction using a Triton kernel
    #    Compute out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) for easier indexing
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape

    # Allocate output tensor
    out_perm = torch.empty_like(left_perm)

    # Strides for the (B, H, N, N) layout
    stride_b, stride_h, stride_i, stride_k = left_perm.stride()
    stride_b = int(stride_b)
    stride_h = int(stride_h)
    stride_i = int(stride_i)
    stride_k = int(stride_k)

    # Tile sizes – tuned for typical GPUs
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32

    grid = (
        B * H,                               # one program per (b, h) pair
        triton.cdiv(N, BLOCK_I),            # i‑tiles
        triton.cdiv(N, BLOCK_J),            # j‑tiles
    )
    _trimul_outgoing_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        B, H, N,
        stride_b, stride_h, stride_i, stride_k,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # Restore shape (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise gating

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in float32 as required by the interface
    return out.to(torch.float32)
