def _gate_mask_fuse_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr,
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elements

    left = tl.load(left_ptr + offs, mask=m, other=0.0)
    right = tl.load(right_ptr + offs, mask=m, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=m, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=m, other=0.0)

    left = left * left_gate
    right = right * right_gate

    if HAS_MASK:
        mask_offs = offs // H
        mask_val = tl.load(mask_ptr + mask_offs, mask=m, other=0.0)
        left = left * mask_val
        right = right * mask_val

    tl.store(left_ptr + offs, left, mask=m)
    tl.store(right_ptr + offs, right, mask=m)
```

"""
TriMul outgoing kernel (AlphaFold3 style) with Triton‑fused mask+gate.

Algorithm
---------
1. Layer‑norm the input tensor (float32) → half precision.
2. Linear projections to the hidden dimension (no bias) in half.
3. Compute three sigmoid gates in half.
4. If a mask is present, a Triton kernel fuses:
      left  = left  * left_gate  * mask
      right = right * right_gate * mask
   otherwise simple element‑wise multiplication is performed.
5. Core O(N³) contraction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM (torch.bmm) over the hidden dimension.
6. Output layer‑norm, element‑wise scaling by out_gate, final linear projection
   back to the original dimension.
7. Cast the result back to float32.

Only the mask+gate fusion is done in Triton; the heavy matrix multiply uses
cuBLAS via torch.bmm, which is already highly optimized.
"""

import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # half* left  (flattened)
    right_ptr,           # half* right (flattened)
    left_gate_ptr,       # half* left_gate  (flattened)
    right_gate_ptr,      # half* right_gate (flattened)
    mask_ptr,            # half* mask (flattened, broadcast over hidden dim)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present else 0
    H: tl.constexpr,          # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Optional mask (broadcast over hidden dim)
    if HAS_MASK:
        # One mask value per (batch, i, k) group of H hidden entries
        mask_offs = offs // H
        mask_val = tl.load(mask_ptr + mask_offs, mask=mask, other=0.0)
        left = left * mask_val
        right = right * mask_val

    # Store the fused results back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of gating and optional masking using Triton.
    All tensors must be contiguous torch.float16.
    """
    total = left.numel()
    BLOCK = 1024                      # threads per program
    grid = (triton.cdiv(total, BLOCK),)
    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when mask absent

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # No return; tensors are modified in place


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/float) – may be None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half precision)
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16), bias=None)

    # ------------------------------------------------------------------
    # 3. Gates (half precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask + gates (Triton) or simple multiplication
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # Convert mask to half and flatten
        mask_h = mask.to(torch.float16).contiguous().view(-1)               # [B*N*N]
        # Flatten all tensors to 1‑D for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(left_f, right_f, left_gate_f, right_gate_f, mask_h, hidden_dim)

        # Reshape back to original layout
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask: just multiply by the gates
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out[...,i,j,d] = Σ_k left[...,i,k,d] * right[...,j,k,d]
    #    Implemented as a batched GEMM over the hidden dimension.
    # ------------------------------------------------------------------
    # Bring hidden dimension to the second axis for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)             # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)           # (B*H, N, N)

    # Batched matrix multiplication (left @ right^T)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # Restore layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    # Cast back to float32 for the final output
    return out.to(torch.float32)
