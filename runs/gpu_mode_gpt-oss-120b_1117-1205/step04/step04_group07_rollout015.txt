# Implementation of batched_gemm_kernel

@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute start indexes
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Range for K
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for boundary
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        # Load A: shape (BLOCK_M, BLOCK_K)
        a_ptrs = A_ptr + batch_id * stride_a_batch + offs_m[:, None] * stride_a_row + cur_k[None, :] * stride_a_col
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B: shape (BLOCK_K, BLOCK_N)
        b_ptrs = B_ptr + batch_id * stride_b_batch + cur_k[:, None] * stride_b_row + offs_n[None, :] * stride_b_col
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = C_ptr + batch_id * stride_c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass for AlphaFold3.

- LayerNorm + linear projections & gating are performed in PyTorch (float16).
- A small Triton kernel fuses mask, left‑gate and right‑gate elementwise.
- The O(N³) contraction is implemented as a batched GEMM using a custom
  Triton kernel (one program per (batch, i‑tile, j‑tile) block).
- Output LayerNorm, out‑gate scaling and final linear projection finish the
  forward pass. The result is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # * half
    right_ptr,         # * half
    left_gate_ptr,     # * half
    right_gate_ptr,    # * half
    mask_ptr,          # * half (broadcasted)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):
    """Fuse mask, left‑gate and right‑gate into the projected tensors."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    # Load data
    left = tl.load(left_ptr + offs, mask=active, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=active, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=tl.float16(0.0))

    if HAS_MASK:
        # each hidden entry shares the same mask value
        mask_idx = offs // HIDDEN_DIM
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=tl.float16(1.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def _apply_gate_mask_fuse(
    left: torch.Tensor,
    right: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    hidden_dim: int,
):
    """
    In‑place fusion of mask (if present) and gates using the Triton kernel above.
    All tensors must be contiguous 1‑D half‑precision views.
    """
    total = left.numel()
    BLOCK = 32768  # must be a multiple of 32
    grid = (triton.cdiv(total, BLOCK),)
    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when no mask

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        HIDDEN_DIM=hidden_dim,
    )


@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: C[b] = A[b] @ B[b]^T   with A,B shape (M,K) and (K,N).
    Batch dimension is the first program ID.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile (M x K)
        a_ptrs = A_ptr + batch_id * stride_a_batch \
                 + offs_m[:, None] * stride_a_row \
                 + cur_k[None, :] * stride_a_col
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=tl.float16(0.0))

        # Load B tile (K x N)
        b_ptrs = B_ptr + batch_id * stride_b_batch \
                 + cur_k[:, None] * stride_b_row \
                 + offs_n[None, :] * stride_b_col
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=tl.float16(0.0))

        # Accumulate
        acc += tl.dot(a, b)

    # Write output
    c_ptrs = C_ptr + batch_id * stride_c_batch \
             + offs_m[:, None] * stride_c_row \
             + offs_n[None, :] * stride_c_col
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, dim]  (float32)
            - mask_tensor:  [B, N, N]      (bool/float) – may be ignored if config["nomask"] is True
            - weights: dictionary with the required weight tensors
            - config: at least {"dim": int, "hidden_dim": int, "nomask": bool (optional)}

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float16 for speed)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half precision)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Compute gates (half precision + sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) with gating using Triton
    # ------------------------------------------------------------------
    if not nomask:
        # flatten tensors for the kernel
        mask_f = mask.to(torch.float16).contiguous().view(-1)           # [B*N*N]
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_mask_fuse(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            hidden_dim,
        )
        # restore shapes
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: batched GEMM (outgoing update)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Bring hidden_dim into the batch dimension for the GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B_h, H, _, _ = left_perm.shape
    batch = B_h * H

    left_mat = left_perm.view(batch, N, N)   # (batch, M=N, K=N)
    right_mat = right_perm.view(batch, N, N)  # (batch, K=N, N)

    out_mat = torch.empty_like(left_mat)

    # Tile sizes – chosen to fit well on typical B200 GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B_h, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in float32 as required
    return out.to(torch.float32)
