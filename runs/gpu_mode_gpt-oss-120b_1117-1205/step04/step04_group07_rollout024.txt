def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, N, K,
    stride_ah, stride_ai, stride_ak,
    stride_bh, stride_bk, stride_bn,
    stride_ch, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Compute mask for out-of-bound rows/cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A block (batch, row=m, col=k)
        a = tl.load(a_ptr + pid_batch * stride_ah + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak,
                    mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B block (batch, row=k, col=n) - B is transposed
        b = tl.load(b_ptr + pid_batch * stride_bh + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn,
                    mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Write back result
    c = acc.to(tl.float16)
    tl.store(c_ptr + pid_batch * stride_ch + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj,
             c, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass with a fused Triton kernel for the
O(N³) contraction.

The implementation follows the reference PyTorch module:
1. LayerNorm on the input.
2. Linear projections to a hidden dimension (half precision).
3. Sigmoid gates (half) and optional mask.
4. Batched matrix multiplication:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   performed by a custom Triton kernel that processes the
   (B*hidden_dim) independent N×N GEMMs.
5. Output LayerNorm, out‑gate scaling and final linear projection.
All heavy arithmetic uses float16 with float32 accumulation in the
kernel; the final result is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for many independent N×N matrix multiplies
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr,                # left matrix   [batch, N, K] (K == N)
    b_ptr,                # right (transposed) matrix [batch, K, N]
    c_ptr,                # output matrix [batch, N, N]
    batch: tl.int32,      # number of independent GEMMs  (B * hidden_dim)
    N: tl.int32,          # rows/cols of each square matrix
    K: tl.int32,          # inner dimension (== N)
    stride_ah, stride_ai, stride_ak,   # strides for a_ptr
    stride_bh, stride_bk, stride_bn,   # strides for b_ptr
    stride_ch, stride_ci, stride_cj,   # strides for c_ptr
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)          # which GEMM
    pid_m = tl.program_id(1)              # block row within the GEMM
    pid_n = tl.program_id(2)              # block col within the GEMM

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in FP32 for better precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a tile of A: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_batch * stride_ah
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load a tile of B (already transposed): shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_batch * stride_bh
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)   # fused multiply‑add

    # Write back the result in FP16
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_batch * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor:  torch.Tensor [B, N, N] (float32 or bool) or None
            - weights: dict with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict with at least "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias = weights["norm.bias"].to(torch.float16)

    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(torch.float16)  # work in half precision from here

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x, left_proj_w, bias=None)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w, bias=None)  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid) and optional mask
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x, left_gate_w, bias=None).sigmoid()
    right_gate = F.linear(x, right_gate_w, bias=None).sigmoid()
    out_gate = F.linear(x, out_gate_w, bias=None).sigmoid()  # used later

    if not nomask and mask is not None:
        # mask: [B, N, N] -> broadcast to hidden_dim
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Batched GEMM: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    # Move hidden dimension to leading axis for easy batching
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) = right transposed

    M = B * hidden_dim                        # total number of GEMMs
    left_mat = left_perm.view(M, N, N)        # (M, N, N)
    right_mat = right_perm.view(M, N, N)      # (M, N, N) ; already transposed
    out_mat = torch.empty_like(left_mat)      # (M, N, N)

    # Triton kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        M,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M,
        N,
        N,  # K = N
        left_mat.stride(0),
        left_mat.stride(1),
        left_mat.stride(2),
        right_mat.stride(0),
        right_mat.stride(1),
        right_mat.stride(2),
        out_mat.stride(0),
        out_mat.stride(1),
        out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, out_gate scaling and final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate  # out_gate already broadcasted

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w, bias=None)  # [B,N,N,dim] in half

    return out.to(torch.float32)
