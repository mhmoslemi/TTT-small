"""
Outgoing TriMul (AlphaFold3) forward pass – Triton‑accelerated core contraction.

Algorithm
---------
1. LayerNorm + half‑precision cast of input.
2. Linear projections to hidden dimension (left/right) and their sigmoid gates.
   Optional mask is fused by simple elementwise multiplication (no extra kernel).
3. Core O(N³) contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a tiled batched matrix multiplication in a Triton kernel.
   This avoids the costly permute+contiguous memory re‑layout of the reference.
4. Output LayerNorm over hidden dim, elementwise gating, final linear projection
   back to original dimension.
5. Return float32 tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    left_ptr,                                   # *fp16
    right_ptr,                                  # *fp16
    out_ptr,                                    # *fp16
    B, N, H,                                    # dimensions
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out = left @ rightᵀ for each (b, h) slice."""
    pid_i = tl.program_id(0)          # block row
    pid_j = tl.program_id(1)          # block col
    pid_bh = tl.program_id(2)         # batch * hidden

    batch_idx = pid_bh // H
    hidden_idx = pid_bh % H

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kk * BLOCK_K

        # ---- load left block (i,k) ----
        offs_i = i_start + tl.arange(0, BLOCK_M)[:, None]
        offs_k = k_start + tl.arange(0, BLOCK_K)[None, :]

        mask_i = offs_i < N
        mask_k = offs_k < N
        mask_left = mask_i & mask_k

        left_offset = (
            batch_idx * stride_left_b
            + hidden_idx * stride_left_h
            + offs_i * stride_left_i
            + offs_k * stride_left_k
        )
        left = tl.load(left_ptr + left_offset, mask=mask_left, other=tl.float16(0.0))

        # ---- load right block (k,j) ----
        offs_j = j_start + tl.arange(0, BLOCK_N)[None, :]

        mask_j = offs_j < N
        mask_right = mask_k[:, None] & mask_j

        right_offset = (
            batch_idx * stride_right_b
            + hidden_idx * stride_right_h
            + offs_k * stride_right_k
            + offs_j * stride_right_i
        )
        right = tl.load(right_ptr + right_offset, mask=mask_right, other=tl.float16(0.0))

        # ---- dot product accumulation ----
        acc += tl.dot(left, right, out_dtype=tl.float32)

    # ---- write out tile ----
    mask_i_out = (i_start + tl.arange(0, BLOCK_M)) < N
    mask_j_out = (j_start + tl.arange(0, BLOCK_N)) < N
    mask_out = mask_i_out[:, None] & mask_j_out[None, :]

    out_offset = (
        batch_idx * stride_out_b
        + hidden_idx * stride_out_h
        + i_start * stride_out_i
        + j_start * stride_out_j
    )
    tl.store(out_ptr + out_offset, acc.to(tl.float16), mask=mask_out)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of parameter tensors (float32)
            - config: dict with at least "dim" and "hidden_dim"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    inp, mask, weights, config = data
    device = inp.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> half)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"].to(torch.float16),
        bias=weights["norm.bias"].to(torch.float16),
    )
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections to hidden dim
    # -----------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
    )  # [B, N, N, H]

    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
    )  # [B, N, N, H]

    # -----------------------------------------------------------------
    # 3. Gate tensors (sigmoid)
    # -----------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
    ).sigmoid()  # [B, N, N, H]

    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
    ).sigmoid()  # [B, N, N, H]

    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
    ).sigmoid()  # [B, N, N, H]

    # Fuse gates (in‑place) -------------------------------------------
    left = left * left_gate
    right = right * right_gate
    del left_gate, right_gate  # free memory

    # Optional mask ----------------------------------------------------
    if (mask is not None) and (not nomask):
        # mask shape [B, N, N] -> broadcast over hidden dim
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # -----------------------------------------------------------------
    # 4. Core O(N³) contraction via Triton
    # -----------------------------------------------------------------
    B, N1, N2, H = left.shape
    assert N1 == N2 == N, "Sequence dimensions must be square."

    out = torch.empty_like(left)  # allocate output tensor (half)

    # Triton launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows
        triton.cdiv(N, BLOCK_N),          # cols
        B * H,                            # batch * hidden
    )

    # Strides (int64)
    stride_left_b = left.stride(0)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)
    stride_left_h = left.stride(3)

    stride_right_b = right.stride(0)
    stride_right_i = right.stride(1)   # j dimension uses the same stride as i
    stride_right_k = right.stride(2)
    stride_right_h = right.stride(3)

    stride_out_b = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_h = out.stride(3)

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_i,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # free proj tensors early
    del left, right

    # -----------------------------------------------------------------
    # 5. Output LayerNorm, gating, final linear
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
    )  # back to dim

    # Return float32 as required
    return out.to(torch.float32)
