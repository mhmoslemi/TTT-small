def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)
    device = x.device

    # LayerNorm
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x_norm.to(torch.float16)

    # Combined linear
    combined_weight = torch.cat([
        weights["left_proj.weight"],
        weights["left_gate.weight"],
        weights["right_proj.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).to(torch.float16)   # shape (5*hidden_dim, dim)

    # flatten x_h
    B, N, _, _ = x.shape
    x_flat = x_h.view(B * N * N, dim)
    proj = torch.nn.functional.linear(x_flat, combined_weight, bias=None)  # shape (B*N*N, 5*hidden_dim)
    proj = proj.view(B, N, N, 5 * hidden_dim)   # shape (B,N,N,5*H)

    # split
    left_proj, left_gate, right_proj, right_gate, out_gate = torch.split(proj, hidden_dim, dim=-1)
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    # mask & gate fuse
    if not nomask and mask is not None:
        # mask -> half
        mask_h = mask.to(torch.float16).contiguous()
        # collapse to flat view
        left_f = left_proj.contiguous()
        right_f = right_proj.contiguous()
        left_gate_f = left_gate.contiguous()
        right_gate_f = right_gate.contiguous()
        # apply mask/fuse kernel
        _apply_gate_and_mask(left_f, right_f, left_gate_f, right_gate_f, mask_h, hidden_dim)
        left = left_f
        right = right_f
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Core contraction
    # Permute + reshape
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B_, H_, N_, _ = left_perm.shape
    left_mat = left_perm.view(B_ * H_, N_, N_)
    right_mat = right_perm.view(B_ * H_, N_, N_)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # shape (B,N,N,H)

    # Output layernorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    return out.to(torch.float32)

"""
TriMul outgoing kernel (AlphaFold3 style) – fast forward pass.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32 → half).
2. Fuse all five linear projections (left_proj, left_gate, right_proj,
   right_gate, out_gate) into a single matmul with a stacked weight matrix.
   Split the result into the five tensors and apply sigmoid to the gates.
3. Fuse mask, left‑gate and right‑gate using a Triton kernel
   (in‑place on `left` and `right` tensors).
4. Core contraction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM: for each hidden channel d we compute
   L_d @ R_dᵀ using `torch.bmm`.  This requires a permutation and view
   (contiguous copy) because the hidden dimension must become the batch
   dimension.
5. Output layer‑norm, scale by `out_gate`, final linear projection back to
   the original channel dimension.
6. Cast the result back to float32.

Only the mask‑gate fusion is performed in Triton; the rest uses highly–
optimized PyTorch kernels (batched GEMM, LayerNorm, Linear).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # half* left tensor (flattened)
    right_ptr,           # half* right tensor (flattened)
    left_gate_ptr,       # half* left‑gate tensor (flattened)
    right_gate_ptr,      # half* right‑gate tensor (flattened)
    mask_ptr,            # half* mask tensor (flattened, broadcast over hidden)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present, else 0
    H: tl.constexpr,          # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elements

    # load tensors
    left_val = tl.load(left_ptr + offs, mask=m, other=0.0)
    right_val = tl.load(right_ptr + offs, mask=m, other=0.0)
    left_gate_val = tl.load(left_gate_ptr + offs, mask=m, other=0.0)
    right_gate_val = tl.load(right_gate_ptr + offs, mask=m, other=0.0)

    if HAS_MASK:
        # one mask value per (b,i,k) – repeated over hidden dim
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=m, other=0.0)
        left_val = left_val * left_gate_val * mask_val
        right_val = right_val * right_gate_val * mask_val
    else:
        left_val = left_val * left_gate_val
        right_val = right_val * right_gate_val

    tl.store(left_ptr + offs, left_val, mask=m)
    tl.store(right_ptr + offs, right_val, mask=m)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of gating and (optional) masking using Triton.
    All tensors must be contiguous half‑precision.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # No return – tensors are modified in place


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be None if config["nomask"]=True
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Combined linear projection for left/right values and gates
    # --------------------------------------------------------------
    # Stack the five weight matrices: left_proj, left_gate, right_proj,
    # right_gate, out_gate   → shape (5*hidden_dim, dim)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # (5*H, dim)

    # Flatten input to (B*N*N, dim) and apply one big linear
    x_flat = x_h.view(B * N * N, dim)
    proj = F.linear(x_flat, combined_weight, bias=None)  # (B*N*N, 5*H)
    proj = proj.view(B, N, N, 5 * hidden_dim)  # (B, N, N, 5*H)

    # Split the combined projection
    left_proj, left_gate, right_proj, right_gate, out_gate = torch.split(
        proj, hidden_dim, dim=-1
    )
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    # --------------------------------------------------------------
    # 3. Fuse mask (if any) and gates into left/right tensors
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).contiguous().view(-1)  # [B*N*N]
        left_f = left_proj.contiguous()
        right_f = right_proj.contiguous()
        left_gate_f = left_gate.contiguous()
        right_gate_f = right_gate.contiguous()
        _apply_gate_and_mask(
            left_f, right_f, left_gate_f, right_gate_f, mask_h, hidden_dim
        )
        left = left_f
        right = right_f
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 4. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as batched GEMM (left @ right^T) per hidden channel.
    # --------------------------------------------------------------
    # Bring hidden dimension into the batch dimension for torch.bmm.
    # Shape after permute: (B, H, N, N) → then view as (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B_, H_, N_, _ = left_perm.shape
    left_mat = left_perm.view(B_ * H_, N_, N_)          # (B*H, N, N)
    right_mat = right_perm.view(B_ * H_, N_, N_)        # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore layout: (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5. Output LayerNorm, gating, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in float32 as required
    return out.to(torch.float32)
