def custom_kernel(data):
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm (float32) then cast to half
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x = x.to(torch.float16)

    # Projections
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16), bias=None)

    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(torch.float16), bias=None))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(torch.float16), bias=None))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(torch.float16), bias=None))

    # Apply mask and gates
    if not nomask:
        # mask broadcast
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_h * left_gate
        right = right * mask_h * right_gate
    else:
        left = left * left_gate
        right = right * right_gate

    # Ensure contiguity
    left = left.contiguous()
    right = right.contiguous()

    # Output tensor
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # Strides
    stride_b, stride_i, stride_k, stride_h = left.stride()  # for left; they are same for right

    # Kernel launch
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 16
    BLOCK_H = 8
    num_hblocks = (hidden_dim + BLOCK_H - 1) // BLOCK_H
    grid = (B * num_hblocks, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_b,
        stride_i,
        stride_k,
        stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # After contraction: out shape [B,N,N,H]
    # Output layernorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # broadcast (same shape)
    # Final linear projection to dim
    out = F.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    # Cast to float32
    return out.to(torch.float32)

    right = right.contiguous()

"""
TriMul outgoing kernel (AlphaFold3 style).

Algorithm
---------
1. Layer‑norm the input and cast to float16.
2. Linear projections → left/right tensors of shape [B, N, N, H].
3. Compute three gating vectors (left_gate, right_gate, out_gate) with sigmoid.
4. Apply mask (if present) and element‑wise gating to left/right.
5. Core contraction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a custom Triton kernel that performs a batched
   matrix‑multiply for each hidden‑block (H is split into small tiles).
   The kernel loads a (M×K×HB) block of left and a (K×N×HB) block of right,
   accumulates the product for each hidden slice using tl.dot,
   and writes the result to the output tensor.
6. Layer‑norm over the hidden dimension, scale by out_gate,
   and a final linear projection back to the original channel size.
7. Return the result in float32.

The kernel fuses the N³ contraction across all hidden channels,
reducing memory traffic compared to a naïve torch.bmm loop.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,          # half*  (B, N, N, H)
    right_ptr,         # half*  (B, N, N, H)
    out_ptr,           # half*  (B, N, N, H)
    B, N, H,
    stride_b, stride_i, stride_k, stride_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid0 = tl.program_id(0)   # batch * hidden‑block
    pid1 = tl.program_id(1)   # i‑block
    pid2 = tl.program_id(2)   # j‑block

    # decode batch and hidden‑block indices
    num_hblocks = tl.cdiv(H, BLOCK_H)
    b = pid0 // num_hblocks
    hb = pid0 % num_hblocks

    # block origins
    i0 = pid1 * BLOCK_M
    j0 = pid2 * BLOCK_N
    h0 = hb * BLOCK_H

    # index vectors
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_h = h0 + tl.arange(0, BLOCK_H)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # accumulator: (HB, M, N) in fp32
    acc = tl.zeros((BLOCK_H, BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k_offset in range(0, N, BLOCK_K):
        cur_k = tl.minimum(BLOCK_K, N - k_offset)
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----------- left block (M, K, HB) -------------
        left_ptrs = (
            left_ptr
            + b * stride_b
            + offs_i[:, None, None] * stride_i
            + offs_k[None, :, None] * stride_k
            + offs_h[None, None, :] * stride_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # ----------- right block (K, N, HB) ------------
        right_ptrs = (
            right_ptr
            + b * stride_b
            + offs_k[:, None, None] * stride_k
            + offs_j[None, :, None] * stride_i
            + offs_h[None, None, :] * stride_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # ----------- inner product for each hidden slice ----------
        for hh in range(0, BLOCK_H):
            a = left[:, :, hh].to(tl.float32)    # (M, K)
            b_mat = right[:, :, hh].to(tl.float32)  # (K, N)
            acc[hh] += tl.dot(a, b_mat)          # (M, N)

    # ----------- write back ------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_b
        + offs_i[:, None, None] * stride_i
        + offs_j[None, :, None] * stride_k
        + offs_h[None, None, :] * stride_h
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
    )


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/float) – optional
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections to hidden dimension (half)
    # -----------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # -----------------------------------------------------------------
    # 3. Gates (half)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(
            x,
            weights["left_gate.weight"].to(torch.float16),
            bias=None,
        )
    )
    right_gate = torch.sigmoid(
        F.linear(
            x,
            weights["right_gate.weight"].to(torch.float16),
            bias=None,
        )
    )
    out_gate = torch.sigmoid(
        F.linear(
            x,
            weights["out_gate.weight"].to(torch.float16),
            bias=None,
        )
    )

    # -----------------------------------------------------------------
    # 4. Apply mask and gates
    # -----------------------------------------------------------------
    if not nomask:
        # mask broadcast over hidden dimension
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_h * left_gate
        right = right * mask_h * right_gate
    else:
        left = left * left_gate
        right = right * right_gate

    # Ensure contiguous layout for the kernel
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, _ = left.shape

    # -----------------------------------------------------------------
    # 5. Core contraction (TriMul) via Triton
    # -----------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # Strides of left/right tensors (identical)
    stride_b, stride_i, stride_k, stride_h = left.stride()  # (B,N,N,H)

    # Tile sizes – chosen to fit in registers while keeping good occupancy
    BLOCK_M = 32   # i‑dimension tile
    BLOCK_N = 32   # j‑dimension tile
    BLOCK_K = 16   # reduction (k) tile
    BLOCK_H = 4    # hidden‑channel tile

    num_hblocks = (hidden_dim + BLOCK_H - 1) // BLOCK_H
    grid = (
        B * num_hblocks,                     # batch × hidden‑block
        triton.cdiv(N, BLOCK_M),             # i‑tiles
        triton.cdiv(N, BLOCK_N),             # j‑tiles
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_b,
        stride_i,
        stride_k,
        stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gate scaling, and final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in the original dtype
    return out.to(torch.float32)
```
