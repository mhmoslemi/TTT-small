"""
Outgoing TriMul (AlphaFold3) forward pass with Triton mask/gate fusion.

Algorithm:
1. LayerNorm over the last dimension (dim) → half precision.
2. Linear projections to hidden_dim (left, right).
3. Compute three gate tensors (left_gate, right_gate, out_gate) via linear + sigmoid.
4. Fuse mask (if present) and left/right gates in a Triton kernel (in‑place).
5. Core contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM per hidden slice (torch.bmm) using half.
6. Output LayerNorm over hidden_dim, element‑wise out_gate gating, final linear
   projection back to dim.
7. Return float32 tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # *fp16
    right_ptr,         # *fp16
    left_gate_ptr,     # *fp16
    right_gate_ptr,    # *fp16
    mask_ptr,          # *fp16 (broadcasted over hidden dim)
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,   # hidden dimension
):
    """
    In‑place: left = left * left_gate * mask
            right = right * right_gate * mask
    `total_elems` = B * N * N * H
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # Load vectors
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    lgate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # One mask value per (b,i,j) → total_elems // H
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * lgate * mask_val
        right = right * rgate * mask_val
    else:
        left = left * lgate
        right = right * rgate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_mask_and_gate(left, right, left_gate, right_gate, mask, hidden_dim):
    """Launch Triton kernel that fuses mask + left/right gates in‑place."""
    total = left.numel()
    BLOCK = 32768  # large block size, fits into a single thread block
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when mask absent

    _mask_gate_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # In‑place modification; nothing is returned


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of parameter tensors (float32)
            - config: dict with at least "dim" and "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    inp, mask, weights, config = data
    device = inp.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 → half)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections to hidden_dim (no bias)
    # -----------------------------------------------------------------
    left = F.linear(
        x, weights["left_proj.weight"].to(torch.float16)
    )  # [B, N, N, H]
    right = F.linear(
        x, weights["right_proj.weight"].to(torch.float16)
    )

    # -----------------------------------------------------------------
    # 3. Gate tensors (sigmoid)
    # -----------------------------------------------------------------
    left_gate = F.linear(
        x, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = F.linear(
        x, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = F.linear(
        x, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    # -----------------------------------------------------------------
    # 4. Fuse mask + gates (if needed) via Triton
    # -----------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → half, broadcast over hidden dim
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]

        # Flatten tensors for kernel
        left_f = left.view(-1)
        right_f = right.view(-1)
        left_gate_f = left_gate.view(-1)
        right_gate_f = right_gate.view(-1)
        mask_f = mask_h.view(-1)  # length B*N*N

        _apply_mask_and_gate(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            hidden_dim,
        )

        # Reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: batched GEMM over hidden dimension
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape  # left shape = (B, N, N, H)

    # Rearrange to (B*H, N, N) for torch.bmm
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Matrix multiplication: (i,k) * (j,k)^T -> (i,j)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return as float32 as required
    return out.to(torch.float32)

"""
Outgoing TriMul (AlphaFold3) forward pass with Triton mask/gate fusion.

Algorithm
---------
1. LayerNorm over the last dimension (dim) → half precision.
2. Linear projections to hidden_dim (left, right).
3. Compute three gate tensors (left_gate, right_gate, out_gate) via linear + sigmoid.
4. Fuse mask (if present) and left/right gates in a Triton kernel (in‑place).
5. Core contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM per hidden slice (torch.bmm) using half.
6. Output LayerNorm over hidden_dim, element‑wise out_gate gating,
   final linear projection back to dim.
7. Return a float32 tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # *fp16
    right_ptr,         # *fp16
    left_gate_ptr,     # *fp16
    right_gate_ptr,    # *fp16
    mask_ptr,          # *fp16 (broadcasted over hidden dim)
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,   # hidden dimension
):
    """
    In‑place:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    `total_elems` = B * N * N * H.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elems

    left = tl.load(left_ptr + offs, mask=m, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=m, other=tl.float16(0.0))
    lgate = tl.load(left_gate_ptr + offs, mask=m, other=tl.float16(0.0))
    rgate = tl.load(right_gate_ptr + offs, mask=m, other=tl.float16(0.0))

    if HAS_MASK:
        # One mask value per (b,i,j) => total_elems // H entries
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=m, other=tl.float16(0.0))
        left = left * lgate * mask_val
        right = right * rgate * mask_val
    else:
        left = left * lgate
        right = right * rgate

    tl.store(left_ptr + offs, left, mask=m)
    tl.store(right_ptr + offs, right, mask=m)


def _apply_mask_and_gate(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    Launch the Triton kernel that fuses mask + left/right gates in‑place.
    All tensors are 1‑D views of the original data.
    """
    total = left.numel()
    BLOCK = 32768  # large block size, fits in a single thread block
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when mask absent

    _mask_gate_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # In‑place modification; nothing to return


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (float32)
            - config: dict with at least "dim" and "hidden_dim",
                      optionally "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 → half)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections to hidden dimension (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # -----------------------------------------------------------------
    # 3. Gate tensors (sigmoid)
    # -----------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # -----------------------------------------------------------------
    # 4. Fuse mask + gates (if needed) via Triton
    # -----------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → half, broadcast over hidden dim
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]

        # Flatten tensors for the kernel
        left_f = left.view(-1)
        right_f = right.view(-1)
        left_gate_f = left_gate.view(-1)
        right_gate_f = right_gate.view(-1)
        mask_f = mask_h.view(-1)  # length B*N*N

        _apply_mask_and_gate(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            hidden_dim,
        )

        # Reshape back to original layout
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction via batched GEMM over hidden dimension
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape  # left shape = (B, N, N, H)

    # Rearrange to (B*H, N, N) for torch.bmm
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Matrix multiplication: (i,k) @ (j,k)^T → (i,j)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return as float32 as required
    return out.to(torch.float32)
