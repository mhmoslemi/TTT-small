@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,   # *fp16
    right_ptr,
    left_gate_ptr,
    right_gate_ptr,
    mask_ptr,
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    # Load values
    left = tl.load(left_ptr + offs, mask=active, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=active, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=tl.float16(0.0))

        right = right * right_gate

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Input LayerNorm in fp32, then cast to fp16
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # Linear projections
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16), bias=None)

    # Gate values
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    if not nomask:
        # Flatten tensors for kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)

        _apply_gate_mask_fuse(left_f, right_f, left_gate_f, right_gate_f, mask_f, hidden_dim)

        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # Release some memory
    del left_gate, right_gate, x_h

    # Contraction via batched matmul (half precision)
    # Permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Perform batched matrix multiplication
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

"""
TriMul (outgoing) forward pass.

The implementation follows the reference PyTorch module but moves the
element‑wise mask + gating fusion into a custom Triton kernel.  All
linear projections, LayerNorms and the heavy N³ contraction are performed
in half‑precision (fp16) using PyTorch (cuBLAS for the batched GEMM).  The
final output is cast back to fp32.

Key steps:
1. Input LayerNorm (fp32 → fp16)
2. Linear projections to hidden_dim
3. Compute sigmoid gates
4. Fuse mask (if present) and gates with a Triton kernel (in‑place)
5. Batched matrix multiplication: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
6. Output LayerNorm, out‑gate scaling and final linear projection
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # * fp16
    right_ptr,         # * fp16
    left_gate_ptr,     # * fp16
    right_gate_ptr,    # * fp16
    mask_ptr,          # * fp16 (broadcasted)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    # Load values
    left = tl.load(left_ptr + offs, mask=active, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=active, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=tl.float16(0.0))

    if HAS_MASK:
        # Each mask element corresponds to HIDDEN_DIM entries in left/right
        mask_idx = offs // HIDDEN_DIM
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=tl.float16(1.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)

        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)  # [B, N, N, dim]

    # -------------------- 2. Linear projections (to hidden_dim) --------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16), bias=None)

    # -------------------- 3. Compute sigmoid gates --------------------
    left_gate = F.linear(
        x_h, weights["left_gate.weight"].to(torch.float16), bias=None
    ).sigmoid()
    right_gate = F.linear(
        x_h, weights["right_gate.weight"].to(torch.float16), bias=None
    ).sigmoid()
    out_gate = F.linear(
        x_h, weights["out_gate.weight"].to(torch.float16), bias=None
    ).sigmoid()

    # -------------------- 4. Fuse mask + gates (Triton) --------------------
    if not nomask:
        # Flatten tensors for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask_tensor.to(torch.float16).contiguous().view(-1)  # [B*N*N]

        _apply_gate_mask_fuse(
            left_f, right_f, left_gate_f, right_gate_f, mask_f, hidden_dim
        )

        # Reshape back to original layout
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # Simple element‑wise multiplication when no mask is present
        left = left * left_gate
        right = right * right_gate

    # Release intermediate tensors we no longer need
    del left_gate, right_gate, x_h

    # -------------------- 5. Core contraction (batched GEMM) --------------------
    # Shape to (B, H, N, N) for efficient batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)      # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)    # (B*H, N, N)

    # out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # -------------------- 6. Output LayerNorm, out‑gate scaling, final linear --------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out, weights["to_out.weight"].to(torch.float16), bias=None
    )  # (B, N, N, dim)

    # Return in float32 as required by the interface
    return out.to(torch.float32)
```
