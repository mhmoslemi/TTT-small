@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program_id: batch_idx*x + block_i + block_j dims maybe multiple axes grid
    pid_batch = tl.program_id(2)  # third axis
    pid_m = tl.program_id(0)    # block row
    pid_n = tl.program_id(1)    # block col

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # compute start pointers for A and B for this batch
    a_batch_ptr = a_ptr + pid_batch * stride_ab
    b_batch_ptr = b_ptr + pid_batch * stride_bb
    c_batch_ptr = c_ptr + pid_batch * stride_cb

    # loops across K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # load A tile
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < K), other=0.0)
        # load B tile
        b_ptrs = b_batch_ptr + ((k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=(k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N), other=0.0)
        # accumulate
        acc += tl.dot(a, b)

        # advance pointer? Not needed as we recompute each iteration

    # write back
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    # mask for u bound
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)

"""
TriMul outgoing implementation (AlphaFold3).

The kernel performs the heavy O(N³·H) contraction:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
by a batched GEMM (A @ Bᵀ) using Triton.  The surrounding
operations (LayerNorm, linear projections, sigmoid gates,
optional mask+gate fusion, output LayerNorm and final projection)
are executed in PyTorch (float16 where possible).  The
gate+mask fusion and the batched GEMM are both written in Triton,
providing a large speed‑up over the reference PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask (if any) with left/right gating.
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # half* flattened left tensor
    right_ptr,           # half* flattened right tensor
    left_gate_ptr,       # half* flattened left‑gate tensor
    right_gate_ptr,      # half* flattened right‑gate tensor
    mask_ptr,            # half* flattened mask (broadcast over hidden dim)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present, else 0
    H: tl.constexpr,          # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elements

    left = tl.load(left_ptr + offs, mask=m, other=0.0)
    right = tl.load(right_ptr + offs, mask=m, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=m, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=m, other=0.0)

    if HAS_MASK:
        # mask is per (batch,i,k) → one value per group of H hidden values
        mask_off = offs // H
        mask_val = tl.load(mask_ptr + mask_off, mask=m, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=m)
    tl.store(right_ptr + offs, right, mask=m)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """In‑place fusion of gating and optional mask using Triton."""
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when no mask

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # tensors are modified in place


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ where A,B ∈ ℝ^{M×K},
# batch size = B*H (B = batch, H = hidden_dim), M=N=K=seq_len.
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_b = tl.program_id(2)   # batch*hidden index

    # base pointers for this batch
    a_batch = a_ptr + pid_b * stride_ab
    b_batch = b_ptr + pid_b * stride_bb
    c_batch = c_ptr + pid_b * stride_cb

    # offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        cur_k = tl.minimum(BLOCK_K, K - k_offset)

        # A tile: (M_block, K_block)
        a_ptrs = a_batch + (offs_m[:, None] * stride_am
                            + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & (tl.arange(0, BLOCK_K)[None, :] < cur_k)),
                    other=0.0)

        # B tile (treated as Bᵀ): (K_block, N_block)
        b_ptrs = b_batch + ((k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk
                            + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=((tl.arange(0, BLOCK_K)[:, None] < cur_k) & mask_n[None, :]),
                    other=0.0)

        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (float32 or bool); may be None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32) → FP16 for subsequent ops
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections (left/right) to hidden dimension (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16), bias=None)

    # --------------------------------------------------------------
    # 3. Gates (sigmoid after linear)
    # --------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # --------------------------------------------------------------
    # 4. Fuse mask (if present) with gates using Triton
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # Ensure contiguous half‑precision tensors
        mask_h = mask.to(torch.float16).contiguous().view(-1)               # [B*N*N]
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(left_f, right_f, left_gate_f, right_gate_f,
                             mask_h, hidden_dim)

        # reshape back to original layout
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask: simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core O(N³·H) contraction via batched GEMM (Triton)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # Rearrange to (batch, hidden, N, N) and flatten batch*hidden
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N_seq, _ = left_perm.shape
    BH = B * H

    left_mat = left_perm.view(BH, N_seq, N_seq)   # (BH, N, N)   A matrix
    right_mat = right_perm.view(BH, N_seq, N_seq) # (BH, N, N)   B matrix

    # output buffer
    out_mat = torch.empty_like(left_mat)

    # strides (all tensors are contiguous)
    stride_a_batch, stride_am, stride_ak = left_mat.stride()
    stride_b_batch, stride_bk_raw, stride_bn_raw = right_mat.stride()
    # For Bᵀ we need stride over K (original column) and N (original row)
    stride_bk = stride_ak          # = 1
    stride_bn = stride_am          # = N_seq
    stride_c_batch, stride_cm, stride_cn = out_mat.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N_seq, BLOCK_M),
        triton.cdiv(N_seq, BLOCK_N),
        BH,
    )
    _tri_mul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N_seq, N_seq, N_seq,
        stride_am, stride_ak, stride_a_batch,
        stride_bk, stride_bn, stride_b_batch,
        stride_cm, stride_cn, stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N_seq, N_seq).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    # Cast back to FP32 for the final output
    return out.to(torch.float32)
