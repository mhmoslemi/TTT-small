"""
TriMul (outgoing) forward pass with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor (float32 → half).
2. Linear projections to the hidden dimension `H` (half precision).
3. Compute three gating tensors (half) with a sigmoid non‑linearity.
4. Apply optional mask and gates element‑wise.
5. Core contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (left @ rightᵀ) using a custom Triton
   kernel that works on the (B·H, N, N) view of the tensors.
6. Output layer‑norm, element‑wise scaling by `out_gate`,
   and final linear projection back to the original dimension.
7. Return the result in float32.

At least one compute‑intensive part (the N³ contraction) is executed
entirely on the GPU with Triton, satisfying the requirement for a custom
kernel while keeping the rest of the pipeline in highly‑optimized PyTorch
operations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_trmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to the three tensors
    N,                                 # sequence length
    stride_ah, stride_ai, stride_ak,   # strides for a (batch_head, i, k)
    stride_bh, stride_bi, stride_bk,   # strides for b (batch_head, j, k) – we read transposed
    stride_ch, stride_ci, stride_ck,   # strides for c (batch_head, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication C = A @ Bᵀ where
        A : (batch_head, N, N)   (i,k)
        B : (batch_head, N, N)   (j,k)
        C : (batch_head, N, N)   (i,j)

    The kernel iterates over tiles of K and accumulates in FP32.
    """
    pid = tl.program_id(0)

    # number of tiles in the i and j dimensions
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # decode the linear program id into (batch_head, block_i, block_j)
    batch_head = pid // (num_blocks_m * num_blocks_n)
    block_id = pid % (num_blocks_m * num_blocks_n)
    block_i = block_id // num_blocks_n
    block_j = block_id % num_blocks_n

    # tile offsets
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # out‑of‑range masks for the i‑ and j‑dimensions
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load A tile (i,k) -----
        a_ptrs = (
            a_ptr
            + batch_head * stride_ah
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # ----- load B tile transposed (k,j) -----
        # B is stored as (j,k); we need (k,j) for the dot product
        b_ptrs = (
            b_ptr
            + batch_head * stride_bh
            + offs_k[:, None] * stride_bk          # k  becomes row
            + offs_j[None, :] * stride_bi          # j  becomes column
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # accumulate
        acc += tl.dot(a, b)

    # ----- write back C tile (i,j) -----
    c_ptrs = (
        c_ptr
        + batch_head * stride_ch
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_ck
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple (input_tensor, mask, weights_dict, config_dict)
        input_tensor : torch.Tensor [B, N, N, C] (float32)
        mask         : torch.Tensor [B, N, N] (bool or numeric) or None
        weights_dict : dict of model parameters (float32)
        config_dict  : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gates (half) and sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Apply mask (if any) and element‑wise gating
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * left_gate * mask_h
        right = right * right_gate * mask_h
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton (batched GEMM)
    #    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    batch_head = B * hidden_dim

    # reshape to (batch_head, N, N) for the kernel
    left_perm = left.permute(0, 3, 1, 2).contiguous().view(batch_head, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous().view(batch_head, N, N)
    out_perm = torch.empty_like(left_perm)

    # strides (in elements) for the contiguous view (batch_head, N, N)
    stride_ah = left_perm.stride(0)
    stride_ai = left_perm.stride(1)
    stride_ak = left_perm.stride(2)

    stride_bh = right_perm.stride(0)
    stride_bi = right_perm.stride(1)   # row (j) stride
    stride_bk = right_perm.stride(2)   # column (k) stride

    stride_ch = out_perm.stride(0)
    stride_ci = out_perm.stride(1)
    stride_ck = out_perm.stride(2)

    # kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        triton.cdiv(batch_head, 1)
        * triton.cdiv(N, BLOCK_M)
        * triton.cdiv(N, BLOCK_N),
    )

    _batched_trmul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        N,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bi, stride_bk,
        stride_ch, stride_ci, stride_ck,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # ensure kernel completion before next ops
    torch.cuda.synchronize()

    # reshape back to original layout: (B, N, N, H)
    out = out_perm.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 7. Cast back to float32 and return
    # ------------------------------------------------------------------
    return out.to(torch.float32)
