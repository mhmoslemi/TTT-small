"""
TriMul (outgoing) forward pass optimized with:
- Single fused linear projection for left/right projections and gates.
- Triton kernel that fuses gating, optional mask, and element‑wise multiplication.
- Batched half‑precision GEMM for the N³ contraction (torch.bmm, which uses cuBLAS).
All heavy tensors are kept in float16; final result is cast back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half* (flattened left tensor)
    right_ptr,         # half* (flattened right tensor)
    left_gate_ptr,     # half* (flattened left gate)
    right_gate_ptr,    # half* (flattened right gate)
    mask_ptr,          # half* (flattened mask, broadcasted over hidden dim)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present, else 0
    H: tl.constexpr,          # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elements

    left = tl.load(left_ptr + offs, mask=m, other=0.0)
    right = tl.load(right_ptr + offs, mask=m, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=m, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=m, other=0.0)

    if HAS_MASK:
        # mask is per (batch, i, j) entry, i.e. per element of size M = B*N*N.
        # each hidden channel shares the same mask entry.
        mask_idx = offs // H            # integer division, broadcast across hidden dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=m, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=m)
    tl.store(right_ptr + offs, right, mask=m)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place multiplication:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    `mask` may be None → no mask is applied.
    All tensors must be contiguous half‑precision.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when mask absent

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # operation is in‑place; nothing returned

            - config: dict with keys "dim", "hidden_dim", optional "nomask"

        torch.Tensor [B, N, N, dim] (float32)
    """

    device = input_tensor.device

    # config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. Cast weights to half‑precision once
    # --------------------------------------------------------------

    w_norm_b = weights["norm.bias"].to(torch.float16, device=device)

    w_right_proj = weights["right_proj.weight"].to(torch.float16, device=device)

    w_out_norm_b = weights["to_out_norm.bias"].to(torch.float16, device=device)
    w_out = weights["to_out.weight"].to(torch.float16, device=device)

    # 2. Input LayerNorm (half‑precision)
    # --------------------------------------------------------------
    x_h = input_tensor.to(torch.float16, device=device)
    x_norm = F.layer_norm(x_h, (dim,), weight=w_norm_w, bias=w_norm_b)

    # 3. Combined linear projection for left/right and gates
    # --------------------------------------------------------------
    B, N, _, _ = x_norm.shape

    x_flat = x_norm.view(M, dim)                  # (M, dim)

    right = proj[:, h : 2 * h]
    left_gate = proj[:, 2 * h : 3 * h]
    right_gate = proj[:, 3 * h : 4 * h]
    out_gate = proj[:, 4 * h : 5 * h]

    # --------------------------------------------------------------
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    # --------------------------------------------------------------
    # 5. Fuse mask + gating (Triton) or simple elementwise multiply
    # --------------------------------------------------------------
    if not nomask:

        # flatten everything for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

            right_f,
            left_gate_f,

            mask_h,

        right = right_f.view(M, h)

    # --------------------------------------------------------------
    left = left.view(B, N, N, h)

    out_gate = out_gate.view(B, N, N, h)

    #    This is a batched GEMM: for each (b,d) compute

    # --------------------------------------------------------------
    # Reorder to (B, h, N, N) for batched matmul
    left_perm = left.permute(0, 3, 1, 2)   # (B, h, N, N)

    B_h, H_h, N_h, _ = left_perm.shape
    left_mat = left_perm.reshape(B_h * H_h, N_h, N_h)   # (B*h, N, N)
    right_mat = right_perm.reshape(B_h * H_h, N_h, N_h) # (B*h, N, N)

    # Batched matmul (half‑precision, cuBLAS)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*h, N, N)

    out = out_mat.view(B, h, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, h)

    # --------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate scaling and final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w_out_norm_w,
        bias=w_out_norm_b,
    )
    out = out * out_gate  # element‑wise scaling (half)

    out = F.linear(out, w_out, bias=None)  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 9. Cast back to float32 for downstream use
    # --------------------------------------------------------------
    return out.to(torch.float32)
