"""
TriMul forward (outgoing) – optimized implementation.

The heavy computation consists of:
1. LayerNorm and linear projections (FP16) – performed with PyTorch.
2. Gating + optional mask fusion – performed with a small Triton kernel.
3. Core O(N³) contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is equivalent to a batched matrix multiplication
   left_h @ right_h^T for each hidden channel.  Implemented via
   torch.bmm on the (B*hidden_dim)-batched view.
4. Output LayerNorm, out‑gate scaling and final linear projection
   (FP16), returning FP32.

The Triton kernel fuses element‑wise mask (if any) with the left/right
gates, reducing memory traffic and latency.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse gating and optional mask
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # float16*
    right_ptr,         # float16*
    left_gate_ptr,     # float16*
    right_gate_ptr,    # float16*
    mask_ptr,          # float16* (optional)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,  # 1 if mask is present, else 0
    H: tl.constexpr,         # hidden_dim (size of innermost channel)
):
    """In‑place: left = left * left_gate * mask; right = right * right_gate * mask."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # mask broadcast across hidden dimension:
        # each hidden entry shares the same mask entry.
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    Fuse left/right with their gates and optional mask in‑place using Triton.
    All inputs must be contiguous, half‑precision tensors.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when HAS_MASK==0

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # operation performed in‑place; nothing to return.


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) – may be ignored
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half)
    # --------------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # --------------------------------------------------------------
    # 3. Gates (half) with sigmoid
    # --------------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # --------------------------------------------------------------
    # 4. Fuse mask (if any) with gates via Triton kernel
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → half, flattened
        mask_h = mask.to(torch.float16).contiguous().view(-1)  # [B*N*N]

        # Flatten tensors for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_h,
            hidden_dim,
        )
        # Reshape back (views share storage)
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask: simple element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate

    # Gates have been fused; free temporary tensors
    del left_gate, right_gate, x_h

    # --------------------------------------------------------------
    # 5. Core contraction: batched GEMM over hidden dimension
    #    out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    # --------------------------------------------------------------
    # Rearrange to (B, H, N, N) for efficient batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # Batched matrix multiplication: left @ right^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise gating (half)

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in float32 as required by the interface
    return out.to(torch.float32)
