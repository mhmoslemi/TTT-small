@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr,
    total_elements: tl.int32,
    hidden_dim: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left_val = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_val = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate_val = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate_val = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # compute mask offset: each group of hidden_dim values share a mask value.
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        # apply
        left_val = left_val * left_gate_val * mask_val
        right_val = right_val * right_gate_val * mask_val
    else:
        left_val = left_val * left_gate_val
        right_val = right_val * right_gate_val

    tl.store(left_ptr + offs, left_val, mask=mask)
    tl.store(right_ptr + offs, right_val, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    # layernorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # cast to half
    x_h = x.half()
    # linear projections
    left = torch.nn.functional.linear(x_h, weights["left_proj.weight"].half(), bias=None)
    right = torch.nn.functional.linear(x_h, weights["right_proj.weight"].half(), bias=None)

    # gates
    left_gate = torch.nn.functional.linear(x_h, weights["left_gate.weight"].half(), bias=None).sigmoid()
    right_gate = torch.nn.functional.linear(x_h, weights["right_gate.weight"].half(), bias=None).sigmoid()
    out_gate = torch.nn.functional.linear(x_h, weights["out_gate.weight"].half(), bias=None).sigmoid()

    # Fuse mask and gate:
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).contiguous()   # shape [B,N,N]
        # flatten mask to 1D for kernel
        # left,right,gates need to be flattened
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        # kernel
        _apply_gate_and_mask(left, right, left_gate, right_gate, mask_h, hidden_dim)
    else:
        left = left * left_gate
        right = right * right_gate

    # Core contraction
    # Permute
    left_perm = left.permute(0, 3, 1, 2).contiguous()  # B, hidden, N, N
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output layernorm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half()
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"].half(), bias=None)

    # Cast to float32
    return out.float()

"""
TriMul outgoing operator with Triton fused gating+mask.

The kernel fuses the element‑wise mask (if present) and the sigmoid gates
into the left/right projected tensors.  All heavy arithmetic is done in
float16; the expensive O(N³) contraction is performed with a batched
matrix multiplication (cuBLAS) on half‑precision tensors.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr,
    total_elements, hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """Fuse mask (optional) and sigmoid gates into left/right tensors."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load values
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # Each hidden_dim values share the same mask entry.
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store the fused results back in‑place.
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fuse mask (if present) and sigmoid gates via Triton.
    All tensors must be contiguous torch.float16.
    """
    total = left.numel()
    BLOCK = 1024  # configurable block size
    grid = (triton.cdiv(total, BLOCK),)
    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when mask not used

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        hidden_dim,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )
    # left and right are modified in‑place

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored if config["nomask"]=True
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half precision)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w, bias=None)   # [B, N, N, H]
    right = F.linear(x_h, right_proj_w, bias=None) # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Gates (half precision)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_h, left_gate_w, bias=None).sigmoid()
    right_gate = F.linear(x_h, right_gate_w, bias=None).sigmoid()
    out_gate = F.linear(x_h, out_gate_w, bias=None).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) with gates via Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # Prepare mask in half‑precision and flatten
        mask_h = mask.to(torch.float16).contiguous()
        # Ensure contiguous layout for all tensors
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        # Fuse mask + gates in‑place
        _apply_gate_and_mask(left, right, left_gate, right_gate, mask_h, hidden_dim)
    else:
        # Simple element‑wise gating when no mask is required
        left = left * left_gate
        right = right * right_gate

    # Release gate tensors we no longer need
    del left_gate, right_gate, x_h

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched GEMM using cuBLAS (torch.bmm)
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # Batched matrix multiplication (left * right^T)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w, bias=None)

    # Cast back to float32 for the final output
    return out.to(torch.float32)

"""
TriMul outgoing operator with Triton‑fused mask and gating.

The heavy O(N³) contraction is performed with a batched half‑precision
matrix multiplication (cuBLAS).  A small Triton kernel fuses the optional
mask and the sigmoid gates into the left/right projected tensors, reducing
memory traffic.  All linear layers, layer‑norms and the final projection are
executed in float16; the result is cast back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr,
    total_elements, hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """Fuse optional mask and sigmoid gates into left/right tensors."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=valid, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=valid, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=tl.float16(0.0))

    if HAS_MASK:
        # One mask value per (batch, i, k) – shared across hidden_dim entries.
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store back in‑place
    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)

    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) – may be ignored
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half precision)
    # ------------------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )  # [B, N, N, H]
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Gates (half precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) with gates via Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # Ensure contiguous layout in half precision
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_h = mask.to(torch.float16).contiguous()
        _apply_gate_and_mask(left, right, left_gate, right_gate, mask_h, hidden_dim)
    else:
        left = left * left_gate
        right = right * right_gate

    # Release now‑unused tensors
    del left_gate, right_gate, x_h

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Performed as a batched GEMM (cuBLAS) on half‑precision tensors.
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)  # (B*H, N, N)

    # Batched matrix multiplication with transpose on the second operand
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Cast back to float32 for the final output
    return out.to(torch.float32)
