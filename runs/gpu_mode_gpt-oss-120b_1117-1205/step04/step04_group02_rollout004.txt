import triton

@triton.jit
def gate_fuse_kernel(left_ptr, right_ptr, left_gate_ptr, right_gate_ptr,
                    mask_ptr, total_elements, hidden_dim,

    mask = offs < total_elements

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # layernorm
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])
    x = x.to(torch.float16)

    # linear proj
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # gates
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # fuse mask & gates
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).contiguous().view(-1)  # [B*N*N]
        # flatten all tensors for kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        fuse_gates_and_mask(left_f, right_f, left_gate_f, right_gate_f, mask_h, hidden_dim)
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # core contraction: out = left @ right^T per hidden dim
    # permute to (B, hidden_dim, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left.shape
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # output layernorm & projection
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"].to(torch.float16),
                       bias=weights["to_out_norm.bias"].to(torch.float16))
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    return out.to(torch.float32)

"""
Outgoing TriMul (AlphaFold3) forward pass.

Algorithm
---------
1. Layer‑norm over the last dimension (dim) and cast to float16.
2. Linear projections → left/right (dim → hidden_dim) in half precision.
3. Compute three sigmoid gates (left_gate, right_gate, out_gate) in half.
4. Fuse mask (if present) with the two gates using a small Triton kernel:
   left  = left  * left_gate  * mask
   right = right * right_gate * mask
   (the kernel works in‑place on the flattened tensors).
5. Core O(N³) contraction:
   For each batch and hidden channel d we compute
       out_{i,j,d} = Σ_k left_{i,k,d} * right_{j,k,d}
   which is a batched GEMM: left @ rightᵀ.
6. Apply output LayerNorm, scale with out_gate and final linear projection.
7. Return the result in float32.

Only the mask‑gate fusion is implemented in Triton; the heavy
contraction uses torch.bmm (cuBLAS) which is already optimal for the
batched matrix multiplications.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_fuse_kernel(
    left_ptr,          # * half
    right_ptr,         # * half
    left_gate_ptr,     # * half
    right_gate_ptr,    # * half
    mask_ptr,          # * half (broadcasted)
    total_elements,    # i32
    hidden_dim,        # i32
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """In‑place fusion of mask and gates.

    left  = left  * left_gate  * mask (if HAS_MASK)
    right = right * right_gate * mask (if HAS_MASK)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights      : dict of torch.Tensors (float32) containing all linear
                         and LayerNorm parameters
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half precision)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Compute gates (half precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) with the two gates via Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → half, flattened
        mask_h = mask.to(torch.float16).contiguous().view(-1)

        # flatten tensors for the kernel (they are contiguous after view)
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(
            left_f, right_f,
            left_gate_f, right_gate_f,
            mask_h,
            hidden_dim,
        )
        # reshape back to original layout
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask → simple element‑wise gating
        left = left * left_gate
        right = right * right_gate

    # Gates no longer needed
    del left_gate, right_gate, x

    # ------------------------------------------------------------------
    # 5. Core O(N³) contraction via batched GEMM (half precision)
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    (equivalently: out = left @ rightᵀ for each hidden channel)
    # ------------------------------------------------------------------
    # Bring hidden dimension to the second axis: (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left.shape
    left_mat = left.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right.view(B * H, N, N) # (B*H, N, N)

    # Batched GEMM: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in the original dtype (float32)
    return out.to(torch.float32)
