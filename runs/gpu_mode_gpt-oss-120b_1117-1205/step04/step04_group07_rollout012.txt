def _apply_mask_gate_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements: tl.int32,
    hidden_dim: tl.int32,

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)
    left = F.linear(x_h, left_proj_weight, bias=None)
    right = F.linear(x_h, right_proj_weight, bias=None)

    # Gate linear layers
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_h, left_gate_weight, bias=None).sigmoid()
    right_gate = F.linear(x_h, right_gate_weight, bias=None).sigmoid()
    out_gate = F.linear(x_h, out_gate_weight, bias=None).sigmoid()

    # Apply mask and gates using Triton kernel
    if not nomask:
        # mask shape [B, N, N]; expand to half
        mask_h = mask.to(torch.float16).contiguous()
    else:
        mask_h = None

    # Ensure tensors contiguous for kernel
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    _apply_gate_mask_fuse(left, right, left_gate, right_gate, mask_h, hidden_dim)

    # At this point left and right have been gated and masked.

    # Contract via batched GEMM
    # left shape [B,N,N,hidden_dim]; we need shape [B*hidden_dim,N,N]
    B, N, _, _ = left.shape
    # Permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Reshape
    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)

    # Compute out = left_mat @ right_mat^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # Reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # shape [B,N,N,hidden_dim]

    # Output layernorm and gating
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    # Multiply out_gate (already half)
    out = out * out_gate

    # Final linear projection
    to_out_weight = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_weight, bias=None)

    # Return float32
    return out.to(torch.float32)

"""
Optimized outgoing TriMul forward pass.

The implementation follows the reference PyTorch module but performs the
element‑wise mask and gate fusion in a custom Triton kernel.  All linear
projections, layer‑norms and the O(N³) contraction are kept in half‑precision
(`torch.float16`) to exploit Tensor‑Core acceleration.  The contraction is
implemented as a batched GEMM (`torch.bmm`) after reshaping, which is the
fastest available approach on NVIDIA GPUs.  The final result is returned as
`float32`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -----------------------------------------------------------------------
# Triton kernel: fuse mask (if present) and gating into left/right tensors
# -----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,                # * half
    right_ptr,               # * half
    left_gate_ptr,           # * half
    right_gate_ptr,          # * half
    mask_ptr,                # * half (broadcasted)
    total_elements: tl.int32,
    hidden_dim: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    # Load values
    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # mask is sized [B, N, N] – broadcast over hidden_dim
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store the fused results back in‑place
    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def _apply_gate_mask_fuse(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fuse of mask (optional) and gating values using Triton.

    Parameters
    ----------
    left, right : torch.Tensor
        Tensors of shape [B, N, N, hidden_dim] (float16, contiguous).
    left_gate, right_gate : torch.Tensor
        Gating tensors of the same shape (float16, contiguous).
    mask : torch.Tensor or None
        Mask of shape [B, N, N] (float16) or None if not used.
    hidden_dim : int
        Size of the hidden dimension (compile‑time constant for the kernel).
    """
    total = left.numel()
    BLOCK = 32768  # 2^15, good trade‑off between launch overhead and occupancy
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    # If no mask, provide a dummy pointer (won't be dereferenced)
    mask_ptr = mask.view(-1) if mask is not None else left

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        hidden_dim,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
    )


# -----------------------------------------------------------------------
# Entry point
# -----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – may be ignored
        - weights      : dict with keys
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        - config       : dict containing at least "dim", "hidden_dim", optional "nomask"
    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> half)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half)
    # ------------------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gating values (half, sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) with gates using Triton
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask.to(torch.float16).contiguous()
    else:
        mask_h = None

    # Ensure contiguity for the kernel
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    _apply_gate_mask_fuse(left, right, left_gate, right_gate, mask_h, hidden_dim)

    # ------------------------------------------------------------------
    # 5. Core O(N^3) contraction via batched GEMM
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # left shape == right shape == [B, N, N, hidden_dim]

    # Move hidden dimension to the second axis for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Collapse batch and hidden dimensions
    left_mat = left_perm.view(B * hidden_dim, N, N)      # (B*H, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)    # (B*H, N, N)

    # Batched matrix multiplication (Tensor‑Core enabled)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore original layout: [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise gating (half)

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return as float32 to match the reference implementation
    return out.to(torch.float32)
