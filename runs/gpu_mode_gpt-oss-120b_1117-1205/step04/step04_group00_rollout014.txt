@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Limit
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to start of batch
    a_batch_offset = a_ptr + pid_batch * batch_stride_a
    b_batch_offset = b_ptr + pid_batch * batch_stride_b
    c_batch_offset = c_ptr + pid_batch * batch_stride_c

    # Compute a block pointer for a (rows = offs_m, cols = tile K)
    # We'll loop over K dimension.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute k offsets in this block
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_offset + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B_T: shape (BLOCK_K, BLOCK_N) (since we need B transpose)
        # Offsets: row = offs_k (k), col = offs_n (j)
        b_ptrs = b_batch_offset + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Compute
        acc += tl.dot(a, b)

        # Move pointers? Not needed because we compute offsets each iteration.

    # Write back to C
    c_ptrs = c_batch_offset + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass with a fused Triton kernel for the
batch‑wise matrix multiplication C = A @ Bᵀ, where A and B have shape
[batch, N, N] (batch = B * hidden_dim). The rest of the module (LayerNorm,
Linear projections, gating and optional mask) is performed in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """C[batch, i, j] = Σₖ A[batch, i, k] * B[batch, j, k]  (Bᵀ is folded in)."""
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Compute base pointers for this batch
    a_batch = a_ptr + pid_batch * batch_stride_a
    b_batch = b_ptr + pid_batch * batch_stride_b
    c_batch = c_ptr + pid_batch * batch_stride_c

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load Bᵀ tile: we read B[j, k] (original B) as Bᵀ[k, j]
        #   offset = k * stride_bk + n * stride_bn   (k varies fastest)
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_N)

    # Store the result tile
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (optional, bool/float)
            - weights: dict of model parameters (float32)
            - config: dict containing "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----------------------------------------------------------------------
    # 1. Input LayerNorm (float32) -> half precision
    # ----------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ----------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (half)
    # ----------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16), bias=None)

    # ----------------------------------------------------------------------
    # 3. Gates (half)
    # ----------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # ----------------------------------------------------------------------
    # 4. Apply mask (if any) and multiply by gates
    # ----------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape [B, N, N] -> broadcast over hidden_dim
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f * left_gate
        right = right * mask_f * right_gate
    else:
        left = left * left_gate
        right = right * right_gate

    # ----------------------------------------------------------------------
    # 5. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as a batched GEMM (A @ Bᵀ) via a Triton kernel.
    # ----------------------------------------------------------------------
    # Transpose to (B, hidden_dim, N, N) so each [b,d] pair is a matrix.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    batch = B * H

    # Collapse batch and hidden dimension for the GEMM
    left_mat = left_perm.view(batch, N, N).contiguous()   # (batch, N, N)
    right_mat = right_perm.view(batch, N, N).contiguous()  # (batch, N, N)

    # Allocate output tensor
    out_mat = torch.empty_like(left_mat)  # same dtype (float16) and device

    # Strides for the kernel (all are int64)
    a_batch_stride, a_stride_m, a_stride_k = left_mat.stride()
    b_batch_stride, b_stride_row, b_stride_col = right_mat.stride()
    c_batch_stride, c_stride_m, c_stride_n = out_mat.stride()

    # For Bᵀ we need stride over the reduction axis (k) = original column stride,
    # and stride over the output column (j) = original row stride.
    stride_bk = b_stride_col   # = 1
    stride_bn = b_stride_row   # = N

    # Tile sizes (tuned for typical B200 memory hierarchy)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_m_blocks = triton.cdiv(N, BLOCK_M)
    num_n_blocks = triton.cdiv(N, BLOCK_N)

    grid = (batch, num_m_blocks, num_n_blocks)
    _batched_matmul_kernel[grid](
        left_mat, right_mat, out_mat,
        a_batch_stride, a_stride_m, a_stride_k,
        b_batch_stride, stride_bk, stride_bn,
        c_batch_stride, c_stride_m, c_stride_n,
        N, N, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Restore the original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # ----------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    # Return in float32 as required by the API
    return out.to(torch.float32)
