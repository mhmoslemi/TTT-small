"""
Optimized outgoing TriMul forward (AlphaFold3) using Triton for mask+gate fusion.

Algorithm
---------
1. Layer‑norm the input tensor (float32) and cast to float16.
2. Linear projections to hidden_dim (left/right) and gate projections (left_gate,
   right_gate, out_gate) using FP16 weights.
3. Fuse mask (if present) with left/right gates in a single Triton kernel:
      left  = left  * left_gate  * mask
      right = right * right_gate * mask
   If no mask, the kernel simply multiplies with the gates.
4. Core contraction:
      out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   This is implemented as a batched GEMM using torch.bmm:
      (B, H, N, N) -> (B*H, N, N) @ (B*H, N, N)^T
5. Output LayerNorm, out‑gate scaling, and final linear projection back to `dim`.
6. Return result as float32.

Only the mask+gate fusion is performed in Triton; the heavyweight
matrix multiplication leverages cuBLAS via torch.bmm.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse left/right with their gates and optional mask
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # float16*
    right_ptr,         # float16*
    left_gate_ptr,     # float16*
    right_gate_ptr,    # float16*
    mask_ptr,          # float16* (broadcasted, length = B*N*N)
    total_elements,    # int32
    H: tl.constexpr,   # hidden_dim
    HAS_MASK: tl.constexpr,  # 1 if mask is present, 0 otherwise
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    # Load values
    left = tl.load(left_ptr + offs, mask=active, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=active, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=tl.float16(0.0))

    if HAS_MASK:
        # Each hidden entry shares the same mask value -> index by offs // H
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store results back in‑place
    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    Apply (mask) * gate fusion in‑place on `left` and `right` tensors.
    All tensors must be 1‑D contiguous float16 views.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when HAS_MASK==0

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        H=hidden_dim,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )
    # In‑place modification; nothing to return.


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) – may be ignored
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> float16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (FP16)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gates (FP16) with sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) with gates via Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        # Ensure contiguous storage
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        # Mask: cast to float16, flatten
        mask_f = mask.to(torch.float16).contiguous().view(-1)

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            hidden_dim,
        )
        # Reshape back (views share storage)
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask: simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # Gates fused; free temporaries
    del left_gate, right_gate, x

    # ------------------------------------------------------------------
    # 5. Core contraction via batched GEMM (BMM)
    #    out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) for efficient batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N) # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=1e-5,
    )
    out = out * out_gate  # element‑wise gating (FP16)

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in float32 as required by the interface
    return out.to(torch.float32)
