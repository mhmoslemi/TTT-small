def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(2) # combined batch * head
    pid_x = tl.program_id(0) # i block
    pid_y = tl.program_id(1) # j block

    b = pid_z // H
    h = pid_z % H

    # Compute start offsets
    i_offset = pid_x * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_y * BLOCK_N + tl.arange(0, BLOCK_N)
    k_offset = tl.arange(0, BLOCK_K) # for each loop iteration

    # Masks out-of-bound for i, j
    i_mask = i_offset < N
    j_mask = j_offset < N

    # base pointers for current slice (b,i,k,h)
    # left: [B,N,N,H] layout: b,i,k,h with strides (stride_left_b, stride_left_i, stride_left_k, stride_left_h)
    # we compute ptrs: left_ptr + b*stride_left_b + i_offset*stride_left_i + k_offset*stride_left_k + h*stride_left_h

    # right: [B,N,N,H] but we need right[b,j,k,h] (notice second index is j)
    # Strides: (stride_right_b, stride_right_j, stride_right_k, stride_right_h)

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute current k base offset
        k_base = k * BLOCK_K
        cur_k = k_base + k_offset
        k_mask = cur_k < N

        # load left block
        left_ptrs = left_ptr + b*stride_left_b + i_offset[:,None]*stride_left_i + cur_k[None,:]*stride_left_k + h*stride_left_h
        left = tl.load(left_ptrs, mask=i_mask[:,None] & k_mask[None,:], other=0.0)
        # load right block
        right_ptrs = right_ptr + b*stride_right_b + j_offset[:,None]*stride_right_j + cur_k[None,:]*stride_right_k + h*stride_right_h
        right = tl.load(right_ptrs, mask=j_mask[:,None] & k_mask[None,:], other=0.0)

        # compute dot product for each i,j over k block
        acc += tl.dot(left, right)  # tl.dot expects (M,K) and (K,N) -> (M,N)

    # after loop, store to out
    out_ptrs = out_ptr + b*stride_out_b + i_offset[:,None]*stride_out_i + j_offset[None,:]*stride_out_j + h*stride_out_h
    tl.store(out_ptrs, acc.to(tl.float16), mask=i_mask[:,None] & j_mask[None,:])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    # Norm: using torch.nn.functional.layer_norm.
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # cast to half
    x_h = x.to(torch.float16)

    # Projections
    left = torch.nn.functional.linear(x_h, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = torch.nn.functional.linear(x_h, weights["right_proj.weight"].to(torch.float16), bias=None)

    # Gates
    left_gate = torch.nn.functional.linear(x_h, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = torch.nn.functional.linear(x_h, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = torch.nn.functional.linear(x_h, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # Apply gating
    left = left * left_gate
    right = right * right_gate

    if not nomask:
        # mask shape [B,N,N], need to broadcast to last dim
        mask_h = mask.to(torch.float16)
        left = left * mask_h.unsqueeze(-1)
        right = right * mask_h.unsqueeze(-1)
    # Ensure contiguity
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, _ = left.shape # left shape B,N,N,H
    # Output allocation
    out = torch.empty_like(left, dtype=torch.float16)

    # Get strides
    s_left = left.stride()
    s_right = right.stride()
    s_out = out.stride()

    # kernel launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * hidden_dim)

    trimul_contraction_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        s_left[0], s_left[1], s_left[2], s_left[3],
        s_right[0], s_right[1], s_right[2], s_right[3],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Post processing
    # out_norm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    # multiply out_gate
    out = out * out_gate
    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)
    # cast to float32
    out = out.to(torch.float32)
    return out

@triton.jit
def trimul_contraction_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_x = tl.program_id(0)
    pid_y = tl.program_id(1)
    pid_z = tl.program_id(2)

    b = pid_z // H
    h = pid_z % H

    # offsets
    i = pid_x * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_y * BLOCK_N + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)

    i_mask = i < N
    j_mask = j < N

    # base pointers
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = kk * BLOCK_K + k
        k_mask = cur_k < N

        # left block (BM, BK)
        left_ptrs = left_base + i[:, None] * stride_left_i + cur_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)

        # right block (BK, BN)
        # note: right dimensions (j, k) but we need (BK, BN): we load with k as first axis
        right_ptrs = right_base + cur_k[:, None] * stride_right_k + j[None, :] * stride_right_i  # careful: stride for j dimension = stride_right_i (since second dim)
        # However stride_right_i = right.stride(1) = N*H, which is same as stride_left_i.

"""
TriMul (outgoing) forward pass fused with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the 4‑D input (B, N, N, C) using the provided weight/bias.
2. Cast to fp16 and apply two linear projections (left / right) to the hidden
   dimension H.
3. Compute three gate tensors (left_gate, right_gate, out_gate) via linear
   layers followed by sigmoid and apply them element‑wise (optionally also a
   mask).
4. Core contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a tiled GEMM in a Triton kernel.  The kernel processes a
   block of the (i, j) output plane for a specific (batch, head) pair and
   accumulates over k in chunks of size BLOCK_K.
5. Apply output layer‑norm over the hidden dimension, scale by out_gate and
   project back to the original dimension.
6. Return the result in float32.

The Triton kernel replaces the expensive batched `torch.bmm` and
fuses the whole N³ contraction, yielding a substantial speed‑up.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: contract left and right tensors over the k dimension.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_contraction_kernel(
    left_ptr,    # fp16*  [B,N,N,H]
    right_ptr,   # fp16*  [B,N,N,H]
    out_ptr,     # fp16*  [B,N,N,H] (output)
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)  # block index over output rows (i)
    pid_j = tl.program_id(1)  # block index over output cols (j)
    pid_z = tl.program_id(2)  # combined batch * hidden channel index

    # decode batch and hidden channel from pid_z
    b = pid_z // H
    h = pid_z % H

    # offsets for the i, j, and k dimensions
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    k_off = tl.arange(0, BLOCK_K)

    i_mask = i_off < N
    j_mask = j_off < N

    # base pointers (include batch and channel offsets)
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k in tiles of size BLOCK_K
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        cur_k = kk * BLOCK_K + k_off
        k_mask = cur_k < N

        # left block: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + i_off[:, None] * stride_left_i + cur_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=i_mask[:, None] & k_mask[None, :],
                       other=0.0)

        # right block: shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_base + cur_k[:, None] * stride_right_k + j_off[None, :] * stride_right_j
        right = tl.load(right_ptrs,
                        mask=k_mask[:, None] & j_mask[None, :],
                        other=0.0)

        # accumulate dot product over the k‑tile
        acc += tl.dot(left, right)

    # write the final tile (cast to fp16)
    out_ptrs = out_base + i_off[:, None] * stride_out_i + j_off[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=i_mask[:, None] & j_mask[None, :])


# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask         : torch.Tensor [B, N, N] (optional, may be ignored)
        - weights_dict : dict of model parameters (float32)
        - config_dict  : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)  # fp16 for the heavy part

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (fp16)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16), bias=None)

    # ------------------------------------------------------------------
    # 3. Gates (fp16) and sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # ------------------------------------------------------------------
    # 4. Apply gating (and optional mask) in‑place
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate
    if not nomask:
        # mask: [B, N, N] -> broadcast over hidden dim
        mask_h = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_h
        right = right * mask_h

    # Ensure contiguous layout for the kernel
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, H = left.shape  # H == hidden_dim

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton (out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h])
    # ------------------------------------------------------------------
    out = torch.empty_like(left, dtype=torch.float16)

    # strides for the tensors (int64 values)
    s_left = left.stride()
    s_right = right.stride()
    s_out = out.stride()

    # tile sizes (tuned for fp16 GEMM)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # i dimension
        triton.cdiv(N, BLOCK_N),   # j dimension
        B * H,                      # batch * hidden channel
    )

    _trimul_contraction_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_left[0], s_left[1], s_left[2], s_left[3],
        s_right[0], s_right[1], s_right[2], s_right[3],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    # Cast back to float32 for the final output
    return out.to(torch.float32)
