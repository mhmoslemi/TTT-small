"""
TriMul “outgoing” forward pass (optimized).

Algorithm:
1. Input LayerNorm (float32 → float16).
2. Linear projections to hidden_dim (left/right) and gate projections
   (left_gate/right_gate/out_gate) in float16.
3. Fuse mask (if present) with left/right gates via a Triton kernel:
   left  ← left  * left_gate  * mask
   right ← right * right_gate * mask
   (mask broadcast over hidden_dim).
4. Core O(N³) contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a large batched matrix multiplication:
   out = (left @ rightᵀ) for each (batch, hidden_dim) pair.
5. Output LayerNorm on hidden_dim, gating with out_gate, final linear
   projection back to dim.
All heavy arithmetic is performed in float16; the final output is cast to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask (if present) with left/right gates
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,        # half *
    right_ptr,       # half *
    left_gate_ptr,   # half *
    right_gate_ptr,  # half *
    mask_ptr,        # half *
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present, else 0
    H: tl.constexpr,          # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # Load tensors
    left      = tl.load(left_ptr + offs,      mask=mask, other=0.0)
    right     = tl.load(right_ptr + offs,     mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr + offs,mask=mask, other=0.0)

    if HAS_MASK:
        # One mask entry corresponds to H consecutive hidden channels
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left  = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left * left_gate
        right = right * right_gate

    tl.store(left_ptr  + offs, left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _fuse_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of mask (if any) and gates:
        left  ← left  * left_gate  * mask
        right ← right * right_gate * mask
    All tensors are half‑precision and contiguous.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when no mask

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # left and right are modified in place

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) or None
            - weights: dict mapping parameter names to tensors
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> half)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half precision)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )  # [B, N, N, hidden_dim]
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gate values (half precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) and gates via Triton
    # ------------------------------------------------------------------
    if (not nomask) and (mask is not None):
        # mask shape [B, N, N] -> broadcast over hidden_dim
        mask_h = mask.unsqueeze(-1).to(torch.float16).contiguous()
        _fuse_gate_and_mask(
            left, right,
            left_gate, right_gate,
            mask_h,
            hidden_dim,
        )
    else:
        # No mask: simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as batched matrix multiplication (left @ rightᵀ)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # Rearrange to [B, hidden_dim, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]
    # Transpose the last two dimensions of right (so we get rightᵀ)
    right_perm_T = right_perm.transpose(-2, -1).contiguous()  # [B, H, N, N]

    # Merge batch and channel dims for batched GEMM
    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_perm_T.view(B * hidden_dim, N, N)

    # Batched matrix multiplication in FP16
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)

    # Restore original shape [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in float32 as required
    return out.to(torch.float32)
