Inside:

pid_n = tl.program_id(1)
pid_z = tl.program_id(2)

offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)

base_r = right_ptr + pid_z * stride_r_batch

    k_offset = k * BLOCK_K
    # load left tile: shape (BLOCK_M, BLOCK_K)

    acc += tl.dot(a, b)   # a: (M,K), b: (K,N) => (M,N)

# done loops
# store result cast to float16

c_ptrs = base_o + (offs_m[:, None] * stride_o_m) + (offs_n[None, :] * stride_o_n)
tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul outgoing implementation with Triton fusion.

The heavy O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed by a custom batched GEMM kernel.
Masking and gating are fused into a single Triton kernel to avoid
extra memory traffic. All arithmetic is in float16; the final result is
cast back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: fuse mask (if present) and gating into the projected tensors
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # half* flattened left tensor
    right_ptr,           # half* flattened right tensor
    left_gate_ptr,       # half* flattened left‑gate tensor
    right_gate_ptr,      # half* flattened right‑gate tensor
    mask_ptr,            # half* flattened mask (broadcast over hidden dim)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present, else 0
    H: tl.constexpr,          # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elements

    left = tl.load(left_ptr + offs, mask=m, other=0.0)
    right = tl.load(right_ptr + offs, mask=m, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=m, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=m, other=0.0)

    if HAS_MASK:
        # one mask value per (batch, i, k) group of H hidden values
        mask_offs = offs // H
        mask_val = tl.load(mask_ptr + mask_offs, mask=m, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=m)
    tl.store(right_ptr + offs, right, mask=m)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of gating and optional masking using Triton.
    All tensors must be contiguous half‑precision.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when no mask

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # tensors are modified in place


# ----------------------------------------------------------------------
# Kernel 2: Batched GEMM for the O(N³) contraction
# Computes out = left @ rightᵀ for each (batch, hidden) pair.
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    left_ptr,   # half* (B*H, M, K)
    right_ptr,  # half* (B*H, N, K) – note K is the inner dim
    out_ptr,    # half* (B*H, M, N)
    M: tl.int32,
    N: tl.int32,
    K: tl.int32,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_n, stride_r_k,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block column
    pid_z = tl.program_id(2)  # batch * hidden index

    # offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # boundary checks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch/hidden plane
    base_l = left_ptr + pid_z * stride_l_batch
    base_r = right_ptr + pid_z * stride_r_batch
    base_o = out_ptr + pid_z * stride_o_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a tile of left: (BLOCK_M, BLOCK_K)
        a_ptrs = base_l + (offs_m[:, None] * stride_l_m) + (offs_k[None, :] * stride_l_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load a tile of right transposed: (BLOCK_K, BLOCK_N)
        b_ptrs = base_r + (offs_k[:, None] * stride_r_k) + (offs_n[None, :] * stride_r_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Write back
    c = acc.to(tl.float16)
    c_ptrs = base_o + (offs_m[:, None] * stride_o_m) + (offs_n[None, :] * stride_o_n)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half precision)
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16), bias=None)

    # ------------------------------------------------------------------
    # 3. Gates (half precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) with gates via Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).contiguous().view(-1)          # [B*N*N]
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(left_f, right_f, left_gate_f, right_gate_f, mask_h, hidden_dim)

        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched GEMM (left @ rightᵀ) using Triton.
    # ------------------------------------------------------------------
    B, N_seq, _, _ = left.shape  # B, N, N, H
    # Move hidden dimension to the batch axis for GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    BH = B * hidden_dim
    left_bh = left_perm.view(BH, N_seq, N_seq)   # (B*H, N, N)
    right_bh = right_perm.view(BH, N_seq, N_seq) # (B*H, N, N)

    out_bh = torch.empty_like(left_bh)  # (B*H, N, N), float16

    # Kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        triton.cdiv(N_seq, BLOCK_M),
        triton.cdiv(N_seq, BLOCK_N),
        BH,
    )

    _batched_gemm_kernel[grid](
        left_bh,
        right_bh,
        out_bh,
        M=N_seq,
        N=N_seq,
        K=N_seq,
        stride_l_batch=left_bh.stride(0),
        stride_l_m=left_bh.stride(1),
        stride_l_k=left_bh.stride(2),
        stride_r_batch=right_bh.stride(0),
        stride_r_n=right_bh.stride(1),
        stride_r_k=right_bh.stride(2),
        stride_o_batch=out_bh.stride(0),
        stride_o_m=out_bh.stride(1),
        stride_o_n=out_bh.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout: (B, N, N, H)
    out = out_bh.view(B, hidden_dim, N_seq, N_seq).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    # Cast back to float32 for the final output
    return out.to(torch.float32)
