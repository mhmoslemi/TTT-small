"""
TriMul outgoing forward pass (AlphaFold3 style).

The heavy O(N³) contraction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
is implemented as a batched matrix multiplication using a Triton kernel.
Each hidden dimension `d` (and each batch) is handled as an independent
gemm:  left[..., d] @ right[..., d].T .
All other operations (LayerNorm, linear projections, gating, masking) are
performed in PyTorch (float16) for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # half *
    right_ptr,         # half *
    out_ptr,           # half *
    B, N, D,           # int32
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute:
        out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
    where left and right have shape [B, N, N, D].
    """
    pid0 = tl.program_id(0)   # combined batch * hidden dim
    pid1 = tl.program_id(1)   # i tile
    pid2 = tl.program_id(2)   # j tile

    # decode batch and hidden dimension indices
    b = pid0 // D
    d = pid0 % D

    # tile start coordinates
    offs_i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension in blocks
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, i, k, d]   shape (M, K)
        left_ptrs = left_ptr + \
            b * stride_left_b + d * stride_left_d + \
            offs_i[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k
        # right[b, j, k, d]  shape (N, K) (we will transpose inside dot)
        right_ptrs = right_ptr + \
            b * stride_right_b + d * stride_right_d + \
            offs_j[:, None] * stride_right_i + \
            offs_k[None, :] * stride_right_k

        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0).to(tl.float32)

        # right is (N, K); we need its transpose -> (K, N)
        acc += tl.dot(left, right, trans_b=True)

    # write out result in half precision
    out_ptrs = out_ptr + \
        b * stride_out_b + d * stride_out_d + \
        offs_i[:, None] * stride_out_i + \
        offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor: torch.Tensor [B, N, N, dim]  (float32)
        - mask        : torch.Tensor [B, N, N] (bool or numeric) or None
        - weights     : dict of model parameters
        - config      : dict containing "dim" and "hidden_dim"
    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> float16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    ).to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (float16)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )  # [B, N, N, hidden_dim]
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gate values (float16)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Apply mask (if any) and gates
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.unsqueeze(-1).to(torch.float16)  # [B, N, N, 1]
        left = left * left_gate * mask_h
        right = right * right_gate * mask_h
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³) contraction via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty_like(left)  # shape [B, N, N, hidden_dim], float16

    # strides for tensors (int64)
    l_strides = left.stride()   # (B, i, k, d)
    r_strides = right.stride()
    o_strides = out.stride()

    # block sizes – tuned for A100/B200 memory registers
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                 # batch * hidden dimension
        triton.cdiv(N, BLOCK_M),        # i tiles
        triton.cdiv(N, BLOCK_N),        # j tiles
    )

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, hidden_dim,
        l_strides[0], l_strides[1], l_strides[2], l_strides[3],
        r_strides[0], r_strides[1], r_strides[2], r_strides[3],
        o_strides[0], o_strides[1], o_strides[2], o_strides[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in FP32 as required by the interface
    return out.to(torch.float32)
