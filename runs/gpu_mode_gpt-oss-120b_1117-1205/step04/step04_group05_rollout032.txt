@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Compute valid mask
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Compute pointer to start of batch
    a_ptrs = A_ptr + batch_id * stride_Ab
    b_ptrs = B_ptr + batch_id * stride_Bb
    c_ptrs = C_ptr + batch_id * stride_Cb

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a = tl.load(a_ptrs + (offs_m[:, None] * stride_Am) + (offs_k[None, :] * stride_Ak), mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        # Load B tile (K x N)
        b = tl.load(b_ptrs + (offs_k[:, None] * stride_Bk) + (offs_n[None, :] * stride_Bn), mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Compute dot product
        acc += tl.dot(a, b)

        # Increment pointers (optional if we compute offsets each iteration)
        # Not needed since we compute using offsets

    # Write back
    c = acc.to(tl.float16)
    tl.store(c_ptrs + (offs_m[:, None] * stride_Cm) + (offs_n[None, :] * stride_Cn), c, mask=mask_m[:, None] & mask_n[None, :])

Note: In loop, we compute a and b each iteration using current tiles.

But note we need to pass strides: stride_Am = stride of A between rows (M dimension?), slope? Actually for left (A) shape (BATCH, M, K).     right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)

@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for M and N
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to batch start
    a_ptr = A_ptr + batch_id * stride_Ab
    b_ptr = B_ptr + batch_id * stride_Bb
    c_ptr = C_ptr + batch_id * stride_Cb

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr + (offs_m[:, None] * stride_Am) + (offs_k[None, :] * stride_Ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=tl.float16(0.0)
        )
        # Load B tile: shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr + (offs_k[:, None] * stride_Bk) + (offs_n[None, :] * stride_Bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=tl.float16(0.0)
        )
        # Multiply-accumulate
        acc += tl.dot(a, b)

    # Write back
    c = acc.to(tl.float16)
    tl.store(
        c_ptr + (offs_m[:, None] * stride_Cm) + (offs_n[None, :] * stride_Cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :]
    )

"""
TriMul (outgoing) forward pass with Triton acceleration.

The implementation follows the AlphaFold3 TriMul module:
1. Layer‑norm the input.
2. Linear projections to a hidden dimension (no bias) → half precision.
3. Compute sigmoid gates (half precision).
4. Fuse mask (if present) with the left/right projections and their gates
   using a tiny Triton kernel (in‑place).
5. Core O(N³·H) contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is implemented as a batched matrix‑multiply (left @ rightᵀ) using a
   custom Triton kernel that processes each (batch, hidden) pair independently.
6. Output layer‑norm, out‑gate scaling and final linear projection back to the
   original channel dimension.  The final result is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Triton kernel: fuse mask + sigmoid gates into the projected tensors
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # half* flattened left tensor
    right_ptr,           # half* flattened right tensor
    left_gate_ptr,       # half* flattened left‑gate tensor
    right_gate_ptr,      # half* flattened right‑gate tensor
    mask_ptr,            # half* flattened mask (B*N*N)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present, else 0
    H: tl.constexpr,          # hidden dimension size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    within = offs < total_elements

    left = tl.load(left_ptr + offs, mask=within, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=within, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=within, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=within, other=tl.float16(0.0))

    if HAS_MASK:
        # one mask per (batch, i, k) group of H hidden values
        mask_offs = offs // H
        mask_val = tl.load(mask_ptr + mask_offs, mask=within, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=within)
    tl.store(right_ptr + offs, right, mask=within)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """In‑place fusion of gating and optional masking using Triton."""
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when no mask

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # no return – tensors are modified in place


# ----------------------------------------------------------------------
# 2) Triton kernel: batched matrix multiplication (left @ rightᵀ)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the output block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_ptr = A_ptr + batch_id * stride_Ab
    b_ptr = B_ptr + batch_id * stride_Bb
    c_ptr = C_ptr + batch_id * stride_Cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a tile of A (M x K)
        a = tl.load(
            a_ptr + (offs_m[:, None] * stride_Am) + (offs_k[None, :] * stride_Ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=tl.float16(0.0),
        )
        # Load a tile of B (K x N) – B is already transposed
        b = tl.load(
            b_ptr + (offs_k[:, None] * stride_Bk) + (offs_n[None, :] * stride_Bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=tl.float16(0.0),
        )
        acc += tl.dot(a, b)

    # Write back the result
    c = acc.to(tl.float16)
    tl.store(
        c_ptr + (offs_m[:, None] * stride_Cm) + (offs_n[None, :] * stride_Cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# 3) Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/float) – may be None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Load and cast weights to the appropriate dtype/device
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)

    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)

    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half precision)
    # ------------------------------------------------------------------
    left = F.linear(x_h, left_proj_weight, bias=None)   # [B, N, N, H]
    right = F.linear(x_h, right_proj_weight, bias=None)

    # ------------------------------------------------------------------
    # 3. Gates (half precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, left_gate_weight, bias=None).sigmoid()
    right_gate = F.linear(x_h, right_gate_weight, bias=None).sigmoid()
    out_gate = F.linear(x_h, out_gate_weight, bias=None).sigmoid()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) with gates via Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).contiguous()
        # Flatten tensors for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask_h.contiguous().view(-1)

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            hidden_dim,
        )

        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask: simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # Free unnecessary tensors early
    del left_gate, right_gate, x_h

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched GEMM (left @ rightᵀ) using Triton.
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) and flatten batch & head dims
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # Transpose the right operand so that we compute left @ rightᵀ
    right_t = right_mat.transpose(1, 2).contiguous()   # (B*H, N, N)

    out_mat = torch.empty_like(left_mat)  # (B*H, N, N)

    # Strides for the kernel (all tensors are contiguous)
    stride_a_b = left_mat.stride(0)
    stride_a_m = left_mat.stride(1)
    stride_a_k = left_mat.stride(2)

    stride_b_b = right_t.stride(0)
    stride_b_k = right_t.stride(1)
    stride_b_n = right_t.stride(2)

    stride_c_b = out_mat.stride(0)
    stride_c_m = out_mat.stride(1)
    stride_c_n = out_mat.stride(2)

    # Tile sizes – tuned for A200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_matmul_kernel[grid](
        left_mat,
        right_t,
        out_mat,
        B * H,
        N,
        N,
        N,
        stride_a_b,
        stride_a_m,
        stride_a_k,
        stride_b_b,
        stride_b_k,
        stride_b_n,
        stride_c_b,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight, bias=None)

    # Return in float32 as required by the API
    return out.to(torch.float32)
