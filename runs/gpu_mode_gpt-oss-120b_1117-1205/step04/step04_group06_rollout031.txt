import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elems,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,
):

    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)

    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:

        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def custom_kernel(data):
    """
    Outgoing TriMul forward pass (no gradients).

    The implementation follows the reference PyTorch module but
    off‑loads the element‑wise gate+mask fusion to a tiny Triton kernel
    and keeps the expensive contraction in half precision using
    torch.einsum, which maps to a highly tuned batched GEMM.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float16

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----- 1) Input LayerNorm -------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    ).to(dtype)  # cast to fp16 for the rest

    # ----- 2) Linear projections ---------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(dtype), None)   # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"].to(dtype), None)

    # ----- 3) Gates -----------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(dtype), None).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(dtype), None).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(dtype), None).sigmoid()

    # ----- 4) Fuse gates (and optional mask) using Triton --------------------
    # flatten tensors for the kernel
    left_f = left.reshape(-1)
    right_f = right.reshape(-1)
    left_gate_f = left_gate.reshape(-1)
    right_gate_f = right_gate.reshape(-1)

    total = left_f.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 0 if nomask or mask is None else 1
    if has_mask:
        mask_f = mask.to(dtype).contiguous().reshape(-1)   # [B,N,N]
    else:
        # dummy pointer – Triton will ignore it when HAS_MASK==0
        mask_f = left_f

    _gate_mask_fuse_kernel[grid](
        left_f,
        right_f,
        left_gate_f,
        right_gate_f,
        mask_f,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # Reshape back to original layout
    left = left_f.view_as(left)
    right = right_f.view_as(right)

    # Release gate tensors (no longer needed)
    del left_gate, right_gate, left_gate_f, right_gate_f, x

    # ----- 5) Core contraction (batch GEMM) -----------------------------------
    # Einstein summation equivalent to batched matrix multiplication per hidden dim.
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)

    # ----- 6) Output LayerNorm, out_gate, final linear -----------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate  # element‑wise gating

    out = F.linear(out, weights["to_out.weight"].to(dtype), None)

    # Return in float32 as required by the API
    return out.to(torch.float32)

Outgoing TriMul forward pass (AlphaFold3).

- Performs LayerNorm + linear projections in FP16.
- Gates and optional mask are fused in a tiny Triton kernel.
- The O(N³) contraction is carried out with a half‑precision Einstein summation,
  which maps internally to a batched GEMM (highly tuned cuBLAS).
- Output is LayerNormed, gated, and projected back to the original dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # fp16*
    right_ptr,              # fp16*
    left_gate_ptr,          # fp16*
    right_gate_ptr,         # fp16*
    mask_ptr,               # fp16* (broadcasted, unused when HAS_MASK=0)
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,  # 1 → mask present
    H: tl.constexpr,         # hidden_dim
):

    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    right = tl.load(right_ptr + offs, mask=mask, other=0.0)

    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val

        right = right * right_gate

            - mask: torch.Tensor of shape [B, N, N] (optional, may be None)
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float16

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    ).to(dtype)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (FP16)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(dtype), None)   # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"].to(dtype), None)

    # ------------------------------------------------------------------
    # 3. Gates (FP16) + sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(dtype), None).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(dtype), None).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(dtype), None).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse gates (and optional mask) in‑place via Triton
    # ------------------------------------------------------------------
    # Flatten tensors to 1‑D for the kernel
    left_f = left.reshape(-1)
    right_f = right.reshape(-1)
    left_gate_f = left_gate.reshape(-1)
    right_gate_f = right_gate.reshape(-1)

    total = left_f.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 0 if nomask or mask is None else 1
    if has_mask:
        mask_f = mask.to(dtype).contiguous().reshape(-1)   # [B,N,N]
    else:
        mask_f = left_f  # dummy; never read when HAS_MASK==0

    _gate_mask_fuse_kernel[grid](
        left_f,
        right_f,
        left_gate_f,
        right_gate_f,
        mask_f,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )

    # Reshape back to original layout
    left = left_f.view_as(left)
    right = right_f.view_as(right)

    # Release unused tensors
    del left_gate, right_gate, left_gate_f, right_gate_f, x

    # ------------------------------------------------------------------
    # 5. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented via half‑precision einsum (maps to batched GEMM)
    # ------------------------------------------------------------------
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(dtype), None)

    # Cast back to float32 for the API contract
    return out.to(torch.float32)
