    right = tl.load(right_ptr + offs, mask=mask, other=0.0)

        left = left * left_gate
        right = right * right_gate
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

"""TriMul outgoing version with Triton fuse for gate+mask.

The kernel fuses element‑wise multiplication of the projected tensors
with their respective gates and an optional mask. The remaining
computation (the N³ contraction) is performed with a batched GEMM
using torch.bmm, which maps to cuBLAS and is highly optimized. The
forward pass follows the reference PyTorch implementation but keeps
all heavy tensor‑wise operations in half‑precision.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half *
    right_ptr,         # half *
    left_gate_ptr,     # half *
    right_gate_ptr,    # half *
    mask_ptr,          # half *
    total_elements: tl.int32,
    H: tl.constexpr,          # hidden dim
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elements

    left = tl.load(left_ptr + offs, mask=m, other=0.0)
    right = tl.load(right_ptr + offs, mask=m, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=m, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=m, other=0.0)

    if HAS_MASK:
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=m, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=m)
    tl.store(right_ptr + offs, right, mask=m)

def custom_kernel(data):
    """Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # Helper to move weights to half precision on the correct device
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # Load weights
    norm_weight = to_half(weights["norm.weight"])
    norm_bias = to_half(weights["norm.bias"])
    left_proj_weight = to_half(weights["left_proj.weight"])
    right_proj_weight = to_half(weights["right_proj.weight"])
    left_gate_weight = to_half(weights["left_gate.weight"])
    right_gate_weight = to_half(weights["right_gate.weight"])
    out_gate_weight = to_half(weights["out_gate.weight"])
    to_out_norm_weight = to_half(weights["to_out_norm.weight"])
    to_out_norm_bias = to_half(weights["to_out_norm.bias"])
    to_out_weight = to_half(weights["to_out.weight"])

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32) and cast to half
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections to hidden dimension (half precision)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight, bias=None)
    right = F.linear(x, right_proj_weight, bias=None)

    # -----------------------------------------------------------------
    # 3. Compute gating tensors (half precision)
    # -----------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight, bias=None).sigmoid()
    right_gate = F.linear(x, right_gate_weight, bias=None).sigmoid()
    out_gate = F.linear(x, out_gate_weight, bias=None).sigmoid()

    # -----------------------------------------------------------------
    # 4. Fuse mask and gates (Triton kernel) if a mask is present
    # -----------------------------------------------------------------
    if not nomask:
        # Flatten tensors to 1‑D for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        # Mask is broadcast over the hidden dimension
        mask_f = mask.to(torch.float16).contiguous().view(-1)  # shape [B*N*N]

        total = left_f.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)

        _gate_mask_fuse_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            total,
            H=hidden_dim,
            BLOCK=BLOCK,
            HAS_MASK=1,
        )
        # Reshape back to original layout
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # Free gates that are no longer needed
    del left_gate, right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as a batched GEMM (left @ rightᵀ) in half precision.
    # -----------------------------------------------------------------
    # Bring hidden dimension into the batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.reshape(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.reshape(B * H, N, N) # (B*H, N, N)

    # Matrix multiplication with the second operand transposed
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore the [B, N, N, H] layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight, bias=None)

    # Return in the original float32 dtype
    return out.to(torch.float32)

gate and (optional) mask multiplication.  The expensive N³ contraction is
performed as a batched GEMM (left @ rightᵀ) using torch.bmm, which maps to
highly‑optimized cuBLAS kernels.  All heavy arithmetic is kept in
float16, only the final result is returned as float32.
"""

import triton.language as tl

# ----------------------------------------------------------------------

    right_ptr,         # half *
    left_gate_ptr,     # half *

    H: tl.constexpr,          # hidden dimension

):

    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

        right = right * right_gate
