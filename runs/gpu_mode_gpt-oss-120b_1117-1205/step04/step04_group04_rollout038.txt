@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Guard for out of bound M
    if pid_m >= M:
        return

    # Offsets
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for i/j
    i_valid = i_offset < N
    j_valid = j_offset < N

    # Pointer base for this batch
    base_a = a_ptr + pid_m * N * N
    base_b = b_ptr + pid_m * N * N
    base_c = c_ptr + pid_m * N * N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_valid = k_offset < N

        # Load A tile (i,k)
        a_ptrs = base_a + i_offset[:, None] * N + k_offset[None, :]
        a_mask = i_valid[:, None] & k_valid[None, :]
        a = tl.load(a_ptrs, mask=a_mask, other=tl.float16(0.0)).to(tl.float32)

        # Load B tile (k,j) -> B_T
        b_ptrs = base_b + k_offset[:, None] * 1 + j_offset[None, :] * N
        b_mask = k_valid[:, None] & j_valid[None, :]
        b = tl.load(b_ptrs, mask=b_mask, other=tl.float16(0.0)).to(tl.float32)

        # dot
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = base_c + i_offset[:, None] * N + j_offset[None, :]
    c_mask = i_valid[:, None] & j_valid[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)
```

"""
Fast outgoing TriMul implementation.

Algorithm
---------
1. Layer‑Norm the input (float32 → float16).
2. Linear projections to the hidden dimension.
3. Compute sigmoid gates and fuse them (element‑wise) with the projections.
4. (Optional) apply the pairwise mask.
5. Core O(N³·H) contraction is performed as a batched GEMM:
   for each hidden slice h (and batch b) compute
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented in a single Triton kernel that loops over k‑blocks.
6. Output LayerNorm, element‑wise gating with out_gate and final linear
   projection back to the original dimension.
All heavy arithmetic is done in float16 with fp32 accumulation,
final result returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_matmul_kernel(
    a_ptr,          # *half  (left)
    b_ptr,          # *half  (right)
    c_ptr,          # *half  (output)
    M,              # total number of (batch * hidden) matrices
    N,              # sequence length
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication:
    C[m,i,j] = Σ_k A[m,i,k] * B[m,j,k]   (B is accessed as Bᵀ)
    Shapes after flattening: (M, N, N)
    """
    pid_m = tl.program_id(0)  # batch*hidden
    pid_i = tl.program_id(1)  # output i‑block
    pid_j = tl.program_id(2)  # output j‑block

    # Guard against excess batches
    if pid_m >= M:
        return

    # Offsets for the output tile
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_valid = i_offset < N
    j_valid = j_offset < N

    # Base pointers for this batch*hidden slice
    a_base = a_ptr + pid_m * N * N
    b_base = b_ptr + pid_m * N * N
    c_base = c_ptr + pid_m * N * N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_valid = k_offset < N

        # Load A tile (i, k)
        a_ptrs = a_base + i_offset[:, None] * N + k_offset[None, :]
        a_mask = i_valid[:, None] & k_valid[None, :]
        a = tl.load(a_ptrs, mask=a_mask, other=tl.float16(0.0)).to(tl.float32)

        # Load B tile as Bᵀ (k, j)
        b_ptrs = b_base + k_offset[:, None] * 1 + j_offset[None, :] * N
        b_mask = k_valid[:, None] & j_valid[None, :]
        b = tl.load(b_ptrs, mask=b_mask, other=tl.float16(0.0)).to(tl.float32)

        # Accumulate
        acc += tl.dot(a, b)

    # Write back result
    c_ptrs = c_base + i_offset[:, None] * N + j_offset[None, :]
    c_mask = i_valid[:, None] & j_valid[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict mapping parameter names to float32 tensors
            - config: dict containing at least "dim" and "hidden_dim",
                      optionally "nomask" (bool)

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # Unpack
    inp, mask, weights, config = data
    device = inp.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 → half)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections (to hidden)
    # -----------------------------------------------------------------
    left_raw = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right_raw = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # -----------------------------------------------------------------
    # 3. Gates (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x, weights["left_gate.weight"].to(torch.float16))
    )
    right_gate = torch.sigmoid(
        F.linear(x, weights["right_gate.weight"].to(torch.float16))
    )
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"].to(torch.float16))
    )

    # -----------------------------------------------------------------
    # 4. Fuse gates (and optional mask) into left/right
    # -----------------------------------------------------------------
    left = left_raw * left_gate
    right = right_raw * right_gate

    if not nomask and mask is not None:
        # mask shape [B, N, N] → broadcast over hidden dim
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # Release unused tensors to save memory
    del left_raw, right_raw, left_gate, right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction via Triton batched GEMM
    # -----------------------------------------------------------------
    bs, N_seq, _, _ = left.shape  # left shape = [B, N, N, H]
    # Rearrange to (B, H, N, N) and flatten leading dims to M = B*H
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    M = bs * hidden_dim

    left_flat = left_perm.view(M, N_seq, N_seq)
    right_flat = right_perm.view(M, N_seq, N_seq)

    # Allocate output buffer
    out_flat = torch.empty_like(left_flat)

    # Triton block sizes (tuned for typical sequence lengths)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        M,
        triton.cdiv(N_seq, BLOCK_M),
        triton.cdiv(N_seq, BLOCK_N),
    )
    _tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M,
        N_seq,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = (
        out_flat.view(bs, hidden_dim, N_seq, N_seq)
        .permute(0, 2, 3, 1)
        .contiguous()
    )

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return as float32
    return out.to(torch.float32)
