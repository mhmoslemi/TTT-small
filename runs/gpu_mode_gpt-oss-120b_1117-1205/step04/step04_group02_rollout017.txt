def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', True)

    # 1. layerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights['norm.weight'],
        bias=weights['norm.bias']
    )
    x = x.to(torch.float16) # half

    # 2. Linear projections
    # Weights to half
    left = torch.nn.functional.linear(x, weights['left_proj.weight'].to(torch.float16), bias=None)
    right = torch.nn.functional.linear(x, weights['right_proj.weight'].to(torch.float16), bias=None)
    # 3. Gates
    left_gate = torch.nn.functional.linear(x, weights['left_gate.weight'].to(torch.float16), bias=None).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights['right_gate.weight'].to(torch.float16), bias=None).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights['out_gate.weight'].to(torch.float16), bias=None).sigmoid()

        N,

        stride_right_j,

        stride_out_i,
        stride_out_j,

        BLOCK_K=BLOCK_K,
    )

    # reshape out
    out_perm = out_flat.view(B, H, N, N)
    # permute back to original layout (B,N,N,H)
    out = out_perm.permute(0,2,3,1).contiguous()

    # 6. Output layer norm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights['to_out_norm.weight'].to(torch.float16),
        bias=weights['to_out_norm.bias'].to(torch.float16)
    )
    # gating
    out = out * out_gate
    # final linear
    out = torch.nn.functional.linear(
        out,
        weights['to_out.weight'].to(torch.float16),
        bias=None,
    )

    return out.to(torch.float32)

"""
Outgoing TriMul (AlphaFold3) forward pass.

The implementation follows the reference PyTorch code but moves the
element‑wise gate‑and‑mask fusion and the heavy N³ contraction into
custom Triton kernels.  The computation proceeds as:

1. Layer‑norm the input and cast to FP16.
2. Linear projections → left / right (FP16) and three sigmoid gates.
3. Fuse mask (if any) and the left/right gates in a single Triton kernel.
4. Compute the O(N³) update:
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   using a batched GEMM kernel where each (b,d) pair is an
   independent matrix multiplication.
5. Apply output LayerNorm, multiply by the out‑gate and a final linear
   projection back to the original dimension.
6. Return the result in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask + gates (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_fuse_kernel(
    left_ptr,          # * fp16
    right_ptr,         # * fp16
    left_gate_ptr,     # * fp16
    right_gate_ptr,    # * fp16
    mask_ptr,          # * fp16 (broadcasted)
    total_elements,    # i32
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    left  = tl.load(left_ptr + offs, mask=valid, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=valid, other=tl.float16(0.0))
    left_gate  = tl.load(left_gate_ptr + offs, mask=valid, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=tl.float16(0.0))

    if HAS_MASK:
        mask_idx = offs // HIDDEN_DIM          # broadcast mask over hidden dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=tl.float16(1.0))
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    tl.store(left_ptr  + offs, left,  mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """In‑place mask + gate fusion via Triton."""
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when HAS_MASK==0

    gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        HIDDEN_DIM=hidden_dim,
    )
    # The tensors are modified in‑place; nothing to return.


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication for the N³ contraction
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,          # * fp16   (batch, i, k)
    right_ptr,         # * fp16   (batch, j, k)   – we will load transposed
    out_ptr,           # * fp16   (batch, i, j)
    N,                 # i32, sequence length
    # strides for left
    stride_left_batch,   # i32
    stride_left_i,       # i32
    stride_left_k,       # i32
    # strides for right
    stride_right_batch,  # i32
    stride_right_j,      # i32
    stride_right_k,      # i32
    # strides for out
    stride_out_batch,    # i32
    stride_out_i,        # i32
    stride_out_j,        # i32
    # tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)               # batch (B * hidden_dim)
    pid_m = tl.program_id(0)               # i‑tile
    pid_n = tl.program_id(1)               # j‑tile

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A block   (batch, i, k)
        a_ptrs = (left_ptr
                  + pid_b * stride_left_batch
                  + offs_m[:, None] * stride_left_i
                  + offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load Bᵀ block (batch, k, j)  – we transpose on‑the‑fly
        b_ptrs = (right_ptr
                  + pid_b * stride_right_batch
                  + offs_k[:, None] * stride_right_k
                  + offs_n[None, :] * stride_right_j)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Store the result
    out_ptrs = (out_ptr
                + pid_b * stride_out_batch
                + offs_m[:, None] * stride_out_i
                + offs_n[None, :] * stride_out_j)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of torch.Tensors (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → FP16
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)   # now FP16

    # ------------------------------------------------------------------
    # 2. Linear projections (FP16)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Compute gates (FP16)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) with gates via Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).contiguous().view(-1)   # [B*N*N]

        # flatten tensors for in‑place kernel
        left_f      = left.contiguous().view(-1)
        right_f     = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_h,
            hidden_dim,
        )
        # reshape back
        left  = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # Simple element‑wise gating
        left  = left * left_gate
        right = right * right_gate

    # Release temporary tensors
    del left_gate, right_gate, x

    # ------------------------------------------------------------------
    # 5. Core N³ contraction using a batched GEMM Triton kernel
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    # Move hidden dimension before the two sequence axes for convenient batching
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    batch_total = B * H

    # Flatten batch dimension => (batch_total, N, N)
    left_flat  = left_perm.view(batch_total, N, N)
    right_flat = right_perm.view(batch_total, N, N)

    # Allocate output tensor (same dtype and device)
    out_flat = torch.empty_like(left_flat)

    # Tile sizes – chosen to fit well on B200 GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Strides in number of elements (not bytes)
    stride_left_batch, stride_left_i, stride_left_k = left_flat.stride()
    stride_right_batch, stride_right_j, stride_right_k = right_flat.stride()
    stride_out_batch, stride_out_i, stride_out_j = out_flat.stride()

    # Launch kernel
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch_total,
    )
    trimul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_left_batch,
        stride_left_i,
        stride_left_k,
        stride_right_batch,
        stride_right_j,
        stride_right_k,
        stride_out_batch,
        stride_out_i,
        stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, H, N, N) then to (B, N, N, H)
    out_perm = out_flat.view(B, H, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in original dtype (float32)
    return out.to(torch.float32)
