"""
TriMul (outgoing) forward implementation.

Algorithm:
1. Layer‑normalize the input tensor.
2. Linear projections (left/right) and gate linear layers are computed in FP16.
3. A tiny Triton kernel fuses the per‑element mask (if present) with the
   left/right gates:   left = left * left_gate * mask,
                       right = right * right_gate * mask.
4. The O(N³) contraction is a batched matrix multiplication:
        out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   which equals left_{b,d} @ right_{b,d}ᵀ for each hidden channel d.
5. Output LayerNorm, out‑gate scaling and final linear projection restore the
   original channel dimension.
All heavy element‑wise work is fused in Triton; the contraction uses
torch.bmm (cuBLAS) for peak performance.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # float16*
    right_ptr,           # float16*
    left_gate_ptr,       # float16*
    right_gate_ptr,      # float16*
    mask_ptr,            # float16* (broadcasted, ignored when HAS_MASK=0)
    total_elements,      # int32
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask is present
    H: tl.constexpr,          # hidden dimension size
):
    """In‑place: left = left * left_gate * mask; right = right * right_gate * mask."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    # Load data
    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # mask is broadcast over hidden dim: each hidden entry shares the same mask entry.
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store results back (in‑place)
    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    Fuse left/right with their gates and optional mask using the Triton kernel.
    All tensors must be contiguous FP16 views.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when HAS_MASK==0

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # In‑place operation – nothing to return.


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric) – may be ignored
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → FP16 for subsequent ops
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (FP16)
    # ------------------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gates (FP16) with sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) with gates via Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → FP16, flattened
        mask_h = mask.to(torch.float16).contiguous().view(-1)  # shape B*N*N

        # Flatten tensors for kernel launch
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_h,
            hidden_dim,
        )
        # Reshape back to original layout (views share storage)
        left = left_f.view_as(left)
        right = right_f.view_as(right)
        # Release temporaries
        del left_f, right_f, left_gate_f, right_gate_f, mask_h
    else:
        # No mask: simple element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate

    # Gates are no longer needed
    del left_gate, right_gate, x_h

    # ------------------------------------------------------------------
    # 5. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as batched GEMM: L_{b,d} @ R_{b,d}ᵀ
    # ------------------------------------------------------------------
    # Bring hidden dimension to second axis for efficient batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    # Collapse batch and hidden dims
    left_mat = left_perm.view(B * H, N, N)    # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)  # (B*H, N, N)

    # Batched matrix multiplication: L @ Rᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore layout to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise gating (FP16)

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in float32 as required by the interface
    return out.to(torch.float32)
