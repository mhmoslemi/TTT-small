def _gate_mask_fuse_kernel(left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr, total_elements, hidden_dim, HAS_MASK: tl.constexpr, BLOCK: tl.constexpr):

    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

        right = right * right_gate

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

Algorithm
---------
1. Input LayerNorm (float32) → cast to fp16.
2. Linear projections to hidden dimension (fp16):
   left = x @ W_left , right = x @ W_right.
3. Compute three gate projections (fp16) and apply sigmoid.
4. Fuse mask (if present) and gate multiplications into a single
   Triton kernel that updates `left` and `right` in‑place:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
5. Core contraction:
   out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   Realised as a batched GEMM over the (B·hidden_dim) batch dimension.
6. Output LayerNorm (fp16) → multiply by out_gate → final linear projection
   back to the original dimension (fp16 → fp32).

The only Triton code is the gate+mask fusion kernel; the heavy N³
contraction leverages Torch's highly‑optimised batched GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half*   (flattened left tensor)
    right_ptr,         # half*   (flattened right tensor)
    left_gate_ptr,     # half*   (flattened left gate)
    right_gate_ptr,    # half*   (flattened right gate)
    mask_ptr,          # half*   (flattened mask, broadcasted over hidden dim)
    total_elements,    # i32
    hidden_dim,        # i32
    HAS_MASK: tl.constexpr,   # 1 if mask present else 0
    BLOCK: tl.constexpr,      # block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # mask is broadcast over the hidden dimension:
        # each hidden entry shares the same mask entry.
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    `mask` may be ``None`` (no masking). All tensors must be contiguous
    half‑precision tensors.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    # Provide a dummy pointer when mask is absent (kernel does not read it)
    mask_ptr = mask if mask is not None else left

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        hidden_dim,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )
    # No Python return – operation performed in‑place


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (half)
    # ------------------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gate projections (half) + sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) with gates via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # Ensure mask is half‑precision and contiguous
        mask_h = mask.to(torch.float16).contiguous()
        # Flatten tensors for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask_h.view(-1)  # [B*N*N]

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            hidden_dim,
        )
        # Reshape back to original layout
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask – simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # Gates no longer needed
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as batched GEMM (B*H, N, N) @ (B*H, N, N)^T
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)  # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm → out_gate scaling → final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise gating (half)

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in original float32 dtype
    return out.to(torch.float32)
