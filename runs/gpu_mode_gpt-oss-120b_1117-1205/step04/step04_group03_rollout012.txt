```
@triton.jit
def _matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)  # block row
    pid_n = tl.program_id(2)  # block col

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute per batch pointer offsets
    a_batch_ptr = a_ptr + batch_id * batch_stride_a
    b_batch_ptr = b_ptr + batch_id * batch_stride_b
    c_batch_ptr = c_ptr + batch_id * batch_stride_c

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # compute global offset for K dimension
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # load A tile
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0)

        # load B tile (B is KxN)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0)

        # compute dot product
        acc += tl.dot(a, b)

    # write back C
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)
```

def _matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # pointers for this batch
    a_batch_ptr = a_ptr + batch * batch_stride_a
    b_batch_ptr = b_ptr + batch * batch_stride_b
    c_batch_ptr = c_ptr + batch * batch_stride_c

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):

"""
TriMul (outgoing) forward pass with Triton acceleration.

Algorithm
---------
1. LayerNorm the input (float32) → FP16.
2. Linear projections to hidden dimension (FP16).
3. Compute three gate tensors (FP16) + sigmoid.
4. Fuse mask (if present) with the left/right gates using a small Triton kernel.
5. Core contraction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched GEMM (left @ rightᵀ) for each hidden channel.
   Implemented with a custom Triton GEMM kernel that processes the
   (batch * hidden_dim) independent matrix multiplies in one launch.
6. Output LayerNorm, element‑wise out‑gate, final linear projection.
7. Return result in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Kernel 1: fuse mask and per‑channel gates into left/right tensors.
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # fp16*  (flattened left tensor)
    right_ptr,         # fp16*  (flattened right tensor)
    left_gate_ptr,     # fp16*  (flattened left_gate)
    right_gate_ptr,    # fp16*  (flattened right_gate)
    mask_ptr,          # fp16*  (flattened mask, length = B*N*N)
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask supplied, else 0
    H: tl.constexpr,          # hidden dimension
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each H elements share the same mask entry
        mask_idx = offs // H
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * m
        right = right * right_gate * m
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of mask (if any) and gates using Triton.
    All tensors are contiguous FP16 with shape [B, N, N, hidden_dim].
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when absent

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )


# ----------------------------------------------------------------------
# Kernel 2: batched GEMM (FP16×FP16→FP16) for the core contraction.
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers (batch, M, K) / (batch, K, N) / (batch, M, N)
    M, N, K,                      # dimensions
    stride_am, stride_ak,         # A strides (row, col)
    stride_bk, stride_bn,         # B strides (row, col) where B is (K, N)
    stride_cm, stride_cn,         # C strides (row, col)
    batch_stride_a, batch_stride_b, batch_stride_c,  # strides between batch instances
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    a_batch = a_ptr + batch * batch_stride_a
    b_batch = b_ptr + batch * batch_stride_b
    c_batch = c_ptr + batch * batch_stride_c

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(
            a_batch
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=a_mask,
            other=0.0,
        )
        b = tl.load(
            b_batch
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=b_mask,
            other=0.0,
        )
        acc += tl.dot(a, b)

    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(
        c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn,
        acc.to(tl.float16),
        mask=c_mask,
    )


def _batched_gemm_triton(left, right):
    """
    Perform the contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a single Triton kernel.
    left/right: FP16 tensors of shape [B, N, N, H]
    Returns: FP16 tensor of same shape.
    """
    B, N, _, H = left.shape
    # (B, H, N, N) layout, then flatten batch*head
    left_perm = left.permute(0, 3, 1, 2).contiguous()      # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right_t = right_perm.transpose(2, 3).contiguous()     # (B, H, N, N) with K,J order

    left_mat = left_perm.view(B * H, N, N)   # (B*H, M=N, K=N)
    right_mat = right_t.view(B * H, N, N)    # (B*H, K=N, N=N)
    out_mat = torch.empty_like(left_mat)    # (B*H, M, N)

    # strides (row‑major)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)
    stride_bk = right_mat.stride(1)
    stride_bn = right_mat.stride(2)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)

    batch_stride_a = left_mat.stride(0)
    batch_stride_b = right_mat.stride(0)
    batch_stride_c = out_mat.stride(0)

    M = N
    K = N
    Ndim = N

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * H, triton.cdiv(M, BLOCK_M), triton.cdiv(Ndim, BLOCK_N))
    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M,
        Ndim,
        K,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        batch_stride_a,
        batch_stride_b,
        batch_stride_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul forward pass (outgoing) using Triton for mask‑gate fusion and
    the O(N³) contraction.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            input_tensor: torch.Tensor [B, N, N, C] (float32)
            mask: torch.Tensor [B, N, N] (bool / float) – may be ignored if nomask=True
            weights_dict: model parameters as float32 tensors
            config_dict: contains "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → FP16
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half)
    # ------------------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gates (half) and sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) with gates via Triton
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask.to(torch.float16).contiguous().view(-1)  # [B*N*N]
        # flatten tensors for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_h,
            hidden_dim,
        )
        # reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask – simple element‑wise gating
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton GEMM
    # ------------------------------------------------------------------
    out = _batched_gemm_triton(left, right)      # shape [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate scaling and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in original dtype
    return out.to(torch.float32)
