"""
Outgoing TriMul forward (AlphaFold3).

Algorithm
---------
1. LayerNorm on the input tensor (float32) → cast to float16.
2. Two linear projections (left/right) to the hidden dimension (float16).
3. Three gate projections (left, right, out) with sigmoid (float16).
4. Fuse left/right projections with their gates and an optional mask in‑place
   using a small Triton kernel (no extra memory allocation).
5. Core contraction:
       out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   implemented as a batched GEMM (L @ Rᵀ) per hidden channel.
6. Output LayerNorm, out‑gate scaling and final linear projection.
   Result is returned as float32.

The Triton kernel only handles the element‑wise gate‑mask fusion; the heavy
O(N³) contraction leverages cuBLAS via torch.bmm on half‑precision tensors.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,                     # float16*
    right_ptr,                    # float16*
    left_gate_ptr,                # float16*
    right_gate_ptr,               # float16*
    mask_ptr,                     # float16* (broadcasted)
    total_elements: tl.int32,     # total number of elements in left/right
    BLOCK: tl.constexpr,          # threads per block
    HAS_MASK: tl.constexpr,       # 1 if mask is present
    H: tl.constexpr,              # hidden dimension size
):
    """In‑place: left = left * left_gate * mask; right = right * right_gate * mask."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    # load operands
    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # mask is broadcast over the hidden dimension:
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def _fuse_left_right_gate_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    Fuse gates (and optional mask) into `left` and `right` in‑place.
    All tensors must be contiguous and of dtype torch.float16.
    """
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when HAS_MASK==0

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – optional
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)   # keep float32 for layer_norm
    norm_bias = weights["norm.bias"].to(device)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-6,
    )
    x = x.to(torch.float16)  # half for the rest of the pipeline

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (half)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16, device)
    right_proj_w = weights["right_proj.weight"].to(torch.float16, device)

    left = F.linear(x, left_proj_w, bias=None)   # [B, N, N, hidden]
    right = F.linear(x, right_proj_w, bias=None)

    # ------------------------------------------------------------------
    # 3. Gates (half) and optional mask fusion via Triton
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16, device)
    right_gate_w = weights["right_gate.weight"].to(torch.float16, device)
    out_gate_w = weights["out_gate.weight"].to(torch.float16, device)

    left_gate = F.linear(x, left_gate_w, bias=None).sigmoid()
    right_gate = F.linear(x, right_gate_w, bias=None).sigmoid()
    out_gate = F.linear(x, out_gate_w, bias=None).sigmoid()

    if not nomask:
        # mask: [B, N, N] → half and flatten
        mask_h = mask.to(torch.float16, device).contiguous().view(-1)

        # flatten everything for the Triton kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)

        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:

        left = left * left_gate

    del left_gate, right_gate, x

    # bring hidden dimension to the second axis
    left_perm = left.permute(0, 3, 1, 2)   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2) # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    # view as (B*H, N, N)
    left_mat = left_perm.reshape(B * H, N, N)
    right_mat = right_perm.reshape(B * H, N, N)

    # batched matrix multiplication with transposition on the second operand
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, out‑gate scaling and final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16, device)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16, device)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-6,
    )
    out = out * out_gate  # element‑wise gating (half)

    to_out_w = weights["to_out.weight"].to(torch.float16, device)
    out = F.linear(out, to_out_w, bias=None)

    # return as float32
    return out.to(torch.float32)
