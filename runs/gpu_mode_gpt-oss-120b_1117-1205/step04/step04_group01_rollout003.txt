"""
TriMul “outgoing” forward pass (alpha version).

Algorithm
---------
1. Input LayerNorm (float32 → float16).
2. Linear projections `left` and `right` (float16).
3. Compute three gates (`left_gate`, `right_gate`, `out_gate`) via linear + sigmoid.
4. Fuse mask (if present) and the two gates into `left` and `right` with a
   Triton kernel (`_gate_mask_fuse_kernel`).  This performs an in‑place
   element‑wise multiplication:  left = left * left_gate * mask,
   right = right * right_gate * mask.
5. Core contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
   This is a batched matrix multiplication per channel:
   for each d we compute left_d @ right_d.T.
   We implement it with torch.bmm on a reshaped view
   (B·hidden_dim, N, N), which lets cuBLAS handle the O(N³) work.
6. Output LayerNorm, multiply by `out_gate`, final linear projection back to `dim`.
7. Return float32 tensor.
"""

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half precision)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )                     # [B, N, N, hidden_dim]
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gates (half precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) and gates via Triton
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    if not nomask and mask is not None:
        # mask shape: [B, N, N] → half, no extra dim; kernel broadcasts over hidden_dim
        mask_h = mask.to(torch.float16)
        _fuse_gate_and_mask(left, right, left_gate, right_gate, mask_h, hidden_dim)
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched GEMM: for each d, out_d = left_d @ right_d.T
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # shape is [B, N, N, hidden_dim]

    # Rearrange tensors to (B*hidden_dim, N, N) for torch.bmm
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # [B, hidden_dim, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()        # [B, hidden_dim, N, N] (k,j)

    left_flat = left_perm.view(B * hidden_dim, N, N)          # [B*H, N, N]
    right_flat = right_perm.view(B * hidden_dim, N, N)        # [B*H, N, N]

    out_flat = torch.bmm(left_flat, right_flat)               # [B*H, N, N]

    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in float32 as required
    return out.to(torch.float32)
