def custom_kernel(data):
    input_tensor, mask, weights, config = data

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # LayerNorm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # float32

    # Convert to half
    x_h = x.to(torch.float16)

    # Linear projections
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16), bias=None)

    # Gates
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # Apply mask and gating
    if not config.get("nomask", True) and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * left_gate * mask_h
        right = right * right_gate * mask_h
    else:
        left = left * left_gate
        right = right * right_gate

    # Rearrange for matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Flatten batch and head dimensions
    BH = bs * hidden_dim
    left_flat = left_perm.view(BH, seq_len, seq_len)
    right_flat = right_perm.view(BH, seq_len, seq_len)
    # allocate output
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # Get strides
    stride_l_batch = left_flat.stride(0)
    stride_l_i = left_flat.stride(1)
    stride_l_k = left_flat.stride(2)

    stride_r_batch = right_flat.stride(0)
    stride_r_i = right_flat.stride(1)
    stride_r_k = right_flat.stride(2)

    stride_o_batch = out_flat.stride(0)
    stride_o_i = out_flat.stride(1)
    stride_o_j = out_flat.stride(2)

    # Triton matmul kernel
    import triton
    import triton.language as tl

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (triton.cdiv(seq_len, BLOCK_M), triton.cdiv(seq_len, BLOCK_N), BH)

        left_flat,
        right_flat,
        out_flat,
        seq_len,
        stride_l_batch, stride_l_i, stride_l_k,
        stride_r_batch, stride_r_i, stride_r_k,
        stride_o_batch, stride_o_i, stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back
    out_perm = out_flat.view(bs, hidden_dim, seq_len, seq_len)  # (B, H, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # Output layer norm and linear
    out = F.layer_norm(

        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )
    # Return float32
    return out.to(torch.float32)

"""
Optimized outgoing TriMul operator (AlphaFold3).

The implementation follows the reference PyTorch module but moves the
computationally heavy batch matrix multiplication
    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
to a custom Triton kernel.  The kernel computes a batched GEMM
(L × Rᵀ) for each hidden channel (treated as an independent batch
dimension).  All other operations (LayerNorm, linear projections,
gates, optional mask) are performed in PyTorch with half‑precision.
The final output is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_matmul_kernel(
    left_ptr,          # * half, shape (B*H, N, N)
    right_ptr,         # * half, shape (B*H, N, N)
    out_ptr,           # * half, shape (B*H, N, N)
    N,                 # int32, sequence length
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_i, stride_r_k,
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: out = left @ rightᵀ for each batch‑head."""
    pid_m = tl.program_id(0)          # block index in i (row)
    pid_n = tl.program_id(1)          # block index in j (col)
    pid_bh = tl.program_id(2)         # combined batch * hidden index

    # Offsets within the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for out‑of‑bounds rows / cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointer for this batch‑head
    batch_offset = pid_bh * stride_l_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Masks for the K dimension
        mask_k = offs_k < N

        # Load left block (M × K)
        a_ptrs = left_ptr + batch_offset + \
                 offs_m[:, None] * stride_l_i + \
                 offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=tl.float16(0.0))

        # Load right block (K × N) – note the transposed layout
        b_ptrs = right_ptr + batch_offset + \
                 offs_k[:, None] * stride_r_k + \
                 offs_n[None, :] * stride_r_i
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=tl.float16(0.0))

        # Dot‑product accumulation
        acc += tl.dot(a, b)   # a: (M,K), b: (K,N) → (M,N)

    # Write the result back (cast to FP16)
    out_ptrs = out_ptr + batch_offset + \
               offs_m[:, None] * stride_o_i + \
               offs_n[None, :] * stride_o_j
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape [B, N, N, C] (float32)
            - mask: torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights: dict of torch.Tensors (float32) with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → half precision
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # still float32
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half)
    # ------------------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gating (half)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Apply mask + gates
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # broadcast mask over the hidden dimension
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * left_gate * mask_h
        right = right * right_gate * mask_h
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton batched GEMM
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # N = seq_len, hidden_dim = hidden_dim
    # Rearrange to (B, H, N, N) and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Flatten batch and head dimensions → (B*H, N, N)
    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # Strides (all tensors are contiguous)
    stride_l_batch = left_flat.stride(0)
    stride_l_i = left_flat.stride(1)
    stride_l_k = left_flat.stride(2)

    stride_r_batch = right_flat.stride(0)
    stride_r_i = right_flat.stride(1)
    stride_r_k = right_flat.stride(2)

    stride_o_batch = out_flat.stride(0)
    stride_o_i = out_flat.stride(1)
    stride_o_j = out_flat.stride(2)

    # Triton launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        BH,
    )

    trimul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_l_batch, stride_l_i, stride_l_k,
        stride_r_batch, stride_r_i, stride_r_k,
        stride_o_batch, stride_o_i, stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, H, N, N) and transpose to (B, N, N, H)
    out_perm = out_flat.view(B, hidden_dim, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate & final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in float32
    return out.to(torch.float32)
