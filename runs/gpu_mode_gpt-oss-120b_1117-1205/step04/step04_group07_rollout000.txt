"""
Optimized outgoing TriMul forward pass.

Algorithm
---------
1. Layer‑norm the input tensor (float32 → float16).
2. Linear projections to the hidden dimension (float16).
3. Compute sigmoid gates (float16).
4. Fuse mask (if present) with the gates using a Triton kernel.
5. Core O(N³) contraction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM (torch.bmm) after reshaping
   left and right to (B*H, N, N) where H = hidden_dim.
6. Output LayerNorm, out‑gate scaling and final linear projection.
7. Return the result in float32.

Only the mask‑gate fusion is performed in Triton; the heavy tensor
contraction leverages cuBLAS via torch.bmm, which is already highly
optimized on the target B200 GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel: fuse mask (if any) and element‑wise gating for left/right
# -------------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # * half   (flattened left tensor)
    right_ptr,         # * half   (flattened right tensor)
    left_gate_ptr,     # * half   (flattened left_gate)
    right_gate_ptr,    # * half   (flattened right_gate)
    mask_ptr,          # * half   (flattened mask, length B*N*N)
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    # Load tensors
    left      = tl.load(left_ptr      + offs, mask=active, other=tl.float16(0.0))
    right     = tl.load(right_ptr     + offs, mask=active, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=tl.float16(0.0))
    right_gate= tl.load(right_gate_ptr+ offs, mask=active, other=tl.float16(0.0))

    if HAS_MASK:
        # mask is broadcasted across the hidden dimension
        mask_idx = offs // HIDDEN_DIM        # integer division
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=tl.float16(1.0))
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    # Store results in‑place
    tl.store(left_ptr  + offs, left,  mask=active)
    tl.store(right_ptr + offs, right, mask=active)

def _apply_gate_mask_fuse(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of mask and gates using Triton.
    All tensors must be contiguous half‑precision.
    mask may be None (no masking).
    """
    total = left.numel()
    BLOCK = 32768                         # multiple of 32 for good occupancy
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when no mask

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        HIDDEN_DIM=hidden_dim,
    )
    # tensors are modified in‑place; nothing to return

# -------------------------------------------------------------------------
# Entry point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights     : dict with the following keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config      : dict containing at least "dim", "hidden_dim",
                            optional "nomask" (bool, default True)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input layer‑norm (float32 → float16)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections to hidden_dim (float16)
    # -----------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # -----------------------------------------------------------------
    # 3. Gating values (float16, sigmoid)
    # -----------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # -----------------------------------------------------------------
    # 4. Fuse mask with gates (if mask is present) using Triton
    # -----------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → flatten to [B*N*N]
        mask_f = mask.to(torch.float16).contiguous().view(-1)
        # flatten all tensors to 1‑D views (contiguous)
        left_f      = left.contiguous().view(-1)
        right_f     = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f= right_gate.contiguous().view(-1)

        _apply_gate_mask_fuse(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            hidden_dim,
        )
        # reshape back
        left  = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # Simple element‑wise multiplication when no mask
        left  = left  * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as a batched GEMM (torch.bmm)
    # -----------------------------------------------------------------
    # Move hidden_dim to the second axis for batched GEMM
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    # Reshape to (B*H, N, N)
    left_mat  = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Batched matrix multiplication: left_mat @ right_mat^T  → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output layer‑norm, out‑gate scaling, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in float32 as required by the spec
    return out.to(torch.float32)
