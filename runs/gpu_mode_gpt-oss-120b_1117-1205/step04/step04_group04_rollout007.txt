"""
custom_kernel for the outgoing TriMul operator (AlphaFold3).

Algorithm
---------
1. Layer‑norm the input tensor (float32 → float16).
2. Linear projections to the hidden dimension (no bias).
3. Compute three gate tensors (sigmoid).
4. Fuse mask (if present) and the left/right gate multiplications
   in‑place with a Triton kernel.
5. Core contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a single batched GEMM over (B·H) matrices.
6. Output LayerNorm → element‑wise gating with out_gate →
   final linear projection back to the original feature dimension.
7. Return the result as float32.

At least one operation (mask & gate fusion) is executed on the GPU via
a Triton kernel, satisfying the requirement to use Triton while keeping
the rest of the pipeline in PyTorch for simplicity and performance.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # *fp16, shape (total,)
    right_ptr,         # *fp16, shape (total,)
    left_gate_ptr,     # *fp16, shape (total,)
    right_gate_ptr,    # *fp16, shape (total,)
    mask_ptr,          # *fp16, shape (B*N*N,) – broadcasted over hidden dim
    total_elems,       # i32, total number of elements in left/right
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,   # hidden dimension size
):
    """In‑place: left = left * left_gate * mask, right = right * right_gate * mask."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elems

    # Load tensors
    left = tl.load(left_ptr + offs, mask=m, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=m, other=tl.float16(0.0))
    lgate = tl.load(left_gate_ptr + offs, mask=m, other=tl.float16(0.0))
    rgate = tl.load(right_gate_ptr + offs, mask=m, other=tl.float16(0.0))

    if HAS_MASK:
        # Each hidden slice shares the same mask value.
        mask_idx = offs // H            # integer division, broadcast over H
        mask_val = tl.load(mask_ptr + mask_idx, mask=m, other=tl.float16(0.0))
        left = left * lgate * mask_val
        right = right * rgate * mask_val
    else:
        left = left * lgate
        right = right * rgate

    tl.store(left_ptr + offs, left, mask=m)
    tl.store(right_ptr + offs, right, mask=m)


def _apply_mask_and_gate(left, right, left_gate, right_gate, mask):
    """
    Fuse mask (if present) and gate tensors into left/right in‑place.
    All tensors are expected to be contiguous fp16 tensors.
    """
    total = left.numel()
    # Large block size reduces launch overhead; adjust if needed.
    BLOCK = 65536
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when mask unused

    _mask_gate_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=left.shape[-1],
    )
    # In‑place modification; nothing to return.


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (float32)
            - config: dict containing "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    inp, mask, weights, config = data
    device = inp.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> float16)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float16, device=device)
    norm_bias = weights["norm.bias"].to(torch.float16, device=device)

    x = F.layer_norm(
        inp,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    ).to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections to hidden dimension (bias=False)
    # --------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"].to(torch.float16, device=device)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16, device=device)

    left = F.linear(x, left_proj_weight)   # (B, N, N, H)
    right = F.linear(x, right_proj_weight)

    # --------------------------------------------------------------
    # 3. Gate tensors (sigmoid)
    # --------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"].to(torch.float16, device=device)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16, device=device)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16, device=device)

    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    # --------------------------------------------------------------
    # 4. Fuse mask and gates (if needed) via Triton
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] -> [B, N, N, 1] -> flatten to [B*N*N]
        mask_fp16 = mask.to(torch.float16, device=device).unsqueeze(-1).contiguous()
        mask_flat = mask_fp16.view(-1)  # (B*N*N,)
        _apply_mask_and_gate(
            left.view(-1), right.view(-1),
            left_gate.view(-1), right_gate.view(-1),
            mask_flat,
        )
        # Reshape back
        left = left.view_as(left)
        right = right.view_as(right)
    else:
        # Simple element‑wise multiplication when mask is absent
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction: batched GEMM over (B·H) matrices
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # Move hidden dim to the batch axis and make memory contiguous
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_mat = left_t.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_t.view(B * H, N, N) # (B*H, N, N)

    # Batched matrix multiplication: (i,k) × (j,k)^T → (i,j)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection
    # --------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16, device=device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16, device=device)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    out = out * out_gate

    to_out_weight = weights["to_out.weight"].to(torch.float16, device=device)
    out = F.linear(out, to_out_weight)  # (B, N, N, dim), bias=None

    # --------------------------------------------------------------
    # 7. Cast back to float32 and return
    # --------------------------------------------------------------
    return out.to(torch.float32)
