def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', True)

    # Input LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'])
    # Convert to half
    x_h = x.half()

    # Projections
    left = torch.nn.functional.linear(x_h, weights['left_proj.weight'].half())
    right = torch.nn.functional.linear(x_h, weights['right_proj.weight'].half())

    # Gating
    left_gate = torch.nn.functional.linear(x_h, weights['left_gate.weight'].half()).sigmoid()
    right_gate = torch.nn.functional.linear(x_h, weights['right_gate.weight'].half()).sigmoid()
    out_gate = torch.nn.functional.linear(x_h, weights['out_gate.weight'].half()).sigmoid()

    if not nomask:
        mask_h = mask.half().unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    # Apply gates (elementwise)
    left = left * left_gate
    right = right * right_gate

    # Compute contraction using batched GEMM:
    # reshape to (B*H,N,N)
    B, N, _, _ = input_tensor.shape  # dims: B N N dim
    # left: shape [B,N,N,H]
    left = left.permute(0,3,1,2).contiguous()   # [B,H,N,N]
    right = right.permute(0,3,1,2).contiguous() # [B,H,N,N]
    left_mat = left.view(B*hidden_dim, N, N)
    right_mat = right.view(B*hidden_dim, N, N)

    # GEMM: left_mat * right_mat^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))

    # reshape back to [B,N,N,H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()  # [B,N,N,H]

    # Apply out_gate using Triton kernel
    out = elementwise_mul_kernel(out, out_gate) # returns scaled out

    # Output layernorm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights['to_out_norm.weight'].half(),
        bias=weights['to_out_norm.bias'].half(),
    )
    # final linear
    out = torch.nn.functional.linear(out, weights['to_out.weight'].half())
    return out.float()

"""
TriMul (outgoing) forward pass.

The implementation follows the reference PyTorch version but performs the
expensive O(N³) contraction with a batched GEMM (torch.bmm) and uses a small
Triton kernel only for the final element‑wise multiplication with the
output gate. All linear projections, layer‑norms and sigmoid gates are kept in
half precision for speed, and the final result is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _elem_mul_kernel(
    a_ptr,          # * half
    b_ptr,          # * half
    c_ptr,          # * half (output, can be same as a_ptr)
    n_elements: tl.int32,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < n_elements

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(c_ptr + offs, a * b, mask=mask)


def _elem_mul_inplace(out: torch.Tensor, other: torch.Tensor) -> None:
    """In‑place element‑wise multiplication out = out * other using Triton."""
    assert out.is_contiguous() and other.is_contiguous()
    total = out.numel()
    BLOCK = 65536  # 2^16, good trade‑off for large tensors
    grid = (triton.cdiv(total, BLOCK),)
    _elem_mul_kernel[grid](out, other, out, total, BLOCK=BLOCK)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor of shape [B, N, N] (bool or float)
            - weights: dict with the following keys (all torch.Tensor on CUDA):
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict containing at least "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack arguments
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (float32 -> half)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias = weights["norm.bias"].to(torch.float16)
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)
    x_h = x.half()  # [B, N, N, dim] in fp16

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (fp16)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x_h, right_proj_w)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Gating values (fp16, sigmoid)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_h, left_gate_w).sigmoid()
    right_gate = F.linear(x_h, right_gate_w).sigmoid()
    out_gate = F.linear(x_h, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 4. Optional mask application (broadcasted over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # Apply gates (element‑wise, fp16)
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as a batched GEMM.
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape  # B, N, N, dim
    # Rearrange to [B, hidden_dim, N, N] and make contiguous for view
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left_mat = left_perm.view(B * hidden_dim, N, N)    # (B*H, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)  # (B*H, N, N)

    # Batched matrix multiplication: (B*H, N, N) @ (B*H, N, N)^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (fp16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    # ------------------------------------------------------------------
    # 7. Multiply by out_gate using a tiny Triton kernel
    # ------------------------------------------------------------------
    # Ensure tensors are contiguous fp16
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    _elem_mul_inplace(out, out_gate)   # out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to dim (fp16)
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)   # [B, N, N, dim] in fp16

    # Return float32 tensor
    return out.float()
