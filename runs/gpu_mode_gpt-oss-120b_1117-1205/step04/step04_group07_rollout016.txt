"""
TriMul (outgoing) forward pass implemented with mixed‑precision PyTorch
and a small Triton kernel for the out‑gate scaling.

Algorithm:
1. Layer‑norm the input (FP16).
2. Linear projections → left/right (FP16).
3. Compute sigmoid gates (FP16).
4. Apply optional mask and element‑wise gate multiplication (pure PyTorch).
5. Contract left/right via a batched GEMM:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   (implemented with torch.bmm on reshaped tensors).
6. Output LayerNorm (FP16).
7. Scale by out_gate using a Triton kernel (in‑place FP16 multiply).
8. Final linear projection back to `dim` (FP16) and cast to FP32.

A Triton kernel is required (step 7) but the heavy O(N³) work is
performed by cuBLAS via torch.bmm for best performance.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: in‑place element‑wise multiplication of out *= out_gate
# ----------------------------------------------------------------------
@triton.jit
def _out_gate_kernel(
    out_ptr,        # * half
    gate_ptr,       # * half
    total_elements: tl.int32,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    out_val = tl.load(out_ptr + offs, mask=mask, other=0.0)
    gate_val = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    out_val = out_val * gate_val
    tl.store(out_ptr + offs, out_val, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict: contains the following keys (all float32):
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict: must contain "dim", "hidden_dim"; optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # Cast model weights to FP16 once (keeps them on GPU)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device=device, dtype=torch.float16)
    norm_bias = weights["norm.bias"].to(device=device, dtype=torch.float16)
    left_proj_weight = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_weight = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (FP16)
    # --------------------------------------------------------------
    x = input_tensor.to(dtype=torch.float16)
    x = F.layer_norm(x,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # --------------------------------------------------------------
    # 2. Linear projections to hidden dimension
    # --------------------------------------------------------------
    left = F.linear(x, left_proj_weight, bias=None)      # [B, N, N, H]
    right = F.linear(x, right_proj_weight, bias=None)   # [B, N, N, H]

    # --------------------------------------------------------------
    # 3. Gating values (sigmoid, FP16)
    # --------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight, bias=None).sigmoid()
    right_gate = F.linear(x, right_gate_weight, bias=None).sigmoid()
    out_gate = F.linear(x, out_gate_weight, bias=None).sigmoid()

    # --------------------------------------------------------------
    # 4. Optional mask + gate fusion (pure PyTorch for simplicity)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] -> broadcast over hidden dim
        mask_f = mask.to(dtype=torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    left = left * left_gate
    right = right * right_gate

    # Release unused tensors early
    del left_gate, right_gate, x

    # --------------------------------------------------------------
    # 5. Core O(N³) contraction via batched GEMM
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    # Rearrange to (B, H, N, N) for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    # Collapse batch & head dimensions
    left_mat = left_perm.view(B * H, N, N)               # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)             # (B*H, N, N)

    # Batched matrix multiplication (left @ right^T)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm (FP16)
    # --------------------------------------------------------------
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    # --------------------------------------------------------------
    # 7. Scale by out_gate using a Triton kernel (in‑place FP16)
    # --------------------------------------------------------------
    out_flat = out.view(-1)
    out_gate_flat = out_gate.view(-1)
    total = out_flat.numel()
    BLOCK = 32768  # must be a multiple of 32
    grid = (triton.cdiv(total, BLOCK),)
    _out_gate_kernel[grid](
        out_flat,
        out_gate_flat,
        total,
        BLOCK=BLOCK,
    )
    # `out` is now scaled in‑place

    # --------------------------------------------------------------
    # 8. Final linear projection back to `dim`
    # --------------------------------------------------------------
    out = F.linear(out, to_out_weight, bias=None)

    # Cast back to FP32 for the final output
    return out.to(dtype=torch.float32)
