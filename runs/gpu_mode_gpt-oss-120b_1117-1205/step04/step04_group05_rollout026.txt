@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    b = pid // H
    h = pid % H
    # compute block offsets
    block_i = tl.program_id(1)
    block_j = tl.program_id(2)
    i_start = block_i * BLOCK_M
    j_start = block_j * BLOCK_N
    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, N, BLOCK_K):
        cur_k = tl.minimum(N - k, BLOCK_K)
        # offsets for left tile
        left_offset = b * stride_left_b + i_start * stride_left_i + k * stride_left_k + h * stride_left_h
        left_ptrs = left_ptr + left_offset + tl.arange(0, BLOCK_M)[:, None] * stride_left_i + tl.arange(0, cur_k)[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=(i_start + tl.arange(0,BLOCK_M)[:,None] < N) & (k + tl.arange(0,cur_k)[None,:] < N), other=0.0)
        # right tile
        right_offset = b * stride_right_b + j_start * stride_right_j + k * stride_right_k + h * stride_right_h
        right_ptrs = right_ptr + right_offset + tl.arange(0, BLOCK_N)[:, None] * stride_right_j + tl.arange(0, cur_k)[None, :] * stride_right_k
        right = tl.load(right_ptrs, mask=(j_start + tl.arange(0,BLOCK_N)[:,None] < N) & (k + tl.arange(0,cur_k)[None,:] < N), other=0.0)
        # compute dot
        acc += tl.dot(left, right)
    # store out
    out_offset = b * stride_out_b + i_start * stride_out_i + j_start * stride_out_j + h * stride_out_h
    out_ptrs = out_ptr + out_offset + tl.arange(0,BLOCK_M)[:, None] * stride_out_i + tl.arange(0,BLOCK_N)[None, :] * stride_out_j
    mask_store = (i_start + tl.arange(0, BLOCK_M)[:, None] < N) & (j_start + tl.arange(0, BLOCK_N)[None, :] < N)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_store)
```

"""
TriMul (outgoing) forward pass with a Triton kernel.

The kernel implements the O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
for every batch b and hidden channel d.  The input tensor is
layer‑normalized, linearly projected to a hidden dimension, gated
(sigmoid) and optionally masked.  The heavy contraction is done with a
custom Triton kernel that tiles the i‑ and j‑dimensions (BLOCK_M×BLOCK_N)
and accumulates over k in blocks (BLOCK_K).  All arithmetic inside the
kernel is in float16 with float32 accumulation for numerical stability.
After the contraction an output LayerNorm, out‑gate scaling and a final
linear projection bring the result back to the original channel dimension.
The final output is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    left_ptr,          # half*  [B, N, N, H]
    right_ptr,         # half*  [B, N, N, H]
    out_ptr,           # half*  [B, N, N, H]
    B, N, H,           # runtime ints
    stride_b,          # B stride (elements) = N*N*H
    stride_i,          # stride over i or j dimension = N*H
    stride_k,          # stride over k dimension = H
    stride_j,          # stride over j dimension in output = H
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)               # encodes (batch, hidden)
    b = pid // H
    h = pid % H

    pid_i = tl.program_id(1)             # tile over i
    pid_j = tl.program_id(2)             # tile over j

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # ranges for i, j and k
    i_range = i_start + tl.arange(0, BLOCK_M)
    j_range = j_start + tl.arange(0, BLOCK_N)
    k_range = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds i/j
    i_mask = i_range < N
    j_mask = j_range < N

    # base offset for the current (b, h) slice
    base = b * stride_b + h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for kk in range(0, N, BLOCK_K):
        cur_k = kk + k_range
        k_mask = cur_k < N

        # ----- load left tile (i, k) -----
        left_offsets = base + i_range[:, None] * stride_i + cur_k[None, :] * stride_k
        left = tl.load(left_ptr + left_offsets,
                       mask=i_mask[:, None] & k_mask[None, :],
                       other=0.0)

        # ----- load right tile (j, k) -----
        right_offsets = base + j_range[:, None] * stride_i + cur_k[None, :] * stride_k
        right = tl.load(right_ptr + right_offsets,
                        mask=j_mask[:, None] & k_mask[None, :],
                        other=0.0)

        # accumulate dot product over k
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_N)

    # ----- store result -----
    out_offsets = base + i_range[:, None] * stride_i + j_range[None, :] * stride_j
    store_mask = i_mask[:, None] & j_mask[None, :]
    tl.store(out_ptr + out_offsets, acc.to(tl.float16), mask=store_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) – may be None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", and optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------
    # 1. Input LayerNorm (float32) -> half precision
    # -----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # -----------------------------------------------------------
    # 2. Linear projections & gates (half precision)
    # -----------------------------------------------------------
    # weights to half
    w_left = weights["left_proj.weight"].to(torch.float16)
    w_right = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    # projections
    left = F.linear(x, w_left, bias=None)          # (B, N, N, H)
    right = F.linear(x, w_right, bias=None)

    # gates
    left_gate = torch.sigmoid(F.linear(x, w_left_gate, bias=None))
    right_gate = torch.sigmoid(F.linear(x, w_right_gate, bias=None))
    out_gate = torch.sigmoid(F.linear(x, w_out_gate, bias=None))

    # optional mask (broadcast over hidden dim)
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_h
        right = right * mask_h

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # make contiguous for Triton
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, H = left.shape

    # -----------------------------------------------------------
    # 3. Core O(N³) contraction via Triton
    # -----------------------------------------------------------
    out = torch.empty_like(left)   # (B, N, N, H), half

    # strides in number of elements (half precision)
    stride_b = N * N * H          # jump to next batch
    stride_i = N * H              # jump to next i (or j) index
    stride_k = H                  # jump to next k index
    stride_j = H                  # output stride for j dimension

    # tiling parameters (tuned for B200, FP16)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (B * H,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_b,
        stride_i,
        stride_k,
        stride_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------
    # 4. Output LayerNorm, out‑gate scaling and final linear
    # -----------------------------------------------------------
    # to_out_norm weights/bias to half
    w_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_out_norm = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w_out_norm,
        bias=b_out_norm,
    )
    out = out * out_gate

    # final linear projection back to original dim
    w_to_out = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, w_to_out, bias=None)

    # cast back to float32 for the final output
    return out.to(torch.float32)
