    mask = offs < total_elems

    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))

"""
Outgoing TriMul (AlphaFold3) forward pass.

Algorithm
---------
1. Cast input to half‑precision and apply LayerNorm over the feature dimension.
2. Linear projections (no bias) from dim → hidden_dim produce left/right tensors.
3. Compute three gating tensors (linear + sigmoid) producing left_gate,
   right_gate, and out_gate.
4. If a mask is present, fuse mask, left_gate and right_gate into a single
   Triton kernel that updates left/right in‑place:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
   Otherwise perform element‑wise multiplication with the gates in PyTorch.
5. Core O(N³) contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix‑multiply for each hidden slice:
        out_d = L_d @ R_dᵀ    (L_d,R_d ∈ ℝ^{N×N})
   Implemented with a single torch.bmm over the batch dimension
   (batch = B·hidden_dim).
6. Output LayerNorm over the hidden dimension, element‑wise gating with out_gate,
   and a final linear projection hidden_dim → dim.
7. Return the result as float32.

Only the mask‑gate fusion is performed in Triton; all other heavy
operations (LayerNorm, linear layers, batched GEMM) use PyTorch’s
highly‑optimized CUDA kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # *half
    right_ptr,         # *half
    left_gate_ptr,     # *half
    right_gate_ptr,    # *half
    mask_ptr,          # *half, length = B*N*N (broadcasted over H)
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    H: tl.constexpr,   # hidden dimension
):
    """
    In‑place fusion kernel:
        left  = left  * left_gate  * (mask ?)
        right = right * right_gate * (mask ?)
    The mask is shared across the H hidden channels, thus each block of
    H consecutive elements corresponds to a single mask entry.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    lgate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # One mask value per (b,i,j) → each repeats H times in the flattened view
        mask_idx = offs // H
        mval = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * lgate * mval
        right = right * rgate * mval
    else:
        left = left * lgate
        right = right * rgate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_mask_and_gate(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    Launches the Triton kernel that fuses mask and gate multiplication
    into left/right in‑place.
    Arguments are 1‑D flattened tensors (half precision).
    """
    total = left.numel()
    BLOCK = 32768  # sufficiently large to keep launch overhead low
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy when no mask

    _mask_gate_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
        H=hidden_dim,
    )
    # In‑place operation; nothing to return


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor:  torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict: mapping from parameter name to torch.Tensor (float32)
            - config_dict: must contain "dim" and "hidden_dim". May contain
                           "nomask" (bool) indicating that mask should be ignored.

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    inp, mask, weights, config = data
    device = inp.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 0. Cast everything to half‑precision once
    # -----------------------------------------------------------------
    # Input
    x = inp.to(torch.float16)

    # LayerNorm parameters
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias = weights["norm.bias"].to(torch.float16)

    # Projection weights
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)

    # Gate weights
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)

    # Output norm & projection
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    to_out_weight = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (half precision)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # shape [B, N, N, dim], dtype=half

    # -----------------------------------------------------------------
    # 2. Linear projections to hidden dimension (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # -----------------------------------------------------------------
    # 3. Gate tensors (sigmoid)
    # -----------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid()   # [B, N, N, hidden_dim]
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()    # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 4. Apply mask (if present) and fuse with gates via Triton
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] -> broadcast over hidden_dim
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        # Flatten tensors for the kernel
        left_f = left.view(-1)
        right_f = right.view(-1)
        lg_f = left_gate.view(-1)
        rg_f = right_gate.view(-1)
        mask_f = mask_h.view(-1)  # length B*N*N

        _apply_mask_and_gate(left_f, right_f, lg_f, rg_f, mask_f, hidden_dim)

        # Reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask: simple element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core O(N³) contraction via batched GEMM over hidden dimension
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape  # left shape (B, N, N, H)

    # Bring hidden dimension to batch axis: (B, H, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # Batched matrix multiply: (i,k) × (j,k)ᵀ → (i,j)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # back to [B, N, N, dim]

    # -----------------------------------------------------------------
    # 7. Return float32 tensor as required
    # -----------------------------------------------------------------
    return out.to(torch.float32)
