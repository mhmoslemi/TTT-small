"""
TriMul (outgoing) forward pass with fused Triton kernels.

- A first Triton kernel fuses mask and gate multiplications.
- A second Triton kernel computes the O(N³) contraction
  out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
  as a batched GEMM (left @ rightᵀ) with fp16 inputs and fp32 accumulation.
- The remaining operations (LayerNorm, linear projections, output
  LayerNorm and final linear) are performed in PyTorch (fp16) to keep the
  code simple while still achieving the target runtime.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: fuse mask and gating (left/right * gate * mask)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # half* (flattened left)
    right_ptr,         # half* (flattened right)
    left_gate_ptr,     # half* (flattened left_gate)
    right_gate_ptr,    # half* (flattened right_gate)
    mask_ptr,          # half* (flattened mask, broadcasted over hidden dim)
    total_elems: tl.int32,
    H: tl.constexpr,          # hidden dimension
    HAS_MASK: tl.constexpr,   # 1 if mask present else 0
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # load data
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each hidden channel shares a mask entry
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """In‑place fusion of left/right with their gates and optional mask."""
    total = left.numel()
    BLOCK = 2048
    grid = (triton.cdiv(total, BLOCK),)

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when no mask

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        H=hidden_dim,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )


# ----------------------------------------------------------------------
# Kernel 2: batched GEMM for the TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    left_ptr,          # half*   (BH, M, K)   flattened
    right_ptr,         # half*   (BH, N, K)   flattened
    out_ptr,           # half*   (BH, M, N)   flattened
    M, N, K,
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_j, stride_r_k,
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile over rows (i)
    pid_n = tl.program_id(1)   # tile over cols (j)
    pid_b = tl.program_id(2)   # batch*head index

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    left_batch_ptr = left_ptr + pid_b * stride_l_batch
    right_batch_ptr = right_ptr + pid_b * stride_r_batch
    out_batch_ptr = out_ptr + pid_b * stride_o_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k_tile * BLOCK_K

        # left tile (M x K)
        left_tile_ptr = left_batch_ptr + (
            offs_m[:, None] * stride_l_i
            + (k_offset + offs_k)[None, :] * stride_l_k
        )
        left_mask = mask_m[:, None] & ((k_offset + offs_k)[None, :] < K)
        left_tile = tl.load(left_tile_ptr, mask=left_mask, other=0.0)

        # right tile transposed (K x N)  where right_T[k, n] = right[n, k]
        right_tile_ptr = right_batch_ptr + (
            (k_offset + offs_k)[:, None] * stride_r_k
            + offs_n[None, :] * stride_r_j
        )
        right_mask = ((k_offset + offs_k)[:, None] < K) & mask_n[None, :]
        right_tile = tl.load(right_tile_ptr, mask=right_mask, other=0.0)

        acc += tl.dot(left_tile, right_tile)

    # write back
    out_tile_ptr = out_batch_ptr + (
        offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j
    )
    out_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(out_tile_ptr, acc.to(tl.float16), mask=out_mask)


def _batched_gemm_trimul(left, right):
    """
    Compute out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    where left/right are tensors of shape (B, H, N, N) (contiguous, half).
    Returns tensor of same shape.
    """
    B, H, N, _ = left.shape
    BH = B * H
    left_flat = left.view(BH, N, N).contiguous()
    right_flat = right.view(BH, N, N).contiguous()
    out_flat = torch.empty_like(left_flat)

    # strides for the flattened tensors
    stride_l_batch, stride_l_i, stride_l_k = left_flat.stride()
    stride_r_batch, stride_r_j, stride_r_k = right_flat.stride()  # j is row dim, k is col dim
    stride_o_batch, stride_o_i, stride_o_j = out_flat.stride()

    BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        BH,
    )
    _trimul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        N,
        N,
        stride_l_batch,
        stride_l_i,
        stride_l_k,
        stride_r_batch,
        stride_r_j,
        stride_r_k,
        stride_o_batch,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out_flat.view(B, H, N, N)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (float32/bool) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) -> half precision
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (half)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gates (half)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask with gates via Triton (if mask present)
    # ------------------------------------------------------------------
    if not nomask:
        # flatten for kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask.to(torch.float16).contiguous().view(-1)  # [B,N,N]

        _apply_gate_and_mask(
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            hidden_dim,
        )
        # reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    del left_gate, right_gate  # free memory

    # ------------------------------------------------------------------
    # 5. Core contraction: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as a custom Triton batched GEMM.
    # ------------------------------------------------------------------
    # bring hidden dimension to the second axis for easy flattening
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    out_perm = _batched_gemm_trimul(left_perm, right_perm)  # (B, H, N, N)

    # back to original layout
    out = out_perm.permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise scaling

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    return out.to(torch.float32)
