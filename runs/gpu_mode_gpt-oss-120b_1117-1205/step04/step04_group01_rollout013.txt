```
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)
    device = input_tensor.device
    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x = x.to(torch.float16)

    # Projections
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16), bias=None)

    # Gates
    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    # Apply mask and gates
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # shape [B,N,N,1]
        # Use kernel
        total = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)
        HAS_MASK = 1
        _gate_mask_fuse_kernel[grid](
            left, right,
            left_gate, right_gate,
            mask_h,
            total,
            BLOCK=BLOCK,
            HAS_MASK=HAS_MASK,
            H=hidden_dim,
        )
    else:
        left = left * left_gate
        right = right * right_gate

    # Reshape for batched matmul
    B, N, _, _ = left.shape # left shape [B,N,N,hidden_dim]
    # permute to [B, hidden_dim, N, N] then view
    left_mat = left.permute(0,3,1,2).contiguous().view(B*hidden_dim, N, N)
    right_mat = right.permute(0,3,2,1).contiguous().view(B*hidden_dim, N, N)

    # Batched matmul using torch.matmul
    out_mat = torch.matmul(left_mat, right_mat) # shape [B*hidden_dim, N, N]

    # Reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()

    # Output LayerNorm
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"].to(torch.float16), bias=weights["to_out_norm.bias"].to(torch.float16))

    out = out * out_gate

    # Final linear
    out = F.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    return out.to(torch.float32)
```

"""
Optimized TriMul “outgoing” forward pass.

Algorithm:
1. Layer‑norm the input (float32) and cast to fp16.
2. Linear projections → left/right (fp16) and three gate vectors (fp16).
3. Fuse gate multiplication (and optional mask) into a single Triton kernel.
4. Reshape left/right to (B*H, N, N) where H = hidden_dim.
   Right is transposed on its inner pair‑wise dimensions.
5. Batched matrix multiplication (torch.matmul) computes the O(N³) contraction:
       out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,k,j]
6. Reshape back to (B, N, N, H), apply output LayerNorm, out‑gate and final
   linear projection.  All heavy arithmetic stays in fp16; the final tensor
   is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # fp16 *
    right_ptr,         # fp16 *
    left_gate_ptr,     # fp16 *
    right_gate_ptr,    # fp16 *
    mask_ptr,          # fp16 *
    total_elems: tl.int32,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask present, else 0
    H: tl.constexpr,          # hidden dimension (size of innermost channel)
):
    """In‑place elementwise fusion:
       left  = left  * left_gate  * (mask if present)
       right = right * right_gate * (mask if present)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elems

    left      = tl.load(left_ptr      + offs, mask=valid, other=tl.float16(0.0))
    right     = tl.load(right_ptr     + offs, mask=valid, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=tl.float16(0.0))
    right_gate= tl.load(right_gate_ptr+ offs, mask=valid, other=tl.float16(0.0))

    if HAS_MASK:
        # mask is broadcasted over the hidden dimension; one mask entry per H values
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=tl.float16(0.0))
        left  = left  * left_gate  * mask_val
        right = right * right_gate * mask_val
    else:
        left  = left  * left_gate
        right = right * right_gate

    tl.store(left_ptr  + offs, left,  mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of parameter tensors
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> fp16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (fp16)
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )   # [B, N, N, hidden_dim]

    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Gates (fp16)
    # ------------------------------------------------------------------
    left_gate = F.linear(
        x,
        weights["left_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    right_gate = F.linear(
        x,
        weights["right_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(torch.float16),
        bias=None,
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse gates (and optional mask) with a Triton kernel
    # ------------------------------------------------------------------
    total_elems = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_elems, BLOCK),)

    if not nomask and mask is not None:
        # mask: [B, N, N] → broadcast over hidden_dim
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        HAS_MASK = 1
        mask_ptr = mask_h
    else:
        HAS_MASK = 0
        # dummy pointer (won’t be dereferenced because HAS_MASK==0)
        mask_ptr = left

    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total_elems,
        BLOCK=BLOCK,
        HAS_MASK=HAS_MASK,
        H=hidden_dim,
    )

    # ------------------------------------------------------------------
    # 5. Core O(N³) contraction via batched matrix multiplication
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # left shape is [B, N, N, hidden_dim]

    # reshape to (B*H, N, N); right is transposed on its inner pairwise indices
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N)

    # batched matmul (FP16)
    out_mat = torch.matmul(left_mat, right_mat)  # (B*H, N, N)

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in requested precision
    return out.to(torch.float32)
