"""
Outgoing TriMul (AlphaFold3) forward pass.

The kernel fuses mask multiplication with the left/right gating tensors in a
single pass.  All heavy arithmetic (linear projections, the N³ contraction,
layer‑norms and final projection) is performed in float16 to exploit Tensor
Cores; only the final result is cast back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # *half   (B,N,N,H)
    right_ptr,         # *half   (B,N,N,H)
    left_gate_ptr,     # *half   (B,N,N,H)
    right_gate_ptr,    # *half   (B,N,N,H)
    mask_ptr,          # *half   (B,N,N)  broadcasted over H
    total_elems: tl.int32,
    H: tl.constexpr,   # hidden dimension
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """In‑place fusion:
       left  = left  * left_gate  * (mask if present)
       right = right * right_gate * (mask if present)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    lgate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # each H‑contiguous block corresponds to one mask element
        mask_idx = offs // H
        mval = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
        left = left * lgate * mval
        right = right * rgate * mval
    else:
        left = left * lgate
        right = right * rgate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_mask_and_gate(left, right, left_gate, right_gate, mask, hidden_dim):
    """Launch the Triton kernel that fuses mask and gating in‑place."""
    total = left.numel()                       # B * N * N * H
    BLOCK = 65536                               # enough threads per launch
    grid = (triton.cdiv(total, BLOCK),)
    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else left  # dummy pointer when no mask

    _mask_gate_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_ptr,
        total,
        H=hidden_dim,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )
    # inplace modification; nothing to return


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor:  torch.Tensor [B, N, N] (float32/bool) or None
            - weights_dict: mapping from parameter names to tensors (float32)
            - config_dict: contains "dim" and "hidden_dim"; may contain "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    inp, mask, weights, config = data
    device = inp.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------------------------------------------------------------
    # 1. LayerNorm on the input (float16)
    # ---------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float16, device=device)
    norm_bias = weights["norm.bias"].to(torch.float16, device=device)
    x = inp.to(torch.float16)
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # ---------------------------------------------------------------
    # 2. Linear projections to hidden dimension (no bias)
    # ---------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"].to(torch.float16, device=device)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16, device=device)

    left = F.linear(x, left_proj_weight)   # (B,N,N,hidden)
    right = F.linear(x, right_proj_weight)

    # ---------------------------------------------------------------
    # 3. Gates (sigmoid of linear projections)
    # ---------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"].to(torch.float16, device=device)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16, device=device)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16, device=device)

    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    # ---------------------------------------------------------------
    # 4. Fuse mask with gates (if mask is present)
    # ---------------------------------------------------------------
    if not nomask and mask is not None:
        # mask broadcasted over hidden dimension
        mask_h = mask.to(torch.float16, device=device).unsqueeze(-1)   # (B,N,N,1)

        # flatten tensors for the kernel
        left_f = left.view(-1)
        right_f = right.view(-1)
        left_gate_f = left_gate.view(-1)
        right_gate_f = right_gate.view(-1)
        mask_f = mask_h.view(-1)   # length B*N*N

        _apply_mask_and_gate(
            left_f, right_f, left_gate_f, right_gate_f, mask_f, hidden_dim
        )

        # restore shapes
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # simple element‑wise multiplication when mask is absent
        left = left * left_gate
        right = right * right_gate

    # ---------------------------------------------------------------
    # 5. Core O(N³) contraction:
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ---------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # reshape to (B*H, N, N) for batched GEMM
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matrix multiply: (i,k) × (j,k)ᵀ -> (i,j)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection
    # ---------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16, device=device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16, device=device)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate

    to_out_weight = weights["to_out.weight"].to(torch.float16, device=device)
    out = F.linear(out, to_out_weight)   # (B,N,N,dim)

    # ---------------------------------------------------------------
    # 7. Cast back to float32 and return
    # ---------------------------------------------------------------
    return out.to(torch.float32)
