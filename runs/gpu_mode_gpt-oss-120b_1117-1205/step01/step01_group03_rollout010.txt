def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Input normalization
    # Use torch.nn.functional.layer_norm for speed.
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections (FP16)
    x_fp16 = x.to(torch.float16)
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)
    left = torch.nn.functional.linear(x_fp16, left_proj_weight) # shape [bs,N,N,hidden]
    right = torch.nn.functional.linear(x_fp16, right_proj_weight)

    # Handle mask
    if mask is None:
        mask_fp16 = torch.ones(bs, N, N, dtype=torch.float16, device=x.device)
    else:
        mask_fp16 = mask.to(torch.float16)
    mask_fp16 = mask_fp16.unsqueeze(-1) # shape [bs,N,N,1]
    left = left * mask_fp16
    right = right * mask_fp16

    # Gates
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.nn.functional.linear(x_fp16, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_fp16, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_fp16, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Reshape for GEMM
    # Convert shape to (bs, hidden, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()   # (bs, hidden, N, N)
    # Right need to be transposed on last two dims (k,j) => we need right_trans = right.permute(0, 3, 2, 1)
    right = right.permute(0, 3, 2, 1).contiguous()  # (bs, hidden, N, N) where original (j,k) becomes (k,j)

    total_batches = bs * hidden_dim
    left_mat = left.reshape(total_batches, N, N)
    right_mat = right.reshape(total_batches, N, N)
    out_mat = torch.empty_like(left_mat)

    # Determine block sizes
    # Some heuristics: choose block M,N=128 unless N smaller
    if N <= 64:
        BLOCK_M = BLOCK_N = 32
        BLOCK_K = 32
    elif N <= 128:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    # Launch kernel
    grid = (total_batches, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    tri_mul_gemm[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(1), out_mat.stride(2),
        left_mat.stride(0), right_mat.stride(0), out_mat.stride(0),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Reshape back
    out = out_mat.reshape(bs, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)  # shape [bs,N,N,hidden]

    # Output layer norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Apply out gate (convert out_gate to float32)
    out_norm = out_norm * out_gate.to(torch.float32)

    # Final projection
    to_out_weight = weights["to_out.weight"].to(torch.float32)  # shape (dim, hidden)
    output = torch.nn.functional.linear(out_norm, to_out_weight)  # shape [bs,N,N,dim]

    return output
```

```
@triton.jit
def tri_mul_gemm(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # determine valid region (mask)
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # compute base pointers for current batch
    a_batch = a_ptr + pid_batch * stride_batch_a
    b_batch = b_ptr + pid_batch * stride_batch_b
    c_batch = c_ptr + pid_batch * stride_batch_c

    # pointers for A and B
    a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)  # (M,K)*(K,N) -> (M,N)
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store result
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass with a Triton kernel for the core N×N×N
multiplication.

Algorithm
---------
1. Layer‐norm the input tensor (float32).
2. Project to a hidden dimension (float16) with three linear layers:
   left_proj, right_proj and three gated linear layers.
3. Apply the optional mask and the gates.
4. For each batch and hidden channel compute
       out[b, h, i, j] = sum_k left[b, h, i, k] * right[b, h, j, k]
   which is a batched matrix multiplication
       left_mat @ right_mat.T .
   The multiplication is performed by a Triton kernel `tri_mul_gemm`
   that tiles the M‑N dimensions and iterates over K with FP16 loads and
   FP32 accumulation.
5. Reshape the result back to [B, N, N, hidden] (float32),
   apply a second LayerNorm, an output gate and a final linear projection
   to the original channel dimension.

Only the heavy N³ matrix multiplication is offloaded to Triton;
all other operations are kept in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_gemm(
    a_ptr, b_ptr, c_ptr,          # pointers to A, B, C
    M, N, K,                      # matrix dimensions (all = seq_len)
    stride_am, stride_ak,         # strides for A (row, col)
    stride_bk, stride_bn,         # strides for B (row, col) – B is already transposed
    stride_cm, stride_cn,         # strides for C (row, col)
    stride_batch_a, stride_batch_b, stride_batch_c,  # stride between independent batches
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B, where A and B are FP16 and accumulation is FP32."""
    pid_batch = tl.program_id(0)   # batch * hidden index
    pid_m = tl.program_id(1)       # block row
    pid_n = tl.program_id(2)       # block column

    # Base pointers for this batch
    a_batch = a_ptr + pid_batch * stride_batch_a
    b_batch = b_ptr + pid_batch * stride_batch_b
    c_batch = c_ptr + pid_batch * stride_batch_c

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds handling
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Pointers to the current tiles of A and B
    a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store the result (cast back to FP16)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # Extract configuration and weight tensors
    # ------------------------------------------------------------------
    dim = config["dim"]               # input & output channel dimension
    hidden = config["hidden_dim"]     # hidden dimension (c_z)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [bs, N, N, dim], dtype float32

    # ------------------------------------------------------------------
    # 2. Linear projections & gates (float16 for speed)
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # Projections
    left_proj_w = weights["left_proj.weight"].to(torch.float16)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left = F.linear(x_f16, left_proj_w)   # [bs, N, N, hidden]
    right = F.linear(x_f16, right_proj_w)

    # Optional mask (skip if nomask is set)
    if not config.get("nomask", False):
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)  # [bs, N, N, 1]
        left = left * mask_f16
        right = right * mask_f16

    # Gates
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_f16, left_gate_w).sigmoid()
    right_gate = F.linear(x_f16, right_gate_w).sigmoid()
    out_gate = F.linear(x_f16, out_gate_w).sigmoid()   # [bs, N, N, hidden]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Core TriMul: batched GEMM (left @ right.T) via Triton
    # ------------------------------------------------------------------
    # Rearrange to (batch, hidden, N, N) and make right transposed on the
    # last two dimensions.
    left = left.permute(0, 3, 1, 2).contiguous()          # (bs, hidden, N, N)
    right = right.permute(0, 3, 2, 1).contiguous()        # (bs, hidden, N, N)  (k,j)

    bs, _, N, _ = left.shape
    total_batches = bs * hidden

    left_mat = left.reshape(total_batches, N, N)
    right_mat = right.reshape(total_batches, N, N)
    out_mat = torch.empty_like(left_mat)   # fp16

    # Heuristic block sizes
    if N <= 64:
        BLOCK_M = BLOCK_N = 32
        BLOCK_K = 32
    elif N <= 128:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        total_batches,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    tri_mul_gemm[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,                           # M, N, K
        left_mat.stride(1), left_mat.stride(2),   # stride_am, stride_ak
        right_mat.stride(1), right_mat.stride(2), # stride_bk, stride_bn
        out_mat.stride(1), out_mat.stride(2),     # stride_cm, stride_cn
        left_mat.stride(0), right_mat.stride(0), out_mat.stride(0),  # batch strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [bs, N, N, hidden] (float32)
    out = out_mat.reshape(bs, hidden, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 4. Output LayerNorm, output gate and final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )  # [bs, N, N, hidden], float32

    # Apply output gate (convert gate to float32)
    out = out * out_gate.to(torch.float32)

    # Final projection back to original dim
    to_out_w = weights["to_out.weight"].to(torch.float32)   # (dim, hidden)
    output = F.linear(out, to_out_w)                       # [bs, N, N, dim]

    return output
