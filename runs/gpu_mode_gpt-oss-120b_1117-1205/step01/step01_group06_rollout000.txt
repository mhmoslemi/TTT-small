"""
TriMul forward (outgoing) with a Triton kernel for the hidden‑dim LayerNorm.

The implementation follows the PyTorch reference:
1. LayerNorm over the last dimension (`dim`).
2. Linear projections to a hidden dimension (`hidden_dim`) for left/right tensors.
3. Linear gates (left, right, out) with sigmoid activation.
4. Optional pairwise mask applied before gating.
5. Pairwise contraction over the sequence dimension k:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   (implemented with a high‑performance torch.einsum on bfloat16).
6. Hidden‑dim LayerNorm performed by a Triton kernel (per (b,i,j) slice).
7. Multiply by the output gate and a final linear projection back to `dim`.

Only the hidden‑dim LayerNorm runs on Triton; all other heavy work uses
optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_hidden_dim_kernel(
    inp_ptr,               # *float32: [B,N,N,H]
    out_ptr,               # *float32: [B,N,N,H]
    weight_ptr,            # *float32: [H]
    bias_ptr,              # *float32: [H]
    B,                     # tl.int32
    N,                     # tl.int32
    H,                     # tl.int32
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,   # tl.int32
    stride_out_b, stride_out_i, stride_out_j, stride_out_h, # tl.int32
    eps: tl.constexpr,                                   # float
    BLOCK_H: tl.constexpr                               # compile‑time block size >= H
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # Decode (b,i,j) from pid
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # Offsets along the hidden dimension
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # Base pointer for this (b,i,j) slice
    base_in = b * stride_in_b + i * stride_in_i + j * stride_in_j
    base_out = b * stride_out_b + i * stride_out_i + j * stride_out_j

    # Load the hidden‑dim vector
    x = tl.load(
        inp_ptr + base_in + offs_h * stride_in_h,
        mask=mask_h,
        other=0.0
    )

    # ---------- mean ----------
    sum_x = tl.sum(tl.where(mask_h, x, 0.0))
    mean = sum_x / H

    # ---------- variance ----------
    diff = x - mean
    var = tl.sum(tl.where(mask_h, diff * diff, 0.0)) / H

    # ---------- normalize ----------
    inv_std = tl.rsqrt(var + eps)
    norm = diff * inv_std

    # ---------- apply affine ----------
    w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)
    b_ = tl.load(bias_ptr + offs_h, mask=mask_h, other=0.0)
    out = norm * w + b_

    # Store result
    tl.store(
        out_ptr + base_out + offs_h * stride_out_h,
        out,
        mask=mask_h
    )


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be unused)
            - weights: dict of model parameters (all tensors are on the same device)
            - config: dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm over the last dimension
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    x = x.contiguous()
    B, N, _, _ = x.shape

    # --------------------------------------------------------------
    # 2) Linear projections & gating (float32)
    x_flat = x.view(-1, dim)                       # [B*N*N, dim]

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    left_gate = F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    right_gate = F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    out_gate = F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()

    # --------------------------------------------------------------
    # 3) Apply mask (if present) and gates
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(dtype)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 4) Pairwise contraction over k (einsum)
    # Use bfloat16 for bandwidth, then back to float32
    left_bf = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    out = torch.einsum('... i k h, ... j k h -> ... i j h', left_bf, right_bf)  # [B,N,N,H] bfloat16
    out = out.to(dtype).contiguous()

    # --------------------------------------------------------------
    # 5) Hidden‑dim LayerNorm via Triton
    out_norm = torch.empty_like(out)

    # Strides in element units
    s_b, s_i, s_j, s_h = out.stride()
    total_programs = B * N * N
    grid = (triton.cdiv(total_programs, 1),)   # one program per (b,i,j)

    # BLOCK_H must be >= hidden_dim; round up to a power‑of‑two for good compile time.
    BLOCK_H = 128
    if hidden_dim > BLOCK_H:
        # fallback to a larger static block (will still work but may waste threads)
        BLOCK_H = 256

    layernorm_hidden_dim_kernel[grid](
        out,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        B,
        N,
        hidden_dim,
        s_b, s_i, s_j, s_h,
        s_b, s_i, s_j, s_h,
        eps=1e-5,
        BLOCK_H=BLOCK_H,
    )

    # --------------------------------------------------------------
    # 6) Apply output gate
    out_scaled = out_norm * out_gate

    # --------------------------------------------------------------
    # 7) Final linear projection back to `dim`
    out_scaled_flat = out_scaled.view(-1, hidden_dim)       # [B*N*N, H]
    final_flat = F.linear(out_scaled_flat, weights["to_out.weight"])
    final = final_flat.view(B, N, N, dim)

    return final
