def custom_kernel(data):
    x, mask, weights, config = data
    device = x.device
    bs, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    eps = 1e-5
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)
    # Cast to float16
    x_f16 = x_norm.half()
    # Linear projections
    left = torch.nn.functional.linear(x_f16, weights["left_proj.weight"].half())
    right = torch.nn.functional.linear(x_f16, weights["right_proj.weight"].half())
    # Mask
    if mask is not None:
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_f16
        right = right * mask_f16
    # Gates
    left_gate = torch.nn.functional.linear(x_f16, weights["left_gate.weight"].half()).sigmoid()
    right_gate = torch.nn.functional.linear(x_f16, weights["right_gate.weight"].half()).sigmoid()
    out_gate = torch.nn.functional.linear(x_f16, weights["out_gate.weight"].half()).sigmoid()
    left = left * left_gate
    right = right * right_gate
    # Triton core multiplication
    # Permute
    left = left.permute(0,3,1,2).contiguous()   # shape (bs, hidden, N, N)
    right = right.permute(0,3,2,1).contiguous() # shape (bs, hidden, N, N)
    total_batches = bs * hidden_dim
    left_mat = left.view(total_batches, N, N)  # shape (total_batches, M, K)
    right_mat = right.view(total_batches, N, N) # shape (total_batches, K, N)
    out_mat = torch.empty_like(left_mat) # half type
    # Determine block sizes
    if N <= 128:
        BLOCK_M = 32
        BLOCK_N = 32
        BLOCK_K = 32
    elif N <= 256:
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 32
    else: # up to 1024
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 64
    # Grid dimensions
    grid = (total_batches,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))
    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(1), out_mat.stride(2),
        left_mat.stride(0), right_mat.stride(0), out_mat.stride(0),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # Reshape back
    out = out_mat.view(bs, hidden_dim, N, N).permute(0,2,3,1).to(torch.float32)
    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                                            bias=to_out_norm_bias, eps=eps)
    # apply out_gate
    out_norm = out_norm * out_gate.to(torch.float32)
    # final projection
    to_out_weight = weights["to_out.weight"]
    output = torch.nn.functional.linear(out_norm, to_out_weight)
    return output

"""
TriMul outgoing forward implementation with a Triton kernel for the N^3
multiplicative update. The kernel computes, for each batch and hidden
dimension, a matrix multiplication `out = left @ right.T`, where `left`
and `right` are the gated projection tensors of shape (N, N). The rest
of the TriMul module (LayerNorms, linear projections, sigmoid gates) are
performed in PyTorch, while the core O(N³) operation is offloaded to
Triton for speed.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to A, B, C
    M, N, K,                           # matrix dimensions (M x K) @ (K x N) -> (M x N)
    stride_am, stride_ak,              # A strides: row (M) and column (K)
    stride_bk, stride_bn,              # B strides: row (K) and column (N)
    stride_cm, stride_cn,              # C strides: row (M) and column (N)
    stride_batch_a, stride_batch_b, stride_batch_c,   # batch stride for A, B, C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute batched matrix multiplication:
        C[batch] = A[batch] @ B[batch]
    Each batch corresponds to a distinct (batch, hidden) pair
    of the TriMul operator.
    """
    pid_batch = tl.program_id(0)  # batch * hidden index
    pid_m = tl.program_id(1)      # tile row index
    pid_n = tl.program_id(2)      # tile col index

    # base pointers for this specific batch
    a_batch = a_ptr + pid_batch * stride_batch_a
    b_batch = b_ptr + pid_batch * stride_batch_b
    c_batch = c_ptr + pid_batch * stride_batch_c

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # pointers to blocks of A and B
    a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension in tiles
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) => (BLOCK_M, BLOCK_N)
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store result as FP16
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the TriMul "outgoing" operator.
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    # unpack config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # input shape: [bs, N, N, dim]
    bs, N, _, _ = input_tensor.shape

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (over last dimension)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # -----------------------------------------------------------------
    # 2. Linear projections & gating (use FP16 for speed)
    # -----------------------------------------------------------------
    x_f16 = x.half()  # [bs, N, N, dim]  -> half

    # projections
    left = F.linear(x_f16, weights["left_proj.weight"].half())   # [bs, N, N, hidden]
    right = F.linear(x_f16, weights["right_proj.weight"].half())

    # mask (if provided)
    if mask is not None:
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)  # [bs, N, N, 1]
        left = left * mask_f16
        right = right * mask_f16

    # gates
    left_gate = F.linear(x_f16, weights["left_gate.weight"].half()).sigmoid()
    right_gate = F.linear(x_f16, weights["right_gate.weight"].half()).sigmoid()
    out_gate = F.linear(x_f16, weights["out_gate.weight"].half()).sigmoid()

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 3. Core N^3 multiplication via Triton (batched matmul)
    #    out[b, i, j, d] = sum_k left[b, i, k, d] * right[b, j, k, d]
    #    This is equivalent to a batched GEMM per hidden channel.
    # -----------------------------------------------------------------
    # Rearrange tensors to (batch, hidden, N, N) layout.
    # left: (bs, N, N, hidden) -> (bs, hidden, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    # right: (bs, N, N, hidden) -> (bs, hidden, N, N) where the last two dims are (k, j)
    right = right.permute(0, 3, 2, 1).contiguous()

    total_batches = bs * hidden_dim
    # reshape to (total_batches, N, N)
    left_mat = left.view(total_batches, N, N)
    right_mat = right.view(total_batches, N, N)

    # output buffer (FP16)
    out_mat = torch.empty_like(left_mat)

    # Choose block sizes based on N (simple heuristics)
    if N <= 128:
        BLOCK_M, BLOCK_N, BLOCK_K = 32, 32, 32
    elif N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    elif N <= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    else:  # N up to 1024
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    grid = (total_batches,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,                                 # M, N, K
        left_mat.stride(1), left_mat.stride(2),   # stride_am, stride_ak
        right_mat.stride(1), right_mat.stride(2),# stride_bk, stride_bn
        out_mat.stride(1), out_mat.stride(2),    # stride_cm, stride_cn
        left_mat.stride(0), right_mat.stride(0), out_mat.stride(0),  # batch strides
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Restore original layout: (bs, N, N, hidden) in FP32
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # -----------------------------------------------------------------
    # 4. Output LayerNorm, output gate, final linear projection
    # -----------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # apply output gate (convert to float32)
    out_norm = out_norm * out_gate.to(torch.float32)

    # final linear projection back to `dim`
    output = F.linear(out_norm, weights["to_out.weight"])

    return output
```

TriMul outgoing forward implementation with a Triton kernel for the
core N³ multiplicative update. The kernel computes, for each batch and
hidden channel, a matrix multiplication
    out = left @ right.T
where `left` and `right` are the gated projection tensors of shape
(N, N). All other operations (LayerNorms, linear projections, sigmoid

import triton.language as tl

def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to A, B, C
    M, N, K,                           # matrix sizes: (M×K) @ (K×N) → (M×N)
    stride_am, stride_ak,              # A: stride for rows (M) and cols (K)
    stride_bk, stride_bn,              # B: stride for rows (K) and cols (N)
    stride_cm, stride_cn,              # C: stride for rows (M) and cols (N)
    stride_batch_a, stride_batch_b, stride_batch_c,  # batch stride for A, B, C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C[batch] = A[batch] @ B[batch]."""
    pid_batch = tl.program_id(0)  # batch * hidden index
    pid_m = tl.program_id(1)      # tile row
    pid_n = tl.program_id(2)      # tile col

    # Pointers for the current batch
    a_batch = a_ptr + pid_batch * stride_batch_a
    b_batch = b_ptr + pid_batch * stride_batch_b
    c_batch = c_ptr + pid_batch * stride_batch_c

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Pointers to the sub‑tiles
    a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)
        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store result as FP16
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor with shape [bs, seq_len, seq_len, dim] and dtype torch.float32.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    # -----------------------------------------------------------------
    # 0. Extract shapes / config
    # -----------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, N, _, _ = input_tensor.shape  # input: [bs, N, N, dim]

    # -----------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [bs, N, N, dim] float32

    # -----------------------------------------------------------------
    # 2. Linear projections (FP16) and mask handling
    # -----------------------------------------------------------------
    x_f16 = x.half()  # cast once

    # Projections to hidden dimension (no bias)
    left = F.linear(x_f16, weights["left_proj.weight"].half())   # [bs, N, N, hidden]
    right = F.linear(x_f16, weights["right_proj.weight"].half())

    # Apply mask if provided (mask shape: [bs, N, N])
    if mask is not None:
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)   # [bs, N, N, 1]
        left = left * mask_f16
        right = right * mask_f16

    # -----------------------------------------------------------------
    # 3. Gates (computed in FP32 for better accuracy, then cast)
    # -----------------------------------------------------------------
    left_gate_fp32 = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [bs, N, N, hidden]
    right_gate_fp32 = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))         # keep FP32

    left_gate = left_gate_fp32.half()
    right_gate = right_gate_fp32.half()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 4. Core N³ update via Triton batched GEMM
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # -----------------------------------------------------------------
    # Rearrange to (batch, hidden, N, N) where the last two dims are (i,k) and (k,j)
    left = left.permute(0, 3, 1, 2).contiguous()   # (bs, hidden, N, N)
    right = right.permute(0, 3, 2, 1).contiguous()  # (bs, hidden, N, N) with axes (k, j)

    total_batches = bs * hidden_dim
    left_mat = left.view(total_batches, N, N)   # (B*H, M=N, K=N)
    right_mat = right.view(total_batches, N, N)  # (B*H, K=N, N=N)
    out_mat = torch.empty_like(left_mat)        # output buffer (FP16)

    # Heuristic block sizes (tuned for typical N up to 1024)
    if N <= 128:
        BLOCK_M, BLOCK_N, BLOCK_K = 32, 32, 32
    elif N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    elif N <= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    else:  # up to 1024
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    grid = (
        total_batches,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,                                     # M, N, K
        left_mat.stride(1), left_mat.stride(2),       # stride_am, stride_ak
        right_mat.stride(1), right_mat.stride(2),     # stride_bk, stride_bn
        out_mat.stride(1), out_mat.stride(2),         # stride_cm, stride_cn
        left_mat.stride(0), right_mat.stride(0), out_mat.stride(0),  # batch strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to the original layout and convert to FP32
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)  # [bs, N, N, hidden]

    # -----------------------------------------------------------------
    # 5. Output LayerNorm, output gate, final linear projection
    # -----------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_norm = out_norm * out_gate  # out_gate already FP32

    # Final projection back to `dim`
    output = F.linear(out_norm, weights["to_out.weight"])

    return output
