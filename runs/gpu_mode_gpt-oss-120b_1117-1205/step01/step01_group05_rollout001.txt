def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm weights
    norm_weight = weights["norm.weight"].to(device).to(torch.float32)
    norm_bias = weights["norm.bias"].to(device).to(torch.float32)

    # Compute LayerNorm over last dim
    x = input_tensor.to(torch.float32)
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)  # inv sqrt
    x_norm = x_norm * norm_weight + norm_bias

    # Cast to fp16 for projections/gates
    x_h = x_norm.to(torch.float16)

    # Load weights
    w_left = weights["left_proj.weight"].to(device).to(torch.float16)  # shape (hidden_dim, dim)
    w_right = weights["right_proj.weight"].to(device).to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(device).to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(device).to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(device).to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(device).to(torch.float32)
    b_to_out_norm = weights["to_out_norm.bias"].to(device).to(torch.float32)
    w_to_out = weights["to_out.weight"].to(device).to(torch.float32)

    # Linear projections
    left = torch.nn.functional.linear(x_h, w_left)  # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_h, w_right)

    # Apply mask if provided
    if mask is not None:
        mask_exp = mask.to(x_h.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate = torch.nn.functional.linear(x_h, w_left_gate).sigmoid()
    right_gate = torch.nn.functional.linear(x_h, w_right_gate).sigmoid()
    out_gate = torch.nn.functional.linear(x_h, w_out_gate).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Triton batched matmul
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape for kernel
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous()
    right_mat = right_mat.permute(0, 1, 3, 2).contiguous().view(B * H, N, N)

    out_mat = torch.empty((B * H, N, N), dtype=torch.float32, device=device)

    # strides
    stride_az, stride_am, stride_ak = left_mat.stride()
    stride_bz, stride_bk, stride_bn = right_mat.stride()
    stride_cz, stride_cm, stride_cn = out_mat.stride()

    # launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (B * H, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    _tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        N,  # M
        N,  # N
        N,  # K
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=2,
    )

    # reshape output
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # shape [B,N,N,hidden_dim] (float32)

    # Output LayerNorm
    # LN over hidden_dim channel
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) * torch.rsqrt(var_o + eps)
    out_norm = out_norm * w_to_out_norm + b_to_out_norm

    # Apply output gate (out_gate is [B,N,N,hidden_dim] fp16 or fp32) - ensure dtype matches
    out_norm = out_norm * out_gate.to(torch.float32)

    # Final linear projection back to dim
    output = torch.nn.functional.linear(out_norm, w_to_out)  # shape [B,N,N,dim]

    return output

@triton.jit
def _tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # batch id
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer to C tile
    c_ptrs = C_ptr + batch_id * stride_cz + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a_ptrs = A_ptr + batch_id * stride_az + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # Load B tile (K x N)
        b_ptrs = B_ptr + batch_id * stride_bz + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    # Store result
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward kernel.

The heavy computation is the tensor contraction:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a batched matrix multiplication per (b,d):
    out_slice = left_slice @ right_slice.T
We implement this contraction with a Triton GEMM kernel that processes
(a batch of) (N x N) matrices in fp16 (inputs) with fp32 accumulation.
All other operations (LayerNorms, linear projections, gates, mask handling)
are performed in PyTorch. The final result is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: C = A @ B  (B is already transposed)
    A : (batch, M, K)   - fp16
    B : (batch, K, N)   - fp16 (original right slice transposed)
    C : (batch, M, N)   - fp32 output
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)   # block row
    pid_n = tl.program_id(2)   # block col

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the output tile
    c_ptrs = (
        C_ptr
        + batch_id * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile (M x K)
        a_ptrs = (
            A_ptr
            + batch_id * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # load B tile (K x N)
        b_ptrs = (
            B_ptr
            + batch_id * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    # store result
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (float32, bool or None)
            - weights: dict of model parameters
            - config: dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # ------------------------------------------------------------------
    # Input LayerNorm (over last dimension)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device).to(torch.float32)
    norm_bias = weights["norm.bias"].to(device).to(torch.float32)

    x = input_tensor.to(torch.float32)  # [B, N, N, dim]
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Projections & Gates (fp16 for speed)
    # ------------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    w_left = weights["left_proj.weight"].to(device).to(torch.float16)          # (hidden_dim, dim)
    w_right = weights["right_proj.weight"].to(device).to(torch.float16)

    w_left_gate = weights["left_gate.weight"].to(device).to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(device).to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(device).to(torch.float16)

    # linear projections
    left = F.linear(x_h, w_left)      # [B, N, N, hidden_dim]
    right = F.linear(x_h, w_right)

    # optional mask
    if mask is not None:
        mask_exp = mask.to(x_h.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # gates
    left_gate = F.linear(x_h, w_left_gate).sigmoid()
    right_gate = F.linear(x_h, w_right_gate).sigmoid()
    out_gate = F.linear(x_h, w_out_gate).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton batched matrix multiplication: out = left @ right.T
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N) where each batch corresponds to a (b,d) slice
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)          # (B*H, M=N, K=N)
    # right needs to be transposed on its last two dims
    right_mat = right.permute(0, 3, 1, 2).contiguous()                         # (B, H, N, N)
    right_mat = right_mat.permute(0, 1, 3, 2).contiguous().view(B * H, N, N)  # (B*H, K=N, N)

    out_mat = torch.empty((B * H, N, N), dtype=torch.float32, device=device)

    # strides for the three tensors
    stride_az, stride_am, stride_ak = left_mat.stride()
    stride_bz, stride_bk, stride_bn = right_mat.stride()
    stride_cz, stride_cm, stride_cn = out_mat.stride()

    # tile sizes (tuned for B200, fp16 inputs)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        N,  # M
        N,  # N
        N,  # K
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=2,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp32

    # ------------------------------------------------------------------
    # Output LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float32)

    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) * torch.rsqrt(var_o + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # Apply output gate
    out_norm = out_norm * out_gate.to(torch.float32)

    # ------------------------------------------------------------------
    # Final projection back to original dimension
    # ------------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(device).to(torch.float32)  # (dim, hidden_dim)
    output = F.linear(out_norm, w_to_out)  # [B, N, N, dim] float32

    return output
