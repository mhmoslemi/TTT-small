import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

## Triton launch convention

Every failure mode below has appeared in this search. Read this before writing.

A `@triton.jit` function is NOT a Python function. It is launched with a grid
subscript, and its `tl.constexpr` parameters exist only inside it:

```python
@triton.jit
def my_kernel(x_ptr, y_ptr, n, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < n
    tl.store(y_ptr + offs, tl.load(x_ptr + offs, mask=m, other=0.0), mask=m)

def custom_kernel(data):
    ...
    BLOCK = 128                                   # a host variable YOU define
    grid = (triton.cdiv(n, BLOCK),)               # host-side, use triton.cdiv
    my_kernel[grid](x, y, n, BLOCK=BLOCK)         # note the [grid] subscript
```

- `my_kernel(...)` without `[grid]` raises "Cannot call @triton.jit'd outside
  the scope of a kernel".
- `BLOCK`, `BLOCK_M`, `BLOCK_N` are not defined on the host unless you define
  them there. Referencing one you only declared as a `tl.constexpr` parameter
  raises NameError.
- `tl.cdiv` is device-side. On the host use `triton.cdiv`.
- Define the kernel at module top level, never nested inside `custom_kernel`.

You do not have to move everything into Triton. A correct submission that keeps
most of the computation in PyTorch and moves one hot region into a kernel scores;
one that fails to launch scores zero.

TriMul (outgoing) forward pass with a Triton‑accelerated cubic contraction.

The heavy tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched GEMM where each (batch, hidden) pair
forms an independent N×N matrix multiplication.
All surrounding LayerNorms, gating and linear projections are performed
with PyTorch; the core O(N³·H) work runs on the GPU via the
`batched_matmul_kernel` defined below.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr,                      # [BH, M, K]  matrix A
    b_ptr,                      # [BH, K, N]  matrix B (already transposed)
    c_ptr,                      # [BH, M, N]  output matrix C
    M, N, K,                    # dimensions (all equal to seq_len)
    stride_am, stride_ak,       # strides for A: (row, col)
    stride_bk, stride_bn,       # strides for B: (row, col)
    stride_cm, stride_cn,       # strides for C: (row, col)
    stride_batch_a, stride_batch_b, stride_batch_c,  # batch strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)          # row block index
    pid_n = tl.program_id(1)          # column block index
    pid_batch = tl.program_id(2)      # which (batch × hidden) matrix

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        k_offset = k * BLOCK_K

        # pointers for the current K‑tile
        a_ptrs = a_ptr + pid_batch * stride_batch_a \
                 + offs_m[:, None] * stride_am \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak

        b_ptrs = b_ptr + pid_batch * stride_batch_b \
                 + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

        # masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & ((k_offset + tl.arange(0, BLOCK_K))[None, :] < K)
        b_mask = ((k_offset + tl.arange(0, BLOCK_K))[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K)·(BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # write back
    c_ptrs = c_ptr + pid_batch * stride_batch_c \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn

    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul module (outgoing version).

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask:        torch.Tensor [B, N, N]       (float32, may be all‑ones)
            - weights:     dict of model parameters (see module definition)
            - config:      dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # --------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)   # [B, N, N, dim]

    # Work in FP16 for the expensive part
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left  = F.linear(x_fp16, left_proj_w)   # [B, N, N, H]
    right = F.linear(x_fp16, right_proj_w)  # [B, N, N, H]

    # --------------------------------------------------------------------
    # 3. Optional mask
    # --------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)
        left  = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------------
    # 4. Gating
    # --------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------
    # 5. Cubic contraction via batched GEMM (Triton)
    # --------------------------------------------------------------------
    B, N, _, H = left.shape  # left shape: [B, N, N, H]

    # reshape to (B, H, N, N) and transpose right to (k, j)
    A = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N) : rows=i, cols=k
    Bmat = right.permute(0, 3, 2, 1).contiguous()      # (B, H, N, N) : rows=k, cols=j

    # Collapse batch and hidden dimensions -> independent matrices
    BH = B * H
    A_flat = A.reshape(BH, N, N)
    B_flat = Bmat.reshape(BH, N, N)

    # Output buffer
    C_flat = torch.empty_like(A_flat, dtype=torch.float16, device=device)

    # Block sizes (tuned for B200, N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), BH)

    batched_matmul_kernel[grid](
        A_flat,
        B_flat,
        C_flat,
        N, N, N,                              # M, N, K
        A_flat.stride(1), A_flat.stride(2),   # stride_am, stride_ak
        B_flat.stride(1), B_flat.stride(2),   # stride_bk, stride_bn
        C_flat.stride(1), C_flat.stride(2),   # stride_cm, stride_cn
        A_flat.stride(0), B_flat.stride(0), C_flat.stride(0),  # batch strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout: [B, N, N, H]
    out = C_flat.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------------
    # 6. Post‑processing (LayerNorm on hidden dim, final gate, output proj)
    # --------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out_norm = out_norm * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"].to(device)  # (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)          # final shape [B, N, N, dim]

    return output

The TriMul("outgoing") operation requires updating an intermediate tensor using a contraction along the sequence dimensions, followed by gated operations. The main bottleneck is the O(N^3) contraction, which triples the operation time in the given example.

We address this by implementing a custom blocked matrix multiplication kernel optimized for this specific scenario:
- **Blocked Layout**: The kernel breaks the input into manageable blocks along the sequence dimensions.
- **Dependency Handling**: Clearly shows established dependencies between input matrices.
- **Mixed Precision**: Uses float32 for computations requiring high precision.
- **Efficient Memory Access**: Minimum data movement via proper blocking and strides.
- **Balanced Compute/Throughput**: Applies blocking that maximizes parallel execution

We normalize the input tensor along its channel dimension (using LayerNorm parameters from the host). Then map the tensors onto hidden_dim channels. We compute the outgoing multiplicative update by contracting over the sequence index (k) for each hidden channel (d).
