def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1 layernorm on dim
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)

    B,N,_,_ = x.shape

    # flatten x for linear projections
    x_flat = x.view(-1, dim)  # shape (B*N*N, dim)

    # 2 proj
    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    # gates
    left_gate = F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    right_gate = F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    out_gate = F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()

    # mask apply
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # convert to bf16 for contraction
    left_bf = left.to(torch.bfloat16).contiguous()
    right_bf = right.to(torch.bfloat16).contiguous()

    # allocate output bf16
    out_bf = torch.empty_like(left_bf)

    # get strides
    s_b_l, s_i_l, s_k_l, s_h_l = left_bf.stride()
    s_b_r, s_j_r, s_k_r, s_h_r = right_bf.stride()
    s_b_o, s_i_o, s_j_o, s_h_o = out_bf.stride()

    # wrapper for kernel launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * hidden_dim)
    trimul_kernel[grid](
        left_bf,
        right_bf,
        out_bf,
        B,
        N,
        hidden_dim,
        s_b_l, s_i_l, s_k_l, s_h_l,
        s_b_r, s_j_r, s_k_r, s_h_r,
        s_b_o, s_i_o, s_j_o, s_h_o,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # convert back to float32
    out = out_bf.to(torch.float32)

    # hidden_dim layernorm (torch)
    out_norm = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=1e-5)

    # apply out gate
    out_scaled = out_norm * out_gate

    # final projection
    out_scaled_flat = out_scaled.view(-1, hidden_dim)
    final = F.linear(out_scaled_flat, weights["to_out.weight"]).view(B, N, N, dim)

    return final
```

```
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hb = tl.program_id(2)

    # compute i and j start indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # decode batch and hidden index
    b_idx = pid_hb // H
    h_idx = pid_hb % H

    # compute base pointer offsets
    left_base = b_idx * stride_l_b + h_idx * stride_l_h
    right_base = b_idx * stride_r_b + h_idx * stride_r_h
    out_base = b_idx * stride_o_b + h_idx * stride_o_h

    # offsets
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # k loop
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        k_start = kk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile
        a_ptrs = left_ptr + left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0, dtype=tl.bfloat16)

"""
TriMul (outgoing) forward pass with a Triton kernel for the O(N³) contraction.

Algorithm
---------
1. Layer‑norm over the last feature dimension (dim) of the input tensor.
2. Linear projections → left / right (hidden_dim) and three gating vectors
   (left_gate, right_gate, out_gate).  Apply mask (if present) and element‑wise
   gate multiplication.
3. Cast left/right to bfloat16 and compute the pairwise contraction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication (left @ rightᵀ) performed by a
   custom Triton kernel.  The kernel treats each (batch, hidden) slice as a
   separate matrix, uses tiling (BLOCK_M × BLOCK_N × BLOCK_K) and accumulates
   in float32.
4. Cast the result back to float32 and apply a LayerNorm over the hidden
   dimension (torch implementation, which is already highly optimized).
5. Multiply by the out_gate and apply the final linear projection back to `dim`.

Only the heavy O(N³) contraction is written in Triton; all other operations
use fast PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_kernel(
    left_ptr,        # *bfloat16  [B, N, N, H]
    right_ptr,       # *bfloat16  [B, N, N, H]
    out_ptr,         # *bfloat16  [B, N, N, H]
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # Program IDs for the output tile (i‑tile, j‑tile) and the combined (batch, hidden) index
    pid_i = tl.program_id(0)          # tile along output rows i
    pid_j = tl.program_id(1)          # tile along output cols j
    pid_hb = tl.program_id(2)         # combined batch * hidden index

    # start offsets of this tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # decode batch index b and hidden index h from pid_hb
    b_idx = pid_hb // H
    h_idx = pid_hb % H

    # base pointer offsets for this (b, h) slice
    left_base  = b_idx * stride_l_b + h_idx * stride_l_h
    right_base = b_idx * stride_r_b + h_idx * stride_r_h
    out_base   = b_idx * stride_o_b + h_idx * stride_o_h

    # offsets within the tile
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        k_start = kk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + left_base + \
                 offs_i[:, None] * stride_l_i + \
                 offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.bfloat16)

        # right tile (transposed): shape (BLOCK_K, BLOCK_N)
        # we need right[b, j, k, h] -> treat k as rows, j as columns
        b_ptrs = right_ptr + right_base + \
                 offs_k[:, None] * stride_r_k + \
                 offs_j[None, :] * stride_r_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0,
                    dtype=tl.bfloat16)

        acc += tl.dot(a, b)   # accumulates in float32

    # write the result back (cast to bfloat16)
    c_ptrs = out_ptr + out_base + \
             offs_i[:, None] * stride_o_i + \
             offs_j[None, :] * stride_o_j
    tl.store(c_ptrs, acc.to(tl.bfloat16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be ignored)
            - weights: dict of weight tensors
            - config: dict containing "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input layer‑norm over the last dimension
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 2. Linear projections & gating (float32)
    x_flat = x.view(-1, dim)                     # [B*N*N, dim]

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    left_gate = F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    right_gate = F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    out_gate = F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()

    # ------------------------------------------------------------------
    # 3. Apply mask (if any) and gates
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(dtype)      # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Pairwise contraction (batched GEMM) using Triton
    left_bf = left.to(torch.bfloat16).contiguous()
    right_bf = right.to(torch.bfloat16).contiguous()
    out_bf = torch.empty_like(left_bf)   # shape [B, N, N, hidden_dim], bfloat16

    # strides (in element units)
    s_bl, s_il, s_kl, s_hl = left_bf.stride()
    s_br, s_jr, s_kr, s_hr = right_bf.stride()
    s_bo, s_io, s_jo, s_ho = out_bf.stride()

    # launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # i‑tiles
        triton.cdiv(N, BLOCK_N),   # j‑tiles
        B * hidden_dim,            # combined batch‑hidden dimension
    )

    trimul_kernel[grid](
        left_bf,
        right_bf,
        out_bf,
        B,
        N,
        hidden_dim,
        s_bl, s_il, s_kl, s_hl,
        s_br, s_jr, s_kr, s_hr,
        s_bo, s_io, s_jo, s_ho,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # convert back to float32 for the rest of the pipeline
    out = out_bf.to(dtype)

    # ------------------------------------------------------------------
    # 5. LayerNorm over the hidden dimension (torch implementation)
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 6. Apply outgoing gate
    out_scaled = out_norm * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to `dim`
    out_scaled_flat = out_scaled.view(-1, hidden_dim)
    final = F.linear(out_scaled_flat, weights["to_out.weight"]).view(B, N, N, dim)

    return final
