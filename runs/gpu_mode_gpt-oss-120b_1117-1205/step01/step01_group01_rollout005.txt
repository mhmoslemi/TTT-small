def custom_kernel(data):
    x, mask, weights, config = data
    device = x.device

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]

    # LayerNorm
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for heavy ops
    x_half = x.half()

    # Linear Proj
    left_proj_w = weights["left_proj.weight"].to(device).half()
    right_proj_w = weights["right_proj.weight"].to(device).half()
    left = F.linear(x_half, left_proj_w)  # [B,N,N,hidden_dim]
    right = F.linear(x_half, right_proj_w)

    # Mask if provided
    if mask is not None:
        mask_half = mask.unsqueeze(-1).to(device=device, dtype=torch.float16)
        left = left * mask_half
        right = right * mask_half

    # Gates
    left_gate_w = weights["left_gate.weight"].to(device).half()
    right_gate_w = weights["right_gate.weight"].to(device).half()
    out_gate_w = weights["out_gate.weight"].to(device).half()

    left_gate = torch.sigmoid(F.linear(x_half, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_half, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_half, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # Contraction – batched GEMM
    # Rearrange dimensions to (B, hidden, N, N) for left and (B, hidden, N, N) for right (transposed)
    left_perm = left.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) (k on dim2, j on dim3)

    BH = B * hidden_dim
    A = left_perm.reshape(BH, N, N)
    Bmat = right_perm.reshape(BH, N, N)

    C = torch.empty_like(A, dtype=torch.float16, device=device)

    # block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), BH)

    batched_matmul_kernel[grid](
        A,
        Bmat,
        C,
        N, N, N,
        A.stride(1), A.stride(2),
        Bmat.stride(1), Bmat.stride(2),
        C.stride(1), C.stride(2),
        A.stride(0), Bmat.stride(0), C.stride(0),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back
    out_hidden = C.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden_dim]

    # Post-processing
    out_fp32 = out_hidden.to(torch.float32)

    # to_out_norm
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out_norm = out_norm * out_gate.to(torch.float32)

    # final linear projection
    to_out_w = weights["to_out.weight"].to(device)  # shape (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)

    return output
```

"""
TriMul (outgoing) forward pass with Triton‑accelerated batched matrix multiplication.

The heavy O(N³·H) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed as a batched GEMM: for each (b,d) we multiply the
(N,N) matrix left[b,:, :,d] with the transpose of right[b,:, :,d].
All other operations (LayerNorm, linear projections, gating) are
handled in PyTorch (FP16 for the compute‑intensive part). The kernel
operates on half‑precision tensors and accumulates in FP32 for accuracy.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr,                # [BH, M, K] left tensor (FP16)
    b_ptr,                # [BH, K, N] right tensor transposed (FP16)
    c_ptr,                # [BH, M, N] output tensor (FP16)
    M, N, K,              # matrix dimensions (all = seq_len)
    stride_am, stride_ak, # strides of A: row, col
    stride_bk, stride_bn, # strides of B: row, col
    stride_cm, stride_cn, # strides of C: row, col
    stride_batch_a, stride_batch_b, stride_batch_c,  # batch stride
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)      # block row index
    pid_n = tl.program_id(1)      # block col index
    pid_batch = tl.program_id(2)  # batch*hidden index

    # Offsets within the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_blocks):
        k_off = k * BLOCK_K
        # Pointer offsets for the current K‑tile
        a_ptrs = a_ptr + pid_batch * stride_batch_a \
                 + offs_m[:, None] * stride_am \
                 + (k_off + tl.arange(0, BLOCK_K))[None, :] * stride_ak

        b_ptrs = b_ptr + pid_batch * stride_batch_b \
                 + (k_off + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

        # OOB masks
        a_mask = (offs_m[:, None] < M) & ((k_off + tl.arange(0, BLOCK_K))[None, :] < K)
        b_mask = ((k_off + tl.arange(0, BLOCK_K))[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # FP16
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # FP16

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K)·(BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # Write result back to C (FP16)
    c_ptrs = c_ptr + pid_batch * stride_batch_c \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn

    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask:         torch.Tensor of shape [B, N, N] (float32/bool) or None
            - weights:      dict of model parameters, keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config:       dict with "dim" (C) and "hidden_dim" (H)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # [B, N, N, dim]

    # Work in fp16 for the heavy path
    x_h = x.half()

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).half()
    right_proj_w = weights["right_proj.weight"].to(device).half()

    left = F.linear(x_h, left_proj_w)   # [B, N, N, H]
    right = F.linear(x_h, right_proj_w)  # [B, N, N, H]

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_h = mask.unsqueeze(-1).to(device=device, dtype=torch.float16)
        left = left * mask_h
        right = right * mask_h

    # -----------------------------------------------------------------
    # 4. Gating
    # -----------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).half()
    right_gate_w = weights["right_gate.weight"].to(device).half()
    out_gate_w = weights["out_gate.weight"].to(device).half()

    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Cubic contraction via batched GEMM (Triton)
    # -----------------------------------------------------------------
    # reshape to (B, H, N, N) and make right transposed on the last two axes
    left_perm = left.permute(0, 3, 1, 2).contiguous()     # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # (B, H, N, N) – rows = k, cols = j

    BH = B * hidden_dim
    A = left_perm.reshape(BH, N, N)     # [BH, M, K]
    Bmat = right_perm.reshape(BH, N, N)  # [BH, K, N] (already transposed)

    C = torch.empty_like(A, dtype=torch.float16, device=device)  # [BH, M, N]

    # Tuned block sizes for fp16 GEMM on B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), BH)

    batched_matmul_kernel[grid](
        A,
        Bmat,
        C,
        N, N, N,                               # M, N, K
        A.stride(1), A.stride(2),               # stride_am, stride_ak
        Bmat.stride(1), Bmat.stride(2),         # stride_bk, stride_bn
        C.stride(1), C.stride(2),               # stride_cm, stride_cn
        A.stride(0), Bmat.stride(0), C.stride(0),  # batch strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore [B, N, N, H] layout
    out_hidden = C.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # -----------------------------------------------------------------
    # 6. Post‑processing (LayerNorm, final gate, output projection)
    # -----------------------------------------------------------------
    out_fp32 = out_hidden.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    out_norm = F.layer_norm(out_fp32, (hidden_dim,),
                           weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply output gate
    out_norm = out_norm * out_gate.to(torch.float32)

    # Final linear projection back to original channel dimension
    to_out_w = weights["to_out.weight"].to(device)   # shape (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)            # [B, N, N, dim]

    return output
