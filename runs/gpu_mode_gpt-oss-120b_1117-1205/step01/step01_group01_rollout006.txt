@triton.jit
def batched_matmul_kernel(
    a_ptr,                      # [BH, M, K]  matrix A
    b_ptr,                      # [BH, K, N]  matrix B (already transposed)
    c_ptr,                      # [BH, M, N]  output matrix C
    M, N, K,                    # dimensions (all equal to seq_len)
    stride_am, stride_ak,       # strides for A: (row, col)
    stride_bk, stride_bn,       # strides for B: (row, col)
    stride_cm, stride_cn,       # strides for C: (row, col)
    stride_batch_a, stride_batch_b, stride_batch_c,  # batch strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)          # row block index
    pid_n = tl.program_id(1)          # column block index
    pid_batch = tl.program_id(2)      # which (batch × hidden) matrix

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        k_offset = k * BLOCK_K

        # pointers for the current K‑tile
        a_ptrs = a_ptr + pid_batch * stride_batch_a \
                 + offs_m[:, None] * stride_am \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak

        b_ptrs = b_ptr + pid_batch * stride_batch_b \
                 + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

        # masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & ((k_offset + tl.arange(0, BLOCK_K))[None, :] < K)
        b_mask = ((k_offset + tl.arange(0, BLOCK_K))[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

"""
TriMul (outgoing) forward pass with Triton‑accelerated cubic contraction.

The heavy O(N³·H) operation:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is realized as a batched GEMM where each (batch, hidden) pair forms an
independent N×N matrix multiplication. All surrounding LayerNorms,
linear projections and gates are handled in PyTorch (FP16 where possible)
while the contraction runs in a custom Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr,               # [BH, M, K]  left matrix
    b_ptr,               # [BH, K, N]  right matrix (already transposed)
    c_ptr,               # [BH, M, N]  output matrix
    M, N, K,             # matrix dimensions (all equal to seq_len)
    stride_am, stride_ak,   # strides for A: row, col
    stride_bk, stride_bn,   # strides for B: row, col
    stride_cm, stride_cn,   # strides for C: row, col
    stride_batch_a, stride_batch_b, stride_batch_c,   # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_m = tl.program_id(0)      # row‑tile index
    pid_n = tl.program_id(1)      # col‑tile index
    pid_batch = tl.program_id(2)  # which (batch × hidden) matrix

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in FP32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_blocks):
        k_offset = k * BLOCK_K

        # pointers for the current K‑tile of A and B
        a_ptrs = a_ptr + pid_batch * stride_batch_a \
                 + offs_m[:, None] * stride_am \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak

        b_ptrs = b_ptr + pid_batch * stride_batch_b \
                 + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

        # masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & ((k_offset + tl.arange(0, BLOCK_K))[None, :] < K)
        b_mask = ((k_offset + tl.arange(0, BLOCK_K))[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # accumulate the dot product of the tiles
        acc += tl.dot(a, b)

    # write back the result
    c_ptrs = c_ptr + pid_batch * stride_batch_c \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn

    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask:         torch.Tensor of shape [B, N, N] (float32) or None
            - weights:      dict of model parameters
            - config:       dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------------
    # 1. LayerNorm on the input (dim)
    # --------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)  # [B, N, N, dim]

    # Cast to FP16 for the heavy part
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------------
    # 2. Linear projections (no bias) to hidden dimension
    # --------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)   # (H, dim)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # [B, N, N, H]
    right = F.linear(x_fp16, right_proj_w) # [B, N, N, H]

    # --------------------------------------------------------------------
    # 3. Masking (if a mask is provided)
    # --------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------------
    # 4. Gating (sigmoid on linear transforms of the normalized input)
    # --------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))    # will be used later

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------
    # 5. Cubic contraction via batched GEMM (Triton)
    # --------------------------------------------------------------------
    B, N, _, H = left.shape  # B=batch size, H=hidden_dim

    # Rearrange tensors so that each (batch, hidden) pair is an independent matrix
    A = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N) – rows=i, cols=k
    B_mat = right.permute(0, 3, 2, 1).contiguous()    # (B, H, N, N) – rows=k, cols=j (transposed)

    BH = B * H
    A_flat = A.reshape(BH, N, N)      # [BH, M, K]
    B_flat = B_mat.reshape(BH, N, N)  # [BH, K, N]

    # Output buffer
    C_flat = torch.empty_like(A_flat, dtype=torch.float16, device=device)  # [BH, M, N]

    # Block sizes (tuned for typical N values; all test N are multiples of 64)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # number of row tiles
        triton.cdiv(N, BLOCK_N),   # number of col tiles
        BH,                        # one grid entry per (batch * hidden) matrix
    )

    batched_matmul_kernel[grid](
        A_flat,
        B_flat,
        C_flat,
        N, N, N,                                   # M, N, K
        A_flat.stride(1), A_flat.stride(2),        # stride_am, stride_ak
        B_flat.stride(1), B_flat.stride(2),        # stride_bk, stride_bn
        C_flat.stride(1), C_flat.stride(2),        # stride_cm, stride_cn
        A_flat.stride(0), B_flat.stride(0), C_flat.stride(0),   # batch strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout: [B, N, N, H]
    out = C_flat.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # FP16

    # --------------------------------------------------------------------
    # 6. Post‑processing (LayerNorm over hidden dim, final gating, output proj)
    # --------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)   # [B, N, N, H]

    # to_out_norm (LayerNorm on hidden dimension)
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply final gate (broadcasted)
    out_norm = out_norm * out_gate.to(torch.float32)

    # Final linear projection back to original dim
    to_out_w = weights["to_out.weight"].to(device)  # (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)           # [B, N, N, dim] (float32)

    return output
