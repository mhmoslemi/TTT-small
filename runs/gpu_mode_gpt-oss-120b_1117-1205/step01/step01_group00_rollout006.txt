@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile
        a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile
        b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass with a fused Triton batched GEMM.

The heavy N³ contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is equivalent to a batched matrix multiplication
    out[b,d] = left[b,d] @ right[b,d].T
where left/right are reshaped to (B*hidden, N, N).
A custom Triton kernel performs this batched GEMM in half precision
with float‑accumulation. All other operations (LayerNorms, linear
projections, gates, mask) use PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ B) where A: (batch, M, K), B: (batch, K, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)           # batch dimension
    pid_m   = tl.program_id(1)           # block row
    pid_n   = tl.program_id(2)           # block column

    # Compute tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Compute tile multiplication
        acc += tl.dot(a, b)

    # Write back result
    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def triton_batched_gemm(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B where
        A: (batch, M, K)  half‑precision
        B: (batch, K, N)  half‑precision
    Returns:
        C: (batch, M, N) half‑precision
    """
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides (in elements)
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tile sizes – chosen for good occupancy on B200 (Tesla T4‑class)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_gemm_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor : torch.Tensor [B, N, N] (bool or numeric)
            - weights_dict: mapping of parameter names to torch.Tensor
            - config_dict : must contain 'dim', 'hidden_dim', and optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    # ------------------------------------------------------------------
    # Configuration
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Weight shortcuts (store in appropriate dtype)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)

    left_gate_weight  = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight   = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] float32

    x_fp16 = x.to(torch.float16)  # for subsequent linear ops

    # ------------------------------------------------------------------
    # 2. Linear projections (half‑precision)
    # ------------------------------------------------------------------
    left  = F.linear(x_fp16, left_proj_weight)   # [B, N, N, hidden]
    right = F.linear(x_fp16, right_proj_weight)  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left  = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 4. Gates (half‑precision)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_weight).sigmoid()   # [B,N,N,hidden]
    right_gate = F.linear(x_fp16, right_gate_weight).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_weight).sigmoid()   # same shape

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton batched GEMM
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    => out = left @ right.T   (per hidden channel)
    # ------------------------------------------------------------------
    # Move hidden dim to 2nd axis for easy flattening
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N) -> (i,k)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, hidden, N, N)

    # Transpose right on the last two dimensions to match GEMM pattern
    right_t = right_perm.transpose(-2, -1).contiguous()  # (B, hidden, N, N) -> (k,j)

    B, H, N, _ = left_perm.shape
    batch_hidden = B * H

    left_flat  = left_perm.view(batch_hidden, N, N)   # (B*H, N, N)
    right_flat = right_t.view(batch_hidden, N, N)    # (B*H, N, N)

    # Perform batched GEMM in half precision
    out_flat = triton_batched_gemm(left_flat, right_flat)  # (B*H, N, N) fp16

    # Release memory of large intermediates early
    del left, right, left_perm, right_perm, right_t, left_flat, right_flat
    torch.cuda.empty_cache()

    # Reshape back to [B, N, N, hidden]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden]
    out = out.to(torch.float32)  # subsequent ops are in FP32

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate.to(torch.float32)  # broadcast over hidden dim
    out = F.linear(out, to_out_weight)      # [B, N, N, dim] (float32)

    return out
