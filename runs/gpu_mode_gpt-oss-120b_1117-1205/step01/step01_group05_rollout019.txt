custom_kernel(data):

    input_tensor, mask, weights, config = data

    # extract dimensions
    B, N, N2, dim = input_tensor.shape   # N2 same as N
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm on input
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = input_tensor

    # x_norm
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_w + norm_b  # shape [B,N,N,dim]

    # Cast to fp16 for projection/gates
    x_h = x_norm.to(torch.float16)

    # Load weights
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float32)  # keep float32 for LN
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float32)
    to_out_weight = weights["to_out.weight"].to(torch.float32)  # final linear weight

    # Linear proj
    left = F.linear(x_h, left_proj_weight)       # [B,N,N,hidden]
    right = F.linear(x_h, right_proj_weight)

    # mask handling
    if mask is not None:
        # mask shape [B,N,N]
        mask_exp = mask.unsqueeze(-1).to(x_h.dtype)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # Gate values
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Heavy contraction via Triton
    # Permute and reshape
    B_, N_, _, H_ = left.shape # B_, N_, N_, H_
    left_perm = left.permute(0,3,1,2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0,3,2,1).contiguous() # [B, H, N, N] (transpose)
    # flatten batch*hidden
    batch_h = B_ * H_
    left_flat = left_perm.view(batch_h, N_, N_)
    right_flat = right_perm.view(batch_h, N_, N_)

    # allocate output
    out_flat = torch.empty((batch_h, N_, N_), dtype=torch.float16, device=left_flat.device)

    # compute strides
    stride_az, stride_am, stride_ak = left_flat.stride()
    stride_bz, stride_bk, stride_bn = right_flat.stride()
    stride_cz, stride_cm, stride_cn = out_flat.stride()

    # set block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (batch_h, triton.cdiv(N_, BLOCK_M), triton.cdiv(N_, BLOCK_N))
    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        N_, N_, N_,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape output back to [B,N,N,hidden]
    out = out_flat.view(B_, H_, N_, N_).permute(0, 2, 3, 1).contiguous()  # shape [B,N,N,hidden] fp16

    # Convert to float32 for LN and final linear
    out_f32 = out.to(torch.float32)

    # Output LayerNorm (over hidden_dim)

    mean_o = out_f32.mean(dim=-1, keepdim=True)

    out_norm = (out_f32 - mean_o) / torch.sqrt(var_o + eps)

    out_norm = out_norm * out_gate.to(torch.float32)

    # Final linear projection to dim
    output = F.linear(out_norm, to_out_weight) # weight shape (dim, hidden)
    return output

@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
   batch_id = tl.program_id(0)
   pid_m = tl.program_id(1)
   pid_n = tl.program_id(2)

   # compute offsets
   offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
   mask_m = offs_m < M
   mask_n = offs_n < N

   # pointers to C tile
   c_ptrs = C_ptr + batch_id*stride_cz + offs_m[:,None]*stride_cm + offs_n[None,:]*stride_cn
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   # loop over K dimension
   for k in range(0, tl.cdiv(K, BLOCK_K)):
       offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
       mask_k = offs_k < K

       # load A tile (M x K)
       a_ptrs = A_ptr + batch_id*stride_az + offs_m[:,None]*stride_am + offs_k[None,:]*stride_ak
       a = tl.load(a_ptrs, mask=mask_m[:,None] & mask_k[None,:], other=0.0).to(tl.float32)

       # load B tile (K x N)
       b_ptrs = B_ptr + batch_id*stride_bz + offs_k[:,None]*stride_bk + offs_n[None,:]*stride_bn
       b = tl.load(b_ptrs, mask=mask_k[:,None] & mask_n[None,:], other=0.0).to(tl.float32)

       acc += tl.dot(a, b)

   # write back
   c = acc.to(tl.float16)
   tl.store(c_ptrs, c, mask=mask_m[:,None] & mask_n[None,:])

Note: The B_ptr offset uses offs_k as row index, and offs_n as column index. That's correct.

"""
TriMul (outgoing) forward implementation with a Triton batched GEMM kernel.
The heavy tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is rewritten as a batched matrix multiplication of shape
    (B*hidden, N, N) @ (B*hidden, N, N).T
and executed in a custom Triton kernel. All other operations (LayerNorm,
linear projections, gates, final LayerNorm and linear) are performed in
PyTorch using mixed‑precision (fp16 for the large GEMM for bandwidth
reduction, fp32 for the normalizations and final output).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: C = A @ B , where
        A : (batch, M, K)   (row‑major)
        B : (batch, K, N)   (row‑major)
        C : (batch, M, N)   (row‑major, stored as fp16)
    Accumulation is in fp32.
    """
    batch_id = tl.program_id(0)        # combined batch*hidden dimension
    pid_m = tl.program_id(1)           # block row
    pid_n = tl.program_id(2)           # block column

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the output tile
    c_ptrs = (
        C_ptr
        + batch_id * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile (M x K)
        a_ptrs = (
            A_ptr
            + batch_id * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # B tile (K x N)
        b_ptrs = (
            B_ptr
            + batch_id * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        acc += tl.dot(a, b)

    # Store the result as fp16 to save bandwidth
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Args:
        data: Tuple (input, mask, weights, config)
            input  : torch.Tensor of shape [B, N, N, dim] (float32)
            mask   : torch.Tensor of shape [B, N, N]   (bool/float) or None
            weights: dict of torch.Tensors for the linear / norm parameters
            config : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Input LayerNorm (over the last dimension)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(torch.float32)
    norm_b = weights["norm.bias"].to(torch.float32)

    x = input_tensor
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_w + norm_b          # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Mixed‑precision linear projections and gates
    # ------------------------------------------------------------------
    x_h = x_norm.to(torch.float16)  # fp16 for the large matmuls

    # projection weights (fp16)
    w_left = weights["left_proj.weight"].to(torch.float16)    # (hidden_dim, dim)
    w_right = weights["right_proj.weight"].to(torch.float16)

    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    # linear projections
    left = F.linear(x_h, w_left)    # [B, N, N, hidden_dim] fp16
    right = F.linear(x_h, w_right)

    # optional mask (broadcast over hidden_dim)
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x_h.dtype)   # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # gates (sigmoid, still fp16)
    left_gate = torch.sigmoid(F.linear(x_h, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_h, w_right_gate))
    out_gate = torch.sigmoid(F.linear(x_h, w_out_gate))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Heavy contraction: out = left @ right^T over the middle index
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # B, N, N, hidden_dim
    # permute to (B, hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()        # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()      # [B, H, N, N] (transpose k↔j)

    batch_h = B * hidden_dim
    left_mat = left_perm.view(batch_h, N, N)                 # (batch_h, M=N, K=N)
    right_mat = right_perm.view(batch_h, N, N)               # (batch_h, K=N, N=N)

    # allocate output in fp16 (will be cast to fp32 later)
    out_mat = torch.empty((batch_h, N, N), dtype=torch.float16, device=device)

    # strides for the three tensors
    stride_az, stride_am, stride_ak = left_mat.stride()
    stride_bz, stride_bk, stride_bn = right_mat.stride()
    stride_cz, stride_cm, stride_cn = out_mat.stride()

    # tile sizes (tuned for A100 / RTX‑B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_h,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # reshape back to [B, N, N, hidden_dim] (still fp16)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_f32 = out.to(torch.float32)   # now fp32 for the following LN

    # ------------------------------------------------------------------
    # Output LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float32)

    mean_o = out_f32.mean(dim=-1, keepdim=True)
    var_o = out_f32.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out_f32 - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * to_out_norm_w + to_out_norm_b

    # Apply the output gate (convert gate to fp32)
    out_norm = out_norm * out_gate.to(torch.float32)

    # ------------------------------------------------------------------
    # Final projection back to the input dimension
    # ------------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float32)   # (dim, hidden_dim)
    output = F.linear(out_norm, w_to_out)                  # [B, N, N, dim]

    return output
