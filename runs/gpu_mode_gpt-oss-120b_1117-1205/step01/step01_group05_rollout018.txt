@triton.jit
def batched_matmul_outgoing_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    b = tl.program_id(0)   # batch dimension (B*H)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N
    # pointer to output tile
    c_ptrs = C_ptr + b * stride_cz + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile
        a_ptrs = A_ptr + b * stride_az + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        # B tile
        b_ptrs = B_ptr + b * stride_bz + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b_tile = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # multiply-accumulate
        acc += tl.dot(a, b_tile)

    # store result
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm forward
    gamma = weights["norm.weight"]
    beta = weights["norm.bias"]
    x_norm = F.layer_norm(input_tensor, (dim,), weight=gamma, bias=beta, eps=eps)

    # Convert to half for linear ops
    x_h = x_norm.to(torch.float16)

    # Linear projections
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # Masking (if mask provided)
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x_h.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # reshape for batched matmul (outgoing)
    B, N, _, H = left.shape
    # Permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Transpose right's last two axes
    right_perm_T = right_perm.transpose(-2, -1).contiguous()
    # flatten batch*H dimension
    batch_dim = B * H
    left_mat = left_perm.view(batch_dim, N, N)
    right_mat_T = right_perm_T.view(batch_dim, N, N)

    # output mat
    out_mat = torch.empty((batch_dim, N, N), dtype=torch.float16, device=device)

    # Kernel launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_matmul_outgoing_kernel[grid](
        left_mat,

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointer to output tile
    c_ptrs = C_ptr + batch_idx * stride_cz + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """Forward pass of the TriMul (outgoing) module with a Triton batched matmul."""
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----- 1. Input LayerNorm (float32) -----
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ----- 2. Projection & gating (half) -----
    x_h = x.to(torch.float16)

    # Linear projections
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # Apply mask if present (mask shape [B,N,N])
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x_h.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # Gates (half)
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ----- 3. Batched GEMM via Triton (outgoing) -----
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B, H, N, N) and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # transpose the last two dimensions of right (K dimension)
    right_perm_T = right_perm.transpose(-2, -1).contiguous()
    # flatten batch * hidden_dim
    batch_dim = B * H
    left_mat = left_perm.view(batch_dim, N, N)
    right_mat_T = right_perm_T.view(batch_dim, N, N)

    # Allocate output tensor (half)
    out_mat = torch.empty((batch_dim, N, N), dtype=torch.float16, device=device)

    # Triton launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_matmul_outgoing_kernel[grid](
        left_mat,
        right_mat_T,
        out_mat,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat_T.stride(0), right_mat_T.stride(1), right_mat_T.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ----- 4. Restore shape and convert to float32 -----
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out_f32 = out.to(torch.float32)

    # ----- 5. Output LayerNorm (float32) -----
    out_norm = F.layer_norm(
        out_f32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ----- 6. Apply output gate -----
    out_norm = out_norm * out_gate.to(torch.float32)

    # ----- 7. Final projection back to dim (float32) -----
    output = F.linear(out_norm, weights["to_out.weight"])

    return output

"""
TriMul (outgoing) forward pass with a Triton batched GEMM.

The heavy part of the module is the tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication of shape (B*H, N, N) @ (B*H, N, N)^T.
We implement this contraction with a custom Triton kernel that processes
each batch‑channel pair as an independent GEMM.  All other operations
(layer‑norms, linear projections, gating) are performed in PyTorch.
The kernel works on fp16 inputs and accumulates in fp32, then stores fp16.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_outgoing_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM for the outgoing TriMul:
    C = A @ B   where A: (batch, M, K), B: (batch, K, N), C: (batch, M, N)
    All tensors are stored in row‑major layout, A/B/C are fp16, accumulation is fp32.
    """
    batch_idx = tl.program_id(0)   # batch * hidden_dim index
    pid_m = tl.program_id(1)       # block row
    pid_n = tl.program_id(2)       # block column

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointer to the output tile
    c_ptrs = (
        C_ptr
        + batch_idx * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a_ptrs = (
            A_ptr
            + batch_idx * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # Load B tile (K x N)
        b_ptrs = (
            B_ptr
            + batch_idx * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # Store result (fp16)
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """Forward pass of the TriMul (outgoing) module using a Triton kernel.

    Args:
        data: Tuple (input, mask, weights, config)
            - input: torch.Tensor [B, N, N, dim] (float32)
            - mask:  torch.Tensor [B, N, N] (float32/bool) or None
            - weights: dict of model parameters (torch.Tensors)
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # -------------------------------------------------------------
    # 2. Linear projections & gates (fp16)
    # -------------------------------------------------------------
    x_h = x.to(torch.float16)

    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # optional mask
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x_h.dtype)  # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------
    # 3. Batched GEMM (outgoing) via Triton
    # -------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # Reshape to (B, H, N, N) and make contiguous for view
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Transpose the last two axes of right (K dimension)
    right_perm_T = right_perm.transpose(-2, -1).contiguous()

    batch_dim = B * H
    left_mat = left_perm.view(batch_dim, N, N)
    right_mat_T = right_perm_T.view(batch_dim, N, N)

    out_mat = torch.empty((batch_dim, N, N), dtype=torch.float16, device=device)

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_matmul_outgoing_kernel[grid](
        left_mat,
        right_mat_T,
        out_mat,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat_T.stride(0), right_mat_T.stride(1), right_mat_T.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # Restore (B, N, N, H) layout and cast to fp32
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out_f32 = out.to(torch.float32)

    # -------------------------------------------------------------
    # 4. Output LayerNorm (float32)
    # -------------------------------------------------------------
    out_norm = F.layer_norm(
        out_f32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # -------------------------------------------------------------
    # 5. Apply output gate and final linear projection
    # -------------------------------------------------------------
    out_norm = out_norm * out_gate.to(torch.float32)
    output = F.linear(out_norm, weights["to_out.weight"])

    return output
