@triton.jit
def triton_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # ensure within bounding dims
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch_ptr = a_ptr + batch_id * stride_batch_a
    b_batch_ptr = b_ptr + batch_id * stride_batch_b
    c_batch_ptr = c_ptr + batch_id * stride_batch_c

    # Initialize accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Starting pointers for each k tile
    a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = b_batch_ptr + (tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load A and B with mask
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < K - k*BLOCK_K)[None, :],
                    other=0.0)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K) < K - k*BLOCK_K)[:, None] & mask_n[None, :],
                    other=0.0)
        # compute dot product
        acc += tl.dot(a, b)  # result float32
        # increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store result
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    (input_tensor, mask, weights, config) = data

    device = input_tensor.device

    B,N,_dim = input.

    dim = config["dim"]; hidden = config["hidden_dim"]; Ensure match.

    # Step 1: LayerNorm: using given norm weight & bias (broadcast).
    eps = 1e-5
    # compute mean and var across dim (last axis)
    # input shape [B,N,N,dim]
    # Use torch.mean for mean across last dim, keep dim.
    mu = input_tensor.mean(dim=-1, keepdim=True)
    var = ((input_tensor - mu) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (input_tensor - mu) * torch.rsqrt(var + eps)
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]

    # Step 2: Linear projections gating:
    x_f16 = x_norm.to(torch.float16)

    # compute left projection: F.linear(x_f16, w_left) produce shape [B,N,N,hidden]
    left = torch.nn.functional.linear(x_f16, weights["left_proj.weight"].to(torch.float16))
    right = torch.nn.functional.linear(x_f16, weights["right_proj.weight"].to(torch.float16))

    # mask handling
    if mask is None:
        mask_f16 = torch.ones(B,N,N, device=device, dtype=torch.float16)
    else:
        mask_f16 = mask.to(torch.float16)
    mask_f16 = mask_f16.unsqueeze(-1) # shape [B,N,N,1]
    left = left * mask_f16
    right = right * mask_f16

    # gates:
    left_gate = torch.nn.functional.linear(x_f16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x_f16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x_f16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Step 3: core multiplication via kernel.
    # Permute for proper shape
    # left: [B,N,N,hidden] -> (B, hidden, N, N)
    left = left.permute(0,3,1,2).contiguous()
    # right: [B,N,N,hidden] -> we need transpose last dims: (B, hidden, N, N) where axes (k,j)
    right = right.permute(0,3,2,1).contiguous()  # now shape (B, hidden, N, N) with first N = k, second = j

    total_batches = B * hidden
    # reshape to (total_batches, N, N)
    left_mat = left.reshape(total_batches, N, N)
    right_mat = right.reshape(total_batches, N, N)

    # allocate output
    out_mat = torch.empty_like(left_mat)  # fp16

    # Choose block sizes based on N maybe:
    if N <= 256:
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 32
    else:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 64

    # However, we also need to consider hidden dimension (total_batches) maybe small; we can set grid accordingly.
    grid = (
        total_batches,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    # Launch
    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        left_mat.stride(1), left_mat.stride(2),  # stride_am, stride_ak
        right_mat.stride(1), right_mat.stride(2), # stride_bk, stride_bn
        out_mat.stride(1), out_mat.stride(2),  # stride_cm, stride_cn
        left_mat.stride(0), right_mat.stride(0), out_mat.stride(0),  # batch strides
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # reshape back
    out = out_mat.reshape(B, hidden, N, N).permute(0, 2, 3, 1)  # shape [B,N,N,hidden]

    # to_out_norm:
    # compute layernorm on hidden dim axis
    eps = 1e-5
    # out is fp16? out is fp16 by default; we can convert to fp32 for LN.
    out_f32 = out.to(torch.float32)
    mean_out = out_f32.mean(dim=-1, keepdim=True)
    var_out = ((out_f32 - mean_out) ** 2).mean(dim=-1, keepdim=True)
    out_norm = (out_f32 - mean_out) * torch.rsqrt(var_out + eps)
    # apply weight and bias
    out_norm = out_norm * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    # out_gate originally fp16; convert to fp32
    out_gate_f32 = out_gate.to(torch.float32)
    out_norm = out_norm * out_gate_f32

    # final projection
    # to_out weight shape [dim, hidden] as linear weight for hidden->dim
    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return output

import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel for batched matrix multiplication (C = A @ B^T)
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute start indices for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + batch_id * stride_batch_a
    b_batch = b_ptr + batch_id * stride_batch_b
    c_batch = c_ptr + batch_id * stride_batch_c

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # offsets for k within this tile
        cur_k = k * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K) + cur_k
        mask_k = offs_k < K

        # pointers to tiles of A and B
        a_ptrs = a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the `TriMul` (outgoing) operator.
    The heavy N³ multiplication is performed by a fused Triton GEMM.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # unpack config
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]  # c_z in paper

    bs, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (no bias in the reference forward beyond
    #    provided norm weight/bias)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # compute mean/var over last dimension (dim)
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    x = (input_tensor - mean) * torch.rsqrt(var + eps)
    x = x * norm_w + norm_b   # shape [bs, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections + gating (use fp16 for speed)
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # projections
    left = F.linear(x_f16, weights["left_proj.weight"].to(torch.float16))   # [bs,N,N,hidden]
    right = F.linear(x_f16, weights["right_proj.weight"].to(torch.float16))

    # mask (may be None)
    if mask is None:
        mask_f16 = torch.ones(bs, N, N, device=device, dtype=torch.float16)
    else:
        mask_f16 = mask.to(torch.float16)
    mask_f16 = mask_f16.unsqueeze(-1)   # [bs,N,N,1]
    left = left * mask_f16
    right = right * mask_f16

    # gates (sigmoid)
    left_gate = F.linear(x_f16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_f16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_f16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Core TriMul multiplication via Triton
    #    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    #    This is equivalent to left_mat @ right_mat^T per (b,d) slice.
    # ------------------------------------------------------------------
    # reshape to (B*hidden, N, N) where dimensions are (i,k) and (k,j)
    left = left.permute(0, 3, 1, 2).contiguous()          # (bs, hidden, N, N)  i,k
    right = right.permute(0, 3, 2, 1).contiguous()        # (bs, hidden, N, N)  k,j

    total_batches = bs * hidden_dim
    left_mat = left.reshape(total_batches, N, N)
    right_mat = right.reshape(total_batches, N, N)

    out_mat = torch.empty_like(left_mat)   # fp16

    # Choose block sizes heuristically
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    elif N <= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    grid = (
        total_batches,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(1), out_mat.stride(2),
        left_mat.stride(0), right_mat.stride(0), out_mat.stride(0),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to original layout [bs, N, N, hidden]
    out = out_mat.reshape(bs, hidden_dim, N, N).permute(0, 2, 3, 1)  # (bs,N,N,hidden)

    # ------------------------------------------------------------------
    # 4. Output LayerNorm, gating, final linear projection (float32)
    # ------------------------------------------------------------------
    out_f32 = out.to(torch.float32)

    # layer norm on hidden dimension
    mean_o = out_f32.mean(dim=-1, keepdim=True)
    var_o = ((out_f32 - mean_o) ** 2).mean(dim=-1, keepdim=True)
    out_norm = (out_f32 - mean_o) * torch.rsqrt(var_o + eps)

    out_norm = out_norm * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    # apply output gate
    out_norm = out_norm * out_gate.to(torch.float32)

    # final linear projection back to dim
    output = F.linear(out_norm, weights["to_out.weight"])

    return output

"""
TriMul (outgoing) forward pass.

The expensive O(N³·C) operation is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix multiplication
    (i,k) @ (k,j)^T for each (b,d) pair.
A Triton kernel computes this batched GEMM in fp16 (accumulating in fp32).
All other operations (LayerNorms, linear projections, sigmoid gates) are
performed in PyTorch (mixed‑precision) to keep the implementation simple yet
correct.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix dimensions
    stride_am, stride_ak,         # A: (M,K) strides
    stride_bk, stride_bn,         # B: (K,N) strides (B holds the transposed right)
    stride_cm, stride_cn,         # C: (M,N) strides
    stride_batch_a, stride_batch_b, stride_batch_c,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for the current batch
    a_batch = a_ptr + batch_id * stride_batch_a
    b_batch = b_ptr + batch_id * stride_batch_b
    c_batch = c_ptr + batch_id * stride_batch_c

    # Accumulator (fp32 for precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K) + cur_k
        mask_k = offs_k < K

        # Pointers to the current tiles of A and B
        a_ptrs = a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # Store the result (convert back to fp16)
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the Triangle Multiplicative Update (outgoing) operator.
    The heavy N³ multiplication is performed by the batched_matmul_kernel.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # Config & shape
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, N, _, _ = input_tensor.shape  # input shape: [B, N, N, dim]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]

    var = ((input_tensor - mu) ** 2).mean(dim=-1, keepdim=True)
    x = (input_tensor - mu) * torch.rsqrt(var + eps)
    x = x * norm_w + norm_b               # [bs, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections & gates (fp16)
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    left = F.linear(x_f16, weights["left_proj.weight"].to(torch.float16))   # [bs,N,N,hidden]
    right = F.linear(x_f16, weights["right_proj.weight"].to(torch.float16))

    # mask handling (may be None)
    if mask is None:
        mask_f16 = torch.ones(bs, N, N, device=device, dtype=torch.float16)
    else:
        mask_f16 = mask.to(torch.float16)
    mask_f16 = mask_f16.unsqueeze(-1)               # [bs,N,N,1]
    left = left * mask_f16
    right = right * mask_f16

    left_gate = F.linear(x_f16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_f16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_f16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # 3. Core TriMul via Triton GEMM

    # ------------------------------------------------------------------
    # Convert to (B, hidden, N, N) layout; right needs a transpose of its last two axes.
    left = left.permute(0, 3, 1, 2).contiguous()          # (bs, hidden, N, N) – (i,k)
    right = right.permute(0, 3, 2, 1).contiguous()        # (bs, hidden, N, N) – (k,j)

        out_mat,
        N, N, N,

        right_mat.stride(1), right_mat.stride(2), # stride_bk, stride_bn
        out_mat.stride(1), out_mat.stride(2),     # stride_cm, stride_cn
        left_mat.stride(0), right_mat.stride(0), out_mat.stride(0),  # batch strides
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    out = out_mat.reshape(bs, hidden_dim, N, N).permute(0, 2, 3, 1)   # fp16

    out_f32 = out.to(torch.float32)
