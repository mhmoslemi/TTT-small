    left_ptr, right_ptr, out_ptr,
    B_, N, H,  # dims: B, N, H

    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    hidden_dim: tl.constexpr,

    pid_n = tl.program_id(1)
    pid_batch = tl.program_id(2)

    h = pid_batch - b * hidden_dim

"""
TriMul (outgoing) forward pass with a Triton‑accelerated cubic contraction.

The heavy O(N³·H) update
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is performed as a batched matrix multiplication (A @ Bᵀ) where each
(batch, hidden) pair is an independent N×N GEMM.  The kernel works on the
original layout [B, N, N, H] without any extra copies:
* rows   = i  (stride_i)
* cols   = k  (stride_j) for A
* rows   = k  (stride_j) for B (transposed view)
* cols   = j  (stride_i) for B
The batch dimension (B) and hidden dimension (H) are collapsed into a
single program ID so that each (b, h) pair is processed independently.
All arithmetic is in FP16 with FP32 accumulation (Tensor‑Core friendly).
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    a_ptr,               # left   [B, N, N, H] (FP16)
    b_ptr,               # right  [B, N, N, H] (FP16)
    c_ptr,               # out    [B, N, N, H] (FP16)
    N,                   # sequence length
    stride_a_batch,      # stride over B for left
    stride_a_i,          # stride over i (row) for left
    stride_a_j,          # stride over k (col) for left
    stride_a_h,          # stride over hidden for left (usually 1)
    stride_b_batch,      # stride over B for right
    stride_b_i,          # stride over i (col after transposition) for right
    stride_b_j,          # stride over j (row after transposition) for right
    stride_b_h,          # stride over hidden for right
    stride_c_batch,      # stride over B for out
    stride_c_i,          # stride over i for out
    stride_c_j,          # stride over j for out
    stride_c_h,          # stride over hidden for out
    hidden_dim: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)          # block row index
    pid_n = tl.program_id(1)          # block col index
    pid_batch = tl.program_id(2)      # combined batch * hidden index

    # recover batch and hidden indices
    b_idx = pid_batch // hidden_dim
    h_idx = pid_batch - b_idx * hidden_dim

    # base pointers for this (batch, hidden) matrix
    a_base = a_ptr + b_idx * stride_a_batch + h_idx * stride_a_h
    b_base = b_ptr + b_idx * stride_b_batch + h_idx * stride_b_h
    c_base = c_ptr + b_idx * stride_c_batch + h_idx * stride_c_h

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator (FP32 for precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # pointers for A tile (rows i, cols k)
        a_ptrs = a_base + offs_m[:, None] * stride_a_i + offs_k[None, :] * stride_a_j
        # pointers for B tile, transposed view (rows k, cols j)
        b_ptrs = b_base + offs_k[:, None] * stride_b_j + offs_n[None, :] * stride_b_i

        # masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < N) & (offs_k[None, :] < N)
        b_mask = (offs_k[:, None] < N) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K)·(BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # write back
    c_ptrs = c_base + offs_m[:, None] * stride_c_i + offs_n[None, :] * stride_c_j
    c_mask = (offs_m[:, None] < N) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul module (outgoing version).

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask:        torch.Tensor [B, N, N]       (float32, may be all‑ones)
            - weights:     dict of model parameters
            - config:      dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # --------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device),
        bias=weights["norm.bias"].to(device),
    )  # [B, N, N, dim]  FP32

    # --------------------------------------------------------------------
    # 2. Linear projections (FP16)
    # --------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # [B, N, N, H]  FP16
    right = F.linear(x_fp16, right_proj_w)  # [B, N, N, H]  FP16

    # --------------------------------------------------------------------
    # 3. Optional mask
    # --------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------------
    # 4. Gating (FP16)
    # --------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------
    # 5. Cubic contraction via Triton (batched GEMM)
    # --------------------------------------------------------------------
    B, N, _, H = left.shape  # left shape: [B, N, N, H]
    out = torch.empty_like(left, dtype=torch.float16, device=device)

    # Strides (all ints)
    stride_a_batch = left.stride(0)
    stride_a_i = left.stride(1)
    stride_a_j = left.stride(2)
    stride_a_h = left.stride(3)

    stride_b_batch = right.stride(0)
    stride_b_i = right.stride(1)
    stride_b_j = right.stride(2)
    stride_b_h = right.stride(3)

    stride_c_batch = out.stride(0)
    stride_c_i = out.stride(1)
    stride_c_j = out.stride(2)
    stride_c_h = out.stride(3)

    # Tuned block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    trimul_outgoing_kernel[grid](
        left, right, out,
        N,
        stride_a_batch, stride_a_i, stride_a_j, stride_a_h,
        stride_b_batch, stride_b_i, stride_b_j, stride_b_h,
        stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
        hidden_dim=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------------
    # 6. Post‑processing (LayerNorm on hidden dim, final gate, output proj)
    # --------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)  # [B, N, N, H] FP32

    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    out_norm = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # [B, N, N, H] FP32

    out_norm = out_norm * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"].to(device)  # (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)           # [B, N, N, dim] FP32

    return output
