@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # combined batch index (b*H + h)
    pid_m = tl.program_id(1)   # block row (i)
    pid_n = tl.program_id(2)   # block col (j)

    b = pid_batch // H
    h = pid_batch % H

    # compute base pointers
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < K

    # pointers: left[i,k]
    a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
    # right[k,j]
    b_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        acc += tl.dot(a, b, out_dtype=tl.float16)
        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k

    c_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)
    x = x.to(torch.float16)

    # combine projection weights
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    combined_w = torch.cat([left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w], dim=0).to(torch.float16)  # shape (5*hidden, dim)

    # linear projection for all at once
    x_flat = x.reshape(-1, dim)  # shape (bs*N*N, dim)
    proj = torch.nn.functional.linear(x_flat, combined_w)  # (bs*N*N, 5*hidden)

    left = proj[..., 0:hidden]
    right = proj[..., hidden:2*hidden]
    left_gate = proj[..., 2*hidden:3*hidden]
    right_gate = proj[..., 3*hidden:4*hidden]
    out_gate = proj[..., 4*hidden:5*hidden]

    # mask handling
    if mask is not None:
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)  # shape (bs,N,N,1)
        left = left * mask_f16
        right = right * mask_f16

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the TriMul "outgoing" operator.
    The heavy triangular multiplication is performed by a fused batched GEMM kernel.
    Linear projections and gating are merged into a single matrix multiplication to
    reduce memory traffic. All remaining operations run in PyTorch (float16) and the
    final output is returned in float32.
    """
    import torch
    import torch.nn.functional as F
    import triton

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    bs, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ---------- LayerNorm on input ----------
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias = weights["norm.bias"].to(torch.float16)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)
    x = x.to(torch.float16)  # [bs,N,N,dim]

    # ---------- Combined linear projections ----------
    # concatenate all projection weights (out_features = hidden for each)
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    combined_w = torch.cat([left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w], dim=0).to(torch.float16)  # (5*hidden, dim)

    # flatten and apply combined linear
    x_flat = x.reshape(-1, dim)                     # (bs*N*N, dim)
    proj = F.linear(x_flat, combined_w)             # (bs*N*N, 5*hidden)
    proj = proj.view(bs, N, N, hidden * 5)           # (bs,N,N,5*hidden)

    # split tensors
    left = proj[..., 0:hidden]
    right = proj[..., hidden:2*hidden]
    left_gate = proj[..., 2*hidden:3*hidden]
    right_gate = proj[..., 3*hidden:4*hidden]
    out_gate = proj[..., 4*hidden:5*hidden]

    # ---------- Mask ----------
    if mask is not None and not config.get("nomask", False):
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)  # (bs,N,N,1)
        left = left * mask_f16
        right = right * mask_f16

    # ---------- Gating ----------
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- Core TriMul (batched GEMM) ----------
    out = torch.empty_like(left)  # fp16, shape (bs,N,N,hidden)

    # strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    total_batches = bs * hidden

    # block sizes heuristic
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    elif N <= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    grid = (total_batches, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    trimul_outgoing_kernel[grid](
        left, right, out,
        bs, N, hidden,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_k, stride_right_i, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # ---------- Output LayerNorm ----------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(out, (hidden,), weight=to_out_norm_w, bias=to_out_norm_b, eps=eps)

    # ---------- Apply output gate ----------
    out = out * out_gate

    # ---------- Final linear projection ----------
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (dim, hidden)
    output = F.linear(out, to_out_w)  # (bs,N,N,dim) in fp16
    return output.to(torch.float32)

"""
TriMul "outgoing" forward pass with Triton acceleration.

The most expensive step (the O(N³) multiplication) is implemented as a
batched GEMM kernel that directly works on the original 4‑D layout
[B, N, N, H] without any permute/contiguous copies.  All linear
projections (left/right and their gates) are fused into a single matrix
multiply by concatenating the five weight matrices, drastically reducing
memory traffic.  The remaining element‑wise operations (masking,
sigmoid, LayerNorm, final linear) are performed in PyTorch (float16) and
the final result is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Batched GEMM kernel: computes out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,k,j,h]
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_k, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)   # (b * H + h)
    pid_m = tl.program_id(1)       # block row   (i)
    pid_n = tl.program_id(2)       # block col   (j)

    # --------------------------------------------------------------
    # decode batch index into (batch, hidden) pair
    # --------------------------------------------------------------
    b = pid_batch // H
    h = pid_batch % H

    # base pointers for this (b, h) slice
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # --------------------------------------------------------------
    # tile offsets
    # --------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N

    # pointers to tiles:
    #   left : (i, k)
    #   right: (k, j)
    a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
    b_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for _ in range(num_k_blocks):
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=tl.float16(0.0))
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=tl.float16(0.0))
        acc += tl.dot(a, b, out_dtype=tl.float16)
        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k

    c_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

# Entry point called by the benchmark harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul "outgoing" operation.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    eps = 1e-5

    bs, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------- Input LayerNorm ---------------------------
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_w,
                     bias=norm_b,
                     eps=eps)
    x = x.to(torch.float16)               # [bs, N, N, dim]

    # ----------------- Combined linear for proj + gates (5 * hidden) ----------
    w_left   = weights["left_proj.weight"]
    w_right  = weights["right_proj.weight"]
    w_lgate  = weights["left_gate.weight"]
    w_rgate  = weights["right_gate.weight"]
    w_outgate = weights["out_gate.weight"]
    # each weight has shape (hidden, dim)
    combined_w = torch.cat([w_left,
                            w_right,
                            w_lgate,
                            w_rgate,
                            w_outgate], dim=0).to(torch.float16)   # (5*hidden, dim)

    x_flat = x.reshape(-1, dim)                        # (bs*N*N, dim)
    proj   = F.linear(x_flat, combined_w)              # (bs*N*N, 5*hidden)
    proj   = proj.view(bs, N, N, hidden * 5)           # (bs, N, N, 5*hidden)

    # split the concatenated projection
    left      = proj[..., 0:hidden]
    right     = proj[..., hidden:2*hidden]
    left_gate = proj[..., 2*hidden:3*hidden]
    right_gate= proj[..., 3*hidden:4*hidden]
    out_gate  = proj[..., 4*hidden:5*hidden]

    # ------------------------------- Mask --------------------------------
    if mask is not None and not nomask:
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)   # (bs, N, N, 1)
        left  = left  * mask_f16
        right = right * mask_f16

    # ------------------------------- Gates --------------------------------
    left_gate  = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate   = out_gate.sigmoid()

    left  = left  * left_gate
    right = right * right_gate

    # -------------------------- Core TriMul -------------------------------
    out = torch.empty_like(left)                         # fp16, (bs, N, N, hidden)

    # strides of the tensors in memory (torch returns ints)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b,   stride_out_i,   stride_out_j,   stride_out_h = out.stride()

    total_batches = bs * hidden

    # heuristic block sizes
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    elif N <= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    grid = (total_batches,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    # launch kernel
    trimul_outgoing_kernel[grid](
        left, right, out,
        bs, N, hidden,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_k, stride_right_i, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # --------------------------- Output LayerNorm -------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(out,
                       (hidden,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=eps)

    # apply output gate
    out = out * out_gate

    # final linear projection back to original dim
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden)
    output = F.linear(out, to_out_w)                       # (bs, N, N, dim) fp16

    return output.to(torch.float32)
