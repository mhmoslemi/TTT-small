@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0) # block row index
    pj = tl.program_id(1) # block col index
    pb = tl.program_id(2) # batch index

    # Compute offsets
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pj * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Compute base pointers for this batch
    a_ptrs = a_ptr + pb * stride_batch_a + offs_m[:, None] * stride_am + (offs_k)[None, :] * stride_ak
    b_ptrs = b_ptr + pb * stride_batch_b + (offs_k)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Compute number of tiles over K
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        # compute k offset
        k_offset = k * BLOCK_K
        # load A tile
        a = tl.load(a_ptrs + k_offset * stride_ak, mask=mask_m[:, None] & (k_offset + offs_k < K), other=0.0)
        # load B tile
        b = tl.load(b_ptrs + k_offset * stride_bk, mask=(k_offset + offs_k < K)[:, None] & mask_n[None, :], other=0.0)
        # dot
        acc += tl.dot(a, b)

    # Store
    c_ptrs = c_ptr + pb * stride_batch_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # Load weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Convert to appropriate dtype (fp16) for compute
    # layernorm on fp32 is okay
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B,N,N,dim] float32

    # Cast to fp16 for heavy ops
    x_fp16 = x.to(torch.float16)

    # Linear projections
    left = torch.nn.functional.linear(x_fp16, left_proj_weight.to(torch.float16))
    right = torch.nn.functional.linear(x_fp16, right_proj_weight.to(torch.float16))

    # Apply mask
    if mask is not None:
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_fp16
        right = right * mask_fp16

    # Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, left_gate_weight.to(torch.float16)))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, right_gate_weight.to(torch.float16)))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, out_gate_weight.to(torch.float16)))

    left = left * left_gate
    right = right * right_gate

    B, N, _, H = left.shape

    # Permute and make contiguous for matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()      # (B, H, N, N) rows=i, cols=k
    right_perm = right.permute(0, 3, 2, 1).contiguous()    # (B, H, N, N) rows=k, cols=j

    # collapse batch*hidden
    BH = B * H
    A = left_perm.view(BH, N, N)
    Bmat = right_perm.view(BH, N, N)

    # output buffer
    C = torch.empty_like(A)  # same dtype fp16

    # Kernel launch
    BLOCK = 64
    BLOCK_M = BLOCK_N = BLOCK_K = BLOCK

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), BH)

    batched_matmul_kernel[grid](
        A,
        Bmat,
        C,
        N, N, N,
        A.stride(1), A.stride(2),
        Bmat.stride(1), Bmat.stride(2),
        C.stride(1), C.stride(2),
        A.stride(0), Bmat.stride(0), C.stride(0),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to original layout
    out_hidden = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # shape [B,N,N,hidden]

    # Post processing
    # Convert to fp32 for layernorm
    out_fp32 = out_hidden.to(torch.float32)

    # to_out_norm
    out_norm = torch.nn.functional.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # Apply out_gate
    out_norm = out_norm * out_gate.to(torch.float32)

    # Final linear projection to dim
    output = torch.nn.functional.linear(out_norm, to_out_weight)  # weight shape (dim, hidden)

    return output

"""
TriMul (outgoing) forward pass with a Triton‐accelerated cubic contraction.
The expensive O(N³·H) update
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is performed as a batched GEMM where each (batch, hidden) pair corresponds
to an independent N×N matrix multiplication.  Inputs are reshaped to a
contiguous layout so that the kernel enjoys fully coalesced loads.
All surrounding LayerNorms, linear projections and gating operations are
executed in PyTorch (half‑precision where beneficial).  The final output
is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: batched matrix multiplication (A @ B) for many (B·H) pairs
# --------------------------------------------------------------------------- #
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix dimensions (M=N=K=seq_len)
    stride_am, stride_ak,         # A strides: rows, cols
    stride_bk, stride_bn,         # B strides: rows (K), cols (N)
    stride_cm, stride_cn,         # C strides: rows, cols
    stride_batch_a, stride_batch_b, stride_batch_c,  # batch stride (BH)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_b = tl.program_id(2)   # batch * hidden index

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # base pointers for this batch tile
    a_base = a_ptr + pid_b * stride_batch_a
    b_base = b_ptr + pid_b * stride_batch_b

    # pointers for the first K‑tile
    a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K

        # masks for out‑of‑bounds loads
        mask_a = (offs_m[:, None] < M) & ((cur_k + offs_k)[None, :] < K)
        mask_b = ((cur_k + offs_k)[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs + cur_k * stride_ak, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs + cur_k * stride_bk, mask=mask_b, other=0.0)

        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_ptr + pid_b * stride_batch_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


# --------------------------------------------------------------------------- #
# Entry point used by the harness
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask, weights, config)
              input_tensor : [B, N, N, dim]      (float32)
              mask         : [B, N, N] (float32) or None
              weights      : dict of model parameters
              config       : dict with keys "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------- #
    # 1. LayerNorm on the input channel dimension
    # ------------------------------------------------------------------- #
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)   # [B,N,N,dim] float32

    # cast once to FP16 for the heavy work
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------- #
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------- #
    left_proj_w  = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left  = F.linear(x_fp16, left_proj_w)   # [B,N,N,hidden]
    right = F.linear(x_fp16, right_proj_w)  # [B,N,N,hidden]

    # ------------------------------------------------------------------- #
    # 3. Optional mask (applied before gating)
    # ------------------------------------------------------------------- #
    if mask is not None:
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)
        left  = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------- #
    # 4. Gating
    # ------------------------------------------------------------------- #
    left_gate_w  = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5. Cubic contraction via a batched GEMM (Triton)
    # ------------------------------------------------------------------- #
    # Reshape to (B*hidden, N, N) with contiguous layout
    left_cont  = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N) → rows=i, cols=k
    right_cont = right.permute(0, 3, 2, 1).contiguous()  # (B, hidden, N, N) → rows=k, cols=j

    BH = B * hidden
    A = left_cont.view(BH, N, N)
    Bmat = right_cont.view(BH, N, N)
    C = torch.empty_like(A)  # result in fp16

    # block size (tuned for B200, works up to N=1024)
    BLOCK = 64
    BLOCK_M = BLOCK_N = BLOCK_K = BLOCK

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), BH)

    batched_matmul_kernel[grid](
        A,
        Bmat,
        C,
        N, N, N,
        A.stride(1), A.stride(2),          # stride_am, stride_ak
        Bmat.stride(1), Bmat.stride(2),    # stride_bk, stride_bn
        C.stride(1), C.stride(2),          # stride_cm, stride_cn
        A.stride(0), Bmat.stride(0), C.stride(0),  # batch strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to original layout [B, N, N, hidden]
    out_hidden = C.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------- #
    # 6. Post‑processing (LayerNorm on hidden, final gate, output projection)
    # ------------------------------------------------------------------- #
    out_fp32 = out_hidden.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    out_norm = F.layer_norm(out_fp32, (hidden,), weight=to_out_norm_w, bias=to_out_norm_b)

    out_norm = out_norm * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"].to(device)   # shape (dim, hidden)
    output = F.linear(out_norm, to_out_w)           # [B, N, N, dim], float32

    return output
