def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm input
    gamma = weights["norm.weight"]
    beta = weights["norm.bias"]
    x = input_tensor
    # Compute mean, var over last dim
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * gamma + beta   # shape B,N,N,dim

    # Convert to fp16 for projections
    x_h = x_norm.to(torch.float16)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)
    left = F.linear(x_h, left_proj_weight)  # B,N,N,hidden_dim
    right = F.linear(x_h, right_proj_weight)

    # Gates
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_h, left_gate_weight).sigmoid()
    right_gate = F.linear(x_h, right_gate_weight).sigmoid()
    out_gate = F.linear(x_h, out_gate_weight).sigmoid()   # B,N,N,hidden_dim

    # Apply mask and gates using Triton kernel if mask provided
    if mask is not None and mask.numel() > 0:
        # ensure mask shape [B,N,N] and dtype float16
        mask_f16 = mask.to(torch.float16)
        # Preallocate outputs
        left_out = torch.empty_like(left)
        right_out = torch.empty_like(right)
        total_elements = left.numel()
        BLOCK = 1024  # adjust
        grid = (triton.cdiv(total_elements, BLOCK),)
        mask_gate_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_f16,
            out_left_ptr=left_out,
            out_right_ptr=right_out,
            total_elements=total_elements,
            hidden_dim=hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left = left_out
        right = right_out
    else:
        # No mask, just apply gates via elementwise multiplication
        left = left * left_gate
        right = right * right_gate

    # Now contract: batched GEMM
    B, N, _, H = left.shape
    # Reshape for batched matmul
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B*H, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B*H, N, N)
    # Compute out = left @ right.T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)
    # reshape back
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)  # (B,N,N,H)
    out_f32 = out.to(torch.float32)

    # Output layernorm over hidden dim
    gamma_out = weights["to_out_norm.weight"]
    beta_out = weights["to_out_norm.bias"]
    mean_o = out_f32.mean(dim=-1, keepdim=True)
    var_o = out_f32.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out_f32 - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * gamma_out + beta_out

    # Apply out_gate
    out_norm = out_norm * out_gate.to(torch.float32)

    # Final linear projection to dim
    to_out_weight = weights["to_out.weight"].to(torch.float32)
    output = F.linear(out_norm, to_out_weight)   # shape [B,N,N,dim]

    return output

"""
TriMul (outgoing) forward pass.

The heavy tensor contraction 
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix multiplication of shape (B*D, N, N) @ (B*D, N, N)^T.
We perform all cheap element‑wise work (layer‑norm, linear projections,
gating, mask handling) in PyTorch and use a small Triton kernel to fuse
masking with the element‑wise gates.  The contraction itself is done with
torch.bmm, which on an A100‑class GPU (B200) uses tensor‑core accelerated
BF16/FP16 GEMM and is far faster than a naïve Triton GEMM for the batch
sizes used in AlphaFold3.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr,
    out_left_ptr, out_right_ptr,
    total_elements, hidden_dim,
    BLOCK: tl.constexpr,
):
    """
    Element‑wise kernel:
        out_left  = left  * left_gate  * mask
        out_right = right * right_gate * mask
    `mask` is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elements

    # Load the four tensors (fp16) and mask (fp16)
    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    # mask index = floor(offs / hidden_dim) because the mask has no hidden dim
    mask_idx = offs // hidden_dim
    mask_valid = mask_idx < (total_elements // hidden_dim)
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=1.0)

    out_left = left * left_gate * mask_val
    out_right = right * right_gate * mask_val

    tl.store(out_left_ptr + offs, out_left, mask=valid)
    tl.store(out_right_ptr + offs, out_right, mask=valid)


def custom_kernel(data):
    """
    Args:
        data: tuple (input, mask, weights, config)
            input  : torch.Tensor [B, N, N, dim] (float32, on CUDA)
            mask   : torch.Tensor [B, N, N] (float32/bool) or None
            weights: dict of torch.Tensors for all linear / norm parameters
            config : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Input LayerNorm (over last dim)
    # ------------------------------------------------------------------
    gamma = weights["norm.weight"].to(device)
    beta = weights["norm.bias"].to(device)

    x = input_tensor  # [B, N, N, dim], float32
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * gamma + beta                      # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Projections (fp16 for speed)
    # ------------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(torch.float16)   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w)      # [B, N, N, hidden_dim]
    right = F.linear(x_h, right_proj_w)

    # ------------------------------------------------------------------
    # Gates (fp16)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_h, left_gate_w).sigmoid()
    right_gate = F.linear(x_h, right_gate_w).sigmoid()
    out_gate = F.linear(x_h, out_gate_w).sigmoid()   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Apply mask and element‑wise gates via Triton (if a mask is present)
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        # mask: [B, N, N] -> fp16
        mask_f16 = mask.to(torch.float16)

        left_out = torch.empty_like(left)
        right_out = torch.empty_like(right)

        total_elements = left.numel()          # B * N * N * hidden_dim
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)

        _mask_gate_kernel[grid](
            left, right,
            left_gate, right_gate,
            mask_f16,
            left_out, right_out,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left = left_out
        right = right_out
    else:
        # No mask => simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # Batched matrix multiplication: out = left @ right^T over the k dim
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                      # H == hidden_dim
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (batch, M, K)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # (batch, K, N)

    # torch.bmm on fp16 uses tensor cores
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16
    out_f32 = out.to(torch.float32)

    # ------------------------------------------------------------------
    # Output LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------
    gamma_out = weights["to_out_norm.weight"].to(device)
    beta_out = weights["to_out_norm.bias"].to(device)

    mean_o = out_f32.mean(dim=-1, keepdim=True)
    var_o = out_f32.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out_f32 - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * gamma_out + beta_out

    # ------------------------------------------------------------------
    # Apply out_gate (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate.to(torch.float32)

    # ------------------------------------------------------------------
    # Final linear projection back to input dimension
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float32)   # (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)                  # [B, N, N, dim]

    return output
