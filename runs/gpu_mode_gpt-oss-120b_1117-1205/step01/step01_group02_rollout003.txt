"""
Triton‐accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension.
2. Apply left/right linear projections and the corresponding sigmoid gates.
   (All heavy ops are cast to float16 for speed.)
3. Apply an optional mask to the projected tensors.
4. Fuse the cubic operation  out[i,j] = Σ_k left[i,k] * right[j,k]
   by a batched GEMM (left @ rightᵀ) using a custom Triton kernel.
5. Reshape the result back to [B, N, N, hidden_dim] and continue in fp32:
   * Layer‑norm, multiply by the output gate, and a final linear projection
     back to the original channel dimension.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# --------------------------------------------------------------
# Batched GEMM kernel: C = A @ Bᵀ   (A,B : [batch, M, K] , [batch, K, N])
# --------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # input and output pointers
    M, N, K,                           # matrix sizes (all int)
    stride_am, stride_ak,              # A row / col strides (in elements)
    stride_bk, stride_bn,              # B row / col strides (in elements)
    stride_cm, stride_cn,              # C row / col strides (in elements)
    stride_a_batch, stride_b_batch, stride_c_batch,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)      # block row index
    pid_n = tl.program_id(1)      # block col index
    pid_b = tl.program_id(2)      # batch index (B * hidden_dim)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_off = k_tile * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K)
        mask_k = (k_off + offs_k) < K

        # Pointers for the current tile (including batch offset)
        a_ptrs = (
            a_ptr
            + pid_b * stride_a_batch
            + offs_m[:, None] * stride_am
            + (k_off + offs_k)[None, :] * stride_ak
        )
        b_ptrs = (
            b_ptr
            + pid_b * stride_b_batch
            + (k_off + offs_k)[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        # Load tiles (apply masks)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # Write out the result (cast back to fp16)
    c_ptrs = (
        c_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)


# --------------------------------------------------------------
# Entry point expected by the evaluation harness
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32, CUDA)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of torch.Tensor with all linear / LN params
            - config: dict with at least {"dim": int, "hidden_dim": int}

    Returns:
        torch.Tensor [B, N, N, dim] (float32, CUDA)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----------------------------------------------------------
    # 1. Input LayerNorm (over channel dimension)
    # ----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    # Cast to fp16 for the heavy path
    x_fp16 = x.to(torch.float16)

    # ----------------------------------------------------------
    # 2. Linear projections + gating (all fp16)
    # ----------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    left_gate = F.linear(
        x_fp16, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = F.linear(
        x_fp16, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = F.linear(
        x_fp16, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    # ----------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ----------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ----------------------------------------------------------
    # 4. Apply element‑wise gates
    # ----------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------
    # 5. Prepare tensors for the fused GEMM (left @ rightᵀ)
    # ----------------------------------------------------------
    # From [B, N, N, H] -> [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Transpose the last two axes of right so we can multiply A @ B
    right_T = right_perm.transpose(-2, -1).contiguous()

    B, H, N, _ = left_perm.shape
    total_batch = B * H

    # Collapse batch and head dimensions for the kernel
    left_flat = left_perm.view(total_batch, N, N)
    right_flat = right_T.view(total_batch, N, N)
    out_flat = torch.empty((total_batch, N, N), dtype=torch.float16, device=device)

    # Strides in *elements*
    stride_a_batch, stride_am, stride_ak = left_flat.stride()
    stride_b_batch, stride_bk, stride_bn = right_flat.stride()
    stride_c_batch, stride_cm, stride_cn = out_flat.stride()

    # ----------------------------------------------------------
    # 6. Launch the Triton kernel
    # ----------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        total_batch,
    )
    trimul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----------------------------------------------------------
    # 7. Reshape back to [B, N, N, H] and continue in FP32
    # ----------------------------------------------------------
    out = (
        out_flat.view(B, H, N, N)          # [B, H, N, N]
        .permute(0, 2, 3, 1)               # [B, N, N, H]
        .contiguous()
        .to(torch.float32)
    )

    # ----------------------------------------------------------
    # 8. Output LayerNorm, gating and final projection
    # ----------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate.to(torch.float32)  # apply output gate
    output = F.linear(out, weights["to_out.weight"].to(torch.float32))  # [B, N, N, dim]

    return output
