def custom_kernel(data):
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Load weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm
    x_norm = F.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Convert to float16 for heavy ops
    x_h = x_norm.to(torch.float16)

    # Linear projections
    left = F.linear(x_h, left_proj_weight.to(torch.float16))   # shape B,N,N,hidden_dim (fp16)
    right = F.linear(x_h, right_proj_weight.to(torch.float16))

    # Gates
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_weight.to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_weight.to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.to(torch.float16)))  # shape B,N,N,hidden_dim

    # Apply mask, gate (elementwise)
    if mask is not None:
        # convert mask to float16
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)   # shape B,N,N,1
        left = left * mask_f16
        right = right * mask_f16

    left = left * left_gate
    right = right * right_gate

    # Batched matmul using torch.bmm
    B = left.shape[0]
    N = left.shape[1]
    # left shape [B,N,N,hidden_dim]
    # Permute to B,hidden_dim,N,N
    left_mat = left.permute(0,3,1,2).reshape(B * hidden_dim, N, N)
    right_mat = right.permute(0,3,1,2).reshape(B * hidden_dim, N, N)

    # Compute out = left @ right^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))

    # reshape back to B,N,N,hidden_dim
    out = out_mat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()   # shape B,N,N,hidden_dim (fp16)

    # Convert to float32 for layernorm
    out_f32 = out.to(torch.float32)

    # Output LayerNorm
    out_norm = F.layer_norm(out_f32, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Multiply with out_gate (cast to fp32)
    out_gate_f32 = out_gate.to(torch.float32)
    out_norm = out_norm * out_gate_f32

    # Final linear projection
    output = F.linear(out_norm, to_out_weight)   # shape B,N,N,dim

    return output

"""
Optimized TriMul (outgoing) forward implementation.

The heavy tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed as a batched GEMM using torch.bmm (cuBLAS) in FP16,
while a tiny Triton kernel is used for the final element‑wise gating.
All other operations (layer‑norms, linear projections) are carried out
with PyTorch's highly‑optimized CUDA kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiplication of two flat tensors
# ----------------------------------------------------------------------
@triton.jit
def _elemwise_mul_kernel(
    a_ptr,                # pointer to first operand (float16)
    b_ptr,                # pointer to second operand (float16)
    c_ptr,                # pointer to output (float16)
    N,                    # total number of elements
    BLOCK: tl.constexpr,  # block size (elements per program)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N

    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(c_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module.

    Args:
        data: tuple (input, mask, weights, config)
            input  : torch.Tensor [B, N, N, dim] (float32)
            mask   : torch.Tensor [B, N, N] (bool/float) or None
            weights: dict of model parameters (float32)
            config : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Load weights (float32) and cast where needed
    # ------------------------------------------------------------------
    # LayerNorm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # Projections (cast to fp16 for speed)
    w_left = weights["left_proj.weight"].to(torch.float16)
    w_right = weights["right_proj.weight"].to(torch.float16)

    # Gating weights (fp16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    # Output LayerNorm
    w_out_norm = weights["to_out_norm.weight"]
    b_out_norm = weights["to_out_norm.bias"]

    # Final projection (float32)
    w_to_out = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # Cast to fp16 for the heavy path
    # ------------------------------------------------------------------
    x_h = x_norm.to(torch.float16)  # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x_h, w_left)   # [B, N, N, hidden_dim] fp16
    right = F.linear(x_h, w_right) # [B, N, N, hidden_dim] fp16

    # ------------------------------------------------------------------
    # Gating tensors (fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_h, w_left_gate))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x_h, w_right_gate)) # [B, N, N, hidden_dim]
    out_gate = torch.sigmoid(F.linear(x_h, w_out_gate))    # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Apply mask (if any) and gates (element‑wise)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f16
        right = right * mask_f16

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Batched matrix multiplication using torch.bmm (FP16)
    #   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to (B*hidden_dim, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # out_mat = left_mat @ right_mat^T  (size: (B*H, N, N))
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, hidden_dim] (still fp16)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, hidden_dim] fp16

    # ------------------------------------------------------------------
    # Output LayerNorm (float32)
    # ------------------------------------------------------------------
    out_f32 = out.to(torch.float32)  # cast for stable statistics
    out_norm = F.layer_norm(
        out_f32,
        normalized_shape=(hidden_dim,),
        weight=w_out_norm,
        bias=b_out_norm,
        eps=eps,
    )  # [B, N, N, hidden_dim] float32

    # ------------------------------------------------------------------
    # Multiply by out_gate (fp16) using a tiny Triton kernel
    # ------------------------------------------------------------------
    # Cast the normalized tensor back to fp16 for the kernel
    out_norm_h = out_norm.to(torch.float16)
    out_gate_h = out_gate  # already fp16

    # Allocate output for the gated tensor
    out_gated_h = torch.empty_like(out_norm_h)

    total_elems = out_norm_h.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_elems, BLOCK),)

    _elemwise_mul_kernel[grid](
        out_norm_h,
        out_gate_h,
        out_gated_h,
        total_elems,
        BLOCK=BLOCK,
        num_warps=4,
    )

    # Cast back to float32 for the final projection
    out_gated_f32 = out_gated_h.to(torch.float32)

    # ------------------------------------------------------------------
    # Final linear projection back to input dimension
    # ------------------------------------------------------------------
    output = F.linear(out_gated_f32, w_to_out)  # [B, N, N, dim] float32

    return output
