@triton.jit
def batched_gemm_kernel(a_ptr, b_ptr, c_ptr,
                        M, N, K,
                        stride_am, stride_ak, stride_batch_a,
                        stride_bk, stride_bn, stride_batch_b,
                        stride_cm, stride_cn, stride_batch_c,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # program ids: program_id(0) for batch (BH), program_id(1) for block row (M), program_id(2) for block col (N)
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets for block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Compute pointer to batch start
    a_batch_ptr = a_ptr + pid_batch * stride_batch_a
    b_batch_ptr = b_ptr + pid_batch * stride_batch_b
    c_batch_ptr = c_ptr + pid_batch * stride_batch_c

    # Masks for boundary conditions
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Pointers for tiles
    a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_batch_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        # dot does reduction over K dim: we need to compute a * b and add to acc.
        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store result
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul "outgoing" forward pass.
- LayerNorm on the input tensor.
- Linear projections (left/right) and gating (left_gate/right_gate/out_gate) in fp16.
- Core O(N³) operation is a batched matrix multiplication:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
  Implemented with a Triton GEMM kernel (batch*hidden as the outer batch).
- Output LayerNorm, gating, and final linear projection back to the original dim.
All heavy arithmetic (the N³ product) runs in fp16 with fp32 accumulation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM A (M×K) @ B (K×N) -> C (M×N)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,            # pointers
    M, N, K,                        # problem sizes
    stride_am, stride_ak, stride_batch_a,  # strides for A (M, K, batch)
    stride_bk, stride_bn, stride_batch_b,  # strides for B (K, N, batch)
    stride_cm, stride_cn, stride_batch_c,  # strides for C (M, N, batch)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # batch * hidden
    pid_m = tl.program_id(1)       # block row
    pid_n = tl.program_id(2)       # block col

    # Base pointers for this batch
    a_batch = a_ptr + pid_batch * stride_batch_a
    b_batch = b_ptr + pid_batch * stride_batch_b
    c_batch = c_ptr + pid_batch * stride_batch_c

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Pointers to tiles of A and B
    a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)
        acc += tl.dot(a, b)      # (BLOCK_M, BLOCK_K)·(BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store result
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    bs, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b, eps=eps)   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections & gating (fp16)
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # Projections
    w_left = weights["left_proj.weight"].to(torch.float16)      # (hidden, dim)
    w_right = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_f16, w_left)      # [B,N,N,hidden]
    right = F.linear(x_f16, w_right)

    # Optional mask
    if mask is None:
        mask_f16 = torch.ones(bs, N, N, 1, dtype=torch.float16, device=device)
    else:
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
    left = left * mask_f16
    right = right * mask_f16

    # Gates (sigmoid)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_f16, w_left_gate))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x_f16, w_right_gate))
    out_gate = torch.sigmoid(F.linear(x_f16, w_out_gate))    # will be used later (fp32)

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Core N³ multiplication via Triton (batched GEMM)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Rearrange tensors to (batch, hidden, N, N) and make contiguous
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)  -> rows=i, cols=k
    right = right.permute(0, 3, 2, 1).contiguous() # (B, hidden, N, N)  -> rows=k, cols=j

    total_batches = bs * hidden
    left_mat = left.view(total_batches, N, N)
    right_mat = right.view(total_batches, N, N)

    out_mat = torch.empty_like(left_mat)  # fp16 output

    # Choose tile sizes based on N
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        total_batches,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,                                 # M, N, K
        left_mat.stride(1), left_mat.stride(2), left_mat.stride(0),   # A strides
        right_mat.stride(1), right_mat.stride(2), right_mat.stride(0),# B strides
        out_mat.stride(1), out_mat.stride(2), out_mat.stride(0),    # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden] and cast to float32 for the next layers
    out = out_mat.view(bs, hidden, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 4. Output LayerNorm, output gate, and final projection (float32)
    # ------------------------------------------------------------------
    w_to_out_norm = weights["to_out_norm.weight"]
    b_to_out_norm = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden,), weight=w_to_out_norm, bias=b_to_out_norm, eps=eps)

    out_gate = out_gate.to(torch.float32)
    out = out * out_gate

    w_to_out = weights["to_out.weight"]   # (dim, hidden)
    output = F.linear(out, w_to_out)      # [B, N, N, dim]

    return output
