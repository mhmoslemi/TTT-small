    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,

    pid_n = tl.program_id(2) # col block

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm on input
    # weights
    gamma = weights["norm.weight"]
    beta = weights["norm.bias"]
    x = input_tensor
    # compute mean and var over last dim
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * gamma + beta  # shape [B,N,N,dim]

    # Convert to half for projection and gating
    x_h = x_norm.to(torch.float16)

    # Linear weights
    w_left = weights["left_proj.weight"].to(torch.float16)
    w_right = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    # proj
    left = F.linear(x_h, w_left)   # [B,N,N,hidden_dim]
    right = F.linear(x_h, w_right)

    # mask
    if mask is not None:
        # shape mask: [B,N,N]
        mask_ = mask.unsqueeze(-1).to(x_h.dtype)
        left = left * mask_
        right = right * mask_

    # gates
    left_gate = torch.sigmoid(F.linear(x_h, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_h, w_right_gate))
    out_gate = torch.sigmoid(F.linear(x_h, w_out_gate))

    left = left * left_gate
    right = right * right_gate

    B, N_, N_, H = left.shape
    assert N_ == N, "sequence length mismatch"

    # reshape for batched GEMM
    # left: (B, N, N, H) -> permute to (B, H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous()
    # right: need to transpose k and j: (B, N, N, H) -> permute (0,3,2,1)
    right_mat = right.permute(0, 3, 2, 1).contiguous()

    # flatten batch and hidden
    left_mat = left_mat.view(B * H, N, N)
    right_mat = right_mat.view(B * H, N, N)

    # allocate output
    out_mat = torch.empty((B * H, N, N), dtype=torch.float16, device=device)

    # strides
    stride_az, stride_am, stride_ak = left_mat.stride()
    stride_bz, stride_bk, stride_bn = right_mat.stride()
    stride_cz, stride_cm, stride_cn = out_mat.stride()

    # tile sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # reshape back
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # to_out_norm
    gamma_out = weights["to_out_norm.weight"]
    beta_out = weights["to_out_norm.bias"]
    out_f32 = out.to(torch.float32)
    mean_o = out_f32.mean(dim=-1, keepdim=True)
    var_o = out_f32.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out_f32 - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * gamma_out + beta_out

    # apply out_gate
    out_norm = out_norm * out_gate.to(torch.float32)

    # final projection
    w_to_out = weights["to_out.weight"]
    output = F.linear(out_norm, w_to_out)  # shape [B,N,N,dim]

    return output

TriMul (outgoing) forward pass with a custom Triton batched GEMM.
The heavy contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

uses 8 warps and 3 prefetch stages to fully exploit tensor‑cores on an NVIDIA
B200 GPU.

import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(

    C_ptr,

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile (M x K)
        a_ptrs = (
            A_ptr
            + batch_id * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # load B tile (K x N)
        b_ptrs = (
            B_ptr
            + batch_id * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )  # fp16

        # tensor‑core matmul, fp16 inputs → fp32 accumulation
        acc += tl.dot(a, b)

    # store result (convert accumulation to fp16)
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    TriMul forward (outgoing) implementation.
    Args:
        data: (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor [B, N, N, dim] (float32)
            mask         : torch.Tensor [B, N, N]  (bool/float) or None
            weights      : dict of torch.Tensors containing all linear / norm params
            config       : dict with keys "dim" and "hidden_dim"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    gamma = weights["norm.weight"].to(device).to(torch.float32)
    beta = weights["norm.bias"].to(device).to(torch.float32)

    x = input_tensor.to(torch.float32)
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * gamma + beta                      # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Projections & Gates (fp16 for speed)
    # ------------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    w_left = weights["left_proj.weight"].to(device).to(torch.float16)   # (hidden_dim, dim)
    w_right = weights["right_proj.weight"].to(device).to(torch.float16)

    w_left_gate = weights["left_gate.weight"].to(device).to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(device).to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(device).to(torch.float16)

    # Linear projections
    left = F.linear(x_h, w_left)            # [B, N, N, hidden_dim]
    right = F.linear(x_h, w_right)

    # Optional mask (broadcast over the hidden dimension)
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x_h.dtype)   # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # Gating
    left_gate = torch.sigmoid(F.linear(x_h, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_h, w_right_gate))
    out_gate = torch.sigmoid(F.linear(x_h, w_out_gate))

    left = left * left_gate
    right = right * right_gate

    B, N, _, H = left.shape   # H == hidden_dim

    # ------------------------------------------------------------------
    # Batched GEMM: out = left @ right^T  (over K dimension)
    # ------------------------------------------------------------------
    # left: (B, N, N, H) → (B, H, N, N)  (rows=i, cols=k)
    left_mat = left.permute(0, 3, 1, 2).contiguous()
    # right: (B, N, N, H) → (B, H, N, N) then transpose k↔j to obtain (B, H, K, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous()

    # flatten batch & hidden dim
    left_mat = left_mat.view(B * H, N, N)        # (batch, M, K)
    right_mat = right_mat.view(B * H, N, N)      # (batch, K, N)

    out_mat = torch.empty((B * H, N, N), dtype=torch.float16, device=device)

    # strides for the kernel
    stride_az, stride_am, stride_ak = left_mat.stride()
    stride_bz, stride_bk, stride_bn = right_mat.stride()
    stride_cz, stride_cm, stride_cn = out_mat.stride()

    # Tiling parameters (tuned for B200 tensor‑cores)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp16

    # ------------------------------------------------------------------
    # Output LayerNorm (float32)
    # ------------------------------------------------------------------
    gamma_out = weights["to_out_norm.weight"].to(device).to(torch.float32)
    beta_out = weights["to_out_norm.bias"].to(device).to(torch.float32)

    out_f32 = out.to(torch.float32)
    mean_o = out_f32.mean(dim=-1, keepdim=True)
    var_o = out_f32.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out_f32 - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * gamma_out + beta_out

    # Apply output gate
    out_norm = out_norm * out_gate.to(torch.float32)

    # ------------------------------------------------------------------
    # Final linear projection back to input dimension
    # ------------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(device).to(torch.float32)   # (dim, hidden_dim)
    output = F.linear(out_norm, w_to_out)    # [B, N, N, dim]

    return output
