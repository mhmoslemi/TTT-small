import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # row block
    pid_n = tl.program_id(1)  # col block
    pid_batch = tl.program_id(2)  # which (batch*hidden) matrix

    # compute global offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k-tiles
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        # offsets for the current k-tile
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        a_ptrs = a_ptr + pid_batch * stride_batch_a \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + pid_batch * stride_batch_b \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

        # mask out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_ptr + pid_batch * stride_batch_c \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Computes the contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    with fused linear projections, gating, and LayerNorms.
    The heavy O(N³·H) matrix multiplication is executed by a
    custom Triton kernel that treats each (batch, hidden) pair as an
    independent matrix multiplication.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------------------------------------------------------
    # LayerNorm over channel dimension
    # ---------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # [B,N,N,dim]

    # ---------------------------------------------------------
    # Cast to half for the bulk of the work
    # ---------------------------------------------------------
    x_h = x.to(torch.float16)

    # ---------------------------------------------------------
    # Linear projections (no bias)
    # ---------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left = F.linear(x_h, left_proj_w)   # [B,N,N,hidden_dim]
    right = F.linear(x_h, right_proj_w) # [B,N,N,hidden_dim]

    # ---------------------------------------------------------
    # Optional masking
    # ---------------------------------------------------------
    if mask is not None and not nomask:
        mask_h = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_h
        right = right * mask_h

    # ---------------------------------------------------------
    # Gating (sigmoid)
    # ---------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------
    # Batched matrix multiplication via Triton
    # out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    # ---------------------------------------------------------
    B, N, _, H = left.shape   # left shape: [B,N,N,H]

    # reshape to (B*H, N, N) where the inner dimension K = N
    # left: rows i, cols k   (M = N, K = N)
    # right: rows k, cols j   (K = N, N = N)
    left_t = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    right_t = right.permute(0, 3, 2, 1).contiguous()        # (B, H, N, N)

    BH = B * H
    A = left_t.view(BH, N, N)
    B_mat = right_t.view(BH, N, N)

    # allocate output
    C = torch.empty_like(A, dtype=torch.float16, device=device)

    # tile sizes – chosen to give high occupancy on A100 / B200
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), BH)

    batched_matmul_kernel[grid](
        A,
        B_mat,
        C,
        N, N, N,                     # M, N, K
        A.stride(1), A.stride(2),    # stride_am, stride_ak
        B_mat.stride(1), B_mat.stride(2),  # stride_bk, stride_bn
        C.stride(1), C.stride(2),    # stride_cm, stride_cn
        A.stride(0), B_mat.stride(0), C.stride(0),  # batch strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out_h = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------------------------------------------------------
    # Post‑processing: LayerNorm on hidden dim, gating, final linear
    # ---------------------------------------------------------
    out_fp32 = out_h.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out_norm = out_norm * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"].to(device)  # (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)  # [B, N, N, dim] float32

    return output

TriMul (outgoing) forward pass with Triton acceleration.

The expensive O(N³·H) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is expressed as a batched matrix multiplication where each
(batch, hidden) pair forms an independent N×N GEMM.
All surrounding LayerNorms, linear projections and gating are performed
with PyTorch, while the cubic contraction is executed by the
`batched_matmul_kernel` defined below.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers for A, B, C
    M, N, K,                           # matrix dimensions (all = seq_len)
    stride_am, stride_ak,              # strides for A: row, col
    stride_bk, stride_bn,              # strides for B: row, col
    stride_cm, stride_cn,              # strides for C: row, col
    stride_batch_a, stride_batch_b, stride_batch_c,  # batch stride
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)            # block row
    pid_n = tl.program_id(1)            # block column
    pid_b = tl.program_id(2)            # which (batch*hidden) matrix

    # global offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in FP32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension in tiles
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)

        # pointers to the current A and B tiles
        a_ptrs = a_ptr + pid_b * stride_batch_a \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + pid_b * stride_batch_b \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

        # mask loads that would go out of bounds
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # write the result back
    c_ptrs = c_ptr + pid_b * stride_batch_c \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32/bool) or None
            - weights: dict mapping parameter names to torch.Tensors
            - config: dict containing at least ``dim`` and ``hidden_dim``.
                      May also contain ``nomask`` (bool).

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1) Input LayerNorm over the channel dimension (dim)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2) Cast to half‑precision for the heavy work
    # -----------------------------------------------------------------
    x_h = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3) Linear projections (bias=False)
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left = F.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x_h, right_proj_w)  # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 4) Optional mask (applied after projection)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask_h = mask.unsqueeze(-1).to(torch.float16)   # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # -----------------------------------------------------------------
    # 5) Gating (sigmoid) – also in half precision
    # -----------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6) Batched matrix multiplication via Triton:
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim

    # reshape to (B*H, N, N) where the second axis is the summation axis (k)
    left_t = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N) rows = i, cols = k
    right_t = right.permute(0, 3, 2, 1).contiguous()        # (B, H, N, N) rows = k, cols = j

    BH = B * H
    A = left_t.view(BH, N, N)   # [BH, M, K] with M=K=N
    B_mat = right_t.view(BH, N, N)  # [BH, K, N]

    # output buffer
    C = torch.empty_like(A, dtype=torch.float16, device=device)  # [BH, M, N]

    # tile sizes tuned for A100 / B200
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), BH)

    batched_matmul_kernel[grid](
        A,
        B_mat,
        C,
        N, N, N,                     # M, N, K
        A.stride(1), A.stride(2),    # stride_am, stride_ak
        B_mat.stride(1), B_mat.stride(2),  # stride_bk, stride_bn
        C.stride(1), C.stride(2),    # stride_cm, stride_cn
        A.stride(0), B_mat.stride(0), C.stride(0),  # batch strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # restore original layout: [B, N, N, hidden_dim]
    out_h = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 7) Post‑processing: LayerNorm on hidden dim, gating, final projection
    # -----------------------------------------------------------------
    out_fp32 = out_h.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out_norm = out_norm * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"].to(device)  # (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)           # [B, N, N, dim] (float32)

    return output
