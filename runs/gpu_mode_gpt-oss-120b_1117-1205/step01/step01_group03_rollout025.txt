@triton.jit
def batched_matmul_forward(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # pointers for this batch
    a_batch = a_ptr + batch_id * stride_batch_a
    b_batch = b_ptr + batch_id * stride_batch_b
    c_batch = c_ptr + batch_id * stride_batch_c

    # compute starting indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # compute mask for M,N dimensions.
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load B tile
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # compute dot product
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # store result
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # Cast to half for speed
    x_h = x.half()

    # Linear projections (no bias)
    left = torch.nn.functional.linear(x_h, weights["left_proj.weight"].half())  # shape [bs,N,N,hidden]
    right = torch.nn.functional.linear(x_h, weights["right_proj.weight"].half())

    # mask handling
    if mask is None:
        mask_f = torch.ones(bs, N, N, device=device, dtype=x_h.dtype)
    else:
        mask_f = mask.to(x_h.dtype)
    mask_f = mask_f.unsqueeze(-1)  # shape [bs,N,N,1]
    left = left * mask_f
    right = right * mask_f

    # Gates
    left_gate = torch.nn.functional.linear(x_h, weights["left_gate.weight"].half()).sigmoid()
    right_gate = torch.nn.functional.linear(x_h, weights["right_gate.weight"].half()).sigmoid()
    out_gate = torch.nn.functional.linear(x_h, weights["out_gate.weight"].half()).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Permute for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (bs, hidden, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (bs, hidden, k, j)

    total_batches = bs * hidden
    left_mat = left_perm.view(total_batches, N, N)
    right_mat = right_perm.view(total_batches, N, N)

    # Output tensor
    out_mat = torch.empty_like(left_mat)  # half

    # Choose block sizes
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    else:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64  # maybe 64 for large

    grid = (total_batches,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))
    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(1), out_mat.stride(2),
        left_mat.stride(0), right_mat.stride(0), out_mat.stride(0),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # Ensure kernel complete
    torch.cuda.synchronize()

    # Reshape back
    out = out_mat.view(bs, hidden, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # Output layer norm (hidden dim)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # apply output gate
    out = out * out_gate.to(torch.float32)

    # Final projection
    output = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float32))

    return output

"""
TriMul (outgoing) forward pass with a fused Triton batched GEMM.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension (dim).
2. Project the normalized tensor to a hidden dimension with two linear
   layers (left_proj, right_proj).  Apply an optional mask.
3. Compute element‑wise gates (left_gate, right_gate, out_gate) with a sigmoid
   of linear transforms and multiply them into the projected tensors.
4. Perform the core O(N³·C) operation:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication L @ Rᵀ for each (b,d) pair.
   A custom Triton kernel computes the batched GEMM in FP16 with FP32
   accumulation, exploiting Tensor‑Cores.
5. Layer‑norm over the hidden dimension, multiply by out_gate and project back
   to the original dimension.

All heavy arithmetic (step 4) runs on the GPU via Triton; the surrounding
operations use PyTorch (FP16 where possible) to keep the implementation
simple yet fast.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions
    stride_am, stride_ak,              # strides for A (M, K)
    stride_bk, stride_bn,              # strides for B (K, N)
    stride_cm, stride_cn,              # strides for C (M, N)
    stride_batch_a, stride_batch_b, stride_batch_c,  # batch stride
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ B, where A is (M×K) and B is (K×N); each
    batch corresponds to one (batch * hidden) slice of the TriMul tensor."""
    batch_id = tl.program_id(0)   # total_batches = B * hidden_dim
    pid_m = tl.program_id(1)      # block row
    pid_n = tl.program_id(2)      # block column

    # Pointers to the current batch
    a_batch = a_ptr + batch_id * stride_batch_a
    b_batch = b_ptr + batch_id * stride_batch_b
    c_batch = c_ptr + batch_id * stride_batch_c

    # Compute tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K sized chunks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M×K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile (K×N)
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Tensor‑core dot (FP16 × FP16 ⇒ FP32 accumulation)
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # Write out C tile (M×N) as FP16
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor, shape [B, N, N, C] (float32)
            - mask_tensor:  torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict: mapping of weight names to torch.Tensor (float32)
            - config_dict:  {"dim": C, "hidden_dim": H, ...}

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 and apply linear projections (bias=False)
    # ------------------------------------------------------------------
    x_h = x.half()

    w_left = weights["left_proj.weight"].to(torch.float16)
    w_right = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, w_left)   # [B, N, N, H]
    right = F.linear(x_h, w_right) # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Mask handling (optional)
    # ------------------------------------------------------------------
    if mask is None:
        mask_f = torch.ones(bs, N, N, device=device, dtype=torch.float16)
    else:
        mask_f = mask.to(torch.float16)
    mask_f = mask_f.unsqueeze(-1)   # [B, N, N, 1]
    left = left * mask_f
    right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear transforms)
    # ------------------------------------------------------------------
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_h, w_left_gate).sigmoid()
    right_gate = F.linear(x_h, w_right_gate).sigmoid()
    out_gate = F.linear(x_h, w_out_gate).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Prepare tensors for batched GEMM: L @ Rᵀ per (b, hidden) pair
    # ------------------------------------------------------------------
    # left: (B, N, N, H)  →  (B, H, N, N) where axes are (batch, hidden, i, k)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: (B, N, N, H) →  (B, H, N, N) where axes are (batch, hidden, k, j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    total_batches = bs * hidden
    left_mat = left_perm.view(total_batches, N, N)
    right_mat = right_perm.view(total_batches, N, N)

    out_mat = torch.empty_like(left_mat)   # fp16 output

    # ------------------------------------------------------------------
    # 6. Batched GEMM kernel launch
    # ------------------------------------------------------------------
    # Block sizes – chosen to fit in registers and leverage Tensor Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64 if N > 512 else 32

    grid = (
        total_batches,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(1), out_mat.stride(2),
        left_mat.stride(0), right_mat.stride(0), out_mat.stride(0),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 7. Reshape back to [B, N, N, H] and convert to FP32
    # ------------------------------------------------------------------
    out = (
        out_mat.view(bs, hidden, N, N)
        .permute(0, 2, 3, 1)
        .to(torch.float32)
    )

    # ------------------------------------------------------------------
    # 8. Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 9. Apply output gate and final linear projection to original dim
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)      # broadcast over hidden dim
    w_to_out = weights["to_out.weight"].to(torch.float32)  # (dim, hidden)
    output = F.linear(out, w_to_out)           # [B, N, N, dim]

    return output
