"""
TriMul (outgoing) forward pass with a fused Triton GEMM kernel.

Algorithm
---------
1. Layer‑norm over the last dimension of the input.
2. Linear projection to a hidden dimension (left/right) and compute three
   sigmoid gates (left, right, out) from the normalized input.
3. Optionally mask the projected tensors.
4. Apply element‑wise gates to left and right.
5. Perform the heavy O(N³) operation:
        out_{i,j,h} = Σ_k left_{i,k,h} * right_{j,k,h}
   which is a batched matrix multiplication left @ rightᵀ.
   This is executed by a custom Triton kernel that processes all
   (batch * hidden_dim) independent GEMMs in one launch.
6. Layer‑norm the result, apply the output gate, and a final linear
   projection back to the original channel dimension.
All heavy arithmetic is carried out in FP16 with FP32 accumulation for
numerical stability; the final output is FP32.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,  # pointers
    M, N, K,              # matrix dimensions
    stride_am, stride_ak,   # strides for A (row, col)
    stride_bk, stride_bn,   # strides for B (row, col)
    stride_cm, stride_cn,   # strides for C (row, col)
    stride_a_batch, stride_b_batch, stride_c_batch,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: C = A @ B   where A,B,C have shape
        [batch, M, K], [batch, K, N], [batch, M, N].
    Accumulation is performed in FP32.
    """
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_b = tl.program_id(2)   # batch index (batch * hidden_dim)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_offset = k_tile * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K)
        mask_k = (k_offset + offs_k) < K

        # Pointers with batch offset
        a_ptrs = (
            a_ptr
            + pid_b * stride_a_batch
            + offs_m[:, None] * stride_am
            + (k_offset + offs_k)[None, :] * stride_ak
        )
        b_ptrs = (
            b_ptr
            + pid_b * stride_b_batch
            + (k_offset + offs_k)[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)  # a,b are fp16, acc is fp32

    # Write result back as fp16
    c_ptrs = (
        c_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing Triangle Multiplicative
    Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32, CUDA)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict with all linear / layer‑norm parameters
            - config: dict containing at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, same device)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. Input layer‑norm
    # -----------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm_fp16 = x_norm.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections and gates (all fp16)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(
        x_norm_fp16, weights["left_proj.weight"].to(torch.float16)
    )
    right = torch.nn.functional.linear(
        x_norm_fp16, weights["right_proj.weight"].to(torch.float16)
    )

    # Gates are computed from the normalized input (as in the reference)
    left_gate = torch.nn.functional.linear(
        x_norm_fp16, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x_norm_fp16, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x_norm_fp16, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4. Apply element‑wise gates
    # -----------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Prepare tensors for the fused GEMM: left @ rightᵀ
    # -----------------------------------------------------------------
    # Rearrange to [B, H, N, N] for easy flattening
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Transpose the last two axes of right to obtain the needed rightᵀ
    right_T = right_perm.transpose(-2, -1).contiguous()  # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    total_batch = B * H

    left_flat = left_perm.view(total_batch, N, N)   # [B*H, N, N]
    right_flat = right_T.view(total_batch, N, N)    # [B*H, N, N]
    out_flat = torch.empty((total_batch, N, N), dtype=torch.float16, device=device)

    # Strides in element units
    stride_a_batch, stride_am, stride_ak = left_flat.stride()
    stride_b_batch, stride_bk, stride_bn = right_flat.stride()
    stride_c_batch, stride_cm, stride_cn = out_flat.stride()

    # -----------------------------------------------------------------
    # 6. Launch the Triton GEMM kernel
    # -----------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        total_batch,
    )

    trimul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 7. Reshape back to [B, N, N, hidden_dim] (FP32)
    # -----------------------------------------------------------------
    out = (
        out_flat.view(B, H, N, N)          # [B, H, N, N]
        .permute(0, 2, 3, 1)               # [B, N, N, H]
        .contiguous()
        .to(torch.float32)
    )

    # -----------------------------------------------------------------
    # 8. Output LayerNorm, output gate, and final projection
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # Apply output gate (cast gate to FP32)
    out = out * out_gate.to(torch.float32)

    # Final linear back to `dim`
    output = torch.nn.functional.linear(
        out, weights["to_out.weight"].to(torch.float32)
    )  # shape [B, N, N, dim]

    return output
