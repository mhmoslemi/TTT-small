"""
TriMul (outgoing) forward pass.

Algorithm:
1. Layer‑norm the input over the last dimension.
2. Linear projections (left/right) and sigmoid gates (left/right/out) using
   the provided weight tensors; everything is cast to fp16.
3. Optional mask is applied to the projected tensors.
4. The core O(N³) operation is a batched matrix multiplication:
       out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   which is equivalent to a batched GEMM left @ rightᵀ for each (b,h).
   This is implemented with a custom Triton kernel that performs many
   independent GEMMs in parallel, accumulating in FP32 and storing in fp16.
5. The result is reshaped back to [B,N,N,hidden] (float32), layer‑normed,
   multiplied by the output gate and projected back to the model dimension.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton kernel: batched GEMM (A @ Bᵀ) where
#   A : [total_batch, M, K]   (fp16)
#   B : [total_batch, K, N]   (fp16)   (right transposed)
#   C : [total_batch, M, N]   (fp16)
# -----------------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # tensor pointers
    M, N, K,                      # matrix dimensions
    stride_am, stride_ak,         # A strides (row, col)
    stride_bk, stride_bn,         # B strides (row, col)
    stride_cm, stride_cn,         # C strides (row, col)
    stride_a_batch, stride_b_batch, stride_c_batch,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col
    pid_b = tl.program_id(2)          # batch (b·h)

    # ----- offsets within the block -----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- loop over K dimension -----
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_offset = k_tile * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K)
        mask_k = (k_offset + offs_k) < K

        # pointers for the current tile
        a_ptrs = (
            a_ptr
            + pid_b * stride_a_batch
            + offs_m[:, None] * stride_am
            + (k_offset + offs_k)[None, :] * stride_ak
        )
        b_ptrs = (
            b_ptr
            + pid_b * stride_b_batch
            + (k_offset + offs_k)[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        # load tiles (apply masks)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # matrix‑multiply‑accumulate
        acc += tl.dot(a, b)   # a: (BM, BK), b: (BK, BN) → (BM, BN)

    # ----- store result -----
    c_ptrs = (
        c_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)


# ----------------------------------------------------------------------------- #
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32 CUDA)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of torch.Tensor containing all linear / layer‑norm params
            - config: dict containing at least {"dim": int, "hidden_dim": int}

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, same device)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------------- #
    # 1. LayerNorm over the last dimension
    # --------------------------------------------------------------------- #
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    # Cast to fp16 for all heavy ops
    x_norm = x_norm.to(torch.float16)

    # --------------------------------------------------------------------- #
    # 2. Linear projections + gates (all fp16, bias=False)
    # --------------------------------------------------------------------- #
    left = torch.nn.functional.linear(
        x_norm, weights["left_proj.weight"].to(torch.float16)
    )                                            # [B, N, N, H]
    right = torch.nn.functional.linear(
        x_norm, weights["right_proj.weight"].to(torch.float16)
    )

    left_gate = torch.nn.functional.linear(
        x_norm, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x_norm, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x_norm, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()                                   # [B, N, N, H]

    # --------------------------------------------------------------------- #
    # 3. Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------------- #
    if mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------------- #
    # 4. Apply element‑wise gates
    # --------------------------------------------------------------------- #
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 5. Prepare tensors for the fused GEMM (left @ rightᵀ)
    # --------------------------------------------------------------------- #
    # Change layout to [B, H, N, N] and make contiguous for nice strides
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Transpose the last two axes of right so we can multiply directly
    right_T = right_perm.transpose(-2, -1).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    total_batch = B * H

    # Collapse batch & head dimensions → (total_batch, N, N)
    left_flat = left_perm.view(total_batch, N, N)
    right_flat = right_T.view(total_batch, N, N)
    out_flat = torch.empty((total_batch, N, N), dtype=torch.float16, device=device)

    # Strides in elements (not bytes)
    stride_a_batch, stride_am, stride_ak = left_flat.stride()
    stride_b_batch, stride_bk, stride_bn = right_flat.stride()
    stride_c_batch, stride_cm, stride_cn = out_flat.stride()

    # --------------------------------------------------------------------- #
    # 6. Launch the Triton GEMM kernel
    # --------------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        total_batch,
    )
    trimul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                     # M
        N,                     # N
        N,                     # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # --------------------------------------------------------------------- #
    # 7. Reshape back to [B, N, N, H] and continue in FP32
    # --------------------------------------------------------------------- #
    out = (
        out_flat.view(B, H, N, N)      # (B, H, N, N)
        .permute(0, 2, 3, 1)           # (B, N, N, H)
        .contiguous()
        .to(torch.float32)
    )

    # --------------------------------------------------------------------- #
    # 8. Output LayerNorm, output gate and final linear projection
    # --------------------------------------------------------------------- #
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate.to(torch.float32)   # apply output gate

    output = torch.nn.functional.linear(
        out, weights["to_out.weight"].to(torch.float32)
    )  # shape [B, N, N, dim]

    return output
