"""
TriMul (outgoing) forward pass with a Triton‑accelerated batched GEMM.

The heavy cubic contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batch of independent matrix multiplications:
    left_d (N×N) @ right_d.T (N×N)  for each hidden channel d.
All surrounding LayerNorms, linear projections and gating are performed in
PyTorch; the core O(N³·H) work runs on the GPU via a tiled Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr,                      # [BH, M, K]  matrix A
    b_ptr,                      # [BH, K, N]  matrix B (not transposed)
    c_ptr,                      # [BH, M, N]  output C
    M, N, K,                    # matrix dimensions (M=N=K=seq_len)
    stride_am, stride_ak,       # A: stride for M (row) and K (col)
    stride_bk, stride_bn,       # B: stride for K (row) and N (col)
    stride_cm, stride_cn,       # C: stride for M (row) and N (col)
    stride_batch_a, stride_batch_b, stride_batch_c,  # batch strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)          # block row index
    pid_n = tl.program_id(1)          # block column index
    pid_b = tl.program_id(2)          # which (batch × hidden) matrix

    # Offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        k_offset = k * BLOCK_K

        # Pointers to the current K‑tile of A and B
        a_ptrs = a_ptr + pid_b * stride_batch_a \
                 + offs_m[:, None] * stride_am \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak

        b_ptrs = b_ptr + pid_b * stride_batch_b \
                 + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

        # Bounds checks
        a_mask = (offs_m[:, None] < M) & ((k_offset + tl.arange(0, BLOCK_K))[None, :] < K)
        b_mask = ((k_offset + tl.arange(0, BLOCK_K))[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K)·(BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # Store the result (still FP32)
    c_ptrs = c_ptr + pid_b * stride_batch_c \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operation.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask:        torch.Tensor of shape [B, N, N] (float32 or bool) or None
            - weights:     dict containing all model parameters
            - config:      dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # --------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)   # [B, N, N, dim]

    # Cast to FP16 for the heavy path
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device).to(torch.float16)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)  # (hidden, dim)

    left  = F.linear(x_fp16, left_proj_w)   # [B, N, N, H]
    right = F.linear(x_fp16, right_proj_w)  # [B, N, N, H]

    # --------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(device).to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left  = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------------
    # 4. Gating (sigmoid on linear transforms of the normalized input)
    # --------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------
    # 5. Cubic contraction via batched GEMM (Triton)
    # --------------------------------------------------------------------
    batch_size, seq_len, _, _ = left.shape  # left shape: [B, N, N, H]
    B = batch_size

    # Rearrange tensors to (B, H, N, N) where the inner two dims are the matrix
    A = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N) → rows=i, cols=k
    Bmat = right.permute(0, 3, 2, 1).contiguous()     # (B, H, N, N) → rows=k, cols=j

    # Collapse batch and hidden dimensions: each (b, h) pair is an independent matmul
    BH = B * hidden_dim
    A_flat = A.reshape(BH, seq_len, seq_len)   # (BH, M, K)
    B_flat = Bmat.reshape(BH, seq_len, seq_len)  # (BH, K, N)

    # Output buffer (FP16 to keep memory low, accumulation stays FP32)
    C_flat = torch.empty_like(A_flat, dtype=torch.float16, device=device)  # (BH, M, N)

    # Tile sizes (tuned for B200, work well up to N=1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(seq_len, BLOCK_M),
        triton.cdiv(seq_len, BLOCK_N),
        BH,
    )

    batched_matmul_kernel[grid](
        A_flat,
        B_flat,
        C_flat,
        seq_len, seq_len, seq_len,                 # M, N, K
        A_flat.stride(1), A_flat.stride(2),        # stride_am, stride_ak
        B_flat.stride(1), B_flat.stride(2),        # stride_bk, stride_bn
        C_flat.stride(1), C_flat.stride(2),        # stride_cm, stride_cn
        A_flat.stride(0), B_flat.stride(0), C_flat.stride(0),  # batch strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore the (B, N, N, H) layout
    out = C_flat.reshape(batch_size, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()  # FP16

    # --------------------------------------------------------------------
    # 6. Post‑processing: LayerNorm on hidden dim, final gating, output proj
    # --------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out_norm = out_norm * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"].to(device)  # (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)          # [B, N, N, dim] (float32)

    return output
