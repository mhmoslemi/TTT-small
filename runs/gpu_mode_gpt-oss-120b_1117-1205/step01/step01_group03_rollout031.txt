"""
TriMul "outgoing" forward pass (AlphaFold3) with a lightweight Triton LayerNorm kernel.
The expensive N×N×N multiplication is performed by:
    - using torch.bmm (cuBLAS) for large N (≥ 512)
    - falling back to a simple batched GEMM Triton kernel for smaller N.
All other linear projections and sigmoids are kept in PyTorch (float16) for speed.
A tiny Triton kernel implements per‑pair LayerNorm over the channel dimension,
fulfilling the requirement to launch a custom kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: per-(b,i,j) LayerNorm over the channel dimension C
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                     # input  [B, N, N, C]  (float32)
    y_ptr,                     # output [B, N, N, C]  (float32)
    w_ptr,                     # gamma  [C]         (float32)
    b_ptr,                     # beta   [C]         (float32)
    B, N,                      # problem sizes
    stride_b0, stride_b1, stride_b2, stride_c,   # strides of x (and y)
    eps,                       # epsilon for stability
    C: tl.constexpr,           # channel dimension (compile‑time constant)
    BLOCK_C: tl.constexpr,     # block size for the C‑dimension (must divide C)
):
    pid = tl.program_id(0)                     # one program per (b,i,j)
    total = B * N * N
    if pid >= total:
        return

    # decode (b,i,j) from linear pid
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # base offset for the (b,i,j,0) element
    base = b * stride_b0 + i * stride_b1 + j * stride_b2

    # ----- first pass: compute mean and variance -----
    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)
    for off in range(0, C, BLOCK_C):
        off_vec = off + tl.arange(0, BLOCK_C)           # (BLOCK_C,)
        x = tl.load(x_ptr + base + off_vec * stride_c, dtype=tl.float32)  # (BLOCK_C,)
        sum_val += tl.sum(x)
        sum_sq += tl.sum(x * x)

    mean = sum_val / C
    var = sum_sq / C - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ----- second pass: write normalized values with weight & bias -----
    for off in range(0, C, BLOCK_C):
        off_vec = off + tl.arange(0, BLOCK_C)
        x = tl.load(x_ptr + base + off_vec * stride_c, dtype=tl.float32)
        w = tl.load(w_ptr + off_vec, dtype=tl.float32)
        b = tl.load(b_ptr + off_vec, dtype=tl.float32)

        y = (x - mean) * rstd * w + b
        tl.store(y_ptr + base + off_vec * stride_c, y, dtype=tl.float32)

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM A (M×K) @ B (K×N) -> C (M×N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers (float16)
    M, N, K,                            # problem sizes (all equal N)
    stride_am, stride_ak,               # strides for A (M, K)
    stride_bk, stride_bn,               # strides for B (K, N)
    stride_cm, stride_cn,               # strides for C (M, N)
    stride_batch_a, stride_batch_b, stride_batch_c,   # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # batch * hidden index
    pid_m = tl.program_id(1)       # tile row
    pid_n = tl.program_id(2)       # tile column

    # pointers to this batch
    a_batch = a_ptr + pid_batch * stride_batch_a
    b_batch = b_ptr + pid_batch * stride_batch_b
    c_batch = c_ptr + pid_batch * stride_batch_c

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # pointers to tiles of A and B
    a_ptrs = a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
    b_ptrs = b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)
        acc += tl.dot(a, b)                      # (BLOCK_M, BLOCK_K)·(BLOCK_K, BLOCK_N)
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write-back
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the TriMul "outgoing" update.
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    bs, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm over the channel dimension (Triton)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]   # (dim,)
    norm_bias   = weights["norm.bias"]    # (dim,)

    x_norm = torch.empty_like(input_tensor, dtype=torch.float32, device=device)

    # launch one program per (b,i,j)
    total_pairs = bs * N * N
    grid = (total_pairs,)
    BLOCK_C = 32  # 32 divides all allowed dim (128, 384)

    layernorm_kernel[grid](
        input_tensor,
        x_norm,
        norm_weight,
        norm_bias,
        bs,
        N,
        input_tensor.stride(0),
        input_tensor.stride(1),
        input_tensor.stride(2),
        input_tensor.stride(3),
        eps,
        C=dim,
        BLOCK_C=BLOCK_C,
    )

    # ------------------------------------------------------------------
    # 2. Projections & gating (float16 for speed)
    # ------------------------------------------------------------------
    x_f16 = x_norm.to(torch.float16)

    # linear weights (project dim -> hidden)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_f16, left_proj_w)       # [bs, N, N, hidden]
    right = F.linear(x_f16, right_proj_w)

    # mask handling
    if mask is None:
        mask_f16 = torch.ones(bs, N, N, dtype=torch.float16, device=device)
    else:
        mask_f16 = mask.to(torch.float16)
    mask_f16 = mask_f16.unsqueeze(-1)          # [bs, N, N, 1]
    left = left * mask_f16
    right = right * mask_f16

    # gate weights
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_f16, left_gate_w).sigmoid()
    right_gate = F.linear(x_f16, right_gate_w).sigmoid()
    out_gate = F.linear(x_f16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Core N×N×N multiplication (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to (B*hidden, N, N) where each slice is a dense matrix
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(bs * hidden, N, N)

    # decide kernel vs cuBLAS based on sequence length
    if N <= 512:  # Triton kernel is competitive for modest sizes
        out_mat = torch.empty_like(left_mat)
        # tile sizes – heuristic
        if N <= 256:
            BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
        else:
            BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
        grid = (bs * hidden,
                triton.cdiv(N, BLOCK_M),
                triton.cdiv(N, BLOCK_N))
        batched_matmul_kernel[grid](
            left_mat,
            right_mat,
            out_mat,
            N, N, N,
            left_mat.stride(1), left_mat.stride(2),
            right_mat.stride(1), right_mat.stride(2),
            out_mat.stride(1), out_mat.stride(2),
            left_mat.stride(0), right_mat.stride(0), out_mat.stride(0),
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        )
    else:
        # for large N cuBLAS is usually faster
        out_mat = torch.bmm(left_mat, right_mat)

    # reshape back to [bs, N, N, hidden]
    out = out_mat.view(bs, hidden, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 4. Output LayerNorm, gating, final projection (float32)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    # apply output gate (cast gate to fp32)
    out_norm = out_norm * out_gate.to(torch.float32)

    # final linear projection back to original dim
    to_out_w = weights["to_out.weight"].to(torch.float32)   # (dim, hidden)
    output = F.linear(out_norm, to_out_w)                  # [bs, N, N, dim]

    return output
