@triton.jit
def matmul_kernel(
    A, B, C,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)  # batch index
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # init accum
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        a_ptrs = A_ptr + pid_z*stride_az + offs_m[:,None]*stride_am + (k_offset + offs_k)[None,:]*stride_ak
        b_ptrs = B_ptr + pid_z*stride_bz + (k_offset + offs_k)[:,None]*stride_bk + offs_n[None,:]*stride_bn

        a = tl.load(a_ptrs, mask=(offs_m < M)[:,None] & ((k_offset+offs_k) < K)[None,:], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=((k_offset+offs_k) < K)[:,None] & (offs_n < N)[None,:], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)
    
    c_ptrs = C_ptr + pid_z*stride_cz + offs_m[:,None]*stride_cm + offs_n[None,:]*stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_m < M)[:,None] & (offs_n < N)[None,:])
```

"""
TriMul (outgoing) forward implementation with a fused Triton kernel.

The heavy computation is the contraction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
which can be expressed as a batched matrix multiplication
    out[b,:, :, d] = left[b,:, :, d] @ right[b,:, :, d].T .
We flatten the batch and hidden dimensions (B * hidden) and compute the
matmul with a custom Triton GEMM kernel (tiles of 64×64×64). All other
operations (layer‑norms, linear projections, gates, mask handling) are
performed in PyTorch. Mixed‑precision (fp16) is used for the projection
and the Triton kernel, while the final output is returned in fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ B   (A: (M,K), B: (K,N))
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer to output tile
    c_ptrs = (
        C_ptr
        + batch_id * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile (M x K)
        a_ptrs = (
            A_ptr
            + batch_id * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # B tile (K x N)
        b_ptrs = (
            B_ptr
            + batch_id * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        # accumulate
        acc += tl.dot(a, b)

    # write back
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )

# ----------------------------------------------------------------------
# Custom forward kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data: tuple (input, mask, weights, config)
            input : torch.Tensor [B, N, N, dim] (float32)
            mask  : torch.Tensor [B, N, N] (bool or float) – may be None
            weights: dict of torch.Tensors for all linear / norm params
            config : dict with keys "dim", "hidden_dim", "nomask" (bool)
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm on input (over last dim)
    # ------------------------------------------------------------------
    gamma = weights["norm.weight"].to(torch.float32)
    beta = weights["norm.bias"].to(torch.float32)

    x = input_tensor.to(torch.float32)
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * gamma + beta          # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (fp16) and masking
    # ------------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    w_left = weights["left_proj.weight"].to(torch.float16)      # (hidden_dim, dim)
    w_right = weights["right_proj.weight"].to(torch.float16)

    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    # Projections
    left = F.linear(x_h, w_left)          # [B, N, N, hidden_dim]
    right = F.linear(x_h, w_right)

    # Optional mask
    if mask is not None and not config.get("nomask", False):
        mask_exp = mask.unsqueeze(-1).to(x_h.dtype)   # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # Gates (sigmoid)
    left_gate = F.linear(x_h, w_left_gate).sigmoid()
    right_gate = F.linear(x_h, w_right_gate).sigmoid()
    out_gate = F.linear(x_h, w_out_gate).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton batched GEMM: out = left @ right^T  (per hidden dim)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape               # H == hidden_dim
    # reshape to (B*H, N, N) and prepare right as transposed (k as rows)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (M=N, K=N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N) # (K=N, N)

    out_mat = torch.empty((B * H, N, N), dtype=torch.float16, device=device)

    # strides
    stride_az, stride_am, stride_ak = left_mat.stride()
    stride_bz, stride_bk, stride_bn = right_mat.stride()
    stride_cz, stride_cm, stride_cn = out_mat.stride()

    # tiling
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to [B, N, N, hidden_dim] (fp16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out_f32 = out.to(torch.float32)      # for the following layer‑norm

    # ------------------------------------------------------------------
    # Output LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    gamma_out = weights["to_out_norm.weight"].to(torch.float32)
    beta_out = weights["to_out_norm.bias"].to(torch.float32)

    mean_o = out_f32.mean(dim=-1, keepdim=True)
    var_o = out_f32.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out_f32 - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * gamma_out + beta_out

    # Apply output gate (cast gate to fp32)
    out_norm = out_norm * out_gate.to(torch.float32)

    # ------------------------------------------------------------------
    # Final linear projection back to input dimension
    # ------------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float32)
    output = F.linear(out_norm, w_to_out)   # [B, N, N, dim], fp32

    return output
