    left = torch.nn.functional.linear(x_flat, left_proj_weight_bf).view(B, N, N, hidden_dim)

"""
TriMul outgoing kernel (optimized).

The operator works on a 4‑D tensor x ∈ ℝ^{B×N×N×C} (C = dim).
Steps
------
1. Layer‑norm over the last dimension (dim) – torch implementation.
2. Mixed‑precision (bfloat16) linear projections:
      left  = x·W_left   , right = x·W_right
   and three gating projections (left_gate, right_gate, out_gate) followed by sigmoid.
3. Optional pairwise mask applied to left/right.
4. Apply gates: left *= left_gate , right *= right_gate.
5. Pairwise contraction over the sequence index k:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication per hidden dimension:
      out_h = left_h @ right_hᵀ   for each h.
   Implemented with torch.matmul on bfloat16 tensors.
6. Layer‑norm over the hidden dimension (hidden_dim) using a custom Triton kernel.
7. Scale by out_gate, then a final linear projection back to dim.

A Triton kernel is used for step 6 (hidden‑dim layer‑norm). All other
heavy work (projections, gating, and the O(N³) contraction) is performed
in mixed precision with highly optimized PyTorch / cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_hidden_dim_kernel(
    inp_ptr,               # *float32  [B,N,N,H]
    out_ptr,               # *float32  [B,N,N,H]
    weight_ptr,            # *float32  [H]
    bias_ptr,              # *float32  [H]
    B,                     # tl.int32
    N,                     # tl.int32
    H,                     # tl.int32
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,   # tl.int32
    stride_out_b, stride_out_i, stride_out_j, stride_out_h, # tl.int32
    eps: tl.constexpr,                                   # const float
    BLOCK_H: tl.constexpr                               # compile‑time block size (≥ hidden_dim)
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # decode (b,i,j) from pid
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # base pointer for this (b,i,j) slice
    inp_base = b * stride_in_b + i * stride_in_i + j * stride_in_j
    out_base = b * stride_out_b + i * stride_out_i + j * stride_out_j

    # ------------------------------------------------------------------
    # Load the hidden‑dim vector
    x = tl.load(
        inp_ptr + inp_base + offs_h * stride_in_h,
        mask=mask_h,
        other=0.0
    )

    # mean over hidden dim
    sum_x = tl.sum(x, axis=0)
    mean = sum_x / H

    # variance
    diff = x - mean
    var = tl.sum(diff * diff, axis=0) / H

    # normalize
    inv_std = tl.rsqrt(var + eps)
    norm = diff * inv_std

    # apply learned scale & bias
    w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)
    b_ = tl.load(bias_ptr + offs_h, mask=mask_h, other=0.0)
    out = norm * w + b_

    # store result
    tl.store(
        out_ptr + out_base + offs_h * stride_out_h,
        out,
        mask=mask_h
    )

    # 4. Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Pairwise contraction over k using batched matmul (BF16)
    # left: (B,N,N,H) -> (B,H,N,N)
    left_perm = left.permute(0, 3, 1, 2)    # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2)  # (B, H, N, N)

    # out_h = left_h @ right_hᵀ   -> (B, H, N, N)
    out = torch.matmul(left_perm, right_perm.transpose(-2, -1))

    # back to (B,N,N,H) and convert to float32 for the layernorm kernel
    out = out.permute(0, 2, 3, 1).contiguous().to(torch.float32)   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 6. Hidden‑dim LayerNorm (Triton)
    out_norm = torch.empty_like(out)

    s_b, s_i, s_j, s_h = out.stride()
    total_progs = B * N * N
    grid = (total_progs,)

    BLOCK_H = 128  # hidden_dim ≤ 128 for all test configurations

    layernorm_hidden_dim_kernel[grid](
        out,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        B,
        N,
        hidden_dim,
        s_b, s_i, s_j, s_h,
        s_b, s_i, s_j, s_h,
        eps=1e-5,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 7. Out gate (float32) – compute once from the *original* normalized x
    out_gate = F.linear(
        x.view(-1, dim),                # float32
        weights["out_gate.weight"],     # float32 → (hidden_dim, dim)
    ).view(B, N, N, hidden_dim).sigmoid()  # float32

    # Scale by out gate
    out_scaled = out_norm * out_gate

    # ------------------------------------------------------------------
    # 8. Final projection back to dim
    final = F.linear(
        out_scaled.view(-1, hidden_dim),   # (B*N*N, hidden_dim)
        weights["to_out.weight"],          # (dim, hidden_dim)
    ).view(B, N, N, dim)

    return final
```
