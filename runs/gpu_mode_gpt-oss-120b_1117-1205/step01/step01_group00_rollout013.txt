@triton.jit
def triton_tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                          N, H,
                          stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
                          stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
                          stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):

    # grid: (total_batch, num_blocks_M, num_blocks_N)

    batch_id = tl.program_id(0) # combined batch id
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute base offsets: batch_id -> b = batch_id // H, d = batch_id % H
    b = batch_id // H
    d = batch_id % H

    # Compute base pointer for this (b,d)
    left_base = left_ptr + b * stride_left_batch + d * stride_left_h
    right_base = right_ptr + b * stride_right_batch + d * stride_right_h
    out_base = out_ptr + b * stride_out_batch + d * stride_out_h

    # Offset for tile rows and cols
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile (M x K)
        # left[i,k] = left_base + i*stride_i + k*stride_k 
        # where stride_i = stride_left_i (N*H), stride_k = stride_left_k (H)
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load right tile (K x N) but we need right_T shape (K,N) where right_T[k,j] = right[b,j,k,d]
        # right_base + j*stride_right_j + k*stride_right_k
        b_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # compute dot product
        acc += tl.dot(a, b)

    # Store result
    c_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

                          N, H,
                          stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
                          stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
                          stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # grid: (total_batch, n_blocks_m, n_blocks_n)
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute b and d
    hid = batch_id % H
    b = batch_id // H

    # calculate base pointers for this (b, d)
    left_base = left_ptr + b * stride_left_batch + hid * stride_left_h
    right_base = right_ptr + b * stride_right_batch + hid * stride_right_h
    out_base = out_ptr + b * stride_out_batch + hid * stride_out_h

    # tile offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension in tiles
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = kk * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        mask_k = offs_k < N

        # Load A tile left[i, k] dimensions: (M,K)
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right_base + k*stride_right_k + j*stride_right_j

        acc += tl.dot(a, b)
    # Store result: out[i, j]
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j

"""
TriMul (outgoing) forward pass with a Triton‑accelerated batched matrix multiplication.

Algorithm
---------
1. Layer‑norm over the input channel dimension (float32).
2. Linear projections **left** and **right** (float16) and three gating projections
   (float16).  Gate values are obtained with sigmoid and multiplied element‑wise.
3. Optional mask is applied (broadcast to the hidden dimension).
4. Core contraction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication for each (b,d) pair.
   The kernel works directly on the original memory layout
   (B, N, N, H) without any extra transpose/copy, using the strides:
        left : (batch, i, k, h)  stride = (N·N·H, N·H, H, 1)
        right: (batch, j, k, h)  stride = (N·N·H, N·H, H, 1)
   For each hidden slice `d` we compute left @ rightᵀ using
   tiling (BLOCK_M×BLOCK_N) and a K‑loop (BLOCK_K).  Accumulation is in FP32.
5. The result is cast back to FP32, layer‑norm over the hidden dimension,
   multiplied by the output gate, and finally projected back to the original
   channel dimension.

Only the heavy N³ contraction is offloaded to Triton; all other
operations stay in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,            # pointers
    N, H,                                   # matrix size and hidden dim
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched matrix multiplication:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    left  : (B, N, N, H)  (i,k) layout
    right : (B, N, N, H)  (j,k) layout, accessed as transposed (k,j)
    out   : (B, N, N, H)
    """
    batch_id = tl.program_id(0)   # combined (b, d) index
    pid_m   = tl.program_id(1)   # i‑tile
    pid_n   = tl.program_id(2)   # j‑tile

    # decode batch_id into (b, d)
    d = batch_id % H
    b = batch_id // H

    # base pointers for this (b, d) slice
    left_base  = left_ptr  + b * stride_left_batch  + d * stride_left_h
    right_base = right_ptr + b * stride_right_batch + d * stride_right_h
    out_base   = out_ptr   + b * stride_out_batch   + d * stride_out_h

    # tile offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = kk * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i,k] – stride (i) = stride_left_i, (k) = stride_left_k
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # right_T[k,j] = right[j,k] – stride (k) = stride_right_k, (j) = stride_right_j
        b_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    # store result
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or numeric)
            - weights_dict: mapping from parameter names to torch.Tensor
            - config_dict: must contain 'dim', 'hidden_dim', and optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # work in half precision from now on
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half‑precision, no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 4. Gates (half‑precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton
    #    Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # Allocate output tensor (half precision)
    out = torch.empty_like(left, dtype=torch.float16, device=device)

    # Strides (B, N, N, H) layout
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_batch, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    total_batches = B * H
    grid = (
        total_batches,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    triton_tri_mul_kernel[grid](
        left,
        right,
        out,
        N,
        H,
        stride_l_batch,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_batch,
        stride_r_j,
        stride_r_k,
        stride_r_h,
        stride_o_batch,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert result to float32 for the remaining operations
    out_f32 = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, output gate, and final projection
    # ------------------------------------------------------------------
    out_f32 = F.layer_norm(
        out_f32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out_f32 = out_f32 * out_gate.to(torch.float32)
    out_f32 = F.linear(out_f32, weights["to_out.weight"].to(torch.float32))

    return out_f32
