```python
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # sometimes config may include nominal. But we use config.

    # LayerNorm
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_w, bias=norm_b
    )
    # To fp16 for speed
    x = x.to(torch.float16)

    # Linear projections (no bias)
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left = torch.nn.functional.linear(x, left_proj_w) # shape B,N,N,hidden_dim
    right = torch.nn.functional.linear(x, right_proj_w)

    # Gates
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Apply mask and gates using Triton:
    if mask is None:
        # mask is all ones => just elementwise multiply left and left_gate
        # we can skip kernel; but to satisfy Triton usage, we can still call kernel with mask = torch.ones_like(mask)
        mask = torch.ones((input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]), dtype=torch.float16, device=device)
    mask = mask.to(torch.float16) # shape [B,N,N]

    # Allocate output for left/right after gating
    left_gated = torch.empty_like(left)
    right_gated = torch.empty_like(right)

    # Launch Triton kernel
    B, N, _, H = left.shape
    total_el = B * N * N * H
    BLOCK_SIZE = 1024  # tune
    grid = (triton.cdiv(total_el, BLOCK_SIZE),)
    apply_gate_mask_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask,
        out_left_ptr=left_gated,
        out_right_ptr=right_gated,
        B=B, N=N, H=H,
        stride_left_b=left.stride(0), stride_left_n1=left.stride(1),
        stride_left_n2=left.stride(2), stride_left_h=left.stride(3),
        stride_right_b=right.stride(0), stride_right_n1=right.stride(1),
        stride_right_n2=right.stride(2), stride_right_h=right.stride(3),
        stride_lgate_b=left_gate.stride(0), stride_lgate_n1=left_gate.stride(1),
        stride_lgate_n2=left_gate.stride(2), stride_lgate_h=left_gate.stride(3),
        stride_rgate_b=right_gate.stride(0), stride_rgate_n1=right_gate.stride(1),
        stride_rgate_n2=right_gate.stride(2), stride_rgate_h=right_gate.stride(3),
        stride_mask_b=mask.stride(0), stride_mask_n1=mask.stride(1),
        stride_mask_n2=mask.stride(2),
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # Now left_gated and right_gated shape [B,N,N,H]
    # Contraction using batched GEMM
    # Permute to (B, H, N, N) for A: left_gated, B: right_gated transposed
    A = left_gated.permute(0, 3, 1, 2).contiguous()
    Bmat = right_gated.permute(0, 3, 2, 1).contiguous()
    BH = B * H
    A_flat = A.view(BH, N, N)
    B_flat = Bmat.view(BH, N, N)  # Note Bmat already shape (B, H, N, N)

    # Perform batched matrix multiplication
    out_flat = torch.bmm(A_flat, B_flat)  # shape (BH, N, N) fp16

    # Reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # shape [B,N,N,H]

    # Post processing
    out_fp32 = out.to(torch.float32)  # for layernorm and final linear

    # Apply to_out_norm and out_gate
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    out_norm = torch.nn.functional.layer_norm(
        out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b
    )
    out_norm = out_norm * out_gate.to(torch.float32)  # out_gate shape matches out

    to_out_w = weights["to_out.weight"].to(device)  # shape (dim, hidden_dim)
    # final linear
    output = torch.nn.functional.linear(out_norm, to_out_w)  # shape [B,N,N,dim]

    return output
```

```python
def apply_gate_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    stride_left_b, stride_left_n1, stride_left_n2, stride_left_h,
    stride_right_b, stride_right_n1, stride_right_n2, stride_right_h,
    stride_lgate_b, stride_lgate_n1, stride_lgate_n2, stride_lgate_h,
    stride_rgate_b, stride_rgate_n1, stride_rgate_n2, stride_rgate_h,
    stride_mask_b, stride_mask_n1, stride_mask_n2,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total_elements = B * N * N * H
    mask = offs < total_elements

    # Compute coordinates
    # Using integer division and modulo
    # off = offs
    off = offs

    h = tl.remainder(off, H)
    off = off // H
    j = tl.remainder(off, N)
    off = off // N
    i = tl.remainder(off, N)
    b = off // N

    # Compute addresses
    left_offset = b * stride_left_b + i * stride_left_n1 + j * stride_left_n2 + h * stride_left_h
    right_offset = b * stride_right_b + i * stride_right_n1 + j * stride_right_n2 + h * stride_right_h
    left_gate_offset = b * stride_lgate_b + i * stride_lgate_n1 + j * stride_lgate_n2 + h * stride_lgate_h
    right_gate_offset = b * stride_rgate_b + i * stride_rgate_n1 + j * stride_rgate_n2 + h * stride_rgate_h

    # mask offset (no H)
    mask_offset = b * stride_mask_b + i * stride_mask_n1 + j * stride_mask_n2

    # Load
    left_val = tl.load(left_ptr + left_offset, mask=mask, other=tl.float16(0))
    right_val = tl.load(right_ptr + right_offset, mask=mask, other=tl.float16(0))
    left_gate_val = tl.load(left_gate_ptr + left_gate_offset, mask=mask, other=tl.float16(0))
    right_gate_val = tl.load(right_gate_ptr + right_gate_offset, mask=mask, other=tl.float16(0))
    mask_val = tl.load(mask_ptr + mask_offset, mask=mask, other=tl.float16(0))

    out_left = left_val * left_gate_val * mask_val
    out_right = right_val * right_gate_val * mask_val

    tl.store(out_left_ptr + left_offset, out_left, mask=mask)
    tl.store(out_right_ptr + right_offset, out_right, mask=mask)
```

"""
TriMul (outgoing) forward pass with Triton‑fused mask‑gate multiplication
and a cuBLAS batched GEMM for the cubic contraction.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def apply_gate_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offs < total

    off = offs
    h = off % H
    off = off // H
    j = off % N
    off = off // N
    i = off % N
    b = off // N

    left_off = b * stride_left_b + i * stride_left_i + j * stride_left_j + h * stride_left_h
    right_off = b * stride_right_b + i * stride_right_i + j * stride_right_j + h * stride_right_h
    lgate_off = b * stride_lgate_b + i * stride_lgate_i + j * stride_lgate_j + h * stride_lgate_h
    rgate_off = b * stride_rgate_b + i * stride_rgate_i + j * stride_rgate_j + h * stride_rgate_h

    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    left_val = tl.load(left_ptr + left_off, mask=mask, other=tl.float16(0.0))
    right_val = tl.load(right_ptr + right_off, mask=mask, other=tl.float16(0.0))
    lgate_val = tl.load(left_gate_ptr + lgate_off, mask=mask, other=tl.float16(0.0))
    rgate_val = tl.load(right_gate_ptr + rgate_off, mask=mask, other=tl.float16(0.0))
    mask_val = tl.load(mask_ptr + mask_off, mask=mask, other=tl.float16(0.0))

    out_l = left_val * lgate_val * mask_val
    out_r = right_val * rgate_val * mask_val

    tl.store(out_left_ptr + left_off, out_l, mask=mask)
    tl.store(out_right_ptr + right_off, out_r, mask=mask)

def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul.
    Input:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_fp16 = torch.float16

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # --------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device),
        bias=weights["norm.bias"].to(device)
    )
    x = x.to(dtype_fp16)

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(device).to(dtype_fp16))
    right = F.linear(x, weights["right_proj.weight"].to(device).to(dtype_fp16))

    # --------------------------------------------------------------
    # 3. Gates (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(device).to(dtype_fp16)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(device).to(dtype_fp16)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(device).to(dtype_fp16)))

    # --------------------------------------------------------------
    # 4. Apply mask and gates via a tiny Triton kernel
    # --------------------------------------------------------------
    # mask may be None or a tensor of shape [B, N, N]
    if mask is None:
        mask = torch.ones_like(input_tensor[..., 0])  # shape [B,N,N]
    mask = mask.to(dtype_fp16)

    B, N, _, H = left.shape  # H == hidden_dim

    left_gated = torch.empty_like(left)
    right_gated = torch.empty_like(right)

    total_elements = B * N * N * H
    BLOCK_SIZE = 1024  # adjust for occupancy
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

    apply_gate_mask_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask,
        out_left_ptr=left_gated,
        out_right_ptr=right_gated,
        B=B, N=N, H=H,
        stride_left_b=left.stride(0), stride_left_i=left.stride(1), stride_left_j=left.stride(2), stride_left_h=left.stride(3),
        stride_right_b=right.stride(0), stride_right_i=right.stride(1), stride_right_j=right.stride(2), stride_right_h=right.stride(3),
        stride_lgate_b=left_gate.stride(0), stride_lgate_i=left_gate.stride(1), stride_lgate_j=left_gate.stride(2), stride_lgate_h=left_gate.stride(3),
        stride_rgate_b=right_gate.stride(0), stride_rgate_i=right_gate.stride(1), stride_rgate_j=right_gate.stride(2), stride_rgate_h=right_gate.stride(3),
        stride_mask_b=mask.stride(0), stride_mask_i=mask.stride(1), stride_mask_j=mask.stride(2),
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # --------------------------------------------------------------
    # 5. Cubic contraction via batched GEMM (cuBLAS)
    # --------------------------------------------------------------
    # left_gated shape [B,N,N,H]
    # right_gated shape [B,N,N,H]
    # We need sum_k left[i,k,d] * right[j,k,d] -> matmul left (i,k) @ right_transposed (k,j)
    A = left_gated.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    Bmat = right_gated.permute(0, 3, 2, 1).contiguous()      # (B, H, N, N)

    BH = B * H
    A_flat = A.view(BH, N, N)
    B_flat = Bmat.view(BH, N, N)

    out_flat = torch.bmm(A_flat, B_flat)                     # (BH, N, N), fp16

    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # --------------------------------------------------------------
    # 6. Post‑processing (LayerNorm, output gate, final linear)
    # --------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    out_norm = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device),
        bias=weights["to_out_norm.bias"].to(device)
    )
    out_norm = out_norm * out_gate.to(torch.float32)

    output = F.linear(out_norm, weights["to_out.weight"].to(device))
    return output

TriMul (outgoing) forward pass.

* LayerNorm + linear projections are done in PyTorch (fp16 for speed).
* A tiny Triton kernel fuses mask application and per‑position gating.
* The cubic contraction Σ_k left[i,k,d]·right[j,k,d] is rewritten as a batched
  matrix multiplication (left @ right.T) and executed by cuBLAS via torch.bmm,
  which is far faster than a pure Triton implementation for the large
  N×N × hidden_dim workloads encountered in AlphaFold3.
* Post‑processing (LayerNorm, output gate, final linear) is performed in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def apply_gate_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offs < total

    # Decompose linear index into (b,i,j,h)
    off = offs
    h = off % H
    off = off // H
    j = off % N
    off = off // N
    i = off % N
    b = off // N

    # Compute flat offsets for each tensor
    left_off = b * stride_left_b + i * stride_left_i + j * stride_left_j + h * stride_left_h
    right_off = b * stride_right_b + i * stride_right_i + j * stride_right_j + h * stride_right_h
    lgate_off = b * stride_lgate_b + i * stride_lgate_i + j * stride_lgate_j + h * stride_lgate_h
    rgate_off = b * stride_rgate_b + i * stride_rgate_i + j * stride_rgate_j + h * stride_rgate_h
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    left_val = tl.load(left_ptr + left_off, mask=mask, other=tl.float16(0))
    right_val = tl.load(right_ptr + right_off, mask=mask, other=tl.float16(0))
    lgate_val = tl.load(left_gate_ptr + lgate_off, mask=mask, other=tl.float16(0))
    rgate_val = tl.load(right_gate_ptr + rgate_off, mask=mask, other=tl.float16(0))
    mask_val = tl.load(mask_ptr + mask_off, mask=mask, other=tl.float16(0))

    out_l = left_val * lgate_val * mask_val
    out_r = right_val * rgate_val * mask_val

    tl.store(out_left_ptr + left_off, out_l, mask=mask)
    tl.store(out_right_ptr + right_off, out_r, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32) or None
            - weights: dict of module parameters
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_fp16 = torch.float16

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # ------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device),
        bias=weights["norm.bias"].to(device),
    )
    x = x.to(dtype_fp16)

    # ------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(device).to(dtype_fp16))
    right = F.linear(x, weights["right_proj.weight"].to(device).to(dtype_fp16))

    # ------------------------------------------------------------
    # 3. Gating (sigmoid)
    # ------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x, weights["left_gate.weight"].to(device).to(dtype_fp16))
    )
    right_gate = torch.sigmoid(
        F.linear(x, weights["right_gate.weight"].to(device).to(dtype_fp16))
    )
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"].to(device).to(dtype_fp16))
    )

    # ------------------------------------------------------------
    # 4. Fuse mask and gates with a tiny Triton kernel
    # ------------------------------------------------------------
    if mask is None:
        mask = torch.ones_like(input_tensor[..., 0])  # shape [B, N, N]
    mask = mask.to(dtype_fp16)

    B, N, _, H = left.shape  # H == hidden_dim
    left_gated = torch.empty_like(left)
    right_gated = torch.empty_like(right)

    total_el = B * N * N * H
    BLOCK_SIZE = 1024  # a good compromise for occupancy
    grid = (triton.cdiv(total_el, BLOCK_SIZE),)

    apply_gate_mask_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask,
        out_left_ptr=left_gated,
        out_right_ptr=right_gated,
        B=B,
        N=N,
        H=H,
        stride_left_b=left.stride(0),
        stride_left_i=left.stride(1),
        stride_left_j=left.stride(2),
        stride_left_h=left.stride(3),
        stride_right_b=right.stride(0),
        stride_right_i=right.stride(1),
        stride_right_j=right.stride(2),
        stride_right_h=right.stride(3),
        stride_lgate_b=left_gate.stride(0),
        stride_lgate_i=left_gate.stride(1),
        stride_lgate_j=left_gate.stride(2),
        stride_lgate_h=left_gate.stride(3),
        stride_rgate_b=right_gate.stride(0),
        stride_rgate_i=right_gate.stride(1),
        stride_rgate_j=right_gate.stride(2),
        stride_rgate_h=right_gate.stride(3),
        stride_mask_b=mask.stride(0),
        stride_mask_i=mask.stride(1),
        stride_mask_j=mask.stride(2),
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # ------------------------------------------------------------
    # 5. Cubic contraction via batched GEMM (cuBLAS)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------
    # left_gated: [B, N, N, H]  (i,k,d)
    # right_gated: [B, N, N, H] (j,k,d) -> need transpose k↔j
    A = left_gated.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    Bmat = right_gated.permute(0, 3, 2, 1).contiguous()      # (B, H, N, N)

    BH = B * H
    A_flat = A.view(BH, N, N)
    B_flat = Bmat.view(BH, N, N)

    out_flat = torch.bmm(A_flat, B_flat)                     # (BH, N, N), fp16
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------
    # 6. Post‑processing (LayerNorm, output gate, final linear)
    # ------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    out_norm = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device),
        bias=weights["to_out_norm.bias"].to(device),
    )
    out_norm = out_norm * out_gate.to(torch.float32)

    output = F.linear(out_norm, weights["to_out.weight"].to(device))
    return output
