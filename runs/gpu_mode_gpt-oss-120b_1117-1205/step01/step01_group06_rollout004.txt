"""
TriMul (outgoing) forward pass with Triton acceleration.

Algorithm:
1. Layer‑norm the input over the last dimension (dim) using PyTorch.
2. Linear projections and gating (left/right projections + left/right/out gates).
3. Optional pairwise mask multiplication (if nomask=False).
4. Compute the cubic contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is implemented as a batched matrix multiplication per hidden dimension
   using torch.matmul on bfloat16 tensors for high throughput.
5. Apply a LayerNorm over the hidden dimension `hidden_dim` using a small
   Triton kernel that processes one (b,i,j) slice at a time.
6. Multiply the normalized tensor by the outgoing gate.
7. Final linear projection back to `dim`.

Only the hidden‑dim LayerNorm is written in Triton; all other heavy work
leverages highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def hidden_layernorm_kernel(
    inp_ptr,   # *float32  [B,N,N,H]
    out_ptr,   # *float32  [B,N,N,H]
    weight_ptr,  # *float32  [H]
    bias_ptr,    # *float32  [H]
    B, N, H,
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    eps: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # decode (b,i,j) from pid
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # vector offsets for the hidden dimension
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # base pointer for this (b,i,j) slice
    inp_base = b * stride_in_b + i * stride_in_i + j * stride_in_j
    out_base = b * stride_out_b + i * stride_out_i + j * stride_out_j

    # ------------------------------------------------------------------
    # Load hidden‑dim slice
    x = tl.load(inp_ptr + inp_base + offs_h * stride_in_h,
                mask=mask_h,
                other=0.0)

    # Compute mean
    sum_x = tl.sum(x, axis=0)
    mean = sum_x / H

    # Compute variance
    diff = x - mean
    sq = diff * diff
    var = tl.sum(sq, axis=0) / H

    # Normalize
    inv_std = tl.rsqrt(var + eps)
    norm = diff * inv_std

    # Apply learned weight and bias
    w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)
    b_ = tl.load(bias_ptr + offs_h, mask=mask_h, other=0.0)
    out = norm * w + b_

    # Store result
    tl.store(out_ptr + out_base + offs_h * stride_out_h,
             out,
             mask=mask_h)


def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward pass.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (optional, bool/int/float)
        - weights      : dict of torch.Tensors with the model parameters
        - config       : dict containing "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over last dimension (dim)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    # Ensure contiguous layout
    x = x.contiguous()

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 2. Linear projections & gating (all in float32)
    x_flat = x.view(-1, dim)  # [B*N*N, dim]

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    left_gate = F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    right_gate = F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    out_gate = F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()

    # ------------------------------------------------------------------
    # 3. Apply mask (if present) and gates
    if not nomask:
        # mask shape [B,N,N] -> [B,N,N,1] broadcast
        mask_exp = mask.to(dtype).unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Pairwise contraction over k using batched matmul (BF16 for speed)
    left_bf = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    # Bring hidden dimension to leading axis: [B, H, N, N]
    left_bf = left_bf.permute(0, 3, 1, 2).contiguous()
    right_bf = right_bf.permute(0, 3, 1, 2).contiguous()

    # Batched matmul: (B*H) x N x N  times (B*H) x N x N^T -> (B*H) x N x N
    out_bf = torch.matmul(left_bf, right_bf.transpose(-2, -1))

    # Back to original layout: [B, N, N, H]
    out = out_bf.permute(0, 2, 3, 1).contiguous().to(dtype)

    # ------------------------------------------------------------------
    # 5. Hidden‑dim LayerNorm via Triton
    out_norm = torch.empty_like(out)
    # Strides in element units
    s_b, s_i, s_j, s_h = out.stride()
    total_programs = B * N * N
    grid = (total_programs,)

    # Choose a compile‑time block size >= hidden_dim
    if hidden_dim <= 64:
        BLOCK_H = 64
    elif hidden_dim <= 128:
        BLOCK_H = 128
    elif hidden_dim <= 256:
        BLOCK_H = 256
    else:
        BLOCK_H = 384  # fits the max hidden_dim in our tests

    hidden_layernorm_kernel[grid](
        out,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        B,
        N,
        hidden_dim,
        s_b,
        s_i,
        s_j,
        s_h,
        s_b,
        s_i,
        s_j,
        s_h,
        eps=1e-5,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 6. Apply outgoing gate
    out_scaled = out_norm * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to `dim`
    out_scaled_flat = out_scaled.view(-1, hidden_dim)  # [B*N*N, hidden_dim]
    final_flat = F.linear(out_scaled_flat, weights["to_out.weight"])
    final = final_flat.view(B, N, N, dim)

    return final
