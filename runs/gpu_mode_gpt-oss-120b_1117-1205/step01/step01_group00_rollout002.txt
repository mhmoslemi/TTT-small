"""TriMul (outgoing) forward pass.

The implementation follows the reference PyTorch module but offloads the
heavy O(N³) contraction to a fused Triton kernel.  The steps are:

1. LayerNorm over the channel dimension (float32).
2. Linear projections (left/right) and three gating linear layers (half‑precision).
3. Optional mask broadcasted over the hidden dimension.
4. Apply the gates to the projected tensors.
5. Compute out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   via a batched matrix multiplication using a Triton kernel
   (treated as (B·D, N, N) @ (B·D, N, N)ᵀ).
6. Output LayerNorm, output gate, and final linear projection back to the
   original channel dimension.

All heavy computation (step 5) is performed in half‑precision with
float‑accumulation, then converted back to float32 for the remaining steps.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ B) where A: (batch, M, K), B: (batch, K, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute C = A @ B for a batch of matrices.

    A: (batch, M, K), B: (batch, K, N), C: (batch, M, N).
    Accumulation is performed in float32, output stored as float16.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)

        # Load B tile (K x N)
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def triton_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B where
    A: (batch, M, K), B: (batch, K, N), all torch.float16 and contiguous.
    Returns C of shape (batch, M, N) as torch.float16.
    """
    assert A.dtype == B.dtype == torch.float16
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tile sizes – tuned for typical A200/B200 memory hierarchy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or numeric)
            - weights_dict: mapping name -> torch.Tensor
            - config_dict: must contain 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (half‑precision, no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16), bias=None)
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16), bias=None)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)  # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 4. Gates (half‑precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16), bias=None).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16), bias=None).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16), bias=None).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton (batched GEMM)
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    # Permute to bring hidden dimension into the batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, k, j)

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Batched matrix multiplication: (B*H, N, N) @ (B*H, N, N)ᵀ
    out_flat = triton_matmul(left_flat, right_flat)   # (B*H, N, N), fp16

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, output gate, and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32), bias=None)

    return out
