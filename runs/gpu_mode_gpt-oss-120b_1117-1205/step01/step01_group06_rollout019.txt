def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)

    # layernorm on last dim.
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)

    # flatten for linear
    B,N,_,_ = x.shape
    x_flat = x.view(-1, dim)

    # Proj
    left = F.linear(x_flat, weights["left_proj.weight"]).view(B,N,N,hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B,N,N,hidden_dim)

    # Gates
    left_gate = F.linear(x_flat, weights["left_gate.weight"]).view(B,N,N,hidden_dim).sigmoid()
    right_gate = F.linear(x_flat, weights["right_gate.weight"]).view(B,N,N,hidden_dim).sigmoid()
    out_gate = F.linear(x_flat, weights["out_gate.weight"]).view(B,N,N,hidden_dim).sigmoid()

    # mask
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(x.dtype) # shape [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # Contraction via Triton
    out = torch.empty_like(left) # same shape dtype

    # compute strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_I), triton.cdiv(N, BLOCK_J), B*hidden_dim)

    trmul_outgoing_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K
    )
    # Layernorm hidden dim
    out_norm = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=1e-5)
    # Apply out_gate
    out_scaled = out_norm * out_gate
    # final linear
    out_scaled_flat = out_scaled.view(-1, hidden_dim)
    final_flat = F.linear(out_scaled_flat, weights["to_out.weight"])
    final = final_flat.view(B, N, N, dim)
    return final

@triton.jit
def trmul_outgoing_kernel(left_ptr, right_ptr, out_ptr,
                         B, N, H,
                         stride_left_b, stride_left_i, stride_left_k, stride_left_h,
                         stride_right_b, stride_right_j, stride_right_k, stride_right_h,
                         stride_out_b, stride_out_i, stride_out_j, stride_out_h,
                         BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)
    # total bh = B*H
    b = pid_bh // H
    h = pid_bh % H

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointer arithmetic for left
        left_ptrs = left_ptr + b * stride_left_b + h * stride_left_h + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # pointer arithmetic for right
        right_ptrs = right_ptr + b * stride_right_b + h * stride_right_h + offs_j[:, None] * stride_right_j + offs_k[None, :] * stride_right_k
        right_tile = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # accumulate
        acc += tl.dot(left_tile, right_tile, trans_b=True)

    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul outgoing kernel (forward only).

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension `dim`.
2. Linear projections → left/right (hidden_dim) and three gate vectors
   (left_gate, right_gate, out_gate) using the supplied weight matrices.
3. Apply optional pairwise mask and element‑wise gates:
       left  = left  * mask * left_gate
       right = right * mask * right_gate
4. Compute the pairwise contraction over the sequence dimension k:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This O(N³) operation is implemented as a custom Triton kernel that
   performs a tiled matrix‑multiply for each (batch, hidden) slice.
5. Apply a LayerNorm over the hidden dimension (`to_out_norm`),
   multiply by `out_gate`, and finally linearly project back to `dim`.
   The remaining operations are performed with PyTorch (highly‑optimized
   kernels).  The Triton kernel is the performance‑critical part.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,                # *float32 pointers
    B, N, H,                                     # int32
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    Grid:
        (i_block, j_block, bh) where bh = b*H + h
    """
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # decode batch and hidden index
    b = pid_bh // H
    h = pid_bh % H

    # start offsets for this tile
    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    # range of i / j indices handled by this program
    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator for the (BLOCK_I, BLOCK_J) sub‑matrix
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # iterate over k in tiles of size BLOCK_K
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left[b,i,k,h]  (shape BLOCK_I × BLOCK_K)
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + h * stride_left_h
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        left_tile = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # load right[b,j,k,h] (shape BLOCK_J × BLOCK_K)
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + h * stride_right_h
            + offs_j[:, None] * stride_right_j
            + offs_k[None, :] * stride_right_k
        )
        right_tile = tl.load(
            right_ptrs,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        )

        # accumulate product over k (dot with transposed right)
        acc += tl.dot(left_tile, right_tile, trans_b=True)

    # store the result tile
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + h * stride_out_h
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (may be None/unused)
            - weights: dict of tensors for all linear / layer‑norm parameters
            - config: dict with keys "dim", "hidden_dim", and "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    B, N, _, _ = x.shape

    # --------------------------------------------------------------
    # 2. Linear projections (no bias) and gates
    x_flat = x.view(-1, dim)  # [B*N*N, dim]

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    left_gate = (
        F.linear(x_flat, weights["left_gate.weight"])
        .view(B, N, N, hidden_dim)
        .sigmoid()
    )
    right_gate = (
        F.linear(x_flat, weights["right_gate.weight"])
        .view(B, N, N, hidden_dim)
        .sigmoid()
    )
    out_gate = (
        F.linear(x_flat, weights["out_gate.weight"])
        .view(B, N, N, hidden_dim)
        .sigmoid()
    )

    # --------------------------------------------------------------
    # 3. Apply mask (if present) and element‑wise gates
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(dtype)  # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous layout for the Triton kernel
    left = left.contiguous()
    right = right.contiguous()

    # --------------------------------------------------------------
    # 4. Pairwise contraction via Triton (the O(N³) core)
    out = torch.empty_like(left)  # shape [B, N, N, hidden_dim]

    # strides (in elements) for pointer arithmetic
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_rj, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # block sizes (tuned for typical N up to 1024)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_I),   # i‑blocks
        triton.cdiv(N, BLOCK_J),   # j‑blocks
        B * hidden_dim,            # combined (batch, hidden) programs
    )

    trmul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        s_lb,
        s_li,
        s_lk,
        s_lh,
        s_rb,
        s_rj,
        s_rk,
        s_rh,
        s_ob,
        s_oi,
        s_oj,
        s_oh,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 5. Hidden‑dim LayerNorm (PyTorch implementation – fast enough)
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # 6. Apply outgoing gate
    out_scaled = out_norm * out_gate

    # 7. Final linear projection back to `dim`
    out_scaled_flat = out_scaled.view(-1, hidden_dim)
    final_flat = F.linear(out_scaled_flat, weights["to_out.weight"])
    final = final_flat.view(B, N, N, dim)

    return final
