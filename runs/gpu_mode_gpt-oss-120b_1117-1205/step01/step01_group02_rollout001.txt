"""
TriMul (outgoing) forward pass with a fused Triton batched GEMM.

Algorithm
---------
1. Layer‑norm the input over the last channel dimension.
2. Compute left/right projections and their sigmoid gates (all in fp16).
3. Apply optional mask and element‑wise gates.
4. For each batch × head we need
        out[i, j] = Σ_k left[i, k] * right[j, k] .
   This is a batched matrix multiplication left @ rightᵀ.
   The heavy O(N³) work is performed by a custom Triton kernel that
   computes many independent GEMMs (total_batch = B·H) in one launch.
5. Reshape the result back, apply the output LayerNorm, the output gate
   and the final linear projection.  The final tensor is returned in float32.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# ----------------------------------------------------------------------------- #
# Triton kernel: batched GEMM A @ Bᵀ  (A: M×K, B: N×K → Bᵀ: K×N)
#   a_ptr : [total_batch, M, K]   (fp16)
#   b_ptr : [total_batch, N, K]   (fp16) – already transposed in host code
#   c_ptr : [total_batch, M, N]   (fp16)
# -----------------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # tensor pointers
    M, N, K,                      # matrix dimensions (int)
    stride_am, stride_ak,         # A strides (row, col)
    stride_bk, stride_bn,         # B strides (row, col) – B is already transposed
    stride_cm, stride_cn,         # C strides (row, col)
    stride_a_batch, stride_b_batch, stride_c_batch,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    pid_b = tl.program_id(2)  # batch index (B * H)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32 for better precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_offset = k_tile * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K)
        mask_k = (k_offset + offs_k) < K

        # Pointers for the current tile (including batch offset)
        a_ptrs = (
            a_ptr
            + pid_b * stride_a_batch
            + offs_m[:, None] * stride_am
            + (k_offset + offs_k)[None, :] * stride_ak
        )
        b_ptrs = (
            b_ptr
            + pid_b * stride_b_batch
            + (k_offset + offs_k)[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # GEMM accumulation
        acc += tl.dot(a, b)

    # Store result (back to fp16)
    c_ptrs = (
        c_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)


# ----------------------------------------------------------------------------- #
# Entry point expected by the evaluation harness
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32, CUDA)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of torch.Tensor with all linear / LN parameters
            - config: dict with at least {"dim": int, "hidden_dim": int}

    Returns:
        torch.Tensor [B, N, N, dim] (float32, on same device)
    """
    # --------------------------------------------------------------------- #
    # Unpack inputs and configuration
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------------- #
    # 1. LayerNorm over the channel dimension
    # --------------------------------------------------------------------- #
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim], float32

    # Cast to half for the heavy part
    x_norm_fp16 = x_norm.to(torch.float16)

    # --------------------------------------------------------------------- #
    # 2. Linear projections (no bias) and gating (all fp16)
    # --------------------------------------------------------------------- #
    left = F.linear(x_norm_fp16, weights["left_proj.weight"].to(torch.float16))   # [B,N,N,hidden]
    right = F.linear(x_norm_fp16, weights["right_proj.weight"].to(torch.float16))

    left_gate = F.linear(x_norm_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_norm_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_norm_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # --------------------------------------------------------------------- #
    # 3. Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------------- #
    if mask is not None:
        # mask shape [B,N,N] -> [B,N,N,1] for broadcasting
        mask_f = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------------- #
    # 4. Apply element‑wise gates
    # --------------------------------------------------------------------- #
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 5. Prepare data for batched GEMM: left @ rightᵀ
    # --------------------------------------------------------------------- #
    # Transform to [B, H, N, N] and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Transpose the last two axes of right so that we can do A @ B (no extra transpose)
    right_T = right_perm.transpose(-2, -1).contiguous()  # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    total_batch = B * H

    # Collapse batch & head dimensions for the kernel
    left_flat = left_perm.view(total_batch, N, N)   # [total_batch, N, N]
    right_flat = right_T.view(total_batch, N, N)   # [total_batch, N, N]
    out_flat = torch.empty((total_batch, N, N), dtype=torch.float16, device=device)

    # Strides expressed in number of elements
    stride_a_batch, stride_am, stride_ak = left_flat.stride()
    stride_b_batch, stride_bk, stride_bn = right_flat.stride()
    stride_c_batch, stride_cm, stride_cn = out_flat.stride()

    # --------------------------------------------------------------------- #
    # 6. Launch the Triton batched GEMM kernel
    # --------------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows
        triton.cdiv(N, BLOCK_N),          # cols
        total_batch,                      # batch
    )
    trimul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------------- #
    # 7. Reshape back to [B, N, N, H] and continue in FP32
    # --------------------------------------------------------------------- #
    out = (
        out_flat.view(B, H, N, N)          # [B, H, N, N]
        .permute(0, 2, 3, 1)               # [B, N, N, H]
        .contiguous()
        .to(torch.float32)
    )

    # --------------------------------------------------------------------- #
    # 8. Output LayerNorm, output gate and final linear projection
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # Apply the output gate (convert gate to fp32 for the multiplication)
    out = out * out_gate.to(torch.float32)

    # Final projection back to `dim`
    output = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return output
