"""
TriMul (outgoing) forward kernel.

The heavy operation is a batched matrix multiplication:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
for each batch b and hidden channel d.
We fuse this contraction into a Triton kernel that computes
(B * hidden_dim) independent GEMMs of size (N, N) in fp16,
accumulating in fp32 for numerical stability.
All other operations (LayerNorm, linear projections, gates, etc.)
are performed in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_az, stride_am, stride_ak,   # A: (batch, M, K)
    stride_bz, stride_bk, stride_bn,   # B: (batch, K, N)
    stride_cz, stride_cm, stride_cn,   # C: (batch, M, N)
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ B where A (M×K) and B (K×N).
    Each batch corresponds to one (hidden_dim) slice of the TriMul.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)  # block row
    pid_n = tl.program_id(2)  # block col

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the output tile
    c_ptrs = (
        C_ptr
        + batch_id * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a_ptrs = (
            A_ptr
            + batch_id * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # Load B tile (K x N)
        b_ptrs = (
            B_ptr
            + batch_id * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        # Accumulate
        acc += tl.dot(a, b)

    # Write back result in FP16 (matches downstream dtype)
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict with the model parameters
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    gamma = weights["norm.weight"].to(torch.float32)
    beta = weights["norm.bias"].to(torch.float32)

    x = input_tensor.to(torch.float32)
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * gamma + beta  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Projections and gates (use fp16 for speed)
    # ------------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    # Linear weights (fp16)
    w_left = weights["left_proj.weight"].to(torch.float16)       # (hidden_dim, dim)
    w_right = weights["right_proj.weight"].to(torch.float16)

    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    # ----- Projections
    left = F.linear(x_h, w_left)   # [B, N, N, hidden_dim]
    right = F.linear(x_h, w_right)

    # ----- Optional mask (applied before gating)
    if mask is not None and not config.get("nomask", False):
        mask_exp = mask.unsqueeze(-1).to(x_h.dtype)  # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ----- Gates (sigmoid)
    left_gate = F.linear(x_h, w_left_gate).sigmoid()
    right_gate = F.linear(x_h, w_right_gate).sigmoid()
    out_gate = F.linear(x_h, w_out_gate).sigmoid()   # shape [B, N, N, hidden_dim]

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton batched GEMM: out = left @ right^T across k dimension
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # reshape to (B*H, N, N) where each batch corresponds to one hidden channel
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (batch, M, K)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # (batch, M, K)

    # We need right.T over the last two dimensions (K,N)
    right_mat_t = right_mat.permute(0, 2, 1).contiguous()                # (batch, K, N)

    out_mat = torch.empty((B * H, N, N), dtype=torch.float16, device=device)

    # Strides (row‑major layout)
    stride_az, stride_am, stride_ak = left_mat.stride()
    stride_bz, stride_bk, stride_bn = right_mat_t.stride()
    stride_cz, stride_cm, stride_cn = out_mat.stride()

    # Tiling parameters – adapt to sequence length
    if N > 512:
        BLOCK_M = BLOCK_N = 128
    else:
        BLOCK_M = BLOCK_N = 64
    BLOCK_K = 64
    NUM_WARPS = 8 if (BLOCK_M * BLOCK_N) >= 8192 else 4

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_gemm_kernel[grid](
        left_mat,
        right_mat_t,
        out_mat,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=NUM_WARPS,
        num_stages=3,
    )

    # Reshape back to [B, N, N, hidden_dim] and promote to fp32
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp16
    out_f32 = out.to(torch.float32)

    # ------------------------------------------------------------------
    # Output LayerNorm (over hidden_dim) and gating
    # ------------------------------------------------------------------
    gamma_out = weights["to_out_norm.weight"].to(torch.float32)
    beta_out = weights["to_out_norm.bias"].to(torch.float32)

    mean_o = out_f32.mean(dim=-1, keepdim=True)
    var_o = out_f32.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out_f32 - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * gamma_out + beta_out

    # Apply out gate (converted to fp32)
    out_norm = out_norm * out_gate.to(torch.float32)

    # ------------------------------------------------------------------
    # Final linear projection back to original dimension
    # ------------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float32)  # (dim, hidden_dim)
    output = F.linear(out_norm, w_to_out)  # [B, N, N, dim]

    return output
