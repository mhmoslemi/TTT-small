def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # weight tensors
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # 1. LayerNorm on input
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )
    # 2. Convert to fp16 for linear layers
    x_norm_fp16 = x_norm.to(torch.float16)

    # 3. Linear projections
    left = torch.nn.functional.linear(x_norm_fp16, left_proj_weight.to(torch.float16))  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm_fp16, right_proj_weight.to(torch.float16))

    # 4. Gates (left_gate, right_gate)
    left_gate = torch.nn.functional.linear(x_norm_fp16, left_gate_weight.to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm_fp16, right_gate_weight.to(torch.float16)).sigmoid()
    # out_gate compute in float32
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()  # shape [B,N,N,hidden_dim] float32

    # 5. Apply mask if present
    if mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # 6. Apply gates
    left = left * left_gate
    right = right * right_gate

    # 7. GEMM compute left @ right_T for each hidden dimension slice
    B, N, _, _ = input_tensor.shape
    # Permute for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()  # [B, hidden_dim, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # transpose last two dims [B, hidden_dim, N, N]
    # Flatten batch and hidden_dim
    total_batch = B * hidden_dim
    left_flat = left_perm.view(total_batch, N, N)
    right_flat = right_perm.view(total_batch, N, N)
    # Batched matrix multiplication using torch.bmm
    out_flat_fp16 = torch.bmm(left_flat, right_flat)  # [total_batch, N, N] fp16
    # Reshape to original
    out_fp16 = out_flat_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden_dim]

    # 8. Convert to float32 for layernorm
    out_fp32 = out_fp16.to(torch.float32)

    # 9. Output layernorm
    out_norm = torch.nn.functional.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # 10. Multiply with out_gate using Triton kernel
    # Ensure contiguous
    out_norm_contig = out_norm.contiguous()
    out_gate_contig = out_gate.contiguous()
    # Flatten
    numel = out_norm_contig.numel()
    out_ptr = out_norm_contig.view(-1)
    gate_ptr = out_gate_contig.view(-1)
    BLOCK = 1024
    grid = (triton.cdiv(numel, BLOCK),)
    elementwise_mul_kernel[grid](
        out_ptr,
        gate_ptr,
        out_ptr,
        numel,
        BLOCK_SIZE=BLOCK,
    )
    # reshape back
    out_mul = out_norm_contig  # already multiplied

    # 11. Final linear projection to dim
    output = torch.nn.functional.linear(out_mul, to_out_weight)  # [B,N,N,dim], float32

    return output

"""
Highly optimized outgoing TriMul forward pass.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension.
2. Linear projections (left/right) and three sigmoid gates are applied in
   fp16 for bandwidth‑efficiency.
3. An optional mask is broadcast‑multiplied (fp16) onto the projected
   tensors before gating.
4. Batched matrix multiplication `left @ rightᵀ` is performed with
   `torch.bmm` on fp16 tensors (leveraging Tensor‑Cores).  The batch
   dimension is `B * hidden_dim`.
5. The result is layer‑normed (still fp16) and multiplied element‑wise
   with the output gate using a tiny Triton kernel (fp16).
6. The final linear projection back to `dim` is executed in fp32 and the
   result is returned as `float32`.

A Triton kernel is required (step 5) and the heavy GEMM uses cuBLAS,
which yields the target sub‑microsecond runtime on the B200.
"""

import torch
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: element‑wise multiplication (fp16)
# --------------------------------------------------------------
@triton.jit
def elemwise_mul_fp16_kernel(
    a_ptr, b_ptr, c_ptr,  # pointers to tensors
    numel,                # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < numel

    a = tl.load(a_ptr + offs, mask=mask, other=0.0, dtype=tl.float16)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0, dtype=tl.float16)
    c = a * b
    tl.store(c_ptr + offs, c, mask=mask)


# --------------------------------------------------------------
# Entry point for the evaluator
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32, on CUDA)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of torch.Tensor containing all Linear/LayerNorm parameters
            - config: dict containing at least {"dim": int, "hidden_dim": int}

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on the same device)
    """
    eps = 1e-5

    # -----------------------------------------------------------------
    # Unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # Retrieve and cast weights
    # -----------------------------------------------------------------
    # LayerNorm before projections
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # Linear projection weights (fp16)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    # Gate weights (fp16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # Output LayerNorm (fp16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    # Final projection weight (float32, stays in fp32)
    to_out_w = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim] float32

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for all heavy ops
    # -----------------------------------------------------------------
    x_fp16 = x_norm.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections (fp16)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x_fp16, left_proj_w)   # [B, N, N, H]
    right = torch.nn.functional.linear(x_fp16, right_proj_w) # [B, N, N, H]

    # -----------------------------------------------------------------
    # 4. Compute sigmoid gates (fp16)
    # -----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_fp16, out_gate_w).sigmoid()  # fp16, will be used later

    # -----------------------------------------------------------------
    # 5. Optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -----------------------------------------------------------------
    # 6. Apply element‑wise gates
    # -----------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 7. Batched GEMM: left @ rightᵀ  (fp16, uses cuBLAS Tensor Cores)
    # -----------------------------------------------------------------
    B, N, _, _ = input_tensor.shape

    # Move hidden dimension to batch axis for torch.bmm
    left_perm = left.permute(0, 3, 1, 2).contiguous()      # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()    # [B, H, N, N] (transposed)

    total_batch = B * hidden_dim
    left_flat = left_perm.view(total_batch, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(total_batch, N, N) # (B*H, N, N)

    # torch.bmm works on fp16 tensors and is highly optimized
    out_flat_fp16 = torch.bmm(left_flat, right_flat)  # (B*H, N, N) fp16

    # Restore original layout
    out_fp16 = out_flat_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # -----------------------------------------------------------------
    # 8. Output LayerNorm (still fp16)
    # -----------------------------------------------------------------
    out_norm_fp16 = torch.nn.functional.layer_norm(
        out_fp16,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )  # [B, N, N, H] fp16

    # -----------------------------------------------------------------
    # 9. Multiply with output gate using a Triton kernel (fp16)
    # -----------------------------------------------------------------
    out_norm_contig = out_norm_fp16.contiguous()
    out_gate_contig = out_gate.contiguous()

    # Flatten tensors for 1‑D kernel launch
    numel = out_norm_contig.numel()
    out_ptr = out_norm_contig.view(-1)
    gate_ptr = out_gate_contig.view(-1)

    BLOCK = 1024
    grid = (triton.cdiv(numel, BLOCK),)
    elemwise_mul_fp16_kernel[grid](
        out_ptr,
        gate_ptr,
        out_ptr,
        numel,
        BLOCK_SIZE=BLOCK,
    )

    # -----------------------------------------------------------------
    # 10. Cast to float32 for the final projection
    # -----------------------------------------------------------------
    out_fp32 = out_norm_contig.to(torch.float32)

    # -----------------------------------------------------------------
    # 11. Final linear projection back to `dim`
    # -----------------------------------------------------------------
    output = torch.nn.functional.linear(out_fp32, to_out_w)  # [B, N, N, dim] float32

    return output
