    right_gate = F.linear(x_f16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_f16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    right = right * right_gate

    right = right.permute(0, 3, 2, 1).contiguous()   # (B, hidden, N, N) with k,j

    left_mat = left.reshape(total_batches, N, N)   # (B*hidden,N,N)
    right_mat = right.reshape(total_batches, N, N)

    out_mat = torch.empty_like(left_mat) # fp16

    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(1), out_mat.stride(2),
        left_mat.stride(0), right_mat.stride(0), out_mat.stride(0),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back
    out = out_mat.reshape(input_tensor.shape[0], hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32) # (B,N,N,hidden)

    # out norm
    out = F.layer_norm(out, normalized_shape=(hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    # apply out gate
    out = out * out_gate.to(torch.float32)

    # final projection
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out

"""
TriMul "outgoing" forward pass with a fused Triton GEMM kernel.

The operator works on an input tensor X of shape [B, N, N, C] (C = dim).
1. LayerNorm over the channel dimension.
2. Linear projections to a hidden dimension H (left_proj, right_proj) and
   three sigmoid gates (left_gate, right_gate, out_gate). All of these are
   performed in float16 for bandwidth reduction.
3. Core update:
       out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, k, j]
   which is a batched GEMM (M=N, K=N, N=N) for each (b, h) pair.
   The GEMM is implemented in Triton with tiling (BLOCK_M, BLOCK_N, BLOCK_K)
   and fp16 inputs, fp32 accumulation.
4. LayerNorm on the hidden dimension, application of out_gate and a final
   linear projection back to dim.

All tensors are assumed to reside on the same CUDA device.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers to A, B, C
    M, N, K,                                 # matrix dimensions
    stride_am, stride_ak,                    # strides for A (M, K)
    stride_bk, stride_bn,                    # strides for B (K, N)
    stride_cm, stride_cn,                    # strides for C (M, N)
    stride_batch_a, stride_batch_b, stride_batch_c,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes C = A @ B where A: (M,K), B: (K,N) for each batch.
    Batch dimension is encoded in program_id(0).
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Pointers to the start of this batch
    a_batch = a_ptr + pid_batch * stride_batch_a
    b_batch = b_ptr + pid_batch * stride_batch_b
    c_batch = c_ptr + pid_batch * stride_batch_c

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Pointers to tiles of A and B
    a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        # fp16 inputs, fp32 accumulation
        acc += tl.dot(a, b, out_dtype=tl.float32)
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store result as fp16
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul "outgoing" update.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask_tensor:  torch.Tensor of shape [B, N, N] or None
            - weights_dict: mapping from string names to torch.Tensor weights
            - config_dict: must contain keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (dtype float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2) Projections & gates (float16)
    # -----------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # Projections
    left = F.linear(x_f16, weights["left_proj.weight"].to(torch.float16))   # [B,N,N,hidden]
    right = F.linear(x_f16, weights["right_proj.weight"].to(torch.float16))

    # Optional mask (broadcasted on hidden dim)
    if mask is not None:
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f16
        right = right * mask_f16

    # Gates (sigmoid, float16)
    left_gate = F.linear(x_f16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_f16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_f16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 3) Core TriMul via Triton batched GEMM
    #    out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, k, j]
    # -----------------------------------------------------------------
    # Permute to (B, hidden, N, N) and make right transpose over (k,j)
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, i, k)
    right = right.permute(0, 3, 2, 1).contiguous()  # (B, hidden, k, j)

    total_batches = B * hidden
    left_mat = left.reshape(total_batches, N, N)   # (B*hidden, M=N, K=N)
    right_mat = right.reshape(total_batches, N, N)  # (B*hidden, K=N, N=N)

    out_mat = torch.empty_like(left_mat)  # fp16 placeholder for result

    # Heuristic block sizes
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        total_batches,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        N,
        N,
        left_mat.stride(1),
        left_mat.stride(2),
        right_mat.stride(1),
        right_mat.stride(2),
        out_mat.stride(1),
        out_mat.stride(2),
        left_mat.stride(0),
        right_mat.stride(0),
        out_mat.stride(0),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden] (float32)
    out = out_mat.reshape(B, hidden, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # -----------------------------------------------------------------
    # 4) Output LayerNorm, gate, final projection (float32)
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
