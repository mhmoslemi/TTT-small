custom_kernel(data):
    input_tensor, mask, weights, config = data
    B,N,_,dim = input_tensor.shape
    hidden_dim = config['hidden_dim']

    # LayerNorm of input
    x = layer_norm_fp32(input_tensor, norm_weight, norm_bias) -> maybe use torch.nn.functional.layer_norm.

    # Projections left,right and gates: left_proj_weight shape (hidden_dim, dim) because linear weight in PyTorch shape (out_features, in_features). Using F.linear: linear(x, weight) does x @ weight.T + bias. Since bias is None, weight shape [out_features, in_features]. So left_proj_weight = weights["left_proj.weight"] of shape (hidden_dim, dim). Right similarly. left_gate_weight shape (hidden_dim, dim). out_gate_weight shape (hidden_dim, dim). Note out_gate weight is same hidden_dim.

    # Compute left = linear(x, left_proj_weight)
    left = F.linear(x, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    if not nomask:
        mask_unsq = mask.to(left.dtype).unsqueeze(-1)   # shape [B,N,N,1]
        left = left * mask_unsq
        right = right * mask_unsq

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # cast to half for matmul
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # Reshape to [B*hidden_dim, N, N] after permuting dims to bring hidden to batch: left_f16.permute(0,3,1,2).contiguous().view(-1,N,N)
    left_f16 = left_f16.permute(0,3,1,2).contiguous()
    right_f16 = right_f16.permute(0,3,1,2).contiguous()
    batch_h = left_f16.shape[0] # B*hidden_dim

    left_f16 = left_f16.view(batch_h,N,N)
    right_f16 = right_f16.view(batch_h,N,N)

    # Compute out = torch.bmm(left_f16, right_f16.transpose(1,2))   shape [batch_h,N,N]
    out_f16 = torch.bmm(left_f16, right_f16.transpose(1,2))  # returns float16

    # Reshape back to original shape: out = out_f16.view(B, hidden_dim, N, N).permute(0,2,3,1) -> shape [B,N,N,hidden_dim]
    out = out_f16.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()
    out = out.to(torch.float32)

    # Output norm
    out = F.layer_norm(out, normalized_shape=(hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=1e-5)
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out

def fuse_left_kernel(
    in_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    MASK_PRESENT: tl.constexpr,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # Compute 4D indices from flat offset
    # using integer division and mod.
    # Compute hidden index h = offs % H
    idx = offs
    h = idx % H
    idx = idx // H
    j = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    # offsets for each tensor
    in_offset = b*stride_in_b + i*stride_in_i + j*stride_in_j + h*stride_in_h
    gate_offset = b*stride_gate_b + i*stride_gate_i + j*stride_gate_j + h*stride_gate_h
    out_offset = b*stride_out_b + i*stride_out_i + j*stride_out_j + h*stride_out_h

    # load values
    a = tl.load(in_ptr + in_offset, mask=mask, other=0.0)
    g = tl.load(gate_ptr + gate_offset, mask=mask, other=0.0)

    res = a * g
    if MASK_PRESENT:
        # mask is scalar without hidden dim; load mask for each (b,i,j)
        mask_scalar = tl.load(mask_ptr + b*stride_mask_b + i*stride_mask_i + j*stride_mask_j, mask=mask, other=0.0)

        res = res * mask_scalar

TriMul forward with Triton-based mask+gate fusion and CuBLAS batched GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def apply_gate_mask_kernel(
    inp_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    MASK_PRESENT: tl.constexpr,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # Decode flat offset into (b,i,j,h)
    h = offs % H
    idx = offs // H
    j = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    # Compute linear offsets for each tensor
    inp_off = b * stride_in_b + i * stride_in_i + j * stride_in_j + h * stride_in_h
    gate_off = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j + h * stride_gate_h
    out_off = b * stride_out_b + i * stride_out_i + j * stride_out_j + h * stride_out_h

    a = tl.load(inp_ptr + inp_off, mask=mask, other=0.0)
    g = tl.load(gate_ptr + gate_off, mask=mask, other=0.0)
    res = a * g

    if MASK_PRESENT:
        mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
        m = tl.load(mask_ptr + mask_off, mask=mask, other=0.0)
        res = res * m

    tl.store(out_ptr + out_off, res.to(tl.float16), mask=mask)

    Triton‑accelerated forward of the “outgoing” TriMul module.

    Parameters

        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor of shape [B, N, N, dim]
        - mask        : torch.Tensor of shape [B, N, N] (bool or float)
        - weights     : dict of torch.Tensors with module parameters
        - config      : dict containing at least "dim", "hidden_dim", optional "nomask"

    Returns

    torch.Tensor of shape [B, N, N, dim]
    """
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # 1. Input LayerNorm

        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,

    # 2. Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    # 3. Gating projections (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B, N, N, H]

    # 4. Fuse mask (if any) and gating into half‑precision tensors via Triton
    left_out = torch.empty_like(left, dtype=torch.float16, device=device)
    right_out = torch.empty_like(right, dtype=torch.float16, device=device)

    if not nomask:
        mask_f = mask.to(torch.float32)
    else:

        mask_f = torch.empty((1,), dtype=torch.float32, device=device)

    s_in_b, s_in_i, s_in_j, s_in_h = left.stride()
    s_gate_b, s_gate_i, s_gate_j, s_gate_h = left_gate.stride()
    s_out_b, s_out_i, s_out_j, s_out_h = left_out.stride()
    s_mask_b = s_mask_i = s_mask_j = 0

        s_mask_b, s_mask_i, s_mask_j,
        s_out_b, s_out_i, s_out_j, s_out_h,
        MASK_PRESENT=MASK_PRESENT,
        BLOCK=BLOCK,

        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    out = F.linear(out, weights["to_out.weight"])

    i = idx % N
    b = idx // N

    # Linear offsets for each tensor
    inp_off = b * stride_in_b + i * stride_in_i + j * stride_in_j + h * stride_in_h
    gate_off = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j + h * stride_gate_h
    out_off = b * stride_out_b + i * stride_out_i + j * stride_out_j + h * stride_out_h

    a = tl.load(inp_ptr + inp_off, mask=msk, other=0.0)       # fp32
    g = tl.load(gate_ptr + gate_off, mask=msk, other=0.0)    # fp32
    res = a * g

    if MASK_PRESENT:
        mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
        m = tl.load(mask_ptr + mask_off, mask=msk, other=0.0)   # fp32 scalar
        res = res * m

    tl.store(out_ptr + out_off, res.to(tl.float16), mask=msk)

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N] (bool or float)
        - weights      : dict of torch.Tensors containing all linear and LN params
        - config       : dict with at least ``dim`` and ``hidden_dim``
                         optional ``nomask`` (bool, default True)

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2. Projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Gating (sigmoid of linear projections)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # 4. Fuse mask & gates into fp16 tensors via Triton
    # -----------------------------------------------------------------
    left_out = torch.empty_like(left, dtype=torch.float16, device=device)
    right_out = torch.empty_like(right, dtype=torch.float16, device=device)

    if not nomask:
        mask_f = mask.to(torch.float32)
    else:
        # Dummy tensor – never read when MASK_PRESENT == 0
        mask_f = torch.empty((1,), dtype=torch.float32, device=device)

    # Strides (elements, not bytes)
    s_in_b, s_in_i, s_in_j, s_in_h = left.stride()
    s_gate_b, s_gate_i, s_gate_j, s_gate_h = left_gate.stride()
    s_out_b, s_out_i, s_out_j, s_out_h = left_out.stride()
    if not nomask:
        s_mask_b, s_mask_i, s_mask_j = mask_f.stride()
    else:
        s_mask_b = s_mask_i = s_mask_j = 0

    total_elements = B * N * N * hidden_dim
    BLOCK = 1024  # max 1024 threads per CUDA block
    grid = (triton.cdiv(total_elements, BLOCK),)
    MASK_PRESENT = int(not nomask)

    # left side
    apply_gate_mask_kernel[grid](
        left, left_gate, mask_f, left_out,
        B, N, hidden_dim,
        s_in_b, s_in_i, s_in_j, s_in_h,
        s_gate_b, s_gate_i, s_gate_j, s_gate_h,
        s_mask_b, s_mask_i, s_mask_j,
        s_out_b, s_out_i, s_out_j, s_out_h,
        MASK_PRESENT=MASK_PRESENT,
        BLOCK=BLOCK,
    )
    # right side (strides are identical layout)
    s_in_b, s_in_i, s_in_j, s_in_h = right.stride()
    s_gate_b, s_gate_i, s_gate_j, s_gate_h = right_gate.stride()
    s_out_b, s_out_i, s_out_j, s_out_h = right_out.stride()

    apply_gate_mask_kernel[grid](
        right, right_gate, mask_f, right_out,
        B, N, hidden_dim,
        s_in_b, s_in_i, s_in_j, s_in_h,
        s_gate_b, s_gate_i, s_gate_j, s_gate_h,
        s_mask_b, s_mask_i, s_mask_j,
        s_out_b, s_out_i, s_out_j, s_out_h,
        MASK_PRESENT=MASK_PRESENT,
        BLOCK=BLOCK,
    )

    # Release intermediate fp32 tensors early
    del left, right, left_gate, right_gate, x

    # -----------------------------------------------------------------
    # 5. Batched GEMM: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a strided batched GEMM (cuBLAS)
    # -----------------------------------------------------------------
    batch_h = B * hidden_dim
    # reshape to (batch_h, N, N) with hidden dim as batch dimension
    left_mat = left_out.permute(0, 3, 1, 2).contiguous().view(batch_h, N, N)
    right_mat = right_out.permute(0, 3, 1, 2).contiguous().view(batch_h, N, N)

    # C = A @ Bᵀ  (Bᵀ because we need sum_k left[i,k] * right[j,k])
    out_fp16 = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original layout and convert to fp32
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # -----------------------------------------------------------------
    # 6. Output norm, gating and final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
