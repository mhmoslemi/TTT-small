"""
TriMul (outgoing) forward pass.

Algorithm
---------
1. Layer‑norm the input.
2. Linear projections (left, right) and sigmoid gates (left_gate, right_gate, out_gate)
   are computed in fp16.
3. A small Triton kernel fuses element‑wise multiplication of the projection,
   its gate and the optional mask (three‑operand multiply).  When no mask is
   provided a two‑operand kernel is used.
4. The heavy N³ computation is performed by a batched matrix multiplication:
   for each head h we compute   out[h] = left[h] @ right[h]ᵀ   using
   torch.bmm (cuBLAS), which is faster than a pure Triton GEMM.
5. The result is layer‑normed, multiplied by the output gate and projected
   back to the original dimension.
All heavy element‑wise work is done in fp16; accumulations for the GEMM are
performed in fp16 via cuBLAS and the final result is cast to fp32.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# --------------------------------------------------------------
# Triton kernels for elementwise multiplication
# --------------------------------------------------------------

@triton.jit
def mul2_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK: tl.constexpr):
    """out = a * b (element‑wise)"""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)

@triton.jit
def mul3_kernel(a_ptr, b_ptr, c_ptr, out_ptr, size, BLOCK: tl.constexpr):
    """out = a * b * c (element‑wise)"""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    c = tl.load(c_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b * c, mask=mask)

# --------------------------------------------------------------
# Entry point
# --------------------------------------------------------------

def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32, CUDA)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of torch.Tensor with all linear / layer‑norm params
            - config: dict containing at least {"dim": int, "hidden_dim": int}

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # -----------------------------------------------------------------
    # 1. unpack
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 2. LayerNorm over the last dimension
    # -----------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm.to(torch.float16)   # fp16 for the heavy part

    # -----------------------------------------------------------------
    # 3. Linear projections & gate logits (fp16)
    # -----------------------------------------------------------------
    left_proj = F.linear(x_norm, weights["left_proj.weight"].to(torch.float16))   # [B,N,N,hidden_dim]
    right_proj = F.linear(x_norm, weights["right_proj.weight"].to(torch.float16))

    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"].to(torch.float16)))  # [B,N,N,hidden_dim]

    # -----------------------------------------------------------------
    # 4. Fuse mask (if any) and gate with a Triton kernel
    # -----------------------------------------------------------------
    total_elems = left_proj.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_elems, BLOCK),)

    # Allocate buffers for gated tensors
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    if mask is None:
        # No mask -> two‑operand multiplication (proj * gate)
        mul2_kernel[grid](
            left_proj, left_gate, left,
            total_elems,
            BLOCK=BLOCK,
        )
        mul2_kernel[grid](
            right_proj, right_gate, right,
            total_elems,
            BLOCK=BLOCK,
        )
    else:
        # Prepare mask broadcasted to hidden_dim
        mask_f = mask.to(torch.float16).unsqueeze(-1)               # [B,N,N,1]
        mask_f = mask_f.expand(-1, -1, -1, hidden_dim).contiguous()  # [B,N,N,hidden_dim]

        # Three‑operand multiplication (proj * gate * mask)
        mul3_kernel[grid](
            left_proj, left_gate, mask_f, left,
            total_elems,
            BLOCK=BLOCK,
        )
        mul3_kernel[grid](
            right_proj, right_gate, mask_f, right,
            total_elems,
            BLOCK=BLOCK,
        )

    # -----------------------------------------------------------------
    # 5. Batched matrix multiplication: out = left @ rightᵀ
    #    Shape: (B, N, N, hidden_dim) -> (B*hidden_dim, N, N)
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    total_batch = B * H

    left_flat = left.permute(0, 3, 1, 2).contiguous().view(total_batch, N, N)   # [B*H, N, N]
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(total_batch, N, N)  # [B*H, N, N]

    # rightᵀ for each batch
    right_T = right_flat.transpose(1, 2).contiguous()

    # cuBLAS batched GEMM (fp16 → fp16)
    out_flat = torch.bmm(left_flat, right_T)   # [B*H, N, N]

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # -----------------------------------------------------------------
    # 6. Output layer‑norm, output gate, final linear projection
    # -----------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    out_norm = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Apply output gate (cast gate to fp32)
    out_gated = out_norm * out_gate.to(torch.float32)

    # Final projection back to `dim`
    output = F.linear(
        out_gated,
        weights["to_out.weight"].to(torch.float32),   # shape (dim, hidden_dim)
    )  # [B, N, N, dim] in fp32

    return output
